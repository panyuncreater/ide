1/**
2 * @file compiler/ir/IRSSA.h
3 * @brief P2-10 IR SSA 基础设施。
4 *
5 * 背景与目标（见 MiniLang优化清单_2026-07-23.md 第 10 项）：
6 * 当前 IR 已是 SSA 风格（vreg 单赋值、单调递增不复用），但缺少：
7 * - 支配树 / 支配边界
8 * - PHI 节点（局部变量未进入 SSA）
9 * - 自然循环检测
10 * - 全局值编号（GVN，当前 CSE 仅基本块内）
11 * - 循环不变代码外提（LICM）
12 * - 函数内联
13 * 本文件实现上述基础设施，使 IR 层具备生产级优化能力。
14 *
15 * 设计要点：
16 * 1. **非侵入式 CFG 视图**：AstIRBuilder 产出的 IRBasicBlock 是"松散块"——单个块
17 * 内可能含多个 LABEL 与终结指令（if/while/for 都在同一 currentBlock_ 中 emit）。
18 * IRCFG 将其展平并按 LABEL/终结指令切分为严格基本块（单入口、单终结），
19 * 不修改 IRFunction 自身结构，因此既有 pass（CSE/常量折叠/循环展开）零影响。
20 * 2. **PHI 生命周期**：PHI 仅在 ssaConstructPass 与 ssaDestructPass 之间存在。
21 * 析构在 lowering 前完成，后端永远不看到 PHI（见 IR.h 的 PHI 注释）。
22 * 3. **安全性**：GVN/LICM 与 CSE 同源约束——替换 dest vreg 引用后，栈式 VM 后端
23 * 会产生栈残留 → OP_POP 栈下溢。因此这些 pass 仅对寄存器式后端启用，
24 * 由 optimizeIR 的 enableGVN / enableLICM 旗标控制（默认 false）。
25 *
26 * 算法出处：
27 * - 支配树：Cooper, Harvey, Kennedy "A Simple, Fast Dominance Algorithm" (2001)
28 * - SSA 构造（PHI 插入）：Cytron 等 "Efficiently Computing Static Single Assignment
29 * Form and the Control Dependence Graph" (1991) —— 支配边界迭代插入
30 * - 自然循环：Aho 等 "Compilers: Principles, Techniques, and Tools" 回边定义
31 *
32 * @see IR.h optimizeIR gvnPass licmPass inlinePass
33 */

48/// CFG 中的一个严格基本块：单入口（一个 LABEL 或函数入口）、单终结指令。
49/// 通过展平 IRFunction 的松散块并按 LABEL/终结指令切分得到。
50struct CFGNode {
51 uint32_t id = 0; // 在 IRCFG::nodes 中的索引
52 uint32_t labelIndex = 0; // 入口 LABEL 索引（入口块用 kEntryLabel）
53 bool hasLabel = false; // 是否以 LABEL 指令起始（入口块可能无 LABEL）
54
55 // 源位置：所属的 IRBasicBlock 索引 + 指令范围 [startInstr, endInstr)
56 // 切分不修改 IRFunction，变换 pass 通过此映射定位并修改原始指令。
57 size_t sourceBlockIdx = 0;
58 size_t startInstr = 0; // 含（指向 LABEL 或首条非终结指令）
59 size_t endInstr = 0; // 不含（指向终结指令的下一条）
60
61 std::vector<uint32_t> preds; // 前驱节点 id 列表
62 std::vector<uint32_t> succs; // 后继节点 id 列表
63
64 static constexpr uint32_t kEntryLabel = 0xFFFFFFFF; // 入口块哨兵 label
65
66 /// 是否为函数入口块（id == 0）
67 bool isEntry() const { return id == 0; }
68};

70// ============================================================
71// IRCFG：从 IRFunction 构建严格 CFG
72// ============================================================

74/// 严格 CFG 构建器。
75///
76/// 切分规则：
77/// - 展平 IRFunction 所有 IRBasicBlock 的指令为单一序列（保持顺序）。
78/// - 一个严格块在以下位置起始：函数第一条指令，或任意 LABEL 指令。
79/// - 一个严格块在以下位置终结：第一条终结指令（JUMP/JUMP_IF_FALSE/RETURN/
80/// RETURN_NULL/THROW），或下一条 LABEL 之前（隐式 fall-through）。
81/// - 后继边：JUMP→目标 LABEL 块；JUMP_IF_FALSE→fall-through 块 + 目标 LABEL 块；
82/// RETURN/RETURN_NULL/THROW→无后继；fall-through→相邻下一块。
83///
84/// 不可变性：构建后 CFG 不随 IRFunction 变更自动更新；变换 pass 修改 IR 后应重建。
85class IRCFG {
86public:
87 /// 从 IRFunction 构建 CFG。空函数返回单入口空块。
88 explicit IRCFG(const IRFunction& ir);
89
90 /// 节点数
91 size_t size() const { return nodes_.size(); }
92 /// 入口节点 id（恒为 0）
93 uint32_t entryId() const { return 0; }
94 /// 取节点（只读）
95 const CFGNode& node(uint32_t id) const { return nodes_[id]; }
96 /// 取所有节点（只读）
97 const std::vector<CFGNode>& nodes() const { return nodes_; }
98 /// 按 LABEL 索引查找节点 id（无则返回 kInvalid）
99 uint32_t nodeOfLabel(uint32_t labelIndex) const;
100
101 static constexpr uint32_t kInvalid = 0xFFFFFFFF;
102
103private:
104 std::vector<CFGNode> nodes_;
105 std::unordered_map<uint32_t, uint32_t> labelToNode_; // labelIndex → node id
106
107 /// 判断指令是否为终结指令（块结束）
108 static bool isTerminator(IROp op);
109};

115/// 支配树。基于 Cooper-Harvey-Kennedy 算法计算 immediate dominator。
116///
117/// 不变量：
118/// - idom_[entry] = entry（自支配）
119/// - idom_[n] 是 n 的严格支配者中离 n 最近的
120/// - dominates(a, b) 当且仅当 a 在从 entry 到 b 的路径上必经 a
121class DominatorTree {
122public:
123 /// 从 CFG 构建支配树
124 explicit DominatorTree(const IRCFG& cfg);
125
126 /// 取 immediate dominator（entry 返回自身，未计算/不可达返回 kInvalid）
127 uint32_t idom(uint32_t node) const { return idom_[node]; }
128 /// a 是否支配 b
129 bool dominates(uint32_t a, uint32_t b) const;
130 /// a 与 b 的最近公共支配者
131 uint32_t nearestCommonDominator(uint32_t a, uint32_t b) const;
132 /// 取 a 在支配树中的直接子节点（被 a 直接支配的节点）
133 const std::vector<uint32_t>& children(uint32_t a) const { return domChildren_[a]; }
134
135 static constexpr uint32_t kInvalid = 0xFFFFFFFF;
136
137private:
138 const IRCFG& cfg_;
139 std::vector<uint32_t> idom_; // node → immediate dominator
140 std::vector<std::vector<uint32_t>> domChildren_; // node → 直接支配的子节点列表
141 std::vector<uint32_t> rpo_; // 反向后序
142 std::vector<uint32_t> postorder_; // 后序（用于 intersect 比较）
143 std::vector<uint32_t> postOrderNum_; // node → postorder 编号
144
145 /// 反向后序遍历（构建 rpo_ 与 postorder_）
146 void computeRPO();
147 /// Cooper-Harvey-Kennedy 迭代求解 idom
148 void computeIDom();
149 /// intersect：沿 postorder 更深者上溯直到相遇（CHK 算法核心）
150 uint32_t intersect(uint32_t b1, uint32_t b2) const;
151};

174/// 自然循环信息。
175///
176/// 回边定义：边 n→h 是回边当且仅当 h 支配 n。
177/// 自然循环 of 回边 n→h = {h} ∪ { 所有不经过 h 能到达 n 的节点 }。
178class NaturalLoopInfo {
179public:
180 struct Loop {
181 uint32_t header; // 循环头（支配者）
182 std::vector<uint32_t> body; // 循环体节点 id（含 header）
183 std::vector<uint32_t> backEdges; // 回边源节点列表（尾→头）
184 };
185
186 NaturalLoopInfo(const IRCFG& cfg, const DominatorTree& domTree);
187
188 /// 所有自然循环
189 const std::vector<Loop>& loops() const { return loops_; }
190 /// 节点 n 是否在任一循环体内
191 bool inLoop(uint32_t n) const { return inLoop_.count(n) > 0; }
192 /// 节点 n 所属的循环索引列表（可能属于多个嵌套循环）
193 const std::vector<size_t>& loopsOf(uint32_t n) const {
194 // BUGFIX-P2 #54: 非循环节点不在 loopsOf_ 中，.at() 会抛 std::out_of_range。
195 // 改为返回静态空 vector 的 fallback，调用方可安全查询任意节点。
196 static const std::vector<size_t> empty;
197 auto it = loopsOf_.find(n);
198 return it != loopsOf_.end() ? it->second : empty;
199 }
200
201private:
202 const IRCFG& cfg_;
203 std::vector<Loop> loops_;
204 std::unordered_map<uint32_t, std::vector<size_t>> loopsOf_; // node → loop indices
205 std::unordered_set<uint32_t> inLoop_;
206 std::vector<std::pair<uint32_t, uint32_t>> backEdges_; // (tail, header) 回边列表
207 std::unordered_set<uint32_t> loopBody_; // collectLoopBody 的临时结果
208
209 void detectBackEdges(const DominatorTree& domTree);
210 /// 收集回边 n→h 的自然循环体（反向 BFS/DFS 从 n 出发，不越过 h）
211 void collectLoopBody(uint32_t header, uint32_t tail);
212};

215// SSA 构造与析构（局部变量 → SSA，含 PHI 插入）

218/// SSA 构造 pass：将函数内局部变量（LOAD_LOCAL/STORE_LOCAL）提升为 SSA 形态。
219///
220/// 步骤（Cytron 算法）：
221/// 1. 收集每个 local slot 的所有定义点（STORE_LOCAL）。
222/// 2. 对每个 slot，在其定义的支配边界迭代插入 PHI（直到不动点）。
223/// 3. 重命名：遍历支配树，将 LOAD_LOCAL/STORE_LOCAL 替换为 vreg 引用，
224/// PHI 在合并点选择对应前驱的 vreg。
225///
226/// 约束：必须在 ssaDestructPass 之前调用；PHI 不可到达后端 lowering。
227/// 返回：是否插入了 PHI（true 表示进入 SSA 形态）。
228bool ssaConstructPass(IRFunction& ir);
229
230/// SSA 析构 pass：将 PHI 节点消除，恢复为 LOAD_LOCAL/STORE_LOCAL 形态。
231///
232/// 策略：对每个 PHI(dest, (pred1, v1), (pred2, v2), ...)，在 pred1 尾部
233/// 插入 STORE_LOCAL slot=v1，在 pred2 尾部插入 STORE_LOCAL slot=v2，
234/// 将 PHI 替换为 LOAD_LOCAL slot→dest。
235///
236/// 返回：是否消除了 PHI（true 表示已退出 SSA 形态）。
237bool ssaDestructPass(IRFunction& ir);
