45/// 将 64 位 NaN-box 原始位模式格式化为 0x 前缀的 16 位十六进制字符串。
46std::string bitsToHex(uint64_t bits) {
47 std::ostringstream os;
48 os << "0x" << std::hex << std::setfill('0') << std::setw(16) << bits;
49 return os.str();
50}

52/// 将 64 位位模式逐位展开为长度 64 的 "0/1" 字符串（最高位在索引 0）。
53std::string bitsToBinary(uint64_t bits) {
54 std::string s(64, '0');
55 for (int i = 0; i < 64; ++i) {
56 if (bits & (1ULL << (63 - i)))
57 s[i] = '1';
58 }
59 return s;
60}
