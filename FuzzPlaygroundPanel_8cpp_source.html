1// ============================================================
2// FuzzPlaygroundPanel.cpp — 模糊测试游乐场教学面板实现
3// ------------------------------------------------------------
4// 复用 cli/fuzz_core.h 的公开 API：
5// - fuzzThreeAgree(source)：单次三后端差分（Interpreter/StackVM/RegisterVM）
6// - runFuzzBatch(opts)：批量生成/变异执行 + 汇总（崩溃/分歧用例）
7// - formatSummaryText(summary)：文本格式汇总（用于统计报告）
8//
9// 单次模式：sourceEdit → fuzzThreeAgree → 渲染 3 行表格 + 详情
10// 批量模式：opts(seed/iter/mode) → runFuzzBatch → 渲染 1 行汇总
11// + 分歧用例表（最多 20 个）+ 详情
12// 三后端在主线程串行执行（教学面板可接受阻塞调用，避免线程复杂化）。

214/// 运行：根据模式分发到单次差分或批量生成/变异。
215void FuzzPlaygroundPanel::onRun() {
216 QString mode = modeCombo_->currentText();
217 if (mode == QString::fromUtf8("单次差分")) {
218 std::string src = sourceEdit_->toPlainText().toStdString();
219 if (src.empty()) {
220 consistencyLabel_->setText(QString::fromUtf8("❌ 源码为空，请输入 MiniLang 源码"));
221 consistencyLabel_->setStyleSheet(QStringLiteral("color:#C0392B;"));
222 return;
223 }
224 // 单次差分：阻塞调用 fuzzThreeAgree
225 minilang_fuzz::FuzzResult r = minilang_fuzz::fuzzThreeAgree(src);
226 lastSingleResult_ = r;
227 hasSingleResult_ = true;
228
229 setupDiffTableForSingle();
230 renderSingleMode(r);
231 renderReportSingle(r);
232 renderConsistencySingle(r);
233 clearDetail();
234 tabWidget_->setCurrentIndex(0); // 切换到差分结果 Tab
235 } else {
236 // 批量生成 / 批量变异
237 minilang_fuzz::FuzzOptions opts;
238 opts.seed = static_cast<uint64_t>(seedSpin_->value());
239 opts.iterations = iterSpin_->value();
240 opts.mode = (mode == QString::fromUtf8("批量变异")) ? minilang_fuzz::FuzzMode::Mutate
241 : minilang_fuzz::FuzzMode::Generate;
242 opts.quiet = true;
243 opts.dumpCrashes = false;
244
245 // 批量执行（阻塞调用，可能耗时）
246 minilang_fuzz::FuzzSummary s = minilang_fuzz::runFuzzBatch(opts);
247 lastBatchSummary_ = s;
248 hasBatchResult_ = true;
249
250 setupDiffTableForBatch();
251 renderBatchMode(s, mode);
252 renderDisagreeTable(s);
253 renderReportBatch(s, mode);
254 renderConsistencyBatch(s);
255 clearDetail();
256 // 有分歧用例时切换到分歧详情 Tab，否则停留在差分结果 Tab
257 tabWidget_->setCurrentIndex(s.disagreementCases.empty() ? 0 : 1);
258 }
259}

358/// durationMs 为三后端串行总耗时（仅显示在第 1 行，其余为"—"）。
359void FuzzPlaygroundPanel::renderSingleMode(const minilang_fuzz::FuzzResult& r) {
360 struct RowInfo {
361 QString name;
362 std::string output;
363 };
364 RowInfo rows[3] = {{QString::fromUtf8("Interpreter"), r.interpOutput},
365 {QString::fromUtf8("StackVM"), r.stackvmOutput},
366 {QString::fromUtf8("RegisterVM"), r.regvmOutput}};
367
368 for (int i = 0; i < 3; ++i) {
369 // 后端名
370 auto* nameItem = new QTableWidgetItem(rows[i].name);
371 nameItem->setTextAlignment(Qt::AlignCenter);
372 nameItem->setForeground(QColor(backendColor(i)));
373 diffTable_->setItem(i, 0, nameItem);
374
375 // 输出（HTML 转义 + 单行化）
376 auto* outItem = new QTableWidgetItem(summarizeOutput(rows[i].output, 200));
377 diffTable_->setItem(i, 1, outItem);
378
379 // 状态
380 QString status = statusFromOutput(rows[i].output);
381 auto* statusItem = new QTableWidgetItem(status);
382 statusItem->setTextAlignment(Qt::AlignCenter);
383 if (status.startsWith(QString::fromUtf8("✅"))) {
384 statusItem->setForeground(QColor(QStringLiteral("#27AE60")));
385 } else {
386 statusItem->setForeground(QColor(QStringLiteral("#C0392B")));
387 }
388 diffTable_->setItem(i, 2, statusItem);
389
390 // 耗时（仅第 1 行显示总耗时，其余为"—"）
391 QString timeText = (i == 0) ? QString::number(static_cast<qint64>(r.durationMs)) : QString::fromUtf8("—");
392 auto* timeItem = new QTableWidgetItem(timeText);
393 timeItem->setTextAlignment(Qt::AlignCenter);
394 if (i == 0) {
395 timeItem->setToolTip(mlTr("三后端串行总耗时（fuzzThreeAgree 内部计时）"));
396 }
397 diffTable_->setItem(i, 3, timeItem);
398 }
399}

421/// 渲染分歧详情表：从 s.disagreementCases 取前 20 个，3 列（序号/源码片段/分歧类型）。
422void FuzzPlaygroundPanel::renderDisagreeTable(const minilang_fuzz::FuzzSummary& s) {
423 disagreeTable_->setRowCount(0);
424 int shown = std::min<int>(20, static_cast<int>(s.disagreementCases.size()));
425 if (shown <= 0) {
426 disagreeTable_->setRowCount(1);
427 auto* emptyItem = new QTableWidgetItem(QString::fromUtf8("（无分歧用例）"));
428 emptyItem->setTextAlignment(Qt::AlignCenter);
429 emptyItem->setForeground(QColor(QStringLiteral("#6E6E6E")));
430 disagreeTable_->setItem(0, 0, emptyItem);
431 disagreeTable_->setSpan(0, 0, 1, 3);
432 return;
433 }
434 disagreeTable_->setRowCount(shown);
435 for (int i = 0; i < shown; ++i) {
436 const auto& d = s.disagreementCases[static_cast<size_t>(i)];
437
438 auto* idxItem = new QTableWidgetItem(QString::number(i + 1));
439 idxItem->setTextAlignment(Qt::AlignCenter);
440
441 // 源码片段：取前 80 字符，换行替换为空格
442 QString snippet = escapeHtml(d.source);
443 snippet.replace(QLatin1Char('\n'), QStringLiteral(" "));
444 snippet.replace(QLatin1Char('\r'), QString());
445 if (snippet.length() > 80)
446 snippet = snippet.left(80) + QStringLiteral("…");
447 auto* srcItem = new QTableWidgetItem(snippet);
448
449 auto* typeItem = new QTableWidgetItem(escapeHtml(d.errorMessage));
450 typeItem->setTextAlignment(Qt::AlignCenter);
451 typeItem->setForeground(QColor(QStringLiteral("#C0392B")));
452
453 disagreeTable_->setItem(i, 0, idxItem);
454 disagreeTable_->setItem(i, 1, srcItem);
455 disagreeTable_->setItem(i, 2, typeItem);
456 }
457}

459/// 渲染单次模式统计报告 HTML：差分摘要 + 错误归一化说明 + 教学解释。
460void FuzzPlaygroundPanel::renderReportSingle(const minilang_fuzz::FuzzResult& r) {
461 QString html =
462 QStringLiteral("<html><body style='font-family:%1;padding:4px;'>").arg(GuiTextUtils::uiFontFamilyQss());
463
464 html += QStringLiteral("<h3 style='margin:0 0 8px 0;color:#2C3E50;'>单次差分摘要</h3>");
465 html += QStringLiteral("<table cellspacing='0' cellpadding='4' style='width:100%;'>"
466 "<tr style='background:#ECF0F1;font-weight:bold;'>"
467 "<td width='30%'>项</td>"
468 "<td width='70%'>值</td>"
469 "</tr>");
470 html += QStringLiteral("<tr><td>源码长度</td><td>%1 字节</td></tr>").arg(static_cast<int>(r.source.size()));
471 html +=
472 QStringLiteral("<tr><td>总耗时</td><td>%1 ms（三后端串行）</td></tr>").arg(static_cast<qint64>(r.durationMs));
473 html += QStringLiteral("<tr><td>崩溃检测</td><td>%1</td></tr>")

814/// 根据 fuzz 输出判断后端状态文本。
815/// <parse-fail> → 解析失败，<compile:...> → 编译错误，<runtime:...> → 运行时错误；其他 → 成功。
816QString FuzzPlaygroundPanel::statusFromOutput(const std::string& output) {
817 if (isParseFailure(output)) {
818 return QString::fromUtf8("❌ 解析失败");
819 }
820 if (isCompileFailure(output)) {
821 return QString::fromUtf8("❌ 编译错误");
822 }
823 if (isRuntimeError(output)) {
824 return QString::fromUtf8("❌ 运行时错误");
825 }
826 return QString::fromUtf8("✅ 成功");
827}

829/// HTML 转义 std::string → QString。
830QString FuzzPlaygroundPanel::escapeHtml(const std::string& s) {
831 return QString::fromUtf8(s.data(), static_cast<int>(s.size())).toHtmlEscaped();
832}

834/// 摘要输出：HTML 转义 + 换行单行化 + 截断。
835QString FuzzPlaygroundPanel::summarizeOutput(const std::string& output, int maxLen) {
836 QString q = escapeHtml(output);
837 q.replace(QLatin1Char('\n'), QStringLiteral("\\n"));
838 q.replace(QLatin1Char('\r'), QString());
839 if (q.length() > maxLen) {
840 q = q.left(maxLen) + QStringLiteral("…");
841 }
842 return q;
843}
