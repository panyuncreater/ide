27/// 规范化模块路径为 loader 请求路径（保留大小写）。见文件头注释。
28inline std::string normalizeModulePathKey(const std::string& raw) {
29 std::string p;
30 p.reserve(raw.size());
31 // '\' → '/'，同时折叠连续 '/'（与 ModuleIsolation::pathHash 的 R53 折叠对齐）
32 bool prevSlash = false;
33 for (char c : raw) {
34 if (c == '\\')
35 c = '/';
36 if (c == '/') {
37 if (!prevSlash)
38 p.push_back('/');
39 prevSlash = true;
40 } else {
41 p.push_back(c);
42 prevSlash = false;
43 }
44 }
45 // 循环剥前缀 "./"（AUDIT-R3 P2-10：只剥一次会漏 "././m.mini"）
46 while (p.size() >= 2 && p[0] == '.' && p[1] == '/') {
47 p.erase(0, 2);
48 }
49 // 折叠中间 "/./" 段
50 for (size_t dotPos = p.find("/./"); dotPos != std::string::npos; dotPos = p.find("/./")) {
51 p.erase(dotPos, 2);
52 }
53 return p;
54}

56/// BUG-M4 fix: 从已规范化的 loader 路径派生缓存去重键。
57/// Windows 文件系统大小写不敏感：缓存键统一 ASCII 小写，使同一文件的不同大小写
58/// 拼写去重为同一模块。仅折叠 ASCII A-Z，不触碰 UTF-8 多字节序列。
59/// Linux/macOS（大小写敏感文件系统）不折叠，与平台语义一致。
60/// 调用方：用 normalizeModulePathKey 的返回值调 loader/mtimeChecker（保留大小写），
61/// 用本函数的返回值作所有缓存/去重 map 键与 pathHash 输入。
62inline std::string moduleCacheKey(const std::string& normalizedLoaderPath) {
63#ifdef _WIN32
64 std::string key = normalizedLoaderPath;
65 for (char& c : key) {
66 if (c >= 'A' && c <= 'Z')
67 c = static_cast<char>(c - 'A' + 'a');
68 }
69 return key;
70#else
71 return normalizedLoaderPath;
72#endif
73}
