2 * @file ast/MacroExpander.cpp
3 * @brief 七特性 MVP 阶段 2：宏展开器实现——表达式子集克隆 + 参数替换。
4 *
5 * 设计说明见 MacroExpander.h。实现要点：
6 * - VarRef 命中替换表 → 直接返回实参 shared_ptr（共享，不克隆）
7 * - 其他节点 → 深克隆（新节点，line/column 沿用模板节点，便于错误定位）
8 * - 不支持的节点类型（语句/声明/match/yield 等）→ 返回 nullptr 报告节点名

18/// 克隆表达式列表的辅助函数。任一元素失败返回 false。
19bool cloneList(const std::vector<std::shared_ptr<ASTNode>>& src, const SubstMap& subst, int depth,
20 std::string* unsupportedNodeName, std::vector<std::shared_ptr<ASTNode>>& out) {
21 out.reserve(src.size());
22 for (const auto& e : src) {
23 auto cloned = cloneWithSubstitution(e.get(), subst, depth, unsupportedNodeName);
24 if (!cloned)
25 return false;
26 out.push_back(std::move(cloned));
27 }
28 return true;
29}

33std::shared_ptr<ASTNode> cloneWithSubstitution(const ASTNode* node, const SubstMap& subst, int depth,
34 std::string* unsupportedNodeName) {
35 if (!node) {
36 if (unsupportedNodeName)
37 *unsupportedNodeName = "<null>";
38 return nullptr;
39 }
40 if (depth >= RuntimeLimits::MAX_MACRO_EXPANSION_DEPTH) {
41 if (unsupportedNodeName)
42 *unsupportedNodeName = "<depth-limit>";
43 return nullptr;
44 }
45 const int d = depth + 1;
46
47 switch (node->nodeType) {
48 case NodeType::NODE_VAR_REF: {
49 const auto* v = static_cast<const VarRef*>(node);
50 auto it = subst.find(v->name);
51 if (it != subst.end()) {
52 // 形参命中：返回实参子树（shared_ptr 共享，同 C 宏的文本替换语义）
53 return it->second;
54 }
55 return std::make_shared<VarRef>(v->name, v->line, v->column);
56 }
57 case NodeType::NODE_NUMBER_LITERAL: {
58 const auto* n = static_cast<const NumberLiteral*>(node);
59 if (n->isInt())
60 return std::make_shared<NumberLiteral>(n->intVal(), n->line, n->column);
61 return std::make_shared<NumberLiteral>(n->floatVal(), n->line, n->column);
62 }
63 case NodeType::NODE_STRING_LITERAL: {
64 const auto* s = static_cast<const StringLiteral*>(node);
65 return std::make_shared<StringLiteral>(s->value, s->line, s->column);
66 }
67 case NodeType::NODE_BOOL_LITERAL: {
68 const auto* b = static_cast<const BoolLiteral*>(node);
69 return std::make_shared<BoolLiteral>(b->value, b->line, b->column);
70 }
71 case NodeType::NODE_NULL_LITERAL:
72 return std::make_shared<NullLiteral>(node->line, node->column);
73 case NodeType::NODE_BINARY_OP: {
74 const auto* b = static_cast<const BinaryOp*>(node);
75 auto l = cloneWithSubstitution(b->left.get(), subst, d, unsupportedNodeName);
76 if (!l)
77 return nullptr;
78 auto r = cloneWithSubstitution(b->right.get(), subst, d, unsupportedNodeName);
79 if (!r)
80 return nullptr;
81 return std::make_shared<BinaryOp>(b->opType, std::move(l), std::move(r), b->line, b->column);
82 }
83 case NodeType::NODE_UNARY_OP: {
84 const auto* u = static_cast<const UnaryOp*>(node);
85 auto o = cloneWithSubstitution(u->operand.get(), subst, d, unsupportedNodeName);
86 if (!o)
87 return nullptr;
88 return std::make_shared<UnaryOp>(u->opType, std::move(o), u->line, u->column);
89 }
90 case NodeType::NODE_FUN_CALL: {
91 const auto* f = static_cast<const FunCall*>(node);
92 std::vector<std::shared_ptr<ASTNode>> args;
93 if (!cloneList(f->arguments, subst, d, unsupportedNodeName, args))
94 return nullptr;
95 if (f->callee) {
96 auto callee = cloneWithSubstitution(f->callee.get(), subst, d, unsupportedNodeName);
97 if (!callee)
98 return nullptr;
99 return std::make_shared<FunCall>(std::move(callee), std::move(args), f->line, f->column);
100 }
101 return std::make_shared<FunCall>(f->name, std::move(args), f->line, f->column);
102 }
103 case NodeType::NODE_INDEX_ACCESS: {
104 const auto* ia = static_cast<const IndexAccess*>(node);
105 auto obj = cloneWithSubstitution(ia->object.get(), subst, d, unsupportedNodeName);
106 if (!obj)
107 return nullptr;
108 auto idx = cloneWithSubstitution(ia->index.get(), subst, d, unsupportedNodeName);
109 if (!idx)
110 return nullptr;
111 return std::make_shared<IndexAccess>(std::move(obj), std::move(idx), ia->line, ia->column);
112 }
113 case NodeType::NODE_MEMBER_ACCESS: {
114 const auto* ma = static_cast<const MemberAccess*>(node);
115 auto obj = cloneWithSubstitution(ma->object.get(), subst, d, unsupportedNodeName);
116 if (!obj)
117 return nullptr;
118 return std::make_shared<MemberAccess>(std::move(obj), ma->fieldName, ma->line, ma->column);
119 }
120 case NodeType::NODE_METHOD_CALL: {
121 const auto* mc = static_cast<const MethodCall*>(node);
122 auto obj = cloneWithSubstitution(mc->object.get(), subst, d, unsupportedNodeName);
123 if (!obj)
124 return nullptr;
125 std::vector<std::shared_ptr<ASTNode>> args;
126 if (!cloneList(mc->arguments, subst, d, unsupportedNodeName, args))
127 return nullptr;
128 return std::make_shared<MethodCall>(std::move(obj), mc->methodName, std::move(args), mc->line, mc->column);
129 }
130 case NodeType::NODE_ARRAY_LITERAL: {
131 const auto* a = static_cast<const ArrayLiteral*>(node);
132 std::vector<std::shared_ptr<ASTNode>> elems;
133 if (!cloneList(a->elements, subst, d, unsupportedNodeName, elems))
134 return nullptr;
135 return std::make_shared<ArrayLiteral>(std::move(elems), a->line, a->column);
136 }
137 case NodeType::NODE_TUPLE_LITERAL: {
138 const auto* t = static_cast<const TupleLiteral*>(node);
139 std::vector<std::shared_ptr<ASTNode>> elems;
140 if (!cloneList(t->elements, subst, d, unsupportedNodeName, elems))
141 return nullptr;
142 return std::make_shared<TupleLiteral>(std::move(elems), t->line, t->column);
143 }
144 case NodeType::NODE_DICT_LITERAL: {
145 const auto* dl = static_cast<const DictLiteral*>(node);
146 std::vector<std::pair<std::shared_ptr<ASTNode>, std::shared_ptr<ASTNode>>> pairs;
147 pairs.reserve(dl->pairs.size());
148 for (const auto& p : dl->pairs) {
149 auto k = cloneWithSubstitution(p.first.get(), subst, d, unsupportedNodeName);
150 if (!k)
151 return nullptr;
152 auto v = cloneWithSubstitution(p.second.get(), subst, d, unsupportedNodeName);
153 if (!v)
154 return nullptr;
155 pairs.emplace_back(std::move(k), std::move(v));
156 }
157 return std::make_shared<DictLiteral>(std::move(pairs), dl->line, dl->column);
158 }
159 case NodeType::NODE_INTERPOLATED_STRING: {
160 const auto* is = static_cast<const InterpolatedString*>(node);
161 auto cloned = std::make_shared<InterpolatedString>(is->line, is->column);
162 cloned->literals = is->literals;
163 cloned->literalsTotalLen = is->literalsTotalLen;
164 cloned->endLine = is->endLine;
165 if (!cloneList(is->expressions, subst, d, unsupportedNodeName, cloned->expressions))
166 return nullptr;
167 return cloned;
168 }
169 case NodeType::NODE_ENUM_VARIANT_EXPR: {
170 const auto* ev = static_cast<const EnumVariantExpr*>(node);
171 std::vector<std::shared_ptr<ASTNode>> args;
172 if (!cloneList(ev->arguments, subst, d, unsupportedNodeName, args))
173 return nullptr;
174 return std::make_shared<EnumVariantExpr>(ev->enumName, ev->variantName, std::move(args), ev->line, ev->column);
175 }
176 case NodeType::NODE_MACRO_CALL: {
177 // 嵌套宏调用：body 模板解析时内层宏已展开（expanded 非空），
178 // 克隆原始实参与 expanded 子树，保持 Formatter 打印形式。
179 const auto* mc = static_cast<const MacroCallExpr*>(node);
180 std::vector<std::shared_ptr<ASTNode>> args;
181 if (!cloneList(mc->arguments, subst, d, unsupportedNodeName, args))
182 return nullptr;
183 auto exp = cloneWithSubstitution(mc->expanded.get(), subst, d, unsupportedNodeName);
184 if (!exp)
185 return nullptr;
186 auto cloned = std::make_shared<MacroCallExpr>(mc->name, std::move(args), std::move(exp), mc->line, mc->column);
187 cloned->producesValue = mc->producesValue;
188 return cloned;
189 }
190 // ============================================================
191 // 七特性宏升级：语句/块体节点克隆（支持语句模板宏）
192 // ------------------------------------------------------------
193 // 赋值目标名（Assignment.name）的形参替换：若实参为简单 VarRef，则替换为
194 // 实参变量名（支持 swap!(x,y) 中 `a=b` → `x=y` 的左值代入）；否则保留原名。
195 // 引入型绑定名（VarDecl.name）不替换（同 C 宏：宏内部 var 可能与实参名碰撞，
196 // 为已知局限，与文本替换宏的卫生性问题一致）。
197 // ============================================================
198 case NodeType::NODE_BLOCK: {
199 const auto* blk = static_cast<const Block*>(node);
200 std::vector<std::shared_ptr<ASTNode>> stmts;
201 if (!cloneList(blk->statements, subst, d, unsupportedNodeName, stmts))
202 return nullptr;
203 auto cloned = std::make_shared<Block>(std::move(stmts), blk->line, blk->column);
204 cloned->closingBraceLine = blk->closingBraceLine;
205 return cloned;
206 }
207 case NodeType::NODE_VAR_DECL: {
208 const auto* vd = static_cast<const VarDecl*>(node);
209 std::shared_ptr<ASTNode> init;
210 if (vd->initializer) {
211 init = cloneWithSubstitution(vd->initializer.get(), subst, d, unsupportedNodeName);
212 if (!init)
213 return nullptr;
214 }
215 return std::make_shared<VarDecl>(vd->name, vd->typeAnnotation, std::move(init), vd->line, vd->column);
216 }
217 case NodeType::NODE_ASSIGNMENT: {
218 const auto* as = static_cast<const Assignment*>(node);
219 auto val = cloneWithSubstitution(as->value.get(), subst, d, unsupportedNodeName);
220 if (!val)
221 return nullptr;
222 // 左值名替换：若形参对应实参为简单 VarRef，代入其变量名
223 std::string targetName = as->name;
224 auto it = subst.find(as->name);
225 if (it != subst.end() && it->second && it->second->nodeType == NodeType::NODE_VAR_REF) {
226 targetName = static_cast<const VarRef*>(it->second.get())->name;
227 }
228 return std::make_shared<Assignment>(targetName, std::move(val), as->line, as->column);
229 }
230 case NodeType::NODE_IF_STMT: {
231 const auto* is = static_cast<const IfStmt*>(node);
232 auto cond = cloneWithSubstitution(is->condition.get(), subst, d, unsupportedNodeName);
233 if (!cond)
234 return nullptr;
235 auto thenB = cloneWithSubstitution(is->thenBranch.get(), subst, d, unsupportedNodeName);
236 if (!thenB)
237 return nullptr;
238 std::shared_ptr<ASTNode> elseB;
239 if (is->elseBranch) {
240 elseB = cloneWithSubstitution(is->elseBranch.get(), subst, d, unsupportedNodeName);
241 if (!elseB)
242 return nullptr;
243 }
244 return std::make_shared<IfStmt>(std::move(cond), std::move(thenB), std::move(elseB), is->line, is->column);
245 }
246 case NodeType::NODE_WHILE_STMT: {
247 const auto* ws = static_cast<const WhileStmt*>(node);
248 auto cond = cloneWithSubstitution(ws->condition.get(), subst, d, unsupportedNodeName);
249 if (!cond)
250 return nullptr;
251 auto body = cloneWithSubstitution(ws->body.get(), subst, d, unsupportedNodeName);
252 if (!body)
253 return nullptr;
254 return std::make_shared<WhileStmt>(std::move(cond), std::move(body), ws->line, ws->column);
255 }
256 case NodeType::NODE_FOR_STMT: {
257 const auto* fs = static_cast<const ForStmt*>(node);
258 std::shared_ptr<ASTNode> init, cond, upd;
259 if (fs->initializer) {
260 init = cloneWithSubstitution(fs->initializer.get(), subst, d, unsupportedNodeName);
261 if (!init)
262 return nullptr;
263 }
264 if (fs->condition) {
265 cond = cloneWithSubstitution(fs->condition.get(), subst, d, unsupportedNodeName);
266 if (!cond)
267 return nullptr;
268 }
269 if (fs->update) {
270 upd = cloneWithSubstitution(fs->update.get(), subst, d, unsupportedNodeName);
271 if (!upd)
272 return nullptr;
273 }
274 auto body = cloneWithSubstitution(fs->body.get(), subst, d, unsupportedNodeName);
275 if (!body)
276 return nullptr;
277 return std::make_shared<ForStmt>(std::move(init), std::move(cond), std::move(upd), std::move(body), fs->line,
278 fs->column);
279 }
280 case NodeType::NODE_RETURN_STMT: {
281 const auto* rs = static_cast<const ReturnStmt*>(node);
282 std::shared_ptr<ASTNode> val;
283 if (rs->value) {
284 val = cloneWithSubstitution(rs->value.get(), subst, d, unsupportedNodeName);
285 if (!val)
286 return nullptr;
287 }
288 return std::make_shared<ReturnStmt>(std::move(val), rs->line, rs->column);
289 }
290 case NodeType::NODE_PRINT_STMT: {
291 const auto* ps = static_cast<const PrintStmt*>(node);
292 std::vector<std::shared_ptr<ASTNode>> vals;
293 if (!cloneList(ps->values, subst, d, unsupportedNodeName, vals))
294 return nullptr;
295 return std::make_shared<PrintStmt>(std::move(vals), ps->line, ps->column);
296 }
297 case NodeType::NODE_BREAK_STMT:
298 return std::make_shared<BreakStmt>(node->line, node->column);
299 case NodeType::NODE_CONTINUE_STMT:
300 return std::make_shared<ContinueStmt>(node->line, node->column);
301 case NodeType::NODE_THROW_STMT: {
302 const auto* ts = static_cast<const ThrowStmt*>(node);
303 auto expr = cloneWithSubstitution(ts->expression.get(), subst, d, unsupportedNodeName);
304 if (!expr)
305 return nullptr;
306 return std::make_shared<ThrowStmt>(std::move(expr), ts->line, ts->column);
307 }
308 case NodeType::NODE_INDEX_ASSIGN: {
309 const auto* ia = static_cast<const IndexAssign*>(node);
310 auto obj = cloneWithSubstitution(ia->object.get(), subst, d, unsupportedNodeName);
311 if (!obj)
312 return nullptr;
313 auto idx = cloneWithSubstitution(ia->index.get(), subst, d, unsupportedNodeName);
314 if (!idx)
315 return nullptr;
316 auto val = cloneWithSubstitution(ia->value.get(), subst, d, unsupportedNodeName);
317 if (!val)
318 return nullptr;
319 return std::make_shared<IndexAssign>(std::move(obj), std::move(idx), std::move(val), ia->line, ia->column);
320 }
321 case NodeType::NODE_MEMBER_ASSIGN: {
322 const auto* ma = static_cast<const MemberAssign*>(node);
323 auto obj = cloneWithSubstitution(ma->object.get(), subst, d, unsupportedNodeName);
324 if (!obj)
325 return nullptr;
326 auto val = cloneWithSubstitution(ma->value.get(), subst, d, unsupportedNodeName);
327 if (!val)
328 return nullptr;
329 return std::make_shared<MemberAssign>(std::move(obj), ma->fieldName, std::move(val), ma->line, ma->column);
330 }
331 case NodeType::NODE_TRY_STMT: {
332 const auto* ts = static_cast<const TryStmt*>(node);
333 auto tryB = cloneWithSubstitution(ts->tryBlock.get(), subst, d, unsupportedNodeName);
334 if (!tryB)
335 return nullptr;
336 std::shared_ptr<ASTNode> catchB, finallyB;
337 if (ts->catchBlock) {
338 catchB = cloneWithSubstitution(ts->catchBlock.get(), subst, d, unsupportedNodeName);
339 if (!catchB)
340 return nullptr;
341 }
342 if (ts->finallyBlock) {
343 finallyB = cloneWithSubstitution(ts->finallyBlock.get(), subst, d, unsupportedNodeName);
344 if (!finallyB)
345 return nullptr;
346 }
347 auto cloned = std::make_shared<TryStmt>(std::move(tryB), ts->catchVarName, std::move(catchB),
348 std::move(finallyB), ts->line, ts->column);
349 cloned->catchKeywordLine = ts->catchKeywordLine;
350 cloned->finallyKeywordLine = ts->finallyKeywordLine;
351 return cloned;
352 }
353 default:
354 // match/yield/await/lambda/class/enum 声明等仍不在宏体支持范围
355 if (unsupportedNodeName)
356 *unsupportedNodeName = node->nodeName();
357 return nullptr;
358 }
