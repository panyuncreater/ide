135/// 判断 src[pos..] 是否以关键字 kw 开头且其前后为单词边界（非标识符字符）。
136bool matchKeywordAt(const std::string& src, size_t pos, const std::string& kw) {
137 if (pos + kw.size() > src.size())
138 return false;
139 if (src.compare(pos, kw.size(), kw) != 0)
140 return false;
141 if (pos > 0) {
142 char pc = src[pos - 1];
143 if (std::isalnum(static_cast<unsigned char>(pc)) || pc == '_')
144 return false;
145 }
146 size_t after = pos + kw.size();
147 if (after < src.size()) {
148 char nc = src[after];
149 if (std::isalnum(static_cast<unsigned char>(nc)) || nc == '_')
150 return false;
151 }
152 return true;
153}

155/// 把源码按顶层结构切成「函数定义」与「其余顶层语句（调用）」两部分。
156/// - 顶层 `fun 名(...) {...}`（含可选结尾 ;）归入 first（defs）
157/// - 其余顶层语句（如 print(...);）归入 second（rest / 调用）
158/// 花括号与字符串内容均被跳过，不误计嵌套深度。
159/// 用途：学生代码提供函数实现，测试用例参考代码提供调用；把两者组合后运行，
160/// 才能真正以学生实现判分（原实现直接跑参考代码 tc.code，与学生代码无关，恒满分）。
161std::pair<std::string, std::string> splitDefsAndRest(const std::string& src) {
162 std::string defs;
163 std::string rest;
164 size_t i = 0;
165 const size_t n = src.size();
166 while (i < n) {
167 char c = src[i];
168 if (std::isspace(static_cast<unsigned char>(c))) {
169 ++i;
170 continue;
171 }
172 if (matchKeywordAt(src, i, "fun")) {
173 const size_t start = i;
174 size_t j = i;
175 bool inStr = false;
176 char strCh = 0;
177 // 推进到函数体起始 '{'
178 while (j < n) {
179 char cj = src[j];
180 if (inStr) {
181 if (cj == '\\') {
182 j += 2;
183 continue;
184 }
185 if (cj == strCh)
186 inStr = false;
187 ++j;
188 continue;
189 }
190 if (cj == '"' || cj == '\'') {
191 inStr = true;
192 strCh = cj;
193 ++j;
194 continue;
195 }
196 if (cj == '{')
197 break;
198 ++j;
199 }
200 if (j >= n) { // 无函数体（异常）：剩余全归 rest
201 rest.append(src, start, n - start);
202 i = n;
203 break;
204 }
205 // 从 '{' 做花括号匹配
206 int depth = 0;
207 inStr = false;
208 strCh = 0;
209 while (j < n) {
210 char cj = src[j];
211 if (inStr) {
212 if (cj == '\\') {
213 j += 2;
214 continue;
215 }
216 if (cj == strCh)
217 inStr = false;
218 ++j;
219 continue;
220 }
221 if (cj == '"' || cj == '\'') {
222 inStr = true;
223 strCh = cj;
224 ++j;
225 continue;
226 }
227 if (cj == '{') {
228 ++depth;
229 } else if (cj == '}') {
230 --depth;
231 if (depth == 0) {
232 ++j;
233 break;
234 }
235 }
236 ++j;
237 }
238 // 吞掉紧随的可选 ';' 与前导空白
239 while (j < n && std::isspace(static_cast<unsigned char>(src[j])))
240 ++j;
241 if (j < n && src[j] == ';')
242 ++j;
243 defs.append(src, start, j - start);
244 defs.push_back('\n');
245 i = j;
246 } else {
247 // 顶层语句：推进到 depth 0 的分号（尊重字符串与括号）
248 const size_t start = i;
249 int depth = 0;
250 bool inStr = false;
251 char strCh = 0;
252 size_t j = i;
253 while (j < n) {
254 char cj = src[j];
255 if (inStr) {
256 if (cj == '\\') {
257 j += 2;
258 continue;
259 }
260 if (cj == strCh)
261 inStr = false;
262 ++j;
263 continue;
264 }
265 if (cj == '"' || cj == '\'') {
266 inStr = true;
267 strCh = cj;
268 ++j;
269 continue;
270 }
271 if (cj == '{' || cj == '[' || cj == '(') {
272 ++depth;
273 } else if (cj == '}' || cj == ']' || cj == ')') {
274 --depth;
275 } else if (cj == ';' && depth == 0) {
276 ++j;
277 break;
278 }
279 ++j;
280 }
281 rest.append(src, start, j - start);
282 rest.push_back('\n');
283 i = j;
284 }
285 }
286 return {defs, rest};
287}

449void ExerciseGraderPanel::onSubmitGrade() {
450 const auto& exs = ExerciseGraderLibrary::exercises();
451 int idx = exerciseCombo_->currentIndex();
452 if (idx < 0 || idx >= static_cast<int>(exs.size())) {

458 // 防重入：串行执行多用例 + 三后端期间 processEvents 不排除信号/定时器，
459 // 快速双击「提交评分」会触发重复评分与结果错乱。
460 if (grading_)
461 return;
462 grading_ = true;
463 // UX：评分期间禁用按钮 + 文案提示，避免误认为无响应
464 const QString submitOrigText = submitBtn_ ? submitBtn_->text() : QString();
465 if (submitBtn_) {
466 submitBtn_->setEnabled(false);
467 submitBtn_->setText(tr("评分中\xE2\x80\xA6"));
468 }
469 QApplication::processEvents(QEventLoop::ExcludeUserInputEvents);
470
471 // 评分准确性修复：以学生代码判分——提取学生的函数定义，与每个测试用例的

655/// 代码风格检查：逐行扫描 Tab 缩进/连续空行/行尾空格/长行/赋值运算符空格，
656/// 返回得分 0-20，note 填写扣分说明。
657int ExerciseGraderPanel::checkStyle(const std::string& code, QString& note) {
658 if (code.empty()) {
659 note = QString::fromUtf8("代码为空");
660 return 0;
661 }
662
663 int score = 20;
664 QStringList issues;
665
666 // 逐行扫描
667 std::stringstream ss(code);
668 std::string line;
669 int consecutiveBlanks = 0;
670 int maxConsecutiveBlanks = 0;
671 bool hasTab = false;
672 int longLines = 0;
673 int trailingWs = 0;
674
675 while (std::getline(ss, line)) {
676 // Tab 检查
677 if (line.find('\t') != std::string::npos) {
678 hasTab = true;
679 }
680 // 空行检查
681 if (line.empty() || line.find_first_not_of(" \t") == std::string::npos) {
682 ++consecutiveBlanks;
683 maxConsecutiveBlanks = std::max(maxConsecutiveBlanks, consecutiveBlanks);
684 } else {
685 consecutiveBlanks = 0;
686 }
687 // 行尾空格
688 if (!line.empty() && (line.back() == ' ' || line.back() == '\t')) {
689 ++trailingWs;
690 }
691 // 长行检查
692 if (line.size() > 120) {
693 ++longLines;
694 }
695 }
696
697 if (hasTab) {
698 score -= 5;
699 issues << QString::fromUtf8("包含 Tab 缩进（建议使用 4 空格）");
700 }
701 if (maxConsecutiveBlanks > 1) {
702 score -= 4;
703 issues << QString::fromUtf8("存在连续多余空行");
704 }
705 if (trailingWs > 0) {
706 int ded = std::min(trailingWs, 3);
707 score -= ded;
708 issues << QString::fromUtf8("行尾多余空格（%1 处）").arg(trailingWs);
709 }
710 if (longLines > 0) {
711 int ded = std::min(longLines * 2, 5);
712 score -= ded;
713 issues << QString::fromUtf8("超过 120 列的行（%1 行）").arg(longLines);
714 }
715
716 // 运算符空格检查：简单检测 = 赋值两侧无空格（排除 ==, <=, >=, !=, +=, -=）
717 int opIssues = 0;
718 for (size_t i = 1; i + 1 < code.size(); ++i) {
719 char prev = code[i - 1];
720 char cur = code[i];
721 char next = code[i + 1];
722 if (cur == '=' && prev != '=' && next != '=' && prev != '!' && prev != '<' && prev != '>' && prev != '+' &&
723 prev != '-' && prev != '*' && prev != '/' && prev != ' ' && prev != '\t' && prev != '\n' && next != ' ' &&
724 next != '\t' && next != '\n' && next != '=') {
725 ++opIssues;
726 }
727 }
728 if (opIssues > 0) {
729 int ded = std::min(opIssues, 3);
730 score -= ded;
731 issues << QString::fromUtf8("赋值运算符两侧缺少空格（%1 处）").arg(opIssues);
732 }
733
734 if (score < 0) {
735 score = 0;
736 }
737 if (score > 20) {
738 score = 20;
739 }
740
741 if (issues.isEmpty()) {
742 note = QString::fromUtf8("代码风格良好");
743 } else {
744 note = issues.join(QString::fromUtf8("；"));
745 }
746 return score;
747}
