25// 调用约定（栈式 VM 统一框架）
26// ------------------------------------------------------------
27// 每次函数/方法调用都在操作数栈上"就地"构造被调用者的活动记录，布局固定为：
28// [ this(仅方法，slot 0) | field0..fieldN(仅方法) | arg0..argM-1 | local0..localK-1 ]
29// 其中：
30// · basePointer = 推入上述所有槽后，stack_.size() - localCount（即第一个槽的索引）。
31// 被调用者用 stack_[basePointer+slot] 访问局部/参数/字段，与调用者栈完全隔离。
32// · returnIp = 调用指令的下一条指令地址，OP_RETURN 时恢复到调用者帧 .ip，
33// 避免调用者重复执行 CALL 指令（否则会无限递归）。
34// · 参数在调用前已由调用者从右到左压栈（栈顶是最后一个实参）；被调用者弹出参数
35// 后 push(this/字段/参数) 重建完整框架，再补 push 剩余 localCount 个 null 槽。
36// · extraSlots = localCount - 已推入槽数，若为负说明编译期帧布局与实际不一致（损坏）。
37// · 调用返回时 OP_RETURN 执行 stack_.resize(savedBp) 一次性回收整个被调用者框架，
38// 并 push 返回值；方法调用还会把字段槽同步回 this / 接收者变量（见 writeBack 机制）。
39// · 深度防护：frames_.size() >= MAX_FRAMES 时报"调用栈溢出"；默认参数数量须在
40// [requiredArity, arity] 区间，缺失尾部参数从 defaultConstIndices 取字面量填充。
41// ============================================================

43VMResult VM::executeCallOps(OpCode op, size_t& ip) {
44 // S1 fix: 拆分为 8 个独立方法，降低圈复杂度（原 978 行 → 调度器 + 各方法 < 200 行）
45 switch (op) {
46 case OpCode::OP_RETURN:
47 return executeReturn(ip);
48 case OpCode::OP_CALL:
49 return executeCall(ip, false);
50 case OpCode::OP_CALL_EXPR:
51 return executeCall(ip, true);
52 case OpCode::OP_TAIL_CALL:
53 return executeTailCall(ip); // L18 eng-tailcall
54 case OpCode::OP_SUPER_CALL:
55 case OpCode::OP_METHOD_CALL:
56 return executeMethodCall(ip, op);
57 case OpCode::OP_CLOSURE:
58 return executeClosure(ip, op);
59 case OpCode::OP_CLASS_NEW:
60 return executeClassNew(ip, op);
61 case OpCode::OP_DEFINE_CLASS:
62 return executeDefineClass(ip, op);
63 default:
64 return runtimeError(ErrorFormat::formatStd("未知操作码: {}", static_cast<int>(op)));
65 }
66}

272// L18 eng-tailcall: executeTailCall — OP_TAIL_CALL 执行
273// 字节布局与 OP_CALL 相同（nameIdx 2B + argCount 1B），后随 OP_RETURN。
274// TCO 可行条件（全部满足才帧复用）：
275// 1. 目标命中 functionChunks_ 且非生成器（executeCallByName 中 functionChunks_
276// 优先级最高，命中即与普通调用目标一致）
277// 2. 当前帧非方法帧（this/字段槽写回语义不兼容）、非生成器帧（协程快照）
278// 3. argCount 在 [requiredArity, arity] 范围内（越界时降级，由普通路径报错）
279// 否则降级：直接转 executeCall(ip,false)——OP_TAIL_CALL 字节布局与 OP_CALL 相同，
280// returnIp=ip+4 恰指向后随 OP_RETURN，语义与 CALL+RETURN 完全等价。
281// 帧复用步骤：关闭当前帧 open upvalues → 填充默认参数 → 实参搬到 bp →
282// 栈收缩+补局部槽 → 换 chunk/ip/upvalues（保留 returnIp/basePointer）。
283// ============================================================
284VMResult VM::executeTailCall(size_t& ip) {
285 VMCallFrame& frame = currentFrame();
286 const BytecodeChunk& chunk = *frame.chunk;
287 uint16_t idx = chunk.code[ip + 1] | (chunk.code[ip + 2] << 8);
288 uint8_t argCount = chunk.code[ip + 3];
289 if (idx >= chunk.constants.size())
290 return runtimeError("常量池索引越界");
291 const std::string& funName = chunk.constants[idx].stringVal();
292
293 auto it = functionChunks_.find(funName);
294 bool canReuse = it != functionChunks_.end() && !it->second.isGenerator && !frame.isMethodCall &&
295 !frame.isInitCall && !chunk.isGenerator;
296 if (canReuse) {
297 const BytecodeChunk& target = it->second;
298 if (argCount < static_cast<uint8_t>(target.requiredArity) || argCount > static_cast<uint8_t>(target.arity)) {
299 canReuse = false; // arity 越界：降级由普通路径报一致错误
300 }
301 }
302 if (!canReuse) {
303 // 降级：OP_TAIL_CALL 字节布局与 OP_CALL 相同，returnIp=ip+4 指向后随 OP_RETURN
304 return executeCall(ip, false);
305 }
306 const BytecodeChunk& target = it->second;
307
308 // 默认参数填充（与 setupFunctionCallFrame 同逻辑：补推常量到栈顶）
309 if (argCount < static_cast<uint8_t>(target.arity)) {
310 int missingCount = target.arity - argCount;
311 int defaultStartIdx = static_cast<int>(target.defaultConstIndices.size()) - missingCount;
312 if (defaultStartIdx < 0 ||
313 static_cast<size_t>(defaultStartIdx + missingCount) > target.defaultConstIndices.size()) {
314 return executeCall(ip, false); // 异常形态降级，普通路径报错
315 }
316 for (int i = defaultStartIdx; i < defaultStartIdx + missingCount; ++i) {
317 uint16_t constIdx = target.defaultConstIndices[i];
318 if (constIdx == RuntimeLimits::NO_INDEX || constIdx >= target.constants.size()) {
319 popN(static_cast<size_t>(i - defaultStartIdx));
320 return executeCall(ip, false); // 非字面量默认值：降级
321 }
322 push(target.constants[constIdx]);
323 }
324 argCount = static_cast<uint8_t>(target.arity);
325 }
326
327 // 帧复用：先关闭指向当前帧栈槽的 open upvalues（闭包快照当轮值）
328 if (closeUpvaluesFrom(frame.basePointer) != VMResult::VM_OK)
329 return VMResult::VM_RUNTIME_ERROR;
330
331 // 实参搬运：栈顶 argCount 个 → basePointer 起始槽位
332 size_t argStart = stack_.size() - argCount;
333 for (uint8_t i = 0; i < argCount; ++i) {
334 stack_[frame.basePointer + i] = std::move(stack_[argStart + i]);
335 }
336 // 栈收缩到 bp+argCount，再补目标函数的额外局部槽
337 int extraSlots = target.localCount - argCount;
338 if (extraSlots < 0) {
339 return runtimeError(ErrorFormat::formatStd("函数调用帧布局损坏: localCount={} < argCount={}", target.localCount,
340 static_cast<int>(argCount)));
341 }
342 stack_.resize(frame.basePointer + argCount);
343 for (int i = 0; i < extraSlots; ++i) {
344 push(Value::nullValue());
345 }
346
347 // 切换帧目标（保留 returnIp/basePointer：调用者信息不变）
348 size_t savedIp = ip;
349 frame.chunk = &target;
350 frame.ip = 0;
351 frame.functionName = funName;
352 frame.upvalues.clear();
353 auto closureIt = functionClosures_.find(funName);
354 if (closureIt != functionClosures_.end() && closureIt->second.vmClosure()) {
355 frame.upvalues = closureIt->second.vmClosure()->upvalues;
356 }
357 notifyStep(savedIp, OpCode::OP_TAIL_CALL);
358 return VMResult::VM_OK;
359}

430VMResult VM::executeCallConstructor(size_t& ip, OpCode op, VMClassInfo& cls, const std::string& funName,

443
444 // 检查是否有 init 方法（沿继承链查找）
445 const BytecodeChunk* initChunkPtr = findMethodChunk(funName, "init");
446

450 std::vector<Value> defaults;

465 // R97 #11 fix: 三后端递归深度消息统一为"递归深度超过限制 (N)"
466 return runtimeError(
467 ErrorFormat::formatStd(ErrorMessages::kRecursionDepthExceededFmtStd, static_cast<int>(MAX_FRAMES)),
468 DiagCodes::kRecursionDepth);
469 }

471 // 推入 this
472 push(instance);
473 // 按方法 chunk 声明的字段顺序（含继承字段）推入字段值
474 // 性能修复: push(instance) 后 refCount=2，若用非 const fields() 会触发
475 // ensureUnique COW 深拷贝整个 fields unordered_map。改用 std::as_const
476 // 调用 const 重载，仅读不写时不触发 COW。类构造是高频热路径。
477 int fieldCount = 0;
478 if (initChunk.fieldOrder.empty()) {

480 fieldCount = 0;
481 } else {

483 auto fieldIt = std::as_const(instance).fields().find(fieldName);
484 if (fieldIt != std::as_const(instance).fields().end()) {
485 push(fieldIt->second);
486 } else {
487 push(Value::nullValue());
488 }
489 }
490 fieldCount = static_cast<int>(initChunk.fieldOrder.size());
491 }
492 // 推入参数
493 for (const auto& arg : args) {
494 push(arg);
495 }
496
497 // 预分配局部变量栈空间：方法体内 var 声明的局部变量需要栈槽
498 int preAllocated = 1 + fieldCount + argCount; // this + 字段 + 参数
499 int extraSlots = initChunk.localCount - preAllocated;
500 // V-P2-1 fix: extraSlots 为负表示帧布局损坏（fieldCount 与编译期不一致）
501 if (extraSlots < 0) {
502 // AUDIT-P1 fix: 错误返回前清理栈上已推入的 this+fields+args，
503 // 与 executeCall OP_CALL 普通路径（BUG-VM-02 fix）保持栈平衡。
504 // args 已含默认值追加，用 args.size() 反映栈上实际参数数。
505 popN(1 + fieldCount + static_cast<int>(args.size()));
506 return runtimeError(ErrorFormat::formatStd("类 {} 的 init 方法帧布局损坏: localCount={} < preAllocated={}",
507
508 funName, initChunk.localCount, preAllocated));
509 }
510 for (int i = 0; i < extraSlots; ++i) {
511 push(Value::nullValue());
512 }
513
514 VMCallFrame newFrame;
515 newFrame.chunk = initChunkPtr;
516 newFrame.returnIp = ip + 4;
517 newFrame.basePointer = stack_.size() - initChunk.localCount;
518 newFrame.functionName = initChunkPtr->name;
519 newFrame.ip = 0;
520 newFrame.isMethodCall = true; // 使 OP_RETURN 同步字段到 this
521 newFrame.isInitCall = true; // init 返回 this 而非 null
522 size_t savedIp = ip;
523 frames_.push_back(std::move(newFrame));
524
525 notifyStep(savedIp, op);
526 return VMResult::VM_OK;
527 }
528
529 // 无 init 方法：检查是否有多余参数（与解释器行为保持一致）
530 if (argCount > 0) {
531 return runtimeError(ErrorFormat::formatStd("类 {} 没有 init 方法，但传入了 {} 个参数", cls.name,
532
533 static_cast<int>(argCount)),
534 DiagCodes::kArityMismatch);
535 }
536 push(instance);
537 notifyStep(ip, op);
538 ip += 4;
539 return VMResult::VM_OK;
540}

695 }

697 notifyStep(savedIp, op);
698 currentFrame().ip = savedIp + 4;
699 return VMResult::VM_OK;

774// ============================================================
775// executeCallExprValue - 闭包值调用（OP_CALL_EXPR 路径）
776// 栈布局调整 + chunkPtr 获取 + 调用 setupFunctionCallFrame
777// ============================================================
778VMResult VM::executeCallExprValue(size_t& ip, OpCode op) {
779 VMCallFrame& frame = currentFrame();
780 const BytecodeChunk& chunk = *frame.chunk;
781
782 uint8_t argCount = chunk.code[ip + 1];
783 // VM-05/06: 从栈上获取闭包值和参数，执行调用
784 if (stack_.size() < static_cast<size_t>(argCount) + 1) {
785 return runtimeError("栈下溢: OP_CALL_EXPR");
786 }
787
788 // 编译器先 push 闭包值再 push 参数（CompilerStmt.cpp visitFunCall, AstIRBuilder.cpp visitFunCall），
789 // 实际栈布局: [..., closure, arg0, arg1, ..., argN-1]
790 // 需从栈中移除闭包值（在参数下方），保留参数在栈顶供新帧使用。
791 size_t closurePos = stack_.size() - argCount - 1;
792 Value callee = std::move(stack_[closurePos]);
793 for (size_t i = 0; i < static_cast<size_t>(argCount); ++i) {
794 stack_[closurePos + i] = std::move(stack_[closurePos + 1 + i]);
795 }
796 stack_.pop_back();
797
798 if (!callee.isClosure()) {
799 // R3-1 fix: 弹出参数以保持栈平衡
800 popN(argCount);
801 return runtimeError("表达式调用需要函数值", DiagCodes::kTypeMismatch);
802 }
803
804 // M3 fix: 优先使用闭包值中的 chunkPtr（直接指针，无哈希查找）
805 // 回退到名称查找兼容旧闭包（chunkPtr 为 null 时）
806 const BytecodeChunk* targetChunkPtr = nullptr;
807 if (callee.vmClosure() && callee.vmClosure()->chunkPtr) {
808 targetChunkPtr = callee.vmClosure()->chunkPtr;
809 } else {
810 auto chunkIt = functionChunks_.find(callee.closureName());
811 if (chunkIt != functionChunks_.end()) {
812 targetChunkPtr = &chunkIt->second;
813 }
814 }
815 if (!targetChunkPtr) {
816 // R3-1 fix: 弹出参数保持栈平衡
817 popN(argCount);
818 return runtimeError("未找到函数: " + callee.closureName(), "undefined-function");
819 }
820
821 const BytecodeChunk& targetChunk = *targetChunkPtr;
822 // F10: 支持默认参数
823 if (argCount < static_cast<uint8_t>(targetChunk.requiredArity) ||
824 argCount > static_cast<uint8_t>(targetChunk.arity)) {
825 // R3-1 fix: 弹出参数保持栈平衡
826 popN(argCount);
827 return runtimeError(ErrorFormat::formatStd("函数 {} 期望 {}-{} 个参数，但传入了 {} 个",
828
829 callee.closureName(), targetChunk.requiredArity,
830
831 targetChunk.arity, static_cast<int>(argCount)),
832 DiagCodes::kArityMismatch);
833 }
834
835 // R164 D.5: 生成器函数拦截——闭包值调用路径。
836 // OP_CALL_EXPR 是 2 字节指令。
837 if (targetChunk.isGenerator) {
838 // D.7 fix: C++ 参数求值顺序不定——先复制 closureName 到局部变量，避免 std::move(callee)
839 // 先求值导致 callee.box_=null，后续 callee.closureName() 触发 std::abort()。
840 std::string closureNameCopy = callee.closureName();
841 return createCoroutineValue(targetChunk, closureNameCopy, argCount, std::move(callee), ip, 2);
842 }
843
844 // VM-05/06: 绑定闭包 upvalues 到新帧
845 std::vector<std::shared_ptr<VMUpvalue>> upvalues;
846 if (callee.vmClosure()) {
847 upvalues = callee.vmClosure()->upvalues;
848 }
849
850 // OP_CALL_EXPR 是 2 字节指令：opcode + argCount(1B)
851 return setupFunctionCallFrame(targetChunk, callee.closureName(), argCount, ip + 2, ip, op, upvalues);
852}

854// ============================================================
855// setupFunctionCallFrame - 共享帧构造 helper
856// 默认参数填充 + MAX_FRAMES 检查 + extraSlots 预分配 + newFrame 构造 + push frame
857// 被 executeCallFunction 和 executeCallExprValue 共享，统一两路径的帧构造逻辑
858// 注：R123 修复了原两路径差异——错误消息文本"闭包调用帧布局损坏"vs"函数调用帧布局损坏"统一为后者，
859// pop 方式 popN(argCount) vs for 循环 pop 统一为 popN（更高效）
860// ============================================================
861VMResult VM::setupFunctionCallFrame(const BytecodeChunk& targetChunk, const std::string& functionName,
862 uint8_t& argCount, size_t returnIp, size_t savedIp, OpCode op,
863 const std::vector<std::shared_ptr<VMUpvalue>>& upvalues) {
864 // F10: 为缺失的尾部参数填充默认值
865 if (argCount < static_cast<uint8_t>(targetChunk.arity)) {
866 int missingCount = targetChunk.arity - argCount;
867 int defaultStartIdx = static_cast<int>(targetChunk.defaultConstIndices.size()) - missingCount;
868 if (defaultStartIdx < 0 ||
869 static_cast<size_t>(defaultStartIdx + missingCount) > targetChunk.defaultConstIndices.size()) {
870 popN(argCount);
871 return runtimeError("函数 " + functionName + " 默认参数索引越界");
872 }
873 // BUG-023 fix (2026-07-18): 错误路径必须弹出已 push 的默认参数。
874 // 原 popN(argCount) 只清理原始参数，循环中已 push 的 (i-defaultStartIdx)
875 // 个默认参数残留在栈上，导致栈不平衡。修复：错误路径弹出
876 // argCount + pushedDefaults 个，其中 pushedDefaults = i - defaultStartIdx。
877 for (int i = defaultStartIdx; i < defaultStartIdx + missingCount; ++i) {
878 uint16_t constIdx = targetChunk.defaultConstIndices[i];
879 if (constIdx == RuntimeLimits::NO_INDEX) {
880 int pushedDefaults = i - defaultStartIdx;
881 popN(static_cast<size_t>(argCount) + static_cast<size_t>(pushedDefaults));
882 return runtimeError("函数 " + functionName + " 的默认参数包含非字面量表达式，VM 不支持");
883 }
884 if (constIdx >= targetChunk.constants.size()) {
885 int pushedDefaults = i - defaultStartIdx;
886 popN(static_cast<size_t>(argCount) + static_cast<size_t>(pushedDefaults));
887 return runtimeError("函数 " + functionName + " 默认参数常量索引越界");
888 }
889 push(targetChunk.constants[constIdx]);
890 }
891 argCount = static_cast<uint8_t>(targetChunk.arity);
892 }
893
894 if (frames_.size() >= MAX_FRAMES) {
895 popN(argCount);
896 // R97 #11 fix: 三后端递归深度消息统一
897 return runtimeError(
898 ErrorFormat::formatStd(ErrorMessages::kRecursionDepthExceededFmtStd, static_cast<int>(MAX_FRAMES)),
899 DiagCodes::kRecursionDepth);
900 }
901
902 // 预分配局部变量栈空间：函数体内 var 声明的局部变量需要栈槽，
903 // 但帧创建时栈上只有参数，需补推 null 填充额外槽位
904 int extraSlots = targetChunk.localCount - argCount;
905 // V-P2-1 fix: extraSlots 为负表示帧布局损坏
906 if (extraSlots < 0) {
907 popN(argCount);
908 return runtimeError(ErrorFormat::formatStd("函数调用帧布局损坏: localCount={} < argCount={}",
909
910 targetChunk.localCount, static_cast<int>(argCount)));
911 }
912 for (int i = 0; i < extraSlots; ++i) {
913 push(Value::nullValue());
914 }
915
916 VMCallFrame newFrame;
917 newFrame.chunk = &targetChunk;
918 newFrame.returnIp = returnIp;
919 newFrame.basePointer = stack_.size() - targetChunk.localCount;
920 newFrame.functionName = functionName;
921 newFrame.ip = 0;
922 newFrame.upvalues = upvalues; // 拷贝
923
924 frames_.push_back(std::move(newFrame));
925 notifyStep(savedIp, op);
926 return VMResult::VM_OK;
927}

929VMResult VM::executeMethodCall(size_t& ip, OpCode op) {
930 VMCallFrame& frame = currentFrame();
931 const BytecodeChunk& chunk = *frame.chunk;
932 bool isSuperCall = (op == OpCode::OP_SUPER_CALL);
933 const int instrLen = isSuperCall ? 9 : 7; // B1 fix: SUPER_CALL 多了 2 字节 classIdx
934 uint16_t idx = chunk.code[ip + 1] | (chunk.code[ip + 2] << 8);
935 uint8_t argCount = chunk.code[ip + 3];
936 uint16_t receiverVarIdx = chunk.code[ip + 4] | (chunk.code[ip + 5] << 8);
937 uint8_t receiverLocalSlotByte = chunk.code[ip + 6];
938 if (idx >= chunk.constants.size())
939 return runtimeError("常量池索引越界");
940 const std::string& methodName = chunk.constants[idx].stringVal();
941
942 // C8: 用 const 引用访问接收者，避免非变异方法的深拷贝
943 const Value& obj = peek(argCount);
944 // V-P2-25 fix: peek 栈下溢时已设置 hasError_ 并返回 null 哨兵，立即退出避免后续误报
945 if (hasError_)
946 return VMResult::VM_RUNTIME_ERROR;
947 // C10: 方法名一次性分类为枚举
948 BuiltinMethod method = classifyBuiltinMethod(methodName);
949
950 // ---- 数组内置方法 ----
951 if (obj.isArray()) {
952 // B7 fix: 提取到 dispatchArrayBuiltin，降低 executeCallOps 圈复杂度
953 VMResult r = dispatchArrayBuiltin(obj, method, methodName, argCount, receiverVarIdx, receiverLocalSlotByte, ip,
954 op, instrLen);
955 if (r != VMResult::VM_OK)
956 return r;
957 return VMResult::VM_OK;
958 }
959
960 // ---- 字典内置方法 ----

962 VMResult r = dispatchDictBuiltin(obj, method, methodName, argCount, receiverVarIdx, receiverLocalSlotByte, ip,
963 op, instrLen);
964 if (r != VMResult::VM_OK)
965 return r;
966 return VMResult::VM_OK;
967 }
968
969 // ---- 字符串内置方法（全部非变异，使用 const 引用）----
970 if (obj.isString()) {
971 VMResult r = dispatchStringBuiltin(obj, method, methodName, argCount, ip, op, instrLen);
972 if (r != VMResult::VM_OK)
973 return r;
974 return VMResult::VM_OK;

978 // 同步对象内部状态通过 shared_ptr<Inner> 共享，方法调用不修改 Value 本身，无需 writeBack
979 if (obj.isChannel() || obj.isMutex() || obj.isRwLock() || obj.isThread()) {
980 VMResult r = dispatchSyncObjectBuiltin(obj, methodName, argCount, ip, op, instrLen);
981 if (r != VMResult::VM_OK)
982 return r;
983 return VMResult::VM_OK;
984 }

998 }

1001 // 先保存类型信息，因为 pop 会使 obj 引用失效
1002 bool wasInstance = obj.isInstance();
1003 // P2 fix (null-access): 捕获 null 标志，供下方发出 null-access 诊断码
1004 bool wasNull = obj.isNull();
1005 std::string clsName = wasInstance ? obj.className() : "";

1007 // 类型（null/int/float/bool/closure）调用方法时，原消息"方法调用需要类实例"
1008 // 具有误导性。改为与 RegisterVM/Interpreter 一致的"类型 X 不支持方法 Y"。
1009 std::string typeName = wasInstance ? "" : obj.typeName();
1010 for (uint8_t i = 0; i < argCount; ++i)
1011 pop();
1012 pop();
1013 if (wasInstance) {
1014 return runtimeError("类 " + clsName + " 没有方法 " + methodName, "undefined-function");
1015 } else if (wasNull) {
1016 // P2 fix (null-access): null 值方法调用给出明确的 null-access 诊断码
1017 return runtimeError("不能在 null 值上访问属性或调用方法", DiagCodes::kNullAccess);
1018 } else {
1019 return runtimeError("类型 " + typeName + " 不支持方法 " + methodName, "undefined-function");
1020 }
1021}

1027VMResult VM::executeInstanceMethodCall(size_t& ip, OpCode op, Value obj, const std::string& methodName,
1028 uint8_t argCount, uint16_t receiverVarIdx, uint8_t receiverLocalSlotByte) {
1029 VMCallFrame& frame = currentFrame();
1030 const BytecodeChunk& chunk = *frame.chunk;
1031 bool isSuperCall = (op == OpCode::OP_SUPER_CALL);
1032 const int instrLen = isSuperCall ? 9 : 7; // B1 fix: SUPER_CALL 多了 2 字节 classIdx
1033
1034 // obj 已是值传递副本，无需再次拷贝（原代码 const Value objCopy = obj 已不再需要）
1035 // V-P2-12 fix: 使用 const 避免后续 fields()/className() 触发 COW 深拷贝
1036 const Value& objCopy = obj;
1037 // B1 fix: super 调用使用编译时编码的类名（而非运行时实例类名）
1038 // 避免 3+ 级继承时 super 查找回到子类导致死循环
1039 std::string searchClassName = objCopy.className();
1040 if (isSuperCall) {
1041 uint16_t classIdx = chunk.code[ip + 7] | (chunk.code[ip + 8] << 8);
1042 // V-P1-7 fix: classIdx 越界应报错而非静默降级到运行时类名（可能导致错误的 super 查找）
1043 if (classIdx >= chunk.constants.size()) {
1044 return runtimeError(
1045 ErrorFormat::formatStd("内部错误: super 调用的类名常量索引越界 ({})", static_cast<int>(classIdx)));
1046 }
1047 searchClassName = chunk.constants[classIdx].stringVal();
1048 auto clsIt = classInfo_.find(searchClassName);
1049 if (clsIt == classInfo_.end() || clsIt->second.superClassName.empty()) {
1050 return runtimeError("类 " + searchClassName + " 没有父类，不能使用 super");
1051 }
1052 searchClassName = clsIt->second.superClassName;
1053 }
1054 // 沿继承链查找方法（父类方法也可调用）
1055 const BytecodeChunk* targetChunkPtr = findMethodChunk(searchClassName, methodName);
1056 if (targetChunkPtr == nullptr) {
1057 // 方法未找到：pop 参数和接收者，返回错误（与原 executeMethodCall 末尾错误路径一致）
1058 popN(argCount + 1);
1059 return runtimeError("类 " + objCopy.className() + " 没有方法 " + methodName, "undefined-function");
1060 }
1061 const BytecodeChunk& targetChunk = *targetChunkPtr;
1062
1063 // F10: 支持默认参数
1064 if (argCount < static_cast<uint8_t>(targetChunk.requiredArity) ||
1065 argCount > static_cast<uint8_t>(targetChunk.arity)) {
1066 // PERF-12 fix: 批量 pop 用 popN（参数 + 接收者）
1067 popN(argCount + 1);
1068 return runtimeError(ErrorFormat::formatStd("方法 {} 期望 {}-{} 个参数，但传入了 {} 个", methodName,
1069
1070 targetChunk.requiredArity, targetChunk.arity,
1071
1072 static_cast<int>(argCount)),
1073 DiagCodes::kArityMismatch);
1074 }
1075
1076 if (frames_.size() >= MAX_FRAMES) {
1077 // BUG-VM-03 fix: 错误返回前 popN(argCount + 1) 清理栈上参数 + 接收者
1078 popN(argCount + 1);
1079 // R97 #11 fix: 三后端递归深度消息统一
1080 return runtimeError(
1081 ErrorFormat::formatStd(ErrorMessages::kRecursionDepthExceededFmtStd, static_cast<int>(MAX_FRAMES)),
1082 DiagCodes::kRecursionDepth);
1083 }
1084
1085 // 收集参数（反向填充，省去 reverse）
1086 if (stack_.size() < static_cast<size_t>(argCount) + 1)
1087 return runtimeError("栈下溢: OP_METHOD_CALL");
1088 SmallArgs<Value> args(argCount);
1089 for (int i = argCount - 1; i >= 0; --i) {
1090 args[i] = pop();
1091 }
1092 pop(); // 移除栈上的原始实例
1093
1094 // 推入 this（拷贝，方法内修改会被 writeBack 写回）
1095 push(objCopy);
1096 // 按方法 chunk 声明的字段顺序推入实例字段值
1097 int fieldCount = 0;
1098 if (targetChunk.fieldOrder.empty()) {
1099 // IR 路径方法不预留字段槽（见 executeClassNew 同名分支注释）
1100 fieldCount = 0;
1101 } else {
1102 for (const auto& fieldName : targetChunk.fieldOrder) {
1103 auto fieldIt = objCopy.fields().find(fieldName);
1104 if (fieldIt != objCopy.fields().end()) {
1105 push(fieldIt->second);
1106 } else {
1107 push(Value::nullValue());
1108 }
1109 }
1110 fieldCount = static_cast<int>(targetChunk.fieldOrder.size());
1111 }
1112 // 推入参数
1113 for (const auto& arg : args) {
1114 push(arg);
1115 }
1116
1117 // F10: 为缺失的尾部参数填充默认值
1118 if (argCount < static_cast<uint8_t>(targetChunk.arity)) {
1119 int missingCount = targetChunk.arity - argCount;
1120 int defaultStartIdx = static_cast<int>(targetChunk.defaultConstIndices.size()) - missingCount;
1121 if (defaultStartIdx < 0 ||
1122 static_cast<size_t>(defaultStartIdx + missingCount) > targetChunk.defaultConstIndices.size()) {
1123 // AUDIT-P1 fix: 错误返回前清理栈上已推入的 this+fields+args，
1124 // 与 executeCall OP_CALL 普通路径（BUG-VM-02 fix）保持栈平衡。
1125 // 此处 argCount 是原始传入参数数（尚未被默认值追加），fieldCount 已在上方计算。
1126 popN(1 + fieldCount + argCount);
1127 return runtimeError("方法 " + methodName + " 默认参数索引越界");
1128 }
1129 // BUG-023 fix (2026-07-18): 错误路径必须弹出已 push 的默认参数（见 OP_CALL 路径注释）
1130 for (int i = defaultStartIdx; i < defaultStartIdx + missingCount; ++i) {
1131 uint16_t constIdx = targetChunk.defaultConstIndices[i];
1132 if (constIdx == RuntimeLimits::NO_INDEX) {
1133 int pushedDefaults = i - defaultStartIdx;
1134 popN(static_cast<size_t>(1 + fieldCount + argCount) + static_cast<size_t>(pushedDefaults));
1135 return runtimeError("方法 " + methodName + " 的默认参数包含非字面量表达式，VM 不支持");
1136 }
1137 if (constIdx >= targetChunk.constants.size()) {
1138 int pushedDefaults = i - defaultStartIdx;
1139 popN(static_cast<size_t>(1 + fieldCount + argCount) + static_cast<size_t>(pushedDefaults));
1140 return runtimeError("方法 " + methodName + " 默认参数常量索引越界");
1141 }
1142 push(targetChunk.constants[constIdx]);
1143 }
1144 argCount = static_cast<uint8_t>(targetChunk.arity);
1145 }
1146
1147 // 预分配局部变量栈空间：方法体内 var 声明的局部变量需要栈槽
1148 int preAllocated = 1 + fieldCount + argCount; // this + 字段 + 参数
1149 int extraSlots = targetChunk.localCount - preAllocated;
1150 // V-P2-1 fix: extraSlots 为负表示帧布局损坏
1151 if (extraSlots < 0) {
1152 // AUDIT-P3-ROUND50 fix: 错误路径未清理栈上已推入的 this + fields + args，
1153 // 与同函数其他错误路径（L705/L714/L759/L765/L769 的 popN）不一致。
1154 popN(1 + fieldCount + argCount);
1155 return runtimeError(ErrorFormat::formatStd("方法 {} 帧布局损坏: localCount={} < preAllocated={}",
1156
1157 methodName, targetChunk.localCount, preAllocated));
1158 }
1159 for (int i = 0; i < extraSlots; ++i) {
1160 push(Value::nullValue());
1161 }
1162
1163 VMCallFrame newFrame;
1164 newFrame.chunk = targetChunkPtr;
1165 newFrame.returnIp = ip + instrLen; // B1 fix: SUPER_CALL 是 9 字节
1166 newFrame.basePointer = stack_.size() - targetChunk.localCount;
1167 newFrame.functionName = targetChunk.name;
1168 newFrame.ip = 0;
1169 newFrame.isMethodCall = true;
1170 newFrame.isInitCall = (methodName == "init"); // init 返回 this 而非 null
1171 // 记录接收者变量名（用于 writeBack 到 globals_）
1172 if (receiverVarIdx != RuntimeLimits::NO_INDEX && receiverVarIdx < chunk.constants.size() &&
1173 chunk.constants[receiverVarIdx].isString()) {
1174 newFrame.receiverVarName = chunk.constants[receiverVarIdx].stringVal();
1175 }
1176 // 记录接收者局部变量 slot（用于 writeBack 到调用者栈帧）
1177 newFrame.receiverLocalSlot = (receiverLocalSlotByte == RuntimeLimits::NO_SLOT) ? -1 : receiverLocalSlotByte;
1178 // W3-2-Bug2 fix: 若方法有捕获的 upvalue（函数内定义的类），填充帧的 upvalue 列表。
1179 // methodUpvalues_ 在 OP_DEFINE_CLASS 时为有 upvalue 描述符的方法创建 VMClosureData 捕获。
1180 auto upvIt = methodUpvalues_.find(targetChunk.name);
1181 if (upvIt != methodUpvalues_.end() && upvIt->second) {
1182 newFrame.upvalues = upvIt->second->upvalues;
1183 }
1184 size_t savedIp = ip;
1185 frames_.push_back(std::move(newFrame));
1186
1187 notifyStep(savedIp, op);
1188 return VMResult::VM_OK;
1189}

1191VMResult VM::executeClosure(size_t& ip, OpCode op) {
1192 VMCallFrame& frame = currentFrame();
1193 const BytecodeChunk& chunk = *frame.chunk;
1194 uint16_t idx = chunk.code[ip + 1] | (chunk.code[ip + 2] << 8);
1195 uint8_t upvalueCount = chunk.code[ip + 3]; // VM-05/06: 改为 upvalue 数量
1196 if (idx >= chunk.constants.size())
1197 return runtimeError("常量池索引越界");
1198 const std::string& funName = chunk.constants[idx].stringVal();
1199
1200 // 创建闭包值（参数名由 OP_CALL 按 arity 绑定，此处不填充假名）
1201 Value closure = Value::makeClosure(funName, nullptr, {});
1202
1203 // VM-05/06: 创建 VM 闭包数据并绑定 upvalue
1204 auto vmClosureData = std::make_shared<VMClosureData>();
1205 vmClosureData->functionName = funName;
1206 vmClosureData->upvalues.resize(upvalueCount);
1207 // M3 fix: 闭包值直接持有函数 chunk 指针（OP_CALL_EXPR 使用，避免名称查找）
1208 auto chunkIt = functionChunks_.find(funName);
1209 if (chunkIt != functionChunks_.end()) {
1210 vmClosureData->chunkPtr = &chunkIt->second;
1211 }
1212
1213 size_t instrBase = ip + 4; // OP_CLOSURE(1) + nameIdx(2) + upvalueCount(1)
1214
1215 for (uint8_t i = 0; i < upvalueCount; ++i) {
1216 uint8_t isLocal = chunk.code[instrBase + i * 2];
1217 uint8_t uvIndex = chunk.code[instrBase + i * 2 + 1];
1218
1219 if (isLocal) {
1220 // 直接捕获：创建新 upvalue 指向调用者帧的栈槽
1221 auto uv = std::make_shared<VMUpvalue>();
1222 uv->stackSlot = frame.basePointer + uvIndex;
1223 uv->isClosed = false;
1224 // #18 fix: 记录所属帧索引（当前帧 = frames_.size()-1），
1225 // 供 OP_SET_UPVALUE 跳过 O(frames_) 线性扫描
1226 uv->owningFrameIdx = frames_.size() - 1;
1227 vmClosureData->upvalues[i] = uv;
1228 // B5 fix: 插入有序索引（multimap，按 stackSlot 排序）
1229 openUpvalues_.emplace(uv->stackSlot, uv);
1230 } else {
1231 // 透传：复用调用者帧的 upvalue
1232 if (static_cast<size_t>(uvIndex) < frame.upvalues.size()) {
1233 vmClosureData->upvalues[i] = frame.upvalues[uvIndex];
1234 } else {
1235 // 降级：创建空 upvalue
1236 vmClosureData->upvalues[i] = std::make_shared<VMUpvalue>();
1237 vmClosureData->upvalues[i]->value = Value::nullValue();
1238 vmClosureData->upvalues[i]->isClosed = true;
1239 }
1240 }
1241 }
1242 closure.vmClosure() = vmClosureData;
1243
1244 // 注册到函数闭包表（OP_CALL 按名称查找时使用）
1245 functionClosures_[funName] = closure;
1246
1247 // V-P2-22 fix: closure 后续不再使用，std::move 入栈（map 已持有拷贝）
1248 push(std::move(closure));
1249 notifyStep(ip, op);
1250 ip = instrBase + upvalueCount * 2; // 跳过 upvalue 描述符
1251 return VMResult::VM_OK;
1252}

1285 }

1296 // （与解释器 visitVarDecl 一致：Point p; 自动调用 0 参数 init）

1300 if (argCount > 0) {
1301 shouldCallInit = true;
1302 } else if (initChunkPtr->requiredArity == 0) {
1303 shouldCallInit = true; // 0 必需参数（含全默认参数 init），自动构造时调用
1304 }
1305 }

1307 if (shouldCallInit) {
1308 // 创建 init 帧执行初始化
1309 const BytecodeChunk& initChunk = *initChunkPtr;

1311 std::vector<Value> defaults;
1312 if (!fillDefaultArgs(initChunk, argCount, className, defaults)) {
1313 return runtimeError(ErrorFormat::formatStd("构造函数 init 期望 {}-{} 个参数，但传入了 {} 个",
1314
1315 initChunk.requiredArity, initChunk.arity,
1316
1317 static_cast<int>(argCount)),
1318 DiagCodes::kArityMismatch);
1319 }
1320 // 将默认参数追加到 args 末尾
1321 for (auto& d : defaults) {
1322 args.push_back(std::move(d));
1323 }
1324
1325 if (frames_.size() >= MAX_FRAMES) {
1326 // R97 #11 fix: 三后端递归深度消息统一
1327 return runtimeError(
1328 ErrorFormat::formatStd(ErrorMessages::kRecursionDepthExceededFmtStd, static_cast<int>(MAX_FRAMES)),
1329 DiagCodes::kRecursionDepth);
1330 }
1331
1332 // 推入 this
1333 push(instance);
1334 // 按方法 chunk 声明的字段顺序（含继承字段）推入字段值
1335 int fieldCount = 0;
1336 if (initChunk.fieldOrder.empty()) {
1337 // IR 路径方法不预留字段槽（字段通过 this.field 成员访问，非本地槽）。
1338 // Compiler 路径方法总有 fieldOrder（即使空类也设为空 vector，此时
1339 // instance.fields() 也为空），故此处推 0 个字段对两端均安全。
1340 fieldCount = 0;
1341 } else {
1342 for (const auto& fieldName : initChunk.fieldOrder) {
1343 auto fieldIt = instance.fields().find(fieldName);
1344 if (fieldIt != instance.fields().end()) {
1345 push(fieldIt->second);
1346 } else {
1347 push(Value::nullValue());
1348 }
1349 }
1350 fieldCount = static_cast<int>(initChunk.fieldOrder.size());
1351 }
1352 // 推入参数
1353 // V-P2-24 fix: args 后续不再使用，std::move 入栈
1354 for (auto& arg : args) {
1355 push(std::move(arg));
1356 }
1357
1358 // 预分配局部变量栈空间：方法体内 var 声明的局部变量需要栈槽
1359 int preAllocated = 1 + fieldCount + argCount; // this + 字段 + 参数
1360 int extraSlots = initChunk.localCount - preAllocated;
1361 // V-P2-1 fix: extraSlots 为负表示帧布局损坏
1362 if (extraSlots < 0) {
1363 // AUDIT-P1 fix: 错误返回前清理栈上已推入的 this+fields+args，
1364 // 与 executeCallConstructor 同路径（L410-413）保持栈平衡。
1365 popN(1 + fieldCount + static_cast<int>(args.size()));
1366 return runtimeError(ErrorFormat::formatStd("类 {} 的 init 方法帧布局损坏: localCount={} < preAllocated={}",
1367
1368 className, initChunk.localCount, preAllocated));
1369 }
1370 for (int i = 0; i < extraSlots; ++i) {
1371 push(Value::nullValue());
1372 }
1373
1374 VMCallFrame newFrame;
1375 newFrame.chunk = initChunkPtr;
1376 newFrame.returnIp = ip + 4;
1377 newFrame.basePointer = stack_.size() - initChunk.localCount;
1378 newFrame.functionName = initChunkPtr->name;
1379 newFrame.ip = 0;
1380 newFrame.isMethodCall = true; // 使 OP_RETURN 同步字段到 this
1381 newFrame.isInitCall = true; // init 返回 this 而非 null
1382 size_t savedIp = ip;
1383 frames_.push_back(std::move(newFrame));
1384
1385 notifyStep(savedIp, op);
1386 return VMResult::VM_OK;
1387 }
1388
1389 // 无 init 或 init.arity 不匹配 argCount：推入实例（OP_INIT_FIELD 或手动 init 后续处理）
1390 // V-P1-8 fix: init 存在但参数不匹配（argCount==0 且 init.requiredArity>0）时报错，而非静默跳过
1391 if (initChunkPtr != nullptr && !shouldCallInit) {
1392 return runtimeError(ErrorFormat::formatStd("类 {} 的 init 期望至少 {} 个参数，但传入了 {} 个", cls.name,
1393
1394 initChunkPtr->requiredArity, static_cast<int>(argCount)),
1395 DiagCodes::kArityMismatch);
1396 }
1397
1398 // 无 init 但有参数：报错（与解释器一致）
1399 // BUG-VM-01 fix: 检查必须在 push(instance) 之前，错误返回时栈上不残留 instance。
1400 // 与同文件 executeCall 类构造路径（L327-336 检查在 push 之前）顺序一致。
1401 if (initChunkPtr == nullptr && argCount > 0) {
1402 return runtimeError(ErrorFormat::formatStd("类 {} 没有 init 方法，但传入了 {} 个参数", cls.name,
1403
1404 static_cast<int>(argCount)),
1405 DiagCodes::kArityMismatch);
1406 }
1407
1408 push(instance);
1409
1410 notifyStep(ip, op);
1411 ip += 4;
1412 return VMResult::VM_OK;
1413}

1415VMResult VM::executeDefineClass(size_t& ip, OpCode op) {
1416 VMCallFrame& frame = currentFrame();
1417 const BytecodeChunk& chunk = *frame.chunk;
1418 // 操作数: nameIdx(2B) + superNameIdx(2B)
1419 // superNameIdx == RuntimeLimits::NO_INDEX 表示无父类
1420 uint16_t idx = chunk.code[ip + 1] | (chunk.code[ip + 2] << 8);
1421 uint16_t superIdx = chunk.code[ip + 3] | (chunk.code[ip + 4] << 8);
1422 if (idx >= chunk.constants.size())
1423 return runtimeError("常量池索引越界");
1424 const std::string& className = chunk.constants[idx].stringVal();
1425 std::string superClassName;
1426 if (superIdx != RuntimeLimits::NO_INDEX && superIdx < chunk.constants.size()) {
1427 superClassName = chunk.constants[superIdx].stringVal();
1428 }
1429 // V-P2-17 fix: templateInstance 只读不写，改为 const 避免 fields() 触发 COW 深拷贝
1430 const Value templateInstance = pop();
1431
1432 // 从模板实例提取当前类字段（M3 fix: 使用 pendingFieldOrder_ 保持声明顺序）
1433 std::vector<std::string> ownFieldOrder = std::move(pendingFieldOrder_);
1434 std::unordered_map<std::string, Value> ownFieldDefaults;
1435 if (templateInstance.isInstance()) {
1436 for (const auto& fieldName : ownFieldOrder) {
1437 auto it = templateInstance.fields().find(fieldName);
1438 if (it != templateInstance.fields().end()) {
1439 ownFieldDefaults[fieldName] = it->second;
1440 }
1441 }
1442 } else {
1443 return runtimeError("OP_DEFINE_CLASS: 模板值不是实例");
1444 }
1445
1446 // 注册类信息（先注册以支持循环引用安全查找）
1447 VMClassInfo info;
1448 info.name = className;
1449 info.superClassName = superClassName;
1450 classInfo_[className] = info;
1451
1452 // BUG-INH-AUDIT-7 fix: 父类重定义时，所有子类的 fieldOrder/fieldDefaults/methodCache
1453 // 缓存失效（原实现仅更新当前类，子类仍用旧父类字段布局）。
1454 // 遍历所有已注册类，若其继承链包含当前重定义的类，清空其缓存。
1455 // methodCache 是 mutable 需单独清空，fieldOrder/fieldDefaults 在下次构造时
1456 // 由 executeClassNew 重新合并（实际上 executeDefineClass 已合并，这里清空
1457 // 子类缓存的 fieldOrder/fieldDefaults 强制下次重定义时重新计算）。
1458 // 注意：单次运行（单文件编译执行）下，类定义按声明顺序处理，子类总在父类
1459 // 之后定义，此循环通常是空操作。此修复主要针对 REPL 场景下的重定义。
1460 for (auto& kv : classInfo_) {
1461 if (kv.first == className)
1462 continue; // 跳过当前类
1463 // 沿继承链查找是否依赖当前重定义的类
1464 std::string cur = kv.second.superClassName;
1465 for (int guard = 0; guard < 64 && !cur.empty(); ++guard) {
1466 if (cur == className) {
1467 // 子类 kv.first 依赖重定义的父类 className，清空缓存
1468 kv.second.fieldOrder.clear();
1469 kv.second.fieldDefaults.clear();
1470 kv.second.methodCache.clear();
1471 break;
1472 }
1473 auto it = classInfo_.find(cur);
1474 if (it == classInfo_.end())
1475 break;
1476 cur = it->second.superClassName;
1477 }
1478 }
1479
1480 // 沿继承链合并父类字段（父类字段在前，子类覆盖同名字段）
1481 // 先按"父→子"顺序收集字段名，子类已存在的字段不重复添加
1482 std::vector<std::string> mergedOrder;
1483 std::unordered_map<std::string, Value> mergedDefaults;
1484
1485 std::vector<std::string> chain;
1486 std::string cur = superClassName;
1487 for (int guard = 0; guard < MAX_INHERITANCE_DEPTH && !cur.empty(); ++guard) {
1488 auto clsIt = classInfo_.find(cur);
1489 if (clsIt == classInfo_.end()) {
1490 return runtimeError("未定义的父类: " + cur, "undefined-function");
1491 }
1492 chain.push_back(cur);
1493 cur = clsIt->second.superClassName;
1494 }
1495 // V-P2 fix: 循环继承检测 — guard 耗尽但 cur 仍非空说明存在继承环
1496 if (!cur.empty()) {
1497 return runtimeError("类继承链过深或存在循环继承: " + className + " -> " + cur);
1498 }
1499 // AUDIT-BUG-C3 fix: 正序遍历链（直接父类在前，根祖先在后），
1500 // 配合 "if not found then add" 去重——直接父类同名字段先入表，根祖先被跳过。
1501 // 原倒序遍历导致根祖先先入表，直接父类同名字段被跳过，三后端不一致。
1502 // 与 Interpreter InterpreterCalls.cpp:317-337 语义对齐。
1503 for (const auto& clsName : chain) {
1504 auto clsIt = classInfo_.find(clsName);
1505 if (clsIt == classInfo_.end())
1506 continue;
1507 for (const auto& fieldName : clsIt->second.fieldOrder) {
1508 if (mergedDefaults.find(fieldName) == mergedDefaults.end()) {
1509 mergedOrder.push_back(fieldName);
1510 mergedDefaults[fieldName] = clsIt->second.fieldDefaults[fieldName];
1511 }
1512 }
1513 }
1514 // 当前类字段最后处理（覆盖父类同名字段）
1515 for (const auto& fieldName : ownFieldOrder) {
1516 if (mergedDefaults.find(fieldName) == mergedDefaults.end()) {
1517 mergedOrder.push_back(fieldName);
1518 }
1519 mergedDefaults[fieldName] = ownFieldDefaults[fieldName];
1520 }
1521
1522 VMClassInfo& registered = classInfo_[className];
1523 registered.fieldOrder = std::move(mergedOrder);
1524 registered.fieldDefaults = std::move(mergedDefaults);
1525
1526 // 在全局变量中存储类标记（与解释器语义一致：类名是类型标识，不是实例）
1527 Value classVal(std::string("class:") + className);
1528 auto classGsIt = globalNameToSlot_.find(className);
1529 // V-P2-23 fix: classVal 后续不再使用，std::move 写入（两分支互斥）
1530 if (classGsIt != globalNameToSlot_.end()) {
1531 globalSlots_[classGsIt->second] = std::move(classVal);
1532 } else {
1533 globals_[className] = std::move(classVal);
1534 }
1535
1536 // W3-2-Bug2 fix: 为有 upvalue 的方法创建 upvalue 捕获。
1537 // 类定义在外层函数内时，方法可能引用外层函数的局部变量（upvalue）。
1538 // 此时仍在定义类的外层函数帧中，可以正确捕获栈槽/upvalue。
1539 // 方法调用时从 methodUpvalues_ 读取并填充方法帧的 upvalues。
1540 captureMethodUpvalues(className);
1541
1542 notifyStep(ip, op);
1543 ip += 5; // nameIdx(2B) + superNameIdx(2B) + opcode(1B)
1544 return VMResult::VM_OK;
1545}

1550 if (methodsIt == methodsByClass_.end())

1574 } else {

1576 uv->value = Value::nullValue();

1578 closureData->upvalues[i] = uv;
1579 }
1580 }
1581 }
1582 methodUpvalues_[chunkPtr->name] = closureData;
1583 }

1586// ============================================================
1587// R98 W2: 同步调用闭包值（供高阶函数共享层回调）
1588// ============================================================
1589// 手动构造调用帧（模拟 OP_CALL_EXPR 的帧设置），压入 frames_，
1590// 运行内部指令循环直到该帧弹出，从栈顶读取返回值。
1591//
1592// 关键不变量：
1593// 1. 调用前后 frames_.size() 不变（push 一次，pop 一次）
1594// 2. 调用前后 stack_.size() 不变（push args + locals，pop 全部 + push result）
1595// 3. 调用者 ip 引用可能因 frames_.push_back 重分配而悬垂——调用方负责保存/恢复
1596// 4. DoS 防护：内部循环使用本地计数器，限制为 maxInstructions（与外层独立）
1597// 5. 错误传播：hasError_ + diagnostics_ 由内部 executeOneInstruction 设置，
1598// 调用方检测 VM_RUNTIME_ERROR 后通过 getLastError() 获取错误信息（P2-12）
1599
1600VMResult VM::invokeClosureSync(const Value& closure, const Value* args, size_t argCount, int /*line*/, int /*column*/,
1601 Value& result) {
1602 if (!closure.isClosure()) {
1603 return runtimeError("高阶函数的参数必须是函数", DiagCodes::kTypeMismatch);
1604 }
1605
1606 // 查找闭包 chunk（与 OP_CALL_EXPR 路径一致）
1607 const BytecodeChunk* targetChunkPtr = nullptr;
1608 if (closure.vmClosure() && closure.vmClosure()->chunkPtr) {
1609 targetChunkPtr = closure.vmClosure()->chunkPtr;
1610 } else {
1611 auto chunkIt = functionChunks_.find(closure.closureName());
1612 if (chunkIt != functionChunks_.end()) {
1613 targetChunkPtr = &chunkIt->second;
1614 }
1615 }
1616 if (!targetChunkPtr) {
1617 return runtimeError("未找到函数: " + closure.closureName(), "undefined-function");
1618 }
1619 const BytecodeChunk& targetChunk = *targetChunkPtr;
1620
1621 // 参数数量检查
1622 if (argCount < static_cast<size_t>(targetChunk.requiredArity) ||
1623 argCount > static_cast<size_t>(targetChunk.arity)) {
1624 return runtimeError(ErrorFormat::formatStd("函数 {} 期望 {}-{} 个参数，但传入了 {} 个",
1625
1626 closure.closureName(), targetChunk.requiredArity,
1627
1628 targetChunk.arity, argCount),
1629 DiagCodes::kArityMismatch);
1630 }
1631
1632 // MAX_FRAMES 检查
1633 if (frames_.size() >= MAX_FRAMES) {
1634 return runtimeError(
1635 ErrorFormat::formatStd(ErrorMessages::kRecursionDepthExceededFmtStd, static_cast<int>(MAX_FRAMES)),
1636 DiagCodes::kRecursionDepth);
1637 }
1638
1639 // 压入参数（左到右，arg0 在栈低位）
1640 for (size_t i = 0; i < argCount; ++i) {
1641 push(args[i]);
1642 }
1643 // 填充默认参数
1644 size_t effectiveArgCount = argCount;
1645 if (argCount < static_cast<size_t>(targetChunk.arity)) {
1646 int missingCount = targetChunk.arity - static_cast<int>(argCount);
1647 int defaultStartIdx = static_cast<int>(targetChunk.defaultConstIndices.size()) - missingCount;
1648 if (defaultStartIdx < 0) {
1649 popN(argCount);
1650 return runtimeError("函数 " + closure.closureName() + " 默认参数索引越界");
1651 }
1652 for (int i = defaultStartIdx; i < defaultStartIdx + missingCount; ++i) {
1653 uint16_t constIdx = targetChunk.defaultConstIndices[i];
1654 if (constIdx == RuntimeLimits::NO_INDEX || constIdx >= targetChunk.constants.size()) {
1655 popN(effectiveArgCount);
1656 return runtimeError("函数 " + closure.closureName() + " 默认参数无效");
1657 }
1658 push(targetChunk.constants[constIdx]);
1659 ++effectiveArgCount;
1660 }
1661 }
1662
1663 // 预分配局部变量槽
1664 int extraSlots = targetChunk.localCount - static_cast<int>(effectiveArgCount);
1665 if (extraSlots < 0) {
1666 popN(effectiveArgCount);
1667 return runtimeError("闭包调用帧布局损坏");
1668 }
1669 for (int i = 0; i < extraSlots; ++i) {
1670 push(Value::nullValue());
1671 }
1672
1673 // 构造调用帧
1674 VMCallFrame newFrame;
1675 newFrame.chunk = &targetChunk;
1676 // R136 fix: 保存调用者帧的当前 IP，OP_RETURN 会用 returnIp 恢复调用者帧 ip。
1677 // 原设 returnIp=0 作哨兵、由调用方负责恢复，但 join 路径（dispatchSyncObjectBuiltin）
1678 // 只执行 ip += instrLen 而未覆盖 OP_RETURN 设入的 0，导致调用者帧 ip 被重置为 0，
1679 // 主线程从头重新执行字节码形成无限循环（spawn(worker)+join() 表现为 worker 反复执行）。
1680 // 高阶函数路径（executeCallHigherOrder）和 spawn 路径（executeCallSpawn）在
1681 // invokeClosureSync 返回后会显式设置 currentFrame().ip = savedIp + 4 覆盖此值，
1682 // 不受影响。设置 returnIp = currentFrame().ip 让 OP_RETURN 自动恢复调用者帧 ip。
1683 newFrame.returnIp = currentFrame().ip;
1684 newFrame.basePointer = stack_.size() - targetChunk.localCount;
1685 newFrame.functionName = closure.closureName();
1686 newFrame.ip = 0;
1687 if (closure.vmClosure()) {
1688 newFrame.upvalues = closure.vmClosure()->upvalues;
1689 }
1690
1691 size_t savedFrameCount = frames_.size();
1692 size_t savedTryStackSize = tryStack_.size(); // P3-A1: 快照 tryStack_ 检测异常穿透
1693 frames_.push_back(std::move(newFrame));
1694
1695 // 内部指令循环：执行直到帧弹出
1696 // DoS 防护：本地计数器限制为 maxInstructions（与外层 execute() 独立计数，
1697 // 总上限为 2x maxInstructions，对教学场景足够）
1698 int64_t localInstrCount = 0;
1699 int64_t maxInstr = RuntimeLimits::RuntimeConfig::instance().maxInstructions();
1700 while (frames_.size() > savedFrameCount && !hasError_) {
1701 if (++localInstrCount > maxInstr) {
1702 runtimeError("指令执行数超过上限，疑似无限循环");
1703 break;
1704 }
1705 VMResult r = executeOneInstruction();
1706 // L14: VM_EXCEPTION_THROW 表示异常被 try/catch 捕获，继续执行
1707 if (r == VMResult::VM_EXCEPTION_THROW)
1708 continue;
1709 if (r != VMResult::VM_OK || hasError_) {
1710 break;
1711 }
1712 }
1713
1714 if (hasError_) {
1715 // P2-12: 错误已写入 diagnostics_，调用方检测 VM_RUNTIME_ERROR 后通过 getLastError() 读取
1716 return VMResult::VM_RUNTIME_ERROR;
1717 }
1718
1719 // P3-A1: 异常穿透检测
1720 // spawn 闭包内 throw 触发 throwException 弹出闭包帧 + 弹出外层 try handler + push thrownValue
1721 // 到栈顶 + 设 main 帧 catchIp。此时 frames_ 已退回 savedFrameCount，tryStack_ 缩小，
1722 // 栈顶是 thrownValue（不应被当作闭包返回值 pop）。返回 VM_EXCEPTION_THROW 让调用方
1723 // （dispatchSyncObjectBuiltin / executeCallSpawn 等）不操作 ip/stack，由主循环保留 catchIp
1724 // 继续 catch 块。判别信号：spawn 闭包内 throw 后 join 报"栈下溢"-> 检查此处是否误 pop。
1725 if (tryStack_.size() < savedTryStackSize) {
1726 return VMResult::VM_EXCEPTION_THROW;
1727 }
1728
1729 // 闭包返回值在栈顶（OP_RETURN 已 push）
1730 if (stack_.empty()) {
1731 return runtimeError("闭包调用未返回值");
1732 }
1733 result = pop();
1734 return VMResult::VM_OK;
1735}

1737// ============================================================
1738// R164 协程/生成器：OP_YIELD 指令执行（重放模式 + 真挂起模式）
1739// ============================================================
1740// 语义: pop 栈顶 yield 值，递增运行时 yield 执行计数器。
1741// 真挂起模式（MINILANG_CORO_FIBER=1 且处于 .next() 挂起上下文）：
1742// 每个 yield 都挂起——push yieldValue 回栈（yield 表达式结果，对齐重放
1743// skip 语义），快照生成器执行状态进 CoroutineData::vmSuspension 后抛出
1744// VMYieldSignal，由 callCoroutineNext 捕获并返回给 .next() 调用方。
1745// 重放模式（MINILANG_CORO_FIBER=0 平台）：
1746// - 若计数器 == 当前重放目标 yieldId：抛出 VMYieldSignal(yieldValue)，被 callCoroutineNext 捕获
1747// - 若计数器 < 目标：push yieldValue 回栈（作为 yield 表达式的结果），继续执行
1748// ============================================================
1749
1750#if MINILANG_CORO_FIBER
1751namespace {
1752// CoroutineData 为 Value 私有嵌套类型，经公有访问器返回类型推导取得指针类别
1753using VmCoroutineDataRawPtr = std::decay_t<decltype(*std::declval<Value&>().coroutineData())>*;

1771VMResult VM::executeCoroutineOps(OpCode op, size_t& ip) {
1772 // 七特性 MVP 阶段 4：OP_AWAIT——同步 drain 协程到完成（与 Interpreter::visitAwaitExpr 对齐）。
1773 // pop 栈顶值：非协程恒等 push 回；协程循环 callCoroutineNext 直到 done，
1774 // push 最后一次 next() 的值。异常穿透与错误传播仿 dispatchCoroutineBuiltin 的 .next() 路径。
1775 if (op == OpCode::OP_AWAIT) {
1776 if (stack_.empty()) {
1777 return runtimeError("栈下溢: OP_AWAIT");
1778 }
1779 Value v = pop();
1780 if (!v.isCoroutine()) {
1781 push(std::move(v)); // await 同步值：恒等
1782 ip += 1;
1783 return VMResult::VM_OK;
1784 }
1785 auto* cd = v.coroutineData();
1786 Value last = cd->currentValueBox.empty() ? Value::nullValue() : cd->currentValueBox.front();
1787 size_t savedTryStackSize = tryStack_.size();
1788 int64_t guard = 0;
1789 while (!cd->done) {
1790 last = callCoroutineNext(v);
1791 if (hasError_) {
1792 return VMResult::VM_RUNTIME_ERROR;
1793 }
1794 if (tryStack_.size() < savedTryStackSize) {
1795 // 协程体内未捕获 throw 穿透：throwException 已截断栈并 push 异常值，
1796 // 不可 push/推进 ip（仿 dispatchCoroutineBuiltin 的 AUDIT-R7 F2 模式）
1797 return VMResult::VM_EXCEPTION_THROW;
1798 }
1799 if (++guard > RuntimeLimits::MAX_LOOP_ITERATIONS) {
1800 return runtimeError("await 驱动协程超出迭代上限（可能是无限生成器）");
1801 }
1802 }
1803 push(std::move(last));
1804 ip += 1; // OP_AWAIT 是 1 字节指令
1805 return VMResult::VM_OK;
1806 }
1807 if (op != OpCode::OP_YIELD) {
1808 return runtimeError(ErrorFormat::formatStd("未知协程操作码: {}", static_cast<int>(op)));
1809 }
1810 // 防御性检查：yield 在非生成器函数体中出现（Compiler 已在 visitYieldExpr 拦截，
1811 // 但字节码注入路径或动态构造的 chunk 可能绕过——此处兜底报错而非崩溃）
1812 if (currentCoroutineTargetYieldId_ < 0) {
1813 return runtimeError("yield 只能在 fun* 生成器函数体内出现");
1814 }
1815 if (stack_.empty()) {
1816 return runtimeError("栈下溢: OP_YIELD");
1817 }
1818 Value yieldValue = pop();
1819 // 运行时 yield 执行计数：每次 OP_YIELD 调用递增
1820 // 用于区分循环内同一 yield 节点的多次执行（编译期 yieldId 无法区分）
1821 int thisExecutionId = currentYieldExecutionCount_++;
1822#if MINILANG_CORO_FIBER
1823 if (currentSuspendingCoro_) {
1824 // 真挂起模式：每个 yield 都挂起（重放目标 gating 不适用）。yield 表达式
1825 // 结果 push 回栈（对齐重放 skip 语义），状态快照进 CoroutineData 后抛出
1826 // VMYieldSignal——callCoroutineNext 捕获后跳过生成器帧清理、保留快照供
1827 // 下次 .next() 恢复。
1828 Value exprResult = yieldValue; // 栈上表达式结果与 .next() 返回值各自独立持有
1829 push(std::move(exprResult));
1830 auto* cd = static_cast<VmCoroutineDataRawPtr>(currentSuspendingCoro_);
1831 auto sus = std::make_shared<VMCoroutineSuspension>();
1832 sus->baseStack = coroSuspendBaseStack_;
1833 sus->baseFrames = coroSuspendBaseFrames_;
1834 sus->stackSuffix.reserve(stack_.size() - coroSuspendBaseStack_);
1835 for (size_t i = coroSuspendBaseStack_; i < stack_.size(); ++i)
1836 sus->stackSuffix.push_back(std::move(stack_[i]));
1837 sus->frames.reserve(frames_.size() - coroSuspendBaseFrames_);
1838 for (size_t i = coroSuspendBaseFrames_; i < frames_.size(); ++i) {
1839 sus->frames.push_back(std::move(frames_[i]));
1840 sus->frames.back().basePointer -= coroSuspendBaseStack_; // 绝对→相对
1841 }
1842 for (const auto& t : tryStack_) {
1843 if (t.frameIndex >= coroSuspendBaseFrames_) {
1844 sus->tryStack.push_back(t);
1845 sus->tryStack.back().stackBase -= coroSuspendBaseStack_;
1846 sus->tryStack.back().frameIndex -= coroSuspendBaseFrames_;
1847 }
1848 }
1849 sus->resumeIp = frames_.back().ip + 1; // OP_YIELD 为 1 字节指令（触发抛出时 ip 未推进）
1850 for (auto it = openUpvalues_.lower_bound(coroSuspendBaseStack_); it != openUpvalues_.end(); ++it) {
1851 sus->openUpvalues.emplace_back(it->first - coroSuspendBaseStack_, it->second);
1852 if (auto uv = it->second.lock()) {
1853 uv->stackSlot -= coroSuspendBaseStack_;
1854 uv->owningFrameIdx -= coroSuspendBaseFrames_;
1855 }
1856 }
1857 openUpvalues_.erase(openUpvalues_.lower_bound(coroSuspendBaseStack_), openUpvalues_.end());
1858 cd->vmSuspension = std::move(sus);
1859 throw VMYieldSignal(std::move(yieldValue));
1860 }
1861#endif
1862 if (thisExecutionId == currentCoroutineTargetYieldId_) {
1863 // 命中目标 yield：抛出 VMYieldSignal，被 callCoroutineNext 捕获
1864 throw VMYieldSignal(std::move(yieldValue));
1865 }
1866 // thisExecutionId < target：跳过此 yield（重放模式核心）
1867 // push yieldValue 回栈作为 yield 表达式的结果，继续执行
1868 push(std::move(yieldValue));
1869 ip += 1; // OP_YIELD 是 1 字节指令
1870 return VMResult::VM_OK;
1871}

1873// ============================================================
1874// R164 D.5: 生成器函数调用拦截——创建协程值
1875// ============================================================
1876// 从栈弹出参数，填充默认参数，构造 Coroutine 值并 push 到栈顶。
1877// 被 executeCallFunction（OP_CALL）和 executeCallExprValue（OP_CALL_EXPR）共用。
1878// instrLen 为调用指令长度（OP_CALL=4, OP_CALL_EXPR=2），用于推进 ip。
1879// ============================================================
1880VMResult VM::createCoroutineValue(const BytecodeChunk& genChunk, const std::string& funName, uint8_t argCount,
1881 Value closureVal, size_t& ip, int instrLen) {
1882 // 弹出参数
1883 if (stack_.size() < static_cast<size_t>(argCount))
1884 return runtimeError("栈下溢: 生成器调用 " + funName);
1885 std::vector<Value> args(argCount);
1886 for (int i = argCount - 1; i >= 0; --i) {
1887 args[i] = pop();
1888 }
1889 // 填充默认参数
1890 if (argCount < static_cast<uint8_t>(genChunk.arity)) {
1891 int missingCount = genChunk.arity - argCount;
1892 int defaultStartIdx = static_cast<int>(genChunk.defaultConstIndices.size()) - missingCount;
1893 if (defaultStartIdx < 0) {
1894 return runtimeError("生成器 " + funName + " 默认参数索引越界");
1895 }
1896 for (int i = defaultStartIdx; i < defaultStartIdx + missingCount; ++i) {
1897 uint16_t constIdx = genChunk.defaultConstIndices[i];
1898 if (constIdx == RuntimeLimits::NO_INDEX || constIdx >= genChunk.constants.size()) {
1899 return runtimeError("生成器 " + funName + " 默认参数无效");
1900 }
1901 args.push_back(genChunk.constants[constIdx]);
1902 }
1903 }
1904 // 创建协程值并 push 到栈顶
1905 Value coro = Value::makeCoroutineVM(&genChunk, std::move(closureVal), std::move(args), genChunk.yieldCount);
1906 push(std::move(coro));
1907 ip += instrLen;
1908 return VMResult::VM_OK;
1909}

1911// ============================================================
1912// R164 D.5: 协程方法分发（.next() / .done()）
1913// ============================================================
1914// 协程内部状态通过 CoroutineData* 共享指针修改，无需 writeBack（与同步对象语义一致）。
1915// 栈布局: [..., receiver, arg0, ..., argN-1]
1916// 执行后: [..., result]（pop receiver + args，push result）
1917// ============================================================
1918VMResult VM::dispatchCoroutineBuiltin(Value& obj, const std::string& methodName, uint8_t argCount, size_t& ip,
1919 OpCode /*op*/, int instrLen) {
1920 if (methodName == "next") {
1921 if (argCount != 0) {
1922 // 清理栈上参数和接收者
1923 popN(argCount);
1924 pop();
1925 return runtimeError("coroutine.next() 不接受参数", DiagCodes::kArityMismatch);
1926 }
1927 // AUDIT-R7 F2 fix: 快照 tryStack_ 检测生成器体内未捕获 throw 的穿透（仿
1928 // dispatchSyncObjectBuiltin 的 P3-A1）。穿透时 throwException 已截断栈
1929 // （receiver 已被清除）并 push 异常值 + 设 catchIp，不可 pop/push/推进 ip。
1930 size_t savedTryStackSize = tryStack_.size();
1931 Value result = callCoroutineNext(obj);
1932 if (hasError_) {
1933 return VMResult::VM_RUNTIME_ERROR;
1934 }
1935 if (tryStack_.size() < savedTryStackSize) {
1936 return VMResult::VM_EXCEPTION_THROW;
1937 }
1938 // 设置 lastMutatedReceiver_ 为接收者原值（对齐 finishSharedBuiltin 和 RegisterVM
1939 // 的 lastMutatedReceiverReg_ = objReg）。IR 路径对所有 isVarRef 方法调用无条件
1940 // 发射 LOAD_MUTATED + STORE，若不设置，LOAD_MUTATED 会读到过期/null 值，STORE
1941 // 覆盖接收者变量（如协程值被 null 覆盖）。
1942 lastMutatedReceiver_ = obj;
1943 pop(); // pop receiver
1944 push(std::move(result)); // push .next() result
1945 ip += instrLen;
1946 return VMResult::VM_OK;
1947 }
1948 if (methodName == "done") {
1949 if (argCount != 0) {
1950 popN(argCount);
1951 pop();
1952 return runtimeError("coroutine.done() 不接受参数", DiagCodes::kArityMismatch);
1953 }
1954 auto* cd = obj.coroutineData();
1955 Value result(cd->done);
1956 // 同 .next()：设置 lastMutatedReceiver_ 为接收者原值
1957 lastMutatedReceiver_ = obj;
1958 pop();
1959 push(std::move(result));
1960 ip += instrLen;
1961 return VMResult::VM_OK;
1962 }
1963 // 未知方法：清理栈并报错
1964 popN(argCount);
1965 pop();
1966 return runtimeError("coroutine 类型不支持方法 " + methodName, "undefined-function");
1967}

1969// ============================================================
1970// R164 D.5: 协程 .next() 重放执行
1971// ============================================================
1972// 核心思路（与 Interpreter::callCoroutineNext 一致）：
1973// 1. 若已耗尽（done=true），返回 currentValueBox 中的最后值
1974// 2. 保存当前帧/栈状态，为生成器 body 设置新帧
1975// 3. 设置 currentCoroutineTargetYieldId_ = cd->currentYieldId，重置计数器
1976// 4. 运行内部指令循环，OP_YIELD 命中目标时抛出 VMYieldSignal
1977// 5. 捕获 VMYieldSignal → 保存 yield 值，递增 currentYieldId
1978// 6. 函数体自然结束（OP_RETURN）→ 标记 done=true，返回 return 值
1979// 7. 恢复帧/栈状态，返回 yield/return 值
1980//
1981// 重放模式：每次 .next() 从 ip=0 重新执行生成器 body，用运行时计数器跳过已返回的 yield。
1982// 对齐 Interpreter 重放模式语义，保证四后端一致性。
1983// ============================================================
1984Value VM::callCoroutineNext(Value& coroVal) {
1985 auto* cd = coroVal.coroutineData();
1986
1987 // 已耗尽：返回最后一次 yield/return 的值
1988 if (cd->done) {
1989 return cd->currentValueBox.empty() ? Value::nullValue() : cd->currentValueBox.front();
1990 }
1991
1992 const BytecodeChunk* genChunk = cd->vmChunk;
1993 if (!genChunk) {
1994 runtimeError("协程缺少生成器字节码块");
1995 return Value::nullValue();
1996 }
1997
1998 // MAX_FRAMES 检查
1999 if (frames_.size() >= MAX_FRAMES) {
2000 runtimeError(ErrorFormat::formatStd(ErrorMessages::kRecursionDepthExceededFmtStd, static_cast<int>(MAX_FRAMES)),
2001 DiagCodes::kRecursionDepth);
2002 return Value::nullValue();
2003 }
2004
2005 // 保存调用方上下文
2006 size_t savedFrameCount = frames_.size();
2007 size_t savedStackSize = stack_.size();
2008 // AUDIT-R7 F2 fix: 快照 tryStack_ 检测异常穿透（仿 invokeClosureSync 的 P3-A1）。
2009 // 生成器体内未捕获 throw 经 throwException 弹出生成器帧、消费外层 handler、
2010 // 截断栈并 push 异常值 + 设 catchIp。此时 hasError_ 为 false 且帧数回落，
2011 // 原实现误入"正常结束"分支：无条件置 done=true、dispatch 层再 pop 掉 catch
2012 // 期待的异常值并推进已被改写的 ip → catch 目标错位 + done 状态与 Interpreter 分歧。
2013 size_t savedTryStackSize = tryStack_.size();
2014 int savedTargetYieldId = currentCoroutineTargetYieldId_;
2015 int savedYieldExecCount = currentYieldExecutionCount_;
2016
2017#if MINILANG_CORO_FIBER
2018 if (cd->vmSuspension) {
2019 // —— 真挂起恢复：帧链/栈切片/try 栈/开放 upvalue 回位（相对→绝对重定基），
2020 // 从挂起点 ip 继续执行。快照消费后清空（再次挂起时产生新快照）。
2021 auto* sus = static_cast<VMCoroutineSuspension*>(cd->vmSuspension.get());
2022 for (auto& v : sus->stackSuffix)
2023 stack_.push_back(std::move(v));
2024 for (auto& f : sus->frames) {
2025 f.basePointer += savedStackSize; // 相对→绝对（重定基到本次入口深度）
2026 frames_.push_back(std::move(f));
2027 }
2028 for (auto& t : sus->tryStack) {
2029 t.stackBase += savedStackSize;
2030 t.frameIndex += savedFrameCount;
2031 tryStack_.push_back(t);
2032 }
2033 for (auto& entry : sus->openUpvalues) {
2034 openUpvalues_.emplace(entry.first + savedStackSize, entry.second);
2035 if (auto uv = entry.second.lock()) {
2036 uv->stackSlot += savedStackSize;
2037 uv->owningFrameIdx += savedFrameCount;
2038 }
2039 }
2040 frames_.back().ip = sus->resumeIp;
2041 // 最外层生成器帧的 returnIp 指向首次 .next() 的调用点；挂起模式下自然
2042 // 完成（OP_RETURN 以 returnIp 恢复调用方 ip，再由 dispatch 的 ip +=
2043 // instrLen 推进）可能发生在任意一次 .next() 中——重定基到本次调用点
2044 // （调用方帧即恢复前的栈顶）。帧链内部帧的 returnIp 指向生成器体内
2045 // 调用点，与调用方无关，保持不变。
2046 frames_[savedFrameCount].returnIp = frames_[savedFrameCount - 1].ip;
2047 savedFrameCount = sus->baseFrames;
2048 savedStackSize = sus->baseStack;
2049 cd->vmSuspension.reset();
2050 currentCoroutineTargetYieldId_ = cd->currentYieldId; // 过 OP_YIELD 防御检查（挂起模式不做目标比较）
2051 } else
2052#endif
2053 {
2054 // 压入参数（左到右，arg0 在栈低位）
2055 uint8_t argCount =
2056 static_cast<uint8_t>(std::min(cd->args.size(), static_cast<size_t>(std::numeric_limits<uint8_t>::max())));
2057 for (const auto& arg : cd->args) {
2058 push(arg);
2059 }
2060
2061 // 填充默认参数
2062 uint8_t effectiveArgCount = argCount;
2063 if (argCount < static_cast<uint8_t>(genChunk->arity)) {
2064 int missingCount = genChunk->arity - argCount;
2065 int defaultStartIdx = static_cast<int>(genChunk->defaultConstIndices.size()) - missingCount;
2066 if (defaultStartIdx >= 0) {
2067 for (int i = defaultStartIdx; i < defaultStartIdx + missingCount; ++i) {
2068 uint16_t constIdx = genChunk->defaultConstIndices[i];
2069 if (constIdx != RuntimeLimits::NO_INDEX && constIdx < genChunk->constants.size()) {
2070 push(genChunk->constants[constIdx]);
2071 ++effectiveArgCount;
2072 }
2073 }
2074 }
2075 }
2076
2077 // 预分配局部变量槽
2078 int extraSlots = genChunk->localCount - static_cast<int>(effectiveArgCount);
2079 if (extraSlots < 0) {
2080 // 帧布局损坏，清理并报错
2081 stack_.resize(savedStackSize);
2082 runtimeError("生成器 " + genChunk->name + " 帧布局损坏");
2083 return Value::nullValue();
2084 }
2085 for (int i = 0; i < extraSlots; ++i) {
2086 push(Value::nullValue());
2087 }
2088
2089 // 构造调用帧
2090 VMCallFrame newFrame;
2091 newFrame.chunk = genChunk;
2092 newFrame.returnIp = currentFrame().ip;
2093 newFrame.basePointer = stack_.size() - genChunk->localCount;
2094 newFrame.functionName = genChunk->name;
2095 newFrame.ip = 0;
2096 // 绑定闭包 upvalues
2097 // R164 fixup: vmClosure 改为单元素容器（绕开 Value 不完整类型 C2079），
2098 // 访问时取首元素再调用 Value::vmClosure() 取 shared_ptr<VMClosureData>
2099 if (!cd->vmClosureBox.empty()) {
2100 auto& closureVal = cd->vmClosureBox.front();
2101 if (closureVal.isClosure() && closureVal.vmClosure()) {
2102 newFrame.upvalues = closureVal.vmClosure()->upvalues;
2103 }
2104 }
2105 frames_.push_back(std::move(newFrame));
2106
2107 // 设置协程重放上下文
2108 currentCoroutineTargetYieldId_ = cd->currentYieldId;
2109 currentYieldExecutionCount_ = 0;
2110 }
2111
2112#if MINILANG_CORO_FIBER
2113 // 嵌套 .next()（await drain / 生成器体内再调 .next()）：登记前保存外层挂起
2114 // 上下文，全部退出路径恢复——外层执行体随后的 OP_YIELD 仍需以外层身份挂起。
2115 void* outerSuspendingCoro = currentSuspendingCoro_;
2116 size_t outerSuspendBaseStack = coroSuspendBaseStack_;
2117 size_t outerSuspendBaseFrames = coroSuspendBaseFrames_;
2118 auto restoreOuterCoroCtx = [&]() {
2119 currentSuspendingCoro_ = outerSuspendingCoro;
2120 coroSuspendBaseStack_ = outerSuspendBaseStack;
2121 coroSuspendBaseFrames_ = outerSuspendBaseFrames;
2122 };
2123 // 真挂起上下文登记：OP_YIELD 据此进入快照路径（含全新启动与恢复两种进入方式）
2124 currentSuspendingCoro_ = cd;
2125 coroSuspendBaseStack_ = savedStackSize;
2126 coroSuspendBaseFrames_ = savedFrameCount;
2127#endif
2128
2129 Value result = Value::nullValue();
2130 bool needCleanup = false; // true = 需手动清理帧/栈（yield 信号或错误路径）
2131
2132 try {
2133 // 内部指令循环：执行直到生成器帧弹出
2134 int64_t localInstrCount = 0;
2135 int64_t maxInstr = RuntimeLimits::RuntimeConfig::instance().maxInstructions();
2136 while (frames_.size() > savedFrameCount && !hasError_) {
2137 if (++localInstrCount > maxInstr) {
2138 runtimeError("指令执行数超过上限，疑似无限循环");
2139 break;
2140 }
2141 VMResult r = executeOneInstruction();
2142 // L14: VM_EXCEPTION_THROW 表示异常被 try/catch 捕获，继续执行
2143 if (r == VMResult::VM_EXCEPTION_THROW)
2144 continue;
2145 if (r != VMResult::VM_OK || hasError_) {
2146 break;
2147 }
2148 }
2149
2150 if (hasError_) {
2151 needCleanup = true;
2152 // AUDIT-R7 F2 fix（发现 3）: 不可恢复错误不置 done/currentValueBox（与
2153 // Interpreter/RegisterVM 对齐：两者错误路径均保持 done=false）。
2154 } else if (tryStack_.size() < savedTryStackSize) {
2155 // AUDIT-R7 F2 fix: 异常穿透到调用方 catch——不置 done、不动结果、不清理
2156 // （throwException 已就位 catch 状态：栈顶是异常值、调用方 ip=catchIp）。
2157 // 由 dispatchCoroutineBuiltin 同样检测并返回 VM_EXCEPTION_THROW。
2158 currentCoroutineTargetYieldId_ = savedTargetYieldId;
2159 currentYieldExecutionCount_ = savedYieldExecCount;
2160#if MINILANG_CORO_FIBER
2161 restoreOuterCoroCtx();
2162#endif
2163 return Value::nullValue();
2164 } else {
2165 if (stack_.size() > savedStackSize) {
2166 // 正常结束：OP_RETURN 已弹出帧并 push 返回值
2167 result = pop();
2168 }
2169 cd->done = true;
2170 cd->currentValueBox.clear();
2171 cd->currentValueBox.push_back(result);
2172 }
2173 } catch (const VMYieldSignal& e) {
2174 // 命中目标 yield：保存 yield 值，递增 currentYieldId
2175 result = std::move(e.yieldValue);
2176 cd->currentValueBox.clear();
2177 cd->currentValueBox.push_back(result);
2178 cd->currentYieldId++;
2179#if MINILANG_CORO_FIBER

2181 // 真挂起：执行状态已在 OP_YIELD 处快照进 cd->vmSuspension。最终 yield
