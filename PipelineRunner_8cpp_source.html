13bool PipelineRunner::runLexer(const std::string& source) {
14 // R54-1 fix: 直接调用 runLexer/runParser 会修改 lexer_/parser_ 内部状态（诊断、
15 // lastTokens_、astRoot_），若不失效缓存，后续 runFrontendPipeline 命中陈旧缓存
16 // 时返回的 diagnostics 裸指针指向已被改写的 DiagnosticBag，且 astRoot_ 已被替换
17 // 为其他源码（如 VmStackSandbox 关卡代码）的 AST，导致编辑器代码执行错误。
18 // invalidatePipelineCache() 已存在但从未被调用——此处补全。
19 invalidatePipelineCache();
20 lastTokens_ = lexer_.scan(source);
21 emit diagnosticsReady(lexer_.getDiagnostics());
22 return !lexer_.getDiagnostics().hasErrors();
23}

25bool PipelineRunner::runParser() {
26 // R54-1 fix: 同 runLexer，直接调用会改变管线状态，必须失效缓存。
27 invalidatePipelineCache();
28 astRoot_ = parser_.parse(lastTokens_);
29 emit diagnosticsReady(parser_.getDiagnostics());

33bool PipelineRunner::runCompiler() {
34 if (!astRoot_)
35 return false;
36
37 // P1-7 + L17: 字节码磁盘缓存。
38 // - StackVM 路径(useRegisterVM_=false): tryLoad/store，magic "MLBC"，扩展名 .mbc
39 // - RegisterVM 路径(useRegisterVM_=true): tryLoadRegister/storeRegister，
40 // magic "MLRC"，扩展名 .mrc（L17 新增，复用 L11 序列化层）
41 // 缓存 key = FNV-1a(source) + compilerMode(direct=0/IR=1) + optFlags（AUDIT-R3 P2-4）。
42 if (bytecodeCacheEnabled_ && !lastSource_.empty()) {
43 BytecodeCache::CacheKey key;
44 key.source = lastSource_;
45 key.compilerMode = compiler_.getUseIR() ? 1 : 0;
46 // AUDIT-R3 P2-4 fix: 优化开关入键——切换 irOptimize/irSSAOptimize 后
47 // 旧缓存不再命中，避免产出与当前优化配置不一致的字节码。
48 key.optFlags =
49 static_cast<uint8_t>((compiler_.getIROptimize() ? 1 : 0) | (compiler_.getIRSSAOptimize() ? 2 : 0));
50 // mtime 暂不校验(PipelineRunner 不持有文件路径);依赖 source hash 失效
51 key.mtime = 0;
52
53 if (compiler_.getUseRegisterVM()) {
54 // L17: RegisterVM 路径缓存
55 // PERF: 进程内 memo 命中免磁盘 IO + 反序列化（见 BytecodeCache::tryMemo）
56 if (auto memo = bytecodeCache_.tryMemoRegister(key)) {
57 compiler_.setLastRegisterResult(std::move(*memo));
58 compiler_.clearDiagnostics();
59 emit diagnosticsReady(compiler_.getDiagnostics());
60 return true;
61 }
62
63 auto cachedReg = bytecodeCache_.tryLoadRegister(key);
64 if (cachedReg) {
65 // 缓存命中:跳过编译,直接注入到 compiler_.lastRegisterResult_
66 compiler_.setLastRegisterResult(std::move(*cachedReg));
67 compiler_.clearDiagnostics();
68 emit diagnosticsReady(compiler_.getDiagnostics());
69 return true;
70 }
71
72 // 缓存未命中:正常编译
73 lastCompileResult_ = compiler_.compile(*astRoot_);
74 emit diagnosticsReady(compiler_.getDiagnostics());
75
76 // 仅在编译成功时写入缓存(避免缓存错误结果)
77 if (!compiler_.getDiagnostics().hasErrors()) {
78 bytecodeCache_.storeRegister(key, compiler_.getLastRegisterResult());
79 bytecodeCache_.storeMemoRegister(key, compiler_.getLastRegisterResult());
80 }
81 return !compiler_.getDiagnostics().hasErrors();
82 }
83
84 // StackVM 路径缓存
85 // PERF: 进程内 memo 命中免磁盘 IO + 反序列化（见 BytecodeCache::tryMemo）
86 if (auto memo = bytecodeCache_.tryMemo(key)) {
87 lastCompileResult_ = std::move(*memo);
88 compiler_.clearDiagnostics();
89 emit diagnosticsReady(compiler_.getDiagnostics());
90 return true;
91 }
92
93 auto cached = bytecodeCache_.tryLoad(key);
94 if (cached) {
95 // 缓存命中:跳过编译,直接使用反序列化的 CompileResult
96 lastCompileResult_ = std::move(*cached);
97 compiler_.clearDiagnostics(); // 清空陈旧诊断(上次某次编译的)
98 emit diagnosticsReady(compiler_.getDiagnostics());
99 return true; // 缓存命中的结果必然是成功编译的产物
100 }
101
102 // 缓存未命中:正常编译
103 lastCompileResult_ = compiler_.compile(*astRoot_);
104 emit diagnosticsReady(compiler_.getDiagnostics());
105
106 // 仅在编译成功时写入缓存(避免缓存错误结果)
107 if (!compiler_.getDiagnostics().hasErrors()) {
108 bytecodeCache_.store(key, lastCompileResult_);
109 bytecodeCache_.storeMemo(key, lastCompileResult_);
110 }
111 return !compiler_.getDiagnostics().hasErrors();
112 }
113
114 lastCompileResult_ = compiler_.compile(*astRoot_);
115 emit diagnosticsReady(compiler_.getDiagnostics());
116 return !compiler_.getDiagnostics().hasErrors();
117}

128PipelineRunner::PipelineResult PipelineRunner::runFrontendPipeline(const std::string& source) {
129 // P1-7: 记录源码,供 runCompiler 查询字节码缓存
130 lastSource_ = source;
131
132 // BUG-ORCH-7 fix: 源码级缓存——同一源码的连续调用（如 blockIfHasErrors + prepareRun）
133 // 直接复用上次的 Lexer/Parser 结果，避免重复执行前端管线
134 if (!cachedPipelineSource_.empty() && cachedPipelineSource_ == source) {
135 return cachedPipelineResult_;
136 }
137
138 PipelineResult result;
139
140 // 词法分析
141 try {
142 if (!runLexer(source)) {
143 result.status = PipelineStatus::LexerFailed;
144 result.setDiagnostics(lexer_.getDiagnostics()); // AUDIT-R5 R7 fix: 自持快照
145 cachePipelineResult(source, result);
146 return result;
147 }
148 } catch (const std::exception& e) {
149 result.status = PipelineStatus::LexerFailed;
150 result.errorMessage = e.what();
151 cachePipelineResult(source, result);
152 return result;
153 }
154
155 // 语法分析
156 try {
157 if (!runParser()) {
158 result.status = PipelineStatus::ParserFailed;
159 result.setDiagnostics(parser_.getDiagnostics()); // AUDIT-R5 R7 fix: 自持快照
160 cachePipelineResult(source, result);
161 return result;
162 }
163 } catch (const std::exception& e) {
164 result.status = PipelineStatus::ParserFailed;
165 result.errorMessage = e.what();
166 cachePipelineResult(source, result);
167 return result;
168 }
169
170 cachePipelineResult(source, result);
171 return result; // OK
172}
