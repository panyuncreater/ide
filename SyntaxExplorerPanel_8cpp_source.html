217/// 运行当前样例代码：Lexer 扫描 → Parser 解析（遇错显示诊断摘要）→
218/// Interpreter 执行，标准输出（含异常信息）写入右侧输出区。
219void SyntaxExplorerPanel::onRunSample() {
220 if (currentItemIndex_ < 0) {
221 statusLabel_->setText(QString::fromUtf8("请先选择产生式"));
222 return;
223 }
224 std::string source = codeEditor_->toPlainText().toStdString();
225 if (source.empty()) {
226 outputEdit_->setPlainText(QString::fromUtf8("（空代码）"));
227 return;
228 }
229
230 std::ostringstream out;
231 try {
232 Lexer lex;
233 auto tokens = lex.scan(source);
234 Parser parser;
235 auto ast = parser.parse(tokens);
236 if (parser.hasErrors()) {
237 const auto& diags = parser.getDiagnostics();
238 out << "[Parser 错误]\n" << diags.summary() << "\n";
239 outputEdit_->setPlainText(QString::fromUtf8(out.str().c_str()));
240 return;
241 }
242 Interpreter interp;
243 interp.setOutputCallback([&out](const std::string& s) { out << s << "\n"; });
244 interp.execute(*ast);
245 out << "[运行完成]";
246 } catch (const std::exception& e) {
247 out << "[异常] " << e.what();
248 }
249 outputEdit_->setPlainText(QString::fromUtf8(out.str().c_str()));
250 statusLabel_->setText(QString::fromUtf8("运行结束"));
251}
