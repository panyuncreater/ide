197/// 开始竞赛：对三后端各运行 N 次，汇总平均/最小/最大耗时，渲染表格、
198/// 柱状图、分析报告与一致性标签。三后端在主线程串行执行。
199void PerformanceRacePanel::onRunRace() {
200 std::string src = sourceEdit_->text().toStdString();
201 if (src.empty()) {
202 consistencyLabel_->setText(QString::fromUtf8("❌ 源码为空，请输入 MiniLang 源码"));
203 consistencyLabel_->setStyleSheet(QStringLiteral("color:#C0392B;"));
204 return;
205 }
206
207 int runCount = runCountSpin_->value();
208 if (runCount < 1) {
209 runCount = 1;
210 }
211
212 // Bug #72 fix: 执行期间禁用按钮，防止重复点击
213 raceBtn_->setEnabled(false);
214
215 // 串行执行三后端（主线程，避免引擎层非线程安全问题）
216 BackendPerfResult interp =
217 runBackendMultiple(src, QString::fromUtf8("Interpreter"), runCount, &PerformanceRacePanel::runInterpreterOnce);
218 BackendPerfResult stackvm =
219 runBackendMultiple(src, QString::fromUtf8("StackVM (IR)"), runCount, &PerformanceRacePanel::runStackVM_IR_Once);
220 BackendPerfResult regvm = runBackendMultiple(src, QString::fromUtf8("RegisterVM (IR)"), runCount,
221 &PerformanceRacePanel::runRegVM_IR_Once);
222
223 // 渲染表格行
224 renderResultRow(0, interp);
225 renderResultRow(1, stackvm);
226 renderResultRow(2, regvm);
227
228 // 渲染柱状图与分析报告
229 renderBarChart(interp, stackvm, regvm);
230 renderAnalysis(interp, stackvm, regvm);
231
232 // 一致性检查
233 renderConsistency(interp, stackvm, regvm);
234
235 // Bug #72 fix: 恢复按钮
236 raceBtn_->setEnabled(true);
237}

257/// 多次运行汇总：调用指定单次执行函数 N 次，记录每次耗时，计算平均/最小/最大。
258/// 任一次运行失败则整体失败（status 取最后一次失败状态，runs 仅记录成功次数）。
259BackendPerfResult
260PerformanceRacePanel::runBackendMultiple(const std::string& src, const QString& backendName, int runCount,
261 SingleRunResult (PerformanceRacePanel::*runner)(const std::string&)) {
262 BackendPerfResult result;
263 result.backendName = backendName;
264
265 if (runCount < 1) {
266 runCount = 1;
267 }
268
269 SingleRunResult last;
270 for (int i = 0; i < runCount; ++i) {
271 SingleRunResult r = (this->*runner)(src);
272 last = r;
273 // Bug #72 fix: 让出 UI 事件循环，避免长时间冻结
274 QApplication::processEvents(QEventLoop::ExcludeUserInputEvents);
275 if (r.success) {
276 result.runs.push_back(static_cast<double>(r.elapsedMs));
277 } else {
278 // 任一次失败即整体失败，提前结束
279 result.success = false;
280 result.status = r.status;
281 result.output = r.output;
282 result.instrCount = r.instrCount;
283 return result;
284 }
285 }
286
287 // 计算平均/最小/最大
288 if (!result.runs.empty()) {
289 double sum = 0.0;
290 double mn = result.runs[0];
291 double mx = result.runs[0];
292 for (double v : result.runs) {
293 sum += v;
294 mn = std::min(mn, v);
295 mx = std::max(mx, v);
296 }
297 result.avgMs = sum / static_cast<double>(result.runs.size());
298 result.minMs = mn;
299 result.maxMs = mx;
300 result.success = true;
301 result.status = QString::fromUtf8("✅ 成功");
302 result.output = last.output;
303 result.instrCount = last.instrCount;
304 } else {
305 result.success = false;
306 result.status = QString::fromUtf8("❌ 无有效运行");
307 result.output = last.output;
308 result.instrCount = last.instrCount;
309 }
310 return result;
311}

348/// 渲染柱状图：用 HTML/CSS（嵌套 table + bgcolor）绘制水平柱状图，

354

357 int successCount = 0;
358 for (int i = 0; i < 3; ++i) {
359 if (backends[i]->success && backends[i]->avgMs > maxAvg) {
360 maxAvg = backends[i]->avgMs;
361 }
362 if (backends[i]->success) {
363 ++successCount;
364 }
365 }
366
367 QString html =
368 QStringLiteral("<html><body style='font-family:\"Consolas\",\"Microsoft YaHei\",monospace;padding:4px;'>"
369 "<h3 style='margin:0 0 8px 0;color:#2C3E50;'>三后端平均耗时对比（柱状图）</h3>");
370
371 if (successCount == 0 || maxAvg <= 0.0) {
372 html += QStringLiteral("<p style='color:#C0392B;'>所有后端均失败或无有效耗时数据，无法绘制柱状图。</p>");
373 html += QStringLiteral("</body></html>");

377

379

406 "<td width='%5%' height='22'></td>"

408 "</table>"
409 "</td>"
410 "<td width='15%' align='right' style='color:%6;'>%7</td>"

414 .arg(color)

419 }
420
421 html += QStringLiteral("</table>");
422
423 // 图例
424 html += QStringLiteral("<p style='margin-top:10px;color:#6E6E6E;font-size:11px;'>"
425 "图例：<span style='color:%1;'>■</span> Interpreter "

427 "（栈式字节码） | <span style='color:%3;'>■</span> RegisterVM（寄存器式）"
428 "</p>")
429 .arg(backendColor(0))
430 .arg(backendColor(1))
431 .arg(backendColor(2));
432
433 // 运行次数提示（取三后端中最大成功运行次数，避免某个后端失败时显示 0）
434 size_t maxRuns = std::max({interp.runs.size(), stackvm.runs.size(), regvm.runs.size()});
435 html += QStringLiteral("<p style='color:#6E6E6E;font-size:11px;'>注：柱长按平均耗时相对最慢后端归一化，"
436 "每个后端运行次数 = %1，仅成功后端参与归一化。</p>")

439 html += QStringLiteral("</body></html>");

443/// 渲染性能分析报告：最快后端 / 加速比 / 性能差异原因 / 指令数对比。
444void PerformanceRacePanel::renderAnalysis(const BackendPerfResult& interp, const BackendPerfResult& stackvm,
445 const BackendPerfResult& regvm) {
446 const BackendPerfResult* backends[3] = {&interp, &stackvm, &regvm};
447
448 QString html =
449 QStringLiteral("<html><body style='font-family:\"Consolas\",\"Microsoft YaHei\",monospace;padding:4px;'>"
450 "<h3 style='margin:0 0 8px 0;color:#2C3E50;'>性能分析报告</h3>");
451
452 // 统计成功后端（fastest/slowest 由此循环得出，无需单独计数）
453 const BackendPerfResult* fastest = nullptr;
454 const BackendPerfResult* slowest = nullptr;
455 for (int i = 0; i < 3; ++i) {
456 if (!backends[i]->success)
457 continue;
458 if (!fastest || backends[i]->avgMs < fastest->avgMs)
459 fastest = backends[i];
460 if (!slowest || backends[i]->avgMs > slowest->avgMs)
461 slowest = backends[i];
462 }
463
464 // 1. 最快后端
465 html += QStringLiteral("<h4 style='color:#27AE60;margin:8px 0 4px 0;'>1. 最快后端</h4>");
466 if (fastest) {
467 html += QStringLiteral("<p>%1 以 <b>%2</b> 平均耗时胜出。</p>")
468 .arg(fastest->backendName)
469 .arg(formatMs(fastest->avgMs));
470 } else {
471 html += QStringLiteral("<p style='color:#C0392B;'>所有后端均失败，无最快后端。</p>");
472 }
473
474 // 2. 加速比
475 html += QStringLiteral("<h4 style='color:#2980B9;margin:8px 0 4px 0;'>2. 加速比</h4>");
476 if (fastest && slowest && fastest != slowest && fastest->avgMs > 0.0) {
477 double speedup = slowest->avgMs / fastest->avgMs;

548/// 渲染一致性检查：比较三后端输出是否完全一致。
549void PerformanceRacePanel::renderConsistency(const BackendPerfResult& interp, const BackendPerfResult& stackvm,
550 const BackendPerfResult& regvm) {
551 // 至少一个后端失败时给出部分一致性提示
552 bool interpOk = interp.success;
553 bool stackvmOk = stackvm.success;
554 bool regvmOk = regvm.success;
555
556 if (!interpOk || !stackvmOk || !regvmOk) {
557 QString failed;
558 if (!interpOk)
559 failed += interp.backendName + QStringLiteral(" ");
560 if (!stackvmOk)
561 failed += stackvm.backendName + QStringLiteral(" ");
562 if (!regvmOk)
563 failed += regvm.backendName + QStringLiteral(" ");
564 consistencyLabel_->setText(QString::fromUtf8("⚠ 部分后端失败：") + failed.trimmed() +
565 QStringLiteral("，无法完整验证一致性"));
566 consistencyLabel_->setStyleSheet(QStringLiteral("color:#D35400;"));
567 return;
568 }
569
570 // 三后端均成功，比较输出
571 if (interp.output == stackvm.output && stackvm.output == regvm.output) {
572 consistencyLabel_->setText(
573 QString::fromUtf8("✅ 三后端输出完全一致（语义等价性验证通过）：") +
574 (interp.output.isEmpty() ? QString::fromUtf8("（无输出）") : interp.output.trimmed().left(80)));
575 consistencyLabel_->setStyleSheet(QStringLiteral("color:#27AE60;"));
576 } else {
577 consistencyLabel_->setText(QString::fromUtf8("❌ 三后端输出不一致（可能存在语义 bug）——") +
578 QString::fromUtf8("Interpreter: 「") + interp.output.trimmed().left(40) +
579 QString::fromUtf8("」| StackVM: 「") + stackvm.output.trimmed().left(40) +
580 QString::fromUtf8("」| RegisterVM: 「") + regvm.output.trimmed().left(40) +
581 QString::fromUtf8("」"));
582 consistencyLabel_->setStyleSheet(QStringLiteral("color:#C0392B;"));
583 }
584}
