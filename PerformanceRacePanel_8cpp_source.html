200/// 开始竞赛：对三后端各运行 N 次，汇总平均/最小/最大耗时，渲染表格、
201/// 柱状图、分析报告与一致性标签。三后端在主线程串行执行。

209 }
210 std::string src = sourceEdit_->text().toStdString();
211 if (src.empty()) {

213 consistencyLabel_->setStyleSheet(
214 (QStringLiteral("color:") + TeachingTheme::dangerRed().name() + QStringLiteral(";")));
215 return;
216 }

268/// 多次运行汇总：调用指定单次执行函数 N 次，记录每次耗时，计算平均/最小/最大。
269/// 任一次运行失败则整体失败（status 取最后一次失败状态，runs 仅记录成功次数）。
270BackendPerfResult
271PerformanceRacePanel::runBackendMultiple(const std::string& src, const QString& backendName, int runCount,
272 SingleRunResult (PerformanceRacePanel::*runner)(const std::string&)) {
273 BackendPerfResult result;
274 result.backendName = backendName;
275
276 if (runCount < 1) {
277 runCount = 1;
278 }
279
280 SingleRunResult last;
281 for (int i = 0; i < runCount; ++i) {
282 SingleRunResult r = (this->*runner)(src);
283 last = r;
284 // Bug #72 fix: 让出 UI 事件循环，避免长时间冻结
285 QApplication::processEvents(QEventLoop::ExcludeUserInputEvents);
286 if (r.success) {
287 result.runs.push_back(static_cast<double>(r.elapsedMs));
288 } else {
289 // 任一次失败即整体失败，提前结束
290 result.success = false;
291 result.status = r.status;
292 result.output = r.output;
293 result.instrCount = r.instrCount;
294 return result;
295 }
296 }
297
298 // 计算平均/最小/最大
299 if (!result.runs.empty()) {
300 double sum = 0.0;
301 double mn = result.runs[0];
302 double mx = result.runs[0];
303 for (double v : result.runs) {
304 sum += v;
305 mn = std::min(mn, v);
306 mx = std::max(mx, v);
307 }
308 result.avgMs = sum / static_cast<double>(result.runs.size());
309 result.minMs = mn;
310 result.maxMs = mx;
311 result.success = true;
312 result.status = QString::fromUtf8("✅ 成功");
313 result.output = last.output;
314 result.instrCount = last.instrCount;
315 } else {
316 result.success = false;
317 result.status = QString::fromUtf8("❌ 无有效运行");
318 result.output = last.output;
319 result.instrCount = last.instrCount;
320 }
321 return result;
322}

458/// 渲染性能分析报告：最快后端 / 加速比 / 性能差异原因 / 指令数对比。
459void PerformanceRacePanel::renderAnalysis(const BackendPerfResult& interp, const BackendPerfResult& stackvm,
460 const BackendPerfResult& regvm) {
461 const BackendPerfResult* backends[3] = {&interp, &stackvm, &regvm};
462
463 QString html =
464 (QStringLiteral("<html><body style='font-family:\"Consolas\",\"Microsoft YaHei\",monospace;padding:4px;'>") +
465 QStringLiteral("<h3 style='margin:0 0 8px 0;color:") + TeachingTheme::midnight().name() +
466 QStringLiteral(";'>性能分析报告</h3>"));
467
468 // 统计成功后端（fastest/slowest 由此循环得出，无需单独计数）
469 const BackendPerfResult* fastest = nullptr;
470 const BackendPerfResult* slowest = nullptr;
471 for (int i = 0; i < 3; ++i) {
472 if (!backends[i]->success)
473 continue;
474 if (!fastest || backends[i]->avgMs < fastest->avgMs)
475 fastest = backends[i];
476 if (!slowest || backends[i]->avgMs > slowest->avgMs)
477 slowest = backends[i];

574void PerformanceRacePanel::renderConsistency(const BackendPerfResult& interp, const BackendPerfResult& stackvm,
575 const BackendPerfResult& regvm) {
576 // 至少一个后端失败时给出部分一致性提示
577 bool interpOk = interp.success;
578 bool stackvmOk = stackvm.success;
579 bool regvmOk = regvm.success;
580
581 if (!interpOk || !stackvmOk || !regvmOk) {
582 QString failed;
583 if (!interpOk)

585 if (!stackvmOk)

587 if (!regvmOk)
588 failed += regvm.backendName + QStringLiteral(" ");
589 consistencyLabel_->setText(QString::fromUtf8("⚠ 部分后端失败：") + failed.trimmed() +
590 QStringLiteral("，无法完整验证一致性"));
591 consistencyLabel_->setStyleSheet(
592 (QStringLiteral("color:") + TeachingTheme::orangeDeep().name() + QStringLiteral(";")));
593 return;
594 }
595
596 // 三后端均成功，比较输出
597 if (interp.output == stackvm.output && stackvm.output == regvm.output) {
598 consistencyLabel_->setText(
599 QString::fromUtf8("✅ 三后端输出完全一致（语义等价性验证通过）：") +
600 (interp.output.isEmpty() ? QString::fromUtf8("（无输出）") : interp.output.trimmed().left(80)));
601 consistencyLabel_->setStyleSheet(
602 (QStringLiteral("color:") + TeachingTheme::backendStackVm().name() + QStringLiteral(";")));
603 } else {
604 consistencyLabel_->setText(QString::fromUtf8("❌ 三后端输出不一致（可能存在语义 bug）——") +
605 QString::fromUtf8("Interpreter: 「") + interp.output.trimmed().left(40) +
606 QString::fromUtf8("」| StackVM: 「") + stackvm.output.trimmed().left(40) +
607 QString::fromUtf8("」| RegisterVM: 「") + regvm.output.trimmed().left(40) +
608 QString::fromUtf8("」"));
609 consistencyLabel_->setStyleSheet(
610 (QStringLiteral("color:") + TeachingTheme::dangerRed().name() + QStringLiteral(";")));
