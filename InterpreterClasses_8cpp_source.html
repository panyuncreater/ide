15void Interpreter::visitClassDecl(ClassDecl& node) {
16 checkBreak(&node);
17
18 ClassInfo cls;
19 cls.name = node.name;
20 cls.superClassName = node.superClassName;
21 cls.closureEnv = currentEnv_; // O5: 捕获类定义时的环境（闭包）
22 // R163 泛型扩展：记录类的类型参数，供 invokeMethod 合并类泛型 + 方法泛型参数
23 cls.typeParams = node.typeParams;
24 // 不再存储 superClass 裸指针，运行时通过 superClassName 查找
25
26 // 如果有父类，验证父类是否已定义
27 if (!node.superClassName.empty()) {
28 // R97 #10 fix: 用 lookupClassSafely 替代 find + end() + runtimeError 三步模式
29 // （引用必须保留以触发查找副作用，否则编译器可能优化掉导致父类未定义检查被跳过）
30 (void)lookupClassSafely(node.superClassName, "未定义的父类: " + node.superClassName, node.line, node.column);
31 }
32
33 // 注册类名到环境
34 Value classVal(std::string("class:") + node.name);
35 currentEnv_->define(node.name, classVal);
36
37 // 先注册类信息（以便方法中可以递归引用自身）
38 classRegistry_[node.name] = cls;
39 // C6 fix: 任何类定义/重定义都递增 gen，使所有 ClassInfo::methodCache_ 条目失效，
40 // 防止子类缓存指向已被替换的父类旧方法指针。
41 classRegistryGen_++;
42
43 // 处理类成员
44 for (auto& member : node.members) {
45 // VarDecl: 字段默认值
46 if (member->nodeType == NodeType::NODE_VAR_DECL) {
47 VarDecl* varDecl = static_cast<VarDecl*>(member.get());
48 Value defaultVal = Value::nullValue();
49 if (varDecl->initializer) {
50 defaultVal = evaluate(varDecl->initializer.get());
51 }
52 classRegistry_[node.name].fields[varDecl->name] = defaultVal; // #2 fix: 直接索引，避免registeredCls失效
53 continue;
54 }
55
56 // FunDecl: 方法
57 if (member->nodeType == NodeType::NODE_FUN_DECL) {
58 // A3 fix: 使用 static_pointer_cast 共享 AST 节点所有权
59 auto funDecl = std::static_pointer_cast<FunDecl>(member);
60 classRegistry_[node.name].methods[funDecl->name] = funDecl; // #2 fix
61 continue;
62 }
63 }
64
65 lastValue_ = std::move(classVal);
66 return;
67}

69void Interpreter::visitMemberAccess(MemberAccess& node) {
70 checkBreak(&node);
71
72 Value obj = evaluate(node.object.get());
73
74 // P0-3 fix: 使用 const 引用避免在只读访问时触发 COW 深拷贝
75 const Value& objC = obj;
76
77 // 类实例的成员访问
78 if (objC.isInstance()) {
79 // O1: super.field — 字段查找不变（实例已含继承字段），方法查找从父类开始
80 bool isSuperAccess = (node.object && node.object->nodeType == NodeType::NODE_SUPER_EXPR);
81
82 const auto& flds = objC.fields();
83 auto it = flds.find(node.fieldName);
84 if (it != flds.end()) {
85 lastValue_ = it->second;
86 return;
87 }
88
89 // 检查是否访问的是方法（返回一个标记值）
90 auto classIt = classRegistry_.find(objC.className());
91 if (classIt != classRegistry_.end()) {
92 ClassInfo* searchClass = &classIt->second;
93 if (isSuperAccess) {
94 // AUDIT-R5 R4 fix: 以词法类（classContextStack_ 顶部 = 当前执行方法的
95 // 定义类）而非运行时实例类解析 super 搜索起点，对齐 visitMethodCall
96 // 的 isSuperCall 路径（L3344-3351）与 VM 的 OP_SUPER_CALL（编译期编码
97 // currentClassName_）。原实现用 objC.className()：A←B←C 且实例为 C 时，
98 // B 方法内的 super 成员访问从 C 的父类 B 开始搜索，可能重新命中 B
99 // 自身方法造成错误分派甚至无限递归。
100 const std::string& lexicalClassName =
101 !classContextStack_.empty() ? classContextStack_.back() : objC.className();
102 auto lexIt = classRegistry_.find(lexicalClassName);
103 ClassInfo* lexClass = (lexIt != classRegistry_.end()) ? &lexIt->second : &classIt->second;
104 if (lexClass->superClassName.empty()) {
105 runtimeError("类 " + lexClass->name + " 没有父类，不能使用 super", node.line, node.column);
106 }
107 auto superIt = classRegistry_.find(lexClass->superClassName);
108 if (superIt != classRegistry_.end()) {
109 searchClass = &superIt->second;
110 }
111 }
112 FunDecl* method = findMethod(*searchClass, node.fieldName);
113 if (method) {
114 // 方法作为字段访问，返回特殊标记
115 // PERF-05 fix: 驻留 "method:Class.field" 字符串，避免每次成员访问重复构造
116 Value methodVal(StringIntern::internConcat(StringIntern::internConcat("method:", objC.className()),
117 std::string(".") + node.fieldName));
118 lastValue_ = std::move(methodVal);
119 return;
120 }
121 }
122
123 // 字段和方法都不存在，报告错误
124 runtimeError("类 " + objC.className() + " 没有字段或方法 '" + node.fieldName + "'", node.line, node.column);
125 }
126
127 // 字典的成员访问（同索引访问，键为 string 类型字段名）
128 if (objC.isDict()) {
129 const auto& dict = objC.dictVal();
130 auto it = dict.find(Value::DictKey{node.fieldName});
131 if (it != dict.end()) {
132 lastValue_ = it->second;
133 return;
134 }
135 lastValue_ = Value::nullValue();
136 return;
137 }
138
139 // P2 fix (null-access): null 值成员访问给出明确的 null-access 诊断码，
140 // 供 ErrorHintEngine 按 code 精确匹配教学提示（而非依赖子串匹配）。
141 if (objC.isNull()) {
142 runtimeError("不能在 null 值上访问属性或调用方法", node.line, node.column, DiagCodes::kNullAccess);
143 }
144 runtimeError(ErrorMessages::kTypeNotMemberAccessible, node.line, node.column, DiagCodes::kTypeMismatch);
145}

179// P1-3 fix: 继承链查找模板，消除 findMethod/findFieldDefault 的重复遍历结构。
180// 沿 superClassName 上移，MAX_INHERITANCE_DEPTH 防止循环继承。
181template <typename Lookup, typename NotFound>
182auto lookupInheritanceChain(const ClassInfo& cls, const std::unordered_map<std::string, ClassInfo>& registry,
183 Lookup lookup, NotFound notFound) -> decltype(lookup(cls)) {
184 const ClassInfo* cur = &cls;
185 int depth = 0;
186 while (cur) {
187 if (++depth > RuntimeLimits::MAX_INHERITANCE_DEPTH)
188 return notFound;
189 if (auto r = lookup(*cur))
190 return r;
191 if (!cur->superClassName.empty()) {
192 auto superIt = registry.find(cur->superClassName);
193 cur = (superIt != registry.end()) ? &superIt->second : nullptr;
194 } else {
195 cur = nullptr;
196 }
197 }
198 return notFound;
199}

201FunDecl* Interpreter::findMethod(const ClassInfo& cls, const std::string& methodName) {
202 // C6 fix: 方法分派缓存。先查缓存（O(1)），gen 不匹配或未命中才走继承链（O(depth)）。
203 auto cacheIt = cls.methodCache_.find(methodName);
204 if (cacheIt != cls.methodCache_.end() && cacheIt->second.second == classRegistryGen_) {
205 return cacheIt->second.first.get(); // A3 fix: cache 持有 shared_ptr，返回裸指针
206 }
207 // P2-7 fix: const 正确性 — 不修改 cls，使用 const 指针遍历继承链
208 // P1-3 fix: 委托给 lookupInheritanceChain 模板
209 // A3 fix: methods 表持有 shared_ptr<FunDecl>，lambda 返回 shared_ptr
210 std::shared_ptr<FunDecl> result = lookupInheritanceChain(
211 cls, classRegistry_,
212 [&methodName](const ClassInfo& c) -> std::shared_ptr<FunDecl> {
213 auto it = c.methods.find(methodName);
214 return (it != c.methods.end()) ? it->second : nullptr;
215 },
216 nullptr);
217 // 写入缓存（记录当前 gen，类重定义时 gen 递增使此条目失效）
218 cls.methodCache_[methodName] = {result, classRegistryGen_};
219 return result.get(); // 返回裸指针，调用方在 ClassInfo 存活期间安全使用
220}

222std::string Interpreter::findMethodDefiningClassName(const ClassInfo& startCls, const std::string& methodName) {
223 // AUDIT-P1-CORRECT fix: 查找方法实际定义所在的类名，用于 classContextStack_ 压入。
224 // 原实现压入 searchClass->name（搜索起始类），当中间类未定义方法时，
225 // findMethod 沿继承链向上找到祖先类的方法，但栈中压入的是中间类名，
226 // 导致后续 super 调用从错误的类开始搜索，可能找到同一个方法形成无限递归。
227 // 本方法不走缓存（缓存只存方法指针不存定义类），直接遍历继承链。
228 // 与 findMethod 的遍历逻辑完全一致，仅额外返回 ClassInfo::name。
229 const ClassInfo* cur = &startCls;
230 int depth = 0;
231 while (cur) {
232 if (++depth > RuntimeLimits::MAX_INHERITANCE_DEPTH)
233 break;
234 if (cur->methods.find(methodName) != cur->methods.end()) {
235 return cur->name;
236 }
237 if (!cur->superClassName.empty()) {
238 auto superIt = classRegistry_.find(cur->superClassName);
239 cur = (superIt != classRegistry_.end()) ? &superIt->second : nullptr;
240 } else {
241 cur = nullptr;
242 }
243 }
244 return startCls.name; // fallback（方法未找到时不应到达）
245}
