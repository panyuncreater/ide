2// AstToyLevels.cpp — AST 搭建玩具题目静态数据实现（功能 4）
3// ------------------------------------------------------------
4// 6 道题由浅入深，覆盖叶子节点 → 二元运算 → 优先级 → 括号改变结构 →
5// 函数调用 → 完整语句。核心教学时刻为第 3 题 vs 第 4 题：
6// - "1 + 2 * 3" → 根是 +，右子树是 *
7// - "(1 + 2) * 3" → 根是 *，左子树是 +
8// 同样的数字和运算符，仅括号不同就导致完全不同的树结构。
9//
10// 教学理念：抽象语法树（AST）是"代码的结构化骨架"。源码是一维的
11// 文本，AST 是二维的树——运算符当根、操作数当树枝。搭建 AST 的过程，
12// 就是让学习者亲手"看见"优先级与括号如何变成树的形状。

144const AstToyLevel* AstToyLibrary::findByLevel(int level) {
145 const auto& lvls = levels();
146 for (const auto& l : lvls) {
147 if (l.level == level)
148 return &l;
149 }
150 return nullptr;
151}
