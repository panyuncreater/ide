23static std::string escapeStringContent(const std::string& s) {
24 std::string escaped;
25 escaped.reserve(s.size());
26 for (size_t i = 0; i < s.size(); ++i) {
27 unsigned char c = static_cast<unsigned char>(s[i]);
28 switch (c) {
29 case '\\':
30 escaped += "\\\\";
31 break;
32 case '"':
33 escaped += "\\\"";
34 break;
35 case '\n':
36 escaped += "\\n";
37 break;
38 case '\t':
39 escaped += "\\t";
40 break;
41 case '\r':
42 escaped += "\\r";
43 break;
44 case '\0':
45 escaped += "\\0";
46 break;
47 case '\b':
48 escaped += "\\b";
49 break;
50 case '\f':
51 escaped += "\\f";
52 break;
53 case '\a':
54 escaped += "\\a";
55 break;
56 case '\v':
57 escaped += "\\v";
58 break;
59 case '{':
60 // AUDIT-R3 P1 fix: 字面 '{' 必须转义——Lexer 在字符串内将裸 '{' 无条件
61 // 视为插值起始（string() → handleInterpolation），且不支持 \\{ 转义。
62 // 若原样输出，含 \\x7b/\\u{7b} 转义的字符串经格式化后重新解析会报
63 // "未终止的插值表达式"或语义变为插值，破坏往返等价性。
64 // 输出为 \\x7b（Lexer handleHexEscape 可回读）。
65 escaped += "\\x7b";
66 break;
67 default:
68 // AUDIT-P3.12 fix: 处理多字节 UTF-8 字符（非 ASCII）
69 if (c >= 0x80) {
70 uint32_t codepoint = 0;
71 int extraBytes = 0;
72 if ((c & 0xE0) == 0xC0) {
73 codepoint = c & 0x1F;
74 extraBytes = 1;
75 } else if ((c & 0xF0) == 0xE0) {
76 codepoint = c & 0x0F;
77 extraBytes = 2;
78 } else if ((c & 0xF8) == 0xF0) {
79 codepoint = c & 0x07;
80 extraBytes = 3;
81 } else {
82 // 无效 UTF-8 首字节，按单字节输出
83 escaped += static_cast<char>(c);
84 continue;
85 }
86 bool valid = true;
87 for (int j = 0; j < extraBytes; ++j) {
88 if (i + 1 + j >= s.size()) {
89 valid = false;
90 break;
91 }
92 unsigned char next = static_cast<unsigned char>(s[i + 1 + j]);
93 if ((next & 0xC0) != 0x80) {
94 valid = false;
95 break;
96 }
97 codepoint = (codepoint << 6) | (next & 0x3F);
98 }
99 if (valid) {
100 if (codepoint > 0xFFFF) {
101 // AUDIT-P3.12 fix: 非 BMP 字符输出 \u{XXXXXX}
102 char buf[12];
103 std::snprintf(buf, sizeof(buf), "\\u{%x}", codepoint);
104 escaped += buf;
105 } else {
106 // BMP 字符保持原字节输出（含中文等 3 字节 UTF-8）
107 escaped += s.substr(i, static_cast<size_t>(1 + extraBytes));
108 }
109 i += static_cast<size_t>(extraBytes); // 跳过多字节序列的剩余字节
110 } else {
111 // 不完整 UTF-8 序列，按单字节输出
112 escaped += static_cast<char>(c);
113 }
114 continue; // 已处理，跳过下方控制字符检查
115 }
116 // AUDIT-P3.11 fix: 转义所有未显式处理的 C0 控制字符和 DEL
117 if (c < 0x20 || c == 0x7F) {
118 char buf[5];
119 std::snprintf(buf, sizeof(buf), "\\x%02x", c);
120 escaped += buf;
121 } else {
122 escaped += static_cast<char>(c);
123 }
124 break;
125 }
126 }
127 return escaped;
128}

228/// 判断节点类型是否是自终止的复合语句（以 } 结尾，不需要额外 ;）
229static bool isSelfTerminating(ASTNode* node) {
230 if (!node)
231 return false;
232 switch (node->nodeType) {
233 case NodeType::NODE_IF_STMT:
234 case NodeType::NODE_WHILE_STMT:
235 case NodeType::NODE_FOR_STMT:
236 case NodeType::NODE_FUN_DECL:
237 case NodeType::NODE_CLASS_DECL:
238 case NodeType::NODE_BLOCK:
239 case NodeType::NODE_DESTRUCTURE_BINDING: // P1 #26 fix: 解构绑定自带 ';'，自终止，避免 formatBlock 双重分号
240 case NodeType::NODE_ENUM_DECL: // P1 #29 fix: enum 以 '}' 结尾，自终止
241 case NodeType::NODE_MACRO_DECL: // 七特性 MVP 阶段 2: 宏声明以 '}' 结尾，自终止
242 case NodeType::NODE_TRAIT_DECL: // 七特性 MVP 阶段 3: trait 声明以 '}' 结尾，自终止
243 case NodeType::NODE_TRY_STMT: // P0 fix: try/catch 以 } 结尾，自终止
244 return true;
245 case NodeType::NODE_IMPORT_STMT: // P0 fix: visitImportStmt 已自行添加 ;
246 return true;

284 // P2 #61 fix: 返回注释形式而非裸 "null"，避免与用户代码中的 null 字面量混淆

291 formatDepth_++;

293 int& d;

676/// 判断子表达式是否需要加括号
677static bool needsParens(ASTNode* child, BinOpType parentOpType, bool isRight) {
678 if (!child || child->nodeType != NodeType::NODE_BINARY_OP)
679 return false;
680 BinaryOp* childBin = static_cast<BinaryOp*>(child);
681 int parentPrec = Formatter::opPrecedence(parentOpType);
682 int childPrec = Formatter::opPrecedence(childBin->opType);
683 // 子优先级更低 → 需要括号
684 if (childPrec < parentPrec)
685 return true;
686 // 同优先级时，右结合运算符的右操作数不需要括号，左操作数也不需要
687 // 但对于左结合运算符的右操作数，如果子也是同优先级，需要括号（如 a - (b - c)）
688 if (childPrec == parentPrec) {
689 if (Formatter::isRightAssoc(parentOpType)) {
690 return !isRight; // 右结合：左操作数需要括号
691 } else {
692 // 左结合：右操作数同优先级子表达式必须加括号以保持原始分组。
693 // AUDIT-FMT-P0 fix: 原实现仅对 ADD/SUB/MUL/DIV/MOD 加括号（#15 fix），
694 // EQ/NEQ/LT/GT/LTE/GTE/AND/OR 漏处理，导致 1 == (2 == 3) 被格式化为 1 == 2 == 3，
695 // 重新解析得到 (1 == 2) == 3，AST 结构改变，往返不变量被破坏。
696 // 修复方案：移除运算符特判，对所有左结合运算符的右操作数同优先级子表达式统一加括号。
697 // 理由：
698 // - 浮点算术不满足结合律：(a+b)+c ≠ a+(b+c)（舍入误差）
699 // - SUB/DIV/MOD 本就不满足结合律
700 // - EQ/NEQ/LT/GT/LTE/GTE 返回 bool，重新分组会改变比较语义
701 // （如 (1 < 2) < 3 → true < 3 → true；1 < (2 < 3) → 1 < true → 1 < 1 → false）
702 // - AND/OR 虽短-circuit 行为在多数情况下等价，但 AST 分组改变违反保形变换原则
703 // - 左操作数同优先级无需括号（左结合天然保持分组）
704 if (isRight) {
705 return true;
706 }
707 return false;
708 }
709 }
710 return false;
711}

713/// 格式化二元运算：递归格式化左右操作数，依据 needsParens 在子表达式优先级不足时补加括号，
714/// 保证重新解析后 AST 结构与原树等价（往返不变量）。
715std::string Formatter::formatBinaryOp(BinaryOp& node) {

717 if (needsParens(node.left.get(), node.opType, false)) {
718 left = "(" + left + ")";
719 }

721 if (needsParens(node.right.get(), node.opType, true)) {
722 right = "(" + right + ")";
723 }
724 return left + binOp(BinaryOp::opTypeStr(node.opType)) + right;
