444/// 抓取一次 VM 单步快照（轨迹单步/断点的数据来源）：
445/// 从未初始化的 VM 直接返回占位；否则从 controller 取当前 IP、OpCode 名、
446/// 栈帧数、源码行号，并通过 getVmStack() 取得操作数栈快照（每个 Value 经
447/// toString 序列化，堆类型用 try/catch 包裹防异常）。快照追加到 traceHistory_
448/// （超 kMaxTraceEntries 则丢弃最旧），刷新表格并自动选中最新行。
449/// 该轨迹即字节码单步/断点的可视化数据源，每行对应一条指令执行后的完整状态。
450void BytecodeTracePanel::captureCurrentState() {
451 if (!controller_) {
452 liveStatusLabel_->setText(tr("状态：未绑定 controller"));
453 return;
454 }
455 if (!controller_->isVmInitialized()) {
456 liveStatusLabel_->setText(tr("状态：VM 未初始化（启动 VM 单步以捕获轨迹）"));
457 return;
458 }
459 // AUDIT-P2 fix: 对齐 CallStackPanel/VariableInspectorPanel/BreakpointConditionPanel/
460 // MemoryModelPanel 的 isVmRunning() 守卫。VM RUN 批之间捕获轨迹会混入用户
461 // 未主动请求的中间状态数据，且未来若 VmStepper 改为多线程会升级为数据竞争。
462 if (controller_->isVmRunning()) {
463 liveStatusLabel_->setText(tr("状态：VM 运行中（暂停后可捕获）"));
464 return;
465 }
466
467 TraceEntry e;
468 // OPT-2 fix: step 延迟到指纹检查通过后再赋值，避免状态未变时 stepCounter_
469 // 误增导致后续捕获步号跳号（autoTimer 2s 安全网可能在 VM 状态未变时触发）。
470 e.ip = controller_->getVmCurrentIP();
471 e.opCodeName = controller_->getVmCurrentOpCodeName();
472 e.frameCount = static_cast<int>(controller_->getVmFrameCount());
473 e.line = controller_->getVmCurrentLine();
474
475 auto stack = controller_->getVmStack();
476 e.stackSnapshot.reserve(stack.size());
477 // AUDIT-P2 fix: Value::toString 对堆类型（数组/字典/实例）走 toStringImpl，
478 // 极端情况（循环引用、深度嵌套、bad_alloc）可能抛异常。对齐 CallStackPanel/
479 // VariableInspectorPanel 的 try/catch 防护，避免异常传播到 QTimer 槽。
480 for (const auto& v : stack) {
481 std::string s;
482 try {
483 s = v.toString();
484 } catch (...) {
485 s = "<error>";
486 }
487 e.stackSnapshot.push_back(std::move(s));
488 }
489
490 // OPT-2 fix: autoTimer 2s 安全网无指纹对比会盲目累积重复轨迹（累积式 push_back）。
491 // 在 e 各字段已填充后计算指纹，与上次比对——状态未变则跳过本次捕获。
492 // 指纹不含 stackSnapshot 内容（避免长栈 O(n) 字符串拼接），仅用大小即可
493 // 区分绝大多数"状态未变"场景；如栈大小相同但内容变化，会多捕获一条，
494 // 不影响正确性，仅极小性能开销。
495 std::string fingerprint = std::to_string(e.ip) + "|" + e.opCodeName + "|" + std::to_string(e.frameCount) + "|" +
496 std::to_string(e.stackSnapshot.size());
497 if (fingerprint == lastCaptureFingerprint_) {
498 // 状态未变，跳过本次捕获（不更新 stepCounter_，不追加历史，不刷新表格）
499 return;
500 }
501 lastCaptureFingerprint_ = std::move(fingerprint);
502
503 // 指纹检查通过，确认本次捕获有效，递增步号
504 e.step = ++stepCounter_;
505
506 liveStatusLabel_->setText(tr("状态：已捕获 step=%1 | IP=%2 | OpCode=%3 | 帧数=%4")
507 .arg(e.step)
508 .arg(e.ip)
509 .arg(QString::fromUtf8(e.opCodeName.c_str()))
510 .arg(e.frameCount));
511
512 if (static_cast<int>(traceHistory_.size()) >= kMaxTraceEntries) {
513 // AUDIT-P1 fix: deque pop_front O(1)（原 vector erase(begin()) O(n) 移位）
514 traceHistory_.pop_front();
515 // R51-2 fix: 标记 deque 已滑动，refreshTraceTable 需全量重建
516 dequeShifted_ = true;
517 }
518 traceHistory_.push_back(std::move(e));
519 refreshTraceTable();
520
521 // 自动选中最新条目
522 int lastRow = traceTable_->rowCount() - 1;
523 if (lastRow >= 0) {
524 traceTable_->selectRow(lastRow);
525 onTraceRowSelected();
526 }
527}
528
529/// 将 traceHistory_ 渲染为轨迹表。AUDIT-P1 fix: 改为增量更新——
530/// 仅追加新行 + 移除溢出行，避免每次 capture 都 setRowCount(0) 全量重建 N×5 个 item。
531/// R51-2 fix: deque 容量满后 pop_front+push_back 保持 size 不变但索引偏移，
532/// 增量更新无法感知，需检测 dequeShifted_ 标志后全量重建。
533void BytecodeTracePanel::refreshTraceTable() {
534 int histSize = static_cast<int>(traceHistory_.size());
535
536 // R51-2 fix: deque 滑动后索引整体偏移，必须全量重建
537 if (dequeShifted_) {
538 traceTable_->setRowCount(0);
539 dequeShifted_ = false;
540 }
541
542 int tableRows = traceTable_->rowCount();
543
544 // 表行多于历史（清空场景或溢出移除）→ 移除多余行
545 while (tableRows > histSize) {
546 traceTable_->removeRow(tableRows - 1);
547 --tableRows;
548 }
549
550 // 追加新行（仅新增部分）
551 for (int i = tableRows; i < histSize; ++i) {
552 traceTable_->insertRow(i);
553 }
554
555 // 填充新行的单元格数据（仅新行，旧行数据不变）
556 for (int i = tableRows; i < histSize; ++i) {
557 const auto& e = traceHistory_[i];
558 auto* c0 = new QTableWidgetItem(QString::number(e.step));
559 auto* c1 = new QTableWidgetItem(QString::number(e.ip));
560 auto* c2 = new QTableWidgetItem(QString::fromUtf8(e.opCodeName.c_str()));
561 auto* c3 = new QTableWidgetItem(QString::number(e.frameCount));
562 auto* c4 = new QTableWidgetItem(QString::number(e.stackSnapshot.size()));
563 c0->setTextAlignment(Qt::AlignCenter);
564 c1->setTextAlignment(Qt::AlignCenter);
565 c3->setTextAlignment(Qt::AlignCenter);
566 c4->setTextAlignment(Qt::AlignCenter);
567 traceTable_->setItem(i, 0, c0);
568 traceTable_->setItem(i, 1, c1);
569 traceTable_->setItem(i, 2, c2);
570 traceTable_->setItem(i, 3, c3);
571 traceTable_->setItem(i, 4, c4);
572 }
573 traceTable_->scrollToBottom();
574}
