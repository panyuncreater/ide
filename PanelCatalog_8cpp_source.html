102/// 按面板 id 在分类树中查找对应条目；未找到返回 nullptr。
103const PanelEntry* PanelCatalog::findById(const std::string& id) {
104 for (const auto& cat : categories()) {
105 for (const auto& leaf : cat.leaves) {
106 if (id == leaf.id) {
107 return &leaf;
108 }
109 }
110 }
111 return nullptr;
112}

114// ============================================================
115// 活动 id → 面板 id 的别名映射
116// ------------------------------------------------------------
117// 处理四类别名：
118// 1. lab-* → lab-manual（实验手册章节）
119// 2. bug-hunt-* → bug-hunt（Bug 狩猎难度变体）
120// 3. op-priority-challenge → ast-toy（运算符优先级挑战路由到 AST 玩具）
121// 4. visited-* → 浏览型面板进入即完成活动（P1-F2/F）
122// 有效面板 id 原样返回。welcome / freeform-project 等需调用方特殊处理的 id
123// 返回空字符串。
124// ============================================================
125static bool startsWith(const std::string& s, const std::string& prefix) {
126 return s.size() >= prefix.size() && s.compare(0, prefix.size(), prefix) == 0;
127}

129/// 将学习活动 id 规范化为其对应的面板 id。
130std::string PanelCatalog::canonicalPanelId(const std::string& activityId) {
131 // 别名：lab-01 ~ lab-99 → lab-manual
132 if (startsWith(activityId, "lab-") && activityId != "lab-manual") {
133 return "lab-manual";
134 }
135 // 别名：bug-hunt-beginner / intermediate / expert → bug-hunt
136 if (startsWith(activityId, "bug-hunt-")) {
137 return "bug-hunt";
138 }
139 // 别名：运算符优先级挑战 → AST 玩具
140 if (activityId == "op-priority-challenge") {
141 return "ast-toy";
142 }
143 // P1-F2/F fix: 浏览型面板的"进入即完成"活动别名
144 // 学习路径中的 visited-* 活动对应一个真实面板，需能被 canonicalPanelId 识别。
145 if (activityId == "visited-glossary")
146 return "glossary";
147 if (activityId == "visited-pipeline")
148 return "pipeline";
149 if (activityId == "visited-memory-model")
150 return "memory-model";
151 if (activityId == "visited-bytecode-trace")
152 return "bytecode-trace";
153 if (activityId == "visited-exception-flow")
154 return "exception-flow";
155 if (activityId == "visited-closure-inspector")
156 return "closure-inspector";
157 // 校验是否为已注册面板 id
158 if (findById(activityId) != nullptr) {
159 return activityId;
160 }
161 // welcome / freeform-project / 未知 id：返回空，调用方特殊处理
162 return std::string();
163}
