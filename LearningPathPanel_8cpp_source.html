167/// 重新读取进度并刷新整个学习路径视图。
168void LearningPathPanel::refresh() {
169 // M9: 清空键盘导航状态（旧 row 即将被销毁，引用不再有效）
170 activityRows_.clear();
171 currentNavIndex_ = -1;
172 highlightedSavedStyle_.clear();
173
174 // 清空旧的阶段卡片（保留末尾的 stretch）
175 // 注：必须用 deleteLater() 而非 delete，因为 refresh() 可能从
176 // row->clicked 信号槽调用链中触发（onActivityClicked→onActivityRequested
177 // →showTeachingPanel→markActivityCompleted→refresh），立即删除信号发送者
178 // 会在 Qt 信号分发期间引发 use-after-free。旧 widget 从 layout 移除后
179 // 不可见，deleteLater 在事件循环返回时安全清理。
180 while (stagesLayout_->count() > 1) {
181 QLayoutItem* item = stagesLayout_->takeAt(0);
182 if (item->widget()) {
183 item->widget()->deleteLater();
184 }
185 delete item;
186 }
187
188 auto& store = LearnerProgressStore::instance();
189 const auto& all = allActivities();
190
191 // 顶部总进度
192 int overall = store.overallProgress(all);
193 overallProgress_->setValue(overall);
194 overallLabel_->setText(mlTr("总进度: %1%").arg(overall));
195
196 // P1-2 fix (F8): 更新当前阶段提示 label
197 int curStage = store.data().currentStage;
198 if (stageLabel_) {
199 if (curStage >= LearningPathData::stageCount()) {
200 // 全部通关
201 stageLabel_->setText(QString::fromUtf8("🎉 %1").arg(mlTr("已通关所有阶段")));
202 stageLabel_->setStyleSheet(
203 QString::fromUtf8("QLabel { padding: 2px 8px; border-radius: 4px;"
204 " background: %1; color: white; font-weight: bold; }")
205 .arg(TeachingTheme::learningStageColor(LearningPathData::stageCount() - 1).name()));
206 } else {
207 // 显示当前阶段标题（标题已含「阶段X：标题」格式）
208 stageLabel_->setText(QString::fromUtf8("📍 %1").arg(stageTitle(curStage)));
209 stageLabel_->setStyleSheet(QString::fromUtf8("QLabel { padding: 2px 8px; border-radius: 4px;"
210 " background: %1; color: white; font-weight: bold; }")
211 .arg(TeachingTheme::learningStageColor(curStage).name()));
212 }
213 }
214
215 // P2-3 fix (F9): 更新薄弱点提示
216 if (weakPointsLabel_) {
217 auto weakPoints = store.getWeakPoints(all);
218 if (weakPoints.empty()) {
219 weakPointsLabel_->hide();
220 } else {
221 // 取第 1 个薄弱点活动标题展示（点击可跳转）
222 const LearningActivity* wp = nullptr;
223 for (const auto& a : all) {
224 if (a.id == weakPoints.front().activityId) {
225 wp = &a;
226 break;
227 }
228 }
229 QString wpText;
230 if (wp) {
231 wpText = mlTr("⚠ 薄弱点: %1 (失败 %2 次)")
232 .arg(QString::fromStdString(wp->title))
233 .arg(weakPoints.front().fails);
234 } else {
235 wpText = mlTr("⚠ 薄弱点: %1 个").arg(weakPoints.size());
236 }
237 // 若有多个薄弱点，tooltip 列出全部
238 if (weakPoints.size() > 1) {
239 QStringList detailLines;
240 for (const auto& wp2 : weakPoints) {
241 const LearningActivity* a2 = nullptr;
242 for (const auto& a : all) {
243 if (a.id == wp2.activityId) {
244 a2 = &a;
245 break;
246 }
247 }
248 QString title = a2 ? QString::fromStdString(a2->title) : QString::fromStdString(wp2.activityId);
249 detailLines
250 << QString::fromUtf8("%1 (尝试 %2 / 失败 %3)").arg(title).arg(wp2.attempts).arg(wp2.fails);
251 }
252 weakPointsLabel_->setToolTip(detailLines.join(QStringLiteral("\n")));
253 } else {
254 weakPointsLabel_->setToolTip(QString());
255 }
256 weakPointsLabel_->setText(wpText);
257 weakPointsLabel_->show();
258 }
259 }
260
261 // P2-3 fix (F9): 更新预计剩余时间
262 if (remainingLabel_) {
263 int remaining = store.estimatedRemainingMinutes(all);
264 int spent = store.totalSpentMinutes();
265 QString remText = mlTr("⏱ 剩余 ~%1 分钟").arg(remaining);
266 if (spent > 0) {
267 remText += mlTr(" · 已用 %1 分钟").arg(spent);
268 }
269 remainingLabel_->setText(remText);
270 remainingLabel_->setToolTip(mlTr("剩余 = 所有未完成且已解锁活动的预计耗时之和"));
271 remainingLabel_->show();
272 }
273
274 // 阶段卡片
275 for (int stage = 0; stage < LearningPathData::stageCount(); ++stage) {
276 auto* card = buildStageCard(stage);
277 if (card) {
278 stagesLayout_->insertWidget(stagesLayout_->count() - 1, card);
279 }
280 }
281
282 // 不再调用 PanelAnimator::fadeInWidget(scrollContent_)：
283 // QGraphicsOpacityEffect 会对含 100+ 子 widget 的 scrollContent_ 做
284 // 离屏 pixmap 合成（每帧 O(n) 复杂度），是章节切换卡顿与偶发崩溃的根因。
285 // 崩溃机制：deleteLater 延迟删除旧 widget 时，effect 的 pixmap 缓存仍
286 // 引用已销毁子 widget → QPainter use-after-free。
287 // 页面级过渡动画改由 Ide::showTeachingPanel 中的轻量 pos 滑入实现。
288}

293QFrame* LearningPathPanel::buildStageCard(int stage) {
294 auto* frame = new QFrame(scrollContent_);
295 frame->setFrameShape(QFrame::StyledPanel);
296 QString color = stageColor(stage);
297
298 frame->setStyleSheet(
299 QString::fromUtf8("QFrame#stageCard { border: 1px solid %1; border-radius: 6px; }").arg(color));
300 frame->setObjectName(QString::fromUtf8("stageCard"));
301
302 auto* layout = new QVBoxLayout(frame);
303 layout->setContentsMargins(8, 6, 8, 6);
304 layout->setSpacing(4);
305
306 // 阶段标题 + 阶段进度条
307 auto* header = new QHBoxLayout;
308 QString title = stageTitle(stage);
309 auto* titleLabel = new QLabel(QString::fromUtf8("<b style='color:%1;'>%2</b>").arg(color).arg(title), frame);
310
311 auto& store = LearnerProgressStore::instance();
312 const auto& all = allActivities();
313 int sp = store.stageProgress(stage, all);
314
315 auto* stageProgress = new QProgressBar(frame);
316 stageProgress->setRange(0, 100);
317 stageProgress->setValue(sp);
318 stageProgress->setTextVisible(true);
319 stageProgress->setFormat(QString::fromUtf8("%p%"));
320 stageProgress->setFixedWidth(120);
321 stageProgress->setFixedHeight(16);
322
323 header->addWidget(titleLabel, 1);
324 header->addWidget(stageProgress, 0);
325 layout->addLayout(header);
326
327 // 该阶段下所有活动项
328 std::string recommendedId = store.nextRecommended(all);
329 auto stageActivities = LearningPathData::byStage(stage);
330 // 搜索过滤：按 id/title/description 大小写不敏感包含匹配
331 QString filter = searchEdit_ ? searchEdit_->text().trimmed() : QString();
332 for (const auto* act : stageActivities) {
333 if (!filter.isEmpty()) {
334 bool matches = QString::fromStdString(act->id).contains(filter, Qt::CaseInsensitive) ||
335 QString::fromStdString(act->title).contains(filter, Qt::CaseInsensitive) ||
336 QString::fromStdString(act->description).contains(filter, Qt::CaseInsensitive);
337 if (!matches) {
338 continue; // 跳过不匹配的活动
339 }
340 }
341
342 bool unlocked = store.isUnlocked(act->id, all);
343 bool completed = false;
344 auto cit = store.data().completed.find(act->id);
345 if (cit != store.data().completed.end() && cit->second)
346 completed = true;
347 bool recommended = (act->id == recommendedId);
348
349 auto* row = buildActivityRow(*act, unlocked, completed, recommended);
350 layout->addWidget(row);
351 }
352
353 return frame;
354}

359/// 构建单个活动的可点击行控件。
360QWidget* LearningPathPanel::buildActivityRow(const LearningActivity& activity, bool unlocked, bool completed,
361 bool recommended) {
362 // 使用 QPushButton 作为行容器，flat 模式去除按钮样式
363 auto* row = new QPushButton(scrollContent_);
364 row->setFlat(true);
365 row->setCheckable(false);
366 row->setFocusPolicy(Qt::NoFocus);
367 row->setSizePolicy(QSizePolicy::Expanding, QSizePolicy::Preferred);
368 row->setCursor(unlocked ? Qt::PointingHandCursor : Qt::ArrowCursor);
369 row->setEnabled(unlocked); // 未解锁的活动禁用点击
370
371 auto* layout = new QHBoxLayout(row);
372 layout->setContentsMargins(4, 2, 4, 2);
373 layout->setSpacing(8);
374
375 // 图标：☑ / ☐ / 🔒
376 QString icon;
377 if (completed) {
378 icon = QString::fromUtf8("\xe2\x98\x91"); // ☑ U+2611
379 } else if (!unlocked) {
380 icon = QString::fromUtf8("\xf0\x9f\x94\x92"); // 🔒 lock emoji
381 } else {
382 icon = QString::fromUtf8("\xe2\x98\x90"); // ☐ U+2610
383 }
384 auto* iconLabel = new QLabel(icon, row);
385 iconLabel->setFixedSize(22, 22);
386 iconLabel->setAlignment(Qt::AlignCenter);
387 layout->addWidget(iconLabel, 0);
388
389 // 中间：标题 + 描述
390 auto* mid = new QVBoxLayout;
391 mid->setSpacing(1);
392
393 QString title = QString::fromStdString(activity.title);
394 if (!unlocked) {
395 title = QString::fromUtf8("<span style='color:#6E6E6E;'>%1</span>").arg(title);
396 } else if (completed) {
397 title = QString::fromUtf8("<span style='color:#4CAF50;'><b>%1</b> ✓</span>").arg(title);
398 } else {
399 title = QString::fromUtf8("<b>%1</b>").arg(title);
400 }
401
402 auto* titleLabel = new QLabel(title, row);
403 titleLabel->setTextFormat(Qt::RichText);
404 mid->addWidget(titleLabel);
405
406 QString desc = QString::fromStdString(activity.description);
407 desc = QString::fromUtf8("<span style='color:#666; font-size:90%%;'>%1</span>").arg(desc);
408 auto* descLabel = new QLabel(desc, row);
409 descLabel->setTextFormat(Qt::RichText);
410 descLabel->setWordWrap(true);
411 mid->addWidget(descLabel);
412
413 layout->addLayout(mid, 1);
414
415 // 预计耗时
416 QString timeStr = mlTr("~%1 分钟").arg(activity.estimatedMinutes);
417 auto& store = LearnerProgressStore::instance();
418 int spent = store.getSpentMinutes(activity.id);
419 if (spent > 0) {
420 // P2-3 fix (F9): 已有实际耗时数据时，显示「~X 分钟 / 已用 Y 分钟」
421 timeStr = mlTr("~%1 分钟 / 已用 %2").arg(activity.estimatedMinutes).arg(spent);
422 }
423 auto* timeLabel = new QLabel(timeStr, row);
424 timeLabel->setStyleSheet(QString::fromUtf8("color:#6E6E6E;"));
425 layout->addWidget(timeLabel, 0);
426
427 // P2-3 fix (F9): 学情画像——显示得分/星级徽章（仅有记录时显示）
428 int score = store.getScore(activity.id);
429 int stars = store.getBestStars(activity.id);
430 if (score > 0 || stars > 0) {
431 QStringList parts;
432 if (score > 0) {
433 parts << mlTr("得分 %1").arg(score);
434 }
435 if (stars > 0) {
436 // 用 ★ 字符显示星级
437 QString starStr;
438 for (int i = 0; i < stars && i < 3; ++i) {
439 starStr += QString::fromUtf8("\xe2\x98\x85"); // ★ U+2605
440 }
441 parts << starStr;
442 }
443 if (!parts.isEmpty()) {
444 auto* badge = new QLabel(parts.join(QString::fromUtf8(" · ")), row);
445 badge->setStyleSheet(QString::fromUtf8("color: %1; font-weight: bold; padding: 1px 6px;"
446 " border: 1px solid %2; border-radius: 3px;")
447 .arg(TeachingTheme::success().name().left(7), TeachingTheme::success().name()));
448 layout->addWidget(badge, 0);
449 }
450 }
451
452 // 推荐标签
453 if (recommended && unlocked && !completed) {
454 auto* recLabel = new QLabel(QString::fromUtf8("<b style='color:%1;'>\xe2\x86\x90 %2</b>")
455 .arg(stageColor(activity.stage))
456 .arg(mlTr("当前推荐")),
457 row);
458 recLabel->setTextFormat(Qt::RichText);
459 layout->addWidget(recLabel, 0);
460
461 // 高亮整行背景
462 // AUDIT-P2 fix: 原 rgba(%1, 0.18) 传 hex 字符串 (#859900) 给 QSS rgba() 函数
463 // 不合法，QSS rgba 只接受数字参数 rgba(r, g, b, a)。改为展开为数字格式。
464 QColor sc = stageColor(activity.stage);
465 row->setStyleSheet(QString::fromUtf8("QPushButton { background-color: rgba(0,0,0,0); border-radius: 4px; }"
466 "QPushButton:hover { background-color: rgba(%1, %2, %3, 46); }")
467 .arg(sc.red())
468 .arg(sc.green())
469 .arg(sc.blue()));
470 } else if (unlocked) {
471 row->setStyleSheet(QString::fromUtf8("QPushButton { background-color: rgba(0,0,0,0); border: none; }"
472 "QPushButton:hover { background-color: rgba(0,0,0,0.06); }"));
473 } else {
474 row->setStyleSheet(
475 QString::fromUtf8("QPushButton { background-color: rgba(0,0,0,0); border: none; color: #6E6E6E; }"));
476 // 未解锁：tooltip 提示需要先完成的前置活动
477 QStringList prereqTitles;
478 for (const auto& prereqId : activity.prerequisites) {
479 const LearningActivity* prereq = LearningPathData::findById(prereqId);
480 if (prereq) {
481 prereqTitles << QString::fromStdString(prereq->title);
482 }
483 }
484 if (!prereqTitles.isEmpty()) {
485 row->setToolTip(mlTr("需要先完成：%1 才能解锁").arg(prereqTitles.join(", ")));
486 } else {
487 row->setToolTip(mlTr("此活动尚未解锁"));
488 }
489 }
490
491 // 连接点击信号
492 if (unlocked) {
493 QString actId = QString::fromStdString(activity.id);
494 connect(row, &QPushButton::clicked, this, [this, actId]() { onActivityClicked(actId); });
495 // M9: 收集到键盘导航列表中（仅已解锁项可被导航选中）
496 activityRows_.append(row);
497
498 // P2-3 fix (F9): 已解锁活动的 tooltip——展示完整学情画像
499 int attempts = 0;
500 {
501 auto it = store.data().attemptCount.find(activity.id);
502 if (it != store.data().attemptCount.end())
503 attempts = it->second;
504 }
505 int fails = store.getFailCount(activity.id);
506 if (attempts > 0 || fails > 0 || score > 0 || spent > 0 || stars > 0) {
507 QStringList lines;
508 lines << mlTr("尝试 %1 次").arg(attempts);
509 if (fails > 0)
510 lines << mlTr("失败 %1 次").arg(fails);
511 if (score > 0)
512 lines << mlTr("得分 %1").arg(score);
513 if (stars > 0)
514 lines << mlTr("星级 %1/3").arg(stars);
515 if (spent > 0)
516 lines << mlTr("已用 %1 分钟").arg(spent);
517 row->setToolTip(lines.join(QStringLiteral(" · ")));
518 }
519 }
520
521 return row;
522}
