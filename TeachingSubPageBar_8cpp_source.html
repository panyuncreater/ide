24int TeachingSubPageBar::addPage(const QString& buttonText, QWidget* page) {
25 auto* btn = new QPushButton(buttonText);
26 btn->setObjectName("teachingSubPageBtn");
27 btn->setCheckable(true);
28 const int idx = buttons_.size();
29 buttons_.append(btn);
30 // 插入到 stretch（最后一个 item）之前
31 buttonBar_->insertWidget(buttonBar_->count() - 1, btn);
32 stack_->addWidget(page);
33
34 // 互斥 + 切换
35 connect(btn, &QPushButton::clicked, this, [this, idx]() { setCurrentIndex(idx); });
36 if (idx == 0) {
37 btn->setChecked(true);
38 }
39 return idx;
40}

42void TeachingSubPageBar::setCurrentIndex(int idx) {
43 if (idx < 0 || idx >= stack_->count() || idx == stack_->currentIndex()) {
44 return;
45 }
46 stack_->setCurrentIndex(idx);
47 // 更新按钮选中态
48 for (int i = 0; i < buttons_.size(); ++i) {
49 buttons_[i]->setChecked(i == idx);

51 updateButtonStyles();

55 }
