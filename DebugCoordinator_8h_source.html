124 /// R121 调用栈帧切换：按帧索引获取该帧的局部变量（Interpreter 路径）。
125 /// @param depth 帧索引（0=栈底 main，递增到栈顶）。越界或无 env 返回空。
126 /// @note 通过 interpreter_->getCallStackSnapshot() 直接访问 CallFrame.env->snapshotLocalVariables()，
127 /// 避免每次调用都通过 debugger_->getCallStack() 触发完整 callback 重建栈快照。
128 /// @note R121-build fix: 改为 inline 以避免 VariableInspectorPanel.cpp 依赖
129 /// DebugCoordinator.cpp 的非 inline 符号（minilang_tests 不链接 DebugCoordinator.cpp）。
130 /// 与 R117 WatchExpressionLibrary 抽取 Library.cpp 的解法不同——此处逻辑简单且
131 /// DebugCoordinator.h 已传递包含 Interpreter.h / CallFrame.h / Environment.h。
132 std::vector<std::pair<std::string, Value>> getFrameLocalsAt(int depth) const {
133 std::vector<std::pair<std::string, Value>> result;
134 try {
135 auto stack = interpreter_->getCallStackSnapshot();
136 if (depth < 0 || depth >= static_cast<int>(stack.size())) {
137 return result;
138 }
139 const auto& frame = stack[depth];
140 if (frame.env) {
141 // P0-2 fix: 用 snapshotLocalVariables() 获取值拷贝，避免直接遍历
142 // Environment::variables 的 const 引用（worker 线程可能并发修改）。
143 auto locals = frame.env->snapshotLocalVariables();
144 for (const auto& kv : locals) {
145 result.emplace_back(kv.first, kv.second);
146 }
147 }
148 } catch (const std::exception& e) {
149 Logger::Warning(std::string("getFrameLocalsAt 异常: ") + e.what(), "Debugger");
150 } catch (...) {
151 Logger::Warning("getFrameLocalsAt 发生未知异常", "Debugger");
152 }
153 return result;
154 }
