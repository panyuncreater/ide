21// PERF: 判断表达式是否可证明为整数类型（保守策略）
22// 规则：
23// 1. NumberLiteral 且 isInt() == true
24// 2. VarRef 且类型注解为 "int"
25// 3. UnaryOp(NEGATE) 且操作数为已知 int
26// 4. BinaryOp(+,-,*,%) 且两个操作数均为已知 int
27// 其他情况均返回 false（回退到通用操作码）
28bool Compiler::isExprKnownInt(const ASTNode* expr) const {
29 if (!expr)
30 return false;
31 switch (expr->nodeType) {
32 case NodeType::NODE_NUMBER_LITERAL: {
33 const auto* num = static_cast<const NumberLiteral*>(expr);
34 return num->isInt();
35 }
36 case NodeType::NODE_VAR_REF: {
37 const auto* ref = static_cast<const VarRef*>(expr);
38 const std::string* type = findVarType(ref->name);
39 return type && *type == "int";
40 }
41 case NodeType::NODE_UNARY_OP: {
42 const auto* unary = static_cast<const UnaryOp*>(expr);
43 if (unary->opType == UnaryOp::UnaryOpType::UOP_NEGATE)
44 return isExprKnownInt(unary->operand.get());
45 return false;
46 }
47 case NodeType::NODE_BINARY_OP: {
48 const auto* bin = static_cast<const BinaryOp*>(expr);
49 // 仅对产生 int 结果的运算符递归检查（除法可能产生 float，比较产生 bool）
50 if (bin->opType == BinOpType::BIN_ADD || bin->opType == BinOpType::BIN_SUB ||
51 bin->opType == BinOpType::BIN_MUL || bin->opType == BinOpType::BIN_MOD) {
52 return isExprKnownInt(bin->left.get()) && isExprKnownInt(bin->right.get());
53 }
54 return false;
55 }
56 default:
57 return false;
58 }
59}

61void Compiler::visitBinaryOp(BinaryOp& node) {
62 // 常量折叠：编译期求值常量表达式
63 Value folded;
64 if (tryFoldBinary(node.opType, node.left.get(), node.right.get(), folded, node.line)) {
65 emitConstant(folded, node.line);
66 return;
67 }
68
69 // 短路运算特殊处理
70 if (node.opType == BinOpType::BIN_AND) {
71 compileNode(node.left.get());
72 // 如果左操作数为假，跳过右操作数
73 size_t jumpPatch = chunk_.code.size();
74 chunk_.writeOp(OpCode::OP_JUMP_IF_FALSE, node.line);
75 chunk_.writeShort(0, node.line); // 占位
76 chunk_.writeOp(OpCode::OP_POP, node.line); // 弹出左操作数
77 compileNode(node.right.get());
78 // 修补跳转地址
79 uint16_t jumpTarget = safeCodeOffset();
80 chunk_.code[jumpPatch + 1] = static_cast<uint8_t>(jumpTarget & 0xFF);
81 chunk_.code[jumpPatch + 2] = static_cast<uint8_t>((jumpTarget >> 8) & 0xFF);
82 // M1 fix: 移除 NOT NOT 双重取反，保留操作数原始值（JS 语义）
83 return;
84 }
85
86 if (node.opType == BinOpType::BIN_OR) {
87 compileNode(node.left.get());
88 // 如果左操作数为真，跳过右操作数，保留左值
89 size_t jumpPatch = chunk_.code.size();
90 chunk_.writeOp(OpCode::OP_JUMP_IF_FALSE, node.line);
91 chunk_.writeShort(0, node.line); // 占位（先跳到 or 右侧）
92 // 如果到这里，左操作数为真 — 保留左值，跳到末尾
93 size_t jumpEnd = chunk_.code.size();
94 chunk_.writeOp(OpCode::OP_JUMP, node.line);
95 chunk_.writeShort(0, node.line); // 占位
96 // 修补第一个跳转：左操作数为假，求值右操作数
97 uint16_t rightStart = safeCodeOffset();
98 chunk_.code[jumpPatch + 1] = static_cast<uint8_t>(rightStart & 0xFF);
99 chunk_.code[jumpPatch + 2] = static_cast<uint8_t>((rightStart >> 8) & 0xFF);
100 // 弹出左操作数，求值右操作数
101 chunk_.writeOp(OpCode::OP_POP, node.line);
102 compileNode(node.right.get());
103 // 修补第二个跳转
104 uint16_t endTarget = safeCodeOffset();
105 chunk_.code[jumpEnd + 1] = static_cast<uint8_t>(endTarget & 0xFF);
106 chunk_.code[jumpEnd + 2] = static_cast<uint8_t>((endTarget >> 8) & 0xFF);
107 // M1 fix: 移除 NOT NOT 双重取反，保留操作数原始值（JS 语义）
108 return;
109 }
110
111 // 普通二元运算
112 compileNode(node.left.get());
113 compileNode(node.right.get());

116 // 保守策略：仅当 100% 确定两侧均为 int 时才使用特化版本

126 break;
127 case BinOpType::BIN_MUL:
128 op = bothInt ? OpCode::OP_MUL_INT_SPEC : OpCode::OP_MULTIPLY;
129 break;
130 case BinOpType::BIN_DIV:
131 op = OpCode::OP_DIVIDE;
132 break;
133 case BinOpType::BIN_MOD:
134 op = OpCode::OP_MODULO;
135 break;
136 case BinOpType::BIN_EQ:
137 op = OpCode::OP_EQUAL;
138 break;
139 case BinOpType::BIN_NEQ:
140 op = OpCode::OP_NOT_EQUAL;
141 break;
142 case BinOpType::BIN_LT:
143 op = bothInt ? OpCode::OP_LT_INT_SPEC : OpCode::OP_LESS;
144 break;
145 case BinOpType::BIN_GT:
146 op = OpCode::OP_GREATER;
147 break;
148 case BinOpType::BIN_LTE:
149 op = OpCode::OP_LESS_EQUAL;
150 break;
151 case BinOpType::BIN_GTE:
152 op = OpCode::OP_GREATER_EQUAL;

294 } else {

296 error("变量 '" + node.name + "' 已在当前作用域中定义", node.line, node.column);

298 chunk_.write(static_cast<uint8_t>(it->second), node.line);
299 }

301 // 避免函数内循环 var 声明累积导致栈溢出。
302 chunk_.writeOp(OpCode::OP_POP, node.line);
303 } else {

309 chunk_.writeShort(static_cast<uint16_t>(slot), node.line);
310 } else {
311 uint16_t nameIdx = identifierIndex(node.name);
312 chunk_.writeOp(OpCode::OP_DEFINE_VAR, node.line);
313 chunk_.writeShort(nameIdx, node.line);
314 }
315 // B4: topLevelGlobals_ 已删除（write-only 死代码，从未被读取）
316 }
317 return;

393 currentUpvalues_.push_back(desc);

396 }

399 // 原代码将此检查嵌套在步骤 2 内部，导致永远不会执行（变量不可能同时是外层局部变量和 upvalue）
400 auto outerUvIt = outerUpvalueNames_.find(name);
401 if (outerUvIt != outerUpvalueNames_.end()) {
402 UpvalueDesc desc;
403 desc.isLocal = false;
404 desc.index = outerUvIt->second; // 外层 upvalue 索引
405 desc.name = name; // 条件断点修复(R75): 记录upvalue变量名
406 int idx = static_cast<int>(currentUpvalues_.size());
407 currentUpvalues_.push_back(desc);
408 currentUpvalueNames_[name] = idx;
409 return idx;
410 }
411 return -1;
412}

427void Compiler::collectFreeVars(const ASTNode& node, std::vector<std::unordered_set<std::string>>& scopes,
428 std::unordered_set<std::string>& freeVars) {
429 switch (node.nodeType) {
430 case NodeType::NODE_VAR_REF: {
431 const auto& ref = static_cast<const VarRef&>(node);
432 if (!isDefinedInScopes(scopes, ref.name)) {
433 freeVars.insert(ref.name);
434 }
435 break;
436 }
437 case NodeType::NODE_ASSIGNMENT: {
438 const auto& assign = static_cast<const Assignment&>(node);
439 if (!isDefinedInScopes(scopes, assign.name)) {
440 freeVars.insert(assign.name);
441 }
442 if (assign.value)
443 collectFreeVars(*assign.value, scopes, freeVars);
444 break;
445 }
446 case NodeType::NODE_VAR_DECL: {
447 const auto& decl = static_cast<const VarDecl&>(node);
448 if (decl.initializer)
449 collectFreeVars(*decl.initializer, scopes, freeVars);
450 scopes.back().insert(decl.name);
451 break;
452 }
453 case NodeType::NODE_FUN_DECL: {
454 const auto& nestedFn = static_cast<const FunDecl&>(node);
455 for (const auto& dv : nestedFn.defaultValues) {
456 if (dv)
457 collectFreeVars(*dv, scopes, freeVars);
458 }
459 std::vector<std::unordered_set<std::string>> nestedScopes;
460 nestedScopes.emplace_back();
461 for (const auto& p : nestedFn.params)
462 nestedScopes.back().insert(p);
463 // R98 W3: 匿名 lambda（name 为空）不插入空名到作用域
464 if (!nestedFn.name.empty()) {
465 nestedScopes.back().insert(nestedFn.name);
466 }
467 std::unordered_set<std::string> nestedFree;
468 if (nestedFn.body)
469 collectFreeVars(*nestedFn.body, nestedScopes, nestedFree);
470 for (const auto& name : nestedFree) {
471 if (!isDefinedInScopes(scopes, name)) {
472 freeVars.insert(name);
473 }
474 }
475 // R98 W3: 匿名 lambda 不注册到外层作用域
476 if (!nestedFn.name.empty()) {
477 scopes.back().insert(nestedFn.name);
478 }
479 break;
480 }
481 case NodeType::NODE_FUN_CALL: {
482 const auto& call = static_cast<const FunCall&>(node);
483 if (call.callee) {
484 collectFreeVars(*call.callee, scopes, freeVars);
485 } else if (!call.name.empty()) {
486 if (!isDefinedInScopes(scopes, call.name)) {
487 freeVars.insert(call.name);
488 }
489 }
490 for (const auto& arg : call.arguments) {
491 if (arg)
492 collectFreeVars(*arg, scopes, freeVars);
493 }
494 break;
495 }
496 case NodeType::NODE_BLOCK: {
497 const auto& block = static_cast<const Block&>(node);
498 scopes.emplace_back();
499 for (const auto& stmt : block.statements) {
500 if (stmt)
501 collectFreeVars(*stmt, scopes, freeVars);
502 }
503 scopes.pop_back();
504 break;
505 }
506 case NodeType::NODE_FOR_STMT: {
507 const auto& forStmt = static_cast<const ForStmt&>(node);
508 scopes.emplace_back();
509 if (forStmt.initializer)
510 collectFreeVars(*forStmt.initializer, scopes, freeVars);
511 if (forStmt.condition)
512 collectFreeVars(*forStmt.condition, scopes, freeVars);
513 if (forStmt.update)
514 collectFreeVars(*forStmt.update, scopes, freeVars);
515 if (forStmt.body)
516 collectFreeVars(*forStmt.body, scopes, freeVars);
517 scopes.pop_back();
518 break;
519 }
520 case NodeType::NODE_TRY_STMT: {
521 const auto& tryStmt = static_cast<const TryStmt&>(node);
522 if (tryStmt.tryBlock)
523 collectFreeVars(*tryStmt.tryBlock, scopes, freeVars);
524 if (tryStmt.catchBlock) {
525 scopes.emplace_back();
526 if (!tryStmt.catchVarName.empty())
527 scopes.back().insert(tryStmt.catchVarName);
528 collectFreeVars(*tryStmt.catchBlock, scopes, freeVars);
529 scopes.pop_back();
530 }
531 break;
532 }
533 case NodeType::NODE_CLASS_DECL: {
534 const auto& cls = static_cast<const ClassDecl&>(node);
535 scopes.back().insert(cls.name);
536 break;
537 }
538 default:
539 for (auto* child : node.children()) {
540 if (child)
541 collectFreeVars(*child, scopes, freeVars);
542 }
543 break;
544 }
545}

547std::unordered_set<std::string> Compiler::computeFreeVars(const FunDecl& fn) {
548 std::vector<std::unordered_set<std::string>> scopes;
549 scopes.emplace_back();
550 for (const auto& param : fn.params)
551 scopes.back().insert(param);
552 scopes.back().insert(fn.name);
553
554 std::unordered_set<std::string> freeVars;
555 for (const auto& dv : fn.defaultValues) {
556 if (dv)
557 collectFreeVars(*dv, scopes, freeVars);
558 }
559 if (fn.body)
560 collectFreeVars(*fn.body, scopes, freeVars);
561 return freeVars;
562}

595// R98 元组与解构：解构绑定 var (a, b, c) = expr 编译
596// 策略：
597// 1. 编译 initializer，栈顶为元组值
598// 2. 对每个 name，按位置 i 生成：
599// DUP（复制元组到栈顶）+ OP_INT i + OP_INDEX_GET（取第 i 个元素）
600// + 变量绑定（OP_DEFINE_VAR / OP_SET_LOCAL）
601// 3. 最后 OP_POP 清理元组（除非在表达式上下文，但解构绑定是语句，无返回值消费）
602// 4. 末尾 OP_POP 清理 DUP 产生的栈顶元组副本
603// 注意：与 visitVarDecl 不同，解构绑定是语句，不保留栈上值。
604void Compiler::visitDestructureBinding(DestructureBinding& node) {
605 if (node.names.size() > 255) {
606 error("解构绑定变量数超过 255 个上限", node.line, node.column);
607 return;
608 }
609
610 // 编译 initializer，栈顶为元组
611 compileNode(node.initializer.get());
612
613 // 为每个变量取元素并绑定
614 for (size_t i = 0; i < node.names.size(); ++i) {
615 // DUP 复制元组到栈顶（保留原元组供下一轮使用）
616 chunk_.writeOp(OpCode::OP_DUP, node.line);
617 // push 索引 i
618 uint16_t idxConst = chunk_.addConstant(Value(static_cast<int64_t>(i)));
619 chunk_.writeOp(OpCode::OP_INT, node.line);
620 chunk_.writeShort(idxConst, node.line);
621 // 取元素：tuple[i]
622 chunk_.writeOp(OpCode::OP_INDEX_GET, node.line);
623 // AUDIT-R6 F4 fix: per-name 类型注解运行时校验（对齐 Interpreter
624 // visitDestructureBinding 的 L20 checkType）。原仅 Interpreter 校验，
625 // `var (a: int, b) = ("s", 2)` 在 VM 静默通过——三后端不一致。
626 // OP_TYPE_CHECK peek 不弹栈，在绑定前检查栈顶元素。
627 emitTypeCheck(node.nameTypeAt(i), node.line);
628 // 栈顶为 tuple[i]，绑定到变量 name
629 bindDestructureVar(node.names[i], node.line, node.column);
630 }
631
632 // 清理栈顶元组（DUP 已被消费，仅剩原始元组）
633 chunk_.writeOp(OpCode::OP_POP, node.line);
634 return;
635}

637// R98 元组与解构：解构绑定单个变量的绑定逻辑（复用 visitVarDecl 的局部/全局分支）
638void Compiler::bindDestructureVar(const std::string& name, int line, int col) {
639 if (inFunction_) {
640 auto it = currentLocals_.find(name);
641 if (it == currentLocals_.end()) {
642 int slot = static_cast<int>(currentLocals_.size());
643 if (slot > 255) {
644 error("函数局部变量数量超过限制（最大 256 个，含 this/参数/字段）", line, col);
645 return;
646 }
647 currentLocals_[name] = slot;
648 peakLocals_ = std::max(peakLocals_, static_cast<int>(currentLocals_.size()));
649 if (static_cast<size_t>(slot) >= localSlotNames_.size()) {
650 localSlotNames_.resize(slot + 1);
651 }
652 localSlotNames_[slot] = name;
653 slotNameRanges_.push_back({static_cast<uint8_t>(slot), name, chunk_.code.size(), 0});
654 chunk_.writeOp(OpCode::OP_SET_LOCAL, line);
655 chunk_.write(static_cast<uint8_t>(slot), line);
656 } else {
657 error("变量 '" + name + "' 已在当前作用域中定义", line, col);
658 chunk_.writeOp(OpCode::OP_SET_LOCAL, line);
659 chunk_.write(static_cast<uint8_t>(it->second), line);
660 }
661 // OP_SET_LOCAL 用 peek(0) 不消费栈顶，补发 OP_POP 清理元素
662 chunk_.writeOp(OpCode::OP_POP, line);
663 } else {
664 int slot = (blockDepth_ > 0) ? -1 : lookupGlobalSlot(name);
665 if (slot >= 0) {
666 chunk_.writeOp(OpCode::OP_DEFINE_GLOBAL, line);
667 chunk_.writeShort(static_cast<uint16_t>(slot), line);
668 } else {
669 uint16_t nameIdx = identifierIndex(name);
670 chunk_.writeOp(OpCode::OP_DEFINE_VAR, line);
671 chunk_.writeShort(nameIdx, line);
672 }
673 }
674}

680// enum 声明编译：将 enum 名绑定为标记值 "enum:Name"。
681// 不需要向 VM 注册 variant 元信息——OP_BUILD_ENUM_VARIANT 通过
682// 常量池中的 enumName/variantName 字符串直接构造 variant 值，
683// 类型校验由 Interpreter（visitEnumVariantExpr）和 TypeChecker 负责。
684void Compiler::visitEnumDecl(EnumDecl& node) {
685 // 编译期：在当前作用域定义 enum 名为标记值 "enum:Name"
686 // 运行时：OP_STRING + OP_DEFINE_VAR/OP_DEFINE_GLOBAL
687 std::string marker = std::string("enum:") + node.name;
688 uint16_t markerIdx = chunk_.addConstant(Value(marker));
689 chunk_.writeOp(OpCode::OP_STRING, node.line);
690 chunk_.writeShort(markerIdx, node.line);
691
692 if (inFunction_) {
693 // 函数内：使用局部变量槽位
694 auto it = currentLocals_.find(node.name);
695 if (it == currentLocals_.end()) {
696 int slot = static_cast<int>(currentLocals_.size());
697 if (slot > 255) {
698 error("函数局部变量数量超过限制（最大 256 个，含 this/参数/字段）", node.line, node.column);
699 chunk_.writeOp(OpCode::OP_POP, node.line); // 平衡栈
700 return;
701 }
702 currentLocals_[node.name] = slot;
703 peakLocals_ = std::max(peakLocals_, static_cast<int>(currentLocals_.size()));
704 if (static_cast<size_t>(slot) >= localSlotNames_.size()) {
705 localSlotNames_.resize(slot + 1);
706 }
707 localSlotNames_[slot] = node.name;
708 slotNameRanges_.push_back({static_cast<uint8_t>(slot), node.name, chunk_.code.size(), 0});
709 chunk_.writeOp(OpCode::OP_SET_LOCAL, node.line);
710 chunk_.write(static_cast<uint8_t>(slot), node.line);
711 chunk_.writeOp(OpCode::OP_POP, node.line); // OP_SET_LOCAL 用 peek，需补 POP
712 } else {
713 // 重定义：直接写入已有槽位
714 chunk_.writeOp(OpCode::OP_SET_LOCAL, node.line);
715 chunk_.write(static_cast<uint8_t>(it->second), node.line);
716 chunk_.writeOp(OpCode::OP_POP, node.line);
717 }
718 } else {
719 // 顶层：使用全局变量槽位或慢路径
720 int slot = (blockDepth_ > 0) ? -1 : lookupGlobalSlot(node.name);
721 if (slot >= 0) {
722 chunk_.writeOp(OpCode::OP_DEFINE_GLOBAL, node.line);
723 chunk_.writeShort(static_cast<uint16_t>(slot), node.line);
724 } else {
725 uint16_t nameIdx = identifierIndex(node.name);
726 chunk_.writeOp(OpCode::OP_DEFINE_VAR, node.line);
727 chunk_.writeShort(nameIdx, node.line);
728 }
729 }

731 // R99 enum 校验：收集 enum 元信息到 pendingEnumInfos_，compile() 完成时
732 // 写入 CompileResult.enumInfos 供 VM 启动时加载到 enumRegistry_。
733 // 对齐 Interpreter::visitEnumDecl 的 enumRegistry_ 注册，使 OP_BUILD_ENUM_VARIANT
734 // 能在运行时校验 variant 名存在性 + 参数 arity 一致性，保证三后端一致。
735 VMEnumInfo info;
736 info.name = node.name;
737 // AUDIT-R6 F3 fix: 携带字段类型注解与泛型参数，使 OP_BUILD_ENUM_VARIANT 能对齐
738 // Interpreter 的字段类型校验（原仅 arity）。
739 info.typeParams = node.typeParams;
740 info.variants.reserve(node.variants.size());
741 for (const auto& v : node.variants) {
742 VMEnumVariantInfo vi;
743 vi.name = v.name;
744 vi.arity = static_cast<int>(v.paramTypes.size());
745 vi.paramTypes = v.paramTypes;
746 info.variants.push_back(std::move(vi));
747 }
748 pendingEnumInfos_.push_back(std::move(info));
749 return;
750}

753// 栈布局：依次 push 各参数（按声明顺序），然后 OP_BUILD_ENUM_VARIANT 消费 argCount 个参数。
754void Compiler::visitEnumVariantExpr(EnumVariantExpr& node) {

756 error("enum variant 参数数量超过 255 个上限", node.line, node.column);

760 for (auto& arg : node.arguments) {

773// match 表达式编译：
774// 1. 编译 scrutinee，push 到栈
775// 2. 对每个 case：
776// - WILDCARD/DEFAULT：直接执行 body
777// - LITERAL：DUP + literal + OP_EQUAL + JUMP_IF_FALSE → next_case
778// - VARIANT：DUP + OP_ENUM_VARIANT_NAME + JUMP_IF_FALSE → next_case
779// 然后对每个 binding：DUP + OP_INT i + OP_ENUM_VARIANT_FIELD + bind
780// - 编译 body（push 结果）
781// - OP_SWAP（结果与 scrutinee 交换）+ OP_POP（删除 scrutinee）
782// - JUMP → end
783// 3. 所有 case 未匹配：OP_POP（删 scrutinee）+ OP_NULL（push null 作为结果）
784// 4. end：
785void Compiler::visitMatchExpr(MatchExpr& node) {
786 // 编译 scrutinee，栈顶为 scrutinee 值
787 compileNode(node.scrutinee.get());
788
789 // 收集所有 case 的 JUMP_IF_FALSE 占位（用于跳到 next case）
790 std::vector<size_t> caseSkipPatches;
791 // 收集所有 case body 末尾的 JUMP（跳到 end）占位
792 std::vector<size_t> endJumps;
793
794 auto savedLocals = currentLocals_;
795 size_t branchSlotBase = currentLocals_.size();
796 bool needCloseUpvalue = inFunction_;
797
798 for (auto& mc : node.cases) {
799 // case 匹配检查（emitMatchPattern 处理 WILDCARD/LITERAL/VARIABLE/VARIANT/TUPLE/OR 六种模式）
800 if (!mc.isDefault && mc.pattern) {
801 emitMatchPattern(*mc.pattern, caseSkipPatches);
802 }
803 // default case 无需 pattern 检查
804 // R133: guard 编译——pattern 匹配成功后求值 guard，false 则跳到下一 case
805 if (mc.guard) {
806 compileNode(mc.guard.get());
807 size_t guardSkipPatch = chunk_.code.size();
808 chunk_.writeOp(OpCode::OP_JUMP_IF_FALSE, mc.guard->line);
809 chunk_.writeShort(0, mc.guard->line);
810 caseSkipPatches.push_back(guardSkipPatch);
811 chunk_.writeOp(OpCode::OP_POP, mc.guard->line); // 弹出 guard bool
812 }
813
814 // 编译 case body（push 结果到栈），body 可能是 Block 或单表达式
815 compileMatchBody(mc.body.get(), node.line);
816
817 // 关闭 case 作用域内声明的 upvalue（与 visitIfStmt 一致）
818 if (needCloseUpvalue && currentLocals_.size() > branchSlotBase && branchSlotBase <= 255) {
819 chunk_.writeOp(OpCode::OP_CLOSE_UPVALUE, node.line);
820 chunk_.write(static_cast<uint8_t>(branchSlotBase), node.line);
821 }
822 closeSlotRanges(branchSlotBase);
823 currentLocals_ = savedLocals;
824
825 // 栈布局：[scrutinee, body_result] → OP_SWAP → [body_result, scrutinee] → OP_POP → [body_result]
826 chunk_.writeOp(OpCode::OP_SWAP, node.line); // 交换 scrut 和 result
827 chunk_.writeOp(OpCode::OP_POP, node.line); // 弹出 scrut
828
829 // JUMP → end
830 size_t endJump = chunk_.code.size();
831 chunk_.writeOp(OpCode::OP_JUMP, node.line);
832 chunk_.writeShort(0, node.line);
833 endJumps.push_back(endJump);
834
835 // R133: 回填当前 case 的所有 skip patch（pattern + guard）跳到下一 case 起始
836 // 原 R99 代码仅回填单个 patch；扩展后每个 case 可能有多个 patch（嵌套 pattern + guard）
837 if (!caseSkipPatches.empty()) {
838 uint16_t nextCaseStart = safeCodeOffset();
839 while (!caseSkipPatches.empty()) {
840 size_t skipPatch = caseSkipPatches.back();
841 caseSkipPatches.pop_back();
842 chunk_.code[skipPatch + 1] = static_cast<uint8_t>(nextCaseStart & 0xFF);
843 chunk_.code[skipPatch + 2] = static_cast<uint8_t>((nextCaseStart >> 8) & 0xFF);
844 }
845 // JUMP_IF_FALSE 不消费条件值，跳过时栈上仍 bool，补 OP_POP
846 chunk_.writeOp(OpCode::OP_POP, node.line);
847 }
848 }
849
850 // L6 fix: 所有 case 未匹配时，对齐 Interpreter 的 runtimeError 行为（抛异常而非静默返回 null）
851 // 无 default 分支时：OP_POP 删 scrut + OP_STRING(msg) + OP_THROW 抛出异常
852 chunk_.writeOp(OpCode::OP_POP, node.line);
853 uint16_t errIdx = chunk_.addConstant(Value(std::string("match 表达式没有匹配的 case")));
854 chunk_.writeOp(OpCode::OP_STRING, node.line);
855 chunk_.writeShort(errIdx, node.line);
856 chunk_.writeOp(OpCode::OP_THROW, node.line);
857
858 // 回填所有 end jumps
859 uint16_t endTarget = safeCodeOffset();
860 for (size_t endJump : endJumps) {
861 chunk_.code[endJump + 1] = static_cast<uint8_t>(endTarget & 0xFF);
862 chunk_.code[endJump + 2] = static_cast<uint8_t>((endTarget >> 8) & 0xFF);
863 }
864 return;
865}

871void Compiler::emitMatchPattern(const MatchPattern& p, std::vector<size_t>& caseSkipPatches) {
872 // R133 模式匹配扩展：递归处理六种 pattern（WILDCARD/LITERAL/VARIABLE/VARIANT/TUPLE/OR）。
873 // 栈布局不变量：调用前 [scrut]，调用后 [scrut]（保持 scrut 在栈顶供后续 case 复用）。
874 // 匹配失败跳转 patch 追加到 caseSkipPatches 由调用方回填到下一 case 起始。
875 //
876 // L27 重构：原 211 行 switch 按模式类型分发到 6 个子函数，降低圈复杂度。
877 switch (p.kind) {
878 case MatchPatternKind::WILDCARD:
879 // 永远匹配，无需检查
880 break;
881 case MatchPatternKind::LITERAL:
882 emitLiteralMatchPattern(p, caseSkipPatches);
883 break;
884 case MatchPatternKind::VARIABLE:
885 emitVariableMatchPattern(p);
886 break;
887 case MatchPatternKind::VARIANT:
888 emitVariantMatchPattern(p, caseSkipPatches);
889 break;
890 case MatchPatternKind::TUPLE:
891 emitTupleMatchPattern(p, caseSkipPatches);
892 break;
893 case MatchPatternKind::OR:
894 emitOrMatchPattern(p, caseSkipPatches);
895 break;
896 }
897}

899void Compiler::emitLiteralMatchPattern(const MatchPattern& p, std::vector<size_t>& caseSkipPatches) {
900 // DUP scrutinee + literal + OP_EQUAL + JUMP_IF_FALSE → next + OP_POP
901 chunk_.writeOp(OpCode::OP_DUP, p.line);
902 compileNode(p.literal.get());
903 chunk_.writeOp(OpCode::OP_EQUAL, p.line);
904 size_t skipPatch = chunk_.code.size();
905 chunk_.writeOp(OpCode::OP_JUMP_IF_FALSE, p.line);
906 chunk_.writeShort(0, p.line);
907 caseSkipPatches.push_back(skipPatch);
908 chunk_.writeOp(OpCode::OP_POP, p.line);
909}

918void Compiler::emitVariantMatchPattern(const MatchPattern& p, std::vector<size_t>& caseSkipPatches) {
919 // DUP scrutinee + OP_ENUM_VARIANT_NAME + JUMP_IF_FALSE → next + OP_POP
920 chunk_.writeOp(OpCode::OP_DUP, p.line);
921 uint16_t enumIdx = chunk_.addConstant(Value(p.enumName));
922 uint16_t varIdx = chunk_.addConstant(Value(p.variantName));
923 chunk_.writeOp(OpCode::OP_ENUM_VARIANT_NAME, p.line);
924 chunk_.writeShort(enumIdx, p.line);
925 chunk_.writeShort(varIdx, p.line);
926 size_t skipPatch = chunk_.code.size();
927 chunk_.writeOp(OpCode::OP_JUMP_IF_FALSE, p.line);
928 chunk_.writeShort(0, p.line);
929 caseSkipPatches.push_back(skipPatch);
930 chunk_.writeOp(OpCode::OP_POP, p.line); // 弹出 bool
931 // 绑定 variant 字段（递归子 pattern）
932 // R133: subPatterns 替代 R99 的 bindings（vector<string>），支持嵌套
933 for (size_t i = 0; i < p.subPatterns.size(); ++i) {
934 // DUP scrutinee + OP_INT i + OP_ENUM_VARIANT_FIELD + 递归 emitMatchPattern(sub, ...)
935 chunk_.writeOp(OpCode::OP_DUP, p.line);
936 uint16_t idxConst = chunk_.addConstant(Value(static_cast<int64_t>(i)));
937 chunk_.writeOp(OpCode::OP_INT, p.line);
938 chunk_.writeShort(idxConst, p.line);
939 chunk_.writeOp(OpCode::OP_ENUM_VARIANT_FIELD, p.line);
940 // 栈顶是 fields[i]，递归匹配子 pattern
941 // R133 fix: 子 pattern fail 用独立 subFailPatches，失败时先 POP field 再跳到 caseSkip（同 TUPLE pattern）
942 // R133 fix-2: 成功路径 OP_POP 后必须 JUMP 跳过失败路径代码（同 TUPLE pattern）
943 std::vector<size_t> subFailPatches;
944 emitMatchPattern(*p.subPatterns[i], subFailPatches);
945 chunk_.writeOp(OpCode::OP_POP, p.line); // 成功路径 POP field
946 emitSubPatternFailPath(subFailPatches, caseSkipPatches, p.line);
947 }
948}

950void Compiler::emitTupleMatchPattern(const MatchPattern& p, std::vector<size_t>& caseSkipPatches) {
951 // R133: 元组模式 - 检查是 tuple + 元素数匹配 + 递归匹配每个元素
952 // 软类型检查：DUP + OP_TYPE_TEST "tuple"（push bool 不抛错）+ JUMP_IF_FALSE next + POP
953 // 注：用 OP_TYPE_TEST 而非 OP_TYPE_CHECK——后者不匹配时 runtimeError，
954 // 而 TUPLE pattern 不匹配时应 fall through 到下一 case 而非抛错。
955 chunk_.writeOp(OpCode::OP_DUP, p.line);
956 uint16_t tupleTypeNameConst = chunk_.addConstant(Value(std::string("tuple")));
957 chunk_.writeOp(OpCode::OP_TYPE_TEST, p.line);
958 chunk_.writeShort(tupleTypeNameConst, p.line);
959 size_t typeSkipPatch = chunk_.code.size();
960 chunk_.writeOp(OpCode::OP_JUMP_IF_FALSE, p.line);
961 chunk_.writeShort(0, p.line);
962 caseSkipPatches.push_back(typeSkipPatch);
963 chunk_.writeOp(OpCode::OP_POP, p.line);
964 // 元素数检查：DUP + OP_LEN + OP_INT size + OP_EQUAL + JUMP_IF_FALSE next + POP
965 chunk_.writeOp(OpCode::OP_DUP, p.line);
966 chunk_.writeOp(OpCode::OP_LEN, p.line);
967 uint16_t sizeConst = chunk_.addConstant(Value(static_cast<int64_t>(p.subPatterns.size())));
968 chunk_.writeOp(OpCode::OP_INT, p.line);
969 chunk_.writeShort(sizeConst, p.line);
970 chunk_.writeOp(OpCode::OP_EQUAL, p.line);
971 size_t sizeSkipPatch = chunk_.code.size();
972 chunk_.writeOp(OpCode::OP_JUMP_IF_FALSE, p.line);
973 chunk_.writeShort(0, p.line);
974 caseSkipPatches.push_back(sizeSkipPatch);
975 chunk_.writeOp(OpCode::OP_POP, p.line);
976 // 递归匹配每个元素
977 for (size_t i = 0; i < p.subPatterns.size(); ++i) {
978 // DUP tuple + OP_INT i + OP_INDEX_GET + 递归 emitMatchPattern(sub, ...) + POP
979 chunk_.writeOp(OpCode::OP_DUP, p.line);
980 uint16_t idxConst = chunk_.addConstant(Value(static_cast<int64_t>(i)));
981 chunk_.writeOp(OpCode::OP_INT, p.line);
982 chunk_.writeShort(idxConst, p.line);
983 chunk_.writeOp(OpCode::OP_INDEX_GET, p.line);
984 // R133 fix: 子 pattern 的 fail 用独立 subFailPatches，失败时先 POP element 再跳到 caseSkip。
985 // 原实现把子 pattern fail 直接加入 caseSkipPatches，失败时栈上有 [scrut, element, bool]，
986 // caseSkip 处只 POP bool，element 残留导致后续 case 栈错乱。
987 // 正确做法：子 pattern fail 回填到"POP element + JUMP caseSkip"，先清栈再跳转。
988 // R133 fix-2: 成功路径 OP_POP 后必须 JUMP 跳过失败路径代码，否则会跌入失败路径
989 // 的 OP_POP+OP_JUMP，导致栈下溢。结构：
990 // [子 pattern body] → [scrut, elem, bool]
991 // OP_JUMP_IF_FALSE subFailPos (失败跳走，栈 [scrut, elem, bool])
992 // OP_POP (成功路径 LITERAL 自己的 pop bool) → [scrut, elem]
993 // OP_POP (成功路径 TUPLE loop 的 pop elem) → [scrut]
994 // OP_JUMP afterFail ← 成功路径跳过失败代码
995 // subFailPos:
996 // OP_POP (失败路径 pop bool) → [scrut, elem]
997 // OP_JUMP caseSkip → [scrut, elem] (caseSkip 处 OP_POP 消费 elem)
998 // afterFail: (下一 sub-pattern)
999 std::vector<size_t> subFailPatches;
1000 emitMatchPattern(*p.subPatterns[i], subFailPatches);
1001 // 成功路径：POP element 继续
1002 chunk_.writeOp(OpCode::OP_POP, p.line);
1003 emitSubPatternFailPath(subFailPatches, caseSkipPatches, p.line);
1004 }

1007void Compiler::emitSubPatternFailPath(std::vector<size_t>& subFailPatches, std::vector<size_t>& caseSkipPatches,
1008 int line) {
1009 // VARIANT/TUPLE 共用：失败路径回填 + POP elem/field + JUMP caseSkip + 回填成功 JUMP。
1010 // 调用前须已 emit 子 pattern 与成功路径 POP（elem/field）。
1011 // subFailPatches 为空时无失败路径代码可 emit，直接返回。
1012 if (subFailPatches.empty()) {
1013 return;
1014 }
1015 // 成功路径 JUMP 跳过失败路径代码（占位，稍后回填）
1016 size_t successJumpOver = chunk_.code.size();
1017 chunk_.writeOp(OpCode::OP_JUMP, line);
1018 chunk_.writeShort(0, line);
1019 // 失败路径入口
1020 uint16_t subFailPos = safeCodeOffset();
1021 for (auto& sp : subFailPatches) {
1022 chunk_.code[sp + 1] = static_cast<uint8_t>(subFailPos & 0xFF);
1023 chunk_.code[sp + 2] = static_cast<uint8_t>((subFailPos >> 8) & 0xFF);
1024 }
1025 chunk_.writeOp(OpCode::OP_POP, line); // 失败路径 POP elem/field
1026 size_t jumpToSkip = chunk_.code.size();
1027 chunk_.writeOp(OpCode::OP_JUMP, line);
1028 chunk_.writeShort(0, line);
1029 caseSkipPatches.push_back(jumpToSkip);
1030 // 回填成功路径 JUMP 到此（下一 sub-pattern 或末尾）
1031 uint16_t afterFail = safeCodeOffset();
1032 chunk_.code[successJumpOver + 1] = static_cast<uint8_t>(afterFail & 0xFF);
1033 chunk_.code[successJumpOver + 2] = static_cast<uint8_t>((afterFail >> 8) & 0xFF);
1034 subFailPatches.clear();
1035}

1037void Compiler::emitOrMatchPattern(const MatchPattern& p, std::vector<size_t>& caseSkipPatches) {
1038 // R133: OR pattern - 任一子 pattern 匹配即成功
1039 // 编译策略：依次尝试每个子 pattern，任一成功跳到 OR 成功位置（OR 末尾）；
1040 // 全部失败时 emit 永假检查让 case 整体跳到下一 case。
1041 std::vector<size_t> successJumps; // 各子 pattern 成功后跳到 OR 末尾
1042 for (size_t i = 0; i < p.subPatterns.size(); ++i) {
1043 // 编译子 pattern，失败的 patch 暂存到 subPatches
1044 std::vector<size_t> subPatches;
1045 emitMatchPattern(*p.subPatterns[i], subPatches);
1046 // 子 pattern 成功：跳到 OR 末尾
1047 size_t successJump = chunk_.code.size();
1048 chunk_.writeOp(OpCode::OP_JUMP, p.line);
1049 chunk_.writeShort(0, p.line);
1050 successJumps.push_back(successJump);
1051 // 子 pattern 失败位置：所有 subPatches 回填到同一位置，然后只写一次 POP
1052 // R133 fix: 原实现每个 sp 都 writeOp POP，导致 failPos 处有多个 POP，
1053 // 第一个 sp 跳转到 failPos 后会连续执行多个 POP，栈下溢。
1054 // 正确做法：所有 patch 共享同一 failPos，只写一次 POP。
1055 uint16_t failPos = safeCodeOffset();
1056 for (auto& sp : subPatches) {
1057 chunk_.code[sp + 1] = static_cast<uint8_t>(failPos & 0xFF);
1058 chunk_.code[sp + 2] = static_cast<uint8_t>((failPos >> 8) & 0xFF);
1059 }
1060 if (!subPatches.empty()) {
1061 // 失败路径有 bool 残留需 POP（JUMP_IF_FALSE 不消费条件值）
1062 chunk_.writeOp(OpCode::OP_POP, p.line);
1063 }
1064 }
1065 // 所有子 pattern 失败：OR 失败
1066 // R133 fix: caseSkip handler 约定——所有 fail patch 跳转时栈须为 [scrut, X]
1067 // （X 为 JUMP_IF_FALSE 残留的 bool 或 TUPLE/VARIANT sub-pattern fail 残留的 elem）。
1068 // handler 执行一次 OP_POP 消费 X，回到 [scrut] 供下一 case 使用。
1069 // OR 整体失败时栈已回到 [scrut]（sub-pattern failPos 已 POP bool），无残值 X，
1070 // 故 push 一个 OP_NULL 作为占位残值，让 caseSkip handler 的 OP_POP 正确消费。
1071 chunk_.writeOp(OpCode::OP_NULL, p.line);
1072 size_t orFailJump = chunk_.code.size();
1073 chunk_.writeOp(OpCode::OP_JUMP, p.line);
1074 chunk_.writeShort(0, p.line);
1075 caseSkipPatches.push_back(orFailJump);
1076 // OR 成功位置：回填所有 successJumps 跳到这里
1077 uint16_t orSuccess = safeCodeOffset();
1078 for (auto& sj : successJumps) {
1079 chunk_.code[sj + 1] = static_cast<uint8_t>(orSuccess & 0xFF);
1080 chunk_.code[sj + 2] = static_cast<uint8_t>((orSuccess >> 8) & 0xFF);
1081 }
1082}

1084void Compiler::compileMatchBody(ASTNode* body, int line) {
1085 // 编译 case body 并保留值在栈顶（栈不变量：调用前 []，调用后 [body_result]）。
1086 // - Block body：编译除最后一条外的所有语句（带 POP），最后一条用 compileNode 保留值；
1087 // 非表达式语句补 OP_NULL。
1088 // - 单表达式 body：直接 compileNode。
1089 // - 空 body：push OP_NULL。
1090 // 表达式节点类型判定（与 IR.cpp visitMatchExpr 的 needsPopForExprStmt 对齐，
1091 // AUDIT-R3 P2-7：不再写死数量，以枚举清单为准）：
1092 // ASSIGNMENT/FUN_CALL/METHOD_CALL/BINARY_OP/UNARY_OP/VAR_REF/MEMBER_ACCESS/INDEX_ACCESS
1093 // /NUMBER/STRING/BOOL/NULL_LITERAL/SUPER_EXPR/ARRAY/DICT/TUPLE/INTERPOLATED/ENUM_VARIANT/MATCH
1094 if (!body) {
1095 // 空 body：push null
1096 chunk_.writeOp(OpCode::OP_NULL, line);
1097 return;
1098 }
1099
1100 if (body->nodeType != NodeType::NODE_BLOCK) {
1101 compileNode(body);
1102 return;
1103 }
1104
1105 // Block：执行所有语句，最后一条语句的值作为结果
1106 Block* blk = static_cast<Block*>(body);
1107 if (blk->statements.empty()) {
1108 chunk_.writeOp(OpCode::OP_NULL, line);
1109 return;
1110 }
1111 // 编译除最后一条外的所有语句（compileStatement 会消费表达式值）
1112 for (size_t i = 0; i + 1 < blk->statements.size(); ++i) {
1113 compileStatement(blk->statements[i].get());
1114 }
1115 // 最后一条语句用 compileNode 保留值在栈顶（不弹）
1116 ASTNode* last = blk->statements.back().get();
1117 compileNode(last);
1118 // 若 last 非表达式节点（如 VarDecl/IfStmt），不会产生栈值，补 push null
1119 // AUDIT-R4 BUG-04 fix: 改用共享谓词 needsPopForExprStmt（ExprStmtPop.h），
1120 // 消除与 compileStatement/AstIRBuilder 的第三份重复清单。
1121 // “产生栈值的表达式节点”与“表达式语句需 POP 的节点”是同一集合。
1122 // AUDIT-R5 BUG-01 fix: 改用节点级重载——若 last 是匿名 lambda，闭包值已在
1123 // 栈顶作为 case 结果，原类型谓词对 FUN_DECL 返回 false 会误补 OP_NULL，
1124 // 造成双值栈不平衡（case 结果变 null + 泄漏闭包值）。
1125 if (!needsPopForExprStmt(last)) {
1126 // 声明/控制流：无值产生，补 null 作为 case 结果
1127 chunk_.writeOp(OpCode::OP_NULL, line);
1128 }
1129}

1155void Compiler::visitIndexAssign(IndexAssign& node) {
1156 // 根据左值对象类型选择赋值策略
1157 VarRef* objVar = (node.object && node.object->nodeType == NodeType::NODE_VAR_REF)
1158 ? static_cast<VarRef*>(node.object.get())
1159 : nullptr;
1160
1161 if (objVar) {
1162 // 简单路径：arr[i] = val（基是 VarRef）
1163 auto localIt = currentLocals_.find(objVar->name);
1164 if (localIt != currentLocals_.end()) {
1165 compileNode(node.index.get());
1166 compileNode(node.value.get());
1167 uint8_t slot = static_cast<uint8_t>(localIt->second);
1168 chunk_.writeOp(OpCode::OP_INDEX_SET_LOCAL, node.line);
1169 chunk_.write(slot, node.line);
1170 return;
1171 }
1172 // W3-2-Bug1c fix: upvalue 接收者的索引赋值（闭包内 arr[i] = val）
1173 // 原 bug: 只检查 currentLocals_，未检查 upvalue，导致 upvalue 接收者错误走全局变量路径
1174 // 报"未定义的变量"。修复: 用 OP_GET_UPVALUE 获取数组引用 + OP_INDEX_SET 修改元素。
1175 // W3-2-Bug1b 对齐: OP_INDEX_SET 的 COW ensureUnique 在数组 refcount > 1（upvalue 与栈
1176 // 副本同时引用）时产生新数组副本，lastMutatedReceiver_ 持有新副本但 upvalue 仍指向旧值。
1177 // 必须发射 OP_WRITEBACK_INDEX_UPVALUE 将变异后的新数组写回 upvalue。
1178 // 对齐 IR 路径 visitIndexAssign 的 #7 fix（IR.cpp L3035-3039）与 visitMemberAssign 的
1179 // W3-2-Bug1b fix（Compiler.cpp L4718-4734）。
1180 if (inFunction_) {
1181 int uvIdx = resolveUpvalue(objVar->name, node.line);
1182 if (uvIdx >= 0) {
1183 // 栈序: [obj, idx, val] — OP_INDEX_SET 弹出 val/idx/obj
1184 chunk_.writeOp(OpCode::OP_GET_UPVALUE, node.line);
1185 chunk_.write(static_cast<uint8_t>(uvIdx), node.line);
1186 compileNode(node.index.get());
1187 compileNode(node.value.get());
1188 chunk_.writeOp(OpCode::OP_INDEX_SET, node.line);
1189 // COW detach 后写回 upvalue（整体替换语义，lastMutatedReceiver_ = 变异后数组）
1190 chunk_.writeOp(OpCode::OP_WRITEBACK_INDEX_UPVALUE, node.line);
1191 chunk_.write(static_cast<uint8_t>(uvIdx), node.line);
1192 return;
1193 }
1194 }
1195 // 全局变量路径（原逻辑）
1196 compileNode(node.index.get());
1197 compileNode(node.value.get());
1198 uint16_t nameIdx = identifierIndex(objVar->name);
1199 chunk_.writeOp(OpCode::OP_INDEX_SET_VAR, node.line);
1200 chunk_.writeShort(nameIdx, node.line);
1201 return;
1202 }
1203
1204 // M1 fix: 嵌套索引赋值（2 层）
1205 // 检测 node.object 是否是 IndexAccess(VarRef) 或 MemberAccess(VarRef)
1206 IndexAccess* outerIdx = (node.object && node.object->nodeType == NodeType::NODE_INDEX_ACCESS)
1207 ? static_cast<IndexAccess*>(node.object.get())
1208 : nullptr;
1209 MemberAccess* outerMem = (node.object && node.object->nodeType == NodeType::NODE_MEMBER_ACCESS)
1210 ? static_cast<MemberAccess*>(node.object.get())
1211 : nullptr;
1212 VarRef* baseVar = nullptr;
1213 if (outerIdx && outerIdx->object && outerIdx->object->nodeType == NodeType::NODE_VAR_REF)
1214 baseVar = static_cast<VarRef*>(outerIdx->object.get());
1215 else if (outerMem && outerMem->object && outerMem->object->nodeType == NodeType::NODE_VAR_REF)
1216 baseVar = static_cast<VarRef*>(outerMem->object.get());
1217
1218 if (baseVar) {
1219 auto localIt = currentLocals_.find(baseVar->name);
1220 bool isLocal = (localIt != currentLocals_.end());
1221
1222 // W3-2-Bug1c fix: 嵌套索引赋值的 upvalue 接收者（闭包内 b.field[i] = val / arr[j][i] = val）
1223 // 原 bug: 嵌套路径只处理 isLocal / 全局变量，未处理 upvalue，导致 upvalue 接收者错误走全局
1224 // 变量路径报"未定义的变量"。修复: 用 OP_GET_UPVALUE + OP_INDEX_SET/OP_MEMBER_SET + WRITEBACK_*_UPVALUE
1225 // 链传播变异。对齐 IR 路径 emitNestedAssignWriteback 的 UPVALUE 分支（IR.cpp L3006-3013）。
1226 int uvIdx = -1;
1227 if (!isLocal && inFunction_) {
1228 uvIdx = resolveUpvalue(baseVar->name, node.line);
1229 }
1230 if (uvIdx >= 0) {
1231 // 步骤 1-3: 编译外层表达式 → push base.outer（副本） + 内层索引 + 值
1232 compileNode(node.object.get());
1233 compileNode(node.index.get());
1234 compileNode(node.value.get());
1235 // 步骤 4: OP_INDEX_SET 弹出 val/idx/obj，修改 obj（COW detach），lastMutatedReceiver_ = new_base.outer
1236 chunk_.writeOp(OpCode::OP_INDEX_SET, node.line);
1237 // 步骤 5: 加载基变量 + 设置基变量字段/索引为 new_base.outer，产生 new_base
1238 // 栈序: [base, outerIdx, mut]（outerIdx 路径）或 [base, mut]（outerMem 路径）
1239 chunk_.writeOp(OpCode::OP_GET_UPVALUE, node.line);
1240 chunk_.write(static_cast<uint8_t>(uvIdx), node.line);
1241 if (outerIdx) {
1242 // base[outerIdx] = mut
1243 compileNode(outerIdx->index.get());
1244 chunk_.writeOp(OpCode::OP_LOAD_MUTATED, node.line);
1245 chunk_.writeOp(OpCode::OP_INDEX_SET, node.line); // lastMutatedReceiver_ = new_base
1246 chunk_.writeOp(OpCode::OP_WRITEBACK_INDEX_UPVALUE, node.line);
1247 chunk_.write(static_cast<uint8_t>(uvIdx), node.line);
1248 } else {
1249 // base.field = mut
1250 uint16_t fieldIdx = identifierIndex(outerMem->fieldName);
1251 chunk_.writeOp(OpCode::OP_LOAD_MUTATED, node.line);
1252 chunk_.writeOp(OpCode::OP_MEMBER_SET, node.line); // lastMutatedReceiver_ = new_base
1253 chunk_.writeShort(fieldIdx, node.line);
1254 chunk_.writeOp(OpCode::OP_WRITEBACK_MEMBER_UPVALUE, node.line);
1255 chunk_.write(static_cast<uint8_t>(uvIdx), node.line);
1256 chunk_.writeShort(fieldIdx, node.line);
1257 }
1258 return;
1259 }
1260
1261 // 编译外层表达式 → push base[outerIdx] 或 base.field（正确求值中间值）
1262 // 注意：不再 push base 变量本身——WRITEBACK_*_LOCAL/VAR 和 MEMBER_SET_LOCAL/VAR
1263 // 均不需要 base 在栈上，原 push base 会导致栈泄漏（每次嵌套赋值泄漏 1 个 Value）。
1264 compileNode(node.object.get());
1265 // 编译内层索引 → push innerIdx
1266 compileNode(node.index.get());
1267 // 编译值 → push val
1268 compileNode(node.value.get());
1269 // OP_INDEX_SET: 弹出 val/innerIdx/outerValue → 修改 → lastMutatedReceiver_
1270 chunk_.writeOp(OpCode::OP_INDEX_SET, node.line);
1271 // 将变异后的内层容器写回基变量。
1272 // 对齐 IR 路径（IR.cpp emitNestedAssignWriteback L2696-2708）：
1273 // IR 路径用 LOAD_MUTATED + INDEX_SET(base2, outerIdx, mut) + WRITEBACK 链传播变异；
1274 // 非 IR 路径改用更简洁的 LOAD_MUTATED + INDEX_SET_LOCAL/VAR 直接原地修改基变量。
1275 // R156 fix: 原 outerIdx 路径用 WRITEBACK_INDEX_LOCAL/VAR 做整体替换，
1276 // 将 arr 变量替换为变异后的 arr[i]（而非 arr[i] = mutated_arr[i]），
1277 // 导致 arr[i][j]=val 在非 IR 路径（StackVM 非 IR + JIT）全部失败。
1278 // outerMem 路径（base.field[idx]=val）已在 BUG-INH-AUDIT-4 fix 中修正为
1279 // LOAD_MUTATED + MEMBER_SET_LOCAL/VAR，此路径正确无需修改。
1280 if (isLocal) {
1281 if (outerIdx) {
1282 // base[outerIdx] = mutated → 重新求值 outerIdx + LOAD_MUTATED + INDEX_SET_LOCAL
1283 // INDEX_SET_LOCAL 直接修改 stack_[bp+slot][outerIdx] = mutated（原地）
1284 compileNode(outerIdx->index.get());
1285 chunk_.writeOp(OpCode::OP_LOAD_MUTATED, node.line);
1286 chunk_.writeOp(OpCode::OP_INDEX_SET_LOCAL, node.line);
1287 chunk_.write(static_cast<uint8_t>(localIt->second), node.line);
1288 } else {
1289 // base.field = mutated → LOAD_MUTATED + MEMBER_SET_LOCAL（BUG-INH-AUDIT-4 fix，不变）
1290 uint16_t fieldIdx = identifierIndex(outerMem->fieldName);
1291 chunk_.writeOp(OpCode::OP_LOAD_MUTATED, node.line);
1292 chunk_.writeOp(OpCode::OP_MEMBER_SET_LOCAL, node.line);
1293 chunk_.write(static_cast<uint8_t>(localIt->second), node.line);
1294 chunk_.writeShort(fieldIdx, node.line);
1295 }
1296 } else {
1297 if (outerIdx) {
1298 // base[outerIdx] = mutated → 重新求值 outerIdx + LOAD_MUTATED + INDEX_SET_VAR
1299 uint16_t nameIdx = identifierIndex(baseVar->name);
1300 compileNode(outerIdx->index.get());
1301 chunk_.writeOp(OpCode::OP_LOAD_MUTATED, node.line);
1302 chunk_.writeOp(OpCode::OP_INDEX_SET_VAR, node.line);
1303 chunk_.writeShort(nameIdx, node.line);
1304 } else {
1305 // base.field = mutated → LOAD_MUTATED + MEMBER_SET_VAR（不变）
1306 uint16_t varIdx = identifierIndex(baseVar->name);
1307 uint16_t fieldIdx = identifierIndex(outerMem->fieldName);
1308 chunk_.writeOp(OpCode::OP_LOAD_MUTATED, node.line);
1309 chunk_.writeOp(OpCode::OP_MEMBER_SET_VAR, node.line);
1310 chunk_.writeShort(varIdx, node.line);
1311 chunk_.writeShort(fieldIdx, node.line);
1312 }
1313 }
1314 return;
1315 }
