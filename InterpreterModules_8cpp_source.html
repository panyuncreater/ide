14void Interpreter::visitImportStmt(ImportStmt& node) {
15 checkBreak(&node);
16
17 // 七特性 MVP 阶段 6：沙箱拦截——禁用文件 import 时仅放行 std/ 内建模块。
18 // std/ 前缀是内建模块唯一标识（与 BuiltinModuleRegistry::isBuiltinModule 同口径），
19 // 其他路径视为文件模块，沙箱下拒绝（防不受信任代码读取本地文件）。
20 if (RuntimeLimits::RuntimeConfig::instance().sandboxBlocksImport()) {
21 const std::string& p = node.modulePath;
22 bool isStdModule = p.rfind("std/", 0) == 0;
23 if (!isStdModule) {
24 runtimeError("沙箱模式禁止导入文件模块: " + p, node.line, node.column);
25 return;
26 }
27 }
28
29 // R132-C fix: 拆为 3 个子任务 helper（路径解析/加载+缓存/名称导入），
30 // 原函数 208 行 → thin orchestrator ~35 行 + 3 helper。
31 // BUG-M4 fix: loaderPath 保留大小写（供 loader 请求），cacheKey 在 Windows 上小写折叠
32 // （供所有缓存/去重 map 键），使同一文件的不同大小写拼写去重为同一模块。
33 std::string loaderPath = resolveModulePath(node);
34 std::string cacheKey = moduleCacheKey(loaderPath);
35
36 // F12: 模块加载 — A6 fix: 加锁拷贝 callback 后解锁检查，避免跨线程数据竞争
37 std::function<std::string(const std::string&)> loader;
38 std::function<int64_t(const std::string&)> mtimeChecker;
39 {
40 std::lock_guard<std::mutex> lock(callbackMutex_);
41 loader = moduleLoader_;
42 mtimeChecker = moduleMtimeChecker_; // BUG-REPL-AUDIT-1 fix
43 }
44 if (!loader) {
45 runtimeError("未设置模块加载器，无法执行 import", node.line, node.column);
46 }
47
48 // P2-14 循环导入延迟加载：不再报错，由 loadModuleOrGetCached 返回部分加载的模块环境。
49 // 循环回路闭合时，模块环境已创建但可能尚未填充所有导出名称——访问未定义名称时
50 // 由 Environment::get 抛"未定义变量"错误（保持现有语义），访问已定义名称返回当前值。
51 // 语义参考 ES Modules + Python：模块对象立即创建，内容按执行顺序填充。
52 // 深度保护仍保留（防止 C++ 栈溢出）
53 if (moduleLoadingStack_.size() >= MAX_RECURSION_DEPTH) {
54 runtimeError("模块导入深度超过限制 (" + std::to_string(MAX_RECURSION_DEPTH) + ")", node.line, node.column);
55 // R53-4 fix: 同上，防御性 return
56 return;
57 }
58
59 // 加载或获取缓存的模块环境
60 auto moduleEnv = loadModuleOrGetCached(loaderPath, cacheKey, node, loader, mtimeChecker);
61
62 // 导入名称到当前环境
63 importNamesFromModule(cacheKey, node, moduleEnv);
64
65 lastValue_ = Value::nullValue();
66}

72std::string Interpreter::resolveModulePath(ImportStmt& node) {
73 // 解析模块路径（相对于当前文件）— SEC-1 fix: 路径校验在 loader 检查之前，
74 // 输入验证应在资源加载前，确保即使无 loader 也能拒绝恶意路径。
75 // AUDIT-R5 R6 fix: 规范化改用 normalizeModulePathKey（common/ModulePath.h
76 // 单一事实源，与 clearModuleCache/Compiler/AstIRBuilder/pathHash 五处同步），
77 // 补齐连续斜杠折叠与 Windows 大小写折叠（原仅 pathHash 折叠双斜杠，
78 // 等价拼写导致同一模块双重加载、副作用翻倍）。
79 std::string modulePath = normalizeModulePathKey(node.modulePath);
80 // SEC-1 fix: 路径遍历攻击防护 — 拒绝 ".." 路径段和绝对路径，防止 import 读取项目目录外文件。
81 // 检查 ".." 作为完整路径段（避免误判 "foo..bar" 这样的合法文件名）。
82 if (modulePath.empty()) {
83 runtimeError("模块路径不能为空", node.line, node.column);
84 }
85 // 绝对路径检测（Unix 以 '/' 开头，Windows 以 'X:...' 驱动器路径形式）
86 // BUG-MOD-1 fix: 原实现仅检测 'C:/' 形式，未拒绝 'C:foo'（Windows 驱动器相对路径），
87 // 可能被 loader 解析到模块目录外的文件。修复：拒绝所有 'X:' 开头形式
88 // （X 为任意字符），覆盖 'C:/'、'C:foo'、'D:path' 等。
89 // 注：反斜杠已在上方统一转为正斜杠，无需再检测 '\\'。
90 if (modulePath[0] == '/' || (modulePath.size() >= 2 && modulePath[1] == ':')) {
91 runtimeError("模块路径不能为绝对路径: " + modulePath, node.line, node.column);
92 }
93 // ".." 路径段检测（按 '/' 分割检查每个段）
94 std::string normalized = modulePath;
95 size_t pos = 0;
96 while (pos < normalized.size()) {
97 size_t next = normalized.find('/', pos);
98 std::string segment = (next == std::string::npos) ? normalized.substr(pos) : normalized.substr(pos, next - pos);
99 if (segment == "..") {
100 runtimeError("模块路径不能包含父目录引用 '..': " + modulePath, node.line, node.column);
101 }
102 if (next == std::string::npos)
103 break;
104 pos = next + 1;
105 }
106 return modulePath;
107}

109std::shared_ptr<Environment>
110Interpreter::loadModuleOrGetCached(const std::string& loaderPath, const std::string& cacheKey, ImportStmt& node,
111 const std::function<std::string(const std::string&)>& loader,
112 const std::function<int64_t(const std::string&)>& mtimeChecker) {
113 // BUG-M4 fix: 所有缓存/去重 map 键用 cacheKey（Windows 小写折叠），loader/mtimeChecker
114 // 与错误消息用 loaderPath（保留大小写，供大小写敏感的 loader 正确解析）。
115 // 检查缓存
116 auto cacheIt = moduleCache_.find(cacheKey);
117 std::shared_ptr<Environment> moduleEnv;
118 // P2-14 循环导入延迟加载：缓存命中时区分完整/部分加载
119 if (cacheIt != moduleCache_.end() && moduleLoadingSet_.count(cacheKey) > 0) {
120 // 部分加载（循环导入）：模块正在加载中，返回已创建的部分 env
121 // 不检查 mtime（模块还在加载中，文件未被修改）
122 // 模块环境可能尚未填充所有导出名称——访问未定义名称时由 Environment::get
123 // 抛"未定义变量"错误（保持现有语义），访问已定义名称返回当前值
124 return cacheIt->second;
125 }
126 if (cacheIt != moduleCache_.end()) {
127 // 完整缓存命中：检查文件 mtime 是否变化，若变化则缓存失效
128 if (mtimeChecker) {
129 int64_t currentMtime = mtimeChecker(loaderPath);
130 auto mtimeIt = moduleMtimes_.find(cacheKey);
131 if (mtimeIt != moduleMtimes_.end() && mtimeIt->second != currentMtime) {
132 // 文件已修改，缓存失效
133 moduleCache_.erase(cacheIt);
134 moduleExports_.erase(cacheKey);
135 moduleMtimes_.erase(cacheKey);
136 cacheIt = moduleCache_.end(); // 标记为未命中
137 }
138 }
139 if (cacheIt != moduleCache_.end()) {
140 return cacheIt->second;
141 }
142 }
143
144 // 加载模块源码（A6 fix: 使用已拷贝的 loader，避免跨线程数据竞争）
145 std::string source = loader(loaderPath);
146 // 空源码视为加载失败（模块不存在或 0 字节文件）。production loader
147 // （IdeController/WorkerManager）在文件不存在/无法打开时均返回 ""，
148 // 无法与真正的 0 字节文件区分；对空模块报错可捕获 import 笔误，
149 // 与 Compiler/IR 路径行为一致（三后端统一）。
150 if (source.empty()) {
151 runtimeError("无法加载模块: " + loaderPath, node.line, node.column);
152 }
153
154 // 词法分析 + 语法分析
155 Lexer lexer;
156 auto tokens = lexer.scan(source);
157 // AUDIT-R5 BUG-03 fix: 检查词法错误（对齐 Compiler::loadAndParseModule 的 BUG-FIX）。
158 // 原实现吞掉 Lexer 诊断，非法 token 被丢弃后残缺模块被静默执行。
159 if (lexer.getDiagnostics().hasErrors()) {
160 const auto& diags = lexer.getDiagnostics().all();
161 std::string detail = diags.empty() ? "" : (": " + diags.front().message);
162 runtimeError("模块 " + loaderPath + " 词法错误" + detail, node.line, node.column);
163 }
164 Parser parser;
165 auto ast = parser.parse(tokens);
166 // AUDIT-R5 BUG-03 fix: 检查可恢复语法错误（parser.hasErrors()）——原实现仅检查
167 // AST 非空，Parser 错误恢复后返回的残缺 AST 会被当作成功加载静默执行，
168 // 与 VM/IR 路径（loadAndParseModule 报“模块语法错误”）不一致。
169 if (!ast || parser.hasErrors()) {
170 const auto& diags = parser.getDiagnostics().all();
171 std::string detail = diags.empty() ? "" : (": " + diags.front().message);
172 runtimeError("模块 " + loaderPath + " 语法错误" + detail, node.line, node.column);
173 }
174
175 // P1-2 fix: 使用独立的空父环境，避免模块访问导入方的全局变量
176 // 模块环境隔离：模块只能访问自身定义和导出的名称，不能读写导入方全局变量
177 auto moduleParentEnv = std::make_shared<Environment>(nullptr);
178 moduleEnv = std::make_shared<Environment>(moduleParentEnv);
179
180 // P2-14 循环导入延迟加载：立即入缓存，支持循环回路闭合时返回部分 env
181 // 模块语句执行完成后 env 已被填充，cache 指向同一对象无需重新赋值
182 moduleCache_[cacheKey] = moduleEnv;
183
184 // P2-14 预扫描 export 名到 moduleExports_，让循环导入命中时能验证具名导入
185 // 对齐 VM/IR 路径的 collectModuleExports（预扫描 ExportStmt 包装的声明名）
186 // exportedNames_ 仍按原机制在 visitExportStmt 中填充，加载完成后覆盖此预扫描值
187 {
188 std::unordered_set<std::string> prescanExports;
189 for (auto& stmt : ast->statements) {
190 if (!stmt || stmt->nodeType != NodeType::NODE_EXPORT_STMT)
191 continue;
192 auto* exp = static_cast<ExportStmt*>(stmt.get());
193 if (!exp->declaration)
194 continue;
195 ASTNode* decl = exp->declaration.get();
196 switch (decl->nodeType) {
197 case NodeType::NODE_VAR_DECL:
198 prescanExports.insert(static_cast<VarDecl*>(decl)->name);
199 break;
200 case NodeType::NODE_CLASS_DECL:
201 prescanExports.insert(static_cast<ClassDecl*>(decl)->name);
202 break;
203 case NodeType::NODE_FUN_DECL:
204 prescanExports.insert(static_cast<FunDecl*>(decl)->name);
205 break;
206 default:
207 break;
208 }
209 }
210 moduleExports_[cacheKey] = std::move(prescanExports);
211 }
212
213 auto savedEnv = currentEnv_;
214 auto savedExported = exportedNames_;
215 exportedNames_.clear();
216 currentEnv_ = moduleEnv;
217 moduleLoadingStack_.push_back(cacheKey);
218 moduleLoadingSet_.insert(cacheKey); // D19 fix: 与 stack 同步维护 set
219
220 // RA-A fix: RAII 守卫统一管理异常路径下的状态恢复（moduleEnv close、env、exported、loadingStack/Set），
221 // 消除原 catch(...) + throw; 的 rethrow。正常路径通过 dismiss 跳过守卫清理。
222 // P2-14: 扩展守卫，异常路径清除 moduleCache_/moduleExports_ 中的部分 env，
223 // 避免后续 import 命中缓存返回未填充完整的 env
224 struct ModuleEnvGuard {
225 Interpreter& interp;
226 std::shared_ptr<Environment>& env;
227 std::shared_ptr<Environment>& saved;
228 std::unordered_set<std::string>& exported;
229 std::unordered_set<std::string> savedExported;
230 std::vector<std::string>& loadingStack;
231 std::unordered_set<std::string>& loadingSet;
232 std::unordered_map<std::string, std::shared_ptr<Environment>>& cache;
233 std::unordered_map<std::string, std::unordered_set<std::string>>& exportsCache;
234 const std::string& cacheKey;
235 bool dismissed = false;
236 ~ModuleEnvGuard() {
237 if (!dismissed) {
238 // AUDIT-BUG-F8 fix: 异常路径必须调用 closeCapturedVariables，将模块内闭包的
239 // open upvalues 关闭为最终值快照。原实现仅恢复 env/exported/loadingStack，
240 // moduleEnv 即将析构，逃逸闭包（经 throw 逃出模块）的 capturedVars 保持初始值。
241 env->closeCapturedVariables();
242 interp.currentEnv_ = saved;
243 exported = std::move(savedExported);
244 loadingStack.pop_back();
245 loadingSet.erase(cacheKey); // D19 fix: 同步移除
246 // P2-14: 清除部分 env 和预扫描的 export 名
247 cache.erase(cacheKey);
248 exportsCache.erase(cacheKey);
249 }
250 }
251 } envGuard{
252 *this, moduleEnv, savedEnv, exportedNames_, savedExported, moduleLoadingStack_, moduleLoadingSet_,
253 moduleCache_, moduleExports_, cacheKey};
254
255 for (auto& stmt : ast->statements) {
256 evaluate(stmt.get());
257 }
258 // RA-A fix: 不再需要 catch(...) + throw; — envGuard 析构统一恢复
259
260 // RA-A fix: 正常路径，dismiss 守卫后手动执行完整清理
261 envGuard.dismissed = true;
262 moduleLoadingStack_.pop_back();
263 moduleLoadingSet_.erase(cacheKey); // D19 fix: 同步移除
264 currentEnv_ = savedEnv;
265
266 // P2-14: moduleCache_[cacheKey] 已在创建时入缓存，无需重新赋值
267 // 用实际执行的 export 名覆盖预扫描值（处理条件分支 export 等动态场景）
268 moduleExports_[cacheKey] = exportedNames_;
269 // BUG-REPL-AUDIT-1 fix: 记录模块文件 mtime，下次 import 时检查是否变化
270 if (mtimeChecker) {
271 moduleMtimes_[cacheKey] = mtimeChecker(loaderPath);
272 }
273 exportedNames_ = savedExported;
274
275 // 保留模块 AST（确保函数/类定义指针有效）
276 replAsts_.push_back(std::move(ast));
277
278 return moduleEnv;
279}

281void Interpreter::importNamesFromModule(const std::string& cacheKey, ImportStmt& node,
282 std::shared_ptr<Environment>& moduleEnv) {
283 // 导入名称到当前环境（仅导入已 export 的名称）
284 // BUG-M4 fix: moduleExports_ 用 cacheKey（与 loadModuleOrGetCached 入键一致）。
285 auto& exports = moduleExports_[cacheKey];
286 if (!node.namespaceAlias.empty()) {
287 // P2-11: import * as ns from "path" — 构造包含所有导出名称的字典对象
288 // 使用 Value::DictMap（R97 #2: 含 DictKeyEqual 透明比较器）构造，
289 // 再通过 Value(DictMap&&) 构造堆字典值。键为导出名（string）。
290 Value::DictMap dict;
291 dict.reserve(exports.size());
292 for (const auto& name : exports) {
293 const Value* valPtr = moduleEnv->get(name);
294 if (valPtr) {
295 dict.emplace(std::string(name), *valPtr);
296 }
297 }
298 currentEnv_->define(node.namespaceAlias, Value(std::move(dict)));
299 } else if (node.importAll) {
300 // 导入全部导出名称
301 for (const auto& name : exports) {
302 const Value* valPtr = moduleEnv->get(name);
303 if (valPtr) {
304 currentEnv_->define(name, *valPtr);
305 }
306 }
307 } else {
308 // P2-1 fix: 原子性导入 — 先验证所有名称都存在，再统一定义
309 // 避免部分名称已导入后遇到错误名称导致环境不一致
310 for (const auto& name : node.names) {
311 if (exports.find(name) == exports.end()) {
312 runtimeError("模块 " + cacheKey + " 中未导出名称: " + name, node.line, node.column);
313 }
314 const Value* valPtr = moduleEnv->get(name);
315 if (!valPtr) {
316 runtimeError("模块 " + cacheKey + " 中未找到导出名称: " + name, node.line, node.column);
317 }
318 }
319 // 全部验证通过后，统一导入
320 for (const auto& name : node.names) {
321 const Value* valPtr = moduleEnv->get(name);
322 currentEnv_->define(name, *valPtr);
323 }
324 }
325}
