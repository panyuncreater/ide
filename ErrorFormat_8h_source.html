46/// @brief 将整数写入栈缓冲并返回字符串（无堆分配）。
47///
48/// 替代 std::to_string，避免 MSVC 上的 locale 查询开销。
49/// 使用 std::to_chars（C++17）实现，零堆分配。
50///
51/// @tparam T 整型类型（int / int64_t / size_t / 等）
52/// @param value 待转换的整数值
53/// @return 转换后的 std::string
54template <typename T, typename = std::enable_if_t<std::is_integral_v<T>>> inline std::string intToString(T value) {
55 char buf[32];
56 auto res = std::to_chars(buf, buf + sizeof(buf), value);
57 return std::string(buf, res.ptr);
58}

60/// @brief 格式化错误消息（栈缓冲，无 std::to_string 调用）。
61///
62/// P3 fix: 替代 "msg" + std::to_string(n) + "msg" 模式，避免多次堆分配。
63/// 使用 snprintf 到 512 字节栈缓冲，足以容纳绝大多数运行时错误消息。
64///
65/// 支持的占位符：
66/// - `%d` - int / int64_t / size_t（整型）
67/// - `%s` - const char* / std::string
68/// - `%zu` - size_t（显式）
69///
70/// @param fmt printf 风格格式字符串
71/// @param args 格式化参数（须与 fmt 中的占位符匹配）
72/// @return 格式化后的 std::string；格式化失败时回退返回 fmt 原文
73/// @note 消息超过 512 字节时截断为 511 字节（保证 null 终止）
74template <typename... Args> inline std::string format(const char* fmt, Args&&... args) {
75 // 简单实现：用 snprintf 到栈缓冲，避免 std::to_string 的 locale 开销。
76 // 缓冲区 512 字节足以容纳绝大多数运行时错误消息。
77 char buf[512];
78 int n = std::snprintf(buf, sizeof(buf), fmt, std::forward<Args>(args)...);
79 if (n < 0)
80 return std::string(fmt); // 格式化失败，回退原始字符串
81 if (static_cast<size_t>(n) < sizeof(buf)) {
82 return std::string(buf, static_cast<size_t>(n));
83 }
84 // 截断（消息过长）：返回 512 字节
85 return std::string(buf, sizeof(buf) - 1);
86}

89// P3-16 fix: formatStd — std::format 风格的类型安全格式化（推荐）
90// ------------------------------------------------------------
91// 与 format（printf 风格）对比：
92// - 类型安全：编译期检查参数与占位符匹配，避免 %d/%s 误用导致 UB
93// - 占位符：{} 而非 %d/%s/%zu，无需为不同整型选择正确说明符
94// - std::string 参数无需 .c_str()
95// - 性能：std::format 内部使用编译期格式串解析 + 栈缓冲，性能与 snprintf 相当
96//
97// 迁移策略：
98// - 新代码应使用 formatStd + ErrorMessages 中 *FmtStd 常量
99// - 老 format 调用可逐步迁移（不强制），保持两套 API 共存
100// - CrashHandler.cpp 保留 snprintf（异步信号安全约束）
101//
102// 用法：
103// ErrorFormat::formatStd("数组索引越界: {}, 有效范围 [0, {})", idx, size);
104// ErrorFormat::formatStd("未定义的函数: {}", funName); // funName 可为 std::string
105// ============================================================
106template <typename... Args> inline std::string formatStd(std::format_string<Args...> fmt, Args&&... args) {
107 return std::format(fmt, std::forward<Args>(args)...);
108}

117/// @brief 为错误消息附加行号/列号定位信息。
118///
119/// 输出格式：`<msg> (行 <line>, 列 <col>)`
120/// 用于 runtimeError 等需要源码定位的错误消息构造。
121///
122/// @param msg 原始错误消息
123/// @param line 源码行号
124/// @param col 源码列号
125/// @return 附加定位信息后的错误消息
126inline std::string formatWithLocation(const std::string& msg, int line, int col) {
127 // 预估容量：msg + " (行 " + 11 位 int + ", 列 " + 11 位 int + ")"
128 std::string result;
129 result.reserve(msg.size() + 32);
130 result = msg;
131 result += " (行 ";
132 std::array<char, 24> buf{};
133 auto r1 = std::to_chars(buf.data(), buf.data() + buf.size(), line);
134 result.append(buf.data(), r1.ptr);
135 result += ", 列 ";
136 auto r2 = std::to_chars(buf.data(), buf.data() + buf.size(), col);
137 result.append(buf.data(), r2.ptr);
138 result += ')';
139 return result;
140}
