3// ============================================================
4// RegisterBytecode.h — PERF-14: 寄存器式字节码 ISA
5// ------------------------------------------------------------
6// 与现有栈式 BytecodeChunk 并存，通过 RegisterVM 执行。
7// IR 三地址码（dest, src1, src2）直接 lowering 为寄存器式指令，
8// 无需栈式序列化（LOAD_CONST × 2 + ADD → REG_ADD r1, r2, r3）。
9//
10// 寄存器模型：
11// - 每个调用帧 32 个虚拟寄存器（R0-R31），8 位编码
12// - R0..R(arity-1)：参数寄存器
13// - R(arity)..R(localCount-1)：局部变量寄存器
14// - R(localCount)..R31：临时寄存器（IR vreg 分配）
15// - 寄存器窗口大小 = max(localCount, nextVReg)，上限 32
16//
17// 指令编码（变长但紧凑）：
18// 1B op + 按需操作数（每寄存器 1B，常量索引/slot 2B）
19// 例：REG_ADD dst(1B) src1(1B) src2(1B) = 4B
20// REG_LOAD_CONST dst(1B) constIdx(2B) = 4B
21// REG_RETURN src(1B) = 2B
22// ============================================================
23

233 /// L1 fix: 按 ip 反查 reg 对应的变量名。优先在 slotNameRanges 中查找
234 /// startIp <= ip < endIp && slot == N 的 range；未命中则回退到 localRegNames。
235 const std::string& resolveSlotName(size_t slot, size_t ip) const {
236 for (const auto& range : slotNameRanges) {
237 if (range.slot == slot && range.startIp <= ip && ip < range.endIp) {
238 return range.name;
239 }
240 }
241 if (slot < localRegNames.size()) {
242 return localRegNames[slot];
243 }
244 static const std::string empty;
245 return empty;
246 }
