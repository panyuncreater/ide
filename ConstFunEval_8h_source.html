3// ============================================================
4// ConstFunEval.h — L18 lang-constfun: const fun 编译期沙箱求值
5// ------------------------------------------------------------
6// 三后端共享的编译期求值 helper（header-only，Compiler / AstIRBuilder
7// 在 visitFunCall 折叠点调用；Interpreter 运行时直接调用语义等价，无需折叠）。
8//
9// 折叠条件（全部满足）：
10// 1. 直接名称调用 f(args)（callee == nullptr）且 f 是已声明的 const fun
11// 2. 实参全部为字面量（number/string/bool/null）
12// 3. arity 在 [requiredParamCount, params.size()] 范围内
13// 4. 沙箱求值成功且无副作用观测（无 output/input、无运行时错误/异常）
14// 5. 结果为原始类型（int/float/bool/string/null）
15// 任一条件不满足 → 返回 nullopt，调用方回退为普通运行时调用（语义安全）。
16//
17// 副作用安全性：
18// - 沙箱是独立 Interpreter 实例，print/input 回调仅置 impure 标志
19// - 引用全局变量/未注册函数 → 沙箱内未定义 → 运行时错误 → 回退
20// - 死循环/深递归 → 沙箱内建迭代/递归上限报错 → 回退
21// 已知限制（v1）：clock()/随机类内建在沙箱中会被固化为编译期值——
22// 与 C++ constexpr 的确定性约定一致，const fun 用户意图即编译期求值。
23// ============================================================

35/// 判断 AST 节点是否为可编译期求值的字面量
36inline bool isLiteralNode(const ASTNode* n) {
37 if (!n)
38 return false;
39 switch (n->nodeType) {
40 case NodeType::NODE_NUMBER_LITERAL:
41 case NodeType::NODE_STRING_LITERAL:
42 case NodeType::NODE_BOOL_LITERAL:
43 case NodeType::NODE_NULL_LITERAL:
44 return true;
45 default:
46 return false;
47 }
48}

50/// 尝试编译期求值 const fun 调用。成功返回结果 Value，失败返回 nullopt（回退运行时）。
51/// @param constFuns 已声明的 const fun 注册表（name → 非拥有 FunDecl 指针，AST 生命周期内有效）
52/// @param call 调用点 FunCall 节点
53inline std::optional<Value> tryEvaluate(const std::unordered_map<std::string, FunDecl*>& constFuns, FunCall& call) {
54 if (call.callee != nullptr)
55 return std::nullopt;
56 auto it = constFuns.find(call.name);
57 if (it == constFuns.end())
58 return std::nullopt;
59 FunDecl* decl = it->second;
60 if (!decl || decl->isGenerator)
61 return std::nullopt;
62 if (call.arguments.size() < static_cast<size_t>(decl->requiredParamCount) ||
63 call.arguments.size() > decl->params.size())
64 return std::nullopt;
65 for (auto& a : call.arguments) {
66 if (!isLiteralNode(a.get()))
67 return std::nullopt;
68 }
69
70 try {
71 Interpreter sandbox;
72 bool impure = false;
73 sandbox.setOutputCallback([&](const std::string&) { impure = true; });
74 sandbox.setInputCallback([&](const std::string&) -> std::string {
75 impure = true;
76 return "";
77 });
78 // 注册全部 const fun（互相调用支持）：非拥有 shared_ptr 别名，
79 // AST 在编译期间存活，沙箱生命周期短于 AST，安全。
80 std::vector<std::shared_ptr<ASTNode>> decls;
81 decls.reserve(constFuns.size());
82 for (const auto& kv : constFuns) {
83 decls.push_back(std::shared_ptr<ASTNode>(kv.second, [](ASTNode*) {}));
84 }
85 Block declBlock(std::move(decls), call.line, call.column);
86 sandbox.execute(declBlock);
87 if (sandbox.getDiagnostics().hasErrors() || impure)
88 return std::nullopt;
89 Value result = sandbox.evaluateExpr(&call);
90 if (sandbox.getDiagnostics().hasErrors() || impure)
91 return std::nullopt;
92 // 仅折叠原始类型（容器/闭包/实例不折叠——常量池语义与共享性复杂）
93 if (!(result.isInt() || result.isFloat() || result.isBool() || result.isString() || result.isNull()))
94 return std::nullopt;
95 return result;
96 } catch (...) {
97 // 沙箱内任何异常（运行时错误/递归超限/迭代超限）→ 回退运行时调用
98 return std::nullopt;
99 }
100}
