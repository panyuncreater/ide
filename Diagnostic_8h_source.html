3 * @brief 统一诊断体系（错误/警告/信息/建议）。
4 *
5 * 为 Lexer、Parser、Compiler、Interpreter、VM、Formatter、IDE、TypeChecker
6 * 等所有模块提供统一的诊断报告格式，便于 IDE 集中展示和处理。
7 *
8 * 核心类型：
9 * - DiagLevel：诊断严重级别（Error/Warning/Info/Hint）
10 * - DiagSource：诊断来源模块标识
11 * - Diagnostic：单条诊断信息，含 format() 格式化输出
12 * - DiagnosticBag：诊断收集器，O(1) 错误/警告计数 + errorLines() 编辑器高亮
13 *
14 * 线程安全性：DiagnosticBag 内部使用 std::vector，**非线程安全**。
15 * 多线程场景需通过外层互斥锁保护（如 IdeController::diagnosticsMutex_）。
16 *
17 * @see IBackend::getDiagnostics
18 */

54/// 单条诊断信息
55struct Diagnostic {
56 DiagLevel level;
57 std::string message;
58 int line;
59 int column;
60 DiagSource source;
61 /// P2 fix (错误码优先匹配): 稳定诊断码，如 "missing-semicolon" / "undefined-variable"。
62 /// 与 ErrorHintEngine::errorPatterns() 表中的 tag 对应。空字符串表示未设置——
63 /// ErrorHintEngine 会回退到现有的中/英子串匹配兜底逻辑。
64 /// 引擎模块逐步迁移到带 code 的 Diagnostic 构造，未迁移时行为不变（兼容性保证）。
65 std::string code;
66 /// P2-12 fix (错误恢复): 是否为致命错误——致命错误阻止错误恢复继续编译，
67 /// 可恢复错误（isFatal=false）允许编译器在记录后跳过当前节点继续处理后续兄弟节点。
68 /// 仅在 DiagLevel::Error 级别下区分 Fatal/Recoverable；Warning/Info/Hint 恒为非致命。
69 /// 默认 false（向后兼容：现有 addError 调用视为可恢复）。
70 bool isFatal = false;
71
72 Diagnostic(DiagLevel lv, const std::string& msg, int ln, int col, DiagSource src)
73 : level(lv), message(msg), line(ln), column(col), source(src) {}

134 /// 格式化为 IDE 输出面板显示文本
135 /// 格式：[来源] 级别 (行 X, 列 Y): 消息
136 std::string format() const {
137 std::string result = "[" + sourceString() + "] " + levelString();
138 if (line > 0) {
139 result += " (行 " + std::to_string(line);
140 if (column > 0) {

143 result += ")";

150/// 诊断收集器：聚合来自多个模块的诊断信息
151class DiagnosticBag {
152public:
153 /// 添加一条诊断
154 void add(const Diagnostic& diag) {
155 diagnostics_.push_back(diag);
156 // P2 fix: 维护计数器实现 O(1) 查询
157 if (diag.isError())
158 ++errorCount_;
159 else if (diag.isWarning())
160 ++warningCount_;
161 }
162
163 /// 便捷方法：添加错误
164 void addError(const std::string& msg, int line, int col, DiagSource src) {
165 diagnostics_.emplace_back(DiagLevel::Error, msg, line, col, src);
166 ++errorCount_; // P2 fix: O(1) 计数
167 }
168
169 /// P2 fix: 添加带 code 的错误
170 void addError(const std::string& msg, int line, int col, DiagSource src, const std::string& diagCode) {
171 diagnostics_.emplace_back(DiagLevel::Error, msg, line, col, src, diagCode);
172 ++errorCount_;
173 }
174
175 /// P2-12 fix (错误恢复): 添加致命错误——致命错误阻止错误恢复继续编译。
176 /// 致命错误场景：模块加载失败、内部不变量违反（未支持的 AST 节点/BinOp）、
177 /// 父类未定义导致字段索引全错等。可恢复错误用 addError 添加（默认 isFatal=false）。
178 void addErrorFatal(const std::string& msg, int line, int col, DiagSource src, const std::string& diagCode = "") {
179 diagnostics_.emplace_back(DiagLevel::Error, msg, line, col, src, diagCode, /*isFatal=*/true);
180 ++errorCount_;
181 ++fatalErrorCount_;
182 }
183
184 /// 便捷方法：添加警告
185 void addWarning(const std::string& msg, int line, int col, DiagSource src) {
186 diagnostics_.emplace_back(DiagLevel::Warning, msg, line, col, src);
187 ++warningCount_; // P2 fix: O(1) 计数
188 }
189
190 /// P2 fix: 添加带 code 的警告
191 void addWarning(const std::string& msg, int line, int col, DiagSource src, const std::string& diagCode) {
192 diagnostics_.emplace_back(DiagLevel::Warning, msg, line, col, src, diagCode);
193 ++warningCount_;
194 }
195
196 /// 便捷方法：添加信息
197 void addInfo(const std::string& msg, int line, int col, DiagSource src) {
198 diagnostics_.emplace_back(DiagLevel::Info, msg, line, col, src);
199 }
200
201 /// P2 fix: 添加带 code 的信息
202 void addInfo(const std::string& msg, int line, int col, DiagSource src, const std::string& diagCode) {
203 diagnostics_.emplace_back(DiagLevel::Info, msg, line, col, src, diagCode);
204 }
205
206 /// 是否有错误
207 bool hasErrors() const { return errorCount_ > 0; }
208
209 /// P2-12 fix (错误恢复): 是否有致命错误——编译器仅对致命错误中止，
210 /// 可恢复错误记录后跳过当前节点继续处理后续兄弟节点。
211 bool hasFatalErrors() const { return fatalErrorCount_ > 0; }
212
213 /// 是否有警告
214 bool hasWarnings() const { return warningCount_ > 0; }
215
216 /// 错误数量
217 int errorCount() const { return errorCount_; }
218
219 /// 警告数量
220 int warningCount() const { return warningCount_; }
221
222 /// 获取所有诊断
223 const std::vector<Diagnostic>& all() const { return diagnostics_; }
224
225 /// 获取错误行号集合（用于编辑器标记）
226 std::vector<int> errorLines() const {
227 std::vector<int> lines;
228 for (const auto& d : diagnostics_) {
229 if (d.isError() && d.line > 0) {
230 lines.push_back(d.line);
231 }
232 }
233 std::sort(lines.begin(), lines.end());
234 lines.erase(std::unique(lines.begin(), lines.end()), lines.end());
235 return lines;
236 }

238 /// 清空所有诊断
239 void clear() {
240 diagnostics_.clear();
241 errorCount_ = 0;
242 warningCount_ = 0;
243 fatalErrorCount_ = 0; // P2-12 fix
244 }
245
246 /// 诊断数量
247 size_t size() const { return diagnostics_.size(); }
248
249 /// 是否为空
250 bool empty() const { return diagnostics_.empty(); }
251
252 /// 按行号排序（同文件内的诊断按位置排列）
253 void sortByLocation() {
254 std::stable_sort(diagnostics_.begin(), diagnostics_.end(), [](const Diagnostic& a, const Diagnostic& b) {
255 if (a.line != b.line)
256 return a.line < b.line;
257 return a.column < b.column;
258 });
259 }
260
261 /// 获取摘要文本（如 "3 个错误, 1 个警告"）
262 std::string summary() const {
263 int errs = errorCount();
264 int warns = warningCount();
265 std::string result;
266 if (errs > 0)
267 result += std::to_string(errs) + " 个错误";
268 if (warns > 0) {
269 if (!result.empty())
270 result += ", ";
271 result += std::to_string(warns) + " 个警告";
272 }
273 if (result.empty())
274 result = "无错误";
275 return result;
276 }

282 int fatalErrorCount_ = 0; // P2-12 fix: O(1) 致命错误计数（错误恢复用）
283};
