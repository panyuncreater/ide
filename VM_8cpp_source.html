17// ============================================================
18// VM 虚拟机实现（栈式后端）
19// ------------------------------------------------------------
20// 本文件实现 MiniLang 的"栈式"执行后端（与 RegisterVM 语义等价）。
21// 核心数据模型：
22// · 操作数栈 stack_：所有值运算的结果都压入此栈，指令从栈顶取/放操作数。
23// · 调用帧栈 frames_：每次函数调用压入一帧，帧记录返回地址(returnIp)、
24// 本帧在 stack_ 中的基准(basePointer)、当前指令指针(ip)、所属字节码块(chunk)。
25// · basePointer 机制：被调用者的局部变量存放在 stack_[basePointer+slot]，
26// 调用返回时只需 stack_.resize(basePointer) 即可一次性回收全部临时/局部槽。
27// · 指令分发：executeOneInstruction 先按"指令类别"两级 switch 转发到
28// execute*Ops，再由具体 opcode 处理（两级拆分仅为可维护性，不改变语义）。
29// · OpCode 编码：定长操作数多为 16 位小端索引（idx = code[ip+1] | code[ip+2]<<8），
30// 指向常量池 constants_；局部变量槽为 8 位；全局变量既有名称寻址(globals_)
31// 也有整数槽位寻址(globalSlots_)两条快速路径。
32// 本文件不含函数调用/容器/杂项指令的具体实现——它们分别在 VMCalls.cpp、
33// VMContainers.cpp（按 execute*Ops 方法归属拆分，便于并行维护）。
34// ============================================================

167VMResult VM::throwException(Value thrownValue) {
168 // F11: 异常处理 — 搜索 try 处理器或跨帧传播
169 // AUDIT-P1.1 fix: 异常传播中断 break/continue 续跳链，清空 pendingJumpStack_。
170 // 否则 finally 内部 throw 异常后，外层 finally 的 OP_FINALLY_END 会错误跳转到
171 // 已被中断的 break/continue 目标（异常优先于控制流转移）。
172 pendingJumpStack_.clear();
173 while (true) {
174 size_t currentFrameIdx = frames_.size() - 1;
175
176 // 从 tryStack_ 顶部搜索当前帧的处理器
177 // tryStack_ 是栈结构，顶部可能是当前帧或更内层帧的处理器
178 while (!tryStack_.empty()) {
179 const auto& handler = tryStack_.back();
180 if (handler.frameIndex == currentFrameIdx) {
181 // 找到 catch 处理器：截断栈、推入异常值、跳转到 catch
182 // P0-5 fix: 截断栈前关闭指向被截断栈槽的 open upvalues，防止悬垂指针
183 if (handler.stackBase <= stack_.size()) {
184 // AUDIT-P2.5 fix: closeUpvaluesFrom 返回错误时立即终止异常处理，避免在损坏状态上继续
185 if (closeUpvaluesFrom(handler.stackBase) != VMResult::VM_OK)
186 return VMResult::VM_RUNTIME_ERROR;
187 stack_.resize(handler.stackBase);
188 }
189 push(std::move(thrownValue));
190 currentFrame().ip = handler.catchIp;
191 tryStack_.pop_back();
192 // L14: 返回 VM_EXCEPTION_THROW 而非 VM_OK，通知调用方 ip 已被设置为 catchIp，
193 // 不应再递增 ip。dispatch 循环将此结果视为"异常已捕获，继续执行"。
194 return VMResult::VM_EXCEPTION_THROW;
195 }
196 if (handler.frameIndex < currentFrameIdx) {
197 // 处理器在外层帧 — 当前帧无处理器，需弹出当前帧
198 break;
199 }
200 // handler.frameIndex > currentFrameIdx: 内层帧残留处理器，弹出
201 tryStack_.pop_back();
202 }
203
204 // 当前帧无处理器 — 弹出帧，传播到调用者
205 if (frames_.size() <= 1) {
206 // 无更多帧 — 未捕获的异常
207 // P2-5 fix: 限制异常值 toString 长度，防止循环引用对象导致超长输出
208 std::string str = thrownValue.toString();
209 ErrorFormat::truncateForError(str);
210 return runtimeError("未捕获的异常: " + str);
211 }
212
213 size_t savedReturnIp = frames_.back().returnIp;
214 size_t savedBp = frames_.back().basePointer;
215 size_t poppedFrameIdx = frames_.size() - 1;
216 // P0-5 fix: 弹帧前关闭该帧关联的 open upvalues
217 // AUDIT-P2.5 fix: closeUpvaluesFrom 返回错误时立即终止异常展开，避免在损坏状态上继续
218 if (closeUpvaluesFrom(savedBp) != VMResult::VM_OK)
219 return VMResult::VM_RUNTIME_ERROR;
220 frames_.pop_back();
221 // Bug3 fix: 清理属于被弹出帧的 tryStack_ handler（与 OP_RETURN 保持一致）
222 while (!tryStack_.empty() && tryStack_.back().frameIndex >= poppedFrameIdx) {
223 tryStack_.pop_back();
224 }
225
226 // 截断栈到调用者的 basePointer
227 if (savedBp <= stack_.size()) {
228 stack_.resize(savedBp);
229 }
230
231 // 设置调用者的 ip，继续在调用者帧中搜索
232 currentFrame().ip = savedReturnIp;
233 // 循环回到顶部，在调用者帧中搜索 tryStack_
234 }
235}

237VMResult VM::closeUpvaluesFrom(size_t fromSlot) {
238 // F11-fix: 关闭指向 [fromSlot, stack_.size()) 范围内栈槽的 open upvalues
239 // B5 fix: openUpvalues_ 现为按 stackSlot 排序的 multimap，lower_bound(fromSlot) 定位起始点，
240 // 一次性关闭并擦除 [fromSlot, ∞) 全部条目，复杂度从 O(n) 降为 O(log n + k)。
241 // AUDIT-P2.5 fix: 返回类型 void→VMResult，调用方可 fail-fast 传播错误。
242 bool hadError = false;
243 auto it = openUpvalues_.lower_bound(fromSlot);
244 while (it != openUpvalues_.end()) {
245 if (auto uv = it->second.lock()) { // closure 仍持有强引用则有效
246 if (!uv->isClosed) {
247 if (uv->stackSlot < stack_.size()) {
248 uv->value = stack_[uv->stackSlot];
249 } else {
250 // AUDIT-P2.4 fix: slot 越界改为 runtimeError 以 fail-fast 暴露问题。
251 // 原实现仅打 Warning 继续执行，闭包静默捕获 null 难以排查根因。
252 // runtimeError 设置 hasError_=true，VM 在后续指令分发终止执行；
253 // 仍执行 isClosed=true + erase 以关闭 upvalue（保留默认 null 值），防止悬垂引用。
254 // AUDIT-P2.5 fix: 记录错误标志，函数末尾返回 VM_RUNTIME_ERROR，调用方立即 return。
255 (void)runtimeError("closeUpvaluesFrom: slot " + std::to_string(uv->stackSlot) + " >= stack size " +
256 std::to_string(stack_.size()));
257 hadError = true;
258 }
259 uv->isClosed = true;
260 }
261 }
262 it = openUpvalues_.erase(it);
263 }
264 return hadError ? VMResult::VM_RUNTIME_ERROR : VMResult::VM_OK;
265}

474 break;
475 }
476 // ---- 局部变量写入 ----
477 case OpCode::OP_SET_LOCAL: {
478 if (ip + 1 >= code.size())
479 break;
480 wt.isWrite = true;
481 wt.localSlot = code[ip + 1];
482 wt.varName = slotToLocalName(wt.localSlot);
483 break;
484 }
485 // ---- upvalue 写入 ----
486 case OpCode::OP_SET_UPVALUE: {
487 if (ip + 1 >= code.size())
488 break;
489 wt.isWrite = true;
490 // upvalue 名称通过 chunk.upvalues[uvIdx].name 反查（R75 条件断点修复已记录）
491 uint8_t uvIdx = code[ip + 1];
492 if (uvIdx < frame.chunk->upvalues.size()) {
493 wt.varName = frame.chunk->upvalues[uvIdx].name;
494 }
495 break;
496 }
497 // ---- 字段写入（仅 fieldNameIdx）----
498 case OpCode::OP_MEMBER_SET:
499 case OpCode::OP_INIT_FIELD: {
500 if (ip + 2 >= code.size())
501 break;
502 wt.isWrite = true;
503 wt.isFieldWrite = true;
504 wt.fieldName = constStr(readShort(ip + 1));
505 break;
506 }
507 // ---- 字段写入（varIdx + fieldIdx）----
508 case OpCode::OP_MEMBER_SET_VAR:
509 case OpCode::OP_WRITEBACK_MEMBER_VAR: {
510 if (ip + 4 >= code.size())
511 break;
512 wt.isWrite = true;
513 wt.isFieldWrite = true;
514 wt.varName = constStr(readShort(ip + 1));
515 wt.fieldName = constStr(readShort(ip + 3));
516 break;
517 }
518 // ---- 字段写入（slot + fieldIdx）----
519 case OpCode::OP_MEMBER_SET_LOCAL:
520 case OpCode::OP_WRITEBACK_MEMBER_LOCAL: {
521 if (ip + 3 >= code.size())
522 break;
523 wt.isWrite = true;
524 wt.isFieldWrite = true;
525 wt.localSlot = code[ip + 1];
526 wt.varName = slotToLocalName(wt.localSlot);
527 wt.fieldName = constStr(readShort(ip + 2));
528 break;
529 }
530 // ---- 字段写入（uvIdx + fieldIdx）----
531 case OpCode::OP_WRITEBACK_MEMBER_UPVALUE: {
532 if (ip + 3 >= code.size())
533 break;
534 wt.isWrite = true;
535 wt.isFieldWrite = true;
536 uint8_t uvIdx = code[ip + 1];
537 if (uvIdx < frame.chunk->upvalues.size()) {
538 wt.varName = frame.chunk->upvalues[uvIdx].name;
539 }
540 wt.fieldName = constStr(readShort(ip + 2));
541 break;
542 }
543 // ---- 索引写入（无操作数，全在栈上）----
544 case OpCode::OP_INDEX_SET: {
545 wt.isWrite = true;
546 wt.isIndexWrite = true;
547 break;
548 }
549 // ---- 索引写入（varIdx）----
550 case OpCode::OP_INDEX_SET_VAR:
551 case OpCode::OP_WRITEBACK_INDEX_VAR: {
552 if (ip + 2 >= code.size())
553 break;
554 wt.isWrite = true;
555 wt.isIndexWrite = true;
556 wt.varName = constStr(readShort(ip + 1));
557 break;
558 }
559 // ---- 索引写入（slot）----
560 case OpCode::OP_INDEX_SET_LOCAL:
561 case OpCode::OP_WRITEBACK_INDEX_LOCAL: {
562 if (ip + 1 >= code.size())
563 break;
564 wt.isWrite = true;
565 wt.isIndexWrite = true;
566 wt.localSlot = code[ip + 1];
567 wt.varName = slotToLocalName(wt.localSlot);
568 break;
569 }
570 // ---- 索引写入（uvIdx）----
571 case OpCode::OP_WRITEBACK_INDEX_UPVALUE: {
572 if (ip + 1 >= code.size())
573 break;
574 wt.isWrite = true;
575 wt.isIndexWrite = true;
576 uint8_t uvIdx = code[ip + 1];
577 if (uvIdx < frame.chunk->upvalues.size()) {
578 wt.varName = frame.chunk->upvalues[uvIdx].name;
579 }
580 break;
581 }
582 default:
583 break;
584 }
585 return wt;
586}

791const BytecodeChunk* VM::findMethodChunk(const std::string& className, const std::string& methodName) const {
792 // P4 fix: 先查缓存
793 auto clsIt = classInfo_.find(className);
794 if (clsIt != classInfo_.end()) {
795 auto& cache = clsIt->second.methodCache;
796 auto cacheIt = cache.find(methodName);
797 if (cacheIt != cache.end()) {
798 return cacheIt->second; // 缓存命中（含 nullptr 表示方法不存在）
799 }
800 }
801
802 // #12 fix: 冷路径走两级 unordered_map 索引（methodsByClass_），消除每层
803 // 继承链 "Class.method" 字符串拼接。语义等价：自底向上沿继承链查找首个命中。
804 std::string cur = className;
805 for (int guard = 0; guard < MAX_INHERITANCE_DEPTH && !cur.empty(); ++guard) {
806 auto clsIdxIt = methodsByClass_.find(cur);
807 if (clsIdxIt != methodsByClass_.end()) {
808 auto mIt = clsIdxIt->second.find(methodName);
809 if (mIt != clsIdxIt->second.end()) {
810 // P4: 写入缓存
811 if (clsIt != classInfo_.end()) {
812 clsIt->second.methodCache[methodName] = mIt->second;
813 }
814 return mIt->second;
815 }
816 }
817 auto nextIt = classInfo_.find(cur);
818 if (nextIt == classInfo_.end())
819 break;
820 cur = nextIt->second.superClassName;
821 }
822 // P4: 缓存 nullptr（方法不存在），避免重复查找
823 if (clsIt != classInfo_.end()) {
824 clsIt->second.methodCache[methodName] = nullptr;
825 }
826 return nullptr;
827}

829// 数值运算类型枚举（避免字符串比较）
830enum { OP_ADD_INT = 0, OP_SUB_INT, OP_MUL_INT, OP_DIV_INT, OP_MOD_INT };
831
832// ============================================================
833// 拓展二期·语言：运算符重载（instance 算术 dunder 分派）
834// ------------------------------------------------------------
835// 栈布局 [left,right] 恰好就是方法调用的 [receiver,arg]，按
836// executeInstanceMethodCall 的帧协议注入新帧（returnIp=ip+1，
837// OP_ADD 系列为单字节指令），方法 OP_RETURN 时结果自然压栈——
838// 与“弹二压一”算术语义吹合。receiver 写回禁用（receiverVarName 空，
839// receiverLocalSlot=-1）：运算符方法内 this 字段变异不写回接收者，
840// 与 Interpreter 的左值拷贝语义一致。
841// ============================================================
842bool VM::tryOperatorOverload(size_t& ip, OpCode op, VMResult& outResult) {
843 if (stack_.size() < 2) {
844 return false; // 栈下溢由 numericOp 统一报错
845 }
846 const Value& left = stack_[stack_.size() - 2];
847 if (!left.isInstance()) {
848 return false;
849 }
850 const char* dunderName = nullptr;
851 switch (op) {
852 case OpCode::OP_ADD:
853 dunderName = "__add";
854 break;
855 case OpCode::OP_SUBTRACT:
856 dunderName = "__sub";
857 break;
858 case OpCode::OP_MULTIPLY:
859 dunderName = "__mul";
860 break;
861 case OpCode::OP_DIVIDE:
862 dunderName = "__div";
863 break;
864 case OpCode::OP_MODULO:
865 dunderName = "__mod";
866 break;
867 default:
868 return false;
869 }
870 const BytecodeChunk* targetChunkPtr = findMethodChunk(left.className(), dunderName);
871 if (targetChunkPtr == nullptr) {
872 return false; // 无 dunder 方法 → 回退到“算术运算需要数值类型”报错
873 }
874 const BytecodeChunk& targetChunk = *targetChunkPtr;
875
876 // 与 Interpreter 一致：运算符方法必须恰好 1 个参数
877 if (targetChunk.requiredArity != 1 || targetChunk.arity != 1) {
878 outResult = runtimeError(std::string("运算符方法 ") + dunderName + " 必须恰好接受 1 个参数");
879 return true;
880 }
881 if (frames_.size() >= MAX_FRAMES) {
882 outResult = runtimeError(
883 ErrorFormat::formatStd(ErrorMessages::kRecursionDepthExceededFmtStd, static_cast<int>(MAX_FRAMES)),
884 DiagCodes::kRecursionDepth);
885 return true;
886 }
887
888 // 栈重排：[left,right] → [this(=left), 字段..., arg(=right)] + 局部变量槽
889 Value right = pop();
890 Value objCopy = pop();
891 push(objCopy); // this
892 int fieldCount = 0;
893 if (!targetChunk.fieldOrder.empty()) {
894 for (const auto& fieldName : targetChunk.fieldOrder) {
895 auto fieldIt = objCopy.fields().find(fieldName);
896 push(fieldIt != objCopy.fields().end() ? fieldIt->second : Value::nullValue());
897 }
898 fieldCount = static_cast<int>(targetChunk.fieldOrder.size());
899 }
900 push(std::move(right)); // 唯一参数
901
902 int preAllocated = 1 + fieldCount + 1; // this + 字段 + 1 参数
903 int extraSlots = targetChunk.localCount - preAllocated;
904 if (extraSlots < 0) {
905 popN(static_cast<size_t>(preAllocated));
906 outResult = runtimeError(ErrorFormat::formatStd("方法 {} 帧布局损坏: localCount={} < preAllocated={}",
907 dunderName, targetChunk.localCount, preAllocated));
908 return true;
909 }
910 for (int i = 0; i < extraSlots; ++i) {
911 push(Value::nullValue());
912 }
913
914 VMCallFrame newFrame;
915 newFrame.chunk = targetChunkPtr;
916 newFrame.returnIp = ip + 1; // OP_ADD 系列均为单字节指令
917 newFrame.basePointer = stack_.size() - targetChunk.localCount;
918 newFrame.functionName = targetChunk.name;
919 newFrame.ip = 0;
920 newFrame.isMethodCall = true;
921 newFrame.isInitCall = false;
922 newFrame.receiverLocalSlot = -1; // 接收者不写回（见函数头注释）
923 auto upvIt = methodUpvalues_.find(targetChunk.name);
924 if (upvIt != methodUpvalues_.end() && upvIt->second) {
925 newFrame.upvalues = upvIt->second->upvalues;
926 }
927 size_t savedIp = ip;
928 frames_.push_back(std::move(newFrame));
929 notifyStep(savedIp, op);
930 outResult = VMResult::VM_OK;
931 return true;
932}

934VMResult VM::numericOp(int opType) {
935 // 使用 peek 访问栈顶避免深拷贝，然后调整栈指针
936 if (stack_.size() < 2)
937 return runtimeError("栈下溢：二元运算需要两个操作数");
938
939 const Value& rightRef = stack_[stack_.size() - 1];
940 const Value& leftRef = stack_[stack_.size() - 2];
941
942 // 字符串拼接（仅加法）
943 if (opType == OP_ADD_INT) {
944 if (leftRef.isString() && rightRef.isString()) {
945 const auto& ls = leftRef.stringVal();
946 const auto& rs = rightRef.stringVal();
947 std::string concat;
948 concat.reserve(ls.size() + rs.size());
949 concat.append(ls).append(rs);
950 stack_[stack_.size() - 2] = Value(std::move(concat));
951 stack_.pop_back();
952 return VMResult::VM_OK;
953 }
954 if (leftRef.isString() || rightRef.isString()) {
955 // S5 fix: 字符串侧用 stringVal() 引用，非字符串侧才 toString()，消除冗余拷贝
956 std::string result;
957 if (leftRef.isString()) {
958 const auto& ls = leftRef.stringVal();
959 auto rs = rightRef.toString();
960 result.reserve(ls.size() + rs.size());
961 result.append(ls).append(rs);
962 } else {
963 auto ls = leftRef.toString();
964 const auto& rs = rightRef.stringVal();
965 result.reserve(ls.size() + rs.size());
966 result.append(ls).append(rs);
967 }
968 stack_[stack_.size() - 2] = Value(std::move(result));
969 stack_.pop_back();
970 return VMResult::VM_OK;
971 }
972 }
973
974 // 非数值类型检查（字符串拼接已在上面处理）
975 if (!leftRef.isNumber() || !rightRef.isNumber()) {
976 return runtimeError("算术运算需要数值类型", DiagCodes::kTypeMismatch);
977 }
978
979 // C12 fix: 调用共享算术运算逻辑，消除与 Interpreter 的重复实现
980 using namespace NumericOps;
981 ArithOp op;
982 switch (opType) {
983 case OP_ADD_INT:
984 op = ArithOp::Add;
985 break;
986 case OP_SUB_INT:
987 op = ArithOp::Sub;
988 break;
989 case OP_MUL_INT:
990 op = ArithOp::Mul;
991 break;
992 case OP_DIV_INT:
993 op = ArithOp::Div;
994 break;
995 case OP_MOD_INT:
996 op = ArithOp::Mod;
997 break;
998 default:
999 return runtimeError("未知算术运算类型");
1000 }
1001
1002 ArithResult r = computeArith(op, leftRef.isInt(), leftRef.isInt() ? leftRef.intVal() : 0, leftRef.toDouble(),
1003 rightRef.isInt(), rightRef.isInt() ? rightRef.intVal() : 0, rightRef.toDouble());
1004 switch (r.status) {
1005 case ArithStatus::DivByZero:
1006 return runtimeError("除零错误", DiagCodes::kDivisionByZero);
1007 case ArithStatus::IntOverflow:
1008 return runtimeError("整数运算溢出");
1009 case ArithStatus::NotNumeric:
1010 return runtimeError("算术运算需要数值类型", DiagCodes::kTypeMismatch);
1011 case ArithStatus::OK:
1012 stack_[stack_.size() - 2] = r.isIntResult ? Value(r.intVal) : Value(r.floatVal);
1013 break;
1014 default:
1015 // ArithStatus 是封闭枚举，落空表示内部错误；不 break 以避免
1016 // stack_[size-2] 未被写入却在下方 pop_back 后成为脏结果。
1017 return runtimeError("内部错误: 未知算术状态");
1018 }
1019
1020 stack_.pop_back(); // 弹出 right，保留结果在 left 原位
1021 return VMResult::VM_OK;
1022}

1036// ============================================================
1037// writeBack 接收者写回机制（ARCH-24 设计说明）
1038// ------------------------------------------------------------
1039// 背景：MiniLang 方法调用语义要求变异方法（如 arr.push()、dict.remove()、
1040// 实例字段写入）的修改回写到原始变量/字段。由于 VM 栈式执行，方法调用时
1041// 接收者被拷贝到新栈帧，方法内的修改不会自动反映到调用者栈帧。
1042//
1043// 编译期编码（Compiler::visitMethodCall）：
1044// 编译器检测接收者是否为简单 VarRef：
1045// - 全局变量：编码 receiverVarIdx（常量池索引，NO_INDEX 表示无）
1046// - 局部变量：编码 receiverLocalSlot（栈槽偏移，NO_SLOT 表示无）
1047// - 复杂表达式（如 obj.x.y.f()）：不写回，结果存入 lastMutatedReceiver_
1048// 这些编码附加在 OP_METHOD_CALL 指令后，VM 执行时读取。
1049//
1050// 运行期写回（VM::writeBackReceiver）：
1051// 方法返回时（OP_RETURN），VM 根据 receiverVarIdx/receiverLocalSlot 判断
1052// 写回目标，执行三分支写回：
1053// 1. receiverVarIdx != RuntimeLimits::NO_INDEX → 写入 globalSlots_（槽位寻址）或 globals_（名称寻址）
1054// 2. receiverLocalSlotByte != RuntimeLimits::NO_SLOT → 写入栈帧（含字段同步到实例）
1055// 3. 否则 → 存入 lastMutatedReceiver_（供调用方通过 OP_GET_LAST_MUTATED 读取）
1056//
1057// 字段同步优化（PERF-13 fix）：
1058// fieldsModified 脏标记由 OP_SET_LOCAL/OP_MEMBER_SET_LOCAL 在写入字段槽时置位。
1059// OP_RETURN 仅在 fieldsModified=true 时执行字段同步循环，跳过只读方法
1060// 和无字段写入的 init，避免冗余拷贝。
1061//
1062// 链式调用（dispatchArrayBuiltin/dispatchDictBuiltin）：
1063// 内建方法分发器在变异路径（ARR_PUSH/POP/REMOVE、DICT_REMOVE）后调用
1064// writeBackReceiver(..., true) 强制写回。非变异路径（ARR_LEN/CONTAINS、
1065// DICT_LEN/KEYS/VALUES/HAS/GET）不调用写回。
1066//
1067// P7 fix：字段同步先拷贝后 move 的顺序
1068// receiverLocalSlotByte == 0 时（this 槽位），先拷贝 mutatedObj 到字段，
1069// 再 move 到栈槽。若先 move 则拷贝时 mutatedObj 已被掏空。
1070// ============================================================
1071
1072// 写回变异方法调用后的接收者（统一数组/字典/实例三处写回逻辑）
1073// 判断链：
1074// 1. receiverVarIdx != RuntimeLimits::NO_INDEX → 写入 globalSlots_ 或 globals_
1075// 2. receiverLocalSlotByte != RuntimeLimits::NO_SLOT → 写入栈帧（含字段同步）
1076// 3. 否则 → 存入 lastMutatedReceiver_
1077VMResult VM::writeBackReceiver(uint16_t receiverVarIdx, uint8_t receiverLocalSlotByte, Value& mutatedObj,
1078 bool fieldsModified) {
1079 const BytecodeChunk& chunk = currentChunk();
1080 if (receiverVarIdx != RuntimeLimits::NO_INDEX && receiverVarIdx < chunk.constants.size()) {
1081 // 防御性检查：常量条目应为字符串（编译期由 IR visitMethodCall 写入）。
1082 // 若字节码被恶意构造或 lowering 存在 bug 写入了非字符串常量，
1083 // stringVal() 行为未定义。显式报错优于静默 UB。
1084 if (!chunk.constants[receiverVarIdx].isString()) {
1085 return runtimeError("writeBackReceiver: 接收者常量非字符串类型");
1086 }
1087 const std::string& recvName = chunk.constants[receiverVarIdx].stringVal();
1088 auto gsIt = globalNameToSlot_.find(recvName);
1089 // AUDIT-R3 P2-3 fix: 补 slot 范围校验，与 resolveMutableGlobal 防御口径一致
1090 if (gsIt != globalNameToSlot_.end() && gsIt->second >= 0 &&
1091 gsIt->second < static_cast<int>(globalSlots_.size())) {
1092 globalSlots_[gsIt->second] = std::move(mutatedObj);
1093 } else {
1094 globals_[recvName] = std::move(mutatedObj);
1095 }
1096 } else if (receiverLocalSlotByte != RuntimeLimits::NO_SLOT) {
1097 size_t bp = currentFrame().basePointer;
1098 // P7 fix: 先拷贝到字段（需要完整副本），再 move 到栈槽
1099 if (fieldsModified && receiverLocalSlotByte > 0 && bp < stack_.size() && stack_[bp].isInstance()) {
1100 VMCallFrame& curFrame = currentFrame();
1101 if (curFrame.chunk && receiverLocalSlotByte <= curFrame.chunk->fieldOrder.size()) {
1102 const std::string& fn = curFrame.chunk->fieldOrder[receiverLocalSlotByte - 1];
1103 stack_[bp].fields()[fn] = mutatedObj; // 拷贝（move 前）
1104 }
1105 }
1106 if (bp + receiverLocalSlotByte < stack_.size()) {
1107 stack_[bp + receiverLocalSlotByte] = std::move(mutatedObj); // move 在后
1108 }
1109 } else {
1110 lastMutatedReceiver_ = std::move(mutatedObj);
1111 }
1112 return VMResult::VM_OK;
1113}

1405}

1424 }

1426 return false;
1427 ip += instrSize;
1428 }

1573// R114 阶段 3：restoreFromSnapshot — 从快照恢复 VM 状态（状态回滚）
1574// ============================================================
1575bool VM::restoreFromSnapshot(const std::vector<Value>& stackValues,
1576 const std::vector<std::pair<std::string, Value>>& globalsValues, size_t targetIp,
1577 size_t targetFrameCount) {
1578 if (!initialized_) {
1579 // P2-12: 改为通过 diagnostics_ 报告错误（消除 lastError_ 字段）
1580 diagnostics_.addError("VM 未初始化，无法回滚", 0, 0, DiagSource::VM);
1581 hasError_ = true;
1582 return false;
1583 }
1584 if (targetFrameCount > frames_.size()) {
1585 diagnostics_.addError("回滚目标帧数超过当前帧数", 0, 0, DiagSource::VM);
1586 hasError_ = true;
1587 return false;
1588 }
1589
1590 // (1) 截断 frames_ 到 targetFrameCount（仅截断不扩展）
1591 while (frames_.size() > targetFrameCount) {
1592 frames_.pop_back();
1593 }
1594 // 设置栈顶帧 ip = targetIp（若仍有帧）
1595 if (!frames_.empty()) {
1596 currentFrame().ip = targetIp;
1597 }
1598
1599 // (2) 替换 stack_ 为 stackValues（截断到 MAX_STACK_SIZE）
1600 stack_.clear();
1601 size_t copyCount = std::min(stackValues.size(), MAX_STACK_SIZE);
1602 for (size_t i = 0; i < copyCount; ++i) {
1603 stack_.push_back(stackValues[i]);
1604 }
1605
1606 // (3) 按 name 更新 globalSlots_/globals_（已有变量覆盖，新变量忽略）
1607 for (const auto& kv : globalsValues) {
1608 const std::string& name = kv.first;
1609 const Value& val = kv.second;
1610 auto slotIt = globalNameToSlot_.find(name);
1611 if (slotIt != globalNameToSlot_.end()) {
1612 int slot = slotIt->second;
1613 if (slot >= 0 && slot < static_cast<int>(globalSlots_.size())) {
1614 globalSlots_[slot] = val;
1615 }
1616 } else {
1617 // globals_ 中的 runtime-defined 变量
1618 auto it = globals_.find(name);
1619 if (it != globals_.end()) {
1620 it->second = val;
1621 }
1622 // 不创建新变量——回滚不应引入编译期未注册的变量
1623 }
1624 // 同步失效 per-chunk varCache_（递增版本号，下次 lookup 重建）
1625 }
1626 ++globalsVersion_;
1627
1628 // (4) 清理 openUpvalues_（slots >= 新栈大小的 open upvalue 已悬垂）
1629 // AUDIT-R3 P1-2 fix: openUpvalues_ 按 stackSlot 升序，需清理的是尾部区间
1630 // [lower_bound(newStackSize), end())。原实现从 begin()（最小 slot）开始且
1631 // 首元素 slot < newStackSize 即 break——只要存在任一低位 open upvalue，
1632 // 所有高位悬垂条目全部残留（后续 closeUpvaluesFrom 报 slot 越界错误，
1633 // 或闭包读到回滚后被复用栈槽的错误值）。对齐 closeUpvaluesFrom 的
1634 // lower_bound 定位；回滚后原栈值已不可用，标记 isClosed（保留默认值）
1635 // 防止仍被闭包持有的 upvalue 悬垂读栈。
1636 size_t newStackSize = stack_.size();
1637 {
1638 auto it = openUpvalues_.lower_bound(newStackSize);
1639 while (it != openUpvalues_.end()) {
1640 if (auto uv = it->second.lock()) {
1641 if (!uv->isClosed) {
1642 uv->isClosed = true; // 无法拷值（栈已回滚），保留默认值防悬垂
1643 }
1644 }
1645 it = openUpvalues_.erase(it);
1646 }
1647 }
1648
1649 // (5) 清理 tryStack_（frameIndex >= targetFrameCount 的 handler 已悬垂）
1650 while (!tryStack_.empty() && tryStack_.back().frameIndex >= targetFrameCount) {
1651 tryStack_.pop_back();
1652 }
1653
1654 // (6) 保守清空 pendingJumpStack_（break/continue 跨回滚未定义）
1655 pendingJumpStack_.clear();
1656
1657 // (7) 清理错误/暂存状态
1658 // P2-12: lastError_/lastErrorLine_ 已移除，hasError_+diagnostics_ 统一管理
1659 // 与 RegisterVM::restoreFromSnapshot 行为一致——清空 diagnostics_ 使回滚后
1660 // hasError()/getLastError() 反映干净状态（调试器回滚后不应显示旧错误）。
1661 hasError_ = false;
1662 diagnostics_.clear();
1663 lastMutatedReceiver_ = Value::nullValue();
1664 pendingFieldOrder_.clear();
1665
1666 return true;
1667}

1669VMResult VM::popFrameAndPushReturn() {
1670 // P1 dedup: 提取自 stepOnce() 和 execute() 中"chunk 执行完毕 → 弹帧"的防御性路径。
1671 // 调用前需确认 ip >= chunk.code.size()（当前 chunk 已执行完毕）。
1672 VMCallFrame& frame = currentFrame();
1673 size_t returnIp = frame.returnIp;
1674 size_t savedBp = frame.basePointer;
1675 bool wasInit = frame.isInitCall;
1676 // P1-5 fix: 记录帧索引并清理 tryStack_ handler
1677 size_t returningFrameIdx = frames_.size() - 1;
1678 // AUDIT-P2.5 fix: closeUpvaluesFrom 返回错误时立即终止返回，避免在损坏状态上继续
1679 if (closeUpvaluesFrom(savedBp) != VMResult::VM_OK) // P1-5 fix: 关闭 open upvalues
1680 return VMResult::VM_RUNTIME_ERROR;
1681 frames_.pop_back();
1682 while (!tryStack_.empty() && tryStack_.back().frameIndex >= returningFrameIdx) {
1683 tryStack_.pop_back(); // P1-5 fix: 清理残留 handler
1684 }
1685 if (!frames_.empty()) {
1686 currentFrame().ip = returnIp; // 恢复调用者 ip，避免重复执行调用指令
1687 // V-P1-5 fix: 先截断栈清理 callee 残留局部变量，再 push 返回值
1688 // AUDIT-BUG-V1 fix: wasInit 分支访问 stack_[savedBp] 需 savedBp < stack_.size()，
1689 // 原条件 savedBp <= stack_.size() 允许 savedBp == top_ 触发 VMStack::operator[] abort。
1690 if (savedBp <= stack_.size()) {
1691 if (wasInit && savedBp < stack_.size()) {
1692 Value thisVal = stack_[savedBp]; // 先拷出 this
1693 stack_.resize(savedBp); // 清理 callee 残留
1694 push(std::move(thisVal));
1695 } else {
1696 stack_.resize(savedBp);
1697 push(Value::nullValue());
1698 }
1699 } else {
1700 push(Value::nullValue());
1701 }
1702 }
1703 return VMResult::VM_OK;
1704}

1706VMResult VM::stepOnce() {
1707 // 帧已空 → 执行完毕
1708 if (frames_.empty())
1709 return VMResult::VM_OK;
1710
1711 // 已有错误 → 不再执行
1712 if (hasError_)
1713 return VMResult::VM_RUNTIME_ERROR;
1714
1715 // AUDIT-R4 BUG-10 fix: 单步执行期间同样抑制增量 GC 回调（同 execute()）。
1716 // 分配仅发生在指令执行内，每步构造/析构 suppressor 的开销仅在调试
1717 // 单步模式下发生（UI 驱动，非热路径）。
1718 GcManager::CallbackSuppressor gcSuppressor;
1719
1720 // P1 fix: stepOnce 累计指令预算检查，防止通过循环调用 stepOnce 绕过 DoS 防护
1721 // 注意：execute() 使用每次调用重新开始的局部计数器，而 stepOnce 使用成员
1722 // stepInstructionCount_ 跨调用累计——因为单步模式下程序由 IDE 多次调用推进，
1723 // 若每次 stepOnce 都从 0 计数，无限循环单步将绕过指令数上限。
1724 // L7 fix: 改为读取 RuntimeConfig 运行时配置（教学场景可调）
1725 const int64_t dynMaxInstr = RuntimeLimits::RuntimeConfig::instance().maxInstructions();
1726 if (++stepInstructionCount_ > dynMaxInstr) {
1727 // P2-12: 错误信息直接写入 diagnostics_（消除 lastError_ 字段）
1728 diagnostics_.addError(
1729 ErrorFormat::formatStd("指令执行数超过上限 {}，疑似无限循环", static_cast<long long>(dynMaxInstr)), 0, 0,
1730 DiagSource::VM);
1731 hasError_ = true;
1732 return VMResult::VM_RUNTIME_ERROR;
1733 }
1734
1735 VMCallFrame& frame = currentFrame();
1736 const BytecodeChunk& chunk = *frame.chunk;
1737 size_t& ip = frame.ip;
1738
1739 // 当前 chunk 执行完毕 → 弹帧（防御性路径，正常情况由 OP_RETURN 处理）
1740 if (ip >= chunk.code.size()) {
1741 return popFrameAndPushReturn();
1742 }
1743
1744 // L14: VM_EXCEPTION_THROW 表示异常被 try/catch 捕获，ip 已在 catchIp。
1745 // 对单步调用方而言此步已完成（非错误），转换为 VM_OK 保持向后兼容。
1746 VMResult r = executeOneInstruction();
1747 return (r == VMResult::VM_EXCEPTION_THROW) ? VMResult::VM_OK : r;
1748}

1754VMResult VM::execute(const CompileResult& result) {
1755 initExecution(result);
1756
1757 // AUDIT-R4 BUG-10 fix: VM 执行期间抑制增量 GC 触发回调。
1758 // VM 操作数栈/全局槽上的临时 Value 不在 GcManager 的根集中（回调属
1759 // Interpreter，仅收集环境链根）；若分配阈值在 VM 执行中途触发回调，
1760 // 仅经 VM 栈可达的容器会被误判为不可达孤岛（默认模式子元素被掉空，
1761 // GcOnly 模式直接 delete → UAF/双重释放）。与 JIT 路径的既有防护
1762 // （JIT.cpp CallbackSuppressor + safepoint）对齐；抑制期间产生的循环
1763 // 引用孤岛由下一轮 Interpreter execute() 入口的兜底 collectCycle 回收。
1764 GcManager::CallbackSuppressor gcSuppressor;
1765
1766 // S-02 fix: 指令执行预算，防止恶意字节码导致 DoS
1767 int64_t instructionCount = 0;
1768 // PERF-03 fix: 在循环入口加载一次指令预算上限到局部变量，避免主循环每条指令
1769 // 都执行 RuntimeConfig::instance().maxInstructions()（含 atomic load + 单例访问）。
1770 // 语义等价性：RuntimeConfig 设计为执行前配置（教学场景可调），execute() 期间
1771 // 不应被修改；下一次 execute() / stepOnce() 调用会读取最新值。stepOnce() 路径
1772 // 仍逐次读取以支持 IDE 单步调试时实时调整预算。
1773 const int64_t dynMaxInstr = RuntimeLimits::RuntimeConfig::instance().maxInstructions();

1775 VMCallFrame& frame = currentFrame();
1776 const BytecodeChunk& chunk = *frame.chunk;
1777 size_t& ip = frame.ip;
1778
1779 if (ip >= chunk.code.size()) {
1780 // 当前 chunk 执行完毕 → 弹帧（防御性路径，正常情况由 OP_RETURN 处理）
1781 VMResult r = popFrameAndPushReturn();
1782 if (r != VMResult::VM_OK)
1783 return r;
1784 continue;
1785 }
1786
1787 if (hasError_) [[unlikely]]
1788 return VMResult::VM_RUNTIME_ERROR;
1789 // S-02 fix: 检查指令预算
1790 // PERF-03 fix: dynMaxInstr 已在循环入口缓存到局部变量，避免每条指令 atomic load。
1791 if (++instructionCount > dynMaxInstr) [[unlikely]] {
1792 // P2-12: 错误信息直接写入 diagnostics_（消除 lastError_ 字段）
1793 diagnostics_.addError(
1794 ErrorFormat::formatStd("指令执行数超过上限 {}，疑似无限循环", static_cast<long long>(dynMaxInstr)), 0,
1795 0, DiagSource::VM);
1796 hasError_ = true;
1797 return VMResult::VM_RUNTIME_ERROR;
1798 }
1799 VMResult r = executeOneInstruction();
1800 // L14: VM_EXCEPTION_THROW 表示 runtimeError 被 try/catch 捕获，ip 已设置为 catchIp。
1801 // 不视为错误，继续执行（不递增 ip，catchIp 由 throwException 写入 frame.ip）。
1802 if (r == VMResult::VM_EXCEPTION_THROW)
1803 continue;
1804 if (r != VMResult::VM_OK || hasError_)
1805 return VMResult::VM_RUNTIME_ERROR;
1806 }
1807
1808 if (hasError_)
1809 return VMResult::VM_RUNTIME_ERROR;
1810 return VMResult::VM_OK;
1811}

1815// ============================================================
1816// A3 注记：MSVC 对密集 switch 已自动生成跳转表（jump table），
1817// 函数指针表的额外重构（1600+ 行拆分为 50+ 方法）收益极小。
1818// 保留 switch 形式，确保代码可维护性。
1819
1820// ============================================================
1821// executeOneInstruction() — 单条指令执行（按指令类别转发到私有方法）
1822// ============================================================
1823
1824VMResult VM::executeOneInstruction() {
1825 // 单条指令执行的核心分派点，execute() 与 stepOnce() 共用此函数。
1826 // 执行流程：① 判空帧/hasError_ 短路；② 取 opcode；③ (仅 debug) 计算完整指令
1827 // 长度并做字节码截断边界检查——release 由 initExecution 的加载期预扫描覆盖；
1828 // ④ 两级转发——先按指令类别 switch 到 executeXxxOps，再由各方法按具体 opcode 处理。
1829 // 两级拆分仅为可维护性（每类方法 < 200 行），不改变执行语义。
1830 if (hasError_) [[unlikely]]
1831 return VMResult::VM_RUNTIME_ERROR;
1832 VMCallFrame& frame = currentFrame();
1833 const BytecodeChunk& chunk = *frame.chunk;
1834 size_t& ip = frame.ip;
1835
1836 OpCode op = static_cast<OpCode>(chunk.code[ip]);
1837
1838#ifdef MINILANG_VM_PROFILING
1839 // C3: 热路径 profiling — 每条指令执行前递增对应 opcode 计数器
1840 ++opProfileCounts_[static_cast<uint8_t>(op)];
1841#endif
1842
1843#ifndef NDEBUG
1844 // M-新2 fix: OP_CLOSURE 是变长指令，需要计算完整长度再做边界检查。
1845 // 截断校验已上移至 initExecution 的加载期预扫描，release 主循环不再逐指令
1846 // 查表+比较；debug 保留逐指令校验作为双保险。
1847 size_t instrSize = BytecodeChunk::instructionSize(op);
1848 if (op == OpCode::OP_CLOSURE && ip + 3 < chunk.code.size()) [[unlikely]] {
1849 uint8_t uvCount = chunk.code[ip + 3];
1850 instrSize = 4 + static_cast<size_t>(uvCount) * 2;
1851 }
1852 if (ip + instrSize > chunk.code.size()) [[unlikely]] {
1853 return runtimeError("字节码截断: 指令不完整");
1854 }
1855#endif
