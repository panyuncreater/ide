65// 2026-06-29: 共享运行时类型匹配函数（VM/RegisterVM/Interpreter 复用）
66// ============================================================
67// 提取自 Interpreter::typeMatch，供三后端统一类型注解强制逻辑。
68// 处理：原始类型(int/float/bool/string)、null（兼容所有注解）、
69// array/dict、数组元素类型注解(如 "int[]")、实例精确类名匹配。
70//
71// 实例继承链检查由调用方扩展（Interpreter 用 classRegistry_，VM 用 classInfo_）。
72// BUG-014 审计结论（2026-07-18）：三后端语义一致，非误报——
73// - Interpreter::typeMatch 内联继承链遍历（classRegistry_）
74// - StackVM OP_TYPE_CHECK (VMContainers.cpp:683-704) 在 typeMatchValue 失败后
75// 执行相同继承链 fallback 检查（classInfo_）
76// - RegisterVM REG_TYPE_CHECK (RegisterVM.cpp:1337-1357) 同样有继承链 fallback
77// 三个后端对相同输入产生一致结果，仅代码组织不同（共享 helper + 调用方 fallback
78// vs 内联实现）。null 兼容性、T? 剥离、fun() 类型、数组/dict 元素递归均已对齐。
79inline bool typeMatchValue(const Value& val, const std::string& annotation) {
80 if (annotation.empty())
81 return true;
82 // null 兼容所有类型注解（2026-06-29 用户决策）
83 if (val.isNull())
84 return true;
85 // 可选类型 T?：剥离末尾 '?' 后递归匹配（对齐 Interpreter::typeMatch AUDIT-P2.7）
86 if (annotation.size() >= 2 && annotation.back() == '?') {
87 return typeMatchValue(val, annotation.substr(0, annotation.size() - 1));
88 }
89 if (annotation == TypeName::INT)
90 return val.isInt();
91 if (annotation == TypeName::FLOAT)
92 return val.isFloat() || val.isInt();
93 if (annotation == TypeName::BOOL)
94 return val.isBool();
95 if (annotation == TypeName::STRING)
96 return val.isString();
97 if (annotation == TypeName::ARRAY)
98 return val.isArray();
99 if (annotation == TypeName::DICT)
100 return val.isDict();
101 // R98 元组与解构：tuple 类型注解匹配
102 if (annotation == TypeName::TUPLE)
103 return val.isTuple();
104 if (annotation == TypeName::NULL_T)
105 return val.isNull();
106 // R99 枚举与 ADT：enum variant 类型注解匹配
107 // - "enum"：泛型注解，匹配任意 enum variant
108 // - "<EnumName>"：精确 enum 名匹配（如 "Color" 匹配 Color.Red/Color.Green 等）
109 if (annotation == TypeName::ENUM) {
110 return val.isEnumVariant();
111 }
112 if (val.isEnumVariant()) {
113 return val.enumVariantEnumName() == annotation;
114 }
115 // dict[K:V] 泛型字典类型注解：key 支持 string/int/bool/float（L4 fix），递归检查每个 value
116 // （对齐 Interpreter::typeMatch AUDIT-P2.7）
117 if (annotation.size() >= 7 && annotation.substr(0, 5) == "dict[" && annotation.back() == ']') {
118 if (!val.isDict())
119 return false;
120 std::string inner = annotation.substr(5, annotation.size() - 6);
121 size_t colonPos = inner.find(':');
122 if (colonPos == std::string::npos)
123 return false;
124 std::string keyType = inner.substr(0, colonPos);
125 std::string valType = inner.substr(colonPos + 1);
126 for (const auto& kv : val.dictVal()) {
127 // L4 fix: kv.first 是 DictKey variant，按 keyType 检查键类型
128 if (keyType == "string") {
129 if (!std::holds_alternative<std::string>(kv.first))
130 return false;
131 } else if (keyType == "int") {
132 if (!std::holds_alternative<int64_t>(kv.first))
133 return false;
134 } else if (keyType == "bool") {
135 if (!std::holds_alternative<bool>(kv.first))
136 return false;
137 } else if (keyType == "float") {
138 if (!std::holds_alternative<double>(kv.first))
139 return false;
140 } else {
141 return false; // 未知 keyType
142 }
143 if (!typeMatchValue(kv.second, valType))
144 return false;
145 }
146 return true;
147 }
148 // 函数类型 fun(...):ret：仅检查是闭包即可（对齐 Interpreter::typeMatch AUDIT-P2.7，参数/返回类型不做验证）
149 if (annotation.size() >= 5 && annotation.substr(0, 4) == "fun(") {
150 return val.isClosure();
151 }
152 // 数组元素类型注解，如 "int[]"、多维 "int[][]"（后缀 [] 剥离递归，对齐 Interpreter::typeMatch AUDIT-P2-CORRECT）
153 if (annotation.size() >= 2 && annotation.substr(annotation.size() - 2) == "[]") {
154 if (!val.isArray())
155 return false;
156 std::string elemType = annotation.substr(0, annotation.size() - 2);
157 for (const auto& elem : val.arrayVal()) {
158 if (!typeMatchValue(elem, elemType))
159 return false;
160 }
161 return true;
162 }
163 // 实例类型注解：精确类名匹配（继承链由调用方扩展）
164 if (val.isInstance()) {
165 return val.className() == annotation;
166 }
167 return false;
168}

198/// 类型信息（预留，未来扩展）
199struct TypeInfo {
200 TypeKind kind = TypeKind::UNKNOWN;
201 std::string className{}; // INSTANCE 类型的类名
202 std::vector<TypeInfo> typeArgs{}; // 泛型类型参数（ARRAY 的元素类型、CLOSURE 的参数/返回类型）
203
204 // 显式构造函数（非聚合化）：避免 TypeInfo{TypeKind::X} 部分初始化触发
205 // GCC -Wextra 的 missing-field-initializers 警告（CI -Werror 拦截）。
206 TypeInfo() = default;
207 explicit TypeInfo(TypeKind k) : kind(k) {}
208 TypeInfo(TypeKind k, std::string cls) : kind(k), className(std::move(cls)) {}
209
210 bool isNumeric() const { return kind == TypeKind::INT || kind == TypeKind::FLOAT; }
211
212 /// 类型兼容性检查：判断 other 类型的值是否可赋给 this 类型的变量。
213 /// 规则：UNKNOWN/ANY 兼容一切；float 接受 int（数值提升）；
214 /// ARRAY/DICT/INSTANCE 按 className/typeArgs 递归检查；其余严格匹配。
215 bool isCompatible(const TypeInfo& other) const {
216 // UNKNOWN 或 ANY 兼容一切（保守假设 / 通配符）
217 if (kind == TypeKind::UNKNOWN || kind == TypeKind::ANY)
218 return true;
219 if (other.kind == TypeKind::UNKNOWN || other.kind == TypeKind::ANY)
220 return true;
221 // null 兼容所有类型（MiniLang 动态语言惯例）
222 if (other.kind == TypeKind::NULL_T)
223 return true;
224 // float 接受 int（数值提升）
225 if (kind == TypeKind::FLOAT && other.kind == TypeKind::INT)
226 return true;
227 // 同类型直接兼容
228 if (kind == other.kind) {
229 // INSTANCE 需要类名匹配
230 if (kind == TypeKind::INSTANCE)
231 return className.empty() || other.className.empty() || className == other.className;
232 return true;
233 }
234 return false;
235 }

237 /// 从类型注解字符串构造 TypeInfo
238 static TypeInfo fromAnnotation(const std::string& ann) {
239 if (ann.empty())
240 return TypeInfo{TypeKind::UNKNOWN};
241 if (ann == "int")
242 return TypeInfo{TypeKind::INT};
243 if (ann == "float")
244 return TypeInfo{TypeKind::FLOAT};
245 if (ann == "bool")
246 return TypeInfo{TypeKind::BOOL};
247 if (ann == "string")
248 return TypeInfo{TypeKind::STRING};
249 if (ann == "null")
250 return TypeInfo{TypeKind::NULL_T};
251 if (ann == "array")
252 return TypeInfo{TypeKind::ARRAY};
253 if (ann == "dict")
254 return TypeInfo{TypeKind::DICT};
255 // 以 [] 结尾的数组元素类型
256 if (ann.size() >= 2 && ann.substr(ann.size() - 2) == "[]")
257 return TypeInfo{TypeKind::ARRAY};
258 // fun(...) 函数类型
259 if (ann.size() >= 5 && ann.substr(0, 4) == "fun(")
260 return TypeInfo{TypeKind::CLOSURE};
261 // 其他视为实例类型（类名）
262 return TypeInfo{TypeKind::INSTANCE, ann};
263 }
