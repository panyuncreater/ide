66// 2026-06-29: 共享运行时类型匹配函数（VM/RegisterVM/Interpreter 复用）
67// ============================================================
68// 提取自 Interpreter::typeMatch，供三后端统一类型注解强制逻辑。
69// 处理：原始类型(int/float/bool/string)、null（兼容所有注解）、
70// array/dict、数组元素类型注解(如 "int[]")、实例精确类名匹配。
71//
72// 实例继承链检查由调用方扩展（Interpreter 用 classRegistry_，VM 用 classInfo_）。
73// BUG-014 审计结论（2026-07-18）：三后端语义一致，非误报——
74// - Interpreter::typeMatch 内联继承链遍历（classRegistry_）
75// - StackVM OP_TYPE_CHECK (VMContainers.cpp:683-704) 在 typeMatchValue 失败后
76// 执行相同继承链 fallback 检查（classInfo_）
77// - RegisterVM REG_TYPE_CHECK (RegisterVM.cpp:1337-1357) 同样有继承链 fallback
78// 三个后端对相同输入产生一致结果，仅代码组织不同（共享 helper + 调用方 fallback
79// vs 内联实现）。null 兼容性、T? 剥离、fun() 类型、数组/dict 元素递归均已对齐。
80/// （2026-10-22）dict[inner] 键/值分隔冒号定位：从左贪婪消费一个完整类型注解
81///（与 Parser::parseTypeAnnotation 的生成文法同构：dict[K:V] / fun(p):r 复合
82/// 形态递归 + []/? 后缀堆叠），其后的 ':' 即键/值分隔符。
83/// 修复"取首个 ':'"在键类型自带冒号形态下的截断：
84/// dict[dict[string:K]:int] —— 首冒号在嵌套 dict 方括号内；
85/// dict[fun():K:string] —— fun 返回类型冒号与分隔冒号在括号外同深度，
86/// 必须贪婪消费键类型 fun():K 后定位分隔符。
87/// 两侧共用（annotationReferencesTypeParam 与 typeMatchValue），保证擦除判定
88/// 与运行时校验看到同一把切分尺。
89/// 返回分隔冒号下标；格式异常返回 npos（调用方保守走原拒绝路径）。
90inline size_t skipTypeAnnotationFrom(const std::string& inner, size_t start) {
91 const auto isIdentChar = [](char c) {
92 return (c >= 'a' && c <= 'z') || (c >= 'A' && c <= 'Z') || (c >= '0' && c <= '9') || c == '_';
93 };
94 size_t j = start;
95 if (j >= inner.size() || !isIdentChar(inner[j]))
96 return std::string::npos;
97 while (j < inner.size() && isIdentChar(inner[j]))
98 ++j;
99 const std::string base = inner.substr(start, j - start);
100 if (base == "dict" && j < inner.size() && inner[j] == '[' &&
101 !(j + 1 < inner.size() && inner[j + 1] == ']')) {
102 // dict[K:V]：递归消费键类型、':'、值类型、']'（对齐 parseTypeAnnotation；
103 // "dict[]" 不进入此分支，按数组后缀处理）
104 ++j; // '['
105 j = skipTypeAnnotationFrom(inner, j);
106 if (j == std::string::npos || j >= inner.size() || inner[j] != ':')
107 return std::string::npos;
108 j = skipTypeAnnotationFrom(inner, j + 1);
109 if (j == std::string::npos || j >= inner.size() || inner[j] != ']')
110 return std::string::npos;
111 ++j;
112 } else if (base == "fun" && j < inner.size() && inner[j] == '(') {
113 // fun(params):ret：参数列表 ':' 返回类型（对齐 parseTypeAnnotation，
114 // 返回类型必填；',' 后紧跟 ')' 的尾随逗号同样被容忍）
115 ++j; // '('
116 if (j < inner.size() && inner[j] != ')') {
117 for (;;) {
118 j = skipTypeAnnotationFrom(inner, j);
119 if (j == std::string::npos)
120 return std::string::npos;
121 if (j < inner.size() && inner[j] == ',') {
122 ++j;
123 if (j < inner.size() && inner[j] == ')')
124 break; // 尾随逗号：',' 已消费，参数列表结束
125 continue;
126 }
127 break;
128 }
129 }
130 if (j >= inner.size() || inner[j] != ')')
131 return std::string::npos;
132 ++j;
133 if (j >= inner.size() || inner[j] != ':')
134 return std::string::npos;
135 j = skipTypeAnnotationFrom(inner, j + 1);
136 if (j == std::string::npos)
137 return std::string::npos;
138 }
139 // 后缀 [] 与 ? 任意堆叠（对齐 parseTypeAnnotation 尾循环）
140 for (;;) {
141 if (j + 1 < inner.size() && inner[j] == '[' && inner[j + 1] == ']')
142 j += 2;
143 else if (j < inner.size() && inner[j] == '?')
144 ++j;
145 else
146 break;
147 }
148 return j;
149}

151inline size_t findDictKVSeparator(const std::string& inner) {
152 const size_t end = skipTypeAnnotationFrom(inner, 0);
153 if (end == std::string::npos || end >= inner.size() || inner[end] != ':')
154 return std::string::npos;
155 return end;
156}

158inline bool typeMatchValue(const Value& val, const std::string& annotation) {
159 if (annotation.empty())
160 return true;
161 // null 兼容所有类型注解（2026-06-29 用户决策）
162 if (val.isNull())
163 return true;
164 // 可选类型 T?：剥离末尾 '?' 后递归匹配（对齐 Interpreter::typeMatch AUDIT-P2.7）
165 if (annotation.size() >= 2 && annotation.back() == '?') {
166 return typeMatchValue(val, annotation.substr(0, annotation.size() - 1));
167 }
168 if (annotation == TypeName::INT)
169 return val.isInt();
170 if (annotation == TypeName::FLOAT)
171 return val.isFloat() || val.isInt();
172 if (annotation == TypeName::BOOL)
173 return val.isBool();
174 if (annotation == TypeName::STRING)
175 return val.isString();
176 if (annotation == TypeName::ARRAY)
177 return val.isArray();
178 if (annotation == TypeName::DICT)
179 return val.isDict();
180 // R98 元组与解构：tuple 类型注解匹配
181 if (annotation == TypeName::TUPLE)
182 return val.isTuple();
183 if (annotation == TypeName::NULL_T)
184 return val.isNull();
185 // R99 枚举与 ADT：enum variant 类型注解匹配
186 // - "enum"：泛型注解，匹配任意 enum variant
187 // - "<EnumName>"：精确 enum 名匹配（如 "Color" 匹配 Color.Red/Color.Green 等）
188 if (annotation == TypeName::ENUM) {
189 return val.isEnumVariant();
190 }
191 if (val.isEnumVariant()) {
192 return val.enumVariantEnumName() == annotation;
193 }
194 // dict[K:V] 泛型字典类型注解：key 支持 string/int/bool/float（L4 fix），递归检查每个 value
195 // （对齐 Interpreter::typeMatch AUDIT-P2.7）
196 if (annotation.size() >= 7 && annotation.substr(0, 5) == "dict[" && annotation.back() == ']') {
197 if (!val.isDict())
198 return false;
199 std::string inner = annotation.substr(5, annotation.size() - 6);
200 // 文法同构切分（2026-10-22）：键类型自带冒号形态（dict[dict[..:K]:..] /
201 // dict[fun():K:string]）此前被首个 ':' 截断，键类型永不命中白名单 → 必拒
202 size_t colonPos = findDictKVSeparator(inner);
203 if (colonPos == std::string::npos)
204 return false;
205 std::string keyType = inner.substr(0, colonPos);
206 std::string valType = inner.substr(colonPos + 1);
207 for (const auto& kv : val.dictVal()) {
208 // L4 fix: kv.first 是 DictKey variant，按 keyType 检查键类型
209 if (keyType == "string") {
210 if (!std::holds_alternative<std::string>(kv.first))
211 return false;
212 } else if (keyType == "int") {
213 if (!std::holds_alternative<int64_t>(kv.first))
214 return false;
215 } else if (keyType == "bool") {
216 if (!std::holds_alternative<bool>(kv.first))
217 return false;
218 } else if (keyType == "float") {
219 if (!std::holds_alternative<double>(kv.first))
220 return false;
221 } else {
222 return false; // 未知 keyType
223 }
224 if (!typeMatchValue(kv.second, valType))
225 return false;
226 }
227 return true;
228 }
229 // 函数类型 fun(...):ret：仅检查是闭包即可（对齐 Interpreter::typeMatch AUDIT-P2.7，参数/返回类型不做验证）
230 if (annotation.size() >= 5 && annotation.substr(0, 4) == "fun(") {
231 return val.isClosure();
232 }
233 // 数组元素类型注解，如 "int[]"、多维 "int[][]"（后缀 [] 剥离递归，对齐 Interpreter::typeMatch AUDIT-P2-CORRECT）
234 if (annotation.size() >= 2 && annotation.substr(annotation.size() - 2) == "[]") {
235 if (!val.isArray())
236 return false;
237 std::string elemType = annotation.substr(0, annotation.size() - 2);
238 for (const auto& elem : val.arrayVal()) {
239 if (!typeMatchValue(elem, elemType))
240 return false;
241 }
242 return true;
243 }
244 // 实例类型注解：精确类名匹配（继承链由调用方扩展）
245 if (val.isInstance()) {
246 return val.className() == annotation;
247 }
248 return false;
249}

251/// R163 泛型扩展：判断类型名是否是类型参数（如 T、E、K、V）。
252/// 用于四后端（Interpreter/StackVM/RegisterVM/IR）的类型参数擦除——
253/// 类型参数运行时不校验，仅跳过类型检查。
254/// （注：曾与 Interpreter::isTypeParameter 成员构成双事实源，后者已于
255/// 2026-10-22 移除，本自由函数为唯一判定入口；复合注解请用
256/// annotationReferencesTypeParam，精确匹配仅适用于已剥离的基础类型名。）
257inline bool isTypeParameter(const std::string& typeName, const std::vector<std::string>& typeParams) {
258 for (const auto& tp : typeParams) {
259 if (typeName == tp) {
260 return true;
261 }
262 }
263 return false;
264}

266/// 泛型擦除补全（2026-10-07）：判断注解是否（递归地）引用了类型参数。
267/// R163 原实现仅精确匹配裸类型名，复合注解（T[] / T? / dict[string:T] / fun(T):T）
268/// 漏擦——Interpreter 侧 typeMatch 将 "T[]" 当具体数组元素类型校验而拒绝合法程序，
269/// 而 VM 路径因编译期发射的 OP_TYPE_CHECK 未发出/通过而接受（四后端不一致，
270/// 由 TestGenerics 探针发现）。规则：注解结构中任一基础类型是类型参数 →
271/// 整个注解视为泛型注解 → 整体跳过运行时校验（对齐 R163"类型参数运行时不校验"；
272/// dict 具体键的校验一并放弃，属记录的简化，重启条件 = 泛型运行时强检立项）。
273/// 注解字符串格式与 Parser::parseTypeAnnotation 的生成格式一一对应。
274// §3.3 fix（2026-10-22）：类型参数词法合并（去重）——嵌套函数/lambda/局部 enum 的
275// 定义期上下文 ∪ 自身声明。同名参数不重复入列（同名遮蔽 = 同一擦除语义，
276// annotationReferencesTypeParam 按名判定）；延迟调用（make<T> 返回 inner 后再调）
277// 以闭包携带的定义期快照为词法事实源（见 Value.h ClosureData::definingTypeParams）。
278inline std::vector<std::string> mergeTypeParams(std::vector<std::string> base,
279 const std::vector<std::string>& extra) {
280 for (const auto& tp : extra) {
281 if (std::find(base.begin(), base.end(), tp) == base.end())
282 base.push_back(tp);
283 }
284 return base;
285}

287inline bool annotationReferencesTypeParam(const std::string& annotation, const std::vector<std::string>& typeParams) {
288 if (annotation.empty() || typeParams.empty())
289 return false;
290 // T? 可选类型：剥离 '?' 后递归
291 if (annotation.back() == '?')
292 return annotationReferencesTypeParam(annotation.substr(0, annotation.size() - 1), typeParams);
293 // T[] / T[][] 数组：剥离 [] 后缀后递归（多维逐层剥离）
294 if (annotation.size() >= 2 && annotation.compare(annotation.size() - 2, 2, "[]") == 0)
295 return annotationReferencesTypeParam(annotation.substr(0, annotation.size() - 2), typeParams);
296 // dict[K:V]：键/值类型分别递归（内嵌嵌套 dict 由值类型递归覆盖）
297 if (annotation.size() >= 7 && annotation.compare(0, 5, "dict[") == 0 && annotation.back() == ']') {
298 std::string inner = annotation.substr(5, annotation.size() - 6);
299 // 文法同构切分（2026-10-22）：与 typeMatchValue 共用 findDictKVSeparator，
300 // 取代"首个 ':'"——键类型自带冒号形态（dict[dict[string:K]:int] /
301 // dict[fun():K:string]）此前被截断为 dict[string / fun()，擦除漏判 →
302 // 运行时校验必拒合法程序（四后端一致地错）
303 size_t colon = findDictKVSeparator(inner);
304 if (colon == std::string::npos)
305 return false;
306 return annotationReferencesTypeParam(inner.substr(0, colon), typeParams) ||
307 annotationReferencesTypeParam(inner.substr(colon + 1), typeParams);
308 }
309 // fun(p1,p2):ret：按括号深度切分参数列表与返回类型，逐段递归
310 // （嵌套 fun(...) 的 '(' ')' 与顶层配对，顶层 ',' 才是参数分隔）
311 if (annotation.size() >= 5 && annotation.compare(0, 4, "fun(") == 0) {
312 size_t depth = 1; // 已消费 "fun(" 的 '('
313 size_t i = 4;
314 std::vector<std::string> parts;
315 size_t start = i;
316 for (; i < annotation.size(); ++i) {
317 char c = annotation[i];
318 if (c == '(') {
319 ++depth;
320 } else if (c == ')') {
321 --depth;
322 if (depth == 0)
323 break; // 参数列表闭合
324 } else if (c == ',' && depth == 1) {
325 parts.push_back(annotation.substr(start, i - start));
326 start = i + 1;
327 }
328 }
329 if (i >= annotation.size())
330 return false; // 格式异常：保守走原校验路径
331 parts.push_back(annotation.substr(start, i - start)); // 末参数（含空参数列表的空串）
332 if (i + 1 < annotation.size() && annotation[i + 1] == ':')
333 parts.push_back(annotation.substr(i + 2)); // 返回类型
334 bool refs = false;
335 for (const auto& p : parts)
336 refs = refs || annotationReferencesTypeParam(p, typeParams);
337 return refs;
338 }
339 return isTypeParameter(annotation, typeParams);
340}

342/// 类型种类（预留，未来扩展）
343enum class TypeKind {
344 UNKNOWN, // 未推断（动态类型）
345 INT, // 整数
346 FLOAT, // 浮点数
347 BOOL, // 布尔
348 STRING, // 字符串
349 NULL_T, // null
350 ARRAY, // 数组（元素类型待推断）
351 DICT, // 字典（键/值类型待推断）
352 INSTANCE, // 类实例（类名在 className 中）
353 CLOSURE, // 闭包（参数/返回类型待推断）
354 ANY // 任意类型（类型通配，与所有类型兼容）
355};

357/// 类型信息（预留，未来扩展）
358struct TypeInfo {
359 TypeKind kind = TypeKind::UNKNOWN;
360 std::string className{}; // INSTANCE 类型的类名
361 std::vector<TypeInfo> typeArgs{}; // 泛型类型参数（ARRAY 的元素类型、CLOSURE 的参数/返回类型）
362
363 // 显式构造函数（非聚合化）：避免 TypeInfo{TypeKind::X} 部分初始化触发
364 // GCC -Wextra 的 missing-field-initializers 警告（CI -Werror 拦截）。
365 TypeInfo() = default;
366 explicit TypeInfo(TypeKind k) : kind(k) {}
367 TypeInfo(TypeKind k, std::string cls) : kind(k), className(std::move(cls)) {}
368
369 bool isNumeric() const { return kind == TypeKind::INT || kind == TypeKind::FLOAT; }
370
371 /// 类型兼容性检查：判断 other 类型的值是否可赋给 this 类型的变量。
372 /// 规则：UNKNOWN/ANY 兼容一切；float 接受 int（数值提升）；
373 /// ARRAY/DICT/INSTANCE 按 className/typeArgs 递归检查；其余严格匹配。
374 bool isCompatible(const TypeInfo& other) const {
375 // UNKNOWN 或 ANY 兼容一切（保守假设 / 通配符）
376 if (kind == TypeKind::UNKNOWN || kind == TypeKind::ANY)
377 return true;
378 if (other.kind == TypeKind::UNKNOWN || other.kind == TypeKind::ANY)
379 return true;
380 // null 兼容所有类型（MiniLang 动态语言惯例）
381 if (other.kind == TypeKind::NULL_T)
382 return true;
383 // float 接受 int（数值提升）
384 if (kind == TypeKind::FLOAT && other.kind == TypeKind::INT)
385 return true;
386 // 同类型直接兼容
387 if (kind == other.kind) {
388 // INSTANCE 需要类名匹配
389 if (kind == TypeKind::INSTANCE)
390 return className.empty() || other.className.empty() || className == other.className;
391 return true;
392 }
393 return false;
394 }

396 /// 从类型注解字符串构造 TypeInfo
397 static TypeInfo fromAnnotation(const std::string& ann) {
398 if (ann.empty())
399 return TypeInfo{TypeKind::UNKNOWN};
400 if (ann == "int")
401 return TypeInfo{TypeKind::INT};
402 if (ann == "float")
403 return TypeInfo{TypeKind::FLOAT};
404 if (ann == "bool")
405 return TypeInfo{TypeKind::BOOL};
406 if (ann == "string")
407 return TypeInfo{TypeKind::STRING};
408 if (ann == "null")
409 return TypeInfo{TypeKind::NULL_T};
410 if (ann == "array")
411 return TypeInfo{TypeKind::ARRAY};
412 if (ann == "dict")
413 return TypeInfo{TypeKind::DICT};
414 // 以 [] 结尾的数组元素类型
415 if (ann.size() >= 2 && ann.substr(ann.size() - 2) == "[]")
416 return TypeInfo{TypeKind::ARRAY};
417 // fun(...) 函数类型
418 if (ann.size() >= 5 && ann.substr(0, 4) == "fun(")
419 return TypeInfo{TypeKind::CLOSURE};
420 // 其他视为实例类型（类名）
421 return TypeInfo{TypeKind::INSTANCE, ann};
422 }
