24/// 滑动展开动画：从 0 高度滑动到 targetHeight，同时淡入
25/// 适用于底部面板/右侧面板的弹出
26// AUDIT-P2 fix: 动画去重——快速连续点击时停止旧动画，避免多个 QPropertyAnimation
27// 同时写入 maximumHeight 导致抖动、目标高度不正确。
28inline void animateShow(QWidget* panel, int targetHeight) {
29 if (!panel)
30 return;
31
32 // 停止正在运行的 maximumHeight 动画
33 const auto anims = panel->findChildren<QPropertyAnimation*>();
34 for (auto* a : anims) {
35 if (a->propertyName() == "maximumHeight" && a->targetObject() == panel) {
36 a->stop();
37 a->deleteLater();
38 }
39 }
40
41 panel->show();
42 panel->setMaximumHeight(0);
43 panel->adjustSize();
44
45 auto* group = new QParallelAnimationGroup(panel);
46 group->setProperty("targetHeight", targetHeight);
47
48 // 高度滑动
49 auto* heightAnim = new QPropertyAnimation(panel, "maximumHeight");
50 heightAnim->setDuration(DURATION_MS);
51 heightAnim->setStartValue(0);
52 heightAnim->setEndValue(targetHeight);
53 heightAnim->setEasingCurve(QEasingCurve::OutCubic);
54
55 // 透明度淡入（通过 GraphicsEffect 或直接用 windowOpacity 不适用于子控件，
56 // 改用 QWidget 的可见性即可，高度动画已足够流畅）
57 group->addAnimation(heightAnim);
58
59 // 动画结束后恢复最大高度约束（允许后续手动调整）
60 QObject::connect(group, &QParallelAnimationGroup::finished, panel, [panel]() {
61 panel->setMaximumHeight(16777215); // QWIDGETSIZE_MAX
62 });
63
64 group->start(QAbstractAnimation::DeleteWhenStopped);
65}

176/// 子页面切换淡入动画：通过 QGraphicsOpacityEffect 驱动 opacity 0→1
177/// 适用于 QStackedWidget 子页切换、列表选中详情刷新等场景
178/// 注：QGraphicsOpacityEffect 由 widget parent 自动释放，无需手动管理
179/// 安全改进：先停止 widget 上正在运行的 opacity 动画，避免多动画堆积
180/// 导致 opacity 属性被多个 QPropertyAnimation 同时写入（引发抖动/崩溃）
181inline void fadeInWidget(QWidget* widget, int duration = FADE_DURATION_MS) {
182 if (!widget)
183 return;
184 auto* effect = qobject_cast<QGraphicsOpacityEffect*>(widget->graphicsEffect());
185 if (!effect) {
186 effect = new QGraphicsOpacityEffect(widget);
187 widget->setGraphicsEffect(effect);
188 }
189 // 停止该 widget 子树中正在运行的 opacity 动画，防止动画堆积
190 const auto anims = widget->findChildren<QPropertyAnimation*>();
191 for (auto* a : anims) {
192 if (a->propertyName() == "opacity" && a->targetObject() == effect) {
193 a->stop();
194 a->deleteLater();
195 }
196 }
197 effect->setOpacity(0.0);
198 auto* anim = new QPropertyAnimation(effect, "opacity", widget);
199 anim->setDuration(duration);
200 anim->setStartValue(0.0);
201 anim->setEndValue(1.0);
202 anim->setEasingCurve(QEasingCurve::OutCubic);
203 anim->start(QAbstractAnimation::DeleteWhenStopped);
204}

206/// 轻量子页面滑入动画：通过 QPropertyAnimation 驱动 widget 的 pos 属性
207/// 从右侧偏移 24px 滑入到目标位置。相比 fadeInWidget：
208/// - 不使用 QGraphicsOpacityEffect，无离屏 pixmap 合成（O(1) vs O(n)）
209/// - 不受子 widget 数量影响，适合复杂面板（含滚动区域/列表/表格）的过渡
210/// 适用于 QStackedWidget 切页时的内容过渡，与 VS Code 编辑器切换节奏一致。
211/// 注：调用方需确保 widget 已被 QStackedWidget 设为 currentWidget，
212/// 其 geometry 已由 stacked layout 就位后再调用本函数。
213inline void slideInWidget(QWidget* widget, int duration = FADE_DURATION_MS) {
214 if (!widget)
215 return;
216 // 停止该 widget 上正在运行的 pos 动画，防止动画堆积
217 const auto anims = widget->findChildren<QPropertyAnimation*>();
218 for (auto* a : anims) {
219 if (a->propertyName() == "pos" && a->targetObject() == widget) {
220 a->stop();
221 a->deleteLater();
222 }
223 }
224 const QPoint finalPos = widget->pos();
225 const int offset = 24;
226 widget->move(finalPos.x() + offset, finalPos.y());
227 auto* anim = new QPropertyAnimation(widget, "pos", widget);
228 anim->setDuration(duration);
229 anim->setStartValue(QPoint(finalPos.x() + offset, finalPos.y()));
230 anim->setEndValue(finalPos);
231 anim->setEasingCurve(QEasingCurve::OutCubic);
232 // 动画结束后确保位置精确归位（防止动画中途被 resize 打断）
233 QObject::connect(anim, &QPropertyAnimation::finished, widget, [widget, finalPos]() { widget->move(finalPos); });
234 anim->start(QAbstractAnimation::DeleteWhenStopped);
235}
