26// WatchpointLibrary — 静态教学场景库
27// ============================================================
28// 6 个典型 watchpoint 场景，覆盖：
29// - 简单变量监视（写入即暂停）
30// - 循环计数器监视
31// - 字段监视（obj.field 写入）
32// - 条件 watchpoint（仅特定值写入时暂停）
33// - 多 watchpoint 协同
34// - 累加器监视（排查累加错误）
35//
36// 帮助学习者理解 watchpoint 机制：
37// - pre-execution peek：在 SET 类指令执行前检查写入目标
38// - 命中即暂停（与行断点不同，不依赖行号）
39// - 命中次数累计（即使条件不满足也计数）
40// - 支持变量监视与字段监视两种目标类型
41

311void WatchpointPanel::refreshLive() {
312 if (!controller_) {
313 liveStatusLabel_->setText(QString::fromUtf8("未绑定控制器"));
314 return;
315 }
316 // 双路径状态守卫：调试运行中（VM 或 Interpreter）且未暂停时不刷新
317 if (controller_->isRunning() && controller_->isDebugRun() && !controller_->isDebugPaused()) {
318 liveStatusLabel_->setText(QString::fromUtf8("状态：调试运行中（暂停后刷新）"));
319 return;
320 }
321 if (controller_->isVmRunning() && !controller_->isDebugPaused()) {
322 liveStatusLabel_->setText(QString::fromUtf8("状态：VM 运行中（暂停后刷新）"));
323 return;
324 }
325
326 // L19: getWatchpoints 返回值拷贝（支持 Interpreter 路径的 std::vector → QVector 转换）
327 QVector<WatchpointInfo> wps = controller_->getWatchpoints();
328 if (wps.isEmpty()) {
329 liveStatusLabel_->setText(QString::fromUtf8("无 Watchpoint"));
330 watchpointTable_->setRowCount(0);
331 return;
332 }
333
334 // 状态文本
335 bool isRunning = controller_->isRunning();
336 bool isDebugRun = controller_->isDebugRun();
337 bool isVmRunning = controller_->isVmRunning();
338 QString modeText;
339 if (isVmRunning) {
340 modeText = controller_->isVmRegisterMode() ? QString::fromUtf8("RegisterVM 调试中")
341 : QString::fromUtf8("StackVM 调试中");
342 } else if (isRunning && isDebugRun) {
343 modeText = QString::fromUtf8("Interpreter 调试中");
344 } else {
345 modeText = QString::fromUtf8("未运行");
346 }
347 liveStatusLabel_->setText(QString::fromUtf8("共 %1 个 Watchpoint | %2").arg(wps.size()).arg(modeText));
348
349 watchpointTable_->setRowCount(wps.size());
350 for (int i = 0; i < wps.size(); ++i) {
351 const WatchpointInfo& wp = wps[i];
352 QString kindText =
353 (wp.kind == WatchpointTargetKind::Field) ? QString::fromUtf8("字段") : QString::fromUtf8("变量");
354 watchpointTable_->setItem(i, 0, new QTableWidgetItem(kindText));
355 watchpointTable_->setItem(i, 1, new QTableWidgetItem(QString::fromUtf8(wp.varName.c_str())));
356 watchpointTable_->setItem(i, 2,
357 new QTableWidgetItem(wp.fieldName.empty() ? QString::fromUtf8("—")
358 : QString::fromUtf8(wp.fieldName.c_str())));
359 watchpointTable_->setItem(i, 3,
360 new QTableWidgetItem(wp.condition.empty() ? QString::fromUtf8("（无条件）")
361 : QString::fromUtf8(wp.condition.c_str())));
362 watchpointTable_->setItem(i, 4, new QTableWidgetItem(QString::number(wp.hitCount)));
363 // 状态：综合显示
364 QString status;
365 if (wp.isConditional()) {
366 status = QString::fromUtf8("条件 Watchpoint");
367 } else if (wp.kind == WatchpointTargetKind::Field) {
368 status = QString::fromUtf8("字段 Watchpoint");
369 } else {
370 status = QString::fromUtf8("变量 Watchpoint");
371 }
372 watchpointTable_->setItem(i, 5, new QTableWidgetItem(status));
373 }

376void WatchpointPanel::populateScenarioDetail(int index) {
377 const auto& items = WatchpointLibrary::scenarios();
378 if (index < 0 || index >= (int)items.size()) {
379 currentSampleCode_.clear();
380 return;
381 }
382 const auto& s = items[index];
383 currentSampleCode_ = s.sampleCode;
384
385 // 散文式字段（description）走 Markdown 渲染
386 QString descFragment = MarkdownRenderer::markdownToHtmlFragment(s.description);
387
388 // HTML 转义——sampleCode/condition 含 < > 字符
389 auto escapeHtml = [](const std::string& str) -> std::string {
390 std::string out;
391 out.reserve(str.size());
392 for (char c : str) {
393 switch (c) {
394 case '&':
395 out += "&amp;";
396 break;
397 case '<':
398 out += "&lt;";
399 break;
400 case '>':
401 out += "&gt;";
402 break;
403 case '"':
404 out += "&quot;";
405 break;
406 case '\'':
407 out += "&#39;";
408 break;
409 default:
410 out += c;
411 break;
412 }
413 }
414 return out;
415 };
416
417 std::ostringstream os;
418 os << "<h3>" << escapeHtml(s.title) << "</h3>";
419 os << "<p><b>监视目标：</b> <code>" << escapeHtml(s.target) << "</code></p>";
420 if (!s.condition.empty()) {
421 os << "<p><b>条件表达式：</b> <code>" << escapeHtml(s.condition) << "</code></p>";
422 }
423 os << "<hr>";
424 os << descFragment.toStdString();
425 os << "<p><b>触发行为：</b> " << escapeHtml(s.expectedBehavior) << "</p>";
426 os << "<h4>示例代码：</h4>";
427 os << "<pre style='background:" << TeachingTheme::surface().name().toStdString()
428 << "; padding:8px; font-family:\"Cascadia Code\",\"Cascadia Mono\",\"Consolas\",\"JetBrains Mono\",\"Source "
429 "Code Pro\",\"Menlo\",\"DejaVu Sans Mono\",\"Courier New\",monospace;'>"
430 << escapeHtml(s.sampleCode) << "</pre>";
431 scenarioDetail_->setHtml(QString::fromUtf8(os.str().c_str()));
432}

434void WatchpointPanel::addWatchpointFromInput() {
435 if (!controller_)
436 return;
437 std::string varName = varNameEdit_->text().toStdString();
438 if (varName.empty()) {
439 liveStatusLabel_->setText(QString::fromUtf8("错误：变量名不能为空"));
440 return;
441 }
442 WatchpointInfo wp;
443 int kindData = kindCombo_->currentData().toInt();
444 wp.kind = static_cast<WatchpointTargetKind>(kindData);
445 wp.varName = varName;
446 if (wp.kind == WatchpointTargetKind::Field) {
447 std::string fieldName = fieldNameEdit_->text().toStdString();
448 if (fieldName.empty()) {
449 liveStatusLabel_->setText(QString::fromUtf8("错误：字段类型需指定字段名"));
450 return;
451 }
452 wp.fieldName = fieldName;
453 }
454 std::string cond = conditionEdit_->text().toStdString();
455 if (!cond.empty()) {
456 wp.condition = cond;
457 }
458 controller_->setWatchpoint(wp);
459 // 清空输入
460 varNameEdit_->clear();
461 fieldNameEdit_->clear();
462 conditionEdit_->clear();
463 refreshLive();
464}
