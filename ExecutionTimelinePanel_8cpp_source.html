204 if (recordBtn_)

206 } else {
207 traceRecorder().endSession();
208 refreshTimer_->stop();

212 if (traceRecorder().size() == 0)

216 // 通知 IdeController 联动 VmStepper / Interpreter 的 recorder 启停
217 // backend_ 由 IdeController 在 setBackendType() 时同步，captureVmStep/captureInterpreterStep
218 // 接收 backend 参数写入每步快照（recorder 本身不持有全局 backend 字段）
219 emit recordingToggled(checked, backend_);
220}

269void ExecutionTimelinePanel::refreshStepList() {
270 size_t total = traceRecorder().size();
271 if (total == lastSeenSize_) {
272 // 数量未变，仅更新步数标签
273 updateStepLabel();
274 return;
275 }
276
277 // 增量追加新步骤（避免全量重建）
278 // 注意：ring buffer 满后旧快照被 erase(begin())，索引会偏移。
279 // 简化处理：若总数量小于已显示数量（ring buffer 偏移），全量重建。
280 if (total < lastSeenSize_ || total - lastSeenSize_ > 100) {
281 // 全量重建
282 stepList_->clear();
283 auto snapshots = traceRecorder().allSnapshots();
284 for (const auto& snap : snapshots) {
285 stepList_->addItem(formatStepItem(snap));
286 }
287 } else {
288 // 增量追加
289 for (size_t i = lastSeenSize_; i < total; ++i) {
290 auto snap = traceRecorder().stepAt(i);
291 if (snap) {
292 stepList_->addItem(formatStepItem(*snap));
293 }
294 }
295 }
296 lastSeenSize_ = total;
297 updateStepLabel();
298 updateSliderRange();
299 updateControlsEnabled();
300
301 // 若用户未选中任何步骤，自动选中最后一步（跟踪最新状态）
302 if (currentStepIdx_ < 0 && total > 0) {
303 navigateToStep(static_cast<int>(total) - 1);
304 }
305}

335// 将导航按钮 / 清空按钮的可用性与当前数据 / 位置同步，避免空数据时可点
336// 、已在首尾时方向键无意义。UX 优化的一部分。
337void ExecutionTimelinePanel::updateControlsEnabled() {
338 const size_t total = traceRecorder().size();
339 const bool hasData = total > 0;
340 const bool recording = recordBtn_ && recordBtn_->isChecked();
341 const int last = static_cast<int>(total) - 1;
342 if (clearBtn_)
343 clearBtn_->setEnabled(hasData && !recording);
344 // 首 / 上一步：当前不在第 0 步且有数据时可用
345 if (firstBtn_)
346 firstBtn_->setEnabled(hasData && currentStepIdx_ != 0);
347 if (prevBtn_)
348 prevBtn_->setEnabled(hasData && currentStepIdx_ != 0);
349 // 下一 / 末步：当前不在末步且有数据时可用
350 if (nextBtn_)
351 nextBtn_->setEnabled(hasData && currentStepIdx_ != last);
352 if (lastBtn_)
353 lastBtn_->setEnabled(hasData && currentStepIdx_ != last);
354}

367void ExecutionTimelinePanel::navigateToStep(int idx) {
368 size_t total = traceRecorder().size();
369 if (total == 0) {
370 currentStepIdx_ = -1;
371 updateStepLabel();
372 return;
373 }
374 // 钳制到合法范围
375 if (idx < 0)
376 idx = 0;
377 if (idx >= static_cast<int>(total))
378 idx = static_cast<int>(total) - 1;
379 currentStepIdx_ = idx;
380
381 // 同步 UI 控件（避免信号回环）
382 // D10 fix: QSignalBlocker RAII 替代手工 blockSignals(true/false) 配对
383 if (slider_->value() != idx) {
384 const QSignalBlocker sliderBlocker(slider_);
385 slider_->setValue(idx);
386 }
387 if (stepList_->currentRow() != idx) {
388 const QSignalBlocker stepListBlocker(stepList_);
389 stepList_->setCurrentRow(idx);
390 }
391 // 滚动到当前行可见
392 stepList_->scrollToItem(stepList_->item(idx), QAbstractItemView::PositionAtCenter);
393
394 showStepDetail(static_cast<size_t>(idx));
395 updateStepLabel();
396 updateControlsEnabled();
397}

424QString ExecutionTimelinePanel::formatDetailHtml(const TraceSnapshot& snap) {
425 QString html;
426 html += "<html><body style='font-family: Consolas, monospace; font-size: 10pt;'>";
427 html += QString("<h3 style='color:#268BD2; margin:0 0 8px 0;'>Step #%1 — %2</h3>")
428 .arg(snap.step)
429 .arg(backendName(snap.backend));
430 html += QString("<table style='margin-bottom:8px;'>"
431 "<tr><td style='color:#8C8C8C;'>行号:</td><td>%1</td>"
432 "<td style='color:#8C8C8C; padding-left:16px;'>IP:</td><td>%2</td>"
433 "<td style='color:#8C8C8C; padding-left:16px;'>帧深:</td><td>%3</td></tr>"
434 "<tr><td style='color:#8C8C8C;'>操作:</td><td colspan='5'>%4</td></tr>"
435 "</table>")
436 .arg(snap.line)
437 .arg(snap.ip)
438 .arg(snap.frameCount)
439 .arg(QString::fromStdString(snap.opOrNodeName));
440
441 // 操作数栈 / 寄存器
442 const auto& stack = (snap.backend == TraceBackend::RegisterVM && !snap.registerStrings.empty())
443 ? snap.registerStrings
444 : snap.stackStrings;
445 if (!stack.empty()) {
446 html += "<h4 style='color:#859900; margin:8px 0 4px 0;'>栈 / 寄存器</h4><div style='background:#F5F5F5; "
447 "padding:4px; border:1px solid #E0E0E0;'>";
448 for (size_t i = 0; i < stack.size(); ++i) {
449 html += QString("<div><span style='color:#8C8C8C;'>[%1]</span> %2</div>")
450 .arg(i)
451 .arg(QString::fromStdString(stack[i]).toHtmlEscaped());
452 }
453 html += "</div>";
454 }
455
456 // 可见变量
457 if (!snap.visibleVarsStrings.empty()) {
458 html += "<h4 style='color:#B58900; margin:8px 0 4px 0;'>可见变量</h4><table style='width:100%;'>";
459 for (const auto& kv : snap.visibleVarsStrings) {
460 html += QString("<tr><td style='color:#268BD2; width:35%;'>%1</td><td>%2</td></tr>")
461 .arg(QString::fromStdString(kv.first).toHtmlEscaped())
462 .arg(QString::fromStdString(kv.second).toHtmlEscaped());
463 }
464 html += "</table>";
465 }
466
467 // 调用栈
468 if (!snap.callStack.empty()) {
469 html += "<h4 style='color:#DC322F; margin:8px 0 4px 0;'>调用栈</h4>";
470 for (size_t i = 0; i < snap.callStack.size(); ++i) {
471 const auto& f = snap.callStack[i];
472 html += QString("<div style='margin:2px 0; padding:2px 4px; background:%1;'>"
473 "<b>%2</b> @ L%3 (depth=%4)</div>")
474 .arg(i == snap.callStack.size() - 1 ? "#CCE4F7" : "#F5F5F5")
475 .arg(QString::fromStdString(f.functionName).toHtmlEscaped())
476 .arg(f.line)
477 .arg(f.depth);
478 if (!f.locals.empty()) {
479 html += "<table style='margin-left:16px;'>";
480 for (const auto& kv : f.locals) {
481 html += QString("<tr><td style='color:#8C8C8C;'>%1</td><td>%2</td></tr>")
482 .arg(QString::fromStdString(kv.first).toHtmlEscaped())
483 .arg(QString::fromStdString(kv.second.toString()).toHtmlEscaped());
484 }
485 html += "</table>";
486 }
487 }
488 }
489
490 // 全局变量
491 if (!snap.globalsStrings.empty()) {
492 html += "<h4 style='color:#8C8C8C; margin:8px 0 4px 0;'>全局变量</h4><table style='width:100%;'>";
493 for (const auto& kv : snap.globalsStrings) {
494 html += QString("<tr><td style='color:#268BD2; width:35%;'>%1</td><td>%2</td></tr>")
495 .arg(QString::fromStdString(kv.first).toHtmlEscaped())
496 .arg(QString::fromStdString(kv.second).toHtmlEscaped());
497 }
498 html += "</table>";
499 }
500
501 html += "</body></html>";
502 return html;
503}
