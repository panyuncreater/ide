1/**
2 * @file parser/Parser.h
3 * @brief MiniLang 递归下降语法分析器。
4 *
5 * 将 Token 流解析为 AST（抽象语法树），根节点为 Block。支持：
6 * - 表达式优先级链（assignment → or → and → equality → comparison
7 * → term → factor → unary → call → primary）
8 * - 控制流（if/elif/else、while、for-in、break、continue）
9 * - 函数声明与 lambda（含默认参数、类型注解）
10 * - 类与继承（class/extends、字段、方法、super）
11 * - 模块系统（import/export，含路径安全校验、循环依赖检测）
12 * - 异常处理（try/catch/finally/throw）
13 * - 字符串插值（解析 `\‍(expr)` 嵌套表达式为 InterpolatedString AST 节点）
14 *
15 * 错误恢复：
16 * - parse() 中 catch ParseError 后调用 synchronize() 跳到下个语句边界继续
17 * - 单条语句错误不影响后续解析，支持 IDE 多错误一次性展示
18 *
19 * DoS 防护：
20 * - MAX_PARSE_DEPTH（256）：递归深度上限，防止恶意嵌套栈溢出
21 * - MAX_BLOCK_DEPTH（256）：块嵌套深度上限
22 *
23 * @see Lexer ASTNode Formatter
24 */

75 // C4 fix: 统一的递归深度 RAII 守卫，消除 6 处局部结构体重复定义。
76 // 构造时递增深度计数器，析构时递减（异常路径也自动递减，防止深度计数泄漏）。

78 int& depth;

80 ~DepthGuard() { --depth; }

85 // R164 协程/生成器：yieldId 计数器 RAII 守卫。
86 // 进入生成器函数 funDecl 时构造，保存旧值并重置为 0；析构时恢复旧值。
87 // 支持嵌套生成器函数（外层 fun* 内定义内层 fun*）的独立 yieldId 计数。
88 // 异常安全：ParseError 抛出时 ~YieldIdScope 自动恢复外层计数器。
89 struct YieldIdScope {
90 int& counter;
91 int saved;
92 explicit YieldIdScope(int& c) : counter(c), saved(c) {}
93 ~YieldIdScope() { counter = saved; }
94 YieldIdScope(const YieldIdScope&) = delete;
95 YieldIdScope& operator=(const YieldIdScope&) = delete;
96 };
97
98 // R164 协程/生成器：bool 字段 RAII 守卫（异常安全保存/恢复）。
99 // 用于 yieldInLoop_（whileStmt/forStmt 入口置 true）和
100 // currentFunHasYieldInLoop_（funDecl 入口重置为 false）。
101 struct BoolScope {
102 bool& flag;
103 bool saved;
104 explicit BoolScope(bool& f) : flag(f), saved(f) {}
105 ~BoolScope() { flag = saved; }
106 BoolScope(const BoolScope&) = delete;
107 BoolScope& operator=(const BoolScope&) = delete;
108 };

116 // R164 协程/生成器：当前生成器函数内 yield 编号计数器。
117 // -1 表示不在生成器上下文（普通函数/全局作用域），>=0 表示当前生成器已分配的 yield 数。
118 // 由 YieldIdScope 在 funDecl 入口保存/重置/恢复，primary() 解析 yield 时分配并递增。

120 // R164 协程/生成器：当前是否处于循环体内部（while/for body）。
121 // 由 BoolScope 在 whileStmt/forStmt 入口保存/置 true/恢复。用于检测 yield-in-loop：
122 // 循环内 yield 的编译期 yieldCount 只是 AST 节点数（1），但运行时可能执行 N 次，
123 // 导致 callCoroutineNext 的 `currentYieldId >= yieldCount` 提前判定 done。
124 // 检测到此类生成器时，yieldCount 设为 INT_MAX（kDynamicYieldCount），
125 // done 改由函数体自然结束路径判定。
126 bool yieldInLoop_ = false;
127 // R164 协程/生成器：当前生成器函数体内是否存在 yield-in-loop。
128 // 由 BoolScope 在 funDecl 入口保存/重置/恢复。primary() 解析 yield 时若 yieldInLoop_
129 // 为 true 则置本字段为 true。funDecl 结尾若本字段为 true，decl->yieldCount = INT_MAX。
130 bool currentFunHasYieldInLoop_ = false;
131 // 七特性 MVP 阶段 4：当前是否在 async fun 体内（await 仅在此上下文合法）。
132 // 由 BoolScope 在 funDecl 入口保存/设置/恢复，嵌套函数独立判定
133 // （async fun 内的普通嵌套 fun 体内 await 非法）。
134 bool currentFunIsAsync_ = false;
