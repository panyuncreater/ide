9// ============================================================
10// SmallMap: 内联平坦存储优化的小型映射容器
11// ============================================================
12// 当条目数 ≤ N 时使用内联数组（线性扫描），超过 N 时回退到 std::unordered_map。
13// 针对 Environment 变量存储优化：大多数作用域仅有 1-8 个变量，
14// 内联模式避免了 unordered_map 的桶数组堆分配（~64 字节最小开销）。
15// 线性扫描对于 ≤8 条目比哈希更快（缓存友好，无哈希计算开销）。
16//
17// AUDIT-R3 P2 契约警示（与 std::unordered_map 的关键差异）：
18// 1. 引用/迭代器失效：第 N+1 次插入触发内联→堆晋升（promoteToHeap），
19// 此前通过 operator[]/try_emplace/find 获得的引用与迭代器全部失效
20// （仍指向已被重置为默认值的内联槽位：读到默认值、写入静默丢失）。
21// unordered_map 保证 rehash 后引用仍有效，本容器不保证。
22// 调用方若需长期持有 V*（如 Environment::boundInstance_），必须保证
23// 持有期间不再新增条目，或在插入后重新定位（参考 H2 fix）。
24// 2. erase 为换尾删除（O(1) 不保序）：遍历中删除会漏访/重访元素，
25// 不要在迭代中 erase。
26

237 template <typename... Args> std::pair<iterator, bool> try_emplace(const K& key, Args&&... args) {
238 if (useHeap_) {
239 auto [it, inserted] = heap_->try_emplace(key, std::forward<Args>(args)...);
240 return {iterator(it), inserted};
241 }
242 // 内联模式：先查找是否已存在
243 for (uint8_t i = 0; i < inlineSize_; ++i) {
244 if (inline_[i].first == key) {
245 return {iterator(&inline_[i]), false};
246 }
247 }
248 // 不存在，需要插入
249 if (inlineSize_ < N) {
250 // 内联空间足够
251 inline_[inlineSize_].first = key;
252 inline_[inlineSize_].second = V(std::forward<Args>(args)...);
253 auto it = iterator(&inline_[inlineSize_]);
254 ++inlineSize_;
255 return {it, true};
256 }
257 // 内联空间满，迁移到堆
258 promoteToHeap();
259 auto [it, inserted] = heap_->try_emplace(key, std::forward<Args>(args)...);
260 return {iterator(it), inserted};
261 }
