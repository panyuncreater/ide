3 * @brief 中间表示层（ARCH-06 完整实现）。

5 * ARCH-06 fix: 引入 IR 抽象层，打破 Compiler 直接生成字节码的紧耦合。
6 * 长期目标：AST → IR → 后端（VM 字节码 / 寄存器字节码 / 未来 JS / WASM）。
7 *
8 * 数据结构层次：
9 * - IROp：IR 指令操作码（约 50 种，覆盖赋值/算术/比较/跳转/调用/

17 * 接口层：
18 * - IRBuilder：抽象接口，将 AST 转换为 IRModule
19 * - AstIRBuilder：具体实现，AST → IRModule
20 * - IRBackend：抽象接口，将 IRModule 转换为目标后端字节码
21 * - BytecodeIRBackend：IR → 栈式 VM 字节码 BytecodeChunk
22 * - RegisterBytecodeBackend：IR → 寄存器式 VM 字节码 RegBytecodeChunk
23 *
24 * IR 优化 pass（C4 增强）：
25 * - constantFoldingPass：常量折叠（编译期可计算的算术/比较）
26 * - copyPropagationPass：复制传播（vreg_a = vreg_b 后续用 vreg_b 替代 vreg_a）
27 * - deadCodeEliminationPass：死代码消除（无副作用指令的 dest vreg 无引用则删除）
28 * - commonSubexpressionEliminationPass：公共子表达式消除（基本块内值编号）
29 * - loopUnrollingPass：循环展开（for 循环体 ≤20 指令且 N≤4 时展开）
30 * - optimizeIR：组合 pass，按 round 重复执行直到收敛（最多 3 轮）
31 *
32 * 设计原则：
33 * - IR 与现有字节码解耦，可独立扩展为 SSA / Dataflow 等高级形式
34 * - 现有 AST→字节码路径保持不变，IR 作为可选中间层（Compiler::setUseIR(true)）
35 * - IRBuilder/IRBackend 为抽象接口，便于未来添加新前端和新后端
36 *
37 * 安全约束：
38 * - Phase-5 fix: POP 指令现携带所消费的 vreg 操作数（对齐 IR spec `POP: [src_vreg]`），
39 * 使 DCE 能看到 POP 对 vreg 的消费关系。因此 DCE/CSE 现在对栈式 VM 后端也是安全的。
40 * - DCE 不能删除带副作用的指令（除零/溢出/LOAD_GLOBAL 未定义等），

44 *
45 * @see Compiler AstIRBuilder BytecodeIRBackend RegisterBytecodeBackend
46 */

119/// 指令操作数容器：≤4 个操作数内联存储免堆分配，超出回落堆缓冲。
120/// 第四轮优化：原 std::vector<IROperand> 使每条 IRInstruction 携带一次独立
121/// 堆分配（IR 构建 / pass 复制 / lowering 全链路），绝大多数指令操作数
122/// ≤4 个（LOAD_CONST/BINOP/JUMP/STORE_LOCAL 等），内联后零分配且提升
123/// 局部性（操作数与指令同处缓存行）。IROperand 为平凡可复制类型，
124/// 实现按 memcpy 语义复制，无元素级构造/析构分叉。
125///
126/// 接口刻意收敛到 IR 层现有使用面（size/empty/operator[]/push_back/
127/// 迭代/initializer_list 赋值），不实现完整 vector 接口——新调用点若
128/// 需要 erase/insert 等能力应先评估是否真适用于 IR 操作数语义。
129class IROperandList {
130public:
131 IROperandList() = default;
132 IROperandList(std::initializer_list<IROperand> ops) {
133 for (const auto& op : ops)
134 push_back(op);
135 }
136 IROperandList(const IROperandList& other) { copyFrom(other); }
137 IROperandList(IROperandList&& other) noexcept { moveFrom(other); }
138 IROperandList& operator=(const IROperandList& other) {
139 if (this != &other) {
140 destroy();
141 copyFrom(other);
142 }
143 return *this;
144 }
145 IROperandList& operator=(IROperandList&& other) noexcept {
146 if (this != &other) {
147 destroy();
148 moveFrom(other);
149 }
150 return *this;
151 }
152 IROperandList& operator=(std::initializer_list<IROperand> ops) {
153 clear();
154 for (const auto& op : ops)
155 push_back(op);
156 return *this;
157 }
158 ~IROperandList() { destroy(); }
159
160 size_t size() const { return size_; }
161 bool empty() const { return size_ == 0; }
162 const IROperand& operator[](size_t i) const { return data()[i]; }
163 IROperand& operator[](size_t i) { return data()[i]; }
164 void push_back(const IROperand& op) {
165 if (size_ < kInlineCapacity && !heap_) {
166 inline_[size_] = op;
167 ++size_;
168 return;
169 }
170 if (heap_ && size_ < heapCapacity_) {
171 heap_[size_] = op;
172 ++size_;
173 return;
174 }
175 grow();
176 heap_[size_++] = op;
177 }
178 void clear() {
179 size_ = 0; // IROperand 平凡可复制，无元素析构；heap_ 保留复用
180 }
181 /// 预留至少 n 个元素的容量（n ≤ 内联容量且未上堆时为 no-op）
182 void reserve(size_t n) {
183 if (n <= kInlineCapacity && !heap_)
184 return;
185 if (heap_ && n <= heapCapacity_)
186 return;
187 size_t newCapacity = heapCapacity_ == 0 ? kInlineCapacity * 2 : heapCapacity_;
188 while (newCapacity < n)
189 newCapacity *= 2;
190 auto* newHeap = new IROperand[newCapacity];
191 for (size_t i = 0; i < size_; ++i)
192 newHeap[i] = data()[i];
193 delete[] heap_;
194 heap_ = newHeap;
195 heapCapacity_ = static_cast<uint32_t>(newCapacity);
196 }
197 IROperand* begin() { return data(); }
198 IROperand* end() { return data() + size_; }
199 const IROperand* begin() const { return data(); }
200 const IROperand* end() const { return data() + size_; }
201
202private:
203 static constexpr size_t kInlineCapacity = 4;
204
205 IROperand* data() { return heap_ ? heap_ : inline_; }
206 const IROperand* data() const { return heap_ ? heap_ : inline_; }
207 void grow() {
208 size_t newCapacity = heapCapacity_ == 0 ? kInlineCapacity * 2 : heapCapacity_ * 2;
209 auto* newHeap = new IROperand[newCapacity];
210 for (size_t i = 0; i < size_; ++i)
211 newHeap[i] = data()[i];
212 delete[] heap_;
213 heap_ = newHeap;
214 heapCapacity_ = static_cast<uint32_t>(newCapacity);
215 }
216 void copyFrom(const IROperandList& other) {
217 size_ = other.size_;
218 heapCapacity_ = other.heapCapacity_;
219 if (other.heap_) {
220 heap_ = new IROperand[other.heapCapacity_];
221 for (uint32_t i = 0; i < size_; ++i)
222 heap_[i] = other.heap_[i];
223 } else {
224 for (uint32_t i = 0; i < size_; ++i)
225 inline_[i] = other.inline_[i];
226 }
227 }
228 void moveFrom(IROperandList& other) noexcept {
229 size_ = other.size_;
230 heapCapacity_ = other.heapCapacity_;
231 heap_ = other.heap_;
232 if (!other.heap_) {
233 for (uint32_t i = 0; i < size_; ++i)
234 inline_[i] = other.inline_[i];
235 }
236 other.size_ = 0;
237 other.heap_ = nullptr;
238 other.heapCapacity_ = 0;
239 }
240 void destroy() {
241 delete[] heap_;
242 heap_ = nullptr;
243 heapCapacity_ = 0;
244 size_ = 0;
245 }
246
247 uint32_t size_ = 0;
248 uint32_t heapCapacity_ = 0; // 0 = 未使用堆
249 IROperand* heap_ = nullptr;
250 IROperand inline_[kInlineCapacity] = {}; // 聚合值初始化（IROperand 无用户构造函数）
251};

529 /// 添加常量，返回索引（已存在则复用）
530 uint32_t addConstant(const Value& v) {
531 // P2-1 fix: 字符串与非字符串常量都用 hash 侧表 O(1) 查找。
532 // 字符串：以 stringVal() 为 key（O(n) 内容比较的 equals 在大常量池下退化）
533 // 数值/布尔：以 std::pair<int64_t,double>+类型 标识为 key，避免线性扫描。
534 // null/Instance/Array/Dict 等复杂类型：保留线性扫描（此类常量极少出现）
535 if (v.isString()) {
536 auto it = stringConstIdx_.find(v.stringVal());
537 if (it != stringConstIdx_.end())
538 return it->second;
539 } else if (v.isInt() || v.isFloat() || v.isBool()) {
540 auto it = scalarConstIdx_.find(scalarKey(v));
541 if (it != scalarConstIdx_.end())
542 return it->second;
543 } else {
544 for (size_t i = 0; i < constants.size(); ++i) {
545 if (constants[i].equals(v))
546 return static_cast<uint32_t>(i);
547 }
548 }
549 // 与 BytecodeChunk/RegBytecodeChunk 一致：常量池索引需 fit 到 uint16_t，
550 // 超限抛异常以避免调用方 static_cast<uint16_t> 静默截断。
551 if (constants.size() >= 65535) {
552 throw std::runtime_error("IR 常量池索引超出 65535 上限");
553 }
554 constants.push_back(v);
555 uint32_t idx = static_cast<uint32_t>(constants.size() - 1);
556 if (v.isString()) {
557 stringConstIdx_[v.stringVal()] = idx;
558 } else if (v.isInt() || v.isFloat() || v.isBool()) {
559 scalarConstIdx_[scalarKey(v)] = idx;
560 }
561 return idx;
562 }
563 /// 添加全局变量名/字段名/函数名，返回索引
564 uint32_t addGlobal(const std::string& globalName) {
565 // perf3 fix: hash 侧表 O(1) 查找替代 O(n) 线性扫描

567 if (it != globalNameIdx_.end())
568 return it->second;
569 // P2-2 fix: 与 addConstant 对齐——globalNames 索引经 writeShort 编码为 uint16_t，
570 // 超限抛异常以避免调用方 static_cast<uint16_t> 静默截断（读写错误全局名）。
571 if (globalNames.size() >= 65535) {
572 throw std::runtime_error("IR 全局名池索引超出 65535 上限");
573 }
574 globalNames.push_back(globalName);
575 uint32_t idx = static_cast<uint32_t>(globalNames.size() - 1);
576 globalNameIdx_[globalName] = idx;
577 return idx;
578 }

580 IRBasicBlock& addBlock(uint32_t labelIdx) {
581 blocks.push_back({});
582 blocks.back().labelIndex = labelIdx;
583 return blocks.back();
584 }
