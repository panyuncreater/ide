252void GcVisualizerPanel::populateTheory() {
253 // 理论概念 HTML
254 QString html = QStringLiteral("<h2>GcManager 垃圾回收原理</h2>"
255 "<p>MiniLang 的堆类型通过侵入式 <b>RefCounted</b> 基类管理引用计数，"
256 "但引用计数<b>无法回收循环引用</b>（如 a.push(a) 形成的自环）。"
257 "<b>GcManager</b> 实现轻量级同步<b>mark-sweep</b>，周期性扫描所有 tracked 容器节点，"
258 "标记从根集可达的对象，释放不可达的循环孤岛。</p>"
259
260 "<h3>一、mark-sweep-finalize 三阶段</h3>"
261 "<p><b>GcManager::collectCycle(roots)</b> 同步阻塞执行三阶段：</p>"
262 "<ul>"
263 "<li><b>Phase 1 - Marking</b>：从 roots 出发深度优先 mark 所有可达容器节点。"
264 "markValue 递归遍历 array / dict / instance / tuple / enum variant / closure 内嵌的"
265 " Value，marked 集合记录可达对象指针。</li>"
266 "<li><b>Phase 2 - Sweeping</b>：遍历 slots_ 登记表，对槽位仍被占用但 marked 未命中的"
267 "对象（即不可达循环孤岛）执行回收。RefCountWithCycleGc 模式清空其子元素打破循环"
268 "（由后续 refCount 归零释放）；GcOnly 模式直接 delete。</li>"
269 "<li><b>Phase 3 - Finalizing</b>：原地压缩 slots_ 登记表——存活对象前移并回写"
270 " gcSlot_，死亡槽位置空回收进 freeSlots_ 复用；无悬垂指针残留。</li>"
271 "</ul>"
272 "<p>collectCycle 结束后 currentPhase_ 回归 Idle；UI 在 animTimer 周期内观察到的常态"
273 "即 Idle，lastMarkedCount / lastCollectedCount 反映上次 GC 结果。</p>"
274
275 "<h3>二、三种 GcMode</h3>"
276 "<p>GcManager 支持三种内存管理策略（仅 reset() 后切换安全）：</p>"
277 "<ul>"
278 "<li><b>RefCountOnly</b>：纯引用计数，跳过 registerTracked，collectCycle 为 no-op。"
279 "循环引用会泄漏直至进程退出。基线对比用。</li>"
280 "<li><b>RefCountWithCycleGc</b>（默认）：引用计数主导 + GC 仅回收循环孤岛。"
281 "sweep 清空不可达孤岛的子元素打破循环，由后续 refCount 归零释放。项目历史模式。</li>"
282 "<li><b>GcOnly</b>：GC 主导，sweep 阶段直接 delete 不可达对象。"
283 "实验性，存在 COW / VM root 限制。</li>"
284 "</ul>"
285
286 "<h3>三、引用计数与 COW 的关系</h3>"
287 "<p>COW（Copy-On-Write）基于 RefCounted 引用计数，而非 GC：</p>"
288 "<ul>"
289 "<li><b>共享</b>：赋值时浅拷贝指针，refCount 加 1，不复制底层数据。</li>"
290 "<li><b>写前 detach</b>：修改前检查 refCount，若 &gt;1 则复制底层数据（detach），"
291 "使修改方独占副本。</li>"
292 "<li><b>COW 克隆也注册 GcManager</b>：detach 后克隆出的新容器会调用"
293 " registerTracked，否则 COW 路径产生的循环引用不会被回收，导致永久内存泄漏。</li>"
294 "</ul>"
295 "<p>引用计数管「正常生命周期」，GcManager 管「引用计数管不到的循环」。"
296 "——两者协同覆盖所有场景。</p>"
297
298 "<h3>教学价值</h3>"
299 "<p>本面板可视化 MiniLang 内存模型中 GC 的核心概念，帮助理解：</p>"
300 "<ul>"
301 "<li>mark-sweep-finalize 三阶段如何协同回收循环引用孤岛</li>"
302 "<li>三种 GcMode 的差异及适用场景</li>"
303 "<li>引用计数、COW、GC 三者的分工与协作关系</li>"
304 "</ul>"
305 "<p style='color:#666;font-size:small;'>"
306 "提示：切换到「② 交互式 GC 模拟器」子页，选择预设场景体验 GC 流程。</p>");
307 theoryBrowser_->setHtml(html);
308

312 for (int i = 0; i < static_cast<int>(phases.size()); ++i) {

314 phaseTable_->setItem(i, 0, new QTableWidgetItem(QString::fromUtf8(p.phase.c_str())));
315 phaseTable_->setItem(i, 1, new QTableWidgetItem(QString::fromUtf8(p.stateField.c_str())));

445GcSimResult GcVisualizerPanel::runScenario(int idx) {
446 GcSimResult result;
447 const auto& scenarios = GcVisualizerLibrary::scenarios();
448 if (idx < 0 || idx >= static_cast<int>(scenarios.size())) {
449 result.summary = "无效的场景索引";
450 return result;
451 }
452 result.scenarioTitle = scenarios[idx].title;
453 result.scenarioCode = scenarios[idx].codeSnippet;
454
455 // 公共前置：完全重置，避免历史状态污染
456 // ARCH-10: 通过 MemoryInspectionAPI 间接调用 GcManager，消除直接依赖
457 MemoryInspectionAPI::resetGc();
458 MemoryInspectionAPI::setGcMode(GcMode::RefCountWithCycleGc);
459
460 switch (idx) {
461 case 0: {
462 // 场景 1：简单孤岛 — 两个互相引用的数组
463 result.modeUsed = GcMode::RefCountWithCycleGc;
464 {
465 auto s = gcStats();
466 result.steps.push_back({"reset 后", s.trackedCount, 0, 0, s.totalGcCount});
467 }
468 {
469 Value a(std::vector<Value>{});
470 Value b(std::vector<Value>{});
471 a.arrayVal().push_back(b); // a → b
472 b.arrayVal().push_back(a); // b → a（循环）
473 // a, b 离开作用域：refCount 各降 1，但互相引用使 refCount=1 无法归零
474 }
475 {
476 auto s = gcStats();
477 result.steps.push_back({"构造循环并离开作用域", s.trackedCount, 0, 0, s.totalGcCount});
478 }
479 {
480 std::vector<const void*> emptyRoots;
481 MemoryInspectionAPI::collectCycle(emptyRoots);
482 }
483 {
484 auto s = gcStats();
485 result.steps.push_back(
486 {"collectCycle(空 roots)", s.trackedCount, s.lastMarkedCount, s.lastCollectedCount, s.totalGcCount});
487 result.expectationMet = s.lastCollectedCount > 0;
488 }
489 result.summary = result.expectationMet ? "✅ 期望达成：循环引用孤岛被 mark-sweep 成功回收，"
490 "引用计数无法处理的循环由 GcManager 接管。"
491 : "❌ 期望未达成：循环引用孤岛未被回收。";
492 break;
493 }
494 case 1: {
495 // 场景 2：多层循环 — A→B→C→A 三层循环
496 result.modeUsed = GcMode::RefCountWithCycleGc;
497 {
498 auto s = gcStats();
499 result.steps.push_back({"reset 后", s.trackedCount, 0, 0, s.totalGcCount});
500 }
501 {
502 Value a(std::vector<Value>{});
503 Value b(std::vector<Value>{});
504 Value c(std::vector<Value>{});
505 a.arrayVal().push_back(b); // A → B
506 b.arrayVal().push_back(c); // B → C
507 c.arrayVal().push_back(a); // C → A（三层循环）
508 // 离开作用域：三个数组互相引用，refCount=1，无法归零
509 }
510 {
511 auto s = gcStats();
512 result.steps.push_back({"构造三层循环并离开作用域", s.trackedCount, 0, 0, s.totalGcCount});
513 }
514 {
515 std::vector<const void*> emptyRoots;
516 MemoryInspectionAPI::collectCycle(emptyRoots);
517 }
518 {
519 auto s = gcStats();
520 result.steps.push_back(
521 {"collectCycle(空 roots)", s.trackedCount, s.lastMarkedCount, s.lastCollectedCount, s.totalGcCount});
522 result.expectationMet = s.lastCollectedCount > 0;
523 }
524 result.summary = result.expectationMet ? "✅ 期望达成：三层循环整条链被 mark-sweep 回收，"
525 "mark 阶段从空根集出发无可达对象，sweep 回收全部循环节点。"
526 : "❌ 期望未达成：多层循环未被回收。";
527 break;
528 }
529 case 2: {
530 // 场景 3：可达保留 — 循环引用但仍有外部根
531 result.modeUsed = GcMode::RefCountWithCycleGc;
532 {
533 auto s = gcStats();
534 result.steps.push_back({"reset 后", s.trackedCount, 0, 0, s.totalGcCount});
535 }
536 {
537 // arr 在内层作用域持有，作为 root
538 Value arr(std::vector<Value>{});
539 arr.arrayVal().push_back(arr); // 自循环
540 {
541 auto s = gcStats();
542 result.steps.push_back({"构造自循环（arr 仍存活）", s.trackedCount, 0, 0, s.totalGcCount});
543 }
544 std::vector<const void*> roots = {arr.gcRootPtr()};
545 MemoryInspectionAPI::collectCycle(roots); // arr 作为 root 标记可达
546 auto s = gcStats();
547 result.steps.push_back(
548 {"collectCycle(roots={arr})", s.trackedCount, s.lastMarkedCount, s.lastCollectedCount, s.totalGcCount});
549 result.expectationMet = (s.lastCollectedCount == 0 && s.trackedCount >= 1);
550 // arr 即将离开作用域：释放后仅剩自循环引用，refCount=1
551 }
552 // arr 已释放，自循环孤岛仍存活（refCount=1），再次 collectCycle 才回收
553 {
554 std::vector<const void*> emptyRoots;
555 MemoryInspectionAPI::collectCycle(emptyRoots);
556 }
557 {
558 auto s = gcStats();
559 result.steps.push_back({"arr 释放后再 collectCycle(空 roots)", s.trackedCount, s.lastMarkedCount,
560 s.lastCollectedCount, s.totalGcCount});
561 }
562 result.summary = result.expectationMet
563 ? "✅ 期望达成：作为 root 的循环数组未被 sweep 回收"
564 "（mark 阶段从 roots 出发标记可达容器）；arr 释放后再 collectCycle 才回收。"
565 : "❌ 期望未达成：可达容器被错误回收。";
566 break;
567 }
568 case 3: {
569 // 场景 4：增量 GC — 多次分配 + GC，trackedCount 趋势
570 result.modeUsed = GcMode::RefCountWithCycleGc;
571 {
572 auto s = gcStats();
573 result.steps.push_back({"reset 后", s.trackedCount, 0, 0, s.totalGcCount});
574 }
575 for (int i = 0; i < 5; ++i) {
576 {
577 Value arr(std::vector<Value>{});
578 arr.arrayVal().push_back(arr); // 自循环
579 }
580 std::vector<const void*> emptyRoots;
581 MemoryInspectionAPI::collectCycle(emptyRoots);
582 auto s = gcStats();
583 result.steps.push_back({"第 " + std::to_string(i + 1) + " 轮 collectCycle 后", s.trackedCount,
584 s.lastMarkedCount, s.lastCollectedCount, s.totalGcCount});
585 }
586 size_t finalTracked = gcStats().trackedCount;
587 result.expectationMet = (finalTracked <= 1);
588 result.summary = result.expectationMet
589 ? "✅ 期望达成：5 轮 collectCycle 后 trackedCount 稳定（≤1），"
590 "Finalizing 阶段压缩 slots_ 登记表仅保留存活节点，无悬垂指针残留。"
591 : "❌ 期望未达成：trackedCount 异常增长，Finalizing 阶段未正确压缩登记表。";
592 break;
593 }
594 default:
595 result.summary = "未知场景索引";
596 break;
597 }
598
599 // 公共清理：恢复干净状态（reset 不重置 gcMode_，需显式恢复默认）
600 MemoryInspectionAPI::resetGc();
601 MemoryInspectionAPI::setGcMode(GcMode::RefCountWithCycleGc);
602 return result;
603}

653void GcVisualizerPanel::onRunSimulation() {
654 // AUDIT-R5 R7 fix (BUG-2): 模拟器 runScenario 会 reset() 全局 GcManager 单例。
655 // 若此时 Worker 线程正在执行用户程序，reset 会清空 slots_ 登记表，
656 // 破坏其 collectCycle 的存活判定（可能提前回收 UAF 或统计崩坏）。
657 // 运行期拒绝模拟（与 BackendExecutionService 的 AUDIT-R2 P1-5 防护策略对齐）。
658 if (controller_ && (controller_->isRunning() || controller_->isVmRunning())) {
659 simBrowser_->setHtml(
660 mlTr("<p style='color:red;'>程序正在运行，GC 模拟会重置全局 GcManager 单例并破坏运行中的对象跟踪。"
661 "请先停止执行后再运行模拟。</p>"));
662 return;
663 }
664 int idx = presetCombo_->currentIndex();
665 if (idx < 0) {
666 simBrowser_->setHtml(mlTr("<p style='color:red;'>请先选择一个预设场景</p>"));
667 return;
668 }
669 GcSimResult result = runScenario(idx);
670 renderScenarioResult(result);
671 refreshStats();
672}
