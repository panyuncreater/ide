3// ============================================================
4// TCO.h — 尾调用优化（TCO）共享识别函数
5// ------------------------------------------------------------
6// 三后端（StackVM 直接路径 / IR 路径 / RegisterVM 路径 / Interpreter 路径）
7// 共享的 AST 结构识别逻辑。L15 扩展支持：
8// - 自递归函数调用 return f(args)（含 upvalue / 默认参数）
9// - 类方法自调用 return this.method(args)（保留 this/字段槽，覆盖参数槽）
10//
11// 识别条件（必须同时满足，才允许 TCO 优化）：
12// 1. return 语句的值是 FunCall 或 MethodCall 节点
13// 2. 对 FunCall：callee == nullptr（直接名称调用 f(args)，非链式 expr(args)）
14// 且 FunCall.name == 当前函数名（SelfFunction）
15// 3. 对 MethodCall：object 是 VarRef("this") 且 methodName == 当前方法名（SelfMethod）
16// 4. 不在 try 块内（由调用方检查 tryDepth_ == 0）
17// 5. currentFunctionDecl_ 非空（由调用方检查，用于获取形参/默认值）
18//
19// L15 放宽的限制（由调用方检查）：
20// - 闭包 upvalue：SelfFunction/SelfMethod 安全（同函数重用同一闭包，
21// upvalue 指向外层作用域，递归不覆盖外层栈槽）
22// - 默认参数：args.size() < params.size() 时用默认值填充缺失参数
23// - 类方法：SelfMethod 保留 slot 0 (this) 和 slot 1..N (字段)，仅覆盖参数槽
24//
25// 仍不支持的形态（需要 VM 层 OP_TAIL_CALL，留作后续工作）：
26// - 互递归 return g(args)（g != 当前函数名，跨 chunk 跳转）
27// - 链式调用 return f(x)(y)（callee != nullptr，闭包目标编译期未知）
28//
29// 设计原则：
30// - 保守识别：仅匹配可直接 JUMP 回当前函数入口的尾调用形态
31// - 三后端一致：所有后端使用同一识别函数，确保语义等价
32// - 无副作用：纯函数，仅读取 AST 结构
33// ============================================================

41struct TailCallInfo {
42 /// 尾调用类型
43 enum class Kind {
44 None, ///< 非尾调用或不支持的形式
45 SelfFunction, ///< return f(args) 自递归函数调用（f == 当前函数名）
46 SelfMethod, ///< return this.method(args) 类方法自调用（method == 当前方法名）

48 ///< 直接名称调用）。需 VM 层 OP_TAIL_CALL/REG_TAIL_CALL
49 ///< 帧复用；调用方需额外检查遮蔽/返回类型注解/非方法体。
50 };

55 /// SelfMethod: 对应的 MethodCall 节点（非拥有，AST 生命周期内有效）

59/// 检查 return 语句是否为可优化的尾调用。
60///
61/// 仅检查 AST 结构。调用方需自行检查：
62/// - tryDepth_ == 0（不在 try-finally / try-catch 块内）
63/// - currentFunctionDecl_ != nullptr（用于获取形参/默认值列表）
64///
65/// L15 放宽：不再要求 currentUpvalues_.empty()（SelfFunction/SelfMethod 安全）
66/// 不再要求 args.size() == params.size()（默认参数填充）
67/// 不再排除类方法（SelfMethod 保留 this/字段槽）
68///
69/// @param node ReturnStmt 节点（可为 nullptr，返回 Kind::None）
70/// @param currentFuncName 当前函数/方法的简单名（不含 "ClassName." 前缀）
71/// @param isMethod 当前是否在类方法体内（true 时识别 SelfMethod）
72/// @return TailCallInfo 描述尾调用类型与节点指针
73inline TailCallInfo identifyTailCall(ReturnStmt* node, const std::string& currentFuncName, bool isMethod) {
74 TailCallInfo info;
75 if (!node || !node->value || currentFuncName.empty()) {
76 return info;
77 }
78
79 // SelfFunction: return f(args) — 直接名称调用且名称匹配当前函数
80 // L18 GeneralCall: return g(args) — 直接名称调用但名称不匹配（互递归/一般尾调用）
81 if (node->value->nodeType == NodeType::NODE_FUN_CALL) {
82 auto* call = static_cast<FunCall*>(node->value.get());
83 // 仅支持直接名称调用：f(args)，不支持链式 expr(args)
84 if (call->callee == nullptr) {
85 if (call->name == currentFuncName && !isMethod) {
86 info.kind = TailCallInfo::Kind::SelfFunction;
87 info.call = call;
88 return info;
89 }
90 if (call->name != currentFuncName) {
91 info.kind = TailCallInfo::Kind::GeneralCall;
92 info.call = call;
93 return info;
94 }
95 }
96 }
97
98 // SelfMethod: return this.method(args) — 方法自调用
99 // object 必须是 VarRef("this")，确保接收者就是当前实例（slot 0 不变）
100 if (node->value->nodeType == NodeType::NODE_METHOD_CALL && isMethod) {
101 auto* methodCall = static_cast<MethodCall*>(node->value.get());
102 if (methodCall->methodName == currentFuncName && methodCall->object &&
103 methodCall->object->nodeType == NodeType::NODE_VAR_REF) {
104 auto* objVar = static_cast<VarRef*>(methodCall->object.get());
105 if (objVar->name == "this") {
106 info.kind = TailCallInfo::Kind::SelfMethod;
107 info.methodCall = methodCall;
108 return info;
109 }
110 }
111 }
112
113 return info;
114}

116/// [废弃] 旧的仅识别自递归函数调用的接口，保留向后兼容。
117/// 新代码应使用 identifyTailCall。
118inline bool isTailRecursiveReturn(ReturnStmt* node, const std::string& funcName) {
119 if (!node || !node->value)
120 return false;
121 if (node->value->nodeType != NodeType::NODE_FUN_CALL)
122 return false;
123 auto* call = static_cast<FunCall*>(node->value.get());
124 if (call->callee != nullptr)
125 return false;
126 return call->name == funcName;
127}
