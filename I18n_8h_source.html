1/**
2 * @file gui/I18n.h
3 * @brief 国际化（i18n）辅助层。
4 *
5 * D2: 提供统一的字符串翻译入口，使 GUI 模块可在不继承 QObject 的情况下
6 * 使用 Qt Linguist 翻译管线。当 MINILANG_ENABLE_I18N 未启用时，所有
7 * mlTr 调用编译期退化为 QString::fromUtf8，保持向后兼容与零运行时开销。
8 *
9 * R117: 新增运行时语言切换 API（switchMiniLangLanguage / currentLocale /
10 * availableLocales），支持菜单栏动态切换 UI 语言；新增 QSettings 持久化
11 * 用户语言偏好（ui/locale 键），覆盖系统 locale 自动检测。
12 *
13 * 用法：
14 * 1. CMake 启用 i18n：cmake -DMINILANG_ENABLE_I18N=ON ...
15 * 2. 源码中包裹用户可见字符串：
16 * label->setText(mlTr("打开文件夹"));
17 * 或在带 context 的场景：
18 * label->setText(mlTrCtx("CodeEditor", "行号"));
19 * 或在 QObject 派生类中直接用 tr("打开文件夹")。
20 * 3. lupdate 自动同步 .ts 条目，lrelease 编译为 .qm。
21 * 4. main.cpp 在 QCoreApplication::setApplicationName 后调用
22 * loadMiniLangTranslations() 加载 .qm 文件（自动读 QSettings 优先级）。
23 * 5. 运行时切换：switchMiniLangLanguage("en_US") 卸载旧 translator 加载新 locale。
24 * 切换后需通知所有 widget retranslateUi() 刷新文本（由 Ide 类订阅实现）。
25 *
26 * 翻译上下文约定：
27 * - "MiniLang" — 顶层菜单/窗口标题/全局对话框（默认）
28 * - "CodeEditor" — 代码编辑器相关
29 * - "ReplPanel" — REPL 面板
30 * - "AstViewer" — AST 查看器
31 * - "IdeController"— 控制器层错误消息
32 *
33 * 实现说明：
34 * mlTr/mlTrCtx 是宏，i18n 启用时展开为 QCoreApplication::translate，
35 * 使 lupdate 能正确识别并提取字符串（lupdate 不识别自定义函数调用）。
36 * 未启用时展开为 QString::fromUtf8，行为与现有代码完全等价。
37 *
38 * @see app/translations/minilang_zh_CN.ts
39 * @see app/translations/minilang_en_US.ts (R117 占位，待 lupdate 同步填充)
40 */

77/// 加载翻译文件（main.cpp 在 QCoreApplication 实例化后调用）
78/// 查找路径顺序：
79/// 1. 可执行文件同目录/translations/
80/// 2. 可执行文件同目录
81/// 3. 当前工作目录
82/// @param locale 目标 locale（如 "zh_CN"），默认读 QSettings → 系统 locale
83/// @return 加载成功返回 true；找不到 .qm 或翻译为空时返回 false
84inline bool loadMiniLangTranslations(const QString& locale = QString()) {
85 // 优先级：参数 > QSettings 持久化偏好 > 系统 locale
86 QString loc = locale;
87 if (loc.isEmpty()) {
88 QSettings s;
89 loc = s.value(kLocaleSettingsKey).toString();
90 }
91 if (loc.isEmpty()) {
92 loc = QLocale::system().name();
93 }
94 QString qmName = QString("minilang_%1.qm").arg(loc);
95
96 QDir appDir(QCoreApplication::applicationDirPath());
97 QStringList searchPaths = {
98 appDir.absoluteFilePath("translations/" + qmName),
99 appDir.absoluteFilePath(qmName),
100 QDir::current().absoluteFilePath(qmName),
101 };
102
103 for (const QString& path : searchPaths) {
104 QFileInfo fi(path);
105 if (fi.exists() && fi.isFile()) {
106 auto* translator = new QTranslator(QCoreApplication::instance());
107 if (translator->load(path)) {
108 // 卸载旧的 translator（若存在）
109 if (g_minilang_translator) {
110 QCoreApplication::removeTranslator(g_minilang_translator);
111 delete g_minilang_translator;
112 }
113 QCoreApplication::installTranslator(translator);
114 g_minilang_translator = translator;
115 g_minilang_current_locale = loc;
116 return true;
117 }
118 delete translator;
119 }
120 }
121 // 加载失败时仍记录 locale（用于 availableLocales 报告与持久化）
122 // 不卸载已有 translator，保留上次成功加载的语言
123 g_minilang_current_locale = loc;
124 return false;
125}

127/// 运行时切换 UI 语言（R117）
128/// @param locale 目标 locale（如 "en_US"），空字符串表示恢复系统 locale
129/// @return 切换成功（找到并加载 .qm）返回 true
130/// @note 切换后需调用方通知所有 widget retranslateUi() 刷新文本
131inline bool switchMiniLangLanguage(const QString& locale) {
132 QSettings s;
133 if (locale.isEmpty()) {
134 s.remove(kLocaleSettingsKey);
135 } else {
136 s.setValue(kLocaleSettingsKey, locale);
137 }
138 return loadMiniLangTranslations(locale);
139}

147/// 扫描 translations/ 目录中可用的 locale 列表（R117）
148/// @return locale 字符串列表（如 ["zh_CN", "en_US"]），不含 .qm 扩展名
149/// @note 始终包含 "zh_CN"（源语言，即使无 .qm 也回退到源字符串可用）
150inline QStringList availableMiniLangLocales() {
151 QStringList result;
152 QDir appDir(QCoreApplication::applicationDirPath());
153 QDir trDir(appDir.absoluteFilePath("translations"));
154 QStringList filters;
155 filters << "minilang_*.qm";
156 QFileInfoList files = trDir.entryInfoList(filters, QDir::Files);
157 for (const QFileInfo& fi : files) {
158 // minilang_zh_CN.qm → zh_CN
159 QString base = fi.completeBaseName(); // minilang_zh_CN
160 QString loc = base.mid(QString("minilang_").length());
161 if (!loc.isEmpty() && !result.contains(loc)) {
162 result.append(loc);
163 }
164 }
165 // 始终包含源语言（zh_CN）作为兜底
166 if (!result.contains("zh_CN")) {
167 result.prepend("zh_CN");
168 }
169 return result;
170}
