91/// 通用加法（FLOAT/FLOAT, STRING/STRING, STRING/其他, 其他/STRING）
92/// @param leftBits 左操作数 raw bits
93/// @param rightBits 右操作数 raw bits
94/// @return 结果 raw bits（FLOAT 或 STRING）
95extern "C" uint64_t jitAddGeneric(JitContext* ctx, uint64_t leftBits, uint64_t rightBits) {
96 Value left = bitsToValue(leftBits);
97 Value right = bitsToValue(rightBits);
98 // 字符串拼接（StackVM numericOp OP_ADD_INT 路径对齐）
99 if (left.isString() && right.isString()) {
100 const auto& ls = left.stringVal();
101 const auto& rs = right.stringVal();
102 std::string concat;
103 concat.reserve(ls.size() + rs.size());
104 concat.append(ls).append(rs);
105 return jitReturn(Value(std::move(concat)));
106 }
107 if (left.isString() || right.isString()) {
108 std::string result;
109 if (left.isString()) {
110 const auto& ls = left.stringVal();
111 auto rs = right.toString();
112 result.reserve(ls.size() + rs.size());
113 result.append(ls).append(rs);
114 } else {
115 auto ls = left.toString();
116 const auto& rs = right.stringVal();
117 result.reserve(ls.size() + rs.size());
118 result.append(ls).append(rs);
119 }
120 return jitReturn(Value(std::move(result)));
121 }
122 // R145 修复：整数加法（INT + INT，INT48 溢出时从 JIT 原生路径降级到此）
123 // StackVM 的 OverflowCheck 检查 INT64 范围，INT48 范围由 Value(int64_t) 自动装箱
124 if (left.isInt() && right.isInt()) {
125 int64_t l = left.intVal();
126 int64_t r = right.intVal();
127 if (OverflowCheck::addOverflow(l, r)) {
128 // INT64 溢出：通过 ctx 报告错误（与 jitDivGeneric 除零路径一致），
129 // 由 emitCallBinaryHelper 的 checkError 路径转为可捕获异常，不再静默返回 0
130 if (ctx && ctx->hasError && ctx->errorBuffer) {
131 *ctx->hasError = true;
132 *ctx->errorBuffer = "整数运算溢出";
133 }
134 return jitReturn(Value(0));
135 }
136 return jitReturn(Value(l + r)); // Value(int64_t) 自动处理 int48 装箱
137 }
138 // 浮点加法（StackVM computeArith 路径对齐）
139 if (left.isFloat() || right.isFloat()) {
140 return jitReturn(Value(left.toDouble() + right.toDouble()));
141 }
142 // 不支持的类型组合
143 return jitReturn(Value(0.0));
144}

248/// 通用有序比较（FLOAT/FLOAT, STRING/STRING）
249/// @param cmpType 0=less, 1=greater, 2=less_equal, 3=greater_equal
250/// @return NaN-boxing BOOL raw bits
251extern "C" uint64_t jitOrderedCompare(uint64_t leftBits, uint64_t rightBits, int64_t cmpType) {
252 Value left = bitsToValue(leftBits);
253 Value right = bitsToValue(rightBits);
254 bool result = false;
255 if (left.isString() && right.isString()) {
256 const auto& ls = left.stringVal();
257 const auto& rs = right.stringVal();
258 switch (cmpType) {
259 case 0:
260 result = ls < rs;
261 break;
262 case 1:
263 result = ls > rs;
264 break;
265 case 2:
266 result = ls <= rs;
267 break;
268 case 3:
269 result = ls >= rs;
270 break;
271 default:
272 result = false;
273 break;
274 }
275 } else {
276 double l = left.toDouble();
277 double r = right.toDouble();
278 switch (cmpType) {
279 case 0:
280 result = l < r;
281 break;
282 case 1:
283 result = l > r;
284 break;
285 case 2:
286 result = l <= r;
287 break;
288 case 3:
289 result = l >= r;
290 break;
291 default:
292 result = false;
293 break;
294 }
295 }
296 return jitReturn(Value(result));

300// R146 阶段 4：数组类型支持（C++ 辅助函数）
301// ============================================================
302// 设计策略：数组涉及堆对象（ArrayData）+ COW（Copy-On-Write）+ 引用计数，
303// 原生 JIT 机器码无法高效处理，全部走 C++ 辅助路径。
304//
305// JIT 栈访问机制：JIT 代码在调用数组辅助函数前，用 `mov [r12+48], r15`
306// 把当前操作数栈顶指针写入 ctx->stackTop。辅助函数通过 ctx->stackTop
307// 读写 JIT 栈（向低地址增长：push = sp -= 1; *sp = val; pop = val = *sp; sp += 1）。
308// 调用后 JIT 代码用 `mov r15, [r12+48]` 恢复 r15（辅助函数可能修改了栈顶）。
309//
310// 辅助函数约定：
311// - 错误通过 ctx->hasError + ctx->errorBuffer 报告（越界、类型不匹配等）
312// - 返回值为结果的 raw bits（OP_INDEX_GET 返回元素，OP_INDEX_SET 返回变异后的容器）
313// - 引用计数：bitsToValue addRef，jitReturn detach 转移所有权
314
315/// OP_BUILD_ARRAY: 从 JIT 栈 pop count 个元素，构造 ArrayData，push 数组 raw bits
316/// @param ctx JitContext 指针（访问 stackTop 读写 JIT 栈）
317/// @param count 元素个数（1 字节，最多 255）
318extern "C" void jitBuildArray(JitContext* ctx, uint8_t count) {
319 if (!ctx || !ctx->stackTop) {
320 return;
321 }
322 int64_t* sp = ctx->stackTop;
323 std::vector<Value> elements(count);
324 // 栈顶是最后一个元素，逆序 pop 填入正确位置
325 for (int i = count - 1; i >= 0; --i) {
326 elements[i] = bitsToValue(*sp);
327 sp += 1; // pop（向高地址移动）
328 }
329 Value arr(std::move(elements));
330 // push 数组 raw bits
331 sp -= 1;
332 *sp = valueToBits(arr); // valueToBits(detach) 转移所有权，arr 被置 null 避免析构 release
333 ctx->stackTop = sp; // 写回更新后的栈顶
334}

336/// OP_INDEX_GET: 读取容器元素（支持 array/dict/string/tuple 多态 + 越界检查）
337/// @param ctx JitContext 指针（错误报告用）
338/// @param objBits 容器 raw bits
339/// @param idxBits 索引 raw bits
340/// @return 元素 raw bits；错误时设置 hasError 并返回 NULL bits
341extern "C" uint64_t jitIndexGet(JitContext* ctx, uint64_t objBits, uint64_t idxBits) {
342 Value obj = bitsToValue(objBits);
343 Value idx = bitsToValue(idxBits);
344 // 数组索引读取（最常见的路径）
345 if (obj.isArray() && idx.isInt()) {
346 const auto& arr = obj.arrayVal(); // const 重载，不触发 COW
347 int64_t i = idx.intVal();
348 if (i < 0 || static_cast<size_t>(i) >= arr.size()) {
349 if (ctx && ctx->hasError && ctx->errorBuffer) {
350 *ctx->hasError = true;
351 *ctx->errorBuffer =
352 "数组索引越界: " + std::to_string(i) + ", 有效范围 [0, " + std::to_string(arr.size()) + ")";
353 }
354 return jitReturn(Value()); // NULL
355 }
356 return jitReturn(arr[static_cast<size_t>(i)]); // addRef by bitsToValue in jitReturn
357 }
358 // 字符串索引读取（R161 fixup: 对齐 StackVM 的 ASCII 快速路径 + UTF-8 慢路径）
359 if (obj.isString() && idx.isInt()) {
360 const auto& str = obj.stringVal();
361 int64_t i = idx.intVal();
362
363 // ASCII 快速路径：纯 ASCII 字符串直接按字节索引 O(1)
364 if (obj.isAsciiString()) {
365 if (i < 0 || static_cast<size_t>(i) >= str.size()) {
366 if (ctx && ctx->hasError && ctx->errorBuffer) {
367 *ctx->hasError = true;
368 *ctx->errorBuffer =
369 "字符串索引越界: " + std::to_string(i) + ", 有效范围 [0, " + std::to_string(str.size()) + ")";
370 }
371 return jitReturn(Value());
372 }
373 return jitReturn(Value(std::string(1, str[static_cast<size_t>(i)])));
374 }
375
376 // UTF-8 慢路径：遍历码位对齐 StackVM executeIndexGet
377 size_t charCount = 0;
378 size_t bytePos = 0;
379 size_t targetBytePos = 0;
380 size_t targetByteLen = 0;
381 bool found = false;
382 while (bytePos < str.size()) {
383 size_t charLen = Utf8::byteLength(static_cast<unsigned char>(str[bytePos]));
384 if (static_cast<size_t>(i) == charCount) {
385 targetBytePos = bytePos;
386 targetByteLen = charLen;
387 found = true;
388 }
389 bytePos += charLen;
390 charCount++;
391 }
392 if (i < 0 || !found) {
393 if (ctx && ctx->hasError && ctx->errorBuffer) {
394 *ctx->hasError = true;
395 *ctx->errorBuffer = "字符串索引越界: " + std::to_string(i) + ", 有效范围 [0, " +
396 std::to_string(static_cast<long long>(charCount)) + ")";
397 }
398 return jitReturn(Value());
399 }
400 return jitReturn(Value(str.substr(targetBytePos, targetByteLen)));
401 }
402 // R147 阶段 4b：字典索引读取（键支持 string/int/bool/float，与 StackVM 对齐）
403 if (obj.isDict()) {
404 auto dk = Value::dictKeyFromValue(idx);
405 if (!dk) {
406 if (ctx && ctx->hasError && ctx->errorBuffer) {
407 *ctx->hasError = true;
408 *ctx->errorBuffer = ErrorMessages::kDictKeyInvalidType;
409 }
410 return jitReturn(Value());
411 }
412 const auto& dict = obj.dictVal(); // const 重载，不触发 COW
413 auto it = dict.find(*dk);
414 if (it == dict.end()) {
415 return jitReturn(Value()); // 不存在键返回 null（与 StackVM 一致）
416 }
417 return jitReturn(it->second);
418 }
419 // R147 阶段 4b：元组索引读取（immutable，与数组语义一致）
420 if (obj.isTuple() && idx.isInt()) {
421 const auto& tup = obj.tupleVal(); // const 重载
422 int64_t i = idx.intVal();
423 if (i < 0 || static_cast<size_t>(i) >= tup.size()) {
424 if (ctx && ctx->hasError && ctx->errorBuffer) {
425 *ctx->hasError = true;
426 *ctx->errorBuffer =
427 "元组索引越界: " + std::to_string(i) + ", 有效范围 [0, " + std::to_string(tup.size()) + ")";
428 }
429 return jitReturn(Value());
430 }
431 return jitReturn(tup[static_cast<size_t>(i)]);
432 }
433 if (obj.isArray()) {
434 if (ctx && ctx->hasError && ctx->errorBuffer) {
435 *ctx->hasError = true;
436 *ctx->errorBuffer = ErrorMessages::kArrayIndexMustBeInt;
437 }
438 return jitReturn(Value());
439 }
440 if (obj.isString()) {
441 if (ctx && ctx->hasError && ctx->errorBuffer) {
442 *ctx->hasError = true;
443 *ctx->errorBuffer = ErrorMessages::kStringIndexMustBeInt;
444 }
445 return jitReturn(Value());
446 }
447 if (obj.isTuple()) {
448 if (ctx && ctx->hasError && ctx->errorBuffer) {
449 *ctx->hasError = true;
450 *ctx->errorBuffer = "元组索引必须是整数";
451 }
452 return jitReturn(Value());
453 }
454 // 其他类型组合暂不支持
455 if (ctx && ctx->hasError && ctx->errorBuffer) {
456 *ctx->hasError = true;
457 *ctx->errorBuffer = ErrorMessages::kTypeNotIndexable;
458 }
459 return jitReturn(Value());
460}

462/// OP_INDEX_SET_LOCAL: 局部变量数组/字典索引赋值（COW detach + 写入 + 写回栈槽）
463/// @param ctx JitContext 指针（访问 stackTop 读写 JIT 栈，错误报告用）
464/// @param slot 局部变量槽位号
465/// @param frameBase 当前帧的 basePointer（r13 值，指向 slot 0）
466/// 栈布局（调用前）：[..., idx, val]（val 在栈顶）
467/// 栈布局（调用后）：[...]（pop idx 和 val，不 push）
468/// 语义：obj = frameBase[-slot]; obj[idx] = val; frameBase[-slot] = obj（COW detach 后写回）
469extern "C" void jitIndexSetLocal(JitContext* ctx, uint8_t slot, int64_t* frameBase) {
470 if (!ctx || !ctx->stackTop || !frameBase) {
471 return;
472 }
473 int64_t* sp = ctx->stackTop;
474 Value val = bitsToValue(*sp);
475 sp += 1; // pop val
476 Value idx = bitsToValue(*sp);
477 sp += 1; // pop idx
478 ctx->stackTop = sp; // 更新栈顶（pop 2 个，不 push）
479
480 // 读取局部变量栈槽（JIT 栈向下增长，slot 0 在 frameBase，slot N 在 frameBase - N）
481 // 注意：frameBase - slot 是 int64_t* 指针运算（元素单位），等价于 frameBase - slot*8 字节
482 int64_t* slotPtr = frameBase - slot;
483 Value obj = bitsToValue(*slotPtr);
484
485 if (obj.isArray() && idx.isInt()) {
486 int64_t i = idx.intVal();
487 // 越界检查用 const 重载避免 COW detach
488 const auto& arr = obj.arrayVal();
489 if (i < 0 || static_cast<size_t>(i) >= arr.size()) {
490 if (ctx->hasError && ctx->errorBuffer) {
491 *ctx->hasError = true;
492 *ctx->errorBuffer =
493 "数组索引越界: " + std::to_string(i) + ", 有效范围 [0, " + std::to_string(arr.size()) + ")";
494 }
495 return;
496 }
497 // 触发 COW detach（非 const arrayVal 重载调用 ensureUnique<ArrayData>）
498 obj.arrayVal()[static_cast<size_t>(i)] = val;
499 // 写回栈槽（detach 后可能产生新 ArrayData，需更新栈槽的 raw bits）
500 // obj 的内部 box_ 已被 ensureUnique 更新，提取新的 raw bits
501 *slotPtr = valueToBits(obj); // detach 转移所有权到栈槽
502 return;
503 }
504 // R147 阶段 4b：字典索引赋值（键支持 string/int/bool/float，与 StackVM 对齐）
505 if (obj.isDict()) {
506 auto dk = Value::dictKeyFromValue(idx);
507 if (!dk) {

696/// R156: OP_CLOSURE — 创建闭包值（含 upvalue 捕获）
697/// 与 StackVM executeClosure 对齐（VMCalls.cpp:1022-1083）
698/// @param ctx JitContext 指针（读取 backendPtr、frameCount、currentBp）
699/// @param funName 函数名（C 字符串，编译期嵌入 movabs，需静态生命期）
700/// @param chunkPtr 函数 chunk 指针（编译期从 result.functionChunks 获取）
701/// @param upvalueCount upvalue 数量
702/// @param upvalueDescs 指向字节码中 upvalue 描述符的指针（[isLocal0, index0, isLocal1, index1, ...]）
703/// @return NaN-boxed 闭包指针 bits（push 到 JIT 栈）
704/// R155: OP_CALL_EXPR — 闭包值调用（含 upvalue 绑定，R156 扩展）
705/// 与 StackVM executeCallExprValue 对齐（VMCalls.cpp:647-709）
706/// 栈布局（调用前）：[..., closure, arg0, arg1, ..., argN-1]（closure 在参数下方）
707/// 栈布局（调用后）：新帧栈 [..., arg0, arg1, ..., argN-1, extraSlots...]，调用者 r15 = callerSpAfterPop
708/// @param ctx JitContext 指针（访问 stackTop/frames/frameCount/funcEntriesPtr/callerBp）
709/// @param argCount 实际参数个数
710/// 通过邮箱返回：ctx->methodEntryPtr = 入口地址, ctx->methodLocalCount = localCount
711extern "C" void jitCallExpr(JitContext* ctx, uint8_t argCount) {
712 if (!ctx || !ctx->stackTop) {
713 return;
714 }
715
716 int64_t* sp = ctx->stackTop;
717
718 // 1. pop args（借用，不 detach）
719 std::array<Value, RuntimeLimits::MAX_JIT_ARGS> args;
720 for (int i = argCount - 1; i >= 0; --i) {
721 args[i] = bitsToValue(*sp); // 借用
722 sp += 1; // pop
723 }
724
725 // 2. pop closure（借用）
726 Value callee = bitsToValue(*sp); // 借用
727 sp += 1; // pop closure
728 int64_t* callerSpAfterPop = sp; // 调用者恢复 r15 用
729
730 // 3. 类型检查
731 if (!callee.isClosure()) {
732 if (ctx->hasError && ctx->errorBuffer) {
733 *ctx->hasError = true;
734 *ctx->errorBuffer = "表达式调用需要函数值";
735 }
736 ctx->stackTop = callerSpAfterPop;
737 return;
738 }
739
740 // 4. 从 funcEntries_ 查找 JIT 入口地址 + localCount
741 if (!ctx->funcEntriesPtr) {
742 if (ctx->hasError && ctx->errorBuffer) {
743 *ctx->hasError = true;
744 *ctx->errorBuffer = "JIT 内部错误: funcEntriesPtr 为空";
745 }
746 ctx->stackTop = callerSpAfterPop;
747 return;
748 }
749 std::string funName = callee.closureName();
750 auto it = ctx->funcEntriesPtr->find(funName);
751
752 // R157: lazy compilation — entryPtr 为 null 时触发按需编译
753 if (it == ctx->funcEntriesPtr->end() || it->second.entryPtr == nullptr) {
754 auto* backend = static_cast<JITBackend*>(ctx->backendPtr);
755 if (backend) {
756 JitEntryFn lazyEntry = backend->triggerLazyCompile(funName);
757 if (lazyEntry) {
758 // 编译成功，重新查找（entryPtr 已更新）
759 it = ctx->funcEntriesPtr->find(funName);
760 }
761 }
762 }
763
764 if (it == ctx->funcEntriesPtr->end()) {
765 if (ctx->hasError && ctx->errorBuffer) {
766 *ctx->hasError = true;
767 *ctx->errorBuffer = "未找到函数: " + funName;
768 }
769 ctx->stackTop = callerSpAfterPop;
770 return;
771 }
772 const JitMethodInfo* foundFunc = &it->second;
773
774 if (foundFunc->entryPtr == nullptr) {
775 if (ctx->hasError && ctx->errorBuffer) {
776 *ctx->hasError = true;
777 *ctx->errorBuffer = "函数 " + funName + " 入口未绑定（lazy compilation 失败）";
778 }
779 ctx->stackTop = callerSpAfterPop;
780 return;
781 }
782
783 // 5. 参数校验
784 if (argCount < static_cast<uint8_t>(foundFunc->requiredArity) ||
785 argCount > static_cast<uint8_t>(foundFunc->arity)) {
786 if (ctx->hasError && ctx->errorBuffer) {
787 *ctx->hasError = true;
788 *ctx->errorBuffer = "函数 " + funName + " 期望 " + std::to_string(foundFunc->requiredArity) + "-" +
789 std::to_string(foundFunc->arity) + " 个参数，但传入了 " + std::to_string(argCount) +
790 " 个";
791 }
792 ctx->stackTop = callerSpAfterPop;
793 return;
794 }
795
796 // 6. 默认参数填充
797 std::array<Value, RuntimeLimits::MAX_DEFAULT_PARAMS> defaults;
798 int defaultCount = 0;
799 if (argCount < static_cast<uint8_t>(foundFunc->arity)) {
800 int missingCount = foundFunc->arity - argCount;
801 int defaultStartIdx = static_cast<int>(foundFunc->defaultConstIndices->size()) - missingCount;
802 if (defaultStartIdx < 0 ||
803 static_cast<size_t>(defaultStartIdx + missingCount) > foundFunc->defaultConstIndices->size()) {
804 if (ctx->hasError && ctx->errorBuffer) {
805 *ctx->hasError = true;
806 *ctx->errorBuffer = "函数 " + funName + " 默认参数索引越界";
807 }
808 ctx->stackTop = callerSpAfterPop;
809 return;
810 }
811 for (int i = defaultStartIdx; i < defaultStartIdx + missingCount; ++i) {
812 uint16_t constIdx = (*foundFunc->defaultConstIndices)[static_cast<size_t>(i)];
813 if (constIdx == RuntimeLimits::NO_INDEX || constIdx >= foundFunc->constants->size()) {
814 if (ctx->hasError && ctx->errorBuffer) {
815 *ctx->hasError = true;
816 *ctx->errorBuffer = "函数 " + funName + " 默认参数常量索引越界";
817 }
818 ctx->stackTop = callerSpAfterPop;
819 return;
820 }
821 defaults[defaultCount++] = (*foundFunc->constants)[constIdx];
822 }
823 }
824
825 // 7. 构造新帧栈：push args + push defaults + push extraSlots
826 // JIT 栈向下增长：push = sp -= 1; *sp = valueToBits(val)
827 // push args
828 for (int i = 0; i < argCount; ++i) {
829 sp -= 1;
830 *sp = valueToBits(args[i]); // detach
831 }
832 // push defaults
833 for (int i = 0; i < defaultCount; ++i) {
834 sp -= 1;
835 *sp = valueToBits(defaults[i]); // detach
836 }
837 // push extraSlots (null)
838 int preAllocated = foundFunc->arity; // 函数 chunk 没有 this/fields
839 int extraSlots = foundFunc->localCount - preAllocated;
840 for (int i = 0; i < extraSlots; ++i) {
841 sp -= 1;
842 *sp = static_cast<int64_t>(JIT_NULL_BITS);
843 }
844
845 // 8. 构造 JitFrame
846 int64_t* callerBp = ctx->callerBp; // JIT 代码在调用前存入 r13
847 size_t idx = *ctx->frameCount;
848 if (idx >= RuntimeLimits::MAX_FRAMES) {
849 if (ctx->hasError && ctx->errorBuffer) {
850 *ctx->hasError = true;
851 *ctx->errorBuffer = ErrorFormat::formatStd(ErrorMessages::kRecursionDepthExceededFmtStd,
852 static_cast<int>(RuntimeLimits::MAX_FRAMES));
853 }
854 ctx->stackTop = sp;
855 return;
856 }
857 JitFrame* frame = &ctx->frames[idx];
858 frame->callerBp = callerBp;
859 frame->callerSp = callerSpAfterPop;
860 // returnAddr 由 JIT 代码在调用后设置（jitCallExpr 无法访问 JIT Label）
861 frame->receiverSlotPtr = nullptr; // 非方法调用，无 writeBack
862 frame->methodFieldOrder = nullptr;
863 frame->methodBp = nullptr;
864 frame->isMethodCall = 0;
865 frame->isInitCall = 0;
866 frame->fieldCount = 0;
867
868 *ctx->frameCount = idx + 1;
869
870 // R156: 推入闭包值的 upvalues 到 frameUpvaluesStack_（与 StackVM setupFunctionCallFrame 对齐）
871 // 普通函数调用（OP_CALL）推入空 vector；闭包值调用（OP_CALL_EXPR）从闭包值提取 upvalues
872 auto* backend = static_cast<JITBackend*>(ctx->backendPtr);
873 if (backend) {
874 if (backend->frameUpvaluesStack_.size() <= idx) {
875 backend->frameUpvaluesStack_.resize(idx + 1);
876 }
877 if (callee.vmClosure()) {
878 backend->frameUpvaluesStack_[idx] = callee.vmClosure()->upvalues;
879 } else {
880 backend->frameUpvaluesStack_[idx].clear();
881 }
882 }
883
884 // 9. 通过邮箱返回函数入口地址 + localCount
885 ctx->methodEntryPtr = foundFunc->entryPtr;
886 ctx->methodLocalCount = foundFunc->localCount;
887
888 // 10. 更新栈顶
889 ctx->stackTop = sp;
890}

957/// OP_BUILD_ENUM_VARIANT：从 JIT 栈 pop argCount 个字段，校验后构造 enum variant 压栈
958/// 栈布局（调用前）：[..., arg0, arg1, ..., argN-1] ← ctx->stackTop 指向 argN-1
959/// 栈布局（调用后）：[..., variant]
960/// @param enumName/variantName C 字符串（编译期从 chunk.constants 解析，生命期跨 execute）
961extern "C" void jitBuildEnumVariant(JitContext* ctx, const char* enumName, const char* variantName, uint8_t argCount) {
962 if (!ctx || !ctx->stackTop || !enumName || !variantName) {
963 return;
964 }
965 auto reportError = [&](std::string msg) {
966 if (ctx->hasError && ctx->errorBuffer) {
967 *ctx->hasError = true;
968 *ctx->errorBuffer = std::move(msg);
969 }
970 };
971
972 // R99 enum 校验：运行时检查 enum 已声明、variant 存在、参数 arity 一致（对齐 VM）
973 const auto* registry = static_cast<const std::unordered_map<std::string, VMEnumInfo>*>(ctx->enumRegistryPtr);
974 if (!registry) {
975 reportError(ErrorFormat::formatStd("未定义的 enum: {}", enumName));
976 return;
977 }
978 auto enumIt = registry->find(enumName);
979 if (enumIt == registry->end()) {
980 reportError(ErrorFormat::formatStd("未定义的 enum: {}", enumName));
981 return;
982 }
983 const VMEnumInfo& info = enumIt->second;
984 const VMEnumVariantInfo* varInfo = nullptr;
985 for (const auto& v : info.variants) {
986 if (v.name == variantName) {
987 varInfo = &v;
988 break;
989 }
990 }
991 if (varInfo == nullptr) {
992 reportError(ErrorFormat::formatStd("enum '{}' 没有 variant '{}'", enumName, variantName));
993 return;
994 }
995 if (static_cast<int>(argCount) != varInfo->arity) {
996 reportError(ErrorFormat::formatStd("enum variant '{}.{}' 期望 {} 个参数，得到 {} 个", enumName, variantName,
997 varInfo->arity, static_cast<int>(argCount)));
998 return;
999 }
1000
1001 // pop 字段（借用，与 jitBuildArray 同一所有权约定：弃置栈槽引用由追踪 GC 回收）
1002 int64_t* sp = ctx->stackTop;
1003 std::vector<Value> fields(argCount);
1004 for (int i = argCount - 1; i >= 0; --i) {
1005 fields[i] = bitsToValue(*sp);
1006 sp += 1; // pop
1007 }
1008
1009 // AUDIT-R6 F3 对齐：字段类型注解校验（泛型类型参数擦除；实例继承链 fallback
1010 // 对齐 VM executeContainerBuildOps 的 classInfo_ 查找模式，JIT 侧用 JitClassInfo）
1011 for (size_t fi = 0; fi < fields.size() && fi < varInfo->paramTypes.size(); ++fi) {
1012 const std::string& expectedType = varInfo->paramTypes[fi];
1013 if (expectedType.empty() || minilang::isTypeParameter(expectedType, info.typeParams))
1014 continue;
1015 if (!minilang::typeMatchValue(fields[fi], expectedType)) {
1016 bool inheritOk = false;
1017 if (fields[fi].isInstance() && ctx->classInfoPtr) {

1105// 并发原语支持（spawn/channel/mutex/rwlock + 同步对象方法）
1106// ============================================================
1107// 共享层单一事实源：构造函数走 executeSharedBuiltinFunction（channel/mutex/rwlock），
1108// spawn 走 executeSharedSpawn（延迟执行模式：闭包存入 ThreadInner.pending 字段，
1109// join() 时由调用线程同步执行），方法分发走 handleSyncObjectMethod。
1110// 与 Interpreter/StackVM/RegisterVM 语义天然一致。
1111//
1112// JIT 特有部分：spawn 的 ClosureInvoker 需要从 C++ 同步调用 JIT 编译的闭包。
1113// 实现为闭包跳板（JITBackend::closureTrampoline_，compileAllChunks 末尾发射）：
1114// 1. jitInvokeClosureSync 将 closure + args 压入 JIT 操作数栈（经 ctx->stackTop）
1115// 2. 跳板（C 可调用）建立与主入口同构的 rbp 帧，调用 jitCallExpr 建帧后
1116// jmp 闭包入口；闭包 OP_RETURN 返回跳板 returnLabel 后 ret 回 C++
1117// 3. trampolineFrameFloors_ 记录帧地板，jitThrow 据此阻止异常跨跳板传播
1118
1119/// 从 C++ 同步调用 JIT 编译的闭包（spawn-join 的 ClosureInvoker 实现）
1120/// @return 闭包返回值；闭包内部错误/未捕获异常时返回 err（hasError 已清除，
1121/// 由调用方（handleSyncObjectMethod join 处理器）重新抛出 RuntimeError）
1122// 注：本文件整体处于 extern "C" 块内，此函数返回 C++ 类型 Result<Value>，
1123// 需显式 extern "C++" 语言链接（否则 MSVC C4190 + /WX 阻断编译）。
1124extern "C++" {
1125static Result<Value> jitInvokeClosureSync(JitContext* ctx, const Value& closure, const Value* args, size_t argCount) {
1126 auto* backend = ctx ? static_cast<JITBackend*>(ctx->backendPtr) : nullptr;
1127 if (!backend || !backend->closureTrampoline_ || !ctx->stackTop) {
1128 return Result<Value>::err("JIT 内部错误: 闭包调用跳板未初始化", 0, 0);
1129 }
1130 if (argCount + 1 > RuntimeLimits::MAX_JIT_ARGS) {
1131 return Result<Value>::err("JIT 闭包调用参数过多", 0, 0);
1132 }
1133
1134 // 压入 closure + args（OP_CALL_EXPR 栈布局：[closure, arg0, ..., argN-1]，栈顶 argN-1）
1135 int64_t* sp = ctx->stackTop;
1136 {
1137 Value c = closure; // 拷贝 addRef 后 detach 转移到栈槽
1138 sp -= 1;
1139 *sp = static_cast<int64_t>(valueToBits(c));
1140 }
1141 for (size_t i = 0; i < argCount; ++i) {
1142 Value v = args[i];
1143 sp -= 1;
1144 *sp = static_cast<int64_t>(valueToBits(v));
1145 }
1146 ctx->stackTop = sp;
1147
1148 // 记录帧地板：jitThrow 据此阻止异常跨跳板边界传播
1149 size_t frameFloor = ctx->frameCount ? *ctx->frameCount : 0;
1150 backend->trampolineFrameFloors_.push_back(frameFloor);
1151 backend->closureTrampoline_(ctx, static_cast<uint64_t>(argCount));
1152 backend->trampolineFrameFloors_.pop_back();
1153
1154 if (ctx->hasError && *ctx->hasError) {
1155 // 错误路径：闭包内 runtimeError/未捕获异常 → 帧深度可能未回退，恢复到地板；
1156 // 清理跳板范围内残留的 try handler / 续跳目标（frameIndex > floor）
1157 if (ctx->frameCount && *ctx->frameCount > frameFloor) {
1158 *ctx->frameCount = frameFloor;
1159 }
1160 while (!backend->tryStack_.empty() && backend->tryStack_.back().frameIndex > frameFloor) {
1161 backend->tryStack_.pop_back();
1162 }
1163 while (!backend->pendingJumpStack_.empty() && backend->pendingJumpStack_.back().frameIndex > frameFloor) {
1164 backend->pendingJumpStack_.pop_back();
1165 }
1166 std::string msg = ctx->errorBuffer ? *ctx->errorBuffer : std::string("JIT 闭包执行错误");
1167 *ctx->hasError = false; // 转交上层：join 处理器将重新抛出 RuntimeError
1168 if (ctx->errorBuffer)
1169 ctx->errorBuffer->clear();
1170 return Result<Value>::err(std::move(msg), 0, 0);
1171 }
1172
1173 // 正常路径：闭包返回值在栈顶（emitReturn 已 push 到 callerSp 位置）
1174 int64_t* resultSp = ctx->stackTop;
1175 Value result = bitsToValue(static_cast<uint64_t>(*resultSp));
1176 ctx->stackTop = resultSp + 1; // pop（弃置栈槽引用，追踪 GC 回收，与其他 helper 一致）
1177 return Result<Value>::ok(std::move(result));
1178}
1179} // extern "C++"

1181/// OP_CALL 分流：并发原语内建函数（spawn/channel/mutex/rwlock）
1182/// 栈布局（调用前）：[..., arg0, ..., argN-1] ← ctx->stackTop 指向 argN-1
1183/// 栈布局（调用后）：[..., result]
1184extern "C" void jitCallConcurrency(JitContext* ctx, const char* funName, uint8_t argCount) {
1185 if (!ctx || !ctx->stackTop || !funName) {
1186 return;
1187 }
1188 auto reportError = [&](std::string msg) {
1189 if (ctx->hasError && ctx->errorBuffer) {
1190 *ctx->hasError = true;
1191 *ctx->errorBuffer = std::move(msg);
1192 }
1193 };
1194
1195 // pop args（借用）
1196 int64_t* sp = ctx->stackTop;
1197 std::array<Value, RuntimeLimits::MAX_JIT_ARGS> args;
1198 for (int i = argCount - 1; i >= 0; --i) {
1199 args[i] = bitsToValue(*sp);
1200 sp += 1; // pop
1201 }
1202 ctx->stackTop = sp;
1203
1204 std::string name(funName);
1205 if (name == "spawn") {
1206 // 与 VM executeCallSpawn 对齐：至少 1 个参数（闭包）
1207 if (argCount < 1) {
1208 reportError("spawn 期望至少 1 个参数（函数），但传入了 0 个");
1209 return;
1210 }
1211 // ClosureInvoker：join() 时在调用线程同步执行（延迟执行模式，无数据竞争）。
1212 // 捕获 ctx：ThreadInner 持有 invoker 的生命期不超过本次 execute（每次 execute
1213 // 重新编译 + 重建全局状态，旧 Thread 值的 join 在新一轮不可达）。
1214 ClosureInvoker invoker = [ctx](const Value& closure, const Value* cargs, size_t cargc, int /*line*/,
1215 int /*col*/) -> Result<Value> {
1216 return jitInvokeClosureSync(ctx, closure, cargs, cargc);
1217 };
1218 auto r = executeSharedSpawn(args[0], args.data() + 1, static_cast<size_t>(argCount) - 1, invoker, 0, 0);
1219 if (r.is_err()) {
1220 reportError(r.error().message);
1221 return;
1222 }
1223 Value threadVal = r.value();
1224 sp -= 1;
1225 *sp = static_cast<int64_t>(valueToBits(threadVal));
1226 ctx->stackTop = sp;
1227 return;
1228 }
1229
1230 // channel()/mutex()/rwlock()：共享层注册表（含 arity 校验，错误文本与三 VM 一致）
1231 auto r = executeSharedBuiltinFunction(name, args.data(), argCount, 0, 0);
1232 if (r.is_err()) {
1233 reportError(r.error().message);
1234 return;
1235 }
1236 Value result = r.value();
1237 sp -= 1;
1238 *sp = static_cast<int64_t>(valueToBits(result));
1239 ctx->stackTop = sp;
1240}

1247// - classInfo_ / pendingFieldOrder_ 通过 JitContext.classInfoPtr / pendingFieldOrderPtr 访问

1251/// OP_CLASS_NEW：创建实例（R149 扩展支持 init 方法调用）
1252/// @param ctx JitContext
1253/// @param className 类名 C 字符串（编译期从常量池解析）
1254/// @param argCount 构造参数个数
1255/// 行为：
1256/// - 类未注册（OP_DEFINE_CLASS 之前的模板创建）：argCount 必须 0，创建空实例
1257/// - 类已注册 + 无 init 方法：argCount 必须 0，复制 fieldDefaults 创建实例
1258/// - 类已注册 + 有 init 方法：
1259/// * argCount > 0：调用 init（支持默认参数填充）
1260/// * argCount == 0 + init.requiredArity == 0：自动调用 init（与 StackVM 一致）
1261/// * argCount == 0 + init.requiredArity > 0：报错（init 期望至少 N 个参数）
1262/// - 类已注册 + 无 init + argCount > 0：报错（无 init 但传入了参数）
1263/// 栈布局（调用前）：[argN-1]...[arg0] ← ctx->stackTop 指向 argN-1
1264/// 栈布局（无 init 调用，返回后）：[instance]
1265/// 栈布局（有 init 调用，返回前）：[extraSlots...][defaults...][args...][fields...][this]
1266/// ctx->methodEntryPtr 非 null 表示已设置 init 帧，JIT 代码需 jmp 到入口
1267extern "C" void jitClassNew(JitContext* ctx, const char* className, uint8_t argCount) {
1268 if (!ctx || !ctx->stackTop || !ctx->classInfoPtr || !className) {
1269 return;
1270 }
1271 // 默认：methodEntryPtr = null（无 init 调用），JIT 代码检测此字段决定是否 jmp
1272 ctx->methodEntryPtr = nullptr;
1273 ctx->methodLocalCount = 0;
1274
1275 int64_t* sp = ctx->stackTop;
1276
1277 // 1. pop args（反向填充：栈顶是 argN-1）
1278 std::array<Value, RuntimeLimits::MAX_JIT_ARGS> args;
1279 for (int i = argCount - 1; i >= 0; --i) {
1280 args[i] = bitsToValue(*sp); // 借用
1281 sp += 1; // pop
1282 }
1283 int64_t* callerSpAfterPop = sp; // 调用者恢复 r15 用
1284
1285 std::string name(className);
1286 auto classIt = ctx->classInfoPtr->find(name);
1287 if (classIt == ctx->classInfoPtr->end()) {
1288 // 类尚未注册（OP_DEFINE_CLASS 之前的模板创建）：argCount 必须 0
1289 if (argCount > 0) {
1290 if (ctx->hasError && ctx->errorBuffer) {
1291 *ctx->hasError = true;
1292 *ctx->errorBuffer = "类 " + name + " 尚未定义，不能带参数构造";
1293 }
1294 ctx->stackTop = callerSpAfterPop;
1295 return;
1296 }
1297 Value instance = Value::makeInstance(name);
1298 sp -= 1;
1299 *sp = valueToBits(instance);
1300 ctx->stackTop = sp;
1301 return;
1302 }
1303
1304 // 2. 类已注册：复制 fieldDefaults 创建实例
1305 const JitClassInfo& cls = classIt->second;
1306 Value instance = Value::makeInstance(cls.name);
1307 instance.fields() = cls.fieldDefaults; // 拷贝默认值（含继承字段）
1308
1309 // 3. 沿继承链查找 init 方法
1310 const JitMethodInfo* initMethod = nullptr;
1311 std::string searchClassName = cls.name;
1312 for (int guard = 0; guard < 64 && !searchClassName.empty(); ++guard) {
1313 std::string key = searchClassName + ".init";
1314 auto it = ctx->methodEntriesPtr->find(key);
1315 if (it != ctx->methodEntriesPtr->end()) {
1316 initMethod = &it->second;
1317 break;
1318 }
1319 auto clsIt2 = ctx->classInfoPtr->find(searchClassName);
1320 if (clsIt2 == ctx->classInfoPtr->end())
1321 break;
1322 searchClassName = clsIt2->second.superClassName;
1323 }
1324
1325 // 4. 决定是否调用 init（与 StackVM executeClassNew 一致）
1326 bool shouldCallInit = false;
1327 if (initMethod != nullptr) {
1328 if (argCount > 0) {
1329 shouldCallInit = true;
1330 } else if (initMethod->requiredArity == 0) {
1331 shouldCallInit = true; // 0 必需参数（含全默认参数 init），自动构造时调用
1332 } else {
1333 // init 存在但参数不匹配：报错
1334 if (ctx->hasError && ctx->errorBuffer) {
1335 *ctx->hasError = true;
1336 *ctx->errorBuffer = "类 " + cls.name + " 的 init 期望至少 " +
1337 std::to_string(initMethod->requiredArity) + " 个参数，但传入了 0 个";
1338 }
1339 ctx->stackTop = callerSpAfterPop;
1340 return;
1341 }
1342 } else if (argCount > 0) {
1343 // 无 init 但有参数：报错
1344 if (ctx->hasError && ctx->errorBuffer) {
1345 *ctx->hasError = true;
1346 *ctx->errorBuffer = "类 " + cls.name + " 没有 init 方法，但传入了 " + std::to_string(argCount) + " 个参数";
1347 }
1348 ctx->stackTop = callerSpAfterPop;
1349 return;
1350 }
1351
1352 if (!shouldCallInit) {
1353 // 无 init 或不应调用：直接 push instance
1354 sp -= 1;
1355 *sp = valueToBits(instance);
1356 ctx->stackTop = sp;
1357 return;
1358 }
1359
1360 // 5. 调用 init：参数校验 + 默认参数填充
1361 if (argCount < static_cast<uint8_t>(initMethod->requiredArity) ||
1362 argCount > static_cast<uint8_t>(initMethod->arity)) {
1363 if (ctx->hasError && ctx->errorBuffer) {
1364 *ctx->hasError = true;
1365 *ctx->errorBuffer = "构造函数 init 期望 " + std::to_string(initMethod->requiredArity) + "-" +
1366 std::to_string(initMethod->arity) + " 个参数，但传入了 " + std::to_string(argCount) +
1367 " 个";
1368 }
1369 ctx->stackTop = callerSpAfterPop;
1370 return;
1371 }
1372
1373 std::array<Value, RuntimeLimits::MAX_DEFAULT_PARAMS> defaults;
1374 int defaultCount = 0;
1375 if (argCount < static_cast<uint8_t>(initMethod->arity)) {
1376 int missingCount = initMethod->arity - argCount;
1377 int defaultStartIdx = static_cast<int>(initMethod->defaultConstIndices->size()) - missingCount;
1378 if (defaultStartIdx < 0 ||
1379 static_cast<size_t>(defaultStartIdx + missingCount) > initMethod->defaultConstIndices->size()) {
1380 if (ctx->hasError && ctx->errorBuffer) {
1381 *ctx->hasError = true;
1382 *ctx->errorBuffer = "方法 init 默认参数索引越界";
1383 }
1384 ctx->stackTop = callerSpAfterPop;
1385 return;
1386 }
1387 for (int i = defaultStartIdx; i < defaultStartIdx + missingCount; ++i) {
1388 uint16_t constIdx = (*initMethod->defaultConstIndices)[static_cast<size_t>(i)];
1389 if (constIdx == RuntimeLimits::NO_INDEX || constIdx >= initMethod->constants->size()) {
1390 if (ctx->hasError && ctx->errorBuffer) {
1391 *ctx->hasError = true;
1392 *ctx->errorBuffer = "方法 init 默认参数常量索引越界";
1393 }
1394 ctx->stackTop = callerSpAfterPop;
1395 return;
1396 }
1397 defaults[defaultCount++] = (*initMethod->constants)[constIdx];
1398 }
1399 }
1400
1401 // 6. 构造新帧：push this + push fields + push args + push defaults + push extraSlots
1402
1403 // R149 fix: valueToBits 会 detach（转移所有权置 v 为 null），所以 push this 后
1404 // 不能再从 instance 读 fields。修复：先拷贝 instance 用于 push this（detach 拷贝），
1405 // 原 instance 保留用于读 fields，函数结束时自动析构。
1406 Value instanceForPush = instance; // 拷贝（addRef）
1407 sp -= 1;
1408 *sp = valueToBits(instanceForPush); // detach instanceForPush，原 instance 仍有效
1409
1410 // push fields（按 init chunk 的 fieldOrder 顺序）
1411 int fieldCount = 0;
1412 if (initMethod->fieldOrder && !initMethod->fieldOrder->empty()) {
1413 const Value& instanceConst = instance;
1414 for (const auto& fieldName : *initMethod->fieldOrder) {
1415 auto fieldIt = instanceConst.fields().find(fieldName);
1416 sp -= 1;
1417 if (fieldIt != instanceConst.fields().end()) {
1418 Value fieldCopy = fieldIt->second; // 拷贝（addRef），valueToBits 需要 non-const
1419 *sp = valueToBits(fieldCopy); // detach（转移所有权到 bits）
1420 } else {
1421 *sp = static_cast<int64_t>(JIT_NULL_BITS);
1422 }
1423 }
1424 fieldCount = static_cast<int>(initMethod->fieldOrder->size());
1425 }
1426
1427 // push args
1428 for (int i = 0; i < argCount; ++i) {
1429 sp -= 1;
1430 *sp = valueToBits(args[i]);
1431 }
1432 // push defaults
1433 for (int i = 0; i < defaultCount; ++i) {
1434 sp -= 1;
1435 *sp = valueToBits(defaults[i]);
1436 }
1437
1438 // push extraSlots (null)
1439 int preAllocated = 1 + fieldCount + initMethod->arity;
1440 int extraSlots = initMethod->localCount - preAllocated;
1441 if (extraSlots < 0) {
1442 if (ctx->hasError && ctx->errorBuffer) {
1443 *ctx->hasError = true;
1444 *ctx->errorBuffer = "类 " + cls.name +
1445 " 的 init 方法帧布局损坏: localCount=" + std::to_string(initMethod->localCount) +
1446 " < preAllocated=" + std::to_string(preAllocated);
1447 }
1448 ctx->stackTop = callerSpAfterPop;
1449 return;
1450 }
1451 for (int i = 0; i < extraSlots; ++i) {
1452 sp -= 1;
1453 *sp = static_cast<int64_t>(JIT_NULL_BITS);
1454 }
1455
1456 // 7. 设置 JitFrame
1457 int64_t* callerBp = ctx->callerBp; // JIT 代码在调用前存入
1458 size_t idx = *ctx->frameCount;
1459 if (idx >= RuntimeLimits::MAX_FRAMES) {
1460 if (ctx->hasError && ctx->errorBuffer) {
1461 *ctx->hasError = true;
1462 *ctx->errorBuffer = ErrorFormat::formatStd(ErrorMessages::kRecursionDepthExceededFmtStd,
1463 static_cast<int>(RuntimeLimits::MAX_FRAMES));
1464 }
1465 ctx->stackTop = sp;
1466 return;
1467 }
1468 JitFrame* frame = &ctx->frames[idx];
1469 frame->callerBp = callerBp;
1470 frame->callerSp = callerSpAfterPop;
1471 frame->receiverSlotPtr = nullptr; // init 调用无 writeBack（this 即新实例）
1472 frame->methodFieldOrder = initMethod->fieldOrder;
1473 frame->methodBp = sp + (initMethod->localCount - 1); // this 在 slot 0 = 最高地址
1474 frame->isMethodCall = 1;
1475 frame->isInitCall = 1; // init 返回 this 而非返回值
1476 frame->fieldCount = fieldCount;
1477
1478 *ctx->frameCount = idx + 1;
1479
1480 // R156: 推入空 upvalues（init 调用不携带闭包 upvalues，但需保持 frameUpvaluesStack_ 对齐）

1824// R149 阶段 5：方法调用支持（OP_METHOD_CALL / OP_SUPER_CALL / init 调用）
1825// ============================================================
1826// 设计要点（与 StackVM executeInstanceMethodCall/executeReturn 对齐）：
1827// - JIT 栈向下增长：push = sp -= 1; pop = sp += 1（int64_t* 指针算术）
1828// - 方法帧布局（向低地址）：
1829// [extraSlots...][args...][fields...][this]
1830// ^methodBp (slot 0 = r13)
1831// this 在 methodBp[0]（slot 0，最高地址）
1832// field_i 在 methodBp[-(i+1)]（slot i+1）
1833// - writeBack：方法返回时，字段槽同步回 this.fields()，再写回 receiverSlotPtr
1834// - init 返回 this 而非返回值
1835// - super 调用：从父类开始沿继承链查找方法

1837/// OP_RETURN 方法调用路径：字段槽同步回 this + init 返回 this + writeBack 到接收者
1838/// @param ctx JIT 上下文
1839/// @param framePtr 当前返回的帧指针（frames[*frameCount] 已减 1）
1840/// @param retvalBits 原始返回值的 raw bits
1841/// @return 最终返回值的 raw bits（init 返回 this，其他返回 retvalBits）
1842extern "C" int64_t jitMethodReturn(JitContext* ctx, JitFrame* framePtr, int64_t retvalBits) {
1843 if (!ctx || !framePtr) {
1844 return retvalBits;
1845 }
1846
1847 int64_t* methodBp = framePtr->methodBp;
1848 int64_t fieldCount = framePtr->fieldCount;
1849 const std::vector<std::string>* fieldOrder = framePtr->methodFieldOrder;
1850
1851 // 1. 将方法帧内字段槽（methodBp[-1..-fieldCount]）同步回 this（methodBp[0]）
1852 // JIT 栈向下增长：this 在 slot 0（methodBp[0]），field_i 在 slot i+1（methodBp[-(i+1)]）
1853 if (methodBp && fieldOrder && fieldCount > 0) {
1854 Value thisVal = bitsToValue(methodBp[0]); // 借用 this
1855 if (thisVal.isInstance()) {
1856 for (int64_t i = 0; i < fieldCount && i < static_cast<int64_t>(fieldOrder->size()); ++i) {
1857 Value fieldVal = bitsToValue(methodBp[-(i + 1)]); // 借用字段槽值
1858 thisVal.fields()[(*fieldOrder)[static_cast<size_t>(i)]] = fieldVal; // COW detach
1859 }
1860 methodBp[0] = valueToBits(thisVal); // 写回 this 到栈槽
1861 }
1862 }
1863
1864 // 2. 确定返回值（init 调用返回 this 而非返回值）
1865 int64_t finalRetval = retvalBits;
1866 if (framePtr->isInitCall && methodBp) {
1867 finalRetval = methodBp[0]; // 返回 this
1868 }
1869
1870 // 3. writeBack 到接收者（如果 receiverSlotPtr 非空）
1871 if (framePtr->receiverSlotPtr && methodBp) {
1872 Value modifiedThis = bitsToValue(methodBp[0]); // 借用修改后的 this
1873 if (modifiedThis.isInstance()) {
1874 int64_t* slotPtr = framePtr->receiverSlotPtr;
1875 Value receiver = bitsToValue(*slotPtr); // 借用接收者
1876 if (receiver.isInstance()) {
1877 // 同步所有字段（COW detach）
1878 for (const auto& field : modifiedThis.fields()) {
1879 receiver.fields()[field.first] = field.second;
1880 }
1881 // R149 fix: 如果 receiver 是 caller 的 this slot（slotPtr == callerBp），
1882 // 必须同步 caller frame 的字段槽，否则 caller return 时字段槽（旧值）
1883 // 会覆盖 this.fields()（新值）。
1884 // 场景：Derived.init 调用 super.init(av)：
1885 // 1. Base.init 修改字段槽 a=10 → return 时同步到 Base.init 的 this
1886 // 2. writeBack 把 Base.init 的 this（a=10）同步到 Derived.init 的 this slot
1887 // 3. 但 Derived.init 的字段槽 a 仍是旧值 0（未更新）
1888 // 4. Derived.init 继续 b=bv → 字段槽 b=20
1889 // 5. Derived.init return 时字段槽 a=0, b=20 同步回 this → this.a=0（覆盖了 10）
1890 // 修复：writeBack 后同步更新 caller frame 的字段槽
1891 // 必须在 *slotPtr = valueToBits(receiver) detach 之前进行（detach 后 receiver 为 null）
1892 if (slotPtr == framePtr->callerBp && framePtr->callerBp) {
1893 // receiver 是 caller 的 this slot（slot 0），同步 caller frame 的字段槽
1894 // caller 帧 = framePtr - 1（JitFrame 指针算术，减 72 字节）
1895 // 安全性：slotPtr == callerBp 且 callerBp 非 null → caller 是方法调用（有 this slot）
1896 // → caller 帧存在（idx > 0），framePtr - 1 有效
1897 JitFrame* callerFrame = framePtr - 1;
1898 if (callerFrame->methodFieldOrder && callerFrame->fieldCount > 0) {
1899 int64_t* callerMethodBp = framePtr->callerBp;
1900 for (int64_t i = 0; i < callerFrame->fieldCount &&
1901 i < static_cast<int64_t>(callerFrame->methodFieldOrder->size());
1902 ++i) {
1903 const std::string& fieldName = (*callerFrame->methodFieldOrder)[static_cast<size_t>(i)];
1904 auto fieldIt = receiver.fields().find(fieldName);
1905 if (fieldIt != receiver.fields().end()) {
1906 Value fieldCopy = fieldIt->second; // 拷贝（addRef）
1907 callerMethodBp[-(i + 1)] = valueToBits(fieldCopy); // detach
1908 }
1909 }
1910 }
1911 }
1912 *slotPtr = valueToBits(receiver); // 写回
1913 }
1914 }
1915 }
1916
1917 return finalRetval;
1918}

1923/// 栈布局（调用后，向低地址增长）：

1936/// （C 类内嵌地址），经它访问 methodCallIC_/triggerLazyCompile 即命中已
1937/// 析构实例（stack-use-after-return，2026-10-07 Linux ASan 首抓）。后端
1938/// 实例一律经 ctx->backendPtr（每执行实例在 execute() 设置）解析

1940 const char* superClassName, uint64_t callSiteId, [[maybe_unused]] void* backendPtr) {

1942 return;

1944 // 同步对象方法内联处理邮箱：每次调用先清零，仅在 channel/mutex/rwlock/thread
1945 // 接收者分发完成后置 1（emitMethodCall 生成的代码据此跳过 jmp methodEntryPtr）
1946 ctx->syncMethodHandled = 0;
1947
1948 uint8_t argCount = static_cast<uint8_t>(packedArgs & 0xFF);
1949 // W4 fix: receiverLocalSlotByte（bits 8-15）由 emitMethodCall 打包但 jitMethodCall
1950 // 通过 receiverSlotPtr 参数完成写回，无需在此解包。
1951 bool isInitCall = (packedArgs & (1LL << 16)) != 0;
1952 bool isSuperCall = (packedArgs & (1LL << 17)) != 0;

1954 int64_t* sp = ctx->stackTop;
1955
1956 // 1. pop argCount 个参数（反向填充：栈顶是 argN-1）
1957 // JIT 栈向下增长，sp += 1 是 pop（向高地址移动）
1958 // argCount 为 uint8_t（上限 255），MAX_JIT_ARGS=256 覆盖全部合法值
1959 std::array<Value, RuntimeLimits::MAX_JIT_ARGS> args;
1960 for (int i = argCount - 1; i >= 0; --i) {
1961 args[i] = bitsToValue(*sp); // 借用
1962 sp += 1; // pop
1963 }
1964 // 2. pop receiver
1965 Value receiver = bitsToValue(*sp); // 借用
1966 sp += 1; // pop
1967 int64_t* callerSpAfterPop = sp; // 调用者恢复 r15 用（pop 完 args+receiver 后的栈顶）

1969 if (!receiver.isInstance()) {
1970 // 并发原语：channel/mutex/rwlock/thread 的方法经共享层 handleSyncObjectMethod
1971 // 内联分发（与 VM dispatchSyncObjectBuiltin 对齐）。同步对象内部状态经
1972 // shared_ptr<Inner> 共享，无需 writeBack。thread.join 的 pending 闭包经
1973 // pendingInvoker（spawn 时由 jitCallConcurrency 注入的跳板 invoker）同步执行。
1974 if (receiver.isChannel() || receiver.isMutex() || receiver.isRwLock() || receiver.isThread()) {
1975 std::vector<Value> argsVec(args.begin(), args.begin() + argCount);
1976 try {
1977 BuiltinMethodResult br = handleSyncObjectMethod(methodName, receiver, argsVec, 0, 0);
1978 Value res = std::move(br.result);
1979 int64_t* pushSp = callerSpAfterPop - 1;
1980 *pushSp = static_cast<int64_t>(valueToBits(res));
1981 ctx->stackTop = pushSp;
1982 ctx->syncMethodHandled = 1;
1983 } catch (const RuntimeError& e) {
1984 if (ctx->hasError && ctx->errorBuffer) {
1985 *ctx->hasError = true;
1986 *ctx->errorBuffer = e.what();
1987 }
1988 ctx->stackTop = callerSpAfterPop;
1989 } catch (const std::exception& e) {
1990 if (ctx->hasError && ctx->errorBuffer) {
1991 *ctx->hasError = true;
1992 *ctx->errorBuffer = e.what();
1993 }
1994 ctx->stackTop = callerSpAfterPop;
1995 }
1996 return;
1997 }
1998 if (ctx->hasError && ctx->errorBuffer) {
1999 *ctx->hasError = true;
2000 *ctx->errorBuffer = "类型 " + receiver.typeName() + " 不支持方法 " + std::string(methodName);
2001 }

2003 return;
2004 }

2006 // 3. 方法调用 IC：缓存 (className) → JitMethodInfo* 避免继承链遍历
2007 // 仅对非 super 调用启用（super 调用需从父类开始查找，语义不同）。
2008 // 后端实例经 ctx->backendPtr 解析（每执行实例），不得使用生成代码携带的
2009 // 烘焙 backendPtr 参数——块缓存命中时它是已析构的创建实例（ADR-008 C 类）
2010 auto* backend = ctx ? static_cast<JITBackend*>(ctx->backendPtr) : nullptr;
2011 const JitMethodInfo* foundMethod = nullptr;
2012
2013 if (!isSuperCall && backend && callSiteId < backend->methodCallIC_.size()) {
2014 auto& icEntry = backend->methodCallIC_[callSiteId];
2015 const char* classPtr = receiver.className().c_str();
2016 if (!icEntry.megamorphic) {
2017 foundMethod = icEntry.lookup(classPtr);
2018 if (foundMethod) {
2019 backend->methodIcHitCount_++;
2020 // IC 命中：跳过继承链查找，直接进入参数校验
2021 goto method_found;
2022 }
2023 }
2024 backend->methodIcMissCount_++;
2025 }

2028 {

2031 std::string superName(superClassName);

2033 if (clsIt == ctx->classInfoPtr->end() || clsIt->second.superClassName.empty()) {
2034 if (ctx->hasError && ctx->errorBuffer) {

2037 }
2038 ctx->stackTop = callerSpAfterPop;
2039 return;
2040 }
2041 searchClassName = clsIt->second.superClassName;
2042 } else {
2043 searchClassName = receiver.className();
2044 }

2046 for (int guard = 0; guard < 64 && !searchClassName.empty(); ++guard) {
2047 std::string key = searchClassName + "." + methodName;
2048 auto it = ctx->methodEntriesPtr->find(key);
2049 if (it != ctx->methodEntriesPtr->end()) {
2050 foundMethod = &it->second;
2051 break;
2052 }
2053 auto clsIt = ctx->classInfoPtr->find(searchClassName);
2054 if (clsIt == ctx->classInfoPtr->end())
2055 break;
2056 searchClassName = clsIt->second.superClassName;
2057 }

2059 if (!foundMethod) {
2060 if (ctx->hasError && ctx->errorBuffer) {
2061 *ctx->hasError = true;

2063 }
2064 ctx->stackTop = callerSpAfterPop;
2065 return;
2066 }

2068 // 更新方法调用 IC（仅非 super 调用）
2069 if (!isSuperCall && backend && callSiteId < backend->methodCallIC_.size()) {

2079 if (foundMethod->entryPtr == nullptr) {

2084 if (lazyEntry) {

2088 if (reIt != ctx->methodEntriesPtr->end()) {
2089 foundMethod = &reIt->second;
2090 }
2091 }
2092 }
2093 if (foundMethod->entryPtr == nullptr) {
2094 if (ctx->hasError && ctx->errorBuffer) {
2095 *ctx->hasError = true;
2096 *ctx->errorBuffer = "方法 " + std::string(methodName) + " 入口未绑定（lazy compilation 失败）";
2097 }
2098 ctx->stackTop = callerSpAfterPop;
2099 return;
2100 }
2101 }

2103 // 4. 参数校验 + 默认参数填充
2104 if (argCount < static_cast<uint8_t>(foundMethod->requiredArity) ||
2105 argCount > static_cast<uint8_t>(foundMethod->arity)) {
2106 if (ctx->hasError && ctx->errorBuffer) {
2107 *ctx->hasError = true;
2108 *ctx->errorBuffer = "方法 " + std::string(methodName) + " 期望 " +
2109 std::to_string(foundMethod->requiredArity) + "-" + std::to_string(foundMethod->arity) +

2111 }
2112 ctx->stackTop = callerSpAfterPop;
2113 return;
2114 }

2116 std::array<Value, RuntimeLimits::MAX_DEFAULT_PARAMS> defaults;
2117 int defaultCount = 0;
2118 if (argCount < static_cast<uint8_t>(foundMethod->arity)) {
2119 int missingCount = foundMethod->arity - argCount;
2120 int defaultStartIdx = static_cast<int>(foundMethod->defaultConstIndices->size()) - missingCount;
2121 if (defaultStartIdx < 0 ||
2122 static_cast<size_t>(defaultStartIdx + missingCount) > foundMethod->defaultConstIndices->size()) {
2123 if (ctx->hasError && ctx->errorBuffer) {
2124 *ctx->hasError = true;
2125 *ctx->errorBuffer = "方法 " + std::string(methodName) + " 默认参数索引越界";
2126 }
2127 ctx->stackTop = callerSpAfterPop;
2128 return;
2129 }
2130 for (int i = defaultStartIdx; i < defaultStartIdx + missingCount; ++i) {
2131 uint16_t constIdx = (*foundMethod->defaultConstIndices)[static_cast<size_t>(i)];
2132 if (constIdx == RuntimeLimits::NO_INDEX || constIdx >= foundMethod->constants->size()) {
2133 if (ctx->hasError && ctx->errorBuffer) {
2134 *ctx->hasError = true;
2135 *ctx->errorBuffer = "方法 " + std::string(methodName) + " 默认参数常量索引越界";
2136 }
2137 ctx->stackTop = callerSpAfterPop;
2138 return;
2139 }
2140 defaults[defaultCount++] = (*foundMethod->constants)[constIdx];
2141 }
2142 }

2144 // 5. 构造新帧：push this + push fields + push args + push defaults + push extraSlots
2145 // JIT 栈向下增长：push = sp -= 1; *sp = valueToBits(val)
2146
2147 // R149 fix: valueToBits 会 detach（转移所有权置 v 为 null），所以 push this 后
2148 // 不能再从 receiver 读 fields。修复：先拷贝 receiver 用于 push this（detach 拷贝），
2149 // 原 receiver 保留用于读 fields，函数结束时自动析构。
2150 Value receiverForPush = receiver; // 拷贝（addRef）
2151 sp -= 1;
2152 *sp = valueToBits(receiverForPush); // detach receiverForPush，原 receiver 仍有效
2153
2154 // push fields（按方法 chunk 的 fieldOrder 顺序，从 receiver.fields 读取）
2155 int fieldCount = 0;
2156 if (foundMethod->fieldOrder && !foundMethod->fieldOrder->empty()) {
2157 // 借用原 receiver 读取字段（不修改）
2158 const Value& receiverConst = receiver;
2159 for (const auto& fieldName : *foundMethod->fieldOrder) {
2160 auto fieldIt = receiverConst.fields().find(fieldName);
2161 sp -= 1;
2162 if (fieldIt != receiverConst.fields().end()) {
2163 Value fieldCopy = fieldIt->second; // 拷贝（addRef），valueToBits 需要 non-const
2164 *sp = valueToBits(fieldCopy); // detach 字段值副本
2165 } else {
2166 *sp = static_cast<int64_t>(JIT_NULL_BITS);
2167 }
2168 }
2169 fieldCount = static_cast<int>(foundMethod->fieldOrder->size());
2170 }
2171
2172 // push args
2173 for (int i = 0; i < argCount; ++i) {
2174 sp -= 1;
2175 *sp = valueToBits(args[i]); // detach
2176 }
2177 // push defaults
2178 for (int i = 0; i < defaultCount; ++i) {
2179 sp -= 1;
2180 *sp = valueToBits(defaults[i]); // detach
2181 }
2182
2183 // push extraSlots (null)
2184 int preAllocated = 1 + fieldCount + foundMethod->arity;
2185 int extraSlots = foundMethod->localCount - preAllocated;
2186 for (int i = 0; i < extraSlots; ++i) {
2187 sp -= 1;
2188 *sp = static_cast<int64_t>(JIT_NULL_BITS);
2189 }
2190
2191 // 6. 设置 JitFrame
2192 int64_t* callerBp = ctx->callerBp; // JIT 代码在调用前存入
2193 size_t idx = *ctx->frameCount;
2194 if (idx >= RuntimeLimits::MAX_FRAMES) {
2195 if (ctx->hasError && ctx->errorBuffer) {
2196 *ctx->hasError = true;
2197 *ctx->errorBuffer = ErrorFormat::formatStd(ErrorMessages::kRecursionDepthExceededFmtStd,
2198 static_cast<int>(RuntimeLimits::MAX_FRAMES));
2199 }
2200 ctx->stackTop = sp;
2201 return;
2202 }
2203 JitFrame* frame = &ctx->frames[idx];
2204 frame->callerBp = callerBp;
2205 frame->callerSp = callerSpAfterPop; // 调用者恢复 r15 用
2206 // returnAddr 由 JIT 代码在调用后设置（jitMethodCall 无法访问 JIT Label）
2207 frame->receiverSlotPtr = receiverSlotPtr;
2208 frame->methodFieldOrder = foundMethod->fieldOrder;
2209 // methodBp = sp + (localCount - 1)（this 在 slot 0 = 最高地址）
2210 frame->methodBp = sp + (foundMethod->localCount - 1);
2211 frame->isMethodCall = 1;
2212 frame->isInitCall = isInitCall ? 1 : 0;
2213 frame->fieldCount = fieldCount;
2214
2215 // *frameCount += 1
2216 *ctx->frameCount = idx + 1;
2217
2218 // R156: 推入空 upvalues（方法调用不携带闭包 upvalues，但需保持 frameUpvaluesStack_ 对齐）
2219 {
2220 if (backend) {
2221 if (backend->frameUpvaluesStack_.size() <= idx) {
2222 backend->frameUpvaluesStack_.resize(idx + 1);
2223 }
2224 backend->frameUpvaluesStack_[idx].clear();
2225 }
2226 }
2227
2228 // 7. 通过 ctx 邮箱返回方法入口地址 + localCount
2229 ctx->methodEntryPtr = foundMethod->entryPtr;
2230 ctx->methodLocalCount = foundMethod->localCount;
2231
2232 // 8. 更新栈顶
2233 ctx->stackTop = sp;
2234}

2499/// R162: 抛出异常 — 搜索 tryStack_、截断操作数栈、跨帧传播
2500///
2501/// 算法（与 StackVM VM::throwException 对齐）：
2502/// 1. 清空 pendingJumpStack_（异常中断 break/continue 续跳链）
2503/// 2. 搜索 tryStack_ 顶部当前帧的 handler：
2504/// - 找到：截断栈到 handler.stackBase，push thrownValue，返回 catchAddr
2505/// - 未找到且当前帧非 main：pop JitFrame、关闭 upvalue、恢复 r13，继续搜索

2509/// 跨函数调用不变，只有 r13（basePointer）和 r15（stackTop）需要恢复。
2510///
2511/// @param ctx JitContext 指针
2512/// @param thrownValueBits 异常值的 NaN-boxing 原始位
2513/// @param currentR13 当前帧的 r13（用于关闭 upvalue）
2514/// @return catch 块地址（找到时）或 nullptr（未捕获）
2515extern "C" void* jitThrow(JitContext* ctx, uint64_t thrownValueBits, int64_t* currentR13) {
2516 if (!ctx || !ctx->backendPtr) {
2517 return nullptr;
2518 }
2519 auto* backend = static_cast<JITBackend*>(ctx->backendPtr);
2520
2521 // 1. 清空 pendingJumpStack_（异常中断 break/continue 续跳链）
2522 backend->pendingJumpStack_.clear();
2523
2524 // 跳板帧地板：spawn-join 闭包经 closureTrampoline_ 执行时，异常不得跨跳板
2525 // 边界向外层 JIT 帧传播（catch 地址属于外层代码，在跳板本机栈上 jmp 过去
2526 // 会造成控制流双重执行）。currentFrameIdx 回退到地板即视为跳板内未捕获。
2527 const bool inTrampoline = !backend->trampolineFrameFloors_.empty();
2528 const size_t frameFloor = inTrampoline ? backend->trampolineFrameFloors_.back() : 0;
2529
2530 int64_t* r13 = currentR13;
2531
2532 while (true) {
2533 size_t currentFrameIdx = ctx->frameCount ? *ctx->frameCount : 0;
2534
2535 // 跳板内未捕获：降级为 hasError，由 jitInvokeClosureSync 转为 Result::err
2536 // （join 处理器重新抛出 RuntimeError，外层以运行时错误终止）。
2537 // 必须在 handler 搜索之前检查：frameIndex == frameFloor 的 handler 属于外层代码。
2538 if (inTrampoline && currentFrameIdx <= frameFloor) {
2539 Value thrownValue = bitsToValue(thrownValueBits);
2540 std::string str = thrownValue.toString();
2541 ErrorFormat::truncateForError(str);
2542 if (ctx->hasError)
2543 *ctx->hasError = true;
2544 if (ctx->errorBuffer)
2545 *ctx->errorBuffer = "未捕获的异常: " + str;
2546 return nullptr;
2547 }
2548
2549 // 2. 搜索 tryStack_ 顶部当前帧的 handler
2550 while (!backend->tryStack_.empty()) {
2551 auto& handler = backend->tryStack_.back();
2552 if (handler.frameIndex == currentFrameIdx) {
2553 // 找到 handler：截断栈、push thrownValue、返回 catchAddr
2554 //
2555 // JIT 栈向下增长（push: sub r15,8; mov [r15],val；pop: mov rax,[r15]; add r15,8），
2556 // r15/stackTop 指向栈顶值（空栈时指向栈基址）。
2557 // handler.stackBase = try_begin 时的 r15（指向当时栈顶）。
2558 // 截断 + push thrownValue：新栈顶 = handler.stackBase - 1（向低地址移动一槽），
2559 // [新栈顶] = thrownValueBits。
2560 if (handler.stackBase) {
2561 // AUDIT-R5 R3 fix: 截断前先关闭指向被丢弃区域（地址 < stackBase，
2562 // 即 try 开始后压入的槽位）的 open upvalue，对齐 StackVM
2563 // throwException 的 P0-5 fix（closeUpvaluesFrom(handler.stackBase)）。
2564 // 常规捕获目标是 r13-relative 局部变量槽（地址高于 stackBase，
2565 // 不受影响），此处为防御性对齐：若未来有 upvalue 指向操作数栈
2566 // 临时区，截断后不至于悬垂（UAF 读脏栈）。
2567 backend->closeUpvaluesFrom(handler.stackBase - 1);
2568 int64_t* newSp = handler.stackBase - 1;
2569 *newSp = static_cast<int64_t>(thrownValueBits);
2570 ctx->stackTop = newSp;
2571 }
2572 ctx->currentBp = r13; // 同帧：r13 不变；跨帧：已更新为调用者 r13
2573 // AUDIT-ASAN fix（2026-10-03 MSVC ASan 门禁捕获）：先复制 catchAddr
2574 // 再 pop_back。上一行 pop_back 销毁栈顶 JitTryHandler，随后经引用读
2575 // handler.catchAddr 是 use-after-pop 容器溢出 UB（MSVC STL 注解
2576 // [size, capacity) 毒化区，ASan container-overflow 拦截 18 项
2577 // TestJIT.R162*/L14RuntimeErrorCatch.* 用例；非 ASan 构建下因读取
2578 // 尚未被覆写的旧槽位而侥幸通过）。
2579 void* catchAddr = handler.catchAddr;
2580 backend->tryStack_.pop_back();
2581 return catchAddr;
2582 }
2583 if (handler.frameIndex < currentFrameIdx) {
2584 break; // handler 在外层帧，需弹出当前帧
2585 }
2586 backend->tryStack_.pop_back(); // handler.frameIndex > currentFrameIdx：残留 handler
2587 }
2588
2589 // 3. 当前帧无 handler
2590 if (currentFrameIdx == 0) {
2591 // main 帧未捕获异常
2592 Value thrownValue = bitsToValue(thrownValueBits);
2593 std::string str = thrownValue.toString();
2594 ErrorFormat::truncateForError(str);
2595 if (ctx->hasError)
2596 *ctx->hasError = true;
2597 if (ctx->errorBuffer)
2598 *ctx->errorBuffer = "未捕获的异常: " + str;
2599 return nullptr;
2600 }
2601
2602 // 4. 弹出当前帧，传播到调用者
2603 size_t frameIdx = currentFrameIdx - 1;
2604 JitFrame& frame = ctx->frames[frameIdx];
2605
2606 // 关闭当前帧的 open upvalues（addr <= r13）
2607 backend->closeUpvaluesFrom(r13);
2608
2609 // 清理属于被弹出帧的 tryStack_ handler
2610 while (!backend->tryStack_.empty() && backend->tryStack_.back().frameIndex >= currentFrameIdx) {
2611 backend->tryStack_.pop_back();
2612 }
2613
2614 // 弹出帧
2615 *ctx->frameCount = currentFrameIdx - 1;
2616
2617 // 恢复调用者的 r13
2618 r13 = frame.callerBp;
2619 }
2620}
