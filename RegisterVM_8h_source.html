4// RegisterVM.h — PERF-14: 寄存器式虚拟机
5// ------------------------------------------------------------
6// 执行 RegBytecodeChunk。与栈式 VM 并存，通过 setUseRegisterVM(true) 启用。
7//
8// 寄存器帧模型：
9// - 每个调用帧有 registerCount 个寄存器（Value 数组）
10// - R0..R(arity-1)：参数寄存器（调用时填充）
11// - R(arity)..R(localCount-1)：局部变量寄存器
12// - R(localCount)..R(registerCount-1)：临时寄存器
13//
14// 与栈式 VM 的关键差异：
15// - 无操作数栈，所有操作通过寄存器号寻址
16// - 函数调用时，参数直接放入被调用帧的参数寄存器
17// - 返回值直接放入调用者帧的目标寄存器
18// ============================================================

51// ============================================================
52// 寄存器式 VM 调用帧
53// ============================================================
54// #8 fix: 将 registers 从 std::vector<Value>（每次调用堆分配）改为
55// std::array<Value, MAX_REGISTERS>（栈内联缓冲，零堆分配）。
56// 寄存器号上限为 32（见 RegisterBytecodeBackend 的 vreg 映射检查），
57// 故定长 32 元素数组足够。registerCount 记录实际激活的寄存器数，
58// 用于 reg() 的边界检查（避免读/写未初始化的尾部槽位）。
59struct RegCallFrame {
60 static constexpr size_t MAX_REGISTERS = 32; // 与 RegisterBytecodeBackend 的上限对齐
61
62 const RegBytecodeChunk* chunk = nullptr; // 当前执行的 chunk
63 size_t ip = 0; // 指令指针
64 size_t returnIp = 0; // 返回后调用者的 ip
65 int returnReg = -1; // 返回值写入调用者的寄存器号（-1 = 丢弃）
66 std::array<Value, MAX_REGISTERS> registers{}; // 寄存器窗口（定长，零堆分配）
67 uint8_t registerCount = 0; // 实际激活的寄存器数（≤ MAX_REGISTERS）
68 // 方法调用相关
69 bool isMethodCall = false;
70 bool isInitCall = false;
71 int receiverReg = -1; // 接收者在调用者帧中的寄存器号
72 std::string receiverVarName; // 接收者变量名（用于写回全局变量）
73 int receiverLocalSlot = -1; // 接收者局部变量槽（用于写回局部变量）
74 bool fieldsModified = false; // 是否修改了实例字段
75 // upvalue 支持
76 std::vector<std::shared_ptr<VMUpvalue>> upvalues;
77};
