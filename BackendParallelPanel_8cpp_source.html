41/// 构造执行轨迹 HTML：编译阶段 + 执行阶段 + 统计信息
42QString buildTraceHtml(const QString& compileStage, const QString& execStage, const QString& instrCount,
43 qint64 elapsedMs, bool hasRuntimeError) {
44 QString stats = QStringLiteral("<b>指令数:</b> %1 | <b>耗时:</b> %2 ms | <b>运行时错误:</b> %3")
45 .arg(instrCount)
46 .arg(elapsedMs)
47 .arg(hasRuntimeError ? QStringLiteral("是") : QStringLiteral("否"));
48 return QStringLiteral("<h3>编译阶段</h3><p>%1</p>"
49 "<h3>执行阶段</h3><pre style='white-space:pre-wrap;'>%2</pre>"
50 "<h3>统计信息</h3><p>%3</p>")
51 .arg(compileStage)
52 .arg(execStage)
53 .arg(stats);
54}

56/// 将服务层 BackendExecResult 转换为面板内部结构（含 traceHtml 渲染）
57BackendParallelPanel::BackendExecResult convertServiceResult(const ::BackendExecResult& sr) {
58 BackendParallelPanel::BackendExecResult r;
59 r.success = sr.success;
60 r.output = QString::fromUtf8(sr.output.c_str());
61 r.elapsedMs = static_cast<qint64>(sr.elapsedMs);
62 r.instrCount = QString::fromUtf8(sr.instrCountText().c_str());
63
64 // 状态文本
65 r.status = QString::fromUtf8(sr.statusText().c_str());
66
67 // 编译阶段描述
68 QString compileStage = QString::fromUtf8(sr.compileStageText().c_str());
69
70 // 执行阶段 HTML：输出 + 运行时错误（若有）
71 QString execStage = QString::fromUtf8(sr.output.c_str()).toHtmlEscaped();
72 bool hasRuntimeError = (!sr.success && sr.errorPrefix == "运行时错误");
73 if (hasRuntimeError) {
74 QString err = QString::fromUtf8(sr.errorMsg.c_str()).toHtmlEscaped();
75 execStage += QStringLiteral("\n❌ 运行时错误: %1").arg(err);
76 } else if (!sr.success) {
77 // 词法/语法/编译错误：execStage 留空，compileStage 已含错误信息
78 execStage = QStringLiteral("");
79 }
80 r.traceHtml = buildTraceHtml(compileStage, execStage, r.instrCount, r.elapsedMs, hasRuntimeError);
81 return r;
82}

194/// 运行三后端：通过 BackendExecutionService 串行执行 Interpreter /
195/// StackVM(IR) / RegisterVM(IR)，渲染表格行与执行轨迹，并检查输出一致性。
196void BackendParallelPanel::onRunAllBackends() {
197 std::string src = sourceEdit_->text().toStdString();
198 if (src.empty()) {
199 consistencyLabel_->setText(QString::fromUtf8("❌ 源码为空，请输入 MiniLang 源码"));
200 consistencyLabel_->setStyleSheet(QStringLiteral("color:#C0392B;"));
201 return;
202 }
203
204 // ARCH-10: 通过服务中间层触发三后端执行（主线程串行，避免引擎层非线程安全问题）
205 BackendExecResult interp = convertServiceResult(BackendExecutionService::execute(src, BackendType::Interpreter));
206 BackendExecResult stackvm = convertServiceResult(BackendExecutionService::execute(src, BackendType::StackVM_IR));
207 BackendExecResult regvm = convertServiceResult(BackendExecutionService::execute(src, BackendType::RegisterVM_IR));
208
209 // 渲染表格行
210 renderResult(0, QString::fromUtf8("Interpreter"), interp);
211 renderResult(1, QString::fromUtf8("StackVM (IR)"), stackvm);
212 renderResult(2, QString::fromUtf8("RegisterVM (IR)"), regvm);
213
214 // 渲染执行轨迹
215 interpTrace_->setHtml(interp.traceHtml);
216 stackVmTrace_->setHtml(stackvm.traceHtml);
217 regVmTrace_->setHtml(regvm.traceHtml);
218
219 // 一致性检查
220 renderConsistency(interp, stackvm, regvm);
221}

223/// 渲染单个后端的表格行（后端名/输出/指令数/状态/耗时）。
224void BackendParallelPanel::renderResult(int row, const QString& backendName, const BackendExecResult& r) {
225 if (row < 0 || row >= resultTable_->rowCount()) {
226 return;
227 }
228 auto* nameItem = new QTableWidgetItem(backendName);
229 nameItem->setTextAlignment(Qt::AlignCenter);
230 auto* outItem = new QTableWidgetItem(r.output);
231 auto* instrItem = new QTableWidgetItem(r.instrCount);
232 instrItem->setTextAlignment(Qt::AlignCenter);
233 auto* statusItem = new QTableWidgetItem(r.status);
234 statusItem->setTextAlignment(Qt::AlignCenter);
235 auto* timeItem = new QTableWidgetItem(QString::number(r.elapsedMs));
236 timeItem->setTextAlignment(Qt::AlignCenter);
237
238 // 状态着色：成功绿色，失败红色
239 if (r.success) {
240 statusItem->setForeground(QColor(QStringLiteral("#27AE60")));
241 } else {
242 statusItem->setForeground(QColor(QStringLiteral("#C0392B")));
243 }
244
245 resultTable_->setItem(row, 0, nameItem);
246 resultTable_->setItem(row, 1, outItem);
247 resultTable_->setItem(row, 2, instrItem);
248 resultTable_->setItem(row, 3, statusItem);
249 resultTable_->setItem(row, 4, timeItem);
250}

252/// 渲染一致性检查标签：三后端均成功且输出完全一致则 ✅，否则 ❌。
253void BackendParallelPanel::renderConsistency(const BackendExecResult& interp, const BackendExecResult& stackvm,
254 const BackendExecResult& regvm) {
255 bool allSuccess = interp.success && stackvm.success && regvm.success;
256 bool outputEqual = (interp.output == stackvm.output) && (stackvm.output == regvm.output);
257 if (allSuccess && outputEqual) {
258 consistencyLabel_->setText(QString::fromUtf8("✅ 三后端输出一致，语义等价性验证通过"));
259 consistencyLabel_->setStyleSheet(QStringLiteral("color:#27AE60; font-weight:bold;"));
260 } else {
261 consistencyLabel_->setText(QString::fromUtf8("❌ 三后端输出不一致，存在语义差异！"));
262 consistencyLabel_->setStyleSheet(QStringLiteral("color:#C0392B; font-weight:bold;"));
263 }
264}
