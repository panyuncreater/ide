4 *
5 * 设计：
6 * - core 层中立类型 HostValue（tagged union：null/bool/int64/double/utf8 string），
7 * 避免 core 反向依赖 capi 头（capi 依赖 core）。
8 * - HostFn = HostValue(*)(const HostValue* args, int argc, void* userdata)。
9 * - 全局单例注册表（name → fn+userdata），三后端在"未定义函数"回退路径查表调用。
10 * - Value ↔ HostValue 转换器仅支持 5 种标量（容器参数报错，见 BuiltinMethods 调用点）。
11 *
12 * 线程模型：注册通常在 eval 前（单线程），执行时只读查表。atomic/mutex 非必需
13 * （MVP 假设注册与执行不并发；GUI 场景由宿主保证时序）。
14 *
15 * @see capi/minilang_capi.h（minilang_register_function / minilang_load_plugin）
16 * @see interpreter/BuiltinMethods.cpp（isBuiltinFunction / executeSharedBuiltinFunction 查表接入）
17 */

35/// 宿主互操作值（tagged union，仅标量）。
36/// string 用 std::string 持有（跨 C ABI 时由 capi 层转换为 const char*）。
37struct HostValue {
38 HostValueType type = HostValueType::Null;
39 bool b = false;
40 int64_t i = 0;
41 double d = 0.0;
42 std::string s;

44 static HostValue makeNull() { return HostValue{}; }
45 static HostValue makeBool(bool v) {
46 HostValue hv;

57 static HostValue makeDouble(double v) {

63 static HostValue makeString(std::string v) {
