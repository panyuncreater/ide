76/// P2-2 fix (F3): 把标题文本转为 HTML anchor id（slug）
77/// 规则：非字母数字字符 → 连字符；保留中文；小写；去首尾连字符
78static QString slugify(const QString& text) {
79 QString slug;
80 for (const QChar& c : text) {
81 if (c.isLetterOrNumber()) {
82 slug += c.toLower();
83 } else if (c == '_' || c == '-' || c.isSpace()) {
84 if (!slug.isEmpty() && !slug.endsWith('-'))
85 slug += '-';
86 }
87 // 其他字符忽略
88 }
89 while (slug.startsWith('-'))
90 slug.remove(0, 1);
91 while (slug.endsWith('-'))
92 slug.chop(1);
93 return slug;
94}

96/// P2-2 fix (F4): 对 MiniLang 代码做简单语法高亮
97/// 输入应已 escapeHtml 转义过。基于正则顺序替换：注释 > 字符串 > 关键字 > 数字
98/// 用 <span class="..."> 包裹，CSS 类定义在 buildStylesheet 中
99///
100/// R51-1 fix: 关键字 "class" 会腐蚀已生成的 span 标签的 class="..." 属性
101/// （\bclass\b 匹配属性中的 class）。改用"提取-替换-还原"法：注释/字符串
102/// 标记后先提取到 spans 列表并用唯一占位符替换，关键字/数字替换仅作用于
103/// 剩余纯文本，最后还原占位符。占位符使用 \uE000-\uE0FF 私用区字符（不会
104/// 出现在 MiniLang 源码中，避免与关键字冲突）。
105static QString highlightMiniLang(const QString& escapedCode) {
106 QString result = escapedCode;
107 QStringList spans;
108
109 // 占位符：\uE000 + 序号 + \uE001，私用区字符不会出现在源码或关键字中
110 auto makePlaceholder = [&](int idx) { return QChar(0xE000) + QString::number(idx) + QChar(0xE001); };
111 auto replaceWithPlaceholder = [&](const QRegularExpression& re, const QString& cls) {
112 int from = 0;
113 QRegularExpressionMatch m;
114 while ((m = re.match(result, from)).hasMatch()) {
115 QString captured = m.captured(0);
116 int idx = static_cast<int>(spans.size());
117 spans.push_back(QStringLiteral("<span class=\"%1\">%2</span>").arg(cls, captured));
118 QString placeholder = makePlaceholder(idx);
119 result.replace(m.capturedStart(), m.capturedLength(), placeholder);
120 from = m.capturedStart() + placeholder.length();
121 }
122 };
123
124 // 1. 注释（先处理，避免注释内的关键字/字符串被误高亮）
125 replaceWithPlaceholder(kReMlBlockComment, QStringLiteral("ml-comment"));
126 replaceWithPlaceholder(kReMlLineComment, QStringLiteral("ml-comment"));
127
128 // 2. 字符串
129 replaceWithPlaceholder(kReMlString, QStringLiteral("ml-string"));
130
131 // 3. 关键字（用 \b 边界避免误命中标识符子串）
132 // 此时注释/字符串已被替换为占位符，关键字正则不会匹配到它们的内容
133 for (const QString& kw : kMiniLangKeywords) {
134 QRegularExpression re(QStringLiteral("\\b%1\\b").arg(kw));
135 result.replace(re, QStringLiteral("<span class=\"ml-keyword\">%1</span>").arg(kw));
136 }
137
138 // 4. 数字
139 result.replace(kReMlNumber, QStringLiteral("<span class=\"ml-number\">\\0</span>"));
140
141 // 5. 还原占位符
142 for (int i = static_cast<int>(spans.size()) - 1; i >= 0; --i) {
143 result.replace(makePlaceholder(i), spans[i]);
144 }
145
146 return result;
147}

149/// 检测表格分隔行：行只包含 | - : 空白，且至少有一个 -
150/// 避免使用字符类正则（[\s:-|] 中的 - 会被解释为范围）
151static bool isTableSeparatorLine(const QString& line) {
152 QString t = line.trimmed();
153 if (t.isEmpty())
154 return false;
155 bool hasDash = false;
156 for (const QChar& c : t) {
157 if (c == '|' || c == ':')
158 continue;
159 if (c == '-') {
160 hasDash = true;
161 continue;
162 }
163 if (c.isSpace())
164 continue;
165 return false; // 包含其他字符，不是分隔行
166 }
167 return hasDash;
168}

170/// 转义 HTML 特殊字符（&, <, >），但不破坏已存在的 HTML 实体。
171/// 用于把原始 Markdown 文本安全嵌入 HTML 之前。
172static QString escapeHtml(const QString& s) {
173 QString out = s;
174 out.replace('&', QStringLiteral("&amp;"));
175 out.replace('<', QStringLiteral("&lt;"));
176 out.replace('>', QStringLiteral("&gt;"));
177 return out;
178}

180/// 渲染行内格式：粗体 **x** / 斜体 *x* / 行内代码 `x`。
181/// 输入应已 escapeHtml 转义过。
182static QString renderInline(const QString& s) {
183 QString out = s;
184 // 行内代码 `code`（先处理，避免 code 内的 ** 被后续规则误处理）
185 out.replace(kReInlineCode, QStringLiteral("<code>\\1</code>"));
186 // 粗体 **text**
187 out.replace(kReBold, QStringLiteral("<b>\\1</b>"));
188 // 斜体 *text*（避免与粗体冲突，要求 * 两侧非 *）
189 out.replace(kReItalic, QStringLiteral("<i>\\1</i>"));
190 // P1-4 fix (F16): 行内链接 [text](url) — 最后处理，避免 code/bold/italic
191 // 内的方括号被误匹配。url 中的 & 已被 escapeHtml 转成 &amp;，
192 // 需在嵌入 href 时反转回来（HTML 属性值中 &amp; 合法但 QTextBrowser
193 // 解析 href 时会再次 unescape，导致 &amp;amp; 双重转义）。
194 QRegularExpressionMatch it;
195 int pos = 0;
196 QString result;
197 while ((it = kReLink.match(out, pos)).hasMatch()) {
198 result += out.mid(pos, it.capturedStart() - pos);
199 QString text = it.captured(1);
200 QString url = it.captured(2);
201 url.replace(QStringLiteral("&amp;"), QStringLiteral("&"));
202 // R51-6 fix: URL 中的双引号会破坏 href 属性边界，需转义为 &quot;
203 url.replace(QChar('"'), QStringLiteral("&quot;"));
204 result += QStringLiteral("<a href=\"%1\">%2</a>").arg(url, text);
205 pos = it.capturedEnd();
206 }
207 result += out.mid(pos);
208 return result;
209}

259QString markdownToHtml(const QString& markdown, const QString& codeBlockBg) {
260 if (markdown.isEmpty()) {
261 return QStringLiteral("<html><head></head><body></body></html>");
262 }
263
264 const QString bg = codeBlockBg.isEmpty() ? QStringLiteral("#F5F5F5") : codeBlockBg;
265 const QString codeBlockStyle =
266 QStringLiteral("background:%1; padding:10px 12px; border-radius:6px; "
267 "border:1px solid #E0E0E0; "
268 "font-family:'Cascadia Code','Cascadia Mono','Consolas','JetBrains Mono','Source Code "
269 "Pro','Menlo','DejaVu Sans Mono','Courier New',monospace; "
270 "font-size:13px; "
271 "white-space:pre-wrap;")
272 .arg(bg);
273
274 // 按行扫描，识别块级结构
275 const QStringList lines = markdown.split('\n');
276 QStringList html;
277 html << QStringLiteral("<html><head>");
278 html << buildStylesheet(bg);
279 html << QStringLiteral("</head><body>");
280
281 bool inCodeBlock = false;
282 QString codeBlockContent;
283 QString codeBlockLang;
284 bool inUl = false; // 无序列表
285 bool inOl = false; // 有序列表
286 bool inBlockquote = false; // 引用块
287 QStringList blockquoteLines; // 引用块累积行
288 QStringList paragraph; // 当前段落累积的行
289
290 // 表格状态
291 bool inTable = false;
292 QStringList tableHeader; // 表头单元格
293 QStringList tableRows; // 数据行（每行一个字符串列表）
294
295 auto closeLists = [&]() {
296 if (inUl) {
297 html << QStringLiteral("</ul>");
298 inUl = false;
299 }
300 if (inOl) {
301 html << QStringLiteral("</ol>");
302 inOl = false;
303 }
304 };
305 auto flushParagraph = [&]() {
306 if (!paragraph.isEmpty()) {
307 QString joined = paragraph.join(QStringLiteral("\n"));
308 html << QStringLiteral("<p>") << renderInline(joined) << QStringLiteral("</p>");
309 paragraph.clear();
310 }
311 };
312 auto flushBlockquote = [&]() {
313 if (!blockquoteLines.isEmpty()) {
314 QString joined = blockquoteLines.join(QStringLiteral("<br>"));
315 html << QStringLiteral("<blockquote>") << renderInline(joined) << QStringLiteral("</blockquote>");
316 blockquoteLines.clear();
317 inBlockquote = false;
318 }
319 };
320 auto flushTable = [&]() {
321 if (inTable && !tableHeader.isEmpty()) {
322 html << QStringLiteral("<table>");
323 // 表头
324 html << QStringLiteral("<tr>");
325 for (const auto& h : tableHeader) {
326 // R51-3 fix: 表格单元格需先 escapeHtml 再 renderInline（对齐列表/标题/引用块）
327 html << QStringLiteral("<th>") << renderInline(escapeHtml(h.trimmed())) << QStringLiteral("</th>");
328 }
329 html << QStringLiteral("</tr>");
330 // 数据行
331 for (const auto& row : tableRows) {
332 QStringList cells = row.split(QStringLiteral("|"));
333 html << QStringLiteral("<tr>");
334 for (const auto& c : cells) {
335 html << QStringLiteral("<td>") << renderInline(escapeHtml(c.trimmed())) << QStringLiteral("</td>");
336 }
337 html << QStringLiteral("</tr>");
338 }
339 html << QStringLiteral("</table>");
340 }
341 inTable = false;
342 tableHeader.clear();
343 tableRows.clear();
344 };
345
346 for (int i = 0; i < lines.size(); ++i) {
347 const QString& line = lines[i];
348
349 // ---- 围栏代码块 ----
350 if (line.trimmed().startsWith(QStringLiteral("```"))) {
351 if (!inCodeBlock) {
352 // 进入代码块：先关闭段落/列表/引用/表格
353 flushParagraph();
354 closeLists();
355 flushBlockquote();
356 flushTable();
357 inCodeBlock = true;
358 codeBlockContent.clear();
359 codeBlockLang = line.trimmed().mid(3).trimmed();
360 } else {
361 // 退出代码块
362 QString escaped = escapeHtml(codeBlockContent);
363 // 去掉尾部多余换行
364 while (escaped.endsWith('\n'))
365 escaped.chop(1);
366 // P2-2 fix (F4): MiniLang 语法高亮
367 // 触发条件：codeBlockLang 为 minilang / ml / mlang / mini（不区分大小写）
368 QString lowerLang = codeBlockLang.toLower();
369 bool isMiniLang = (lowerLang == QStringLiteral("minilang") || lowerLang == QStringLiteral("ml") ||
370 lowerLang == QStringLiteral("mlang") || lowerLang == QStringLiteral("mini"));
371 if (isMiniLang) {
372 escaped = highlightMiniLang(escaped);
373 }
374 // 代码块语言标签（仅作为注释显示在代码上方，不渲染为单独元素）
375 if (!codeBlockLang.isEmpty()) {
376 html << QStringLiteral("<div style=\"font-size:11px;color:#6E6E6E;margin-bottom:2px;\">")
377 << escapeHtml(codeBlockLang) << QStringLiteral("</div>");
378 }
379 html << QStringLiteral("<pre style=\"%1\">").arg(codeBlockStyle) << escaped << QStringLiteral("</pre>");
380 inCodeBlock = false;
381 codeBlockContent.clear();
382 codeBlockLang.clear();
383 }
384 continue;
385 }
386 if (inCodeBlock) {
387 // 代码块内容原样保留（最后整体 escapeHtml）
388 codeBlockContent += line + QStringLiteral("\n");
389 continue;
390 }
391
392 // ---- 空行：段落分隔 ----
393 if (line.trimmed().isEmpty()) {
394 flushParagraph();
395 closeLists();
396 flushBlockquote();
397 flushTable();
398 continue;
399 }
400
401 // ---- 水平分割线 --- （单独成行，至少 3 个 -） ----
402 QString trimmed = line.trimmed();
403 if (trimmed == QStringLiteral("---") || trimmed == QStringLiteral("***")) {
404 flushParagraph();
405 closeLists();
406 flushBlockquote();
407 flushTable();
408 html << QStringLiteral("<hr>");
409 continue;
410 }
411
412 // ---- 标题 # / ## / ### ----
413 QRegularExpressionMatch hm = kReHeading.match(line);
414 if (hm.hasMatch()) {
415 flushParagraph();
416 closeLists();
417 flushBlockquote();
418 flushTable();
419 int level = hm.captured(1).length();
420 QString rawText = hm.captured(2);
421 // P2-2 fix (F3): 标题生成 anchor id（slug），用于章节锚点目录跳转
422 QString anchor = slugify(rawText);
423 QString text = renderInline(escapeHtml(rawText));
424 if (!anchor.isEmpty()) {
425 html << QStringLiteral("<h%1 id=\"%2\">%3</h%1>").arg(level).arg(anchor, text);
426 } else {
427 html << QStringLiteral("<h%1>%2</h%1>").arg(level).arg(text);
428 }
429 continue;
430 }
431
432 // ---- 任务列表 - [ ] / - [x] ----
433 QRegularExpressionMatch tm = kReTaskList.match(line);
434 if (tm.hasMatch()) {
435 flushParagraph();
436 if (inOl) {
437 html << QStringLiteral("</ol>");
438 inOl = false;
439 }
440 if (!inUl) {
441 html << QStringLiteral("<ul>");
442 inUl = true;
443 }
444 QString checked = tm.captured(1).toLower();
445 // R51-7 fix: 移除未使用的 text 变量，直接在输出处调用 renderInline
446 QString checkboxClass = (checked == QStringLiteral("x")) ? QStringLiteral("task-checkbox checked")
447 : QStringLiteral("task-checkbox");
448 QString checkboxSymbol = (checked == QStringLiteral("x")) ? QStringLiteral("✓") : QString();
449 html << QStringLiteral("<li class=\"task-list-item\">")
450 << QStringLiteral("<span class=\"%1\">%2</span>").arg(checkboxClass, checkboxSymbol)
451 << renderInline(escapeHtml(tm.captured(2))) << QStringLiteral("</li>");
452 continue;
453 }

455 // ---- 无序列表 - / * 开头（避免与粗体冲突：要求行首是 - 或 *，后跟空格） ----
456 QRegularExpressionMatch ulm = kReUnorderedList.match(line);
457 if (ulm.hasMatch()) {
458 flushParagraph();
459 flushBlockquote();
460 flushTable();
461 if (inOl) {
462 html << QStringLiteral("</ol>");
463 inOl = false;
464 }
465 if (!inUl) {
466 html << QStringLiteral("<ul>");
467 inUl = true;
468 }
469 html << QStringLiteral("<li>") << renderInline(escapeHtml(ulm.captured(1))) << QStringLiteral("</li>");
470 continue;
471 }
472
473 // ---- 有序列表 1. / 2. 开头 ----
474 QRegularExpressionMatch olm = kReOrderedList.match(line);
475 if (olm.hasMatch()) {
476 flushParagraph();
477 flushBlockquote();
478 flushTable();
479 if (inUl) {
480 html << QStringLiteral("</ul>");
481 inUl = false;
482 }
483 if (!inOl) {
484 html << QStringLiteral("<ol>");
485 inOl = true;
486 }
487 html << QStringLiteral("<li>") << renderInline(escapeHtml(olm.captured(1))) << QStringLiteral("</li>");
488 continue;
489 }
490
491 // ---- 引用块 > text ----
492 QRegularExpressionMatch bqm = kReBlockquote.match(line);
493 if (bqm.hasMatch()) {
494 flushParagraph();
495 closeLists();
496 flushTable();
497 inBlockquote = true;
498 blockquoteLines << escapeHtml(bqm.captured(1));
499 continue;
500 }
501 // 非引用行出现，关闭引用块
502 if (inBlockquote) {
503 flushBlockquote();
504 }
505
506 // ---- 表格 | col1 | col2 | ----
507 QRegularExpressionMatch trm = kReTableRow.match(line);
508 if (trm.hasMatch()) {
509 // 检查下一行是否是分隔行（如果是，说明这是表头）
510 bool isNextSeparator = (i + 1 < lines.size()) && isTableSeparatorLine(lines[i + 1]);
511 if (isNextSeparator && !inTable) {
512 // 这是表头行
513 flushParagraph();
514 closeLists();
515 flushBlockquote();
516 inTable = true;
517 tableHeader = trm.captured(1).split(QStringLiteral("|"));
518 // 跳过分隔行
519 ++i;
520 continue;
521 } else if (inTable) {
522 // 数据行
523 tableRows << trm.captured(1);
524 continue;
525 }
526 // 如果不是表格上下文，当作普通段落处理
527 }
528 // 非表格行出现，关闭表格
529 if (inTable) {
530 flushTable();
531 }
532
533 // ---- 普通段落行 ----
534 // 先 escape，再累积（段落末尾统一 renderInline）
535 paragraph << escapeHtml(line);
536 }
537
538 // 收尾：处理文件末尾未关闭的代码块/段落/列表/引用/表格
539 if (inCodeBlock) {
540 // 代码块未闭合（用户输入不完整），按代码块输出
541 QString escaped = escapeHtml(codeBlockContent);
542 while (escaped.endsWith('\n'))
543 escaped.chop(1);
544 html << QStringLiteral("<pre style=\"%1\">").arg(codeBlockStyle) << escaped << QStringLiteral("</pre>");
545 }
546 flushParagraph();
547 closeLists();
548 flushBlockquote();
549 flushTable();
550
551 html << QStringLiteral("</body></html>");
552 return html.join(QString());
553}

559QString markdownToHtmlFragment(const QString& markdown, const QString& codeBlockBg) {
560 QString full = markdownToHtml(markdown, codeBlockBg);
561 // 剥离 <html><head>...</head><body>...</body></html> 包裹，返回 body 内部片段
562 const QString headStart = QStringLiteral("<html><head>");
563 const QString bodyStart = QStringLiteral("</head><body>");
564 const QString bodyEnd = QStringLiteral("</body></html>");
565 if (full.startsWith(headStart) && full.endsWith(bodyEnd)) {
566 int bodyStartIdx = full.indexOf(bodyStart);
567 if (bodyStartIdx >= 0) {
568 int contentStart = bodyStartIdx + bodyStart.length();
569 int contentEnd = full.length() - bodyEnd.length();
570 return full.mid(contentStart, contentEnd - contentStart);
571 }
572 }
573 // 兼容旧版无 <head> 的格式
574 const QString prefix = QStringLiteral("<html><body>");
575 const QString suffix = QStringLiteral("</body></html>");
576 if (full.startsWith(prefix) && full.endsWith(suffix)) {
577 return full.mid(prefix.length(), full.length() - prefix.length() - suffix.length());
578 }
579 return full;
580}

590std::vector<HeadingEntry> extractHeadings(const QString& markdown) {
591 std::vector<HeadingEntry> headings;
592 if (markdown.isEmpty())
593 return headings;
594
595 const QStringList lines = markdown.split('\n');
596 bool inCodeBlock = false;
597 for (const QString& line : lines) {
598 // 跟踪围栏代码块状态，代码块内的 # 不算标题
599 if (line.trimmed().startsWith(QStringLiteral("```"))) {
600 inCodeBlock = !inCodeBlock;
601 continue;
602 }
603 if (inCodeBlock)
604 continue;
605
606 QRegularExpressionMatch hm = kReHeading.match(line);
607 if (hm.hasMatch()) {
608 HeadingEntry e;
609 e.level = hm.captured(1).length();
610 e.text = hm.captured(2);
611 e.anchor = slugify(e.text);
612 if (!e.anchor.isEmpty()) {
613 headings.push_back(std::move(e));
614 }
615 }
616 }
617 return headings;
618}

620QString buildTableOfContents(const QString& markdown, const QString& tocTitle, int maxLevel) {
621 auto headings = extractHeadings(markdown);
622 if (headings.empty())
623 return QString();
624
625 QStringList html;
626 html << QStringLiteral("<div class=\"toc-box\">");
627 if (!tocTitle.isEmpty()) {
628 html << QStringLiteral("<div class=\"toc-title\">") << escapeHtml(tocTitle) << QStringLiteral("</div>");
629 }
630 html << QStringLiteral("<ul class=\"toc-list\">");
631 for (const auto& h : headings) {
632 if (h.level > maxLevel)
633 continue;
634 // 根据级别缩进（h1 不缩进，h2 缩进 1 级，h3 缩进 2 级）
635 int indent = h.level - 1;
636 if (indent < 0)
637 indent = 0;
638 QString style = (indent > 0) ? QStringLiteral(" style=\"margin-left:%1px;\"").arg(indent * 12) : QString();
639 // 内部锚点链接使用 #anchor 格式（QTextBrowser 支持）
640 html << QStringLiteral("<li%1><a href=\"#%2\">%3</a></li>")
641 .arg(style, h.anchor, renderInline(escapeHtml(h.text)));
642 }
643 html << QStringLiteral("</ul>");
644 html << QStringLiteral("</div>");
645 return html.join(QString());
646}
