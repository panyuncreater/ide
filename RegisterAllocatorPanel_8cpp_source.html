128/// 判断字符是否为标识符字符（字母/数字/下划线）
129bool isIdentChar(char c) {
130 return std::isalnum(static_cast<unsigned char>(c)) != 0 || c == '_';
131}
132
133/// 从字符串的指定位置提取标识符
134std::string extractIdent(const std::string& s, size_t start) {
135 size_t i = start;
136 // 跳过前导空白
137 while (i < s.size() && (s[i] == ' ' || s[i] == '\t')) {
138 ++i;
139 }
140 // 读取标识符
141 std::string name;
142 while (i < s.size() && isIdentChar(s[i])) {
143 name += s[i];
144 ++i;
145 }
146 return name;
147}
148
149/// 去除行首空白
150std::string trimLeft(const std::string& s) {
151 size_t i = 0;
152 while (i < s.size() && (s[i] == ' ' || s[i] == '\t')) {
153 ++i;
154 }
155 return s.substr(i);
156}
157
158/// 检查行中是否包含某个标识符（词边界匹配）
159bool containsIdent(const std::string& line, const std::string& word) {
160 if (word.empty()) {
161 return false;
162 }
163 size_t pos = 0;
164 while ((pos = line.find(word, pos)) != std::string::npos) {
165 bool leftOk = (pos == 0) || !isIdentChar(line[pos - 1]);
166 bool rightOk = (pos + word.size() >= line.size()) || !isIdentChar(line[pos + word.size()]);
167 if (leftOk && rightOk) {
168 return true;
169 }
170 pos += word.size();
171 }
172 return false;
173}

175/// 变量活跃区间（分析内部用）
176struct VarLiveRange {
177 std::string name;
178 int defLine; // 定义行（1-based）
179 int lastUseLine; // 最后使用行
180};
181
182/// 按行分割代码（支持 \n 和 \r\n）
183void splitLines(const std::string& code, std::vector<std::string>& out) {
184 out.clear();
185 std::string current;
186 for (char c : code) {
187 if (c == '\n') {
188 out.push_back(current);
189 current.clear();
190 } else if (c != '\r') {
191 current += c;
192 }
193 }
194 if (!current.empty()) {
195 out.push_back(current);
196 }
197}
198
199/// 从代码中提取变量定义和活跃区间
200std::vector<VarLiveRange> extractLiveRanges(const std::string& code) {
201 std::vector<std::string> lines;
202 splitLines(code, lines);
203
204 std::vector<VarLiveRange> ranges;
205
206 // 第一步：识别变量定义（var 声明 + for 循环变量）
207 for (int i = 0; i < static_cast<int>(lines.size()); ++i) {
208 std::string trimmed = trimLeft(lines[i]);
209 if (trimmed.rfind("var ", 0) == 0) {
210 std::string name = extractIdent(trimmed, 4);
211 if (!name.empty()) {
212 ranges.push_back({name, i + 1, i + 1});
213 }
214 } else if (trimmed.rfind("for ", 0) == 0) {
215 // for <var> in <range> {
216 std::string name = extractIdent(trimmed, 4);
217 if (!name.empty()) {
218 ranges.push_back({name, i + 1, i + 1});
219 }
220 }
221 }
222
223 // 第二步：计算每个变量的最后使用行
224 for (auto& r : ranges) {
225 for (int i = r.defLine - 1; i < static_cast<int>(lines.size()); ++i) {
226 if (containsIdent(lines[i], r.name)) {
227 r.lastUseLine = i + 1;
228 }
229 }
230 }
231
232 return ranges;
233}
234
235/// 计算活跃峰值（同时活跃的最大变量数）
236int computeActivePeak(const std::vector<VarLiveRange>& ranges) {
237 if (ranges.empty()) {
238 return 0;
239 }
240 int maxLine = 0;
241 for (const auto& r : ranges) {
242 maxLine = std::max(maxLine, r.lastUseLine);
243 }
244 int peak = 0;
245 for (int line = 1; line <= maxLine; ++line) {
246 int count = 0;
247 for (const auto& r : ranges) {
248 if (r.defLine <= line && line <= r.lastUseLine) {
249 ++count;
250 }
251 }
252 peak = std::max(peak, count);
253 }
254 return peak;
255}

257/// 活跃区间项（线性扫描内部用）
258struct ActiveItem {
259 VarLiveRange range;
260 int reg; // 0-31 已分配，-1 已溢出，-2 已释放
261};
262
263/// 运行简化版线性扫描算法
264/// - 按 defLine 排序
265/// - 贪心分配最低空闲寄存器
266/// - 无空闲寄存器时溢出结束最远的变量（经典启发式）
267std::vector<VarAllocInfo> runLinearScan(const std::vector<VarLiveRange>& ranges, int& spillCount) {
268 spillCount = 0;
269 std::vector<VarAllocInfo> result;
270
271 if (ranges.empty()) {
272 return result;
273 }
274
275 // 按定义行排序
276 std::vector<VarLiveRange> sorted = ranges;
277 std::sort(sorted.begin(), sorted.end(),
278 [](const VarLiveRange& a, const VarLiveRange& b) { return a.defLine < b.defLine; });
279
280 std::vector<ActiveItem> active;
281 std::vector<bool> regFree(RegisterAllocatorLibrary::kRegisterCount, true);
282
283 for (const auto& r : sorted) {
284 // 释放已结束的活跃区间
285 for (auto& a : active) {
286 if (a.reg >= 0 && a.range.lastUseLine < r.defLine) {
287 regFree[a.reg] = true;
288 a.reg = -2; // 标记已释放
289 }
290 }
291 active.erase(std::remove_if(active.begin(), active.end(), [](const ActiveItem& a) { return a.reg == -2; }),
292 active.end());
293
294 // 查找最低空闲寄存器
295 int freeReg = -1;
296 for (int i = 0; i < RegisterAllocatorLibrary::kRegisterCount; ++i) {
297 if (regFree[i]) {
298 freeReg = i;
299 break;
300 }
301 }
302
303 VarAllocInfo info;
304 info.varName = r.name;
305 info.liveRange = "行" + std::to_string(r.defLine) + "-行" + std::to_string(r.lastUseLine);
306 info.spilled = false;
307
308 if (freeReg >= 0) {
309 // 有空闲寄存器，直接分配
310 regFree[freeReg] = false;
311 info.regName = "R" + std::to_string(freeReg);
312 active.push_back({r, freeReg});
313 } else {
314 // 无空闲寄存器，需要溢出决策
315 // 找到活跃集中结束最远的变量
316 int furthestIdx = -1;
317 int furthestEnd = r.lastUseLine;
318 for (int i = 0; i < static_cast<int>(active.size()); ++i) {
319 if (active[i].range.lastUseLine > furthestEnd) {
320 furthestEnd = active[i].range.lastUseLine;
321 furthestIdx = i;
322 }
323 }
324
325 if (furthestIdx >= 0 && active[furthestIdx].range.lastUseLine > r.lastUseLine) {
326 // 溢出最远的活跃变量，将其寄存器分配给当前变量
327 int reg = active[furthestIdx].reg;
328 // 在结果中找到被溢出的变量并更新
329 for (auto& prev : result) {
330 if (prev.varName == active[furthestIdx].range.name) {
331 prev.regName = "溢出";
332 prev.spilled = true;
333 ++spillCount;
334 break;
335 }
336 }
337 active[furthestIdx] = {r, reg};
338 info.regName = "R" + std::to_string(reg);
339 } else {
340 // 当前变量被溢出（结束时间不比最远的早）
341 active.push_back({r, -1});
342 info.regName = "溢出";
343 info.spilled = true;
344 ++spillCount;
345 }
346 }
347
348 result.push_back(info);
349 }
350
351 return result;
352}

660void RegisterAllocatorPanel::renderRegisterUsage() {
661 // 清空所有寄存器的分配
662 for (int i = 0; i < RegisterAllocatorLibrary::kRegisterCount; ++i) {
663 regUsageTable_->setItem(i, 0, new QTableWidgetItem(QStringLiteral("R%1").arg(i)));
664 regUsageTable_->setItem(i, 1, new QTableWidgetItem(tr("（空闲）")));
665 }
666
667 // 收集每个寄存器分配的变量（可能有多个变量复用同一寄存器）
668 std::vector<std::vector<std::string>> regVars(RegisterAllocatorLibrary::kRegisterCount);
669 for (const auto& a : currentAllocations_) {
670 if (!a.spilled && a.regName.size() > 1 && a.regName[0] == 'R') {
671 // 解析寄存器号
672 int regNum = -1;
673 try {
674 regNum = std::stoi(a.regName.substr(1));
675 } catch (...) {
676 continue;
677 }
678 if (regNum >= 0 && regNum < RegisterAllocatorLibrary::kRegisterCount) {
679 regVars[regNum].push_back(a.varName);
680 }
681 }
682 }
683
684 // 渲染
685 for (int i = 0; i < RegisterAllocatorLibrary::kRegisterCount; ++i) {
686 if (regVars[i].empty()) {
687 regUsageTable_->setItem(i, 1, new QTableWidgetItem(tr("（空闲）")));
688 } else {
689 std::string combined;
690 for (size_t j = 0; j < regVars[i].size(); ++j) {
691 if (j > 0) {
692 combined += ", ";
693 }
694 combined += regVars[i][j];
695 }
696 auto* item = new QTableWidgetItem(QString::fromUtf8(combined.c_str()));
697 item->setToolTip(QString::fromUtf8(combined.c_str()));
698 regUsageTable_->setItem(i, 1, item);
699 }

703void RegisterAllocatorPanel::renderDetails() {

715
716 if (currentSpillCount_ == 0) {
717 html << "<h3>溢出决策</h3>";

724 html << "<h3>溢出决策</h3>";
725 html << "<p>活跃变量数超过寄存器上限 " << RegisterAllocatorLibrary::kRegisterCount
726 << "，以下变量被溢出到栈：</p><ul>";
727 for (const auto& a : currentAllocations_) {
728 if (a.spilled) {
729 html << "<li><b>" << a.varName << "</b>（活跃区间：" << a.liveRange << "）</li>";
730 }
731 }
732 html << "</ul>";
733 html << "<p>溢出策略：<b>选择结束最远的变量溢出</b>"
734 << "（线性扫描经典启发式）。被溢出的变量在需要时从栈加载到寄存器，"
735 << "使用后再写回栈。</p>";
736 }
737
738 html << "<h3>性能影响</h3>";
739 if (currentSpillCount_ == 0) {
740 html << "<p style='color:#2E7D32;'>所有变量访问均在寄存器中，"
741 << "<b>性能最优</b>，无额外内存访问开销。</p>";
742 } else {
743 html << "<p style='color:#C62828;'>溢出变量每次访问需要额外的 <b>load/store</b> 操作，"
744 << "预计增加约 <b>" << currentSpillCount_ * 2 << "</b> 次内存访问。"
745 << "在热点循环中，溢出会显著降低性能。</p>";
746 }

748 html << "<h3>寄存器复用情况</h3>";
749 // 统计寄存器复用
750 int reusedRegs = 0;
751 std::vector<int> regVarCount(RegisterAllocatorLibrary::kRegisterCount, 0);
752 for (const auto& a : currentAllocations_) {
753 if (!a.spilled && a.regName.size() > 1 && a.regName[0] == 'R') {
754 int regNum = -1;
755 try {
756 regNum = std::stoi(a.regName.substr(1));
757 } catch (...) {
758 continue;
759 }
760 if (regNum >= 0 && regNum < RegisterAllocatorLibrary::kRegisterCount) {
761 regVarCount[regNum]++;
762 if (regVarCount[regNum] > 1) {
763 reusedRegs++;
764 }
765 }
766 }
767 }
768 if (reusedRegs > 0) {
769 html << "<p>有 <b>" << reusedRegs << "</b> 个寄存器被多个变量复用"
770 << "（变量死亡后寄存器释放给新变量）。"
771 << "这是线性扫描的寄存器复用优化，减少了对寄存器总数的需求。</p>";
772 } else {
773 html << "<p>无寄存器复用（每个变量独占一个寄存器）。</p>";
774 }
775
776 detailBrowser_->setHtml(QString::fromUtf8(html.str().c_str()));
777}
