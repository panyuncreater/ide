312// 拓展二期：双击「命中条件」(列 4)/「依赖行」(列 5)弹输入框编辑
313// ------------------------------------------------------------
314// 提交到 IdeController facade（双路径同步 DebugController + VmStepper），

317void BreakpointConditionPanel::onBreakpointCellDoubleClicked(int row, int column) {
318 if (!controller_ || row < 0)
319 return;
320 if (column != 4 && column != 5)
321 return;
322 auto* lineItem = breakpointTable_->item(row, 0);
323 if (!lineItem)
324 return;
325 bool lineOk = false;
326 int line = lineItem->text().toInt(&lineOk);
327 if (!lineOk || line <= 0)
328 return;
329
330 if (column == 4) {
331 // 命中条件："N"/"== N"/">= N"/"> N"/"% N"，空=清除
332 std::string cur = controller_->getBreakpointHitCondition(line);
333 bool ok = false;
334 QString text = QInputDialog::getText(
335 this, QString::fromUtf8("命中条件"),
336 QString::fromUtf8("行 %1 的命中条件（如 \">= 3\" / \"% 2\" / \"5\"，留空清除）：").arg(line),
337 QLineEdit::Normal, QString::fromUtf8(cur.c_str()), &ok);
338 if (!ok)
339 return;
340 controller_->setBreakpointHitCondition(line, text.trimmed().toStdString());
341 } else {
342 // 依赖行：仅当依赖行断点命中过后本断点才激活（0 清除）
343 int curDep = controller_->getBreakpointDependency(line);
344 bool ok = false;
345 int dep = QInputDialog::getInt(this, QString::fromUtf8("依赖断点"),
346 QString::fromUtf8("行 %1 依赖的断点行号（0 = 清除依赖）：").arg(line),
347 curDep > 0 ? curDep : 0, 0, 1000000, 1, &ok);
348 if (!ok)
349 return;
350 controller_->setBreakpointDependency(line, dep);
351 }
352 refreshLive();
353}

358void BreakpointConditionPanel::refreshLive() {
359 if (!controller_) {
360 liveStatusLabel_->setText(QString::fromUtf8("未绑定控制器"));
361 return;
362 }
363 // R52-3 fix: 补充 Interpreter 调试 resume 期间的状态守卫。原仅检查 VM 路径，
364 // 缺 Interpreter 调试路径——resume 期间 worker 线程活跃，可能并发更新
365 // debugCoord_ 内部的 breakpointHitCounts_，refreshLive 读取会触发 UB。
366 // 对齐 CallStackPanel / VariableInspectorPanel 的双路径守卫模式。
367 if (controller_->isRunning() && controller_->isDebugRun() && !controller_->isDebugPaused()) {
368 liveStatusLabel_->setText(QString::fromUtf8("状态：调试运行中（暂停后刷新）"));
369 return;
370 }
371 // AUDIT-P2 fix: VM 运行期间读断点元数据（getBreakpointCondition/getBreakpointHitCount）
372 // 可能不一致，对齐 CallStackPanel / VariableInspectorPanel 模式，运行中且未暂停时早退。
373 if (controller_->isVmRunning() && !controller_->isDebugPaused()) {
374 liveStatusLabel_->setText(QString::fromUtf8("状态：VM 运行中（暂停后刷新）"));
375 return;
376 }
377
378 QSet<int> breakpoints = controller_->getBreakpoints();
379 if (breakpoints.isEmpty()) {
380 liveStatusLabel_->setText(QString::fromUtf8("无断点"));
381 breakpointTable_->setRowCount(0);
382 return;
383 }
384
385 // 按行号排序展示
386 QList<int> sortedLines = breakpoints.values();
387 std::sort(sortedLines.begin(), sortedLines.end());
388 breakpointTable_->setRowCount(sortedLines.size());
389
390 bool isRunning = controller_->isRunning();
391 bool isDebugRun = controller_->isDebugRun();
392 bool isVmRunning = controller_->isVmRunning();
393 QString modeText;
394 if (isVmRunning) {
395 modeText = controller_->isVmRegisterMode() ? QString::fromUtf8("RegisterVM 调试中")
396 : QString::fromUtf8("StackVM 调试中");
397 } else if (isRunning && isDebugRun) {
398 modeText = QString::fromUtf8("Interpreter 调试中");
399 } else {
400 modeText = QString::fromUtf8("未运行");
401 }
402 liveStatusLabel_->setText(QString::fromUtf8("共 %1 个断点 | %2").arg(sortedLines.size()).arg(modeText));
403
404 for (int i = 0; i < sortedLines.size(); ++i) {
405 int line = sortedLines[i];
406 std::string cond = controller_->getBreakpointCondition(line);
407 int hitCount = controller_->getBreakpointHitCount(line);
408 // R104: 类型列（Line / Logpoint）
409 BreakpointKind kind = controller_->getBreakpointKind(line);
410 QString kindText =
411 (kind == BreakpointKind::Logpoint) ? QString::fromUtf8("日志断点") : QString::fromUtf8("行断点");
412 std::string logMsg = controller_->getLogpointMessage(line);
413
414 breakpointTable_->setItem(i, 0, new QTableWidgetItem(QString::number(line)));
415 breakpointTable_->setItem(i, 1, new QTableWidgetItem(kindText));
416 QString condDisplay;
417 if (kind == BreakpointKind::Logpoint) {
418 condDisplay = logMsg.empty() ? QString::fromUtf8("（无日志消息）") : QString::fromUtf8(logMsg.c_str());
419 if (!cond.empty())
420 condDisplay += QString::fromUtf8(" [if ") + QString::fromUtf8(cond.c_str()) + QString::fromUtf8("]");
421 } else {
422 condDisplay = cond.empty() ? QString::fromUtf8("（无条件）") : QString::fromUtf8(cond.c_str());
423 }
424 breakpointTable_->setItem(i, 2, new QTableWidgetItem(condDisplay));
425 breakpointTable_->setItem(i, 3, new QTableWidgetItem(QString::number(hitCount)));
426 // 拓展二期：命中条件列 + 依赖行列（双击可编辑）
427 std::string hitCond = controller_->getBreakpointHitCondition(line);
428 int depLine = controller_->getBreakpointDependency(line);
429 breakpointTable_->setItem(
430 i, 4,
431 new QTableWidgetItem(hitCond.empty() ? QString::fromUtf8("（无）") : QString::fromUtf8(hitCond.c_str())));
432 breakpointTable_->setItem(
433 i, 5, new QTableWidgetItem(depLine > 0 ? QString::number(depLine) : QString::fromUtf8("（无）")));
434 // 状态：综合显示
435 QString status;
436 if (kind == BreakpointKind::Logpoint) {
437 status = QString::fromUtf8("记录日志");
438 } else if (cond.empty()) {
439 status = QString::fromUtf8("普通断点");
440 } else {
441 status = QString::fromUtf8("条件断点");
442 }
443 breakpointTable_->setItem(i, 6, new QTableWidgetItem(status)); // 拓展二期：状态列后移至列 6

472void BreakpointConditionPanel::populateScenarioDetail(int index) {
473 const auto& items = BreakpointConditionLibrary::scenarios();
474 if (index < 0 || index >= (int)items.size()) {
475 currentSampleCode_.clear();
476 return;
477 }
478 const auto& s = items[index];
479 currentSampleCode_ = s.sampleCode; // 保存当前示例代码供加载按钮使用
480
481 // 散文式字段（description）走 Markdown 渲染，支持 **粗体** / `code` / 列表
482 QString descFragment = MarkdownRenderer::markdownToHtmlFragment(s.description);
483
484 // AUDIT-P2 fix: HTML 转义——sampleCode 含 < > 字符（如 if (n < 0)），
485 // condition 含 < >（如 a > 0 && b < 100）。未转义会导致 HTML 标签注入/渲染错误。
486 auto escapeHtml = [](const std::string& s) -> std::string {
487 std::string out;
488 out.reserve(s.size());
489 for (char c : s) {
490 switch (c) {
491 case '&':
492 out += "&amp;";
493 break;
494 case '<':
495 out += "&lt;";
496 break;
497 case '>':
498 out += "&gt;";
499 break;
500 case '"':
501 out += "&quot;";
502 break;
503 case '\'':
504 out += "&#39;";
505 break;
506 default:
507 out += c;
508 break;
509 }
510 }
511 return out;
512 };
513
514 std::ostringstream os;
515 os << "<h3>" << escapeHtml(s.title) << "</h3>";
516 os << "<p><b>条件表达式：</b> <code>" << escapeHtml(s.condition) << "</code></p>";
517 os << "<hr>";
518 os << descFragment.toStdString();
519 os << "<p><b>触发行为：</b> " << escapeHtml(s.expectedBehavior) << "</p>";
520 os << "<h4>示例代码：</h4>";
521 os << "<pre style='background:" << TeachingTheme::surface().name().toStdString()
522 << "; padding:8px; font-family:\"Cascadia Code\",\"Cascadia Mono\",\"Consolas\",\"JetBrains Mono\",\"Source "
523 "Code Pro\",\"Menlo\",\"DejaVu Sans Mono\",\"Courier New\",monospace;'>"
524 << escapeHtml(s.sampleCode) << "</pre>";
525 scenarioDetail_->setHtml(QString::fromUtf8(os.str().c_str()));
526 // 注：移除 fadeInWidget —— opacity 卡 0 导致切换后详情区空白
527}
