31// RuntimeLimits.h — 统一运行时限制常量
32// ------------------------------------------------------------
33// 消除 17 个限制常量散布在 7 个文件的重复定义问题。
34// 所有模块（Lexer/Parser/Interpreter/Compiler/VM/Formatter/Value）
35// 共享同一组常量，避免对齐遗漏（如 MAX_INHERITANCE_DEPTH 曾重复
36// 定义在 Interpreter.h 和 VM.h 两处）。
37//
38// 设计原则：
39// 1. 所有常量为 constexpr，可供编译期常量折叠
40// 2. 仅依赖 <cstdint> 和 <cstddef>，无循环依赖风险
41// 3. 常量值与原各自定义保持一致，纯重构无行为变更
42//
43// L7 fix（2026-07-19）：引入 RuntimeConfig 运行时配置类。
44// - 原审计结论（R93 BUG-020/021）认为可配置化与 constexpr 设计原则冲突，
45// 但用户要求修复此已知限制。折中方案：保留 constexpr 作为编译期默认值，
46// 另引入 RuntimeConfig 单例允许 IDE 在运行时覆盖部分 DoS 防护限制。
47// - 仅 MAX_INSTRUCTIONS / MAX_PARSE_ERRORS / MAX_LOOP_ITERATIONS 三个
48// DoS 防护常量支持运行时覆盖（教学场景下可能需要调整）。
49// - 其他常量（递归深度、序列化深度等）保持 constexpr，因为它们关系到
50// 栈空间分配和编译期优化，运行时修改可能导致栈溢出。
51// ============================================================

63constexpr size_t MAX_FRAMES = 256;

65constexpr size_t MAX_STACK_SIZE = 1024;

67// ---- 继承链 ----
68// 类继承链最大深度（Interpreter 和 VM 共享，避免重复定义）
69constexpr int MAX_INHERITANCE_DEPTH = 64;

71// ---- DoS 防护 ----
72// Interpreter 循环迭代次数上限（约 0.3 秒 CPU 时间）
73constexpr int64_t MAX_LOOP_ITERATIONS = 10000000;
74// VM 指令执行总预算（≤1000 万次循环迭代，约 0.5 秒）
75constexpr int64_t MAX_INSTRUCTIONS = 40000000;
76// range() 函数参数上限（与 MAX_LOOP_ITERATIONS 对齐）
77constexpr int64_t MAX_RANGE = 10000000;
78// AUDIT-P1.2 fix: 条件断点求值步数上限。条件断点求值同步阻塞主线程，
79// 需比 MAX_LOOP_ITERATIONS 严格 100 倍。100000 步在简单条件表达式下为毫秒级，
80// 足以覆盖合理条件（如 i > 10 && arr.len() > 5），同时截断 while(true){} 等无限循环。
81constexpr size_t MAX_CONDITION_STEPS = 100000;
82// channel.recvTimeout(ms) 单次等待上限（毫秒）。spawn 为延迟执行模式，
83// recvTimeout 阻塞期间不会有其他线程 send，过大的超时等价于卡死 worker；
84// 60s 上限阻断恶意/误写的超长阻塞，同时覆盖合理的教学演示场景。
85constexpr int64_t MAX_CHANNEL_TIMEOUT_MS = 60000;
86
87// ---- 编译期深度保护 ----
88// Parser 最大递归深度
89// AUDIT-P0 fix: 从 512 降至 256。原值 512 在 Debug 模式下会先栈溢出：
90// 每层括号递归经过 expression→assignment→or_→and_→equality→comparison→
91// term→factor→unary→call→primary 共 11 个栈帧，但只有 expression/assignment
92// 各 +1 depth。512/2=256 层括号 × 11 帧 × ~3KB ≈ 8MB，远超 1MB 默认栈。
93// 256 在 Release 1MB 栈下安全（128 层 × 11 × 500B ≈ 700KB），在 Debug
94// 8MB 栈下也安全（128 层 × 11 × 3KB ≈ 4.2MB）。
95constexpr int MAX_PARSE_DEPTH = 256;
96// Parser 最大块嵌套深度
97constexpr int MAX_BLOCK_DEPTH = 256;
98// BUG-PARSER-AUDIT-5 fix: Parser 错误数量上限。
99// 原实现无上限，恶意输入（如 100 万个 ';'）可触发 O(N) 诊断内存膨胀（~200MB）。
100// 主流编译器（gcc/clang/MSVC）均在错误数超阈值后停止解析。100 与 MSVC 默认一致。
101constexpr int MAX_PARSE_ERRORS = 100;
102// Compiler 最大编译嵌套深度
103constexpr int MAX_COMPILE_DEPTH = 512;
104// Formatter 最大格式化嵌套深度
105constexpr int MAX_FORMAT_DEPTH = 256;
106// 七特性 MVP 阶段 2：宏展开克隆递归深度上限。
107// 宏 body 是单个表达式（受 MAX_PARSE_DEPTH 约束），嵌套宏调用在 body 解析时
108// 已展开为子树，克隆深度≈展开后 AST 深度。与 MAX_PARSE_DEPTH 对齐。
109constexpr int MAX_MACRO_EXPANSION_DEPTH = 256;
110
111// ---- Value 序列化 / 比较 ----
112// toString 递归深度上限（对齐 Formatter MAX_FORMAT_DEPTH）
113constexpr int MAX_TOSTRING_DEPTH = 256;
114// equals 递归深度上限
115constexpr int MAX_EQUALS_DEPTH = 256;
116// clone 递归深度上限（防止深嵌套/环形结构栈溢出）
117constexpr int MAX_CLONE_DEPTH = 256;
118
119// ---- Lexer 输入限制 ----
120// 源代码最大大小（10MB）
121constexpr size_t MAX_SOURCE_SIZE = 10 * 1024 * 1024;
122// 最大 Token 数量（100万）
123constexpr size_t MAX_TOKEN_COUNT = 1000000;
124// 字符串插值最大嵌套深度
125constexpr int MAX_INTERP_DEPTH = 64;
126
127// ---- GUI 查找/替换限制 ----
128// D14 fix: 统一 GUI 查找/替换上限到 RuntimeLimits，避免硬编码散布
129// 高亮匹配上限（防止超大文档 UI 卡死）
130constexpr int MAX_FIND_HIGHLIGHTS = 1000;
131// 全部替换上限（防止超大文档 UI 卡死）
132constexpr int MAX_REPLACE_ALL = 100000;
133
134// ---- JIT 调用参数上限 ----
135// P3-15 fix: JIT 运行时辅助函数（jitCallExpr/jitCallMethod/jitCallByName）从
136// 操作数栈 pop 参数到本地缓冲区，argCount 编码为 uint8_t（上限 255），
137// 256 是覆盖全部合法 argCount 的最小 2 的幂。原 JIT.cpp 4 处硬编码 Value args[256]，
138// 现统一为 std::array<Value, MAX_JIT_ARGS>，便于静态边界检查与 clang-tidy 审计。
139constexpr size_t MAX_JIT_ARGS = 256;
140// P3-15 fix: JIT 默认参数填充缓冲区上限。默认参数是参数的子集，
141// 64 足够覆盖实际场景（函数参数总数上限 MAX_JIT_ARGS=256，但默认参数数量
142// 远小于此）。原 JIT.cpp 4 处硬编码 Value defaults[64]，现统一为
143// std::array<Value, MAX_DEFAULT_PARAMS>，便于静态边界检查。
144constexpr size_t MAX_DEFAULT_PARAMS = 64;

146// ---- 字节码哨兵值（"无"标记）----
147// P3-14 fix: 提取散布在 Compiler/VM/Bytecode/IR/JIT/RegisterVM 共 40+ 处的
148// 魔法数字哨兵。0xFFFF 表示 uint16 槽位/常量索引"无"，0xFF 表示 uint8

153// ============================================================
154// L7 fix: RuntimeConfig — 运行时可配置的 DoS 防护限制
155// ------------------------------------------------------------
156// 允许 IDE 在运行时覆盖部分 DoS 防护常量（教学场景下可能需要调整）。
157// 设计要点：
158// 1. 单例模式，线程安全（std::atomic 存储）
159// 2. 默认值与上方 constexpr 保持一致
160// 3. 仅暴露三个 DoS 防护常量，其他常量保持 constexpr 不变
161// 4. 调用方应使用 RuntimeConfig::maxInstructions() 等方法，
162// 而非直接引用 RuntimeLimits::MAX_INSTRUCTIONS
163// 5. 测试场景可通过 setter 临时调小验证触发逻辑
164// ============================================================
165class RuntimeConfig {
166public:
167 static RuntimeConfig& instance() {
168 static RuntimeConfig cfg;
169 return cfg;
170 }
171
172 // DoS 防护限制（运行时可配置）
173 int64_t maxInstructions() const { return maxInstructions_.load(std::memory_order_relaxed); }
174 int maxParseErrors() const { return maxParseErrors_.load(std::memory_order_relaxed); }
175 int64_t maxLoopIterations() const { return maxLoopIterations_.load(std::memory_order_relaxed); }
176
177 void setMaxInstructions(int64_t v) { maxInstructions_.store(v, std::memory_order_relaxed); }
178 void setMaxParseErrors(int v) { maxParseErrors_.store(v, std::memory_order_relaxed); }
179 void setMaxLoopIterations(int64_t v) { maxLoopIterations_.store(v, std::memory_order_relaxed); }
180
181 // 重置为编译期默认值（测试 tearDown 调用避免污染后续测试）
182 void resetToDefaults() {
183 maxInstructions_.store(MAX_INSTRUCTIONS, std::memory_order_relaxed);
184 maxParseErrors_.store(MAX_PARSE_ERRORS, std::memory_order_relaxed);
185 maxLoopIterations_.store(MAX_LOOP_ITERATIONS, std::memory_order_relaxed);
186 // 七特性 MVP 阶段 6：沙箱能力开关重置为默认（全放行）
187 sandboxEnabled_.store(false, std::memory_order_relaxed);
188 sandboxAllowInput_.store(true, std::memory_order_relaxed);
189 sandboxAllowImport_.store(true, std::memory_order_relaxed);
190 }
191
192 // ============================================================
193 // 七特性 MVP 阶段 6：能力级沙箱开关
194 // ------------------------------------------------------------
195 // 与 DoS 限制正交：DoS 限制防资源耗尽，沙箱开关控制能力（IO/import）。
196 // 默认全放行（sandboxEnabled=false）；启用后按子开关拦截。
197 // 拦截点：input（共享层 executeSharedInput）、import（三后端 import 路径
198 // 仅放行 std/ 内建模块）、插件加载（C API）。
199 // atomic 存储保证线程安全（GUI 主线程设置 vs Worker 线程读取）。
200 // ============================================================
201 bool sandboxEnabled() const { return sandboxEnabled_.load(std::memory_order_relaxed); }
202 bool sandboxAllowInput() const { return sandboxAllowInput_.load(std::memory_order_relaxed); }
203 bool sandboxAllowImport() const { return sandboxAllowImport_.load(std::memory_order_relaxed); }
204
205 void setSandboxEnabled(bool v) { sandboxEnabled_.store(v, std::memory_order_relaxed); }
206 void setSandboxAllowInput(bool v) { sandboxAllowInput_.store(v, std::memory_order_relaxed); }
207 void setSandboxAllowImport(bool v) { sandboxAllowImport_.store(v, std::memory_order_relaxed); }
208
209 /// 便捷查询：沙箱启用且 input 被禁。
210 bool sandboxBlocksInput() const { return sandboxEnabled() && !sandboxAllowInput(); }
211 /// 便捷查询：沙箱启用且文件 import 被禁（std/ 内建模块仍放行，由调用方判定）。
212 bool sandboxBlocksImport() const { return sandboxEnabled() && !sandboxAllowImport(); }
213
214private:
215 RuntimeConfig()
216 : maxInstructions_(MAX_INSTRUCTIONS), maxParseErrors_(MAX_PARSE_ERRORS),
217 maxLoopIterations_(MAX_LOOP_ITERATIONS), sandboxEnabled_(false), sandboxAllowInput_(true),
218 sandboxAllowImport_(true) {}
219 RuntimeConfig(const RuntimeConfig&) = delete;
220 RuntimeConfig& operator=(const RuntimeConfig&) = delete;
221
222 std::atomic<int64_t> maxInstructions_;
223 std::atomic<int> maxParseErrors_;
224 std::atomic<int64_t> maxLoopIterations_;
225 // 七特性 MVP 阶段 6：沙箱能力开关
226 std::atomic<bool> sandboxEnabled_;
227 std::atomic<bool> sandboxAllowInput_;
228 std::atomic<bool> sandboxAllowImport_;
229};
