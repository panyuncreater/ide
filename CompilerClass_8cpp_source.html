20void Compiler::visitClassDecl(ClassDecl& node) {
21 // ── 类编译总策略 ───────────────────────────────────────────────────
22 // 类在运行时是一个闭包值（携带方法表）。编译分三步：
23 // 1) 字段布局：先收集父类字段（classFieldNames_ 中查 superClassName），再追加
24 // 子类自有字段，得到 [父类字段..., 子类字段...] 完整顺序——这是运行时的实例
25 // 槽布局，决定 OP_GET_FIELD/OP_SET_FIELD 的索引含义（见 VMContainers.cpp）。
26 // 2) 发射 OP_CLASS_NEW <fieldCount> 创建空实例，随后对每个实例字段补发
27 // OP_INIT_FIELD <fieldIdx> 写入由字段初始化表达式求值得到的默认值。
28 // 3) 方法编译：每个方法作为嵌套 FunDecl 编译为独立 chunk，并以闭包形式挂到实例
29 // 方法表（供 OP_METHOD_CALL 通过 this 派发），同时记录到 classFieldNames_
30 // 供其子类继承字段顺序。
31 // 类声明本身 emit 一个"类对象值"到栈顶，由调用方（声明语句）消费或 POP。
32 //
33 // 拆分说明（原 221 行单函数 → orchestrator + 2 子阶段）：
34 // - compileClassMembers：方法编译循环，每迭代创建 CompileContextGuard 并调用
35 // emitMethodBody；currentClassName_ 在循环内手动保存/恢复
36 // - emitMethodBody：单个方法体编译（chunk 设置 + this/字段/参数槽位 + body +
37 // 元数据保存 + 默认参数值 emit）
38 // 不变量保留：R97 #9 fix CompileContextGuard、B1 fix currentClassName_ 保存/恢复、
39 // C-P2-3/4 fix slot>255 early return、BUG 6a fix 默认参数折叠（由 emitDefaultValues 统一）。
40
41 // 类声明：发射 OP_CLASS_NEW + OP_INIT_FIELD 初始化字段 + 编译方法
42 uint16_t nameIdx = identifierIndex(node.name);
43
44 // 收集当前类的自有字段名
45 std::vector<std::string> ownFieldNames;
46 for (auto& member : node.members) {
47 if (member->nodeType == NodeType::NODE_VAR_DECL) {
48 ownFieldNames.push_back(static_cast<VarDecl*>(member.get())->name);
49 }
50 }
51
52 // 构建含继承字段的完整字段列表（父类字段在前，子类字段在后）
53 std::vector<std::string> allFieldNames;
54 if (!node.superClassName.empty()) {
55 auto it = classFieldNames_.find(node.superClassName);
56 if (it != classFieldNames_.end()) {
57 allFieldNames = it->second; // 父类字段在前
58 } else {
59 // C-P2-9 fix: 父类未找到时报错，而非静默丢失继承字段（导致 slot 布局错误）
60 error("类 '" + node.name + "' 的父类 '" + node.superClassName +
61 "' 未定义（不支持前向引用，请确保父类在子类之前声明）",
62 node.line, node.column);
63 }
64 }
65 for (const auto& fn : ownFieldNames) {
66 if (std::find(allFieldNames.begin(), allFieldNames.end(), fn) == allFieldNames.end()) {
67 allFieldNames.push_back(fn); // 子类字段在后（跳过覆盖的同名字段）
68 }
69 }
70
71 // 注册类字段名（供子类编译时查找）
72 classFieldNames_[node.name] = allFieldNames;
73
74 // 先编译所有方法为独立 chunk
75 compileClassMembers(node, allFieldNames);
76
77 // 主 chunk 中：发射 OP_CLASS_NEW（0 参数构造，字段由 OP_INIT_FIELD 设置）
78 chunk_.writeOp(OpCode::OP_CLASS_NEW, node.line);
79 chunk_.writeShort(nameIdx, node.line);
80 chunk_.write(static_cast<uint8_t>(0), node.line); // argCount = 0（字段由 OP_INIT_FIELD 初始化）
81
82 // 此时栈顶是刚创建的空实例，发射 OP_INIT_FIELD 设置每个字段的默认值
83 for (auto& member : node.members) {
84 if (member->nodeType != NodeType::NODE_VAR_DECL)
85 continue;
86 VarDecl* varDecl = static_cast<VarDecl*>(member.get());
87 // 编译字段默认值表达式
88 if (varDecl->initializer) {
89 compileNode(varDecl->initializer.get());
90 } else {
91 chunk_.writeOp(OpCode::OP_NULL, varDecl->line);
92 }
93 // 发射 OP_INIT_FIELD：pop 默认值，设置到栈顶实例的字段中
94 uint16_t fieldIdx = identifierIndex(varDecl->name);
95 chunk_.writeOp(OpCode::OP_INIT_FIELD, varDecl->line);
96 chunk_.writeShort(fieldIdx, varDecl->line);
97 }
98
99 // 将类注册为全局变量（OP_DEFINE_CLASS 从栈上 pop 模板实例并注册类信息）
100 // 操作数: nameIdx(2B) + superNameIdx(2B)
101 // superNameIdx == NO_INDEX 表示无父类；否则为父类名在常量池中的索引
102 chunk_.writeOp(OpCode::OP_DEFINE_CLASS, node.line);
103 chunk_.writeShort(nameIdx, node.line);
104 if (node.superClassName.empty()) {
105 chunk_.writeShort(RuntimeLimits::NO_INDEX, node.line); // 无父类标记
106 } else {
107 uint16_t superIdx = identifierIndex(node.superClassName);
108 chunk_.writeShort(superIdx, node.line);
109 }
110
111 // 栈上的模板实例已被 OP_DEFINE_CLASS 消费，无需额外 OP_POP
112 return;
113}

120 for (auto& member : node.members) {
121 if (member->nodeType != NodeType::NODE_FUN_DECL)
122 continue;
123 FunDecl* funDecl = static_cast<FunDecl*>(member.get());
124 // R97 #9 fix: 使用 CompileContextGuard RAII 自动保存/恢复 15 个编译上下文成员变量
125 // （含异常路径），替代原 15 行 std::move 保存 + 30 行手动恢复（正常路径 + slot 越界错误路径）。
126 // currentClassName_ 不在 CompileContext 中（仅 visitClassDecl 使用），单独手动保存。

134 // 故放在 emitMethodBody 返回之后统一执行（原实现分别在正常路径与错误路径恢复）。

238
239 functionChunks_[methodKey] = std::move(chunk_);

297 uint16_t fieldIdx = identifierIndex(node.fieldName);

392 }
393 // 检查接收者是否为简单变量（VarRef），用于 writeBack
394 uint16_t receiverVarIdx = RuntimeLimits::NO_INDEX; // NO_INDEX = 无全局变量 writeBack
395 uint8_t receiverLocalSlot = RuntimeLimits::NO_SLOT; // NO_SLOT = 无局部变量 writeBack
396 VarRef* objVar = (node.object && node.object->nodeType == NodeType::NODE_VAR_REF)
397 ? static_cast<VarRef*>(node.object.get())
398 : nullptr;
399 if (objVar) {
400 auto localIt = currentLocals_.find(objVar->name);
401 if (localIt == currentLocals_.end()) {
402 // 全局变量：记录变量名索引供 VM writeBack 使用
403 receiverVarIdx = identifierIndex(objVar->name);
404 } else {
405 // 局部变量：记录 slot 号供 VM writeBack 使用
406 receiverLocalSlot = static_cast<uint8_t>(localIt->second);
407 }
408 }
409
410 // H4 fix: super.method() 的接收者是 this（始终在 slot 0），设置写回目标
411 if (!objVar && node.object && node.object->nodeType == NodeType::NODE_SUPER_EXPR && inFunction_) {
412 receiverLocalSlot = 0; // slot 0 = this
413 }
414
415 // B6 fix: 如果接收者是 IndexAccess，预缓存索引值避免写回时重复求值
416 // （对 arr[expr()].method() 这类调用，expr() 只执行一次）
417 std::string cachedIndexVar;
418 if (!objVar && node.object && node.object->nodeType == NodeType::NODE_INDEX_ACCESS) {
419 auto* ia = static_cast<IndexAccess*>(node.object.get());
420 if (ia->object && ia->object->nodeType == NodeType::NODE_VAR_REF && ia->index) {
421 cachedIndexVar = "__wb_idx_" + std::to_string(writebackCounter_++);
422 compileNode(ia->index.get());
423 uint16_t cacheIdx = identifierIndex(cachedIndexVar);
424 chunk_.writeOp(OpCode::OP_DEFINE_VAR, node.line); // #9 fix: DEFINE 而非 SET（首次创建变量）
425 chunk_.writeShort(cacheIdx, node.line);
426 }
427 }
428
429 // B6 fix: 如果索引已缓存，手动内联 IndexAccess 编译（用缓存值代替重新求值）
430 if (!cachedIndexVar.empty()) {
431 auto* ia = static_cast<IndexAccess*>(node.object.get());
432 compileNode(ia->object.get()); // push base (e.g., arr)
433 uint16_t cacheIdx = identifierIndex(cachedIndexVar);
434 chunk_.writeOp(OpCode::OP_GET_VAR, node.line);
435 chunk_.writeShort(cacheIdx, node.line); // push cached index
436 chunk_.writeOp(OpCode::OP_INDEX_GET, node.line); // base[cachedIndex]
437 } else {
438 compileNode(node.object.get());
439 }
440 // C-P2-1 fix: 方法调用参数数量检查（已前移到函数开头，C-P2-11 fix 避免缓存变量泄漏）
441 for (auto& arg : node.arguments) {
442 compileNode(arg.get());
443 }
444 uint16_t nameIdx = identifierIndex(node.methodName);
445 // O1: super.method() 使用 OP_SUPER_CALL（从父类开始方法查找）
446 bool isSuperCall = (node.object && node.object->nodeType == NodeType::NODE_SUPER_EXPR);
447 chunk_.writeOp(isSuperCall ? OpCode::OP_SUPER_CALL : OpCode::OP_METHOD_CALL, node.line);
448 chunk_.writeShort(nameIdx, node.line);
449 chunk_.write(static_cast<uint8_t>(node.arguments.size()), node.line);
450 chunk_.writeShort(receiverVarIdx, node.line); // 接收者全局变量名索引（NO_INDEX = 无全局 writeBack）
451 chunk_.write(receiverLocalSlot, node.line); // 接收者局部变量 slot（NO_SLOT = 无局部 writeBack）
452 if (isSuperCall) {
453 // B1 fix: 编码当前类名索引，VM 用它查找父类（而非运行时实例类名）
454 uint16_t classIdx = identifierIndex(currentClassName_);
455 chunk_.writeShort(classIdx, node.line);
456 }
457
458 // 嵌套访问变异方法写回：当接收者是 MemberAccess/IndexAccess 且基对象是 VarRef 时，
459 // 方法调用后发射写回指令，确保 this.arr.push(42) 等嵌套调用的修改不丢失。
460 // VM 在变异方法调用时将修改后的对象暂存到 lastMutatedReceiver_，
461 // 写回指令从中取值写回基对象的字段/索引位置。
462 // 若接收者是 VarRef（objVar != nullptr），helper 内部 MemberAccess/IndexAccess
463 // 类型检查均不匹配，等价于 no-op（原 if (!objVar && node.object) 守卫的等价简化）。
464 emitMethodCallWriteback(node.object.get(), node.line, cachedIndexVar);
465
466 // #9 fix: 清理 __wb_idx_ 缓存变量，防止永久泄漏到 globals_
467 if (!cachedIndexVar.empty()) {
468 uint16_t cacheIdx = identifierIndex(cachedIndexVar);
469 chunk_.writeOp(OpCode::OP_DELETE_VAR, node.line);
470 chunk_.writeShort(cacheIdx, node.line);
471 }
472 return;
473}

475void Compiler::emitMethodCallWriteback(const ASTNode* receiver, int line, const std::string& cachedIndexVar) {
476 // visitMethodCall 子阶段：嵌套访问变异方法写回
477 // 当接收者是 MemberAccess/IndexAccess 且基对象是 VarRef 时，发射写回指令，
478 // 确保 this.arr.push(42) 等嵌套调用的修改不丢失。
479 //
480 // W3-2-Bug1 fix: 对齐 IR 路径 emitNestedAssignWriteback 的 3 步序列。
481 // 原 bug: 直接发射 OP_WRITEBACK_MEMBER_*，但 lastMutatedReceiver_ 是方法接收者
482 // （变异后的成员，如数组），而非整个基容器（如 Box 实例）。WRITEBACK 的"整体替换"
483 // 语义会把基容器替换为成员，导致后续访问报"该类型不支持成员访问"。
484 // 修复: 先加载基变量 + LOAD_MUTATED + MEMBER_SET/INDEX_SET 在基容器上设置字段/索引，
485 // 产生新的变异基容器（OP_MEMBER_SET/OP_INDEX_SET 会更新 lastMutatedReceiver_ 为
486 // 整个变异后的基容器），再 WRITEBACK 整体替换变量/upvalue。
487 //
488 // 若 receiver 为 nullptr / VarRef / SuperExpr 等非 MemberAccess/IndexAccess 类型，
489 // 内部类型检查均不匹配，函数为 no-op。
490 if (!receiver)
491 return;
492 if (receiver->nodeType == NodeType::NODE_MEMBER_ACCESS) {
493 // MemberAccess 嵌套写回：如 this.arr.push(42)、obj.field.pop()
494 auto* ma = static_cast<const MemberAccess*>(receiver);
495 if (ma->object && ma->object->nodeType == NodeType::NODE_VAR_REF) {
496 auto* baseVar = static_cast<VarRef*>(ma->object.get());
497 uint16_t fieldIdx = identifierIndex(ma->fieldName);
498
499 // 步骤 1-3: 加载基变量 + LOAD_MUTATED + MEMBER_SET
500 // 栈序: [..., base, mutated_val] → MEMBER_SET → [...]
501 // MEMBER_SET 弹出 val 和 obj，设置 obj.field=val，
502 // 并将变异后的整个 obj 存入 lastMutatedReceiver_。
503 emitLoadVariable(baseVar->name, line); // push base (e.g., Box)
504 chunk_.writeOp(OpCode::OP_LOAD_MUTATED, line); // push mutated member (e.g., array)
505 chunk_.writeOp(OpCode::OP_MEMBER_SET, line); // base.field = mutated; lastMutatedReceiver_ = new base
506 chunk_.writeShort(fieldIdx, line);
507
508 // 步骤 4: WRITEBACK 整体替换变量/upvalue 为 lastMutatedReceiver_（新的变异基容器）
509 auto localIt = currentLocals_.find(baseVar->name);
510 if (localIt != currentLocals_.end()) {
511 // 局部变量成员写回：OP_WRITEBACK_MEMBER_LOCAL(slot, fieldIdx)
512 chunk_.writeOp(OpCode::OP_WRITEBACK_MEMBER_LOCAL, line);
513 chunk_.write(static_cast<uint8_t>(localIt->second), line);
514 chunk_.writeShort(fieldIdx, line);
515 } else if (inFunction_) {
516 // upvalue 接收者的成员写回（闭包内 b.data.push(4)）
517 int uvIdx = resolveUpvalue(baseVar->name, line);
518 if (uvIdx >= 0) {
519 chunk_.writeOp(OpCode::OP_WRITEBACK_MEMBER_UPVALUE, line);
520 chunk_.write(static_cast<uint8_t>(uvIdx), line);
521 chunk_.writeShort(fieldIdx, line);
522 } else {
523 // 全局变量成员写回：OP_WRITEBACK_MEMBER_VAR(varIdx, fieldIdx)
524 chunk_.writeOp(OpCode::OP_WRITEBACK_MEMBER_VAR, line);
525 chunk_.writeShort(identifierIndex(baseVar->name), line);
526 chunk_.writeShort(fieldIdx, line);
527 }
528 } else {
529 // 全局变量成员写回：OP_WRITEBACK_MEMBER_VAR(varIdx, fieldIdx)
530 chunk_.writeOp(OpCode::OP_WRITEBACK_MEMBER_VAR, line);
531 chunk_.writeShort(identifierIndex(baseVar->name), line);
532 chunk_.writeShort(fieldIdx, line);
533 }
534 }
535 } else if (receiver->nodeType == NodeType::NODE_INDEX_ACCESS) {
536 // IndexAccess 嵌套写回：如 arr[0].push(42)、dict["key"].remove("x")
537 // B6 fix: 使用预缓存的索引值，避免重复求值（对 arr[expr()].method() 防止副作用执行两次）
538 auto* ia = static_cast<const IndexAccess*>(receiver);
539 if (ia->object && ia->object->nodeType == NodeType::NODE_VAR_REF) {
540 auto* baseVar = static_cast<VarRef*>(ia->object.get());
541 if (cachedIndexVar.empty())
542 return; // 无缓存索引，无法写回（不应发生，visitMethodCall 已保证缓存）
543
544 // 步骤 1-4: 加载基变量 + 加载缓存索引 + LOAD_MUTATED + INDEX_SET
545 // 栈序: [..., base, idx, mutated_val] → INDEX_SET → [...]
546 // INDEX_SET 弹出 val, idx, obj，设置 obj[idx]=val，
547 // 并将变异后的整个 obj 存入 lastMutatedReceiver_。
548 emitLoadVariable(baseVar->name, line); // push base (e.g., arr)
549 uint16_t cacheIdx = identifierIndex(cachedIndexVar);
550 chunk_.writeOp(OpCode::OP_GET_VAR, line); // push cached index
551 chunk_.writeShort(cacheIdx, line);
552 chunk_.writeOp(OpCode::OP_LOAD_MUTATED, line); // push mutated element
553 chunk_.writeOp(OpCode::OP_INDEX_SET, line); // base[idx] = mutated; lastMutatedReceiver_ = new base
554
555 // 步骤 5: WRITEBACK 整体替换变量/upvalue 为 lastMutatedReceiver_（新的变异基容器）
556 auto localIt = currentLocals_.find(baseVar->name);
557 if (localIt != currentLocals_.end()) {
558 // 局部变量索引写回：OP_WRITEBACK_INDEX_LOCAL(slot)
559 chunk_.writeOp(OpCode::OP_WRITEBACK_INDEX_LOCAL, line);
560 chunk_.write(static_cast<uint8_t>(localIt->second), line);
561 } else if (inFunction_) {
562 // upvalue 接收者的索引写回（闭包内 arr[i].push(42)）
563 int uvIdx = resolveUpvalue(baseVar->name, line);
564 if (uvIdx >= 0) {
565 chunk_.writeOp(OpCode::OP_WRITEBACK_INDEX_UPVALUE, line);
566 chunk_.write(static_cast<uint8_t>(uvIdx), line);
567 } else {
568 // 全局变量索引写回：OP_WRITEBACK_INDEX_VAR(varIdx)
569 chunk_.writeOp(OpCode::OP_WRITEBACK_INDEX_VAR, line);
570 chunk_.writeShort(identifierIndex(baseVar->name), line);
571 }
572 } else {
573 // 全局变量索引写回：OP_WRITEBACK_INDEX_VAR(varIdx)
574 chunk_.writeOp(OpCode::OP_WRITEBACK_INDEX_VAR, line);
575 chunk_.writeShort(identifierIndex(baseVar->name), line);
576 }
577 }
578 }
579}

581void Compiler::emitLoadVariable(const std::string& name, int line) {
582 // 发射变量加载指令，逻辑与 visitVarRef 一致。
583 // 优先级: local → upvalue → global slot → global name。
584 if (inFunction_) {
585 auto it = currentLocals_.find(name);
586 if (it != currentLocals_.end()) {
587 chunk_.writeOp(OpCode::OP_GET_LOCAL, line);
588 chunk_.write(static_cast<uint8_t>(it->second), line);
589 return;
590 }
591 int uvIdx = resolveUpvalue(name, line);
592 if (uvIdx >= 0) {
593 chunk_.writeOp(OpCode::OP_GET_UPVALUE, line);
594 chunk_.write(static_cast<uint8_t>(uvIdx), line);
595 return;
596 }
597 }
598 int slot = lookupGlobalSlot(name);
599 if (slot >= 0) {
600 chunk_.writeOp(OpCode::OP_GET_GLOBAL, line);
601 chunk_.writeShort(static_cast<uint16_t>(slot), line);
602 } else {
603 uint16_t nameIdx = identifierIndex(name);
604 chunk_.writeOp(OpCode::OP_GET_VAR, line);
605 chunk_.writeShort(nameIdx, line);
606 }

614void Compiler::visitSuperExpr(SuperExpr& node) {
615 // 注：super 在非方法上下文中的错误为运行时错误（非编译期）。
616 // 直接 Compiler 路径 currentClassName_ 在 visitClassDecl 中设置，方法编译完后恢复。
617 // 嵌套函数（FunDecl）内的 super 调用是合法的——visitFunDecl 入口保存 currentClassName_
618 // 后不会清除，方法体内嵌套函数仍能继承外层方法名作为 super 上下文。
619 // 顶层或非方法体内访问 super 时 currentClassName_ 为空。
620 //
621 // BUG-INH-AUDIT-6 fix: 原实现无条件 emit OP_GET_LOCAL 0，在顶层运行时
622 // 报"内部错误: 局部变量槽越界 (slot 0)"（误导为 VM bug）。现改为：
623 // - 方法上下文（currentClassName_ 非空）：emit OP_GET_LOCAL 0（this 槽）
624 // - 非方法上下文（currentClassName_ 为空）：emit OP_GET_VAR "this"
625 // VM 在 GLOBAL_NAME 查找失败时报 "未定义的变量: this"，与 IR 路径对齐。
626 // 此运行时错误不可被 try/catch 捕获（在 try 块进入前即触发）。
627 // 顶层 / 普通函数内 super 的运行时错误行为由 ConsistencyDiff.H7b 与
628 // AuditSuper_RuntimeErrorNotCatchableByTryCatch 测试覆盖。
629 // 注：Interpreter 报 "super 只能在类方法中使用"（更友好），VM 路径报
630 // "未定义的变量: this"——此 2-way 差异文档化为已知，由 H7b 测试 EXPECT_NE 覆盖。
631 if (currentClassName_.empty()) {
632 // 非方法上下文：emit OP_GET_VAR "this" → 运行时报 "未定义的变量: this"
633 uint16_t nameIdx = chunk_.addConstant(Value(std::string("this")));
634 chunk_.writeOp(OpCode::OP_GET_VAR, node.line);
635 chunk_.write(static_cast<uint8_t>(nameIdx & 0xFF), node.line);
636 chunk_.write(static_cast<uint8_t>((nameIdx >> 8) & 0xFF), node.line);
637 } else {
638 // 方法上下文：emit OP_GET_LOCAL 0（this 槽）
639 chunk_.writeOp(OpCode::OP_GET_LOCAL, node.line);
640 chunk_.write(0, node.line); // slot 0 = this
641 }
642 return;
643}

671bool Compiler::tryFoldBinary(BinOpType opType, ASTNode* left, ASTNode* right, Value& result, int line) {
672 // D8 fix: 使用 extractConstant 递归提取常量值，支持嵌套常量表达式（如 (1+2)*3）
673 Value lv, rv;
674 if (!extractConstant(left, lv, line) || !extractConstant(right, rv, line)) {
675 return false;
676 }
677
678 // 数值运算
679 if (lv.isNumber() && rv.isNumber()) {
680 // 类型提升：int+float → float
681 bool useFloat = lv.isFloat() || rv.isFloat();
682 double ld = useFloat ? (lv.isFloat() ? lv.floatVal() : static_cast<double>(lv.intVal())) : 0;
683 double rd = useFloat ? (rv.isFloat() ? rv.floatVal() : static_cast<double>(rv.intVal())) : 0;
684 int64_t li = lv.isInt() ? lv.intVal() : 0;
685 int64_t ri = rv.isInt() ? rv.intVal() : 0;
686
687 switch (opType) {
688 case BinOpType::BIN_ADD:
689 if (!useFloat) {
690 // B6 fix: 统一使用 OverflowCheck
691 if (OverflowCheck::addOverflow(li, ri))
692 return false;
693 result = Value(li + ri);
694 } else {
695 result = Value(ld + rd);
696 }
697 return true;
698 case BinOpType::BIN_SUB:
699 if (!useFloat) {
700 if (OverflowCheck::subOverflow(li, ri))
701 return false;
702 result = Value(li - ri);
703 } else {
704 result = Value(ld - rd);
705 }
706 return true;
707 case BinOpType::BIN_MUL:
708 if (!useFloat) {
709 if (OverflowCheck::mulOverflow(li, ri))
710 return false;
711 result = Value(li * ri);
712 } else {
713 result = Value(ld * rd);
714 }
715 return true;
716 case BinOpType::BIN_DIV: {
717 double divisor = useFloat ? rd : static_cast<double>(ri);
718 if (divisor == 0)
719 return false; // 除零不折叠，保留运行时错误
720 // B3 fix: INT64_MIN / -1 = 溢出 UB，不折叠
721 if (!useFloat && OverflowCheck::divOverflow(li, ri))
722 return false;
723 result = useFloat ? Value(ld / rd) : Value(li / ri);
724 return true;
725 }
726 case BinOpType::BIN_MOD:
727 if (!useFloat && ri == 0)
728 return false;
729 // B3 fix: INT64_MIN % -1 = 溢出 UB，不折叠
730 if (!useFloat && OverflowCheck::modOverflow(li, ri))
731 return false;
732 if (useFloat)
733 return false;
734 result = Value(li % ri);
735 return true;
736 case BinOpType::BIN_EQ:
737 result = Value(useFloat ? (ld == rd) : (li == ri));
738 return true;
739 case BinOpType::BIN_NEQ:
740 result = Value(useFloat ? (ld != rd) : (li != ri));
741 return true;
742 case BinOpType::BIN_LT:
743 result = Value(useFloat ? (ld < rd) : (li < ri));
744 return true;
745 case BinOpType::BIN_GT:
746 result = Value(useFloat ? (ld > rd) : (li > ri));
747 return true;
748 case BinOpType::BIN_LTE:
749 result = Value(useFloat ? (ld <= rd) : (li <= ri));
750 return true;
751 case BinOpType::BIN_GTE:
752 result = Value(useFloat ? (ld >= rd) : (li >= ri));
753 return true;
754 default:
755 break;
756 }
757 }
758
759 // 字符串拼接
760 if (lv.isString() && rv.isString() && opType == BinOpType::BIN_ADD) {
761 result = Value(lv.stringVal() + rv.stringVal());
762 return true;
763 }
764
765 // 字符串比较
766 if (lv.isString() && rv.isString()) {
767 // PERF-30 fix: 补全字符串字典序比较折叠（<, >, <=, >=）
768 // 原 BIN_EQ/BIN_NEQ 已支持，此处补齐 4 个关系运算符
769 const std::string& ls = lv.stringVal();
770 const std::string& rs = rv.stringVal();
771 switch (opType) {
772 case BinOpType::BIN_EQ:
773 result = Value(ls == rs);
774 return true;
775 case BinOpType::BIN_NEQ:
776 result = Value(ls != rs);
777 return true;
778 case BinOpType::BIN_LT:
779 result = Value(ls < rs);
780 return true;
781 case BinOpType::BIN_GT:
782 result = Value(ls > rs);
783 return true;
784 case BinOpType::BIN_LTE:
785 result = Value(ls <= rs);
786 return true;
787 case BinOpType::BIN_GTE:
788 result = Value(ls >= rs);
789 return true;
790 default:
791 break;
792 }
793 }
794
795 // 布尔逻辑（M1 fix: 短路语义，返回操作数原始值而非 bool）
796 if (lv.isBool() && rv.isBool()) {
797 if (opType == BinOpType::BIN_AND) {
798 result = lv.isTruthy() ? rv : lv;
799 return true;
800 }
801 if (opType == BinOpType::BIN_OR) {
802 result = lv.isTruthy() ? lv : rv;
803 return true;
804 }
805 if (opType == BinOpType::BIN_EQ) {
806 result = Value(lv.boolVal() == rv.boolVal());
807 return true;
808 }
809 if (opType == BinOpType::BIN_NEQ) {
810 result = Value(lv.boolVal() != rv.boolVal());
811 return true;
812 }
813 }
814
815 return false;
816}

818bool Compiler::tryFoldUnary(UnaryOp::UnaryOpType opType, ASTNode* operand, Value& result, int line) {
819 if (!operand)
820 return false;
821
822 // D8 fix: 使用 extractConstant 递归提取常量值，支持嵌套表达式（如 -(3+2)）
823 Value val;
824 if (!extractConstant(operand, val, line))
825 return false;
826
827 if (opType == UnaryOp::UnaryOpType::UOP_NEGATE) {
828 if (val.isInt()) {
829 // B3 fix: -INT64_MIN = 溢出 UB，不折叠
830 if (OverflowCheck::negateOverflow(val.intVal()))
831 return false;
832 result = Value(-val.intVal());
833 return true;
834 }
835 if (val.isFloat()) {
836 result = Value(-val.floatVal());
837 return true;
838 }
839 }
840 if (opType == UnaryOp::UnaryOpType::UOP_NOT && val.isBool()) {
841 result = Value(!val.boolVal());
842 return true;
843 }
844
845 return false;
846}
847
848// D8 fix: 递归提取常量值。支持字面量 + 嵌套 BinaryOp/UnaryOp 常量表达式，
849// 使 tryFoldBinary/tryFoldUnary 能折叠 (1+2)*3、-(-(5)) 等嵌套表达式。
850bool Compiler::extractConstant(ASTNode* node, Value& result, int line) {
851 if (!node)
852 return false;
853
854 switch (node->nodeType) {
855 case NodeType::NODE_NUMBER_LITERAL:
856 result = static_cast<NumberLiteral*>(node)->getValue(); // A1 fix: getValue()
857 return true;
858 case NodeType::NODE_STRING_LITERAL:
859 result = static_cast<StringLiteral*>(node)->getValue(); // A1 fix: getValue()
860 return true;
861 case NodeType::NODE_BOOL_LITERAL:
862 result = static_cast<BoolLiteral*>(node)->getValue(); // A1 fix: getValue()
863 return true;
864 case NodeType::NODE_NULL_LITERAL:
865 result = Value::nullValue();
866 return true;
867 case NodeType::NODE_BINARY_OP: {
868 auto* bin = static_cast<BinaryOp*>(node);
869 return tryFoldBinary(bin->opType, bin->left.get(), bin->right.get(), result, line);
870 }
