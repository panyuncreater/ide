64// P0-REGALLOC fix: 预扫描所有指令，记录每个 vreg 的最后使用 instrIndex。
65// IR 的 vreg 是 SSA-like 的（allocVReg 单调递增不复用），因此生命周期从首次定义到
66// 最后使用。在最后使用点释放寄存器是安全的——后续不会有指令再读取该 vreg。
67//
68// 关键例外：MEMBER_SET/INDEX_SET 的 obj vreg（操作数[0]）在后续 WRITEBACK_* 指令中
69// 被"隐式使用"——RegisterVM 的 MEMBER_SET/INDEX_SET 将 objReg 记录到
70// lastMutatedReceiverReg_，后续 WRITEBACK_* 通过 reg(lastMutatedReceiverReg_) 读取
71// 变异后的容器。如果 obj vreg 在 MEMBER_SET 后被释放并复用，WRITEBACK 会读到错误值。
72// 因此 obj vreg 的最后使用点必须延长到下一个 WRITEBACK_* 指令之后。
73void RegisterBytecodeBackend::collectVRegLastUse(const IRFunction& ir) {
74 vregLastUse_.clear();
75 lastUseToVregs_.clear();
76 size_t globalIdx = 0;
77 // 待延长的 obj vreg（来自 MEMBER_SET/INDEX_SET，等待下一个 WRITEBACK 确定最后使用点）
78 std::vector<uint32_t> pendingMutatedObjs;
79 // METHOD_CALL/SUPER_CALL 的 obj vreg，默认最后使用点为 METHOD_CALL 本身，
80 // 但如果后续有 LOAD_MUTATED（变异方法写回），则延长到 LOAD_MUTATED。
81 // RegisterVM 的 METHOD_CALL 将 objReg 记录到 lastMutatedReceiverReg_，
82 // 后续 LOAD_MUTATED 通过 reg(lastMutatedReceiverReg_) 读取变异后的容器。
83 // 如果 obj vreg 在 METHOD_CALL 后被释放并复用，LOAD_MUTATED 会读到错误值。
84 // 但非变异方法调用（如 len/substr）没有 LOAD_MUTATED，无需延长。
85 // AUDIT-R4 BUG-02 fix: 与 pendingMutatedObjs 对齐为跨基本块保留——原实现
86 // 在每个块开头 clear()，若 METHOD_CALL 与其 LOAD_MUTATED 被块边界分隔
87 // （如 try handler 边、未来 IR 变换插入 LABEL），接收者寄存器会提前
88 // 释放被复用，LOAD_MUTATED 读脏值损坏变异写回。跨块保留仅使无
89 // LOAD_MUTATED 的旧条目被后续 LOAD_MUTATED 保守延长（寄存器多持有
90 // 一段时间，安全方向），不会产生错误释放。
91 std::vector<uint32_t> pendingMethodCallObjs;
92 for (const auto& block : ir.blocks) {
93 for (const auto& instr : block.instructions) {
94 bool isMutatingSet = (instr.op == IROp::MEMBER_SET || instr.op == IROp::INDEX_SET);
95 bool isWriteback =
96 (instr.op == IROp::WRITEBACK_MEMBER_VAR || instr.op == IROp::WRITEBACK_MEMBER_LOCAL ||
97 instr.op == IROp::WRITEBACK_INDEX_VAR || instr.op == IROp::WRITEBACK_INDEX_LOCAL ||
98 instr.op == IROp::WRITEBACK_MEMBER_UPVALUE || instr.op == IROp::WRITEBACK_INDEX_UPVALUE);
99 bool isMethodOrSuperCall = (instr.op == IROp::METHOD_CALL || instr.op == IROp::SUPER_CALL);
100 bool isLoadMutated = (instr.op == IROp::LOAD_MUTATED);
101 for (size_t opi = 0; opi < instr.operands.size(); ++opi) {
102 const auto& op = instr.operands[opi];
103 if (op.kind != IROperandKind::VIRTUAL)
104 continue;
105 // MEMBER_SET/INDEX_SET 的 obj vreg（操作数[0]）跳过——
106 // 其最后使用点延长到下一个 WRITEBACK_* 指令
107 if (isMutatingSet && opi == 0) {
108 pendingMutatedObjs.push_back(op.index);
109 continue;
110 }
111 // METHOD_CALL/SUPER_CALL 的 obj vreg（操作数[1]）：默认最后使用为当前指令，
112 // 但加入 pending 列表，如果后续有 LOAD_MUTATED 则覆盖延长。
113 if (isMethodOrSuperCall && opi == 1) {
114 vregLastUse_[op.index] = globalIdx;
115 pendingMethodCallObjs.push_back(op.index);
116 continue;
117 }
118 vregLastUse_[op.index] = globalIdx;
119 }
120 // 遇到 WRITEBACK_* 时，延长 pending obj vreg 的最后使用点到此处
121 if (isWriteback && !pendingMutatedObjs.empty()) {
122 for (uint32_t v : pendingMutatedObjs) {
123 vregLastUse_[v] = globalIdx;
124 }
125 pendingMutatedObjs.clear();
126 }
127 // 遇到 LOAD_MUTATED 时，延长 METHOD_CALL/SUPER_CALL 的 obj vreg 生命周期到此处。
128 // 这会覆盖之前设的默认值（METHOD_CALL 指令索引），使寄存器在 LOAD_MUTATED 后才释放。
129 if (isLoadMutated && !pendingMethodCallObjs.empty()) {
130 for (uint32_t v : pendingMethodCallObjs) {
131 vregLastUse_[v] = globalIdx;
132 }
133 pendingMethodCallObjs.clear();
134 }
135 ++globalIdx;
136 }
137 }
138 // 未匹配到 WRITEBACK 的 obj vreg 不释放（保守安全，避免寄存器被复用覆盖）
139 // P2-10 fix: 线性扫描不感知循环回边，循环外定义、循环内使用的 vreg 会被提前释放。
140 // 在构建反向映射前，先延长循环不变 vreg 的最后使用点到循环末尾。
141 extendLastUseForLoops(ir);
142 // L4 fix: 线性扫描不感知 finally 块控制流，return 值 vreg 会被 finally 块覆盖。
143 extendLastUseForFinallyJumps(ir);
144
145 // 构建反向映射：instrIndex → 在此处最后使用的 vreg 列表
146 for (const auto& kv : vregLastUse_) {
147 lastUseToVregs_[kv.second].push_back(kv.first);
148 }
149}

151// P2-10 fix: 循环感知最后使用扩展
152// ------------------------------------------------------------
153// 问题：collectVRegLastUse 的线性扫描假设 vreg 的"最后使用"是指令序列中最后一次引用。
154// 但在循环中，vreg 会被多次使用（每轮迭代一次）。如果 vreg 定义在循环外、
155// 使用在循环内（如循环上界常量 v3 = LOAD_CONST 3，用于 LT v8 v22 v3），
156// 线性扫描会在循环内的最后一次引用后释放寄存器。该寄存器可能被循环内的
157// 另一个 vreg 复用（如 MUL v12 v24 v22 复用 v3 的寄存器），导致下一轮迭代
158// LT 指令读到 v12 的值而非 v3 的值（3），循环条件错误。
159//
160// 修复：对每个自然循环，收集循环体内"定义"和"使用"的 vreg 集合。
161// 对使用在循环内但定义在循环外的 vreg（循环不变量），将其最后使用点
162// 延长到循环体的最后一条指令索引，确保寄存器在整个循环期间不被释放复用。
163//
164// 判定"定义"：指令首操作数为 VIRTUAL 且指令属于"写"语义（LOAD_*/算术/比较/CALL 等）。
165// STORE_LOCAL/MEMBER_SET/INDEX_SET 等指令的首操作数虽可能为 VIRTUAL（obj），
166// 但属于"读"语义，不算定义。
167void RegisterBytecodeBackend::extendLastUseForLoops(const IRFunction& ir) {
168 IRCFG cfg(ir);
169 if (cfg.size() < 2)
170 return;
171 DominatorTree domTree(cfg);
172 NaturalLoopInfo loopInfo(cfg, domTree);
173 if (loopInfo.loops().empty())
174 return;
175
176 // 计算每个 IRBasicBlock 在扁平指令序列中的起始偏移
177 // collectVRegLastUse 的 globalIdx = sum(block[0..i-1].instructions.size()) + localIdx
178 std::vector<size_t> blockFlatOffset;
179 blockFlatOffset.reserve(ir.blocks.size());
180 size_t acc = 0;
181 for (const auto& blk : ir.blocks) {
182 blockFlatOffset.push_back(acc);
183 acc += blk.instructions.size();
184 }
185
186 // 辅助：判断指令是否"定义"了首操作数 vreg（写入 dest）
187 // 与 IR.cpp 的 isPureCompute / lowering 代码中的 dest 约定一致：
188 // 有 dest 的指令首操作数为 VIRTUAL；STORE_*/MEMBER_SET/INDEX_SET/WRITEBACK_*
189 // 等无 dest，首操作数为 slot/obj/name_idx（虽可能 VIRTUAL 但非 dest）。
190 static const auto isStoreLikeOp = [](IROp op) {
191 switch (op) {
192 case IROp::STORE_LOCAL:
193 case IROp::STORE_GLOBAL:
194 case IROp::STORE_UPVALUE:
195 case IROp::MEMBER_SET:
196 case IROp::MEMBER_SET_LOCAL: // L7 fix
197 case IROp::INDEX_SET:
198 case IROp::WRITEBACK_MEMBER_VAR:
199 case IROp::WRITEBACK_MEMBER_LOCAL:
200 case IROp::WRITEBACK_INDEX_VAR:
201 case IROp::WRITEBACK_INDEX_LOCAL:
202 case IROp::WRITEBACK_MEMBER_UPVALUE:
203 case IROp::WRITEBACK_INDEX_UPVALUE:
204 case IROp::DEFINE_GLOBAL:
205 case IROp::DELETE_VAR:
206 case IROp::CLOSE_UPVALUE:
207 case IROp::LABEL:
208 case IROp::JUMP:
209 case IROp::JUMP_IF_FALSE:
210 case IROp::RETURN:
211 case IROp::RETURN_NULL:
212 case IROp::THROW:
213 case IROp::POP:
214 case IROp::PRINT:
215 case IROp::YIELD:
216 case IROp::AWAIT:
217 case IROp::INIT_FIELD:
218 return true;
219 default:
220 return false;
221 }
222 };
223
224 for (const auto& loop : loopInfo.loops()) {
225 // 计算循环体的最后一条指令的扁平索引（循环末尾）
226 size_t loopEndIdx = 0;
227 for (uint32_t nodeId : loop.body) {
228 const auto& node = cfg.node(nodeId);
229 size_t nodeEnd = blockFlatOffset[node.sourceBlockIdx] + node.endInstr;
230 if (nodeEnd > loopEndIdx)
231 loopEndIdx = nodeEnd;
232 }
233 if (loopEndIdx == 0)
234 continue;
235
236 // 收集循环体内定义和使用的 vreg
237 std::unordered_set<uint32_t> vregsDefinedInLoop;
238 std::unordered_set<uint32_t> vregsUsedInLoop;
239 for (uint32_t nodeId : loop.body) {
240 const auto& node = cfg.node(nodeId);
241 const auto& blk = ir.blocks[node.sourceBlockIdx];
242 for (size_t i = node.startInstr; i < node.endInstr; ++i) {
243 const auto& instr = blk.instructions[i];
244 bool hasDest = !instr.operands.empty() && instr.operands[0].kind == IROperandKind::VIRTUAL &&
245 !isStoreLikeOp(instr.op);
246 for (size_t opi = 0; opi < instr.operands.size(); ++opi) {
247 const auto& op = instr.operands[opi];
248 if (op.kind != IROperandKind::VIRTUAL)
249 continue;
250 if (opi == 0 && hasDest) {
251 vregsDefinedInLoop.insert(op.index);
252 } else {
253 vregsUsedInLoop.insert(op.index);
254 }
255 }
256 }
257 }
258
259 // 对使用在循环内但定义在循环外的 vreg，延长最后使用点到循环末尾
260 for (uint32_t vreg : vregsUsedInLoop) {
261 if (vregsDefinedInLoop.count(vreg) > 0)
262 continue; // 循环内定义，每轮迭代重新写入，安全释放
263 auto it = vregLastUse_.find(vreg);
264 if (it != vregLastUse_.end() && it->second < loopEndIdx) {
265 it->second = loopEndIdx;
266 }
267 }
268 }
269}

271// L4 fix: finally 块感知最后使用扩展
272// ------------------------------------------------------------
273// 问题：return 在 try-finally 内时，IR 生成如下序列：
274// <define val>
275// PUSH_JUMP_TARGET returnLandingLabel ← 记录返回着陆点
276// JUMP finallyEntryLabel ← 跳到 finally 入口
277// LABEL returnLandingLabel ← 着陆点（控制流从 FINALLY_END 跳回此处）
278// RETURN val ← 使用 return 值 vreg
279// ...
280// LABEL finallyEntryLabel ← finally 块开始
281// <finally block body>
282// FINALLY_END ← finally 块结束（跳回 returnLandingLabel）
283//
284// 线性顺序中 RETURN(val) 在 finally 块之前，故 vregLastUse_[val] = RETURN 索引，
285// finally 块的 vreg 可能复用 val 的寄存器。但控制流上 finally 块在 RETURN 之前执行，
286// 导致 RETURN 读到被 finally 块覆盖的错误值。
287//
288// 修复：扫描 PUSH_JUMP_TARGET 指令，定位对应的 finally 块（finallyEntryLabel→FINALLY_END），
289// 将着陆点后使用但定义在 PUSH_JUMP_TARGET 之前的 vreg 的最后使用点延长到 FINALLY_END。
290void RegisterBytecodeBackend::extendLastUseForFinallyJumps(const IRFunction& ir) {
291 // 扁平化所有指令，构建 label → 全局索引 映射
292 struct FlatInstr {
293 const IRInstruction* instr;
294 size_t globalIdx;
295 };
296 std::vector<FlatInstr> flat;
297 std::unordered_map<uint32_t, size_t> labelToIdx; // label index → global instruction index
298
299 size_t globalIdx = 0;
300 for (const auto& block : ir.blocks) {
301 for (const auto& instr : block.instructions) {
302 flat.push_back({&instr, globalIdx});
303 if (instr.op == IROp::LABEL && !instr.operands.empty() && instr.operands[0].kind == IROperandKind::LABEL) {
304 labelToIdx[instr.operands[0].index] = globalIdx;
305 }
306 ++globalIdx;
307 }
308 }
309 if (flat.empty())
310 return;
311
312 // 扫描 PUSH_JUMP_TARGET，对每个 finally 跳转扩展 vreg 生命周期
313 for (size_t i = 0; i < flat.size(); ++i) {
314 if (flat[i].instr->op != IROp::PUSH_JUMP_TARGET)
315 continue;
316 if (flat[i].instr->operands.empty() || flat[i].instr->operands[0].kind != IROperandKind::LABEL)
317 continue;
318 uint32_t landingPadLabel = flat[i].instr->operands[0].index;
319
320 // 下一条应为 JUMP finallyEntryLabel
321 if (i + 1 >= flat.size() || flat[i + 1].instr->op != IROp::JUMP)
322 continue;
323 if (flat[i + 1].instr->operands.empty() || flat[i + 1].instr->operands[0].kind != IROperandKind::LABEL)
324 continue;
325 uint32_t finallyEntryLabel = flat[i + 1].instr->operands[0].index;
326
327 // 查找着陆点 LABEL 的全局索引
328 auto landingIt = labelToIdx.find(landingPadLabel);
329 if (landingIt == labelToIdx.end())
330 continue;
331 size_t landingIdx = landingIt->second;
332
333 // 查找 finally 入口 LABEL 的全局索引
334 auto finallyEntryIt = labelToIdx.find(finallyEntryLabel);
335 if (finallyEntryIt == labelToIdx.end())
336 continue;
337 size_t finallyEntryIdx = finallyEntryIt->second;
338
339 // 从 finally 入口扫描到 FINALLY_END，确定 finally 块末尾索引
340 size_t finallyEndIdx = finallyEntryIdx;
341 for (size_t j = finallyEntryIdx; j < flat.size(); ++j) {
342 if (flat[j].instr->op == IROp::FINALLY_END) {
343 finallyEndIdx = j;
344 break;
345 }
346 }
347 // 若未找到 FINALLY_END，用 finally 块区域的最大索引兜底
348 if (finallyEndIdx == finallyEntryIdx)
349 finallyEndIdx = flat.size() - 1;
350
351 // 收集着陆点之后、finally 入口之前使用的所有 vreg
352 // （这些 vreg 定义在 PUSH_JUMP_TARGET 之前，需跨越 finally 块存活）
353 // landingIdx 指向 LABEL 指令，从下一条开始扫描到 finallyEntryIdx
354 for (size_t j = landingIdx + 1; j < finallyEntryIdx && j < flat.size(); ++j) {
355 for (const auto& op : flat[j].instr->operands) {
356 if (op.kind != IROperandKind::VIRTUAL)
357 continue;
358 auto it = vregLastUse_.find(op.index);
359 if (it != vregLastUse_.end() && it->second < finallyEndIdx) {
360 it->second = finallyEndIdx;
361 }
362 }
363 }
364 }
365}

367// P0-REGALLOC fix: 释放最后使用点 == instrIndex 的 vreg 的寄存器到 freeRegs_。
368// 在每条指令 lower 完成后调用，确保该指令已读取完所有操作数。
369void RegisterBytecodeBackend::releaseDeadVRegs(size_t instrIndex) {
370 auto it = lastUseToVregs_.find(instrIndex);
371 if (it == lastUseToVregs_.end())
372 return;
373 for (uint32_t vreg : it->second) {
374 auto regIt = vregToReg_.find(vreg);
375 if (regIt != vregToReg_.end()) {
376 freeRegs_.push_back(regIt->second);
377 vregToReg_.erase(regIt);
378 }
379 }
380 lastUseToVregs_.erase(it);
381}

383bool RegisterBytecodeBackend::lower(const IRFunction& ir) {
384 chunk_ = std::make_unique<RegBytecodeChunk>();
385 chunk_->name = ir.name;
386 chunk_->arity = ir.arity;
387 chunk_->requiredArity = ir.requiredArity;
388 // P2-1 fix: localCount 超 32 会导致 chunk_->registerCount > 32，
389 // RegCallFrame::registers 是 std::array<Value, 32>，写入 registers[32+] 为 OOB。
390 // 与 vregToReg 的 reg >= 32 检查互补，硬失败阻止生成损坏字节码。
391 if (ir.localCount > 32) {
392 Logger::Error("RegisterBytecodeBackend: localCount 超出 32 寄存器上限 (" + std::to_string(ir.localCount) +
393 "，函数 " + ir.name + ")，需减少局部变量或实现寄存器溢出",
394 "RegIR");
395 hasError_ = true;
396 return false;
397 }
398 chunk_->localCount = ir.localCount;
399 chunk_->registerCount = ir.localCount; // 初始为 localCount，vreg 分配后增长
400 chunk_->defaultConstIndices = ir.defaultConstIndices;
401 chunk_->upvalues = ir.upvalues;
402 // BUG-IDE-12 fix: 复制 slot→name 映射，供 RegisterVM 条件断点求值反查变量名
403 chunk_->localRegNames = ir.localSlotNames;
404 // R164 协程/生成器：复制生成器标记和 yieldCount（RegisterVM 在 REG_CALL 时检测）
405 chunk_->isGenerator = ir.isGenerator;
406 chunk_->yieldCount = ir.yieldCount;
407
408 // 复制常量池（保持索引一致）
409 for (const auto& c : ir.constants) {
410 chunk_->constants.push_back(c);
411 }
412
413 // P0-REGALLOC fix: 预扫描所有指令，构建 vreg → 最后使用 instrIndex 映射，
414 // 供 releaseDeadVRegs 在 lowering 过程中及时释放死寄存器复用。
415 collectVRegLastUse(ir);
416
417 // 遍历所有基本块的所有指令，逐条 lowering
418 size_t instrIndex = 0;
419 for (const auto& block : ir.blocks) {
420 for (const auto& instr : block.instructions) {
421 irToBytecodeOffset_.push_back({instrIndex, chunk_->code.size()});
422 // D12 fix2: writeOp/writeShort/writeByte 内部以 column=0 补齐 lines/columns
423 // （主循环的 while 补齐是死代码），故 lower 后用 instr.column 修正本指令
424 // 写入的字节区间，使 RegisterVM 调试信息携带列号（BUG-IBACKEND-2）。
425 const size_t codeStart = chunk_->code.size();
426 if (!lowerInstruction(instr, ir) || hasError_) {
427 return false; // lowering 失败或 vreg 溢出：不生成损坏的字节码
428 }
429 for (size_t i = codeStart; i < chunk_->code.size(); ++i) {
430 chunk_->columns[i] = instr.column;
431 }
432 // 填充行号表（writeOp 已同步补齐，此处为防御性兜底）
433 // D12 fix: BUG-IBACKEND-2 修复——columns 填充 instr.column（原恒 0）
434 while (static_cast<int>(chunk_->lines.size()) < static_cast<int>(chunk_->code.size())) {
435 chunk_->lines.push_back(instr.line);
436 chunk_->columns.push_back(instr.column);
437 }
438 // P0-REGALLOC fix: 释放最后使用点 == instrIndex 的 vreg 的寄存器
439 releaseDeadVRegs(instrIndex);
440 ++instrIndex;
441 }
442 }
443 if (!patchJumps())
444 return false;
445
446 // L1 fix: 将 IR 指令范围的 slotNameRanges 翻译为字节码 IP 范围（对齐 BytecodeIRBackend）
447 if (!ir.slotNameRanges.empty() && !irToBytecodeOffset_.empty()) {
448 auto translateInstr = [this](size_t instrIdx) -> size_t {
449 size_t lo = 0, hi = irToBytecodeOffset_.size();
450 while (lo < hi) {
451 size_t mid = (lo + hi) / 2;
452 if (irToBytecodeOffset_[mid].first <= instrIdx)
453 lo = mid + 1;
454 else
455 hi = mid;
456 }
457 if (lo == 0)
458 return irToBytecodeOffset_[0].second;
459 return irToBytecodeOffset_[lo - 1].second;
460 };
461 for (const auto& range : ir.slotNameRanges) {
462 RegBytecodeChunk::SlotNameRange out;
463 out.slot = static_cast<uint8_t>(range.slot);
464 out.name = range.name;
465 out.startIp = translateInstr(range.startInstr);
466 if (range.endInstr == 0) {
467 out.endIp = 0;
468 } else {
469 out.endIp = translateInstr(range.endInstr);
470 }
471 chunk_->slotNameRanges.push_back(std::move(out));
472 }
473 }
474 return true;
475}

477// ============================================================
478// lowerInstruction — 按 IROp 类别分发的 lowering 主入口
479// ------------------------------------------------------------
480// 本函数原为 979 行的巨型 switch，按 IROp 类别拆分为 11 个 lowerXxxOps 私有方法。
481// 主函数保留 switch 外壳做类别分发，每个 case 调用对应的 lowerXxxOps。
482// 三后端语义等价 / 寄存器分配 / patch 位置约定等不变量在拆分后保持一致。
483//
484// 统一的写码约定（理解各 lowerXxxOps 内部 case 的关键）：
485// · writeOp(RegOp) 写 1 字节操作码。
486// · writeReg(reg) 写 1 字节"寄存器号"(0..31)，内部 assert(reg<32)——只用于真正的寄存器。
487// · writeShort(uint16) 写 2 字节小端常量池索引 / 跳转目标 / 名称索引。
488// · writeByte(uint8) 写 1 字节"数量或描述符"(argCount/uvCount/fieldCount/isLocal/idx 等)，
489// 无 assert。正因如此，凡"数量操作数"必须用 writeByte 而非 writeReg，
490// 否则数量 >= 32 时 Debug 构建的 assert 会崩溃（见各处 AUDIT-BUG-C1）。
491// 寄存器式与栈式的本质差异：所有中间结果存在固定寄存器而非操作数栈，故 POP/DUP 退化为
492// REG_MOVE 或 no-op；嵌套左值变异时 obj 寄存器记录在 lastMutatedReceiverReg_，由 WRITEBACK_*
493// 读取——这正是 collectVRegLastUse 需把 obj vreg 寿命延到 WRITEBACK 之后的原因。
494// AUDIT-P2 fix: vregToReg 溢出后设置 hasError_ 并返回占位 reg=0，但此前各 case 不检查
495// hasError_ 仍继续写入损坏字节码。在 lowerInstruction 入口快速失败，避免溢出后后续
496// 指令继续写入。当前指令的损坏字节码由 lower() 行 164 的 hasError_ 检查兜底（return false），
497// chunk_ 不会被 VM 加载使用。
498// ============================================================
499bool RegisterBytecodeBackend::lowerInstruction(const IRInstruction& instr, const IRFunction& ir) {
500 if (hasError_)
501 return false;
502
503 // AUDIT-R4 BUG-05 fix: 操作数 kind 防御——CONSTANT kind 仅在以下白名单指令
504 // 的固定位置合法（LOAD_CONST/TYPE_CHECK/TYPE_TEST/BUILD_ENUM_VARIANT/
505 // ENUM_VARIANT_NAME 的常量池索引位）。其余指令的操作数若出现 CONSTANT
506 // kind（典型来源：copyPropagationPass 将 VIRTUAL 替换为常量索引），
507 // 下方各 lower* 会把常量索引误作 vreg 编号传入 vregToReg 读错寄存器
508 // （静默数据损坏）。此处快速失败转为干净的编译错误，在未来重新
509 // 启用复制传播而 lowering 未先支持常量物化时立即暴露而非静默读错。
510 switch (instr.op) {
511 case IROp::LOAD_CONST:
512 case IROp::TYPE_CHECK:
513 case IROp::TYPE_TEST:
514 case IROp::BUILD_ENUM_VARIANT:
515 case IROp::ENUM_VARIANT_NAME:
516 break; // 白名单：自身按固定位置解释 CONSTANT 操作数
517 default:
518 for (const auto& op : instr.operands) {
519 if (op.kind == IROperandKind::CONSTANT) {
520 Logger::Error("RegisterBytecodeBackend: 指令 " + std::to_string(static_cast<int>(instr.op)) +
521 " 含非法 CONSTANT 操作数（常量索引 " + std::to_string(op.index) +
522 "）——lowering 尚不支持常量物化，请先禁用复制传播",
523 "RegIR");
524 hasError_ = true;
525 return false;
526 }
527 }
528 break;
529 }
530
531 switch (instr.op) {
532 // ---- 常量加载 ----
533 case IROp::LOAD_CONST:
534 case IROp::LOAD_NULL:
535 case IROp::LOAD_TRUE:
536 case IROp::LOAD_FALSE:
537 return lowerConstOps(instr, ir);
538
539 // ---- 局部变量 / 全局变量 ----
540 case IROp::LOAD_LOCAL:
541 case IROp::STORE_LOCAL:
542 case IROp::LOAD_GLOBAL:
543 case IROp::STORE_GLOBAL:
544 case IROp::DEFINE_GLOBAL:
545 case IROp::DELETE_VAR:
546 return lowerVarOps(instr, ir);
547
548 // ---- upvalue / 闭包 ----
549 case IROp::LOAD_UPVALUE:
550 case IROp::STORE_UPVALUE:
551 case IROp::CLOSE_UPVALUE:
552 case IROp::MAKE_CLOSURE:
553 return lowerUpvalueOps(instr, ir);
554
555 // ---- 算术 ----
556 case IROp::ADD:
557 case IROp::SUB:
558 case IROp::MUL:
559 case IROp::DIV:
560 case IROp::MOD:
561 case IROp::NEGATE:
562 return lowerArithOps(instr, ir);
563
564 // ---- 比较 ----
565 case IROp::EQ:
566 case IROp::NEQ:
567 case IROp::LT:
568 case IROp::GT:
569 case IROp::LTE:
570 case IROp::GTE:
571 return lowerCompareOps(instr, ir);
572
573 // ---- 逻辑 ----
574 case IROp::NOT:
575 return lowerLogicOps(instr, ir);
576
577 // ---- 控制流 ----
578 case IROp::LABEL:
579 case IROp::JUMP:
580 case IROp::JUMP_IF_FALSE:
581 case IROp::RETURN:
582 case IROp::RETURN_NULL:
583 // R164 协程/生成器：YIELD 可中断执行（抛 VMYieldSignal），归入控制流组
584 case IROp::YIELD:
585 // 七特性 MVP 阶段 4：AWAIT 同步 drain（可能抛异常/错误），归入控制流组
586 case IROp::AWAIT:
587 return lowerControlOps(instr, ir);
588
589 // ---- 调用 ----
590 case IROp::CALL:
591 case IROp::CALL_EXPR:
592 case IROp::TAIL_CALL: // L18 eng-tailcall
593 case IROp::METHOD_CALL:
594 case IROp::SUPER_CALL:
595 return lowerCallOps(instr, ir);
596
597 // ---- 容器 / 成员访问 ----
598 case IROp::BUILD_ARRAY:
599 case IROp::BUILD_DICT:
600 case IROp::BUILD_TUPLE:
601 case IROp::INDEX_GET:
602 case IROp::INDEX_SET:
603 case IROp::MEMBER_GET:
604 case IROp::SUPER_MEMBER_GET:
605 case IROp::MEMBER_SET:
606 case IROp::MEMBER_SET_LOCAL: // L7 fix
607 // R99 枚举与 ADT：enum variant 构造/检查/取字段（与容器指令同属 lowerContainerOps）
608 case IROp::BUILD_ENUM_VARIANT:
609 case IROp::ENUM_VARIANT_NAME:
610 case IROp::ENUM_VARIANT_FIELD:
611 // R133 模式匹配扩展：容器长度（与容器指令同属 lowerContainerOps）
612 case IROp::LEN:
613 return lowerContainerOps(instr, ir);
614
615 // ---- 类 ----
616 case IROp::DEFINE_CLASS:
617 case IROp::CLASS_NEW:
618 case IROp::INIT_FIELD:
619 return lowerClassOps(instr, ir);
620
621 // ---- 异常 / 写回 / 其他 ----
622 case IROp::TRY_BEGIN:
623 case IROp::TRY_END:
624 case IROp::THROW:
625 case IROp::LOAD_EXCEPTION:
626 case IROp::PUSH_JUMP_TARGET:
627 case IROp::FINALLY_END:
628 case IROp::WRITEBACK_MEMBER_VAR:
629 case IROp::WRITEBACK_MEMBER_LOCAL:
630 case IROp::WRITEBACK_INDEX_VAR:
631 case IROp::WRITEBACK_INDEX_LOCAL:
632 case IROp::WRITEBACK_MEMBER_UPVALUE:
633 case IROp::WRITEBACK_INDEX_UPVALUE:
634 case IROp::PRINT:
635 case IROp::POP:
636 case IROp::DUP:
637 case IROp::LOAD_MUTATED:
638 case IROp::TYPE_CHECK:
639 case IROp::TYPE_TEST: // R133: 软类型测试，与 TYPE_CHECK 同属 misc 类
640 return lowerMiscOps(instr, ir);
641
642 default:
643 Logger::Error("RegisterBytecodeBackend: 未支持的 IR 指令 " + std::to_string(static_cast<int>(instr.op)),
644 "RegIR");
645 return false;
646 }

705// ============================================================
706// lowerVarOps — 变量类（LOAD/STORE_LOCAL + LOAD/STORE/DEFINE_GLOBAL + DELETE_VAR）
707// ============================================================
708bool RegisterBytecodeBackend::lowerVarOps(const IRInstruction& instr, const IRFunction& ir) {
709 int line = instr.line;
710 switch (instr.op) {
711 // ---- 局部变量（在寄存器式中，local slot 即寄存器）----
712 case IROp::LOAD_LOCAL: {
713 // IR: LOAD_LOCAL dest_vreg, slot
714 // → REG_MOVE dest_vreg_reg, slot_reg
715 if (instr.operands.size() < 2)
716 return false;
717 // Bug-14 同型修复：local slot 超 32 寄存器硬上限静默截断会读写错误寄存器
718 if (instr.operands[1].index >= 32) {
719 Logger::Error("RegBytecodeBackend: LOAD_LOCAL local slot 超出 32 寄存器上限 (slot=" +
720 std::to_string(instr.operands[1].index) + ")",
721 "RegBackend");
722 return false;
723 }
724 uint8_t dst = vregToReg(instr.operands[0].index);
725 uint8_t src = static_cast<uint8_t>(instr.operands[1].index);
726 chunk_->writeOp(RegOp::REG_MOVE, line);
727 chunk_->writeReg(dst, line);
728 chunk_->writeReg(src, line);
729 break;
730 }
731 case IROp::STORE_LOCAL: {
732 // IR: STORE_LOCAL slot, src_vreg
733 // → REG_MOVE slot_reg, src_vreg_reg
734 if (instr.operands.size() < 2)
735 return false;
736 if (instr.operands[0].index >= 32) {
737 Logger::Error("RegBytecodeBackend: STORE_LOCAL local slot 超出 32 寄存器上限 (slot=" +
738 std::to_string(instr.operands[0].index) + ")",
739 "RegBackend");
740 return false;
741 }
742 uint8_t dst = static_cast<uint8_t>(instr.operands[0].index);
743 uint8_t src = vregToReg(instr.operands[1].index);
744 chunk_->writeOp(RegOp::REG_MOVE, line);
745 chunk_->writeReg(dst, line);
746 chunk_->writeReg(src, line);
747 break;
748 }
749
750 // ---- 全局变量 ----
751 case IROp::LOAD_GLOBAL: {
752 // IR: LOAD_GLOBAL dest_vreg, global_idx
753 // IMM_UINT → REG_LOAD_GLOBAL dest, slot(2B)
754 // GLOBAL_NAME → REG_LOAD_GLOBAL dest, nameIdx(2B)（统一用 slot 形式，name 也走常量池）
755 if (instr.operands.size() < 2)
756 return false;
757 uint8_t dst = vregToReg(instr.operands[0].index);
758 if (instr.operands[1].kind == IROperandKind::IMM_UINT) {
759 chunk_->writeOp(RegOp::REG_LOAD_GLOBAL, line);
760 chunk_->writeReg(dst, line);
761 chunk_->writeShort(static_cast<uint16_t>(instr.operands[1].index), line);
762 } else {
763 // 名称版：将名称作为常量，用高 bit 标记（slot < 32768, nameIdx >= 32768）
764 // 简化：名称版也走常量池查找 slot
765 uint16_t nameIdx = addStringConstant(globalName(ir, instr.operands[1].index), ir);
766 chunk_->writeOp(RegOp::REG_LOAD_GLOBAL, line);
767 chunk_->writeReg(dst, line);
768 chunk_->writeShort(nameIdx | 0x8000, line); // 高 bit 置 1 表示名称索引
769 }
770 break;
771 }
772 case IROp::STORE_GLOBAL: {
773 if (instr.operands.size() < 2)
774 return false;
775 uint8_t src = vregToReg(instr.operands[1].index);
776 if (instr.operands[0].kind == IROperandKind::IMM_UINT) {
777 chunk_->writeOp(RegOp::REG_STORE_GLOBAL, line);
778 chunk_->writeReg(src, line);
779 chunk_->writeShort(static_cast<uint16_t>(instr.operands[0].index), line);
780 } else {
781 uint16_t nameIdx = addStringConstant(globalName(ir, instr.operands[0].index), ir);
782 chunk_->writeOp(RegOp::REG_STORE_GLOBAL, line);
783 chunk_->writeReg(src, line);
784 chunk_->writeShort(nameIdx | 0x8000, line);
785 }
786 break;
787 }
788 case IROp::DEFINE_GLOBAL: {
789 if (instr.operands.size() < 2)
790 return false;
791 uint8_t src = vregToReg(instr.operands[1].index);
792 if (instr.operands[0].kind == IROperandKind::IMM_UINT) {
793 chunk_->writeOp(RegOp::REG_DEFINE_GLOBAL, line);
794 chunk_->writeReg(src, line);
795 chunk_->writeShort(static_cast<uint16_t>(instr.operands[0].index), line);
796 } else {
797 uint16_t nameIdx = addStringConstant(globalName(ir, instr.operands[0].index), ir);
798 chunk_->writeOp(RegOp::REG_DEFINE_GLOBAL, line);
799 chunk_->writeReg(src, line);
800 chunk_->writeShort(nameIdx | 0x8000, line);
801 }
802 break;
803 }
804 case IROp::DELETE_VAR: {
805 // BUG-IR-TRY-1 fix: 删除全局变量（catch 块退出后清理 catch 变量）
806 // operands: [global_idx]，kind 可为 IMM_UINT (GLOBAL_SLOT) 或 GLOBAL_NAME
807 // → REG_DELETE_GLOBAL nameConstIdx(2B)
808 // 注意：REG_DELETE_GLOBAL 按 name 常量索引删除，无高 bit 标记区分
809 if (instr.operands.size() < 1)
810 return false;
811 uint16_t nameIdx;
812 if (instr.operands[0].kind == IROperandKind::IMM_UINT) {
813 // IMM_UINT (slot) 路径：RegisterBytecodeBackend 无 globalSlotNames 表，
814 // 理论上不会触达（visitTryStmt 用 GLOBAL_NAME kind），此处保守返回 false。
815 Logger::Error("RegisterBytecodeBackend: DELETE_VAR 不支持 IMM_UINT kind（无 slot→name 表）", "RegIR");
816 return false;
817 } else {
818 nameIdx = addStringConstant(globalName(ir, instr.operands[0].index), ir);
819 }
820 chunk_->writeOp(RegOp::REG_DELETE_GLOBAL, line);
821 chunk_->writeShort(nameIdx, line);
822 break;
823 }
824 default:
825 return false;
826 }
827 return true;
828}

830// ============================================================
831// lowerUpvalueOps — 闭包 upvalue 类（LOAD/STORE_UPVALUE + CLOSE_UPVALUE + MAKE_CLOSURE）
832// ============================================================
833bool RegisterBytecodeBackend::lowerUpvalueOps(const IRInstruction& instr, const IRFunction& ir) {
834 int line = instr.line;
835 switch (instr.op) {
836 case IROp::LOAD_UPVALUE: {
837 if (instr.operands.size() < 2)
838 return false;
839 // Bug-14 同型修复：upvalue 索引超 255 静默截断会读写错误槽
840 if (instr.operands[1].index >= 256) {
841 Logger::Error("RegBytecodeBackend: LOAD_UPVALUE upvalue 索引超出 255 上限 (uvIdx=" +
842 std::to_string(instr.operands[1].index) + ")",
843 "RegBackend");
844 return false;
845 }
846 uint8_t dst = vregToReg(instr.operands[0].index);
847 chunk_->writeOp(RegOp::REG_LOAD_UPVALUE, line);
848 chunk_->writeReg(dst, line);
849 chunk_->code.push_back(static_cast<uint8_t>(instr.operands[1].index));
850 while (static_cast<int>(chunk_->lines.size()) < static_cast<int>(chunk_->code.size())) {
851 chunk_->lines.push_back(line);
852 chunk_->columns.push_back(instr.column); // D12 fix: BUG-IBACKEND-2（原恒 0）
853 }
854 break;
855 }
856 case IROp::STORE_UPVALUE: {
857 if (instr.operands.size() < 2)
858 return false;
859 if (instr.operands[0].index >= 256) {
860 Logger::Error("RegBytecodeBackend: STORE_UPVALUE upvalue 索引超出 255 上限 (uvIdx=" +
861 std::to_string(instr.operands[0].index) + ")",
862 "RegBackend");
863 return false;
864 }
865 uint8_t src = vregToReg(instr.operands[1].index);
866 chunk_->writeOp(RegOp::REG_STORE_UPVALUE, line);
867 chunk_->writeReg(src, line);
868 chunk_->code.push_back(static_cast<uint8_t>(instr.operands[0].index));
869 while (static_cast<int>(chunk_->lines.size()) < static_cast<int>(chunk_->code.size())) {
870 chunk_->lines.push_back(line);
871 chunk_->columns.push_back(instr.column); // D12 fix: BUG-IBACKEND-2（原恒 0）
872 }
873 break;
874 }
875 case IROp::CLOSE_UPVALUE: {
876 // B1 fix: operand 现为 slot_base（块作用域基址），运行时关闭所有
877 // 指向 slot >= frameIdx*MAX_REGISTERS+slot_base 的 open upvalues。
878 if (instr.operands.empty())
879 return false;
880 if (instr.operands[0].index >= 256) {
881 Logger::Error("RegBytecodeBackend: CLOSE_UPVALUE slot_base 超出 255 上限 (slot=" +
882 std::to_string(instr.operands[0].index) + ")",
883 "RegBackend");
884 return false;
885 }
886 chunk_->writeOp(RegOp::REG_CLOSE_UPVALUE, line);
887 chunk_->code.push_back(static_cast<uint8_t>(instr.operands[0].index));
888 while (static_cast<int>(chunk_->lines.size()) < static_cast<int>(chunk_->code.size())) {
889 chunk_->lines.push_back(line);
890 chunk_->columns.push_back(instr.column); // D12 fix: BUG-IBACKEND-2（原恒 0）
891 }
892 break;
893 }
894 case IROp::MAKE_CLOSURE: {
895 // IR: MAKE_CLOSURE dest, name_idx, uv_count, [isLocal, idx]×uv_count
896 if (instr.operands.size() < 3)
897 return false;
898 // P2-3 fix: uvCount/isLocal/idx 经 writeByte 编码为 1 字节，
899 // & 0xFF 静默截断会生成错误 upvalue 描述符（闭包捕获错误变量）。
900 if (instr.operands[2].index >= 256) {
901 Logger::Error("RegisterBytecodeBackend: MAKE_CLOSURE uvCount 超出 255 上限 (" +
902 std::to_string(instr.operands[2].index) + ")",
903 "RegIR");
904 hasError_ = true;
905 return false;
906 }
907 uint8_t dst = vregToReg(instr.operands[0].index);
908 uint16_t nameIdx = addStringConstant(globalName(ir, instr.operands[1].index), ir);
909 uint8_t uvCount = static_cast<uint8_t>(instr.operands[2].index);
910 chunk_->writeOp(RegOp::REG_MAKE_CLOSURE, line);
911 chunk_->writeReg(dst, line); // dst 是真正的寄存器号，用 writeReg
912 chunk_->writeShort(nameIdx, line);
913 // AUDIT-REGVB fix: uvCount/isLocal/idx 不是寄存器号，是 upvalue 描述符。
914 // 原用 writeReg（含 assert(reg < 32)），uvCount >= 32 或 idx >= 32 时 Debug 构建崩溃。
915 // 改用 writeByte（无 assert），合法范围 0-255 已由上方 >= 256 检查保证。
916 chunk_->writeByte(uvCount, line);
917 for (uint8_t i = 0; i < uvCount; ++i) {
918 size_t base = 3 + i * 2;
919 if (base + 1 >= instr.operands.size())
920 return false;
921 if (instr.operands[base].index >= 256) {
922 Logger::Error("RegisterBytecodeBackend: MAKE_CLOSURE isLocal 超出 255 上限 (" +
923 std::to_string(instr.operands[base].index) + ")",
924 "RegIR");
925 hasError_ = true;
926 return false;
927 }
928 if (instr.operands[base + 1].index >= 256) {
929 Logger::Error("RegisterBytecodeBackend: MAKE_CLOSURE upvalue idx 超出 255 上限 (" +
930 std::to_string(instr.operands[base + 1].index) + ")",
931 "RegIR");
932 hasError_ = true;
933 return false;
934 }
935 chunk_->writeByte(static_cast<uint8_t>(instr.operands[base].index), line); // isLocal
936 chunk_->writeByte(static_cast<uint8_t>(instr.operands[base + 1].index), line); // idx
937 }
938 break;
939 }
940 default:
941 return false;
942 }
943 return true;
944}

1081// lowerControlOps — 控制流类（LABEL / JUMP / JUMP_IF_FALSE / RETURN / RETURN_NULL）
1082// ============================================================
1083bool RegisterBytecodeBackend::lowerControlOps(const IRInstruction& instr, const IRFunction& /*ir*/) {
1084 int line = instr.line;
1085 switch (instr.op) {
1086 case IROp::LABEL: {
1087 if (instr.operands.empty())
1088 return false;
1089 labelToOffset_[instr.operands[0].index] = chunk_->code.size();

1097 chunk_->writeShort(0, line); // 占位符
1098 break;
1099 }
1100 case IROp::JUMP_IF_FALSE: {
1101 if (instr.operands.size() < 2)
1102 return false;
1103 uint8_t src = vregToReg(instr.operands[0].index);
1104 chunk_->writeOp(RegOp::REG_JUMP_IF_FALSE, line);
1105 chunk_->writeReg(src, line);
1106 pendingJumps_.push_back({chunk_->code.size(), instr.operands[1].index});
1107 chunk_->writeShort(0, line); // 占位符
1108 break;
1109 }
1110 case IROp::RETURN: {
1111 if (instr.operands.empty())
1112 return false;
1113 uint8_t src = vregToReg(instr.operands[0].index);
1114 chunk_->writeOp(RegOp::REG_RETURN, line);
1115 chunk_->writeReg(src, line);
1116 break;
1117 }
1118 case IROp::RETURN_NULL: {
1119 chunk_->writeOp(RegOp::REG_RETURN_NULL, line);
1120 break;
1121 }
1122 // R164 协程/生成器：YIELD dest, src → REG_YIELD dst, src（3 字节）
1123 // 语义：RegisterVM 执行 REG_YIELD 时比较运行时计数器与目标 yieldId，
1124 // - 命中目标：抛 VMYieldSignal 返回 src 寄存器值
1125 // - 未命中：dst = src，继续执行函数体
1126 case IROp::YIELD: {

1128 return false;
1129 uint8_t dst = vregToReg(instr.operands[0].index);
1130 uint8_t src = vregToReg(instr.operands[1].index);
1131 chunk_->writeOp(RegOp::REG_YIELD, line);
1132 chunk_->writeReg(dst, line);
1133 chunk_->writeReg(src, line);
1134 break;
1135 }

1137 // 语义：RegisterVM 执行 REG_AWAIT 时：src 非协程 → dst = src；
1138 // src 为协程 → 驱动到 done，dst = 最后一次 next() 的值。
1139 case IROp::AWAIT: {
1140 if (instr.operands.size() < 2)
1141 return false;
1142 uint8_t dst = vregToReg(instr.operands[0].index);
1143 uint8_t src = vregToReg(instr.operands[1].index);
1144 chunk_->writeOp(RegOp::REG_AWAIT, line);
1145 chunk_->writeReg(dst, line);
1146 chunk_->writeReg(src, line);
1147 break;
1148 }
1149 default:
1150 return false;
1151 }
1152 return true;
1153}

1625 // BUG-INH-1 fix: 新增字段默认值常量索引
1626 // BUG-INH-IR-1 fix: 新增字段表达式临时局部变量槽位（=寄存器号）
1627 // 操作数布局（见 IR.cpp visitClassDecl）：
1628 // [0] className (FUNC_NAME)
1629 // [1] parentName (IMM_UINT, UINT32_MAX=无父类)
1630 // [2] fieldCount (IMM_UINT)
1631 // [3+i*3] field name (FIELD_NAME)
1632 // [3+i*3+1] fieldDefaultConstIdx (IMM_UINT, UINT32_MAX=null/无默认值/有表达式)
1633 // [3+i*3+2] fieldExprLocalSlot (IMM_UINT, UINT32_MAX=使用常量或null, 否则使用临时 local slot)

1645 if (instr.operands.size() < 3 + fieldCount * 3 + 1)
1646 return false;
1647 uint32_t methodCount = instr.operands[3 + fieldCount * 3].index;
1648 if (instr.operands.size() < 3 + fieldCount * 3 + 1 + methodCount * 2)
1649 return false;
1650 // R7 fix: fieldCount/methodCount 经 writeByte 编码为 uint8_t，超 255 时静默截断

1656 }

1658 Logger::Error("RegisterBytecodeBackend: DEFINE_CLASS methodCount 超过 255 上限", "RegIR");

1665 chunk_->writeShort(parentIdx, line);
1666 chunk_->writeByte(static_cast<uint8_t>(fieldCount), line);
1667 for (uint32_t i = 0; i < fieldCount; ++i) {
1668 size_t nameOpIdx = 3 + i * 3;
1669 size_t defaultOpIdx = 3 + i * 3 + 1;
1670 size_t exprSlotOpIdx = 3 + i * 3 + 2;

1672 chunk_->writeShort(fIdx, line);

1677 uint32_t exprSlot = instr.operands[exprSlotOpIdx].index;

1679 // 非字面量表达式：default const 写 NO_INDEX，exprReg 写 slot 号
1680 if (exprSlot >= 32) {
1681 Logger::Error("RegisterBytecodeBackend: DEFINE_CLASS 字段临时局部变量槽位超出 32 寄存器上限",

1684 return false;
1685 }

1687 chunk_->writeByte(static_cast<uint8_t>(exprSlot), line);
1688 } else if (defaultConstIdx == UINT32_MAX) {

1694 hasError_ = true;
1695 return false;
1696 }
1697 const Value& defaultVal = ir.constants[defaultConstIdx];
1698 uint16_t constIdx = chunk_->addConstant(defaultVal);
1699 chunk_->writeShort(constIdx, line);
1700 chunk_->writeByte(RuntimeLimits::NO_SLOT, line); // NO_SLOT = 无表达式寄存器
1701 }
1702 }
1703 chunk_->writeByte(static_cast<uint8_t>(methodCount), line);
1704 for (uint32_t i = 0; i < methodCount; ++i) {
1705 size_t base = 3 + fieldCount * 3 + 1 + i * 2;
1706 uint16_t mIdx = addStringConstant(globalName(ir, instr.operands[base].index), ir);
1707 uint16_t fIdx = addStringConstant(globalName(ir, instr.operands[base + 1].index), ir);
1708 chunk_->writeShort(mIdx, line);
1709 chunk_->writeShort(fIdx, line);
1710 }
1711 break;
1712 }
1713 case IROp::CLASS_NEW: {
1714 // IR: CLASS_NEW dest, name_idx, arg_count, args...
1715 if (instr.operands.size() < 3)
1716 return false;
1717 uint8_t dst = vregToReg(instr.operands[0].index);
1718 uint16_t nameIdx = addStringConstant(globalName(ir, instr.operands[1].index), ir);
1719 if (instr.operands[2].index > 254) {
1720 Logger::Error("RegisterBytecodeBackend: CLASS_NEW 参数数量超过 254 上限（含 instance 共 255）", "RegIR");
1721 hasError_ = true;
1722 return false;
1723 }
1724 uint8_t argCount = static_cast<uint8_t>(instr.operands[2].index & 0xFF);
1725 chunk_->writeOp(RegOp::REG_CLASS_NEW, line);
1726 chunk_->writeReg(dst, line);
1727 chunk_->writeShort(nameIdx, line);
1728 // AUDIT-BUG-C1 fix: argCount 是数量操作数，改用 writeByte（详见 MAKE_CLOSURE:504）。

1730 for (uint8_t i = 0; i < argCount; ++i) {
1731 if (static_cast<size_t>(3 + i) >= instr.operands.size())
1732 return false;
1733 chunk_->writeReg(vregToReg(instr.operands[3 + i].index), line);
1734 }
1735 break;
1736 }
1737 case IROp::INIT_FIELD: {
1738 if (instr.operands.empty())
1739 return false;
1740 uint16_t fieldIdx = addStringConstant(globalName(ir, instr.operands[0].index), ir);
1741 chunk_->writeOp(RegOp::REG_INIT_FIELD, line);
1742 chunk_->writeShort(fieldIdx, line);
1743 break;
1744 }
1745 default:
1746 return false;
1747 }
1748 return true;
1749}

1960bool RegisterBytecodeBackend::patchJumps() {
1961 for (const auto& pj : pendingJumps_) {
1962 auto it = labelToOffset_.find(pj.targetLabel);
1963 if (it == labelToOffset_.end()) {
1964 Logger::Error("RegisterBytecodeBackend: 未找到标签 " + std::to_string(pj.targetLabel), "RegIR");
1965 return false;
1966 }
1967 // 与 BytecodeIRBackend::patchJumps 对齐：字节码体积超过 64KB 时跳转目标截断，
1968 // 会导致 RegisterVM 跳到错误地址执行任意指令。
1969 if (it->second > 65535) {
1970 Logger::Error("RegisterBytecodeBackend: 跳转目标偏移超过 64KB 限制 (offset=" + std::to_string(it->second) +
1971 ", label=" + std::to_string(pj.targetLabel) + ")",
1972 "RegIR");
1973 return false;
1974 }
1975 uint16_t target = static_cast<uint16_t>(it->second);
1976 if (pj.codeOffset + 1 >= chunk_->code.size()) {
1977 Logger::Error("RegisterBytecodeBackend: 跳转操作数越界", "RegIR");
1978 return false;
1979 }
1980 chunk_->code[pj.codeOffset] = static_cast<uint8_t>(target & 0xFF);
1981 chunk_->code[pj.codeOffset + 1] = static_cast<uint8_t>((target >> 8) & 0xFF);
1982 }
1983 return true;
1984}

1986bool RegisterBytecodeBackend::lowerModule(const IRModule& module) {
1987 resetState();
1988
1989 // lower main 函数
1990 if (!module.mainFunction) {
1991 Logger::Error("RegisterBytecodeBackend: module 无 mainFunction", "RegIR");
1992 return false;
1993 }
1994 if (!lower(*module.mainFunction)) {
1995 return false;
1996 }
1997
1998 // R103 W1 fix: mainFunction 的 chunk 保留在 chunk_ 中，由调用方 takeChunk() 获取，
1999 // 不再放入 functionChunks_。原实现把 mainChunk 放入 functionChunks_["main"]，
2000 // 当用户源码含 `fun main() {...}` 时，子函数 "main" 会覆盖 mainFunction 的 chunk，
2001 // 导致 compileViaRegisterIR 的 find("main") 拿到用户函数而非顶层代码，
2002 // RegVM 执行用户函数体而非顶层代码，输出为空。
2003 // 对齐 BytecodeIRBackend::lowerModule 的设计：mainFunction → chunk_，子函数 → functionChunks_。
2004 chunk_->writeOp(RegOp::REG_RETURN_NULL, 0);
2005 chunk_->buildIpMap();
2006
2007 // lower 子函数（每个子函数独立 backend 实例，避免覆盖状态）
2008 for (const auto& fn : module.functions) {
2009 RegisterBytecodeBackend subBackend;
2010 if (!subBackend.lower(*fn)) {
2011 return false;
2012 }
2013 auto subChunk = subBackend.takeChunk();
2014 subChunk->writeOp(RegOp::REG_RETURN_NULL, 0);
2015 subChunk->buildIpMap();
2016 functionChunks_[fn->name] = std::move(*subChunk);
2017 }
2018
2019 return true;
2020}
