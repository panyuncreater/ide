4// 教学目标：通过亲手操作理解"栈式 VM 就是 push 和 pop"。

6// 1. 计算 1 + 2 — 最基本的 push-pop

8// 3. 计算 (1 + 2) * 3 — 同样的数不同的指令顺序
9// 4. 打印 "hello" — 字符串也是值
10// 5. 自由模式 — 自己探索

26/// 沙盒指令类型
27enum class SandboxOpType {
28 PUSH_INT, // 压入 int 常量，operand 是数字字符串
29 PUSH_STRING, // 压入 string 常量，operand 是字符串内容
30 ADD, // 二元加（pop 两个，push 结果）
31 SUB, // 二元减
32 MUL, // 二元乘
33 DIV, // 二元除（除数为 0 报错）
34 MOD, // 二元取模（除数为 0 报错）
35 NEG, // 一元负（pop 一个，push -x）
36 PRINT, // pop 一个并输出
37 HALT // 结束
38};
39
40/// 单条沙盒指令
41struct SandboxOp {
42 SandboxOpType type;
43 std::string operand; // 操作数（PUSH_INT 时是数字字符串，PUSH_STRING 时是字符串内容，其他类型忽略）
44};
45
46/// 单个关卡定义
47struct SandboxLevel {
48 int level; // 1-5
49 std::string goal; // 目标描述（如 "计算 1 + 2"）
50 std::vector<SandboxOp> availableOps; // 可用指令按钮列表
51 std::vector<SandboxOp> expectedSequence; // 正确顺序的指令序列（自由模式可为空）
52 std::string expectedOutput; // 期望输出（如 "3"）
53 std::string teachingPoint; // 教学点
54 std::string hint; // 提示文本
55 int difficulty; // 1-3（⭐/⭐⭐/⭐⭐⭐）
56};
