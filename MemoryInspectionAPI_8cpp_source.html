24std::string bitsToHex(uint64_t bits) {
25 std::ostringstream os;
26 os << "0x" << std::hex << std::setfill('0') << std::setw(16) << bits;
27 return os.str();
28}

30std::string bitsToBinary(uint64_t bits) {
31 std::string s(64, '0');
32 for (int i = 0; i < 64; ++i) {
33 if (bits & (1ULL << (63 - i)))
34 s[i] = '1';
35 }
36 return s;
37}

39std::string ptrToHex(const void* p) {
40 std::ostringstream os;
41 os << "0x" << std::hex << std::setfill('0') << std::setw(16) << reinterpret_cast<uintptr_t>(p);
42 return os.str();
43}

144NaNBoxSnapshot MemoryInspectionAPI::inspectValue(const Value& v) {
145 NaNBoxSnapshot snap;
146 uint64_t bits = 0;
147 switch (v.getType()) {
148 case ValueType::VAL_INT:
149 // AUDIT-P1 fix: BoxedIntData（超大整数装箱为堆对象）的 type==VAL_INT
150 // 但 isPointer()==true，此时 intVal() 返回原始 int64（可能超出 int48 范围），
151 // 调用 NaNBox::fromInt 会触发 canEncodeInt 失败 → std::abort() 崩溃。
152 // 对 BoxedIntData 显示堆指针占位符，与堆类型 default 分支一致。
153 if (v.isPointer()) {
154 bits = 0x7ffbULL << 48; // PTR_TAG_BASE 占位符
155 snap.tagName = "POINTER";
156 snap.tagValue = 4;
157 } else {
158 bits = NaNBox::fromInt(v.intVal()).rawBits();
159 snap.tagName = "INT";
160 snap.tagValue = 1;
161 }
162 break;
163 case ValueType::VAL_FLOAT:
164 bits = NaNBox::fromFloat(v.floatVal()).rawBits();
165 snap.tagName = "FLOAT";
166 snap.tagValue = 0;
167 break;
168 case ValueType::VAL_BOOL:
169 bits = NaNBox::fromBool(v.boolVal()).rawBits();
170 snap.tagName = "BOOL";
171 snap.tagValue = 2;
172 break;
173 case ValueType::VAL_NULL:
174 bits = NaNBox::null().rawBits();
175 snap.tagName = "NUL";
176 snap.tagValue = 3;
177 break;
178 default:
179 // 堆指针类型：指针值不固定，教学展示用 PTR_TAG_BASE 占位符
180 bits = 0x7ffbULL << 48;
181 snap.tagName = "POINTER";
182 snap.tagValue = 4;
183 break;
184 }
185 snap.bits = bits;
186 snap.bitsHex = bitsToHex(bits);
187 snap.bitsBinary = bitsToBinary(bits);
188 return snap;
189}

191NaNBoxSnapshot MemoryInspectionAPI::inspectBits(uint64_t bits) {
192 NaNBoxSnapshot snap;
193 snap.bits = bits;
194 snap.bitsHex = bitsToHex(bits);
195 snap.bitsBinary = bitsToBinary(bits);
196 // 通过 NaNBox 解码获取 tag
197 NaNBox box = NaNBox::fromBits(bits);
198 int t = static_cast<int>(box.tag());
199 snap.tagValue = t;
200 snap.tagName = tagName(t);
201 return snap;
202}

244HeapObjectSnapshot MemoryInspectionAPI::inspectHeap(const Value& v) {
245 HeapObjectSnapshot snap;
246 if (!v.isPointer()) {
247 snap.isPointer = false;
248 snap.valueType = static_cast<int>(v.getType());
249 snap.valueTypeName = valueTypeName(snap.valueType);
250 snap.structName = "<scalar>";
251 return snap;
252 }
253 RefCounted* p = v.asPointer();
254 snap.isPointer = true;
255 snap.addressHex = ptrToHex(p);
256 if (!p) {
257 snap.structName = "<null-ptr>";
258 return snap;
259 }
260 int t = static_cast<int>(p->type);
261 snap.valueType = t;
262 snap.valueTypeName = valueTypeName(t);
263 snap.structName = heapStructNameForType(t);
264 snap.refCount = p->useCount();
265 // 计算字段数/元素数
266 switch (p->type) {
267 case ValueType::VAL_INT:
268 snap.fieldCount = 1; // BoxedIntData.value
269 break;
270 case ValueType::VAL_STRING:
271 snap.fieldCount = static_cast<size_t>(v.codepointCount());
272 break;
273 case ValueType::VAL_ARRAY:
274 snap.fieldCount = v.arrayVal().size();
275 break;
276 case ValueType::VAL_DICT:
277 snap.fieldCount = v.dictVal().size();
278 break;
279 case ValueType::VAL_INSTANCE:
280 snap.fieldCount = v.fields().size();
281 break;
282 case ValueType::VAL_CLOSURE:
283 snap.fieldCount = v.capturedVars().size();
284 break;
285 case ValueType::VAL_TUPLE:
286 snap.fieldCount = v.tupleVal().size();
287 break;
288 case ValueType::VAL_ENUM_VARIANT:
289 snap.fieldCount = v.enumVariantFields().size();
290 break;
291 case ValueType::VAL_CHANNEL:
292 case ValueType::VAL_MUTEX:
293 case ValueType::VAL_RWLOCK:
294 case ValueType::VAL_THREAD:
295 case ValueType::VAL_COROUTINE:
296 // 并发原语/协程句柄：底层资源经 shared_ptr 封装，无可枚举字段
297 snap.fieldCount = 0;
298 break;
299 case ValueType::VAL_NULL:
300 case ValueType::VAL_FLOAT:
301 case ValueType::VAL_BOOL:
302 snap.fieldCount = 0;
303 break;
304 }
305 return snap;
306}
