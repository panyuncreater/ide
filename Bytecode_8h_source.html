339 uint16_t addConstant(const Value& val) {
340 size_t h = hashValue(val);
341 auto& bucket = constantHashMap_[h];
342 for (uint16_t i : bucket) {
343 if (constants[i].getType() == val.getType() && constants[i].equals(val))
344 return i;
345 }
346 if (constants.size() >= 65535) {
347 throw std::runtime_error("编译错误: 常量池超出限制 (65535)");
348 }
349 uint16_t idx = static_cast<uint16_t>(constants.size());
350 constants.push_back(val);
351 bucket.push_back(idx);
352 return idx;
353 }

356 /// 常量值哈希（用于 O(1) 去重）
357 static size_t hashValue(const Value& val) {
358 switch (val.getType()) {
359 case ValueType::VAL_INT:
360 return std::hash<int64_t>{}(val.intVal());
361 case ValueType::VAL_FLOAT:
362 return std::hash<double>{}(val.floatVal());
363 case ValueType::VAL_STRING:
364 return std::hash<std::string>{}(val.stringVal());
365 case ValueType::VAL_BOOL:
366 return std::hash<bool>{}(val.boolVal());
367 case ValueType::VAL_NULL:
368 return 0;
369 default:
370 return static_cast<size_t>(val.getType());
371 }
372 }

384 /// 预计算 ip→指令索引映射（编译完成后调用一次）
385 void buildIpMap() {
386 ipToInstrIndex.assign(code.size(), -1);
387 size_t offset = 0;
388 int instrIdx = 0;
389 while (offset < code.size()) {
390 ipToInstrIndex[offset] = instrIdx;
391 // D7 fix: 统一使用 instructionSizeAt 处理变长指令
392 offset += instructionSizeAt(offset);

397 /// D7 fix: 获取指定偏移处指令的实际长度（字节数）。
398 /// OP_CLOSURE 是唯一变长指令（4 + 2*upvalueCount 字节），
399 /// 此方法封装变长逻辑，消除 buildIpMap/ide 遍历/TestCompiler 遍历 3 处特例处理。
400 /// 若偏移越界或指令截断，返回 1 避免无限循环。
401 // A3 fix: 通过 isOpCodeVariableLength 元数据判断变长，无需硬编码 OP_CLOSURE 特例
402 size_t instructionSizeAt(size_t offset) const {
403 if (offset >= code.size())
404 return 1;
405 OpCode op = static_cast<OpCode>(code[offset]);
406 if (isOpCodeVariableLength(op)) {
407 // 变长指令按 opcode 分派计算实际长度
408 switch (op) {
409 case OpCode::OP_CLOSURE: {
410 // OP_CLOSURE(1) + nameIdx(2) + upvalueCount(1) + upvalueCount*2
411 if (offset + 3 < code.size()) {
412 uint8_t upvalueCount = code[offset + 3];
413 return 4 + static_cast<size_t>(upvalueCount) * 2;
414 }
415 return 1; // 截断，返回 1 避免无限循环
416 }
417 default:
418 // 元数据标记为变长但未实现实际长度计算的指令——返回 baseSize 兜底
419 return instructionSize(op);
420 }
421 }
422 return instructionSize(op);
423 }
