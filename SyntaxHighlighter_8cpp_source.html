68/// 重写：对单个文本块应用语法高亮格式。
69void SyntaxHighlighter::highlightBlock(const QString& text) {
70 // 状态编码:
71 // 0: 正常
72 // 1: 字符串体内
73 // 2: 块注释（向后兼容，深度=1）
74 // 100 + depth (depth >= 1): 嵌套块注释（支持 /* /* */ */ 嵌套）
75 // 200 + braceDepth (braceDepth >= 1): 字符串插值表达式内（支持跨行插值）
76 // 300 + braceDepth*10 + blockCommentDepth: 插值表达式内且处于块注释中
77 // （BUG-SH-1 fix：原编码 blockCommentDepth > 0 优先于 interpBraceDepth > 0，
78 // 导致插值内块注释跨行时 interpBraceDepth 上下文丢失）
79 // 400 + braceDepth (braceDepth >= 1): 插值表达式内的嵌套字符串体内（跨行）
80 // （BUG-SH-2 fix：原实现嵌套字符串为单行扫描，未闭合时下一行错误处理）
81 int pos = 0;
82 int len = text.length();
83
84 int prevState = previousBlockState();
85 if (prevState < 0)
86 prevState = 0;
87
88 bool inString = false;
89 bool inNestedString = false; // BUG-SH-2: 插值内嵌套字符串跨行
90 int blockCommentDepth = 0;
91 int interpBraceDepth = 0;
92 // 解码顺序：高范围优先（400 > 300 > 200 > 100），避免误匹配
93 if (prevState == 1)
94 inString = true;
95 else if (prevState == 2)
96 blockCommentDepth = 1; // 向后兼容
97 else if (prevState >= 400) {
98 inNestedString = true;
99 interpBraceDepth = prevState - 400;
100 } else if (prevState >= 300) {
101 // BUG-SH-1 fix: 插值内块注释组合状态
102 interpBraceDepth = (prevState - 300) / 10;
103 blockCommentDepth = (prevState - 300) % 10;
104 } else if (prevState >= 200)
105 interpBraceDepth = prevState - 200;
106 else if (prevState >= 100)
107 blockCommentDepth = prevState - 100;
108
109 // PERF-22 fix: 用 per-character 掩码数组标记字符串/注释范围，
110 // 将范围检查从 O(ranges) 线性扫描降为 O(1) 数组查找
111 // Perf-Finding3: thread_local 复用底层数组容量，避免每次按键的堆分配。
112 static thread_local std::vector<char> mask;
113 mask.clear();
114 if (len > 0)
115 mask.resize(len, 0);
116
117 QList<QPair<int, int>> stringRanges;
118 QList<QPair<int, int>> commentRanges;
119
120 int stringStart = inString ? 0 : -1;
121 int commentStart = (blockCommentDepth > 0) ? 0 : -1;
122 // BUG-SH-2 fix: 嵌套字符串跨行时，起始位置为行首（0）
123 int nestedStringStart = inNestedString ? 0 : -1;
124
125 // ---- 第一遍：扫描字符串/注释，填充 mask + ranges ----
126 while (pos < len) {
127 // ---- 块注释内（depth > 0）：跟踪嵌套 /* */ ----
128 if (blockCommentDepth > 0) {
129 mask[pos] = 1;
130 if (pos + 1 < len && text[pos] == '/' && text[pos + 1] == '*') {
131 mask[pos + 1] = 1;
132 pos += 2;
133 blockCommentDepth++;
134 continue;
135 }
136 if (pos + 1 < len && text[pos] == '*' && text[pos + 1] == '/') {
137 mask[pos + 1] = 1;
138 pos += 2;
139 blockCommentDepth--;
140 if (blockCommentDepth == 0) {
141 commentRanges.append({commentStart, pos - commentStart});
142 commentStart = -1;
143 }
144 continue;
145 }
146 pos++;
147 continue;
148 }
149
150 // ---- 字符串插值表达式内（braceDepth > 0）：扫描为代码 ----
151 if (interpBraceDepth > 0) {
152 // BUG-SH-2 fix: 处理跨行嵌套字符串（从上一行延续的未闭合嵌套字符串）
153 if (inNestedString) {
154 while (pos < len) {
155 mask[pos] = 1;
156 if (text[pos] == '\\' && pos + 1 < len) {
157 mask[pos + 1] = 1;
158 pos += 2;
159 continue;
160 }
161 if (text[pos] == '"') {
162 pos++;
163 inNestedString = false;
164 break;
165 }
166 pos++;
167 }
168 stringRanges.append({nestedStringStart, pos - nestedStringStart});
169 continue;
170 }
171 // 嵌套字符串：扫描至闭合 " 或行末
172 if (text[pos] == '"') {
173 nestedStringStart = pos;
174 mask[pos] = 1;
175 pos++;
176 bool closed = false;
177 while (pos < len) {
178 mask[pos] = 1;
179 if (text[pos] == '\\' && pos + 1 < len) {
180 mask[pos + 1] = 1;
181 pos += 2;
182 continue;
183 }
184 if (text[pos] == '"') {
185 pos++;
186 closed = true;
187 break;
188 }
189 pos++;
190 }
191 // BUG-SH-2 fix: 未闭合则标记跨行状态，下一行继续扫描
192 if (!closed) {
193 inNestedString = true;
194 }
195 stringRanges.append({nestedStringStart, pos - nestedStringStart});
196 continue;
197 }
198 // 嵌套块注释
199 if (pos + 1 < len && text[pos] == '/' && text[pos + 1] == '*') {
200 commentStart = pos;
201 mask[pos] = 1;
202 mask[pos + 1] = 1;
203 pos += 2;
204 blockCommentDepth = 1;
205 continue;
206 }
207 if (text[pos] == '{') {
208 interpBraceDepth++;
209 pos++;
210 continue;
211 }
212 if (text[pos] == '}') {
213 interpBraceDepth--;
214 if (interpBraceDepth == 0) {
215 mask[pos] = 1; // } 标记为字符串颜色（插值分隔符）
216 pos++;
217 inString = true;
218 stringStart = pos;
219 } else {
220 pos++;
221 }
222 continue;
223 }
224 pos++;
225 continue;
226 }
227
228 // ---- 字符串体内 ----
229 if (inString) {
230 if (text[pos] == '\\' && pos + 1 < len) {
231 mask[pos] = 1;
232 pos++;
233 mask[pos] = 1;
234 pos++;
235 continue;
236 }
237 if (text[pos] == '"') {
238 mask[pos] = 1;
239 pos++;
240 stringRanges.append({stringStart, pos - stringStart});
241 inString = false;
242 stringStart = -1;
243 continue;
244 }
245 // 识别字符串插值 {expr}，进入插值模式（支持跨行）
246 if (text[pos] == '{') {
247 stringRanges.append({stringStart, pos - stringStart});
248 mask[pos] = 1; // { 标记为字符串颜色
249 pos++;
250 interpBraceDepth = 1;
251 continue;
252 }
253 mask[pos] = 1;
254 pos++;
255 continue;
256 }
257
258 // ---- 正常模式 ----
259 if (pos + 1 < len && text[pos] == '/' && text[pos + 1] == '/') {
260 commentRanges.append({pos, len - pos});
261 for (int i = pos; i < len; ++i)
262 mask[i] = 1;
263 break;
264 }
265
266 if (pos + 1 < len && text[pos] == '/' && text[pos + 1] == '*') {
267 commentStart = pos;
268 blockCommentDepth = 1;
269 mask[pos] = 1;
270 mask[pos + 1] = 1;
271 pos += 2;
272 continue;
273 }
274
275 if (text[pos] == '"') {
276 stringStart = pos;
277 inString = true;
278 mask[pos] = 1;
279 pos++;
280 continue;
281 }
282
283 pos++;
284 }
285
286 if (inString && stringStart >= 0) {
287 stringRanges.append({stringStart, len - stringStart});
288 }
289 if (blockCommentDepth > 0 && commentStart >= 0) {
290 commentRanges.append({commentStart, len - commentStart});
291 }
292
293 // 应用字符串/注释格式
294 for (const auto& range : stringRanges) {
295 setFormat(range.first, range.second, stringFormat_);
296 }
297 for (const auto& range : commentRanges) {
298 setFormat(range.first, range.second, commentFormat_);
299 }
300
301 // ---- 第二遍：单遍扫描识别关键字/数字/运算符，完全消除 regex globalMatch ----
302 // PERF-22 fix: 原实现用 2 次 QRegularExpression::globalMatch（数字正则 + 运算符正则），
303 // 每次匹配都需 regex 引擎回溯 + 对每个匹配检查 mask。
304 // 改造后：单遍字符扫描，按首字符分派到 keyword/number/operator 分支，
305 // 总复杂度从 O(n × regex回溯) 降为 O(n)。
306 pos = 0;
307 while (pos < len) {
308 // 跳过字符串/注释内的字符（mask 标记）
309 if (mask[pos]) {
310 pos++;
311 continue;
312 }
313
314 QChar c = text[pos];
315
316 // ---- 标识符/关键字 ----
317 // 首字符：字母或下划线；后续：字母/数字/下划线
318 if (c.isLetter() || c == '_') {
319 int start = pos;
320 while (pos < len && (text[pos].isLetterOrNumber() || text[pos] == '_')) {
321 pos++;
322 }
323 int wordLen = pos - start;
324 // BUG-SH-4 fix: 移除 wordLen <= 20 上限检查。原检查跳过过长标识符的
325 // keywordSet_ 查找，但 QSet::contains 对任意长度都是 O(1)+平均 O(L) 哈希，
326 // 无性能问题。上限 20 会漏掉理论上的长关键字（虽当前无，但为防御性修复）。
327 if (!keywordSet_.isEmpty() && keywordSet_.contains(text.mid(start, wordLen))) {
328 setFormat(start, wordLen, keywordFormat_);
329 }
330 continue;
331 }
332
333 // ---- 数字字面量 ----
334 // 支持：整数(123)、浮点(1.23)、科学计数(1e5/1.23e-5)、
335 // 前导点(.123)、尾点(123.)
336 // 判定条件：首字符是数字，或者首字符是 '.' 且下一个字符是数字
337 // BUG-SH-3 fix: 识别 0x/0b/0o 前缀。MiniLang 不支持这些前缀，将整个
338 // 非法字面量高亮为错误格式（红色），避免被当作普通数字 0 处理后剩余
339 // 字符（如 xFF）被误识别为标识符。
340 if (c == '0' && pos + 1 < len &&
341 (text[pos + 1] == 'x' || text[pos + 1] == 'X' || text[pos + 1] == 'b' || text[pos + 1] == 'B' ||
342 text[pos + 1] == 'o' || text[pos + 1] == 'O')) {
343 int start = pos;
344 pos += 2;
345 while (pos < len && (text[pos].isLetterOrNumber() || text[pos] == '_'))
346 ++pos;
347 QTextCharFormat errFmt;
348 errFmt.setForeground(QColor(0xdc, 0x32, 0x2f));
349 setFormat(start, pos - start, errFmt);
350 continue;
351 }
352 if (c.isDigit() || (c == '.' && pos + 1 < len && text[pos + 1].isDigit())) {
353 int start = pos;
354 bool hasDot = false;
355
356 // 整数部分（如果有，前导点场景无整数部分）
357 while (pos < len && text[pos].isDigit()) {
358 pos++;
359 }
360
361 // 小数点 + 小数部分
362 if (pos < len && text[pos] == '.' && !hasDot) {
363 hasDot = true;
364 pos++;
365 while (pos < len && text[pos].isDigit()) {
366 pos++;
367 }
368 }
369
370 // 指数部分 e/E[+-]?digits
371 if (pos < len && (text[pos] == 'e' || text[pos] == 'E')) {
372 int expPos = pos;
373 pos++;
374 if (pos < len && (text[pos] == '+' || text[pos] == '-')) {
375 pos++;
376 }
377 if (pos < len && text[pos].isDigit()) {
378 while (pos < len && text[pos].isDigit()) {
379 pos++;
380 }
381 } else {
382 // 回退：'e' 后无数字，不是合法指数，回退到 e 前
383 pos = expPos;
384 }
385 }
386
387 setFormat(start, pos - start, numberFormat_);
388 continue;
389 }
390
391 // ---- 运算符/分隔符 ----
392 // 两字符运算符优先匹配：== != <= >=
393 // 单字符运算符：+ - * / % < > = ( ) { } [ ] ; , :
394 if (pos + 1 < len) {
395 QChar c2 = text[pos + 1];
396 if ((c == '=' && c2 == '=') || (c == '!' && c2 == '=') || (c == '<' && c2 == '=') ||
397 (c == '>' && c2 == '=')) {
398 setFormat(pos, 2, operatorFormat_);
399 pos += 2;
400 continue;
401 }
402 }
403 if (c == '+' || c == '-' || c == '*' || c == '/' || c == '%' || c == '<' || c == '>' || c == '=' || c == '(' ||
404 c == ')' || c == '{' || c == '}' || c == '[' || c == ']' || c == ';' || c == ',' || c == ':') {
405 setFormat(pos, 1, operatorFormat_);
406 pos++;
407 continue;
408 }
409
410 // 其他字符（空白等），跳过
411 pos++;
412 }
413
414 // 设置块状态（支持多行字符串、嵌套块注释、跨行插值、插值内嵌套字符串/块注释）
415 // BUG-SH-1 fix: 插值内块注释跨行时用 300+braceDepth*10+blockCommentDepth
416 // 编码同时保留两个上下文。原编码 blockCommentDepth > 0 优先导致
417 // interpBraceDepth 丢失，下一行块注释结束后无法回到插值模式。
418 // BUG-SH-2 fix: 插值内嵌套字符串跨行用 400+braceDepth 编码。
419 // 原实现嵌套字符串仅单行扫描，未闭合时下一行按普通代码处理。
420 // 注：inString 在 interpBraceDepth > 0 期间恒为 true（进入插值时未清除，
421 // 退出插值时显式重置），无需单独编码，200+braceDepth 即可覆盖。
422 int newState = 0;
423 if (inNestedString)
424 newState = 400 + interpBraceDepth;
425 else if (blockCommentDepth > 0 && interpBraceDepth > 0) {
426 // R54-17 fix: blockCommentDepth 占个位（mod 10），达 10 时溢出到 interpBraceDepth
427 // 导致状态损坏。钳制到 9——10 层嵌套块注释极少见，钳制仅影响高亮不影响语义。
428 int clampedDepth = std::min(blockCommentDepth, 9);
429 newState = 300 + interpBraceDepth * 10 + clampedDepth;
430 } else if (blockCommentDepth > 0)
431 newState = 100 + blockCommentDepth;
432 else if (interpBraceDepth > 0)
433 newState = 200 + interpBraceDepth;
434 else if (inString)
435 newState = 1;
436 setCurrentBlockState(newState);
