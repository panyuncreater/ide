84/// 构造面板：组装顶部子页切换（实时调用栈 / 教学场景库）与 QStackedWidget，
85/// 构建两个子页，配置 2s 自动刷新定时器（安全网），并填充场景库列表。
86CallStackPanel::CallStackPanel(QWidget* parent) : QWidget(parent) {
87 auto* outer = new QVBoxLayout(this);
88 outer->setContentsMargins(4, 4, 4, 4);
89 outer->setSpacing(4);
90
91 // 顶部子页切换
92 auto* pageBar = new QHBoxLayout;
93 pageLiveBtn_ = new QPushButton(tr("实时调用栈"));
94 pageLibraryBtn_ = new QPushButton(tr("教学场景库"));
95 pageLiveBtn_->setCheckable(true);
96 pageLibraryBtn_->setCheckable(true);
97 pageLiveBtn_->setChecked(true);
98 pageBar->addWidget(pageLiveBtn_);
99 pageBar->addWidget(pageLibraryBtn_);
100 pageBar->addStretch();
101 outer->addLayout(pageBar);
102
103 stack_ = new QStackedWidget;
104 auto* livePage = new QWidget;
105 auto* libraryPage = new QWidget;
106 buildLivePage(livePage);
107 buildLibraryPage(libraryPage);
108 stack_->addWidget(livePage);
109 stack_->addWidget(libraryPage);
110 outer->addWidget(stack_, 1);
111
112 connect(pageLiveBtn_, &QPushButton::clicked, [this]() {
113 stack_->setCurrentIndex(0);
114 pageLibraryBtn_->setChecked(false);
115 PanelAnimator::slideInWidget(stack_->currentWidget());
116 });
117 connect(pageLibraryBtn_, &QPushButton::clicked, [this]() {
118 stack_->setCurrentIndex(1);
119 pageLiveBtn_->setChecked(false);
120 // 切到教学场景库时确保有选中项 —— 首次进入若 scenarioList_ 无选中，
121 // 显式调用 showScenario(0) 让详情区立即有内容。
122 if (scenarioList_->count() > 0 && scenarioList_->currentRow() < 0) {
123 scenarioList_->setCurrentRow(0);
124 }
125 if (currentScenarioIdx_ < 0 && scenarioList_->count() > 0) {
126 showScenario(0);
127 }
128 PanelAnimator::slideInWidget(stack_->currentWidget());
129 });
130
131 autoTimer_ = new QTimer(this);
132 // OPT-1: 500ms→2000ms。状态变更由 vmStateChanged 监听器即时触发刷新，
133 // QTimer 降级为安全网（覆盖监听器未触达的边角场景），2s 间隔足以兜底且省 CPU。
134 autoTimer_->setInterval(2000);
135 connect(autoTimer_, &QTimer::timeout, this, [this]() {
136 // PERF: 安全网轮询纪元门控——VM 状态纪元未变时跳过 refreshLive
137 // （clear/rebuild + getVmStack/getVmGlobals 状态快照全量深拷贝）。
138 // 状态变更由 OPT-1 vmStateChanged 观察者即时推送，本定时器仅兜底
139 // 监听器未触达的边角场景；用户手动刷新（onRefresh）不受门控影响。
140 if (controller_) {
141 const std::uint64_t epoch = controller_->vmStateEpoch();
142 if (epoch == lastAutoRefreshEpoch_)
143 return;
144 lastAutoRefreshEpoch_ = epoch;
145 }
146 onRefresh();
147 });
148
149 populateScenarios();
150}

153/// 换绑前先反注册旧监听器，避免 controller 持有悬垂回调。
154void CallStackPanel::setController(IdeController* controller) {
155 if (controller_ == controller)
156 return;

163 controller_->addVmStateChangedListener(this, [this] { onVmStateChanged(); });
164 }
165}

273/// 抓取并渲染实时调用栈（栈帧数据来源与展示逻辑）：
274/// 1. 优先判断活跃引擎——Interpreter 调试（isRunning && isDebugRun）走
275/// getDebugCallStack，VM 单步走 getVmCallStack（标记当前后端为 StackVM/RegisterVM）；
276/// 2. 并发防护：调试 resume 或 VM 运行中（worker 可能并发改 frames_）时跳过
277/// 快照获取，避免数据竞争 UB，仅提示「暂停后刷新」；
278/// 3. 将帧序列倒序（栈顶在上）填入 QTreeWidget，每个帧节点再展开其本地变量
279/// 子节点（值/类型经 try/catch 包裹 toString/typeName 防异常）。
280/// 该树形结构即运行时调用栈帧来源：函数名/调用深度/当前行号/局部变量。
281/// R121: 重建期间设置 inRefresh_ 守卫避免 currentItemChanged 误触发 setSelectedFrame。
282/// 重建后尝试恢复选中之前选中的帧（通过 selectedFrame_ 索引匹配 depth 列）。
283void CallStackPanel::refreshLive() {
284 // R121: 守卫避免 stackTree_->clear() 触发 currentItemChanged → onFrameSelected
285 // 在重建期间误调用 controller_->setSelectedFrame 导致循环刷新。
286 inRefresh_ = true;
287 stackTree_->clear();
288 if (!controller_) {
289 liveStatusLabel_->setText(tr("状态：未绑定 controller"));
290 inRefresh_ = false;
291 return;
292 }
293
294 std::vector<CallStackEntry> entries;
295 QString modeLabel;
296 // AUDIT-P1 fix: 优先判断当前活跃引擎（Interpreter 调试 > VM 单步），
297 // 避免 isVmInitialized 一次性永久 true 后 Interpreter 调试仍走 VM 分支显示陈旧数据。
298 // 同时增加 isDebugPaused() 检查，避免调试 resume 期间 worker 活跃时并发访问解释器
299 // 内部 unordered_map/vector 导致 UB（数据竞争）。
300 if (controller_->isRunning() && controller_->isDebugRun()) {
301 if (!controller_->isDebugPaused()) {
302 // 调试运行中（resume 后 worker 活跃），跳过快照获取避免数据竞争
303 liveStatusLabel_->setText(tr("状态：调试运行中（暂停后刷新）"));
304 frameDetail_->clear();
305 inRefresh_ = false;
306 return;
307 }
308 entries = controller_->getDebugCallStack();
309 modeLabel = tr("Interpreter (debug)");
310 } else if (controller_->isVmInitialized()) {
311 // AUDIT-P1-CORRECT fix: VM RUN 模式期间 worker 可能 push_back/pop_back frames_，
312 // 并发遍历触发 UB（迭代器失效）。添加 isVmRunning() 守卫，
313 // 对齐 Interpreter 调试路径的 isDebugPaused() 守卫。
314 if (controller_->isVmRunning()) {
315 liveStatusLabel_->setText(tr("状态：VM 运行中（暂停后刷新）"));
316 frameDetail_->clear();
317 inRefresh_ = false;
318 return;
319 }
320 entries = controller_->getVmCallStack();
321 modeLabel = controller_->getUseRegisterVM() ? tr("RegisterVM") : tr("StackVM");
322 } else {
323 liveStatusLabel_->setText(tr("状态：未运行（启动调试或 VM 单步以查看调用栈）"));
324 frameDetail_->clear();
325 inRefresh_ = false;
326 return;
327 }
328
329 // R121: 状态标签追加选中帧提示（selectedFrame >= 0 时显示）
330 int selFrame = controller_->getSelectedFrame();
331 QString selHint;
332 if (selFrame >= 0 && selFrame < static_cast<int>(entries.size())) {
333 selHint = tr(" | 已选帧: %1 (%2)").arg(selFrame).arg(QString::fromUtf8(entries[selFrame].functionName.c_str()));
334 }
335 liveStatusLabel_->setText(tr("状态：%1 | 帧数：%2%3").arg(modeLabel).arg(entries.size()).arg(selHint));
336
337 if (entries.empty()) {
338 frameDetail_->setPlainText(tr("（调用栈为空）"));
339 inRefresh_ = false;
340 return;
341 }
342
343 QTreeWidgetItem* itemToSelect = nullptr;
344 for (int i = static_cast<int>(entries.size()) - 1; i >= 0; --i) {
345 const auto& e = entries[i];
346 auto* frame = new QTreeWidgetItem();
347 frame->setText(0, QString::fromUtf8(e.functionName.c_str()));
348 frame->setText(1, QString::number(e.line));
349 frame->setText(2, QString::number(e.depth));
350 // 子节点：本地变量
351 for (const auto& [name, val] : e.locals) {
352 auto* localItem = new QTreeWidgetItem(frame);
353 localItem->setText(0, QString::fromUtf8(name.c_str()));
354 // AUDIT-P1 fix: toString() 可能抛异常（NaN-boxing 解码失败或数据竞争残留），
355 // 与 VmStackPanel 保持一致用 try/catch 包裹避免崩溃。
356 std::string valStr, typeStr;
357 try {
358 valStr = val.toString();
359 } catch (...) {
360 valStr = "<error>";
361 }
362 try {
363 typeStr = val.typeName();
364 } catch (...) {
365 typeStr = "<error>";
366 }
367 localItem->setText(1, QString::fromUtf8(valStr.c_str()));
368 localItem->setText(2, QString::fromUtf8(typeStr.c_str()));
369 }
370 stackTree_->addTopLevelItem(frame);
371 // R121: 恢复选中之前选中的帧
372 if (selFrame >= 0 && e.depth == selFrame) {
373 itemToSelect = frame;
374 }
375 }
376 stackTree_->resizeColumnToContents(0);
377 // R121: 重建完成后恢复选中（不触发 setSelectedFrame 循环，因 inRefresh_ 已 false
378 // 但 onFrameSelected 调用 setSelectedFrame 不会重复刷新——selectedFrame 值未变）
379 if (itemToSelect) {
380 stackTree_->setCurrentItem(itemToSelect);
381 }
382 inRefresh_ = false;
383}

385/// 选中栈树节点时渲染详情：顶层帧节点显示函数名/深度/当前行/局部变量数；
386/// 子节点（变量）显示类型与值，写入详情浏览器。
387/// R121: 选中顶层帧节点时通知 controller 切换选中帧（触发 VariableInspector/WatchPanel 联动刷新）。
388/// inRefresh_ 守卫避免 refreshLive 重建树期间误触发。
389void CallStackPanel::onFrameSelected() {
390 auto* cur = stackTree_->currentItem();
391 if (!cur) {
392 frameDetail_->clear();
393 return;
394 }
395
396 // 顶层节点为帧；子节点为变量
397 if (cur->parent() == nullptr) {
398 QString name = cur->text(0);
399 QString line = cur->text(1);
400 QString depth = cur->text(2);
401 // OPT-2 fix: name（函数名）未转义——若函数名含 < > & 等字符会破坏 HTML 结构。
402 // 对齐变量分支 varVal 的 toHtmlEscaped 处理。
403 QString html = QString("<h3>栈帧: %1</h3>"
404 "<p><b>调用深度:</b> %2</p>"
405 "<p><b>当前行:</b> %3</p>"
406 "<p><b>本地变量数:</b> %4</p>")
407 .arg(name.toHtmlEscaped())
408 .arg(depth)
409 .arg(line)
410 .arg(cur->childCount());
411 frameDetail_->setHtml(html);
412
413 // R121: 通知 controller 切换选中帧（VariableInspector / WatchPanel 会自动刷新）。
414 // inRefresh_ 守卫避免 refreshLive 重建树时的 currentItemChanged 误触发。
415 if (controller_ && !inRefresh_) {
416 bool ok = false;
417 int depthInt = depth.toInt(&ok);
418 if (ok && depthInt >= 0) {
419 controller_->setSelectedFrame(depthInt);
420 }
421 }
422 } else {
423 // 选中变量子节点
424 QString varName = cur->text(0);
425 QString varVal = cur->text(1);
426 QString varType = cur->text(2);
427 // OPT-2 fix: varName/varType 未转义——变量名或类型名含特殊字符会破坏 HTML。
428 // 仅 varVal 调了 toHtmlEscaped，此处补齐 varName/varType。
429 QString html = QString("<h3>变量: %1</h3>"
430 "<p><b>类型:</b> %2</p>"
431 "<p><b>值:</b> %3</p>")
432 .arg(varName.toHtmlEscaped())
433 .arg(varType.toHtmlEscaped())
434 .arg(varVal.toHtmlEscaped());
435 frameDetail_->setHtml(html);
436 }
437}
