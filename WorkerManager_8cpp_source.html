376/// 强制终止 worker 线程（quit+wait），用于 stopForClose 超时后的兜底清理。
377void WorkerManager::forceStop() {
378 if (workerThread_) {
379 // A5 fix: 协作式取消 — 通过 debugger_->stop() 设置 stopped_ 标志，
380 // 解释器/VM 在 checkBreak() 检测到后抛出 DebugStopException 正常退出。
381 // 不再使用 QThread::terminate()（会导致 UB：互斥锁未释放、堆损坏、悬垂信号）。
382 debugger_->stop();

384 // Worker 已正常停止，安全释放
385 // AUDIT-P1 fix: 与 stopForClose/析构路径对齐，reset 前断开 finished 信号，
386 // 避免已投递的 QueuedConnection lambda 在 forceStop 返回后触发悬垂 this。
387 workerThread_->disconnect(this);
388 worker_.reset();
389 workerThread_.reset();
390 } else {
391 // 崩溃修复: Worker 未在 5 秒内停止（死循环未检查 stopped_）。
392 // 必须 terminate + join，否则 forceStop 返回后 isRunning_=false，
393 // 后续 closeEvent 会 accept 并销毁 Ide，而 worker 线程仍通过
394 // QueuedConnection 回写已析构的 IdeController/WorkerManager 成员 →
395 // use-after-free → 栈损坏。
396 // terminate 风险（pauseMutex_ 可能被持有）：仅当 worker 卡在 checkBreak()
397 // 才会持有该锁，但卡在 checkBreak 的 worker 会检测到 stopped_ 并正常退出，
398 // 因此到达此分支的 worker 通常在执行用户代码（未持有 DebugController 锁）。
399 LOG_ERROR("Worker 未在 5 秒内响应取消请求，回退 terminate() + join（确保 close 路径安全）", "IDE");

401 // AUDIT-LIFECYCLE fix: 原 wait() 无超时。Windows 上 TerminateThread 立即生效，
402 // 但 Unix 上 pthread_cancel 为延迟取消（PTHREAD_CANCEL_DEFERRED），
403 // 若 worker 卡在无取消点的纯 CPU 循环则永不生效，wait() 永久阻塞，
404 // std::_Exit(0) 无法触发。加 2 秒超时，超时后直接 _Exit 退出进程。

406 LOG_ERROR("Worker terminate 后 2 秒仍未退出，强制 _Exit", "IDE");
407 Logger::instance().flush();

409 // （最近工作区、窗口布局、引导标记等）。显式 sync 确保持久化。
410 QSettings("MiniLang", "MiniLang IDE").sync();
411 std::_Exit(0);
412 }

414 workerThread_.reset();

416 // (1) Value::operator= 的 release→assign 窗口被中断 → box_ 指向已释放 RefCounted
417 // → restoreReplState 析构时 double-release UAF
418 // (2) unordered_map rehash / vector realloc 中途被中断 → 容器半重组
419 // → restoreReplState 析构时 double-free 或野指针解引用
420 // (3) CRT 堆 critical section 被死线程永久持有 → 后续堆操作崩溃
421 // 这三种损坏状态都会被 NaNBox::tag() 合法化为 NaN float（0xFFFFFFFFFFFFFFFF），
422 // 掩盖底层 UB。继续执行 restoreReplState/setupMainCallbacks 会立即触发崩溃。
423 //
424 // 修复策略：terminate 是不可恢复的平台级故障，立即调用 std::_Exit(0) 退出进程，
425 // 跳过所有析构（损坏状态由操作系统回收）。这是关闭场景，进程即将退出，
426 // 跳过析构是安全的——比尝试析构损坏状态导致崩溃要好得多。

430 // （最近工作区、窗口布局、引导标记等）。显式 sync 确保持久化。
431 QSettings("MiniLang", "MiniLang IDE").sync();
432 std::_Exit(0);
433 }
434 }

436 isRunning_ = false;
437 isDebugRun_ = false;
438 interpreter_->setDebugMode(false);

444/// worker 线程结束后的清理：删除 worker/thread、恢复回调与 REPL 状态。
445void WorkerManager::cleanupWorker() {
446 isRunning_ = false;
447 isDebugRun_ = false;
448 interpreter_->setDebugMode(false);
449 interpreter_->restoreReplState();
450 debugger_->reset();
451
452 // R54-4 fix: 统一三处销毁顺序为"先 worker 后 thread"，与 ~WorkerManager 对齐。
453 // 原顺序先 reset workerThread_ 再 reset worker_，worker 析构时 Qt 内部可能访问
454 // worker->thread()，此时 QThread 对象已被销毁，指针悬垂 → 潜在 UAF。
455 // 正确顺序：quit+wait（停止线程执行）→ reset worker_（QThread 对象仍存活，
456 // thread() 有效）→ reset workerThread_（销毁 QThread 对象）。
457 // R54-5 fix: wait() 增加超时，与 ~WorkerManager/stopForClose/forceStop 一致，
458 // 避免线程终止异常时永久阻塞主线程导致 UI 冻结。
459 if (workerThread_) {
460 workerThread_->quit();
461 if (!workerThread_->wait(5000)) {
462 LOG_ERROR("cleanupWorker: worker 线程未在 5 秒内退出", "IDE");
463 }
464 }
465 worker_.reset();
466 if (workerThread_)
467 workerThread_.reset();
468
469 // 恢复主线程输出+输入回调（worker 的回调 lambda 捕获了已删除的 worker this 指针）
470 setupMainCallbacks();
471}
