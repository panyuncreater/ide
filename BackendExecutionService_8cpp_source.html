142BackendExecResult BackendExecutionService::execute(const std::string& src, BackendType backend,
143 BackendExecDetail* detail) {
144 BackendExecResult r;
145 r.backend = backend;
146
147 // ------------------------------------------------------------
148 // 阶段 1：Lexer
149 // ------------------------------------------------------------
150 Lexer lex;
151 std::vector<Token> tokens;
152 try {
153 tokens = lex.scan(src);
154 } catch (const std::exception& e) {
155 r.success = false;
156 r.errorPrefix = "词法错误";
157 r.errorMsg = e.what();
158 r.diagnostics = collectDiagnostics(lex.getDiagnostics());
159 return r;
160 }
161 if (lex.getDiagnostics().hasErrors()) {
162 r.success = false;
163 r.errorPrefix = "词法错误";
164 r.errorMsg = joinErrorDiagnostics(lex.getDiagnostics());
165 r.diagnostics = collectDiagnostics(lex.getDiagnostics());
166 return r;
167 }
168
169 // ------------------------------------------------------------
170 // 阶段 2：Parser
171 // ------------------------------------------------------------
172 Parser parser;
173 std::unique_ptr<Block> ast;
174 try {
175 ast = parser.parse(tokens);
176 } catch (const std::exception& e) {
177 r.success = false;
178 r.errorPrefix = "语法错误";
179 r.errorMsg = e.what();
180 r.diagnostics = collectDiagnostics(parser.getDiagnostics());
181 return r;
182 }
183 if (!ast || parser.hasErrors()) {
184 r.success = false;
185 r.errorPrefix = "语法错误";
186 r.errorMsg = ast ? joinErrorDiagnostics(parser.getDiagnostics()) : "AST 为空";
187 r.diagnostics = collectDiagnostics(parser.getDiagnostics());
188 return r;
189 }
190
191 // ------------------------------------------------------------
192 // 阶段 3：根据后端类型分发执行
193 // ------------------------------------------------------------
194 auto t0 = std::chrono::steady_clock::now();
195 std::string out;
196
197 // PERF-DETAIL: 单遍 detail 收集脚手架。detail == nullptr 时全部零开销
198 // （updateDetailPeak 直通返回，stepCallback 不安装，无 trackedCount 查询）。
199 const size_t detailBaseTracked = detail ? GcManager::instance().trackedCount() : 0;
200 size_t detailPeakTracked = 0;
201 auto updateDetailPeak = [&]() {
202 if (!detail)
203 return;
204 size_t cur = GcManager::instance().trackedCount();
205 size_t delta = (cur > detailBaseTracked) ? (cur - detailBaseTracked) : 0;
206 if (delta > detailPeakTracked)
207 detailPeakTracked = delta;
208 };
209
210 if (backend == BackendType::Interpreter) {
211 // Interpreter 直接执行 AST，无字节码
212 Interpreter interp;
213 // PERF-DETAIL: 输出回调顺带采样 tracked 峰值（对齐原两遍 detail 路径）
214 interp.setOutputCallback([&](const std::string& s) {
215 out += s;
216 updateDetailPeak();
217 });
218
219 bool hasRuntimeError = false;
220 std::string runtimeErr;
221 try {
222 interp.execute(*ast);
223 } catch (const RuntimeError& e) {
224 hasRuntimeError = true;
225 runtimeErr = e.what();
226 } catch (const std::exception& e) {
227 hasRuntimeError = true;
228 runtimeErr = e.what();
229 }
230 auto t1 = std::chrono::steady_clock::now();
231 r.elapsedUs = std::chrono::duration_cast<std::chrono::microseconds>(t1 - t0).count();
232 r.elapsedMs = r.elapsedUs / 1000; // AUDIT-R2 P2-2 fix: 由微秒派生，保留兼容
233 r.output = out;
234 r.instrCount = -1; // Interpreter 无字节码
235
236 if (hasRuntimeError) {
237 r.success = false;
238 r.errorPrefix = "运行时错误";
239 r.errorMsg = runtimeErr;
240 } else {
241 r.success = true;
242 }
243 if (detail)
244 detail->peakTrackedCount = detailPeakTracked;
245 return r;
246 }
247
248 // StackVM / RegisterVM 路径：先编译
249 Compiler compiler;
250 if (backend == BackendType::StackVM_IR) {
251 compiler.setUseIR(true);
252 } else { // RegisterVM_IR
253 compiler.setUseRegisterVM(true);
254 }
255
256 if (backend == BackendType::StackVM_IR) {
257 CompileResult cr;
258 try {
259 cr = compiler.compile(*ast);
260 } catch (const std::exception& e) {
261 r.success = false;
262 r.errorPrefix = "编译错误";
263 r.errorMsg = e.what();
264 r.diagnostics = collectDiagnostics(compiler.getDiagnostics());
265 return r;
266 }
267 if (compiler.getDiagnostics().hasErrors()) {
268 r.success = false;
269 r.errorPrefix = "编译错误";
270 r.errorMsg = compiler.getLastError();
271 if (r.errorMsg.empty())
272 r.errorMsg = joinErrorDiagnostics(compiler.getDiagnostics());
273 r.diagnostics = collectDiagnostics(compiler.getDiagnostics());
274 return r;
275 }
276 r.instrCount = static_cast<int64_t>(cr.mainChunk.code.size());
277 // PERF-DETAIL: 生成器 chunk 提取（原在第二遍执行前完成）
278 if (detail)
279 detail->generatorChunks = extractStackGeneratorChunks(cr.functionChunks);
280
281 VM vm;
282 vm.setOutputCallback([&](const std::string& s) { out += s; });
283 // AUDIT-R2 P2-3 fix: 重置计时基线——此前 t0 在编译前取，VM 路径耗时含
284 // 编译时间而 Interpreter 路径不含，三后端耗时对比口径不对称。
285 t0 = std::chrono::steady_clock::now();
286 // PERF-DETAIL: step 回调统计 opcode 频次 + 每 64 条采样 tracked 峰值
287 // （仅 detail 模式安装；无 detail 时 notifyStep 保持默认关闭零开销）
288 uint64_t detailStepCounter = 0;
289 if (detail) {
290 vm.setStepCallback([&](const VMStepInfo& info) {
291 detail->opcodeCounts[static_cast<uint8_t>(info.opcode)]++;
292 if ((++detailStepCounter & 63u) == 0)
293 updateDetailPeak();
294 });
295 vm.setStepCallbackEnabled(true);
296 }
297 // AUDIT-R2 P1-4 fix: 包裹执行异常——与 Interpreter 路径及 executeWithDetail
298 // 的 try/catch 处理对齐（后两者均已包裹），避免 std::bad_alloc 等异常
299 // 穿透到 GUI 槽函数导致 Qt 事件循环终止。
300 bool execThrew = false;
301 std::string execThrewMsg;
302 try {
303 vm.execute(cr);
304 } catch (const std::exception& e) {
305 execThrew = true;
306 execThrewMsg = e.what();
307 } catch (...) {
308 execThrew = true;
309 execThrewMsg = "未知异常";
310 }
311 vm.setStepCallbackEnabled(false);
312
313 auto t1 = std::chrono::steady_clock::now();
314 r.elapsedUs = std::chrono::duration_cast<std::chrono::microseconds>(t1 - t0).count();
315 r.elapsedMs = r.elapsedUs / 1000;
316 r.output = out;
317
318 if (execThrew) {
319 r.success = false;
320 r.errorPrefix = "运行时错误";
321 r.errorMsg = execThrewMsg;
322 } else if (vm.hasError()) {
323 r.success = false;
324 r.errorPrefix = "运行时错误";
325 r.errorMsg = vm.getLastError();
326 } else {
327 r.success = true;
328 }
329 if (detail)
330 detail->peakTrackedCount = detailPeakTracked;
331 r.diagnostics = collectDiagnostics(vm.getDiagnostics());
332 return r;
333 }
334
335 // RegisterVM_IR 路径
336 try {
337 compiler.compile(*ast);
338 } catch (const std::exception& e) {
339 r.success = false;
340 r.errorPrefix = "编译错误";
341 r.errorMsg = e.what();
342 r.diagnostics = collectDiagnostics(compiler.getDiagnostics());
343 return r;
344 }
345 if (compiler.getDiagnostics().hasErrors()) {
346 r.success = false;
347 r.errorPrefix = "编译错误";
348 r.errorMsg = compiler.getLastError();
349 if (r.errorMsg.empty())
350 r.errorMsg = joinErrorDiagnostics(compiler.getDiagnostics());
351 r.diagnostics = collectDiagnostics(compiler.getDiagnostics());
352 return r;
353 }
354
355 const auto& regResult = compiler.getLastRegisterResult();
356 r.instrCount = static_cast<int64_t>(regResult.mainChunk.code.size());
357 // PERF-DETAIL: 生成器 chunk 提取（原在第二遍执行前完成）
358 if (detail)
359 detail->generatorChunks = extractRegGeneratorChunks(regResult.functionChunks);
360
361 RegisterVM vm;
362 vm.setOutputCallback([&](const std::string& s) { out += s; });
363 // AUDIT-R2 P2-3 fix: 重置计时基线（同 StackVM 路径，不含编译时间）
364 t0 = std::chrono::steady_clock::now();
365 // PERF-DETAIL: step 回调统计 opcode 频次 + 每 64 条采样 tracked 峰值
366 uint64_t detailStepCounter = 0;
367 if (detail) {
368 vm.setStepCallback([&](const RegVMStepInfo& info) {
369 detail->opcodeCounts[static_cast<uint8_t>(info.opcode)]++;
370 if ((++detailStepCounter & 63u) == 0)
371 updateDetailPeak();
372 });
373 vm.setStepCallbackEnabled(true);
374 }
375 // AUDIT-R2 P1-4 fix: 包裹执行异常（同 StackVM 路径）
376 bool execThrew = false;
377 std::string execThrewMsg;
378 try {
379 vm.execute(regResult);
380 } catch (const std::exception& e) {
381 execThrew = true;
382 execThrewMsg = e.what();
383 } catch (...) {
384 execThrew = true;
385 execThrewMsg = "未知异常";
386 }
387 vm.setStepCallbackEnabled(false);
388
389 auto t1 = std::chrono::steady_clock::now();
390 r.elapsedUs = std::chrono::duration_cast<std::chrono::microseconds>(t1 - t0).count();
391 r.elapsedMs = r.elapsedUs / 1000;
392 r.output = out;
393
394 if (execThrew) {
395 r.success = false;
396 r.errorPrefix = "运行时错误";
397 r.errorMsg = execThrewMsg;
398 } else if (vm.hasError()) {
399 r.success = false;
400 r.errorPrefix = "运行时错误";
401 r.errorMsg = vm.getLastError();
402 } else {
403 r.success = true;
404 }
405 if (detail)
406 detail->peakTrackedCount = detailPeakTracked;
407 r.diagnostics = collectDiagnostics(vm.getDiagnostics());
408 return r;
409}

414// PERF-DETAIL（2026-10-04）: 现为 execute() 的薄封装——detail 数据由 execute()
415// 的可选出参在单次执行内联收集，不再重新执行第二遍。收集语义：
416// - generatorChunks：扫描 CompileResult/RegisterCompileResult 的 functionChunks，
417// 过滤 isGenerator=true 的 chunk 并提取 yieldCount（动态则标记 isDynamic=true）
418// - opcodeCounts：通过 VM/RegisterVM 的 stepCallback 机制累加每条指令的 opcode
419// - peakTrackedCount：GcManager 跟踪对象相对执行前基线的净增峰值
420// （AUDIT-R2 P1-5 fix：不再 reset 全局单例）
421//
422// Interpreter 路径：
423// - generatorChunks 为空（无字节码概念）
424// - opcodeCounts 全为 0（无 opcode 概念）
425// - peakTrackedCount 仍有效（Interpreter 也使用 GcManager）

429/// 从 BytecodeChunk map 提取生成器 chunk 信息
430std::vector<GeneratorChunkInfo> extractStackGeneratorChunks(const std::map<std::string, BytecodeChunk>& chunks) {
431 std::vector<GeneratorChunkInfo> result;
432 for (const auto& kv : chunks) {
433 if (kv.second.isGenerator) {
434 GeneratorChunkInfo info;
435 info.name = kv.first;
436 if (kv.second.yieldCount == BytecodeChunk::kDynamicYieldCount) {
437 info.isDynamic = true;
438 info.yieldCount = -1;
439 } else {
440 info.isDynamic = false;
441 info.yieldCount = kv.second.yieldCount;
442 }
443 result.push_back(std::move(info));
444 }
445 }
446 return result;
447}

449/// 从 RegBytecodeChunk map 提取生成器 chunk 信息
450std::vector<GeneratorChunkInfo> extractRegGeneratorChunks(const std::map<std::string, RegBytecodeChunk>& chunks) {
451 std::vector<GeneratorChunkInfo> result;
452 for (const auto& kv : chunks) {
453 if (kv.second.isGenerator) {
454 GeneratorChunkInfo info;
455 info.name = kv.first;
456 if (kv.second.yieldCount == RegBytecodeChunk::kDynamicYieldCount) {
457 info.isDynamic = true;
458 info.yieldCount = -1;
459 } else {
460 info.isDynamic = false;
461 info.yieldCount = kv.second.yieldCount;
462 }
463 result.push_back(std::move(info));
464 }
465 }
466 return result;
467}

471BackendExecDetail BackendExecutionService::executeWithDetail(const std::string& src, BackendType backend) {
472 // PERF-DETAIL: 单遍完成执行 + detail 收集（原实现全流程执行两遍——第二遍
473 // 重新 Lexer+Parser+compile+execute 仅为收集 generatorChunks / opcodeCounts /
474 // peakTrackedCount，开销与 execute() 相同）。detail 数据现由 execute() 的
475 // 可选出参在本次执行内联收集；词法/语法/编译错误在 execute() 阶段 1/2 即
476 // 返回，detail 扩展字段保持默认值（与原早退语义一致）。
477 // 注：detail 模式的 elapsedUs 含 step 采样开销（每条指令一次回调），原两遍
478 // 方案首遍计时不含采样——profile 面板展示时口径自洽（计数与耗时同源）。
479 BackendExecDetail d;
480 static_cast<BackendExecResult&>(d) = execute(src, backend, &d);
481 return d;
482}

484// ============================================================
485// BackendExecutionService::createBackend
486// ============================================================
487// ARCH-10 IVmBackend 工厂方法：创建指定类型的后端实例，返回多态 unique_ptr。
488// 调用方持有 unique_ptr 自动管理生命周期，通过 IVmBackend* 多态调用
489// stepOnce / getStack / getCallStack 等方法，无需依赖具体 VM/RegisterVM 类。
490//
491// 设计说明：
492// - Interpreter 不实现 IVmBackend（仅实现 IBackend 的基础接口），
493// 传入 BackendType::Interpreter 返回 nullptr。
494// - 工厂仅创建对象，不执行代码；调用方需自行调用具体子类的 execute 入口
495// 开始执行（VM::execute / RegisterVM::execute），然后通过 IVmBackend*
496// 接口读取状态。
497// - 返回 unique_ptr 而非裸指针，确保 RAII 生命周期管理，避免面板忘记
498// delete 导致泄漏。
499std::unique_ptr<IVmBackend> BackendExecutionService::createBackend(BackendType backend) {
500 switch (backend) {
501 case BackendType::StackVM_IR:
502 return std::make_unique<VM>();
503 case BackendType::RegisterVM_IR:
504 return std::make_unique<RegisterVM>();
505 case BackendType::Interpreter:
506 // Interpreter 不实现 IVmBackend 的步进接口（无 opcode 概念），
507 // 返回 nullptr 表示不支持。调用方应改用 execute() 走完整流程。
508 return nullptr;
509 }
510 return nullptr;
511}
