160std::string valueToBitsHex(const Value& v) {
161 // ARCH-10: 通过 MemoryInspectionAPI 获取 NaN-box 位模式快照，
162 // 消除面板对 interpreter/NaNBox.h 的直接依赖。
163 // 内部实现已处理 BoxedIntData（堆指针）降级为 PTR_TAG_BASE 占位符的边界情况。
164 return MemoryInspectionAPI::inspectValue(v).bitsHex;
165}
166
167// [[maybe_unused]]：位模式展示当前经 bitsHex 路径，二进制转换保留供
168// 教学面板扩展使用，避免 GCC -Werror=unused-function（匿名命名空间）。
169[[maybe_unused]] std::string bitsToBinary(uint64_t bits) {
170 std::string s(64, '0');
171 for (int i = 0; i < 64; ++i) {
172 if (bits & (1ULL << (63 - i)))
173 s[i] = '1';
174 }
175 return s;
176}

309// 拓展二期：双击变量项修改值（仅调试暂停/VM 暂停时生效）
310// ------------------------------------------------------------
311// 交互：双击变量行 → QInputDialog 输入新值（int/float/bool/null/"字符串"）
312// → parseDebugValueText 解析 → IdeController::setDebugVariableValue 写回
313// （Interpreter 路径经 DebugController 写回调；VM 路径经 VmStepper 双后端）。
314// 仅接受第一层变量项（分组的直接子项）；容器子元素编辑暂不支持。
315// ============================================================
316void VariableInspectorPanel::onVariableDoubleClicked(QTreeWidgetItem* item, int column) {
317 Q_UNUSED(column);
318 if (!controller_ || !item)
319 return;
320 // 仅第一层变量项：有父（分组）且父无父（分组是顶层）
321 if (!item->parent() || item->parent()->parent())
322 return;
323 // 仅调试暂停或 VM 暂停（单步间隙）时可写
324 bool interpPaused = controller_->isRunning() && controller_->isDebugRun() && controller_->isDebugPaused();
325 bool vmPaused = !controller_->isRunning() && controller_->isVmInitialized() && !controller_->isVmRunning();
326 if (!interpPaused && !vmPaused) {
327 liveStatusLabel_->setText(tr("状态：仅调试暂停/VM 单步暂停时可修改变量"));
328 return;
329 }
330 const QString name = item->text(0);
331 bool ok = false;
332 const QString text = QInputDialog::getText(
333 this, tr("修改变量"), tr("变量 %1 的新值（int / float / true / false / null / \"字符串\"）：").arg(name),
334 QLineEdit::Normal, item->text(2), &ok);
335 if (!ok || text.isEmpty())
336 return;
337 Value newVal;
338 if (!parseDebugValueText(text.toStdString(), newVal)) {
339 liveStatusLabel_->setText(tr("状态：新值解析失败（支持 int/float/bool/null/\"字符串\"）"));
340 return;
341 }
342 if (controller_->setDebugVariableValue(name.toStdString(), newVal)) {
343 refreshLive(); // 写入成功：刷新树展示新值
344 } else {
345 liveStatusLabel_->setText(tr("状态：变量 %1 写入失败（不存在或当前后端不支持）").arg(name));
346 }
347}

405/// 读取 VM 当前变量状态并刷新实时展示（含异常保护）。
406/// R121: 当 controller_->getSelectedFrame() >= 0 时，locals 改为消费 getSelectedFrameLocals()
407/// 显示所选帧的局部变量；globals 保持显示当前后端的全局变量（VM 模式）。
408/// selectedFrame_ == -1 时保持原行为（栈顶 locals + globals）。
409void VariableInspectorPanel::refreshLive() {
410 varTree_->clear();
411 if (!controller_) {
412 liveStatusLabel_->setText(tr("状态：未绑定 controller"));
413 return;
414 }
415
416 // 收集变量，按来源分组
417 std::vector<VariableSnapshot> locals;
418 std::unordered_map<std::string, Value> globals;
419 QString modeLabel;
420
421 // AUDIT-P1 fix: 优先判断当前活跃引擎（Interpreter 调试 > VM 单步），
422 // 避免 isVmInitialized 一次性永久 true 后 Interpreter 调试仍走 VM 分支显示陈旧数据。
423 // 同时增加 isDebugPaused() 检查，避免调试 resume 期间 worker 活跃时并发访问解释器
424 // 内部 unordered_map/vector 导致 UB（数据竞争）。
425 if (controller_->isRunning() && controller_->isDebugRun()) {
426 if (!controller_->isDebugPaused()) {
427 liveStatusLabel_->setText(tr("状态：调试运行中（暂停后刷新）"));
428 varDetail_->clear();
429 return;
430 }
431 modeLabel = tr("Interpreter (debug)");
432 locals = controller_->getDebugVariableSnapshot();
433 } else if (controller_->isVmInitialized()) {
434 // AUDIT-P1-CORRECT fix: VM RUN 模式（异步 QTimer 批量执行）期间 worker 可能
435 // 修改 globalSlots_/globalNameToSlot_，并发读取触发 UB。添加 isVmRunning() 守卫，
436 // 对齐 Interpreter 调试路径的 isDebugPaused() 守卫。
437 if (controller_->isVmRunning()) {
438 liveStatusLabel_->setText(tr("状态：VM 运行中（暂停后刷新）"));
439 varDetail_->clear();
440 return;
441 }
442 modeLabel = controller_->getUseRegisterVM() ? tr("RegisterVM") : tr("StackVM");
443 globals = controller_->getVmGlobals();
444 } else {
445 liveStatusLabel_->setText(tr("状态：未运行（启动调试或 VM 单步以查看变量）"));
446 varDetail_->clear();
447 return;
448 }
449
450 // R121: 当用户选中调用栈某帧时（selectedFrame >= 0），用 getSelectedFrameLocals()
451 // 覆盖 locals，显示该帧的局部变量。selectedFrame_ == -1 时保持原行为。
452 int selFrame = controller_->getSelectedFrame();
453 QString frameHint;
454 if (selFrame >= 0) {
455 auto selLocals = controller_->getSelectedFrameLocals();
456 if (!selLocals.empty()) {
457 // 用所选帧 locals 替换原 locals（原 locals 是栈顶帧 + 全局混合，已通过 globals 分支保留全局）
458 // 对于 Interpreter 模式，原 locals 已含全局 + upvalue，需保留全局部分
459 // 简化策略：直接清空 locals，用所选帧 locals 填充；globals 独立显示
460 locals.clear();
461 for (const auto& kv : selLocals) {
462 locals.push_back(VariableSnapshot{kv.first, kv.second, "selected-frame"});
463 }
464 frameHint = tr(" | 已选帧: %1").arg(selFrame);
465 }
466 }
467
468 int total = static_cast<int>(locals.size() + globals.size());
469 liveStatusLabel_->setText(tr("状态：%1 | 变量数：%2%3").arg(modeLabel).arg(total).arg(frameHint));
470
471 // 全局变量组
472 auto* globalGroup = new QTreeWidgetItem(varTree_);
473 globalGroup->setText(0, tr("全局变量 (%1)").arg(globals.size()));
474 QFont globalFont = globalGroup->font(0);
475 globalFont.setBold(true);
476 globalGroup->setFont(0, globalFont);
477 // AUDIT-P2 fix: 全局变量按名排序，与局部变量（std::map）排序一致，
478 // 避免 unordered_map hash 桶序导致的非确定展示顺序（对齐 VmStackPanel 行 171-172）
479 std::vector<std::pair<std::string, Value>> globalEntries(globals.begin(), globals.end());
480 std::sort(globalEntries.begin(), globalEntries.end(),
481 [](const auto& a, const auto& b) { return a.first < b.first; });
482 for (const auto& [name, val] : globalEntries) {
483 auto* item = new QTreeWidgetItem(globalGroup);
484 item->setText(0, QString::fromUtf8(name.c_str()));
485 // AUDIT-P1 fix: toString()/typeName() 异常防护，与 VmStackPanel 一致。
486 // P2-UX fix: 捕获异常原因并设为 tooltip，通知用户值可能已损坏。
487 std::string valStr, typeStr;
488 QString errorTooltip;
489 try {
490 typeStr = val.typeName();
491 } catch (const std::exception& e) {
492 typeStr = "<error>";
493 errorTooltip = tr("typeName() 异常：%1（值可能已损坏）").arg(QString::fromUtf8(e.what()));
494 } catch (...) {
495 typeStr = "<error>";
496 errorTooltip = tr("typeName() 抛出未知异常（值可能已损坏）");
497 }
498 try {
499 valStr = val.toString();
500 } catch (const std::exception& e) {
501 valStr = "<error>";
502 errorTooltip = errorTooltip.isEmpty()
503 ? tr("toString() 异常：%1（值可能已损坏）").arg(QString::fromUtf8(e.what()))
504 : errorTooltip + "\n" + tr("toString() 异常：%1").arg(QString::fromUtf8(e.what()));
505 } catch (...) {
506 valStr = "<error>";
507 errorTooltip = errorTooltip.isEmpty() ? tr("toString() 抛出未知异常（值可能已损坏）")
508 : errorTooltip + "\n" + tr("toString() 抛出未知异常");
509 }
510 item->setText(1, QString::fromUtf8(typeStr.c_str()));
511 item->setText(2, QString::fromUtf8(valStr.c_str()));
512 if (!errorTooltip.isEmpty()) {
513 item->setToolTip(0, errorTooltip);
514 item->setToolTip(1, errorTooltip);
515 item->setToolTip(2, errorTooltip);
516 }
517 item->setData(0, Qt::UserRole, QString::fromUtf8(name.c_str()));
518 }
519 globalGroup->setExpanded(true);
520
521 // 局部变量组（按 scope 二次分组）
522 std::map<std::string, std::vector<const VariableSnapshot*>> byScope;
523 for (const auto& s : locals)
524 byScope[s.scope].push_back(&s);
525
526 for (const auto& [scope, vec] : byScope) {
527 auto* group = new QTreeWidgetItem(varTree_);
528 QString label = QString::fromUtf8(scope.c_str());
529 if (label.isEmpty())
530 label = tr("本地作用域");
531 group->setText(0, tr("%1 (%2)").arg(label).arg(vec.size()));
532 QFont f = group->font(0);
533 f.setBold(true);
534 group->setFont(0, f);
535 for (const auto* s : vec) {
536 auto* item = new QTreeWidgetItem(group);
537 item->setText(0, QString::fromUtf8(s->name.c_str()));
538 // AUDIT-P1 fix: toString()/typeName() 异常防护。
539 // P2-UX fix: 捕获异常原因并设为 tooltip，通知用户值可能已损坏。
540 std::string valStr, typeStr;
541 QString errorTooltip;
542 try {
543 typeStr = s->value.typeName();
544 } catch (const std::exception& e) {
545 typeStr = "<error>";
546 errorTooltip = tr("typeName() 异常：%1（值可能已损坏）").arg(QString::fromUtf8(e.what()));
547 } catch (...) {
548 typeStr = "<error>";
549 errorTooltip = tr("typeName() 抛出未知异常（值可能已损坏）");
550 }
551 try {
552 valStr = s->value.toString();
553 } catch (const std::exception& e) {
554 valStr = "<error>";
555 errorTooltip = errorTooltip.isEmpty()
556 ? tr("toString() 异常：%1（值可能已损坏）").arg(QString::fromUtf8(e.what()))
557 : errorTooltip + "\n" + tr("toString() 异常：%1").arg(QString::fromUtf8(e.what()));
558 } catch (...) {
559 valStr = "<error>";
560 errorTooltip = errorTooltip.isEmpty() ? tr("toString() 抛出未知异常（值可能已损坏）")
561 : errorTooltip + "\n" + tr("toString() 抛出未知异常");
562 }
563 item->setText(1, QString::fromUtf8(typeStr.c_str()));
564 item->setText(2, QString::fromUtf8(valStr.c_str()));
565 if (!errorTooltip.isEmpty()) {
566 item->setToolTip(0, errorTooltip);
567 item->setToolTip(1, errorTooltip);
568 item->setToolTip(2, errorTooltip);

571 }
572 group->setExpanded(true);
573 }

576/// 变量列表选中项变化时的详情刷新。
577void VariableInspectorPanel::onVariableSelected() {
578 auto* cur = varTree_->currentItem();
579 if (!cur || !cur->parent()) {
580 varDetail_->clear();
581 return;
582 }
583 // 仅变量节点（非 scope group）显示详情
584 QString name = cur->text(0);
585 QString type = cur->text(1);
586 QString value = cur->text(2);
587
588 // 从 controller 取实际 Value（保证位字段精确）
589 QString bitsHex = tr("（未知）");
590 if (controller_) {
591 Value v;
592 bool found = false;
593 // AUDIT-P1 fix: 优先判断当前活跃引擎（与 refreshLive 一致），
594 // 避免 isVmInitialized 一次性 true 后 Interpreter 调试仍走 VM 分支找不到局部变量。
595 // 同时增加 isDebugPaused() 检查避免调试 resume 期间数据竞争。
596 if (controller_->isRunning() && controller_->isDebugRun()) {
597 if (controller_->isDebugPaused()) {
598 auto locals = controller_->getDebugVariableSnapshot();
599 for (const auto& s : locals) {
600 if (s.name == name.toStdString()) {
601 v = s.value;
602 found = true;
603 break;
604 }
605 }
606 }
607 } else if (controller_->isVmInitialized() && !controller_->isVmRunning()) {
608 // AUDIT-P1-CORRECT fix: VM RUN 期间并发读取 globals 触发 UB，添加 isVmRunning() 守卫
609 auto globals = controller_->getVmGlobals();
610 auto it = globals.find(name.toStdString());
611 if (it != globals.end()) {
612 v = it->second;
613 found = true;
614 }
615 }
616 // AUDIT-P1 fix: 原 `!v.isNull() || v.getType()==VAL_NULL` 是恒真表达式
617 // （false||true=true 或 true||...=true），导致未找到变量时也显示 null 位模式。
618 // 改为 found 标志判断，仅找到时显示位模式。
619 if (found) {
620 bitsHex = QString::fromUtf8(valueToBitsHex(v).c_str());
621 }
622 }
623
624 QString html = QString("<h3>%1: %2</h3>"
625 "<p><b>类型:</b> %3</p>"
626 "<p><b>值 (toString):</b> %4</p>"
627 "<p><b>NaN-boxing 位:</b> <code>%5</code></p>")
628 .arg(name.toHtmlEscaped())
629 .arg(type.toHtmlEscaped())
630 .arg(type.toHtmlEscaped())
631 .arg(value.toHtmlEscaped())
632 .arg(bitsHex.toHtmlEscaped());
633 varDetail_->setHtml(html);
634}
