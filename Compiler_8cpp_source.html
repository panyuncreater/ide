23/**
24 * Compiler —— MiniLang 前端字节码编译器总入口
25 *
26 * 职责：将解析后的 AST（Block 语句列表）翻译为可在 VM 上执行的字节码。
27 * 它是纯单遍（single-pass）递归下降式代码生成器：边遍历 AST 边 emit 指令，
28 * 不构建独立的中间 AST 再遍历（栈式 VM 直接路径）。三条后端路径共享同一套
29 * visit* 访问者，因此语义天然一致。
30 *
31 * ── 三条后端路径（由编译开关选择，见 compile()）─────────────────────────
32 * 1) 栈式 VM 直接路径（默认）：visit* 直接 emit 栈式 OpCode，常量池写入主 chunk。
33 * 2) IR 路径（useIR_）：AST → IR（AstIRBuilder）→ [可选优化 pass] → 栈式字节码
34 * （BytecodeIRBackend）。指令以操作数栈为模型，与路径 1 等价。
35 * 3) 寄存器式 VM 路径（useRegisterVM_）：AST → IR → 寄存器式字节码
36 * （RegisterBytecodeBackend）。指令以固定寄存器窗口为模型，无操作数栈。
37 * 路径 1 与 2/3 的关键差异在于 IR 路径经过统一的 AstIRBuilder / *IRBackend
38 * 中间层，便于做常量折叠、复制传播、DCE 等优化 pass。
39 *
40 * ── 派发机制 ──────────────────────────────────────────────────────────
41 * compileNode(node) 通过 Visitor 模式调用 node->accept(*this)，回调对应
42 * visit* 方法。compileStatement() 在表达式语句之后补发 OP_POP 以平衡栈
43 * （见下“栈平衡纪律”）。
44 *
45 * ── 栈平衡纪律（仅栈式路径相关）──────────────────────────────────────
46 * 每个 visit* 必须保证“离开时栈高度与进入时一致”，除非该节点作为表达式被调用，
47 * 此时约定“产生恰好一个栈顶值，由调用方消费”。compileStatement 对产生栈值
48 * 的表达式节点（赋值/调用/二元运算/字面量等）补发 OP_POP，避免顶层或循环体内
49 * 栈无限累积导致运行时栈溢出。声明与控制流节点已自行平衡，不在其列。
50 *
51 * ── 闭包与自由变量 ────────────────────────────────────────────────────
52 * 嵌套函数通过 collectFreeVars/computeFreeVars 前向分析出捕获的自由变量，
53 * 再由 resolveUpvalue 在编译函数体时登记 UpvalueDesc（区分 isLocal 直接捕获
54 * 与 isLocal=false 透传捕获），支撑 3+ 层嵌套闭包沿外层函数透传 upvalue。
55 *
56 * ── 常量折叠 ──────────────────────────────────────────────────────────
57 * tryFoldBinary/tryFoldUnary 在 visit* 中优先尝试编译期求值；仅当操作数
58 * 均为编译期常量（extractConstant 递归提取）时折叠，并统一经 emitConstant
59 * 写入常量池，避免对带副作用表达式误折叠。
60 *
61 * ── 模块系统 ──────────────────────────────────────────────────────────
62 * import/export 通过模块加载器与 ModuleIsolation 实现跨文件编译；每次编译
63 * 都会重置 linkedModuleSet_/moduleLoadingSet_/moduleExports_ 等状态，
64 * 防止跨次编译复用旧缓存（见 compile() 与 compileVia* 的清理段）。
65 */

559// AUDIT-P1.1 fix: break/continue finally 续跳机制。
560// 若 break/continue 在 try-finally 内，发射 OP_PUSH_JUMP_TARGET + OP_JUMP 续跳字节码。
561// 编译期布局（两层 try-finally 示例，A 内层、B 外层）：
562// OP_PUSH_JUMP_TARGET <breakTarget> ← 最外层 finally 执行完后跳转
563// OP_PUSH_JUMP_TARGET <B.finallyEntry> ← 内层 finally 执行完后跳转
564// OP_JUMP <A.finallyEntry> ← 跳到最内层 finally
565// 执行顺序：A.finally → OP_FINALLY_END(pop B.finallyEntry, jump) → B.finally →
566// OP_FINALLY_END(pop breakTarget, jump) → breakTarget
567// 多层嵌套同理：从外到内依次 push，从内到外依次执行 finally 后续跳。
568bool Compiler::emitFinallyJump(int line, std::vector<size_t>& realTargetPatches) {
569 // 收集所有 enclosing try-finally 块（从内到外）
570 // finallyIndices[0] = innermost, finallyIndices[N-1] = outermost
571 std::vector<size_t> finallyIndices;
572 for (size_t i = tryFinallyStack_.size(); i > 0; --i) {
573 if (tryFinallyStack_[i - 1].hasFinally) {
574 finallyIndices.push_back(i - 1);
575 }
576 }
577 if (finallyIndices.empty())
578 return false;
579
580 // 从最外层到最内层，依次发射 OP_PUSH_JUMP_TARGET
581 // 循环 i 从 N 递减到 1，对应 ctxIdx 从 finallyIndices[N-1]（outermost）到 finallyIndices[0]（innermost）。
582 // push 顺序：outermost → innermost，使 pendingJumpStack_ 弹出顺序为 innermost → outermost，
583 // 与 finally 执行顺序（内层先执行）匹配。
584 // BUG-025 fix: 原代码 `if (i == 1)` 与 `finallyIndices[i - 2]` 索引反转。
585 // 原意：最外层（i==N）push realTarget，内层 push 下一个外层 finally 的 entry。
586 // 实际：最内层（i==1）push realTarget，外层 push 下一个内层 finally 的 entry。
587 // 修复：判据改为 `if (i == finallyIndices.size())`，索引改为 `finallyIndices[i]`。
588 for (size_t i = finallyIndices.size(); i > 0; --i) {
589 chunk_.writeOp(OpCode::OP_PUSH_JUMP_TARGET, line);
590 size_t patch = chunk_.code.size();
591 chunk_.writeShort(0, line); // 占位，待回填
592
593 if (i == finallyIndices.size()) {
594 // 最外层：push 真实跳转目标（breakTarget/continueTarget），由循环编译完成后回填。
595 // BUG-026 fix: realTargetPatches 进入 breakJumps/continueJumps 后，回填代码使用
596 // `code[patch+1]`/`code[patch+2]`（patch 为 opcode 位置）。
597 // 但此处 patch 是 opcode 之后的位置（第一操作数字节），需回退一字节到 opcode 位置。
598 realTargetPatches.push_back(patch - 1);
599 } else {
600 // 内层：push 下一个（更外层）finally 入口。finallyIndices[i] 比 finallyIndices[i-1] 更外。
601 size_t nextCtxIdx = finallyIndices[i];
602 tryFinallyStack_[nextCtxIdx].pendingTargetPatches.push_back(patch);
603 }
604 }
605
606 // 发射 OP_JUMP 到最内层 finally 入口（待回填）
607 chunk_.writeOp(OpCode::OP_JUMP, line);
608 size_t jumpPatch = chunk_.code.size();
609 chunk_.writeShort(0, line); // 占位
610 tryFinallyStack_[finallyIndices[0]].pendingJumpPatches.push_back(jumpPatch);
611
612 return true;
613}
