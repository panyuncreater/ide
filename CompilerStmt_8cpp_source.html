416 chunk_.isGenerator = node.isGenerator;

423 functionChunks_[effectiveName] = std::move(chunk_);

432 }

448 }

456 chunk_.write(static_cast<uint8_t>(funChunk.upvalues[i].index), node.line);
457 }
458 // R98 W3: 匿名 lambda 不存储到任何变量槽——闭包值留在栈上作为表达式结果。
459 // 调用方（如 var f = fun(x){...}; 或 map(arr, fun(x){...});）从栈顶取值。

462 // 注意：若 lambda 出现在表达式语句上下文（如 `fun(x){...}(5);` 立即调用），
463 // 调用结果会被外层 expressionStatement 的 POP 清理。

466 // H5 fix: 内嵌函数存储为局部变量（供 OP_CALL_EXPR 使用），全局函数直接弹出
467 if (isInner) {
468 // 分配局部变量槽位存储闭包值
469 int slot = static_cast<int>(currentLocals_.size());
470 currentLocals_[node.name] = slot;
471 peakLocals_ = std::max(peakLocals_, static_cast<int>(currentLocals_.size()));
472 innerFunctions_.insert(node.name);

475 chunk_.write(static_cast<uint8_t>(slot), node.line);
476 chunk_.writeOp(OpCode::OP_POP, node.line);
477 } else {
478 // R98 W2 fix: 全局函数的闭包值存储到全局槽位（而非 POP 丢弃）。
479 // 原 H5 fix 假设"顶层函数仅通过 OP_CALL 按名查找 functionChunks_/functionClosures_，
480 // 永不作为值使用"，但 W2 高阶函数 map/filter/reduce/forEach/find 打破此假设——
481 // map(arr, double) 中 double 作为参数值被 visitVarRef 编译为 OP_GET_GLOBAL <slot>，
482 // 若全局槽位为 null（闭包值被 POP 丢弃），OP_GET_GLOBAL 取到 null，
483 // invokeClosureSync 报"高阶函数的参数必须是函数"或上游报"未定义的变量: double"。
484 // 修复：顶层函数也用 OP_DEFINE_GLOBAL 将闭包值存入全局槽位，使函数名可作为值引用。
485 // 安全性：(1) OP_CALL 仍通过 functionClosures_ 按名查找，不读取 globalSlots_，不受影响；
486 // (2) W1-BUG2 fix 的 isClassName/isInFunctionChunks 分流仍有效——直接调用 double(5)
487 // 因 isInFunctionChunks=true 走 OP_CALL，不读 globalSlots_；
488 // (3) 函数重定义 fun f(){} fun f(){} 第二次 OP_DEFINE_GLOBAL 覆盖第一次，语义正确。
489 int globalSlot = lookupGlobalSlot(node.name);
490 if (globalSlot >= 0) {
491 chunk_.writeOp(OpCode::OP_DEFINE_GLOBAL, node.line);
492 chunk_.writeShort(static_cast<uint16_t>(globalSlot), node.line);
493 } else {
494 // 无全局槽位（pre-scan 未注册）时退化为 OP_POP 保持向后兼容
495 chunk_.writeOp(OpCode::OP_POP, node.line);
496 }
497 }
498
499 // VM-05/06: 如果在函数内，将本函数注册到 outerLocals_ 和 outerFunctions_ 供更内层捕获
500 if (inFunction_) {
501 // H5 fix: 内嵌函数已有真实槽位，使用它；全局函数使用虚拟槽位
502 int slot = isInner ? innerFunctionSlots_[node.name] : peakLocals_;
503 outerLocals_[node.name] = slot;
504 outerFunctions_[node.name] = slot;
505 }
506 return;
507}

599 stringConstIndex_.clear(); // R164 fixup2: 清空字符串常量去重缓存（新 chunk 有新常量池）

608 currentTypeParams_ = node.typeParams;

642 }

672void Compiler::emitDefaultValues(FunDecl& node) {
673 // F10: 编译默认参数值为常量
674 // 仅支持字面量（Number/String/Bool/Null）和负数字面量，复杂表达式需通过 Interpreter 执行
675 // 本函数由 visitFunDecl 与 emitMethodBody 共用，确保两条路径默认参数语义一致。
676 for (size_t i = 0; i < node.defaultValues.size(); ++i) {
677 if (node.defaultValues[i]) {
678 const ASTNode* dv = node.defaultValues[i].get();
679 Value constVal;
680 bool isConst = false;
681
682 if (dv->nodeType == NodeType::NODE_NUMBER_LITERAL) {
683 constVal = static_cast<const NumberLiteral*>(dv)->getValue(); // A1 fix: getValue()
684 isConst = true;
685 } else if (dv->nodeType == NodeType::NODE_STRING_LITERAL) {
686 constVal = static_cast<const StringLiteral*>(dv)->getValue(); // A1 fix: getValue()
687 isConst = true;
688 } else if (dv->nodeType == NodeType::NODE_BOOL_LITERAL) {
689 constVal = static_cast<const BoolLiteral*>(dv)->getValue(); // A1 fix: getValue()
690 isConst = true;
691 } else if (dv->nodeType == NodeType::NODE_NULL_LITERAL) {
692 constVal = Value::nullValue();
693 isConst = true;
694 } else if (dv->nodeType == NodeType::NODE_UNARY_OP) {
695 // 支持负数字面量: -42, -3.14, --5 (双重否定)
696 // C-P2-2 fix: 递归折叠嵌套一元运算，使 --5 等价于 5
697 const ASTNode* cur = dv;
698 int negateCount = 0;
699 while (cur && cur->nodeType == NodeType::NODE_UNARY_OP) {
700 const auto* unary = static_cast<const UnaryOp*>(cur);
701 if (unary->opType != UnaryOp::UnaryOpType::UOP_NEGATE)
702 break;
703 ++negateCount;
704 cur = unary->operand.get();
705 }
706 if (cur && cur->nodeType == NodeType::NODE_NUMBER_LITERAL && negateCount > 0) {
707 // A1 fix: NumberLiteral 存储标量，用 isInt()/intVal() 直接访问
708 const NumberLiteral* numNode = static_cast<const NumberLiteral*>(cur);
709 if (numNode->isInt()) {
710 int64_t v = numNode->intVal();
711 // 奇数次取反为负，偶数次为正
712 constVal = Value((negateCount % 2 == 1) ? -v : v);
713 } else {
714 double v = numNode->floatVal(); // A1 fix: numNode 标量访问
715 constVal = Value((negateCount % 2 == 1) ? -v : v);
716 }
717 isConst = true;
718 }
719 }
720
721 if (isConst) {
722 uint16_t constIdx = chunk_.addConstant(constVal);
723 chunk_.defaultConstIndices.push_back(constIdx);
724 } else {
725 // 复杂表达式默认值：VM 不支持，记录无效索引（NO_INDEX）
726 // Interpreter 路径仍可正常执行
727 chunk_.defaultConstIndices.push_back(RuntimeLimits::NO_INDEX);
728 }
729 }
730 }
731}

733void Compiler::visitFunCall(FunCall& node) {
734 // L18 lang-constfun: const fun 调用且实参全字面量 → 编译期沙箱求值，
735 // 直接 emit 常量（求值失败/不纯/非原始类型结果 → 回退普通调用）。
736 // 局部名遮蔽防护：当前作用域存在同名局部/upvalue 时不折叠。
737 if (!constFunDecls_.empty() && node.callee == nullptr && currentLocals_.find(node.name) == currentLocals_.end()) {
738 if (auto folded = ConstFunEval::tryEvaluate(constFunDecls_, node)) {
739 emitConstant(*folded, node.line);
740 return;
741 }
742 }
743 // 链式调用 / 表达式调用: callee(args)
744 if (node.callee) {
745 // C-P1-1 fix: 参数数量检查移到编译参数之前，避免截断后栈损坏
746 if (node.arguments.size() > 255) {
747 error("函数调用参数数量超过限制（最大 255 个）", node.line, 0);
748 return;
749 }
750 // 编译被调用表达式（结果应为闭包值，推入栈顶）
751 compileNode(node.callee.get());
752 // 编译参数
753 for (auto& arg : node.arguments) {
754 compileNode(arg.get());
755 }
756 // OP_CALL_EXPR: 栈顶 N 个参数下方为闭包值
757 chunk_.writeOp(OpCode::OP_CALL_EXPR, node.line);
758 chunk_.write(static_cast<uint8_t>(node.arguments.size()), node.line);
759 return;
760 }
761
762 // H5 fix: 内嵌函数通过局部变量中的闭包值调用，避免 functionClosures_ 按名称覆盖
763 auto innerIt = innerFunctionSlots_.find(node.name);
764 if (innerIt != innerFunctionSlots_.end()) {
765 // C-P1-1 fix: 参数数量检查移到编译参数之前
766 if (node.arguments.size() > 255) {
767 error("函数调用参数数量超过限制（最大 255 个）", node.line, 0);
768 return;
769 }
770 // 先 push 闭包值（从局部变量获取）
771 chunk_.writeOp(OpCode::OP_GET_LOCAL, node.line);
772 chunk_.write(static_cast<uint8_t>(innerIt->second), node.line);
773 // 再编译参数
774 for (auto& arg : node.arguments) {
775 compileNode(arg.get());
776 }
777 chunk_.writeOp(OpCode::OP_CALL_EXPR, node.line);
778 chunk_.write(static_cast<uint8_t>(node.arguments.size()), node.line);
779 return;
780 }
781
782 // W1 fix (R103): 对齐 IR.cpp CRITICAL-1 fix——若 node.name 是当前函数的
783 // LOCAL 或 UPVALUE（持有闭包值，如 `var g = f; g();`），走 OP_CALL_EXPR 路径
784 // 通过闭包值调用。原 StackVM 直接路径仅检查 innerFunctionSlots_（嵌套函数声明 slot），
785 // 不覆盖 `var g = f;` 的普通局部变量闭包值场景，导致 StackVM 直接路径报
786 // "未定义的函数: g"。IR 路径已通过 CRITICAL-1 fix 解决，直接路径未对齐。
787 if (inFunction_) {
788 auto localIt = currentLocals_.find(node.name);
789 if (localIt != currentLocals_.end()) {
790 if (node.arguments.size() > 255) {
791 error("函数调用参数数量超过限制（最大 255 个）", node.line, 0);
792 return;
793 }
794 chunk_.writeOp(OpCode::OP_GET_LOCAL, node.line);
795 chunk_.write(static_cast<uint8_t>(localIt->second), node.line);
796 for (auto& arg : node.arguments) {
797 compileNode(arg.get());
798 }
799 chunk_.writeOp(OpCode::OP_CALL_EXPR, node.line);
800 chunk_.write(static_cast<uint8_t>(node.arguments.size()), node.line);
801 return;
802 }
803 int uvIdx = resolveUpvalue(node.name, node.line);
804 if (uvIdx >= 0) {
805 if (node.arguments.size() > 255) {
806 error("函数调用参数数量超过限制（最大 255 个）", node.line, 0);
807 return;
808 }
809 chunk_.writeOp(OpCode::OP_GET_UPVALUE, node.line);
810 chunk_.write(static_cast<uint8_t>(uvIdx), node.line);
811 for (auto& arg : node.arguments) {
812 compileNode(arg.get());
813 }
814 chunk_.writeOp(OpCode::OP_CALL_EXPR, node.line);
815 chunk_.write(static_cast<uint8_t>(node.arguments.size()), node.line);
816 return;
817 }
818 } else {
819 // W1 fix 扩展：顶层 var 持闭包值调用 (var g = makeAdder(10); g(32);)
820 // 顶层 var 注册到 globalSlotAllocator_，此处通过 OP_GET_GLOBAL 加载闭包值
821 // 后走 OP_CALL_EXPR。原 StackVM 直接路径走 OP_CALL 按名查找 functionChunks_，
822 // 找不到 var 持有的闭包值，报"未定义的函数: g"。
823 //
824 // W1-BUG fix (R104): compile() pre-scan 步骤也将顶层 ClassDecl 注册到
825 // globalSlotAllocator_（见第 142-143 行），但 `Foo()` 类实例化必须走 OP_CALL
826 // （VM 运行时检测到名称是类时执行 OP_CLASS_NEW + 字段初始化）。
827 // 原 W1 fix 扩展未区分 class 名与普通 var 闭包值名，导致 `Foo()` 被错误编译为
828 // OP_GET_GLOBAL + OP_CALL_EXPR，VM 在 OP_CALL_EXPR 中找不到类构造入口而失败。
829 // 修复：先用 classFieldNames_ 判定是否为类名，若是则跳过 W1 fix 扩展，
830 // 落到下方 OP_CALL 路径（VM 运行时按名查找 functionChunks_ 或 classRegistry_）。
831 //
832 // W1-BUG2 fix (R104): 模块导入的 FunDecl（如 `import { counter } from "m";`）
833 // 也被 preScanModuleGlobals 注册到 globalSlotAllocator_，但其闭包值在
834 // visitFunDecl 顶层路径被 OP_POP 丢弃，全局槽位为 null。原 W1 fix 扩展未排除
835 // 此场景，counter() 被错误编译为 OP_GET_GLOBAL + OP_CALL_EXPR，运行时
836 // globalSlots_[slot] 为 null，OP_CALL_EXPR 报"表达式调用需要函数值"。
837 // 修复：追加 functionChunks_ 查找——若名称在 functionChunks_ 中（已编译的
838 // 函数声明，含模块导入的 FunDecl），走 OP_CALL 命名调用通过 functionChunks_
839 // 查找。仅 var 持闭包值（不在 functionChunks_ 中）走 OP_CALL_EXPR。
840 bool isClassName = (classFieldNames_.find(node.name) != classFieldNames_.end());
841 bool isInFunctionChunks = (functionChunks_.find(node.name) != functionChunks_.end());
842 int slot = (isClassName || isInFunctionChunks) ? -1 : lookupGlobalSlot(node.name);
843 if (slot >= 0) {
844 if (node.arguments.size() > 255) {
845 error("函数调用参数数量超过限制（最大 255 个）", node.line, 0);
846 return;
847 }
848 chunk_.writeOp(OpCode::OP_GET_GLOBAL, node.line);
849 chunk_.writeShort(static_cast<uint16_t>(slot), node.line);
850 for (auto& arg : node.arguments) {
851 compileNode(arg.get());
852 }
853 chunk_.writeOp(OpCode::OP_CALL_EXPR, node.line);
854 chunk_.write(static_cast<uint8_t>(node.arguments.size()), node.line);
855 return;
856 }
857 }
858
859 // C-P1-1 fix: 参数数量检查移到编译参数之前
860 if (node.arguments.size() > 255) {
861 error("函数调用参数数量超过限制（最大 255 个）", node.line, 0);
862 return;
863 }
864
865 // 编译参数
866 for (auto& arg : node.arguments) {
867 compileNode(arg.get());
868 }
869
870 // 函数名作为常量
871 uint16_t nameIdx = identifierIndex(node.name);
872
873 // 使用 OP_CALL 指令
874 chunk_.writeOp(OpCode::OP_CALL, node.line);
875 chunk_.write(static_cast<uint8_t>(nameIdx & 0xFF), node.line);
876 chunk_.write(static_cast<uint8_t>((nameIdx >> 8) & 0xFF), node.line);
877 chunk_.write(static_cast<uint8_t>(node.arguments.size()), node.line);
878 return;
879}

881void Compiler::visitReturnStmt(ReturnStmt& node) {
882 if (!inFunction_) {
883 error("return 只能在函数体内使用", node.line, 0);
884 return;
885 }
886 // R109/L15 TCO: 尾调用优化。识别 return f(args) 或 return this.method(args) 形态的
887 // 自递归调用，编译为"参数求值 + 逆序 OP_SET_LOCAL 覆盖参数槽 + OP_JUMP 函数入口"。
888 // 跳过 OP_RETURN 的帧弹出，复用当前帧执行下一轮递归，深度无界。
889 //
890 // L15 扩展（与 IR 路径严格对齐）：
891 // - SelfFunction: 放宽 upvalue 限制（同函数重用闭包，upvalue 指向外层栈不变）
892 // - SelfFunction: 支持默认参数（args.size() < params.size() 时用默认值填充）
893 // - SelfMethod: 类方法自调用 return this.method(args)
894 // 保留 slot 0 (this) 和 slot 1..N (字段)，仅覆盖参数槽 N+1..N+params
895 //
896 // 仍要求的条件：
897 // (1) AST 结构匹配（TCO::identifyTailCall 返回非 None）
898 // (2) 不在 try 块内（tryDepth_ == 0）
899 // (3) currentFunctionDecl_ 非空（用于获取形参/默认值）
900 // (4) args.size() <= params.size()（超出视为非尾调用）
901 //
902 // 参数求值顺序：先全部求值到栈顶，再逆序 OP_SET_LOCAL + OP_POP 覆盖参数槽。
903 // 逆序保证：args[i] 可能引用 params[j]（j<i），若顺序赋值会破坏 params[j]。
904 // 栈布局变化：[..., val_0, val_1, ..., val_N-1] → [...] → OP_JUMP（栈深度恢复）
905 const bool isMethod = compilingMethodBody_; // AUDIT-R7 F5 fix: 独立标志（非 currentClassName_ 非空）
906 TCO::TailCallInfo tco = TCO::identifyTailCall(&node, currentFunctionName_, isMethod);
907 // B1-Shadow fix: SelfFunction 名称被局部变量/参数遮蔽（var f = other; return f(x);）
908 // 时不是自递归——非 TCO 路径的局部优先解析会调用被赋值的闭包，而 TCO 会错误
909 // 跳回自身入口形成死循环（实证：三 VM 路径均报"指令数超出上限"，Interpreter
910 // 的运行时重绑定校验正确回退）。currentLocals_ 含当前作用域可见的局部变量与参数。
911 if (tco.kind == TCO::TailCallInfo::Kind::SelfFunction &&
912 currentLocals_.find(tco.call->name) != currentLocals_.end()) {
913 tco.kind = TCO::TailCallInfo::Kind::None;
914 }
915 // L18 eng-tailcall: 互递归/一般尾调用 return g(args)。编译策略：先正常编译
916 // 调用表达式（自动继承 visitFunCall 的遮蔽/upvalue/全局槽分流），若尾部
917 // 恰为 OP_CALL（纯命名调用路径）则原地 patch 为 OP_TAIL_CALL，再补 OP_RETURN。
918 // 运行时帧复用或降级为普通调用（语义与 CALL+RETURN 完全等价）。
919 // 限制条件：
920 // - 非方法体（方法帧的 this/字段槽与 writeBack 语义不兼容帧复用）
921 // - 无返回类型注解（TCO 路径跳过 OP_TYPE_CHECK，目标函数类型未知）
922 // - 非生成器体（协程帧快照不可复用，运行时另有兑底）
923 // - 不在 try 块内（handler 清理语义）
924 if (tco.kind == TCO::TailCallInfo::Kind::GeneralCall && tryDepth_ == 0 && inFunction_ && !compilingMethodBody_ &&
925 currentFunctionReturnType_.empty() && (currentFunctionDecl_ == nullptr || !currentFunctionDecl_->isGenerator)) {
926 compileNode(node.value.get());
927 // 仅当尾部恰为 4 字节 OP_CALL 时 patch（CALL_EXPR 等路径保持普通调用）。
928 // 三重校验防误判操作数字节：opcode 字节 + 常量池名称回查 + argCount 匹配。
929 if (chunk_.code.size() >= 4 && chunk_.code[chunk_.code.size() - 4] == static_cast<uint8_t>(OpCode::OP_CALL)) {
930 size_t callPos = chunk_.code.size() - 4;
931 uint16_t nIdx = static_cast<uint16_t>(chunk_.code[callPos + 1] | (chunk_.code[callPos + 2] << 8));
932 uint8_t ac = chunk_.code[callPos + 3];
933 if (nIdx < chunk_.constants.size() && chunk_.constants[nIdx].isString() &&
934 chunk_.constants[nIdx].stringVal() == tco.call->name && ac == tco.call->arguments.size()) {
935 chunk_.code[callPos] = static_cast<uint8_t>(OpCode::OP_TAIL_CALL);
936 }
937 }
938 chunk_.writeOp(OpCode::OP_RETURN, node.line);
939 return;
940 }
941 if (tco.kind == TCO::TailCallInfo::Kind::GeneralCall) {
942 tco.kind = TCO::TailCallInfo::Kind::None; // 条件不满足：回退普通 return 路径
943 }
944 if (tco.kind != TCO::TailCallInfo::Kind::None && tryDepth_ == 0 && currentFunctionDecl_ != nullptr) {
945 const size_t paramCount = currentFunctionDecl_->params.size();
946 // 计算参数槽位基址：SelfFunction=0，SelfMethod=1+fieldCount（跳过 this 和字段）
947 size_t paramSlotBase = 0;
948 const std::vector<std::shared_ptr<ASTNode>>* args = nullptr;
949 if (tco.kind == TCO::TailCallInfo::Kind::SelfFunction) {
950 args = &tco.call->arguments;
951 } else { // SelfMethod
952 paramSlotBase = 1 + chunk_.fieldOrder.size();
953 args = &tco.methodCall->arguments;
954 }
955 if (args->size() <= paramCount && paramSlotBase + paramCount <= 255) {
956 // 编译所有实参表达式，结果压栈
957 for (auto& arg : *args) {
958 compileNode(arg.get());
959 }
960 // L15: 缺失参数用默认值或 null 填充
961 for (size_t i = args->size(); i < paramCount; ++i) {
962 if (i < currentFunctionDecl_->defaultValues.size() && currentFunctionDecl_->defaultValues[i]) {
963 compileNode(currentFunctionDecl_->defaultValues[i].get());
964 } else {
965 chunk_.writeOp(OpCode::OP_NULL, node.line);
966 }
967 }
968 // 逆序 OP_SET_LOCAL + OP_POP 覆盖参数槽位
969 // 栈布局：[..., val_0, val_1, ..., val_{N-1}]（val_{N-1} 在栈顶）
970 // OP_SET_LOCAL 复制栈顶到 slot（不弹栈），OP_POP 弹栈。
971 // 必须从栈顶（val_{N-1}）开始赋值到 slot N-1，逆序向下，
972 // 否则正向迭代会把 val_{N-1-i} 赋给 slot i（参数顺序反转）。
973 // AUDIT-R7 F3 fix: 覆盖前先关闭 slot >= paramSlotBase 的 open upvalues——
974 // 帧复用覆盖仍被本轮闭包捕获的槽位时，闭包应快照当轮值（实证：原实现
975 // 尾递归内逐轮存入数组的闭包全部读到末轮值 "111"，Interpreter 为 "321"）。
976 // SelfMethod 的 this/字段槽（< paramSlotBase）不被覆盖不关闭。
977 chunk_.writeOp(OpCode::OP_CLOSE_UPVALUE, node.line);
978 chunk_.write(static_cast<uint8_t>(paramSlotBase), node.line);
979 for (size_t i = paramCount; i > 0; --i) {
980 chunk_.writeOp(OpCode::OP_SET_LOCAL, node.line);
981 chunk_.write(static_cast<uint8_t>(paramSlotBase + i - 1), node.line);
982 chunk_.writeOp(OpCode::OP_POP, node.line);
983 }
984 // 跳转到函数/方法入口（OP_JUMP 是绝对跳转）
985 chunk_.writeOp(OpCode::OP_JUMP, node.line);
986 chunk_.writeShort(static_cast<uint16_t>(currentFunctionEntryIp_), node.line);
987 return;
988 }
989 }
990 if (node.value) {
991 compileNode(node.value.get());
992 } else {
993 chunk_.writeOp(OpCode::OP_NULL, node.line);
994 }
995 // BUG-TYPE-1 fix (P1): 函数有返回类型注解时，在 OP_RETURN 前发射 OP_TYPE_CHECK
996 // 检查返回值类型兼容性。对齐 Interpreter::visitReturnStmt 的 checkType 逻辑。
997 // 原实现仅 Interpreter 检查返回类型，StackVM/RegisterVM 静默通过，导致类型安全绕过。
998 // R109 TCO: TCO 路径跳过 TYPE_CHECK 是安全的——递归调用的 return 会再次触发检查。
999 if (!currentFunctionReturnType_.empty()) {
1000 emitTypeCheck(currentFunctionReturnType_, node.line);
1001 }
1002 // L4 fix: return 在 try-finally 块内时，续跳到 finally 入口执行 finally 块。
1003 // finally 块作为 Block 栈平衡（var 声明 OP_SET_LOCAL+OP_POP 不改操作数栈深度），
1004 // 返回值留在栈顶，finally 末尾 OP_FINALLY_END 续跳到紧邻的 OP_RETURN（return landing pad）。
1005 // finally 块内若再次 return/throw，新控制流覆盖原 return（与 Interpreter 对齐）。
1006 std::vector<size_t> returnFinallyPatches;
1007 if (emitFinallyJump(node.line, returnFinallyPatches)) {
1008 // return 在 try-finally 块内：回填 realTarget 到 OP_RETURN 位置
1009 size_t returnIp = chunk_.code.size();
1010 if (returnIp > 65535) {
1011 error("return 续跳目标溢出 65535", node.line, 0);
1012 return;
1013 }
1014 uint16_t target = static_cast<uint16_t>(returnIp);
1015 for (size_t patch : returnFinallyPatches) {
1016 // patch = OP_PUSH_JUMP_TARGET opcode 位置（emitFinallyJump 存 patch-1）
1017 // 回填操作数两字节到 patch+1, patch+2（与 breakJumps 回填一致）
1018 chunk_.code[patch + 1] = static_cast<uint8_t>(target & 0xFF);
1019 chunk_.code[patch + 2] = static_cast<uint8_t>((target >> 8) & 0xFF);
1020 }
1021 }
1022 chunk_.writeOp(OpCode::OP_RETURN, node.line);
1023 return;
1024}

1026void Compiler::visitBreakStmt(BreakStmt& node) {
1027 if (loopStack_.empty()) {
1028 error("break 只能在循环体内使用", node.line, 0);
1029 return;
1030 }
1031 // C-P0-2 fix: 只弹出循环内部的 try handler（与 continue 一致），
1032 // 之前使用 tryDepth_（全局深度）会错误弹出入层函数/外层循环的 try 帧
1033 int tryDepthInLoop = tryDepth_ - loopStack_.back().tryDepthAtStart;
1034 for (int i = 0; i < tryDepthInLoop; ++i) {
1035 chunk_.writeOp(OpCode::OP_TRY_END, node.line);
1036 }
1037 // AUDIT-P2-CORRECT fix: break 跳出循环时需关闭循环体内声明的闭包捕获变量的
1038 // upvalue，对齐正常迭代退出时的 OP_CLOSE_UPVALUE 发射。OP_CLOSE_UPVALUE
1039 // bodySlotBase 关闭 slot >= bodySlotBase 的全部 open upvalues（含嵌套块变量）。
1040 // 原 visitBreakStmt 直接发 OP_JUMP 跳到 breakTarget（在 OP_CLOSE_UPVALUE 之后），
1041 // 跳过 upvalue 关闭，导致闭包捕获变 by-reference（三后端不一致）。
1042 const LoopContext& loopCtx = loopStack_.back();
1043 if (loopCtx.needCloseUpvalue && loopCtx.bodySlotBase <= 255) {
1044 chunk_.writeOp(OpCode::OP_CLOSE_UPVALUE, node.line);
1045 chunk_.write(static_cast<uint8_t>(loopCtx.bodySlotBase), node.line);
1046 }
1047 // AUDIT-P1.1 fix: 若 break 在 try-finally 内，发射续跳字节码（先 push 真实目标，再 jump 到 finally 入口）。
1048 // finally 末尾的 OP_FINALLY_END 从 pendingJumpStack_ 取出真实目标续跳。
1049 // 若不在 try-finally 内，走常规路径（直接 OP_JUMP 到 breakTarget）。
1050 std::vector<size_t> realTargetPatches;
1051 if (emitFinallyJump(node.line, realTargetPatches)) {
1052 // 续跳路径：realTargetPatches 中的 patch 待回填到 breakTarget
1053 for (size_t patch : realTargetPatches) {
1054 loopStack_.back().breakJumps.push_back(patch);
1055 }
1056 } else {
1057 // 常规路径：发射 OP_JUMP，目标在循环编译完成后回填
1058 size_t patch = chunk_.code.size();
1059 chunk_.writeOp(OpCode::OP_JUMP, node.line);
1060 chunk_.writeShort(0, node.line);
1061 loopStack_.back().breakJumps.push_back(patch);
1062 }
1063 return;
1064}

1104void Compiler::visitImportStmt(ImportStmt& node) {
1105 // 七特性 MVP 阶段 6：沙箱拦截——禁用文件 import 时仅放行 std/ 内建模块
1106 // （与 Interpreter::visitImportStmt 同口径，保证三后端沙箱行为一致）。
1107 if (RuntimeLimits::RuntimeConfig::instance().sandboxBlocksImport()) {
1108 const std::string& p = node.modulePath;
1109 if (p.rfind("std/", 0) != 0) {
1110 error("沙箱模式禁止导入文件模块: " + p, node.line, node.column);
1111 return;
1112 }
1113 }
1114 // VM-IMPORT: 编译期模块内联——加载模块源码、解析 AST、预扫描全局槽位、
1115 // 内联编译模块语句。模块代码在编译期被"展开"到主程序中，VM 运行时无需模块加载机制。
1116 //
1117 // BUG-AUDIT-MOD-2 fix: 模块隔离（AST 重写 + 作用域分析）
1118 // 在内联编译前，对模块 AST 调用 ModuleTopLevelRenamer::rename，将模块的
1119 // 非导出顶层声明名前缀化为 `__mod_<hash>__<name>`，并递归重写模块内对这些
1120 // 名字的引用。导入方无法用原名访问模块的非导出名，与 Interpreter 的模块
1121 // 隔离语义（独立 Environment）对齐。导出名保持原名，导入方正常访问。
1122 // - run-once: 同一模块多次 import 时仅编译/执行一次（linkedModuleSet_ 保证）
1123 // - 安全保障: export 标记检查（BUG-AUDIT-MOD-1）+ 深度限制（BUG-AUDIT-MOD-3）
1124 // + AST 隔离（BUG-AUDIT-MOD-2，本处实施）
1125
1126 // 1. 路径规范化与安全校验（SEC-1: 路径遍历防护）
1127 // BUG-M4 fix: loaderPath 保留大小写（供 moduleLoader_/预编译加载）；modulePath 为去重键
1128 // （Windows 小写折叠），供 linkedModuleSet_/moduleLoadingSet_/moduleExports_/rename 等所有
1129 // 去重与隔离用途，使同一文件的不同大小写拼写去重为同一模块（对齐 Interpreter）。
1130 std::string loaderPath = normalizeModulePath(node.modulePath);
1131 if (loaderPath.empty()) {
1132 error("模块路径无效: " + node.modulePath, node.line, 0);
1133 return;
1134 }
1135 std::string modulePath = moduleCacheKey(loaderPath);
1136
1137 // P2-11: 命名空间导入字典构造 lambda（run-once 与首次加载路径共用）
1138 // 对每个 export 名 emit: OP_STRING(key常量) + OP_GET_GLOBAL(value)，
1139 // 最后 OP_BUILD_DICT + OP_DEFINE_GLOBAL(namespaceAlias)。
1140 // 修复说明：
1141 // - 使用 OP_STRING（非已废弃的 OP_CONSTANT）
1142 // - 使用 writeShort（BytecodeChunk 无 writeUint16 方法）
1143 // - 使用 addConstant(Value(name))（BytecodeChunk 无 addStringConstant 方法）
1144 // - run-once 路径也需调用：全局槽位已存在，OP_GET_GLOBAL 可直接读取
1145 auto emitNamespaceDict = [this, &node, &modulePath]() {
1146 if (node.namespaceAlias.empty())
1147 return;
1148 auto expIt = moduleExports_.find(modulePath);
1149 if (expIt != moduleExports_.end() && !expIt->second.empty()) {
1150 // 收集导出名并排序（确保三后端字典 key 顺序一致）
1151 std::vector<std::string> sortedExports(expIt->second.begin(), expIt->second.end());
1152 std::sort(sortedExports.begin(), sortedExports.end());
1153 if (sortedExports.size() > 255) {
1154 error("命名空间导入的导出数量超过 255 上限", node.line, 0);
1155 return;
1156 }
1157 // 逐个 push key(字符串常量) + value(全局槽位值)
1158 for (const auto& name : sortedExports) {
1159 // push key: 字符串常量（D5 fix: OP_STRING 替代废弃的 OP_CONSTANT）
1160 uint16_t keyIdx = chunk_.addConstant(Value(name));
1161 chunk_.writeOp(OpCode::OP_STRING, node.line);
1162 chunk_.writeShort(keyIdx, node.line);
1163 // push value: OP_GET_GLOBAL
1164 // AUDIT-R5 R5 fix: 优先读模块导出快照槽位（detach 后 lookupGlobalSlot(name) 已失效），

1578 // namespace 导入由 emitNamespaceDict 构造字典快照，此处跳过。
1579 if (!node.namespaceAlias.empty())
1580 return;
1581 auto slotIt = moduleExportSlots_.find(modulePath);
1582 if (slotIt == moduleExportSlots_.end())
1583 return;
1584 const auto& slotMap = slotIt->second;
1585 auto emitCopy = [&](const std::string& name) {
1586 auto it = slotMap.find(name);
1587 if (it == slotMap.end())
1588 return;
1589 int moduleSlot = it->second;
1590 int mainSlot = allocateGlobalSlot(name); // name 已 detach，分配新槽位；循环未 detach 时返回原槽位
1591 if (mainSlot == moduleSlot)
1592 return; // 循环导入未 detach：同槽位，跳过自拷贝（保持既有共享行为）
1593 chunk_.writeOp(OpCode::OP_GET_GLOBAL, node.line);
1594 chunk_.writeShort(static_cast<uint16_t>(moduleSlot), node.line);
1595 chunk_.writeOp(OpCode::OP_DEFINE_GLOBAL, node.line);
1596 chunk_.writeShort(static_cast<uint16_t>(mainSlot), node.line);
1597 };
1598 if (node.importAll) {
1599 // 全量导入（bare import "m"）：拷贝所有导出（顺序不影响语义）
1600 for (const auto& kv : slotMap)
1601 emitCopy(kv.first);
1602 } else {
1603 for (const auto& name : node.names)
1604 emitCopy(name);
1605 }
1606}

1610// ============================================================
1611CompileResult Compiler::compileModule(const std::string& source, const std::string& modulePath) {
1612 // 独立编译模块源码，生成可序列化的 CompileResult。
1613 // 与 compile() 的关键区别：
1614 // 1. 模块全局槽位从 0 开始独立编号
1615 // 2. 收集 export 名称到 result.moduleExports
1616 // 3. 不设置 moduleLoader_（模块内 import 仍通过 moduleLoader_ 处理，但模块不应再 import 其他模块）
1617 //
1618 // 生成的 CompileResult 可通过 BytecodeCache::storeToFile 序列化为 .minic 文件。
1619
1620 // 1. 词法分析
1621 Lexer lexer;
1622 auto tokens = lexer.scan(source);
1623 if (lexer.getDiagnostics().hasErrors()) {
1624 const auto& diags = lexer.getDiagnostics().all();
1625 if (!diags.empty()) {
1626 error("模块 '" + modulePath + "' 词法错误: " + diags.front().message, diags.front().line,
1627 diags.front().column);
1628 } else {
1629 error("模块 '" + modulePath + "' 词法错误", 0, 0);
1630 }
1631 return {};
1632 }
1633
1634 // 2. 语法分析
1635 Parser parser;
1636 auto moduleAst = parser.parse(tokens);
1637 if (parser.hasErrors() || !moduleAst) {
1638 const auto& diags = parser.getDiagnostics().all();
1639 if (!diags.empty()) {
1640 error("模块 '" + modulePath + "' 语法错误: " + diags.front().message, diags.front().line,
1641 diags.front().column);
1642 } else {
1643 error("模块 '" + modulePath + "' 语法错误", 0, 0);
1644 }
1645 return {};
1646 }
1647
1648 // 3. 重置编译器状态（与 compile() 相同的状态清理）
1649 chunk_ = BytecodeChunk();
1650 chunk_.name = "module:" + modulePath;
1651 chunk_.arity = 0;
1652 chunk_.reserveCode(1024);
1653 varIndex_.clear();
1654 stringConstIndex_.clear();
1655 diagnostics_.clear();
1656 functionChunks_.clear();
1657 currentLocals_.clear();
1658 inFunction_ = false;
1659 classFieldNames_.clear();
1660 outerLocals_.clear();
1661 writebackCounter_ = 0;
1662 peakLocals_ = 0;
1663 blockDepth_ = 0;
1664 blockSaveCounter_ = 0;
1665 compileDepth_ = 0;
1666 globalSlotAllocator_.clear();
1667 innerFunctions_.clear();
1668 innerFunctionSlots_.clear();
1669 linkedModuleSet_.clear();
1670 moduleLoadingSet_.clear();
1671 moduleLoadingStack_.clear();
1672 moduleExports_.clear();
1673 moduleExportSlots_.clear(); // AUDIT-R6 F6 fix: R5 新增成员同步清理（与 compile() 对齐）
1674 moduleExportVars_.clear(); // AUDIT-R6 F6 fix
1675 moduleAsts_.clear();
1676 pendingEnumInfos_.clear();
1677
1678 // 4. 预扫描模块顶层声明，分配全局槽位
1679 preScanModuleGlobals(*moduleAst);
1680
1681 // 5. 收集模块导出名称
1682 collectModuleExports(modulePath, *moduleAst);
1683
1684 // 6. 内联编译模块语句
1685 for (auto& stmt : moduleAst->statements) {
1686 if (stmt)
1687 compileStatement(stmt.get());
1688 }
1689
1690 // 7. 末尾添加 null + return（模块 mainChunk 必须有返回值）
1691 chunk_.writeOp(OpCode::OP_NULL, 0);
1692 chunk_.writeOp(OpCode::OP_RETURN, 0);
1693
1694 // 8. 构建结果
1695 CompileResult result;
1696 result.mainChunk = std::move(chunk_);
1697 result.functionChunks = std::move(functionChunks_);
1698 result.globalSlotCount = globalSlotAllocator_.count();
1699 result.globalSlotNames = std::move(globalSlotAllocator_.mutableNames());
1700 result.enumInfos = std::move(pendingEnumInfos_);
1701
1702 // 9. 填充 moduleExports（从 moduleExports_ map 提取，排序确保确定性）
1703 auto expIt = moduleExports_.find(modulePath);
1704 if (expIt != moduleExports_.end()) {
1705 result.moduleExports.reserve(expIt->second.size());
1706 for (const auto& name : expIt->second) {
1707 result.moduleExports.push_back(name);
1708 }
1709 std::sort(result.moduleExports.begin(), result.moduleExports.end());
1710 }
1711
1712 // 10. 预计算 IP→指令索引映射
1713 result.mainChunk.buildIpMap();
1714 for (auto& kv : result.functionChunks) {
1715 kv.second.buildIpMap();
1716 }
1717
1718 LOG_INFO("模块预编译完成: " + modulePath + " (" + std::to_string(result.globalSlotCount) + " 全局槽, " +
1719 std::to_string(result.functionChunks.size()) + " 函数chunk, " +
1720 std::to_string(result.moduleExports.size()) + " 导出)",
1721 "Compiler");
1722
1723 return result;
1724}

1729RegisterCompileResult Compiler::compileModuleViaRegisterIR(const std::string& source, const std::string& modulePath) {
1730 // 独立编译模块源码，生成可序列化的 RegisterCompileResult。
1731 // 与 compileViaRegisterIR 的关键区别：
1732 // 1. 模块全局槽位从 0 开始独立编号（不与主程序共享）
1733 // 2. 收集 export 名称到 result.moduleExports
1734 // 3. 不设置 moduleLoader_（模块内 import 仍通过 moduleLoader_ 处理）
1735 //
1736 // 生成的 RegisterCompileResult 可通过 BytecodeCache::storeRegisterToFile
1737 // 序列化为 MLRC 格式 .minic 文件。
1738
1739 // 1. 词法分析
1740 Lexer lexer;
1741 auto tokens = lexer.scan(source);
1742 if (lexer.getDiagnostics().hasErrors()) {
1743 const auto& diags = lexer.getDiagnostics().all();
1744 if (!diags.empty()) {
1745 error("模块 '" + modulePath + "' 词法错误: " + diags.front().message, diags.front().line,
1746 diags.front().column);
1747 } else {
1748 error("模块 '" + modulePath + "' 词法错误", 0, 0);
1749 }
1750 return {};
1751 }
1752
1753 // 2. 语法分析
1754 Parser parser;
1755 auto moduleAst = parser.parse(tokens);
1756 if (parser.hasErrors() || !moduleAst) {
1757 const auto& diags = parser.getDiagnostics().all();
1758 if (!diags.empty()) {
1759 error("模块 '" + modulePath + "' 语法错误: " + diags.front().message, diags.front().line,
1760 diags.front().column);
1761 } else {
1762 error("模块 '" + modulePath + "' 语法错误", 0, 0);
1763 }
1764 return {};
1765 }
1766
1767 // 3. 重置编译器状态（与 compileViaRegisterIR 相同的状态清理）
1768 diagnostics_.clear();
1769 linkedModuleSet_.clear();
1770 moduleLoadingSet_.clear();
1771 moduleLoadingStack_.clear();
1772 moduleExports_.clear();
1773 moduleAsts_.clear();
1774 pendingEnumInfos_.clear();
1775 globalSlotAllocator_.clear();
1776
1777 // 3.5 收集模块导出名称（对齐 compileModule 的 collectModuleExports 调用）
1778 // IR builder 有自己的 moduleExports_，但 result.moduleExports 从 Compiler::moduleExports_ 提取。
1779 collectModuleExports(modulePath, *moduleAst);
1780
1781 // 4. AstIRBuilder 构建模块 IR（独立全局槽位）
1782 AstIRBuilder irBuilder;
1783 // 模块编译不转发 precompiledModuleResolver_（模块内的 import 走源码编译）
1784 lastIR_ = irBuilder.build(*moduleAst);
1785 if (irBuilder.hasError()) {
1786 auto irDiags = irBuilder.takeDiagnostics();
1787 for (const auto& d : irDiags.all()) {
1788 diagnostics_.add(d);
1789 }
1790 if (!irDiags.hasErrors()) {
1791 error("IR 构建失败", 0, 0);
1792 }
1793 return {};
1794 }

1796 error("IR 构建失败", 0, 0);
1797 return {};
1798 }

1800 for (auto& ast : irModuleAsts) {
1801 moduleAsts_.push_back(std::move(ast));
1802 }
1803
1804 IRModule* module = irBuilder.getModule();
1805 module->mainFunction = std::move(lastIR_);
1806 module->globalSlotNames = irBuilder.getGlobalSlotNames();
1807
1808 // 5. IR 优化（对齐 compileViaRegisterIR 的优化配置）
1809 if (irOptimize_) {
1810 if (irSSAOptimize_) {
1811 inlinePass(*module);
1812 }
1813 optimizeIR(*module->mainFunction, /*enableCopyPropagation=*/false, /*enableDCE=*/true);
1814 for (auto& fn : module->functions) {
1815 if (fn)
1816 optimizeIR(*fn, /*enableCopyPropagation=*/false, /*enableDCE=*/true);

1825 }
1826 for (auto& fn : module->functions) {
1827 if (fn) {
1828 bool fnSsaBuilt = ssaConstructPass(*fn);
1829 if (fnSsaBuilt) {
1830 gvnPass(*fn);
1831 licmPass(*fn);
1832 ssaDestructPass(*fn);
1833 optimizeIR(*fn, /*enableCopyPropagation=*/false, /*enableDCE=*/true);
1834 }
1835 }
1836 }
1837 }
1838 }
1839
1840 // 6. IR → RegisterBytecode
1841 RegisterBytecodeBackend backend;
1842 if (!backend.lowerModule(*module)) {
1843 error("Register IR lowering 失败", 0, 0);
1844 return {};
1845 }
1846

1848 auto mainChunk = backend.takeChunk();
1849 if (!mainChunk) {
1850 error("Register IR lowering 未生成 main 字节码", 0, 0);
1851 return {};
1852 }
1853 result.mainChunk = std::move(*mainChunk);
1854 result.functionChunks = backend.takeFunctionChunks();
1855 result.globalSlotNames = irBuilder.getGlobalSlotNames();
1856 result.globalSlotCount = static_cast<int>(result.globalSlotNames.size());
1857 result.enumInfos = irBuilder.takeEnumInfos();
1858
1859 // 7. 填充 moduleExports（从 moduleExports_ map 提取，排序确保确定性）

1861 if (expIt != moduleExports_.end()) {
1862 result.moduleExports.reserve(expIt->second.size());

1864 result.moduleExports.push_back(name);
1865 }
1866 std::sort(result.moduleExports.begin(), result.moduleExports.end());
1867 }
1868
1869 // 8. 预计算 IP→指令索引映射

1872 kv.second.buildIpMap();
1873 }

1879 std::to_string(result.functionChunks.size()) + " 函数chunk, " +
1880 std::to_string(result.moduleExports.size()) + " 导出)",
1881 "Compiler-RegModule");
1882
1883 return result;
1884}

1887// P2-11 预编译模块：重命名 OP_CLOSURE/OP_CALL 引用的函数名
1888// ============================================================
1889void Compiler::renameClosureRefs(BytecodeChunk& chunk, const std::string& prefix,
1890 const std::unordered_set<std::string>& exportSet,
1891 const std::unordered_set<std::string>& moduleFunctions) {
1892 // 遍历 code 中的 OP_CLOSURE / OP_CALL 指令，获取 nameIdx → constants[nameIdx].stringVal()，
1893 // 若名称在 moduleFunctions 中且不在 exportSet 中，则前缀化为 `prefix + name` 并更新常量。
1894 //
1895 // 必须同时处理 OP_CLOSURE 和 OP_CALL：
1896 // - OP_CLOSURE 创建闭包值，引用函数名常量；非导出函数的闭包值需重命名，
1897 // 否则 functionChunks_.find(oldName) 在主程序中会找到错误的函数（如主程序同名函数）。
1898 // - OP_CALL 按名查找函数；模块内调用非导出函数（如 `main` 调用 `helper`）也需重命名，
1899 // 否则运行时按原名查找会命中主程序的同名函数，破坏模块隔离语义。
1900 //
1901 // 通过 moduleFunctions 精确判断 OP_CALL 引用的是模块内函数还是内置函数：
1902 // - 模块内非导出函数（在 moduleFunctions 中，不在 exportSet 中）→ 前缀化
1903 // - 模块内导出函数（在 exportSet 中）→ 保持原名
1904 // - 内置函数（不在 moduleFunctions 中）→ 保持原名
1905 //
1906 // 格式：
1907 // OP_CLOSURE: [op(1B), nameIdx(2B), upvalueCount(1B), ...upvalueDescs]
1908 // OP_CALL: [op(1B), nameIdx(2B), argCount(1B)]
1909 size_t offset = 0;
1910 while (offset < chunk.code.size()) {
1911 if (offset >= chunk.code.size())
1912 break;
1913 OpCode op = static_cast<OpCode>(chunk.code[offset]);
1914 // L18 eng-tailcall: OP_TAIL_CALL 布局同 OP_CALL（nameIdx 在 offset+1..+2），
1915 // 同样需要模块隔离重命名，否则尾调用按原名命中主程序同名函数。
1916 if (op == OpCode::OP_CLOSURE || op == OpCode::OP_CALL || op == OpCode::OP_TAIL_CALL) {
1917 if (offset + 2 < chunk.code.size()) {
1918 uint16_t nameIdx = static_cast<uint16_t>(chunk.code[offset + 1]) |
1919 (static_cast<uint16_t>(chunk.code[offset + 2]) << 8);
1920 if (nameIdx < chunk.constants.size() && chunk.constants[nameIdx].isString()) {
1921 const std::string& fnName = chunk.constants[nameIdx].stringVal();
1922 // 仅重命名模块内非导出函数：在 moduleFunctions 中但不在 exportSet 中
1923 if (moduleFunctions.count(fnName) > 0 && exportSet.count(fnName) == 0) {
1924 chunk.constants[nameIdx] = Value(prefix + fnName);
1925 }
1926 }
1927 }
1928 }
1929 offset += chunk.instructionSizeAt(offset);
1930 }
1931}

1934// L11 预编译模块（RegisterVM）：重命名 REG_CALL/REG_MAKE_CLOSURE 引用的函数名
1935// ============================================================
1936void Compiler::renameRegClosureRefs(RegBytecodeChunk& chunk, const std::string& prefix,
1937 const std::unordered_set<std::string>& exportSet,
1938 const std::unordered_set<std::string>& moduleFunctions) {
1939 // 遍历 code 中的 REG_CALL / REG_MAKE_CLOSURE 指令，获取 nameIdx →
1940 // constants[nameIdx].stringVal()，若名称在 moduleFunctions 中且不在 exportSet 中，
1941 // 则前缀化为 `prefix + name` 并更新常量。
1942 //
1943 // 指令格式（nameIdx 位置相同）：
1944 // REG_CALL: [op(1B), dst(1B), nameIdx(2B LE), argCount(1B), args...]
1945 // REG_MAKE_CLOSURE: [op(1B), dst(1B), nameIdx(2B LE), uvCount(1B), uvDescs...]
1946 // nameIdx 在 offset+2..offset+3
1947 size_t offset = 0;
1948 while (offset < chunk.code.size()) {
1949 RegOp op = static_cast<RegOp>(chunk.code[offset]);
1950 // L18 eng-tailcall: REG_TAIL_CALL 布局同 REG_CALL（nameIdx 在 offset+2..+3），
1951 // 同样需要模块隔离重命名。
1952 if (op == RegOp::REG_CALL || op == RegOp::REG_MAKE_CLOSURE || op == RegOp::REG_TAIL_CALL) {
1953 // nameIdx 在 offset+2..offset+3（2B LE）
1954 if (offset + 3 < chunk.code.size()) {
1955 uint16_t nameIdx = static_cast<uint16_t>(chunk.code[offset + 2]) |
1956 (static_cast<uint16_t>(chunk.code[offset + 3]) << 8);
1957 if (nameIdx < chunk.constants.size() && chunk.constants[nameIdx].isString()) {
1958 const std::string& fnName = chunk.constants[nameIdx].stringVal();
1959 // 仅重命名模块内非导出函数：在 moduleFunctions 中但不在 exportSet 中
1960 if (moduleFunctions.count(fnName) > 0 && exportSet.count(fnName) == 0) {
1961 chunk.constants[nameIdx] = Value(prefix + fnName);
1962 }
1963 }
1964 }
1965 }
1966 offset += chunk.instructionSizeAt(offset);
1967 }
1968}

1970// ============================================================
1971// P2-11 预编译模块：加载 .minic 并合并到当前编译
1972// ============================================================
1973bool Compiler::loadPrecompiledModule(const std::string& modulePath, ImportStmt& node) {
1974 // 策略概述：
1975 // 1. 通过 precompiledModuleResolver_ 获取 .minic 文件路径
1976 // 2. BytecodeCache::tryLoadFromFile 加载预编译 CompileResult
1977 // 3. 为模块的 globalSlotNames 在主程序分配对应全局槽位
1978 // 4. 构建 relocationMap: moduleSlot → mainSlot
1979 // 5. 模块 mainChunk 转为函数 chunk (__mod_<hash>__init)，
1980 // 重命名非导出函数引用，重定位全局槽位
1981 // 6. 模块 functionChunks 重命名+重定位后合并到主程序
1982 // 7. 主 chunk emit OP_CLOSURE + OP_CALL_EXPR + OP_POP 调用模块初始化
1983 // 8. 填充 moduleExports_ 供后续具名导入验证
1984
1985 if (!precompiledModuleResolver_)
1986 return false;
1987
1988 std::string minicPath = precompiledModuleResolver_(modulePath);
1989 if (minicPath.empty())
1990 return false;
1991
1992 // L12: 推导源码路径（与 .minic 同目录、同 stem、.mini 扩展名）
1993 // 用于 tryLoadFromFile 的源码失效校验：若源码 mtime/hash 与 .minic 头部
1994 // 嵌入值不匹配（源码已修改），返回 nullopt 回退到源码编译。
1995 // 推导失败（无 .mini 文件）→ sourcePath 为空，tryLoadFromFile 跳过校验（向后兼容）。
1996 std::string sourcePath;
1997 {
1998 namespace fs = std::filesystem;
1999 fs::path minicP(minicPath);
2000 if (minicP.has_stem()) {
2001 fs::path candidate = minicP.parent_path() / (minicP.stem().string() + ".mini");
2002 std::error_code ec;
2003 if (fs::exists(candidate, ec)) {
2004 sourcePath = candidate.string();
2005 }
2006 }
2007 }
2008
2009 BytecodeCache cache;
2010 auto moduleResult = cache.tryLoadFromFile(minicPath, sourcePath);
2011 if (!moduleResult)
2012 return false; // 加载失败（文件不存在/损坏/校验失败/源码已修改），回退到源码编译
2013
2014 // 生成模块前缀（用于非导出函数名重命名，避免与主程序冲突）
2015 // FNV-1a 哈希确保不同模块路径产生不同前缀
2016 uint64_t hash = 0xcbf29ce484222325ULL;
2017 for (unsigned char c : modulePath) {
2018 hash ^= c;
2019 hash *= 0x100000001b3ULL;
2020 }
2021 std::string modPrefix = "__mod_" + std::to_string(hash) + "_";
2022
2023 // 构建导出名称集合（用于决定哪些函数名不重命名）
2024 std::unordered_set<std::string> exportSet(moduleResult->moduleExports.begin(), moduleResult->moduleExports.end());
2025
2026 // P2-11 fix: 构建模块所有函数名集合（导出+非导出），用于 renameClosureRefs
2027 // 精确判断 OP_CALL/OP_CLOSURE 引用的是模块内函数还是内置函数。
2028 // 仅模块内非导出函数需要前缀化；内置函数（如 print）保持原名。
2029 std::unordered_set<std::string> moduleFunctions;
2030 for (const auto& [fnName, _] : moduleResult->functionChunks) {
2031 moduleFunctions.insert(fnName);
2032 }
2033
2034 // 1. 为模块的全局槽位在主程序分配对应槽位
2035 // relocationMap[moduleSlot] = mainSlot（-1 表示不重定位）
2036 std::vector<int> relocationMap(moduleResult->globalSlotCount, -1);
2037 for (int i = 0; i < moduleResult->globalSlotCount; ++i) {
2038 const std::string& name = moduleResult->globalSlotNames[i];
2039 // 导出名直接在主程序分配槽位（导入方需通过此槽位访问）
2040 // 非导出名也分配槽位（模块内部代码需要），但名称被前缀化避免主程序访问
2041 std::string mainName = (exportSet.count(name) > 0) ? name : (modPrefix + name);
2042 int mainSlot = allocateGlobalSlot(mainName);
2043 relocationMap[i] = mainSlot;
2044 }
2045
2046 // 2. 处理模块 mainChunk：转为函数 chunk
2047 std::string initFnName = modPrefix + "_init";
2048 BytecodeChunk& moduleMainChunk = moduleResult->mainChunk;
2049 moduleMainChunk.name = initFnName;
2050 // 重命名非导出函数引用（OP_CLOSURE/OP_CALL 的常量池条目）
2051 renameClosureRefs(moduleMainChunk, modPrefix, exportSet, moduleFunctions);
2052 // 重定位全局槽位引用
2053 moduleMainChunk.relocateGlobalSlots(relocationMap);
2054
2055 // 3. 处理模块 functionChunks：重命名 + 重定位
2056 for (auto& [oldName, fnChunk] : moduleResult->functionChunks) {

2225void Compiler::visitTryStmt(TryStmt& node) {
2226 // 编译模式:
2227 // OP_TRY_BEGIN <catchOffset>
2228 // <try block>
2229 // OP_TRY_END
2230 // OP_JUMP <afterCatch>
2231 // catchIp:
2232 // <bind exception to catchVar>
2233 // <catch block>
2234 // afterCatch:
2235 //
2236 // BUG-AUDIT-FINALLY-1: 如果有 finally 块，外层再包一个 OP_TRY_BEGIN/END：
2237 // OP_TRY_BEGIN <finallyCatchOffset> ← 外层 try（捕获异常路径）
2238 // <内层 try-catch>
2239 // OP_TRY_END ← 弹出外层 handler
2240 // <finally block> ← 正常路径执行 finally
2241 // OP_JUMP <afterFinally>
2242 // finallyCatchIp: ← 异常路径
2243 // <finally block>（重复一次）
2244 // OP_THROW ← re-throw（异常值在栈顶）
2245 // afterFinally:
2246 //
2247 // break/continue 与 finally（AUDIT-P1.1 修复，三后端一致）：break/continue 在
2248 // try-finally 内时，先 push 真实跳转目标到 pendingJumpStack_，再 jump 到 finally
2249 // 入口，finally 末尾的 OP_FINALLY_END 从栈取出目标续跳——因此 StackVM/RegisterVM
2250 // 与 Interpreter（走 loopFlow_ 状态标志，try 块正常完成后继续执行 finally）行为一致。
2251 // return 三后端均执行 finally 后再传播（L4 fix，一致）。
2252 // 异常值在异常路径的 finally 执行期间保留在栈顶（Block 是栈平衡的），
2253 // OP_THROW pop 并 re-throw。若 finally 自身 throw，throwException 会截断
2254 // 栈到外层 handler 的 stackBase（丢弃原异常值），新异常正常传播。
2255 //
2256 // 拆分说明（原 357 行单函数 → orchestrator + 3 子阶段）：
2257 // - emitTryBlock：OP_TRY_BEGIN + try body + OP_TRY_END + skip-catch OP_JUMP
2258 // - emitCatchBlock：catchOffset 回填 + catch 变量绑定 + catch body（含 cleanup wrap）
2259 // + cleanup + OP_CLOSE_UPVALUE + 恢复 currentLocals_ + 回填 afterCatch
2260 // - emitFinallyBlock：OP_TRY_END + finallyEntry 回填 + finally body + OP_FINALLY_END
2261 // + 异常路径 + 回填 afterFinally
2262 // 控制流不变量：emitCatchBlock 返回 false 时直接 return（不 pop tryFinallyStack_，
2263 // 对齐原实现 catch 块 early return 行为）；emitFinallyBlock 早返回时不 pop
2264 // tryFinallyStack_，由本函数统一 pop（单次 pop 语义）。
2265
2266 // AUDIT-P1.1 fix: push try-finally 编译期上下文
2267 tryFinallyStack_.push_back({node.finallyBlock != nullptr, 0, {}, {}});
2268
2269 // 0. 如果有 finally，发射外层 OP_TRY_BEGIN
2270 size_t outerTryBeginIp = 0;
2271 size_t finallyCatchOffsetPatch = std::string::npos;
2272 if (node.finallyBlock) {
2273 outerTryBeginIp = chunk_.code.size();
2274 chunk_.writeOp(OpCode::OP_TRY_BEGIN, node.line);
2275 finallyCatchOffsetPatch = chunk_.code.size();
2276 chunk_.writeShort(0, node.line); // 占位
2277 ++tryDepth_; // 外层 try 计入深度，使 break/continue 发射对应 OP_TRY_END
2278 }
2279
2280 // BUG-AUDIT-FINALLY-1: try-finally（无 catch）路径。
2281 // catchVarName 为空表示无 catch 子句，异常不应被捕获。
2282 // 外层 OP_TRY_BEGIN（finallyCatchOffset）会捕获异常 → 执行 finally → rethrow。
2283 // 跳过内层 try-catch 的全部字节码（OP_TRY_BEGIN/catchOffset/catch 变量绑定/cleanup）。
2284 if (!node.catchVarName.empty()) {
2285 TryCatchPatchInfo patchInfo;
2286 emitTryBlock(node, patchInfo);
2287 if (!emitCatchBlock(node, patchInfo)) {
2288 // 保留原始控制流：原 catch 块 early return 不 pop tryFinallyStack_。
2289 // tryFinallyStack_ 在此路径下不 pop（与原实现一致）。
2290 return;
2291 }
2292 } else {
2293 // try-finally（无 catch）：只编译 try 块，不发射内层 try-catch。
2294 // 外层 OP_TRY_BEGIN（finallyCatchOffset）会捕获异常 → 执行 finally → rethrow。
2295 if (node.tryBlock) {
2296 compileNode(node.tryBlock.get());
2297 }
2298 }
2299
2300 // 9. BUG-AUDIT-FINALLY-1: finally 块字节码
2301 if (node.finallyBlock) {
2302 emitFinallyBlock(node, outerTryBeginIp, finallyCatchOffsetPatch);
2303 }
2304
2305 // AUDIT-P1.1 fix: pop try-finally 编译期上下文
2306 tryFinallyStack_.pop_back();
2307
2308 return;
2309}

2318 chunk_.writeOp(OpCode::OP_TRY_BEGIN, node.line);

2321

2332 info.skipCatchJumpPatch = chunk_.code.size();

2337bool Compiler::emitCatchBlock(TryStmt& node, const TryCatchPatchInfo& info) {
2338 // L27 重构：原 200 行单函数按阶段拆分为 5 个子函数分发，降低圈复杂度。
2339 // 所有 BUG-7a/7b/7c/BUG-AUDIT-EXC-*/BUG-TRY-1/L1 fix 不变量原样保留。
2340 // 阶段顺序：回填 catchOffset → 绑定 catch 变量 → 编译 catch body（含 cleanup wrap）
2341 // → 恢复作用域 → 回填 afterCatch 跳转。
2342
2343 // 阶段 1：回填 catchOffset
2344 if (!patchCatchOffset(node, info)) {
2345 return false;
2346 }
2347
2348 // 阶段 2：绑定 catch 变量（函数内/顶层，含遮蔽保护）
2349 CatchVarBindInfo bind;
2350 if (!bindCatchVariable(node, bind)) {
2351 return false;
2352 }
2353
2354 // 阶段 3：编译 catch body（含 cleanup wrap），返回 cleanup 跳转 patch
2355 // 注：若 cleanupThrowOffset 溢出，emitCatchBodyWithCleanup 已完成 restoreMapping +
2356 // currentLocals_ 恢复并返回 false，此时不可再调用 restoreCatchScope（会重复恢复 +
2357 // 错误 emit OP_CLOSE_UPVALUE），直接返回 false 保持原 early-return 语义。
2358 size_t skipCleanupThrowJumpPatch = std::string::npos;
2359 if (!emitCatchBodyWithCleanup(node, bind, skipCleanupThrowJumpPatch)) {
2360 return false;
2361 }
2362
2363 // 阶段 4：恢复 catch 作用域（restoreMapping + OP_CLOSE_UPVALUE + closeSlotRanges + 恢复 currentLocals_）
2364 restoreCatchScope(node, bind);
2365
2366 // 阶段 5：回填 afterCatch 跳转目标
2367 if (!patchSkipCatchJumps(node, info, skipCleanupThrowJumpPatch)) {
2368 return false;
2369 }
2370
2371 return true;
2372}

2388bool Compiler::bindCatchVariable(TryStmt& node, CatchVarBindInfo& bind) {
2389 // 5. 在 catchIp 处：异常值已在栈顶，绑定到 catch 变量

2406 peakLocals_ = std::max(peakLocals_, static_cast<int>(currentLocals_.size()));

2443bool Compiler::emitCatchBodyWithCleanup(TryStmt& node, const CatchVarBindInfo& bind,
2444 size_t& outSkipCleanupThrowJumpPatch) {
2445 // 6. 编译 catch 块
2446 // BUG-TRY-1 fix: 若 catch 块内 throw，原实现跳过清理代码，导致 catch 变量泄漏、
2447 // 被遮蔽的全局值未恢复。修复：用 OP_TRY_BEGIN 包装 catch 块，捕获内层 throw，
2448 // 跳到 cleanupThrowIp 执行清理代码后 OP_THROW rethrow。
2449 // 字节码布局：
2450 // catchIp: <bind exception>
2451 // OP_TRY_BEGIN <cleanupThrowOffset>
2452 // <catch block>
2453 // OP_TRY_END
2454 // <cleanup code> ← 正常路径
2455 // OP_JUMP <afterCatch>
2456 // cleanupThrowIp:
2457 // <cleanup code> ← 异常路径（复制）
2458 // OP_THROW ← rethrow（异常值已在栈顶）
2459 // afterCatch:
2460 //
2461 // cleanup 字节码栈平衡为 0（OP_DELETE_VAR 不影响栈；OP_GET_VAR+OP_SET_GLOBAL+OP_DELETE_VAR = 0），
2462 // 异常值保持在栈顶，OP_THROW 可正确 rethrow。
2463 outSkipCleanupThrowJumpPatch = std::string::npos;
2464 bool needsCleanupWrap = bind.needCatchVarCleanup || bind.hasShadowedGlobal;
2465 size_t innerTryBeginIp = 0;
2466 size_t innerCatchOffsetPatch = std::string::npos;
2467 if (needsCleanupWrap) {
2468 innerTryBeginIp = chunk_.code.size();
2469 chunk_.writeOp(OpCode::OP_TRY_BEGIN, node.line);
2470 innerCatchOffsetPatch = chunk_.code.size();
2471 chunk_.writeShort(0, node.line); // 占位，稍后回填为 cleanupThrowOffset

2540void Compiler::restoreCatchScope(TryStmt& node, const CatchVarBindInfo& bind) {

2544 }

2547 // 必须发射 OP_CLOSE_UPVALUE 关闭指向该 slot 的 open upvalue。否则后续代码声明新
2548 // 局部变量会复用该 slot 覆盖原值，逃逸的闭包通过 upvalue 读取到错误值（等价悬垂引用）。

2558 // 的全部 open ranges。closeSlotRanges 必须在 currentLocals_ 恢复前调用，

2564}

2584void Compiler::emitFinallyBlock(TryStmt& node, size_t outerTryBeginIp, size_t finallyCatchOffsetPatch) {
2585 --tryDepth_;
2586 chunk_.writeOp(OpCode::OP_TRY_END, node.line); // 弹出外层 try 处理器
2587
2588 // AUDIT-P1.1 fix: 记录 finally 入口地址，回填 break/continue 的续跳 patches。
2589 // finallyEntryIp 是正常路径 finally 块的入口（OP_TRY_END 之后的第一条指令）。
2590 size_t finallyEntryIp = chunk_.code.size();
2591 tryFinallyStack_.back().finallyEntryIp = finallyEntryIp;
2592 // 回填所有 pendingJumpPatches（break/continue 的 OP_JUMP 目标 = finallyEntryIp）
2593 for (size_t patch : tryFinallyStack_.back().pendingJumpPatches) {
2594 if (finallyEntryIp > 65535) {

2596 // 不 pop tryFinallyStack_，由 visitTryStmt 统一 pop（单次 pop 语义）
2597 return;
2598 }
2599 uint16_t target = static_cast<uint16_t>(finallyEntryIp);
2600 chunk_.code[patch] = static_cast<uint8_t>(target & 0xFF);
2601 chunk_.code[patch + 1] = static_cast<uint8_t>((target >> 8) & 0xFF);
2602 }
2603 // 回填所有 pendingTargetPatches（内层 break/continue 的 OP_PUSH_JUMP_TARGET 目标 = finallyEntryIp）
2604 for (size_t patch : tryFinallyStack_.back().pendingTargetPatches) {
2605 if (finallyEntryIp > 65535) {
2606 error("finally 块入口偏移溢出 65535", node.line, 0);
2607 // 不 pop tryFinallyStack_，由 visitTryStmt 统一 pop（单次 pop 语义）
2608 return;

2611 chunk_.code[patch] = static_cast<uint8_t>(target & 0xFF);
2612 chunk_.code[patch + 1] = static_cast<uint8_t>((target >> 8) & 0xFF);
2613 }

2615 // AUDIT-R6 B1 fix(A): 发射 finally 体（两个副本）期间暂时弹出自身条目。
2616 // 原实现中 finally 体内的 break/continue/return 经 emitFinallyJump 把自身当作
2617 // enclosing finally：新增的 pendingJumpPatches 永不回填（回填点已过），OP_JUMP
2618 // 目标保持占位 0 → 运行时跳回程序开头无限重执行（实证：异常路径每轮泄漏
2619 // 异常值至“栈溢出”）。弹出后 finally 体内的 break 正确链到更外层 finally 或
2620 // 直走常规 breakJumps 回填路径；结束后压回，保持调用方统一 pop 语义。

2622 tryFinallyStack_.pop_back();

2624 // 正常路径：执行 finally
2625 compileNode(node.finallyBlock.get());
2626 // AUDIT-P1.1 fix: finally 块末尾追加 OP_FINALLY_END。
2627 // 若 pendingJumpStack_ 非空（break/continue 触发），pop 目标并跳转；
2628 // 否则继续执行（正常完成）。异常路径的 finally 末尾保持 OP_THROW（re-throw）。
2629 chunk_.writeOp(OpCode::OP_FINALLY_END, node.line);
2630
2631 // 跳过异常路径
2632 size_t skipFinallyExceptionJumpPatch = chunk_.code.size();
2633 chunk_.writeOp(OpCode::OP_JUMP, node.line);
2634 chunk_.writeShort(0, node.line); // 占位
2635

2637 size_t finallyCatchIp = chunk_.code.size();
2638 size_t finallyCatchOffset = finallyCatchIp - (outerTryBeginIp + 3);
2639 if (finallyCatchOffset > 65535) {

2641 // AUDIT-R6 B1: 提前返回前压回自身条目，保持调用方统一 pop 语义

2643 return;

2649 // AUDIT-R6 B1 fix(B): 异常值先从栈上暂存到唯一临时全局（OP_DEFINE_VAR pop），

2659 compileNode(node.finallyBlock.get());

2661 chunk_.writeShort(finallyExcIdx, node.line);
2662 chunk_.writeOp(OpCode::OP_DELETE_VAR, node.line);

2668 tryFinallyStack_.push_back(std::move(selfFinallyCtx));
2669
2670 // afterFinally
2671 size_t afterFinally = chunk_.code.size();

2673 error("代码量过大，跳转目标溢出 65535", node.line, 0);
2674 // 不 pop tryFinallyStack_，由 visitTryStmt 统一 pop（单次 pop 语义）
2675 return;
2676 }
2677 uint16_t afterFinallyTarget = static_cast<uint16_t>(afterFinally);
2678 chunk_.code[skipFinallyExceptionJumpPatch + 1] = static_cast<uint8_t>(afterFinallyTarget & 0xFF);
2679 chunk_.code[skipFinallyExceptionJumpPatch + 2] = static_cast<uint8_t>((afterFinallyTarget >> 8) & 0xFF);
2680}

2714void Compiler::visitBlock(Block& node) {
2715 auto savedLocals = currentLocals_;
2716
2717 if (!inFunction_) {
2718 blockDepth_++;
2719
2720 // 收集块作用域中将要声明的变量名，以便在编译前保存被遮蔽的全局变量
2721 std::vector<std::pair<std::string, std::string>> shadowedSaves; // (blockVarName, tempSaveName)
2722 std::vector<std::pair<std::string, int>> removedSlots; // H1: (name, slot) 用于恢复
2723 for (auto& stmt : node.statements) {
2724 if (stmt && stmt->nodeType == NodeType::NODE_VAR_DECL) {
2725 VarDecl* vd = static_cast<VarDecl*>(stmt.get());
2726 int gsSlot = lookupGlobalSlot(vd->name);
2727 if (gsSlot >= 0) {
2728 std::string saveName = "__blk_save_" + std::to_string(blockDepth_) + "_" +
2729 std::to_string(blockSaveCounter_++) + "_" + vd->name;
2730 shadowedSaves.push_back({vd->name, saveName});
2731 // A2: 使用槽位操作码保存全局变量值
2732 uint16_t saveIdx = identifierIndex(saveName);
2733 chunk_.writeOp(OpCode::OP_GET_GLOBAL, vd->line);
2734 chunk_.writeShort(static_cast<uint16_t>(gsSlot), vd->line);
2735 chunk_.writeOp(OpCode::OP_DEFINE_VAR, vd->line);
2736 chunk_.writeShort(saveIdx, vd->line);
2737 }
2738 }
2739 }
2740
2741 // H1 fix: 保存全局值后，移除被遮蔽的全局槽位条目
2742 // 使块内 compileVarDecl/compileVarRef/compileAssignment 不命中全局槽位，
2743 // 改用 OP_DEFINE_VAR/OP_GET_VAR/OP_SET_VAR → globals_ 路径
2744 for (auto& [varName, saveName] : shadowedSaves) {
2745 int slot = globalSlotAllocator_.removeMapping(varName); // B4: 临时遮蔽
2746 if (slot >= 0) {
2747 removedSlots.push_back({varName, slot});
2748 }
2749 }
2750
2751 // 编译块体
2752 for (auto& stmt : node.statements) {
2753 compileStatement(stmt.get());
2754 }
2755
2756 // 找出块作用域内新声明的变量
2757 std::vector<std::string> blockVars;
2758 for (auto& [name, slot] : currentLocals_) {
2759 if (savedLocals.find(name) == savedLocals.end()) {
2760 blockVars.push_back(name);
2761 }
2762 }
2763
2764 // 恢复外层作用域（P28: swap）
2765 currentLocals_.swap(savedLocals);
2766 blockDepth_--;
2767
2768 // H1 fix: 恢复被移除的全局槽位条目（必须在恢复字节码之前，使 lookupGlobalSlot 正确）
2769 for (auto& [name, slot] : removedSlots) {
2770 globalSlotAllocator_.restoreMapping(name, slot); // B4: 恢复遮蔽
2771 }
2772
2773 // 清理块作用域变量并恢复被遮蔽的全局变量
2774 for (auto& [varName, saveName] : shadowedSaves) {
2775 // A2: 使用槽位操作码恢复全局变量值
2776 int gsSlot = lookupGlobalSlot(varName);
2777 uint16_t saveIdx = identifierIndex(saveName);
2778 chunk_.writeOp(OpCode::OP_GET_VAR, node.line);
2779 chunk_.writeShort(saveIdx, node.line);
2780 if (gsSlot >= 0) {
2781 chunk_.writeOp(OpCode::OP_SET_GLOBAL, node.line);
2782 chunk_.writeShort(static_cast<uint16_t>(gsSlot), node.line);
2783 } else {
2784 uint16_t origIdx = identifierIndex(varName);
2785 chunk_.writeOp(OpCode::OP_SET_VAR, node.line);
2786 chunk_.writeShort(origIdx, node.line);
2787 }
2788 // 删除临时保存变量
2789 chunk_.writeOp(OpCode::OP_DELETE_VAR, node.line);
2790 chunk_.writeShort(saveIdx, node.line);
2791 }
2792
2793 // 删除块作用域变量（含被遮蔽变量的 globals_ 残留值）
2794 for (auto& name : blockVars) {
2795 uint16_t nameIdx = identifierIndex(name);
2796 chunk_.writeOp(OpCode::OP_DELETE_VAR, node.line);
2797 chunk_.writeShort(nameIdx, node.line);
2798 }
2799 } else {
2800 // 函数内块作用域：局部变量使用栈槽。
2801 // BUG-AUDIT-CLOSE-1 fix: 块退出时需发射 OP_CLOSE_UPVALUE，关闭指向本块 slot 的
2802 // open upvalues，使闭包捕获块退出时刻的值快照（by-value），对齐 IR 路径
2803 // leaveBlockScope（AstIRBuilder.cpp leaveBlockScope）和 Interpreter 的 closeCapturedVariables。
2804 // 原实现注释"VM 帧退出时自动释放"是误解——closeUpvaluesFrom 在 OP_RETURN 时
2805 // 关闭 upvalue 产生的是函数返回时刻的快照（by-reference），与块退出快照语义不一致。
2806 size_t slotBase = currentLocals_.size(); // 块内第一个新 slot 的编号
2807 for (auto& stmt : node.statements) {
2808 compileStatement(stmt.get());
2809 }
2810 // 块退出时关闭指向 slot >= slotBase 的全部 open upvalues
2811 if (currentLocals_.size() > slotBase && slotBase <= 255) {
2812 chunk_.writeOp(OpCode::OP_CLOSE_UPVALUE, node.line);
2813 chunk_.write(static_cast<uint8_t>(slotBase), node.line);
2814 }
2815 closeSlotRanges(slotBase); // L1 fix: 回填本块内变量 range 的 endIp
2816 currentLocals_.swap(savedLocals); // P28: swap
2817 }
2818 return;
2819}
