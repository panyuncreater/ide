120// 公式型操作（pops 依赖操作数，在 irStackEffect 的 switch 分支处理）：恰 10 条
121constexpr bool isFormulaOp(IROp op) {
122 switch (op) {
123 case IROp::CALL:
124 case IROp::TAIL_CALL:
125 case IROp::CALL_EXPR:
126 case IROp::METHOD_CALL:
127 case IROp::SUPER_CALL:
128 case IROp::CLASS_NEW:
129 case IROp::BUILD_ARRAY:
130 case IROp::BUILD_DICT:
131 case IROp::BUILD_TUPLE:
132 case IROp::BUILD_ENUM_VARIANT:
133 return true;
134 default:
135 return false;
136 }
137}
138
139// 编译期完备性校验：固定效应表中每个 op 恰出现一次，且全部非公式型
140// IROp（除 PHI——SSA 中间产物，lowerInstruction 的 default 分支显式拒绝，
141// 不应到达效应查询）均已登记。新增 IROp 未同步本表时编译失败。
142constexpr bool tableIsComplete() {
143 constexpr size_t kTableSize = sizeof(kFixedStackEffect) / sizeof(kFixedStackEffect[0]);
144 // 1. 表内无重复
145 for (size_t i = 0; i < kTableSize; ++i) {
146 for (size_t j = i + 1; j < kTableSize; ++j) {
147 if (kFixedStackEffect[i].op == kFixedStackEffect[j].op)
148 return false;
149 }
150 }
151 // 2. 非公式 op 全部在表中（PHI 除外——不可 lower，见注释）
152 for (size_t v = 0; v < static_cast<size_t>(IROp::IROp_COUNT); ++v) {
153 const IROp op = static_cast<IROp>(v);
154 if (op == IROp::IROp_COUNT)
155 continue;
156 if (isFormulaOp(op) || op == IROp::PHI)
157 continue;
158 bool found = false;
159 for (size_t i = 0; i < kTableSize; ++i) {
160 if (kFixedStackEffect[i].op == op) {
161 found = true;
162 break;
163 }
164 }
165 if (!found)
166 return false;
167 }
168 return true;
169}
170
171static_assert(tableIsComplete(), "kFixedStackEffect 未覆盖全部非公式 IROp（或存在重复登记）——"
172 "新增 IROp 时必须同步补充效应表或 irStackEffect 的公式分支");
173

176bool BytecodeIRBackend::irStackEffect(const IRInstruction& instr, StackEffect& fx) {
177 // ---- 公式型（pops 依赖操作数）优先：调用点 pops=公式、pushes=1（返回值）；
178 // 参数在调用点划入被调帧（bp=H-argc，VMCalls.cpp setupFunctionCallFrame），
179 // 返回时 resize 统一回收，与逐字节码模拟等价 ----
180 switch (instr.op) {
181 case IROp::CALL: // [dest, name, argc, ...] -> pops=argc（VMCalls.cpp 各路径 pop argc）
182 case IROp::TAIL_CALL: // 布局同 CALL；降级路径 = executeCall，净效应同
183 if (instr.operands.size() < 3)
184 return false;
185 fx = {static_cast<int>(instr.operands[2].index), 1};
186 return true;
187 case IROp::CALL_EXPR: // [dest, closure_vreg, argc, ...] -> pops=argc+1（闭包值 + 参数）
188 if (instr.operands.size() < 3)
189 return false;
190 fx = {static_cast<int>(instr.operands[2].index) + 1, 1};
191 return true;
192 case IROp::METHOD_CALL: // [dest, obj, name, argc, recvVar, recvSlot] -> pops=argc+1
193 // （receiver 在 peek(argc) 处，VMCalls.cpp executeMethodCall）
194 if (instr.operands.size() < 4)
195 return false;
196 fx = {static_cast<int>(instr.operands[3].index) + 1, 1};
197 return true;
198 case IROp::SUPER_CALL: // [dest, this, name, class_idx, argc, classIdx] -> 同 METHOD_CALL
199 if (instr.operands.size() < 5)
200 return false;
201 fx = {static_cast<int>(instr.operands[4].index) + 1, 1};
202 return true;
203 case IROp::CLASS_NEW: // [dest, name, argc, ...] -> pop argc、push 实例（VMCalls.cpp:1267-1281）
204 if (instr.operands.size() < 3)
205 return false;
206 fx = {static_cast<int>(instr.operands[2].index), 1};
207 return true;
208 case IROp::BUILD_ARRAY: // [dest, count, ...] -> pop count、push 数组
209 case IROp::BUILD_TUPLE:
210 if (instr.operands.size() < 2)
211 return false;
212 fx = {static_cast<int>(instr.operands[1].index), 1};
213 return true;
214 case IROp::BUILD_DICT: // [dest, pairCount, ...] -> key/val 交替 pop 2*pairCount
215 if (instr.operands.size() < 2)
216 return false;
217 fx = {static_cast<int>(instr.operands[1].index) * 2, 1};
218 return true;
219 case IROp::BUILD_ENUM_VARIANT: // [dest, enumName, variantName, argc, ...]
220 if (instr.operands.size() < 4)
221 return false;
222 fx = {static_cast<int>(instr.operands[3].index), 1};
223 return true;
224 default:
225 break;
226 }
227
228 // ---- 固定效应：查表 ----
229 for (const auto& e : kFixedStackEffect) {
230 if (e.op == instr.op) {
231 fx = {static_cast<int>(e.pops), static_cast<int>(e.pushes)};
232 return true;
233 }
234 }
235 // 表 + 公式均未覆盖：仅可能是 PHI（SSA 中间产物，不可 lower）或
236 // 编译期完备性被绕过的新增 op——保守返回 false，交由 checkStackBalance 拒绝
237 return false;
238}

392void BytecodeIRBackend::recordJumpDepth(uint32_t label, int depth) {
393 if (depth == kUnknownDepth)
394 return; // 不可校验的入边（死代码区段的跳转），不记录不校验
395 auto it = labelJumpDepth_.find(label);
396 if (it == labelJumpDepth_.end()) {
397 labelJumpDepth_.emplace(label, depth);
398 return;
399 }
400 if (it->second != depth) {
401 // 合并点深度冲突：报告但不拒绝 lowering（AUDIT-R6 break-discard 模式兼容，
402 // 详见 checkStackBalance LABEL 分支注释）
403 Logger::Error("IR 栈平衡校验: LABEL #" + std::to_string(label) + " 跳转入边深度冲突（已有 " +
404 std::to_string(it->second) + "，本次 " + std::to_string(depth) + "）—— 报告后继续",
405 "IR");
406 }

409bool BytecodeIRBackend::checkStackBalance(const IRInstruction& instr) {
410 // 诊断开关（D1 fix 配套）：设 MINILANG_IR_STACK_DEBUG=1 时向 stderr 打印
411 // 逐指令栈深模拟轨迹（函数名 + IROp + 处理前深度 + 操作数索引），
412 // 用于排查校验器报告的合并点冲突。未设置时零开销（static 初始化只读一次环境变量）。
413 static const bool traceDepth = [] {
414#ifdef _WIN32
415 // MSVC：getenv 触发 C4996 弃用（/WX 下即错），用 secure CRT 版本
416 char buf[8] = {};
417 size_t sz = 0;
418 getenv_s(&sz, buf, sizeof(buf), "MINILANG_IR_STACK_DEBUG");
419 return sz > 0;
420#else
421 // GCC/clang：无 getenv_s（MSVC 专属 CRT），用标准 getenv
422 return std::getenv("MINILANG_IR_STACK_DEBUG") != nullptr;
423#endif
424 }();
425 if (traceDepth) {
426 fprintf(stderr, "[stackdbg:%s] %-22s depth=%d ops=[", traceName_.c_str(), irOpName(instr.op), simStackDepth_);
427 for (size_t i = 0; i < instr.operands.size() && i < 6; ++i)
428 fprintf(stderr, "%s%u", i ? "," : "", instr.operands[i].index);
429 fprintf(stderr, "]\n");
430 }
431 // ---- LABEL：按前驱期望锚定/校验深度 ----
432 if (instr.op == IROp::LABEL) {
433 if (instr.operands.empty() || instr.operands[0].kind != IROperandKind::LABEL)
434 return true; // 无标签号（异常形态），不校验
435 const uint32_t lbl = instr.operands[0].index;
436 auto cit = labelCatchDepth_.find(lbl);
437 if (cit != labelCatchDepth_.end()) {
438 // 异常入边权威：运行时到达 catch 时高度 = TRY_BEGIN 时高度 + 1（异常值）
439 simStackDepth_ = cit->second;
440 } else {
441 auto jit = labelJumpDepth_.find(lbl);
442 if (jit != labelJumpDepth_.end()) {
443 if (simStackDepth_ != kUnknownDepth && simStackDepth_ != jit->second) {
444 // 合并点深度冲突：报告但不拒绝（AUDIT-R6 break-discard 模式兼容，
445 // 详见下方 emplace 冲突分支注释）。按跳转侧期望继续校验。
446 Logger::Error("IR 栈平衡校验: LABEL #" + std::to_string(lbl) + " 深度不一致（期望 " +
447 std::to_string(jit->second) + "，实际 " + std::to_string(simStackDepth_) +
448 "）—— 按跳转侧继续校验",
449 "IR");
450 }
451 simStackDepth_ = jit->second;
452 }
453 // 无任何期望且线性深度未知 → 保持未知（不可校验，不误报）
454 }
455 // 记录/复核本标签深度，供后续回边（循环）校验
456 if (simStackDepth_ != kUnknownDepth) {
457 auto ins = labelJumpDepth_.emplace(lbl, simStackDepth_);
458 if (!ins.second && ins.first->second != simStackDepth_) {
459 // 合并点深度冲突：报告但不拒绝 lowering。
460 // 已知合法形态：AUDIT-R6 的 "break 丢弃挂起 return 值"（try 内 return 压栈
461 // 后 finally 中 break 直接跳出）——break 边深度多 1，运行时由 OP_RETURN
462 // 的帧回收（stack_.resize(savedBp)）清理残留，静态模拟无法表达该丢弃。
463 // 若为真实 lowering 缺陷，运行时会在该合并点之后错位执行，需人工甄别。
464 Logger::Error("IR 栈平衡校验: LABEL #" + std::to_string(lbl) + " 入边深度冲突（跳转 " +
465 std::to_string(ins.first->second) + " vs 本处 " + std::to_string(simStackDepth_) +
466 "）—— 按跳转侧继续校验",
467 "IR");
468 simStackDepth_ = ins.first->second;
469 }
470 }
471 return true;
472 }
473
474 // ---- 普通指令：效应表 + 深度更新 ----
475 StackEffect fx{0, 0};
476 if (!irStackEffect(instr, fx)) {
477 // 表 + 公式均未覆盖：仅可能是 PHI（不可 lower，lowerInstruction 的
478 // default 分支应先行拒绝）或编译期完备性被绕过——D1 第二步起显式
479 // 拒绝，不再保守跳过（静默跳过会让新增 IROp 失去校验）
480 Logger::Error("IR 栈平衡校验: " + std::string(irOpName(instr.op)) + " (line=" + std::to_string(instr.line) +
481 ") 未登记栈效应表 —— 拒绝 lowering",
482 "IR");
483 return false;
484 }
485 if (simStackDepth_ != kUnknownDepth) {
486 if (fx.pops > simStackDepth_) {
487 Logger::Error("IR 栈平衡校验: " + std::string(irOpName(instr.op)) + " (line=" + std::to_string(instr.line) +
488 ") 需弹出 " + std::to_string(fx.pops) + " 个操作数，但栈深仅 " +
489 std::to_string(simStackDepth_) + " —— lowering 栈不平衡",
490 "IR");
491 return false;
492 }
493 simStackDepth_ += fx.pushes - fx.pops;
494 }
495 // 深度未知区段（上一条为 JUMP/RETURN/THROW 且未到 LABEL 的死代码）：
496 // 深度保持未知，下一 LABEL 处按前驱期望重新锚定
497
498 // ---- 控制流特殊处理 ----
499 switch (instr.op) {
500 case IROp::JUMP: {
501 // 无条件跳转：记录目标期望深度；线性流终止（至下一 LABEL 前为死代码）
502 if (!instr.operands.empty() && instr.operands[0].kind == IROperandKind::LABEL) {
503 recordJumpDepth(instr.operands[0].index, simStackDepth_);
504 }
505 simStackDepth_ = kUnknownDepth;
506 break;
507 }
508 case IROp::JUMP_IF_FALSE: {
509 // 条件为 peek：跳转侧与 fall-through 侧深度相同
510 if (instr.operands.size() >= 2 && instr.operands[1].kind == IROperandKind::LABEL) {
511 recordJumpDepth(instr.operands[1].index, simStackDepth_);
512 }
513 break;
514 }
515 case IROp::TRY_BEGIN: {
516 // catch 入口深度 = TRY_BEGIN 时深度 + 1（throwException resize 到
517 // handler.stackBase 后压入异常值，VM.cpp:187-189）
518 if (!instr.operands.empty() && instr.operands[0].kind == IROperandKind::LABEL &&
519 simStackDepth_ != kUnknownDepth) {
520 labelCatchDepth_[instr.operands[0].index] = simStackDepth_ + 1;
521 }
522 break;
523 }
524 case IROp::RETURN:
525 case IROp::RETURN_NULL:
526 case IROp::THROW:
527 // 控制流离开线性路径：至下一 LABEL 前为死代码，深度置未知
528 simStackDepth_ = kUnknownDepth;
529 break;
530 default:
531 break;
532 }
533 return true;
534}
