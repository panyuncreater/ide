162/// 格式化 DiagnosticBag 中的错误条目为 HTML（已转义）
163QString formatDiagnosticErrors(const DiagnosticBag& bag) {
164 QString result;
165 for (const auto& d : bag.all()) {
166 if (d.isError()) {
167 result += QString::fromUtf8(d.format().c_str()).toHtmlEscaped() + QStringLiteral("<br>");
168 }
169 }
170 if (result.isEmpty()) {
171 result = QStringLiteral("（未知错误）");
172 }
173 return result;
174}

176/// HTML 转义辅助
177QString esc(const std::string& s) {
178 return QString::fromUtf8(s.c_str()).toHtmlEscaped();
179}

320/// 解析源码：面板内独立完成 Lexer → Parser → AST，递归构建 QTreeWidget 树。
321/// 解析失败时在详情区显示错误。
322void AstVisualizerPanel::onParse() {
323 astTree_->clear();
324 itemNodeMap_.clear();
325 astRoot_.reset();
326 nodeDetail_->clear();
327
328 std::string source = sourceEdit_->text().toStdString();
329 if (source.empty()) {
330 nodeDetail_->setHtml(QString::fromUtf8("<div style='color:#C0392B; padding:8px;'><b>请输入源码</b></div>"));
331 return;
332 }
333
334 // Lexer → Parser → AST
335 Lexer lex;
336 std::vector<Token> tokens;
337 try {
338 tokens = lex.scan(source);
339 } catch (const std::exception& e) {
340 nodeDetail_->setHtml(QString::fromUtf8("<div style='color:#C0392B; padding:8px;'><b>词法错误:</b> %1</div>")
341 .arg(QString::fromUtf8(e.what()).toHtmlEscaped()));
342 return;
343 }
344 if (lex.getDiagnostics().hasErrors()) {
345 nodeDetail_->setHtml(QString::fromUtf8("<div style='color:#C0392B; padding:8px;'><b>词法错误:</b> %1</div>")
346 .arg(formatDiagnosticErrors(lex.getDiagnostics())));
347 return;
348 }
349
350 Parser parser;
351 std::unique_ptr<Block> ast;
352 try {
353 ast = parser.parse(tokens);
354 } catch (const std::exception& e) {
355 nodeDetail_->setHtml(QString::fromUtf8("<div style='color:#C0392B; padding:8px;'><b>语法错误:</b> %1</div>")
356 .arg(QString::fromUtf8(e.what()).toHtmlEscaped()));
357 return;
358 }
359 if (parser.hasErrors()) {
360 nodeDetail_->setHtml(QString::fromUtf8("<div style='color:#C0392B; padding:8px;'><b>语法错误:</b> %1</div>")
361 .arg(formatDiagnosticErrors(parser.getDiagnostics())));
362 return;
363 }
364
365 // 转移所有权给 astRoot_（Block* → ASTNode*）
366 astRoot_ = std::move(ast);
367 populateAstTree(astRoot_.get());
368
369 if (astTree_->topLevelItemCount() > 0) {
370 astTree_->topLevelItem(0)->setSelected(true);
371 onAstNodeSelected(astTree_->topLevelItem(0), 0);
372 } else {
373 nodeDetail_->setHtml(
374 QString::fromUtf8("<div style='color:#6E6E6E; padding:8px;'><i>（解析产物为空）</i></div>"));
375 }
376}

378/// 递归构建 AST 树：将 ASTNode 树映射为 QTreeWidgetItem 树。
379void AstVisualizerPanel::populateAstTree(ASTNode* root) {
380 if (!root) {
381 return;
382 }
383 auto* rootItem = buildTreeItem(root);
384 astTree_->addTopLevelItem(rootItem);
385 astTree_->expandAll();
386}

388/// 递归构造 QTreeWidgetItem：节点名 + 行:列摘要，子节点按 AST 结构嵌套。
389QTreeWidgetItem* AstVisualizerPanel::buildTreeItem(ASTNode* node) {
390 if (!node) {
391 return nullptr;
392 }
393 QString name = QString::fromStdString(node->nodeName());
394 QString pos = QStringLiteral("%1:%2").arg(node->line).arg(node->column);
395 auto* item = new QTreeWidgetItem({name, pos});
396 itemNodeMap_.emplace_back(item, node);
397
398 for (ASTNode* child : node->children()) {
399 if (child) {
400 item->addChild(buildTreeItem(child));
401 }
402 }
403 return item;
404}

406/// 点击 AST 树节点时，在右侧详情区显示该节点的 HTML 详情。
407void AstVisualizerPanel::onAstNodeSelected(QTreeWidgetItem* item, int /*column*/) {
408 if (!item) {
409 return;
410 }
411 // 查找对应的 ASTNode 裸指针
412 ASTNode* node = nullptr;
413 for (const auto& pair : itemNodeMap_) {
414 if (pair.first == item) {
415 node = pair.second;
416 break;
417 }
418 }
419 if (!node) {
420 nodeDetail_->clear();
421 return;
422 }
423 nodeDetail_->setHtml(buildNodeDetailHtml(node));

426/// 构造节点详情 HTML：节点类型 + 行号/列号 + 属性列表 + 子节点列表。
427QString AstVisualizerPanel::buildNodeDetailHtml(ASTNode* node) {
428 QString html;
429 html += QStringLiteral("<h3>%1</h3>").arg(QString::fromStdString(node->nodeName()).toHtmlEscaped());
430 html += QStringLiteral("<p><b>行号:</b> %1 | <b>列号:</b> %2</p>").arg(node->line).arg(node->column);
431
432 // 属性列表
433 QStringList props = extractNodeProperties(node);
434 html += QStringLiteral("<h4>属性:</h4>");
435 if (props.isEmpty()) {
436 html += QStringLiteral("<p><i>（无属性或未知节点类型）</i></p>");
437 } else {
438 html += QStringLiteral("<ul>");
439 for (const QString& p : props) {
440 html += QStringLiteral("<li>") + p + QStringLiteral("</li>");
441 }
442 html += QStringLiteral("</ul>");
443 }
444
445 // 子节点列表
446 QStringList children = extractChildDescriptions(node);
447 html += QStringLiteral("<h4>子节点:</h4>");
448 if (children.isEmpty()) {
449 html += QStringLiteral("<p><i>（无子节点）</i></p>");
450 } else {
451 html += QStringLiteral("<ul>");
452 for (const QString& c : children) {
453 html += QStringLiteral("<li>") + c + QStringLiteral("</li>");
454 }
455 html += QStringLiteral("</ul>");
456 }
457 return html;
458}

460/// 通过 dynamic_cast 提取节点属性（key: value 形式）。
461/// 仅处理常见节点类型，对不支持的类型返回空列表。
462QStringList AstVisualizerPanel::extractNodeProperties(ASTNode* node) {
463 QStringList props;
464 if (!node) {
465 return props;
466 }
467 // ---- 声明类 ----
468 if (auto* v = dynamic_cast<VarDecl*>(node)) {
469 props << QStringLiteral("name: %1").arg(esc(v->name));
470 props << QStringLiteral("typeAnnotation: %1")
471 .arg(v->typeAnnotation.empty() ? QStringLiteral("(none)") : esc(v->typeAnnotation));
472 props << QStringLiteral("hasInitializer: %1")
473 .arg(v->initializer ? QStringLiteral("true") : QStringLiteral("false"));
474 } else if (auto* f = dynamic_cast<FunDecl*>(node)) {
475 props << QStringLiteral("name: %1").arg(esc(f->name));
476 QString params;
477 for (size_t i = 0; i < f->params.size(); ++i) {
478 if (i > 0)
479 params += QStringLiteral(", ");
480 params += QString::fromStdString(f->params[i]);
481 if (i < f->paramTypes.size() && !f->paramTypes[i].empty()) {
482 params += QStringLiteral(": ") + QString::fromStdString(f->paramTypes[i]);
483 }
484 }
485 props << QStringLiteral("params: [%1]").arg(params);
486 props << QStringLiteral("returnType: %1")
487 .arg(f->returnType.empty() ? QStringLiteral("(none)") : esc(f->returnType));
488 props << QStringLiteral("requiredParamCount: %1").arg(f->requiredParamCount);
489 } else if (auto* c = dynamic_cast<ClassDecl*>(node)) {
490 props << QStringLiteral("name: %1").arg(esc(c->name));
491 props << QStringLiteral("superClassName: %1")
492 .arg(c->superClassName.empty() ? QStringLiteral("(none)") : esc(c->superClassName));
493 props << QStringLiteral("memberCount: %1").arg(c->members.size());
494 } else if (auto* e = dynamic_cast<EnumDecl*>(node)) {
495 props << QStringLiteral("name: %1").arg(esc(e->name));
496 QString tparams;
497 for (size_t i = 0; i < e->typeParams.size(); ++i) {
498 if (i > 0)
499 tparams += QStringLiteral(", ");
500 tparams += QString::fromStdString(e->typeParams[i]);
501 }
502 props << QStringLiteral("typeParams: [%1]").arg(tparams);
503 props << QStringLiteral("variantCount: %1").arg(e->variants.size());
504 // ---- 语句类 ----
505 } else if (dynamic_cast<IfStmt*>(node)) {
506 props << QStringLiteral("hasElseBranch: %1")
507 .arg(static_cast<IfStmt*>(node)->elseBranch ? QStringLiteral("true") : QStringLiteral("false"));
508 } else if (dynamic_cast<WhileStmt*>(node)) {
509 // WhileStmt 无额外属性（condition/body 已在子节点列表）
510 } else if (dynamic_cast<ForStmt*>(node)) {
511 auto* f3 = static_cast<ForStmt*>(node);
512 props << QStringLiteral("hasInitializer: %1")
513 .arg(f3->initializer ? QStringLiteral("true") : QStringLiteral("false"));
514 props
515 << QStringLiteral("hasCondition: %1").arg(f3->condition ? QStringLiteral("true") : QStringLiteral("false"));
516 props << QStringLiteral("hasUpdate: %1").arg(f3->update ? QStringLiteral("true") : QStringLiteral("false"));
517 } else if (auto* r = dynamic_cast<ReturnStmt*>(node)) {
518 props << QStringLiteral("hasValue: %1").arg(r->value ? QStringLiteral("true") : QStringLiteral("false"));
519 } else if (dynamic_cast<BreakStmt*>(node)) {
520 // 无属性
521 } else if (dynamic_cast<ContinueStmt*>(node)) {
522 // 无属性
523 } else if (auto* b = dynamic_cast<Block*>(node)) {
524 props << QStringLiteral("statementCount: %1").arg(b->statements.size());
525 } else if (auto* p = dynamic_cast<PrintStmt*>(node)) {
526 props << QStringLiteral("valueCount: %1").arg(p->values.size());
527 } else if (auto* t = dynamic_cast<TryStmt*>(node)) {
528 props << QStringLiteral("catchVarName: %1").arg(esc(t->catchVarName));
529 props << QStringLiteral("hasFinallyBlock: %1")
530 .arg(t->finallyBlock ? QStringLiteral("true") : QStringLiteral("false"));
531 } else if (auto* t2 = dynamic_cast<ThrowStmt*>(node)) {
532 props << QStringLiteral("hasExpression: %1")
533 .arg(t2->expression ? QStringLiteral("true") : QStringLiteral("false"));
534 } else if (auto* i = dynamic_cast<ImportStmt*>(node)) {
535 props << QStringLiteral("modulePath: %1").arg(esc(i->modulePath));
536 props << QStringLiteral("importAll: %1").arg(i->importAll ? QStringLiteral("true") : QStringLiteral("false"));
537 props << QStringLiteral("nameCount: %1").arg(i->names.size());
538 } else if (auto* e2 = dynamic_cast<ExportStmt*>(node)) {
539 props << QStringLiteral("hasDeclaration: %1")
540 .arg(e2->declaration ? QStringLiteral("true") : QStringLiteral("false"));
541 // ---- 表达式类 ----
542 } else if (auto* n = dynamic_cast<NumberLiteral*>(node)) {
543 if (n->isFloat_) {
544 props << QStringLiteral("value: %1 (float)").arg(n->floatValue_);
545 } else {
546 props << QStringLiteral("value: %1 (int)").arg(n->intValue_);
547 }
548 } else if (auto* s = dynamic_cast<StringLiteral*>(node)) {
549 props << QStringLiteral("value: \"%1\"").arg(esc(s->value));
550 } else if (auto* b2 = dynamic_cast<BoolLiteral*>(node)) {
551 props << QStringLiteral("value: %1").arg(b2->value ? QStringLiteral("true") : QStringLiteral("false"));
552 } else if (dynamic_cast<NullLiteral*>(node)) {
553 props << QStringLiteral("value: null");
554 } else if (auto* b22 = dynamic_cast<BinaryOp*>(node)) {
555 props << QStringLiteral("opType: %1").arg(QString::fromUtf8(BinaryOp::opTypeStr(b22->opType)));
556 props << QStringLiteral("precedence: %1").arg(BinaryOp::precedence(b22->opType));
557 } else if (auto* u = dynamic_cast<UnaryOp*>(node)) {
558 props << QStringLiteral("opType: %1").arg(QString::fromUtf8(UnaryOp::opTypeStr(u->opType)));
559 } else if (auto* f2 = dynamic_cast<FunCall*>(node)) {
560 if (f2->callee) {
561 props << QStringLiteral("callMode: 链式调用（callee 非空）");
562 } else {
563 props << QStringLiteral("name: %1").arg(esc(f2->name));
564 }
565 props << QStringLiteral("argumentCount: %1").arg(f2->arguments.size());
566 } else if (auto* m = dynamic_cast<MemberAccess*>(node)) {
567 props << QStringLiteral("fieldName: %1").arg(esc(m->fieldName));
568 } else if (auto* m2 = dynamic_cast<MethodCall*>(node)) {
569 props << QStringLiteral("methodName: %1").arg(esc(m2->methodName));
570 props << QStringLiteral("argumentCount: %1").arg(m2->arguments.size());
571 } else if (dynamic_cast<IndexAccess*>(node)) {
572 // 无额外属性（object/index 在子节点列表）
573 } else if (dynamic_cast<IndexAssign*>(node)) {
574 // 无额外属性
575 } else if (auto* a = dynamic_cast<Assignment*>(node)) {
576 props << QStringLiteral("name: %1").arg(esc(a->name));
577 } else if (auto* v2 = dynamic_cast<VarRef*>(node)) {
578 props << QStringLiteral("name: %1").arg(esc(v2->name));
579 } else if (auto* a2 = dynamic_cast<ArrayLiteral*>(node)) {
580 props << QStringLiteral("elementCount: %1").arg(a2->elements.size());
581 } else if (auto* d = dynamic_cast<DictLiteral*>(node)) {
582 props << QStringLiteral("pairCount: %1").arg(d->pairs.size());
583 } else if (auto* i2 = dynamic_cast<InterpolatedString*>(node)) {
584 props << QStringLiteral("literalCount: %1").arg(i2->literals.size());
585 props << QStringLiteral("expressionCount: %1").arg(i2->expressions.size());
586 } else if (auto* d2 = dynamic_cast<DestructureBinding*>(node)) {
587 QString names;
588 for (size_t i13 = 0; i13 < d2->names.size(); ++i13) {
589 if (i13 > 0)
590 names += QStringLiteral(", ");
591 names += QString::fromStdString(d2->names[i13]);
592 }
593 props << QStringLiteral("names: [%1]").arg(names);
594 } else if (auto* e22 = dynamic_cast<EnumVariantExpr*>(node)) {
595 props << QStringLiteral("enumName: %1").arg(esc(e22->enumName));
596 props << QStringLiteral("variantName: %1").arg(esc(e22->variantName));
597 props << QStringLiteral("argumentCount: %1").arg(e22->arguments.size());
598 } else if (auto* m24 = dynamic_cast<MatchExpr*>(node)) {
599 props << QStringLiteral("caseCount: %1").arg(m24->cases.size());
600 } else if (auto* t23 = dynamic_cast<TupleLiteral*>(node)) {
601 props << QStringLiteral("elementCount: %1").arg(t23->elements.size());
602 } else if (dynamic_cast<SuperExpr*>(node)) {
603 props << QStringLiteral("note: 引用父类方法/字段");
604 } else {
605 // 未知节点类型：不显示属性（详情区已显示「无属性或未知节点类型」）
606 }
607 return props;
608}

610/// 提取子节点描述列表（使用 ASTNode::children() 接口）。
611QStringList AstVisualizerPanel::extractChildDescriptions(ASTNode* node) {
612 QStringList childrenDesc;
613 if (!node) {
614 return childrenDesc;
615 }
616 // 通过 dynamic_cast 给子节点附上角色标签（如 VarDecl 的 initializer、IfStmt 的
617 // condition/thenBranch/elseBranch），便于学习者理解每个子节点的语义角色。
618 if (auto* v = dynamic_cast<VarDecl*>(node)) {
619 if (v->initializer)
620 childrenDesc << QStringLiteral("initializer: %1")
621 .arg(QString::fromStdString(v->initializer->nodeName()).toHtmlEscaped());
622 } else if (auto* f = dynamic_cast<FunDecl*>(node)) {
623 if (f->body)
624 childrenDesc << QStringLiteral("body: %1").arg(QString::fromStdString(f->body->nodeName()).toHtmlEscaped());
625 for (size_t i = 0; i < f->defaultValues.size(); ++i) {
626 if (f->defaultValues[i]) {
627 childrenDesc << QStringLiteral("defaultValue[%1]: %2")
628 .arg(i)
629 .arg(QString::fromStdString(f->defaultValues[i]->nodeName()).toHtmlEscaped());
630 }
631 }
632 } else if (auto* c = dynamic_cast<ClassDecl*>(node)) {
633 for (size_t i = 0; i < c->members.size(); ++i) {
634 if (c->members[i]) {
635 childrenDesc << QStringLiteral("member[%1]: %2")
636 .arg(i)
637 .arg(QString::fromStdString(c->members[i]->nodeName()).toHtmlEscaped());
638 }
639 }
640 } else if (auto* i = dynamic_cast<IfStmt*>(node)) {
641 if (i->condition)
642 childrenDesc << QStringLiteral("condition: %1")
643 .arg(QString::fromStdString(i->condition->nodeName()).toHtmlEscaped());

645 childrenDesc << QStringLiteral("thenBranch: %1")
646 .arg(QString::fromStdString(i->thenBranch->nodeName()).toHtmlEscaped());

652 childrenDesc << QStringLiteral("condition: %1")
653 .arg(QString::fromStdString(w->condition->nodeName()).toHtmlEscaped());
654 if (w->body)
655 childrenDesc << QStringLiteral("body: %1").arg(QString::fromStdString(w->body->nodeName()).toHtmlEscaped());
656 } else if (auto* f2 = dynamic_cast<ForStmt*>(node)) {
657 if (f2->initializer)
658 childrenDesc << QStringLiteral("initializer: %1")
659 .arg(QString::fromStdString(f2->initializer->nodeName()).toHtmlEscaped());
660 if (f2->condition)
661 childrenDesc << QStringLiteral("condition: %1")
662 .arg(QString::fromStdString(f2->condition->nodeName()).toHtmlEscaped());

664 childrenDesc
665 << QStringLiteral("update: %1").arg(QString::fromStdString(f2->update->nodeName()).toHtmlEscaped());
666 if (f2->body)
667 childrenDesc
668 << QStringLiteral("body: %1").arg(QString::fromStdString(f2->body->nodeName()).toHtmlEscaped());
669 } else if (auto* r = dynamic_cast<ReturnStmt*>(node)) {
670 if (r->value)
671 childrenDesc
672 << QStringLiteral("value: %1").arg(QString::fromStdString(r->value->nodeName()).toHtmlEscaped());
673 } else if (auto* b = dynamic_cast<BinaryOp*>(node)) {
674 if (b->left)
675 childrenDesc << QStringLiteral("left: %1").arg(QString::fromStdString(b->left->nodeName()).toHtmlEscaped());
676 if (b->right)
677 childrenDesc
678 << QStringLiteral("right: %1").arg(QString::fromStdString(b->right->nodeName()).toHtmlEscaped());
679 } else if (auto* u = dynamic_cast<UnaryOp*>(node)) {
680 if (u->operand)
681 childrenDesc

683 } else if (auto* f22 = dynamic_cast<FunCall*>(node)) {

694 } else if (auto* m = dynamic_cast<MemberAccess*>(node)) {

696 childrenDesc
697 << QStringLiteral("object: %1").arg(QString::fromStdString(m->object->nodeName()).toHtmlEscaped());
698 } else if (auto* m2 = dynamic_cast<MethodCall*>(node)) {
699 if (m2->object)
700 childrenDesc
701 << QStringLiteral("object: %1").arg(QString::fromStdString(m2->object->nodeName()).toHtmlEscaped());
702 for (size_t i11 = 0; i11 < m2->arguments.size(); ++i11) {
703 if (m2->arguments[i11]) {
704 childrenDesc << QStringLiteral("arg[%1]: %2")

708 }
709 } else if (auto* i2 = dynamic_cast<IndexAccess*>(node)) {

711 childrenDesc

724 childrenDesc
725 << QStringLiteral("value: %1").arg(QString::fromStdString(i22->value->nodeName()).toHtmlEscaped());

730 } else if (auto* m23 = dynamic_cast<MemberAssign*>(node)) {
