95/// 命中条件判定。支持 "N"/"== N"/">= N"/"> N"/"% N"；
96/// 空表达式或解析失败返回 true（视为无命中条件，不拦截，宽容优于静默失效）。
97inline bool evalHitCondition(const std::string& expr, int count) {
98 size_t b = expr.find_first_not_of(" \t");
99 size_t e = expr.find_last_not_of(" \t");
100 if (b == std::string::npos)
101 return true;
102 std::string s = expr.substr(b, e - b + 1);
103 std::string op = "==";
104 size_t numStart = 0;
105 if (s.rfind(">=", 0) == 0 || s.rfind("==", 0) == 0) {
106 op = s.substr(0, 2);
107 numStart = 2;
108 } else if (s[0] == '>' || s[0] == '%') {
109 op = s.substr(0, 1);
110 numStart = 1;
111 }
112 try {
113 long n = std::stol(s.substr(numStart));
114 if (op == "==")
115 return count == n;
116 if (op == ">=")
117 return count >= n;
118 if (op == ">")
119 return count > n;
120 if (op == "%")
121 return n > 0 && count % n == 0;
122 } catch (...) {
123 // 解析失败：视为无命中条件
124 }
125 return true;
126}

128/// 调试改值（setVariable）新值文本解析：int / float / true / false /
129/// null / "带引号字符串"。不支持任意表达式（副作用隔离）。
130inline bool parseDebugValueText(const std::string& text, Value& out) {
131 size_t b = text.find_first_not_of(" \t");
132 size_t e = text.find_last_not_of(" \t");
133 if (b == std::string::npos)
134 return false;
135 std::string s = text.substr(b, e - b + 1);
136 if (s == "true") {
137 out = Value(true);
138 return true;
139 }
140 if (s == "false") {
141 out = Value(false);
142 return true;
143 }
144 if (s == "null") {
145 out = Value::nullValue();
146 return true;
147 }
148 if (s.size() >= 2 && s.front() == '"' && s.back() == '"') {
149 out = Value(s.substr(1, s.size() - 2));
150 return true;
151 }
152 try {
153 size_t pos = 0;
154 long long iv = std::stoll(s, &pos);
155 if (pos == s.size()) {
156 out = Value(static_cast<int64_t>(iv));
157 return true;
158 }
159 } catch (...) {
160 }
161 try {
162 size_t pos = 0;
163 double dv = std::stod(s, &pos);
164 if (pos == s.size()) {
165 out = Value(dv);
166 return true;
167 }
168 } catch (...) {
169 }
170 return false;
171}
