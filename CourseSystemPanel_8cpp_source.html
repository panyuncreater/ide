195QString CourseSystemLibrary::courseToJson(const Course& course) {
196 QJsonObject root;
197 root["title"] = QString::fromUtf8(course.title.c_str());
198 root["description"] = QString::fromUtf8(course.description.c_str());
199
200 QJsonArray chaptersArr;
201 for (const auto& ch : course.chapters) {
202 QJsonObject chObj;
203 chObj["title"] = QString::fromUtf8(ch.title.c_str());
204 QJsonArray lessonsArr;
205 for (const auto& ls : ch.lessons) {
206 QJsonObject lsObj;
207 lsObj["title"] = QString::fromUtf8(ls.title.c_str());
208 lsObj["code"] = QString::fromUtf8(ls.code.c_str());
209 lsObj["explanation"] = QString::fromUtf8(ls.explanation.c_str());
210 lsObj["expectedOutput"] = QString::fromUtf8(ls.expectedOutput.c_str());
211 lessonsArr.append(lsObj);
212 }
213 chObj["lessons"] = lessonsArr;
214 chaptersArr.append(chObj);
215 }
216 root["chapters"] = chaptersArr;
217
218 QJsonDocument doc(root);
219 return QString::fromUtf8(doc.toJson(QJsonDocument::Indented));
220}

222bool CourseSystemLibrary::parseCourse(const QString& json, Course& out, QString& errMsg) {
223 out = Course{};
224 QJsonParseError parseErr;
225 QJsonDocument doc = QJsonDocument::fromJson(json.toUtf8(), &parseErr);
226 if (parseErr.error != QJsonParseError::NoError) {
227 errMsg = QStringLiteral("JSON 解析失败：%1（偏移 %2）").arg(parseErr.errorString()).arg(parseErr.offset);
228 return false;
229 }
230 if (!doc.isObject()) {
231 errMsg = QStringLiteral("JSON 顶层必须是对象");
232 return false;
233 }
234 QJsonObject root = doc.object();
235
236 // title（必需，字符串）
237 if (!root.contains("title")) {
238 errMsg = QStringLiteral("缺少必需字段：title");
239 return false;
240 }
241 QJsonValue titleVal = root.value("title");
242 if (!titleVal.isString()) {
243 errMsg = QStringLiteral("字段 title 必须是字符串");
244 return false;
245 }
246 out.title = titleVal.toString().toUtf8().constData();
247
248 // description（可选，字符串）
249 if (root.contains("description")) {
250 QJsonValue descVal = root.value("description");
251 if (!descVal.isString()) {
252 errMsg = QStringLiteral("字段 description 必须是字符串");
253 return false;
254 }
255 out.description = descVal.toString().toUtf8().constData();
256 }
257
258 // chapters（必需，数组）
259 if (!root.contains("chapters")) {
260 errMsg = QStringLiteral("缺少必需字段：chapters");
261 return false;
262 }
263 QJsonValue chaptersVal = root.value("chapters");
264 if (!chaptersVal.isArray()) {
265 errMsg = QStringLiteral("字段 chapters 必须是数组");
266 return false;
267 }
268 QJsonArray chaptersArr = chaptersVal.toArray();
269 if (chaptersArr.isEmpty()) {
270 errMsg = QStringLiteral("chapters 不能为空");
271 return false;
272 }
273
274 for (int i = 0; i < chaptersArr.size(); ++i) {
275 QJsonValue chVal = chaptersArr.at(i);
276 if (!chVal.isObject()) {
277 errMsg = QStringLiteral("第 %1 个 chapter 必须是对象").arg(i + 1);
278 return false;
279 }
280 QJsonObject chObj = chVal.toObject();
281 CourseChapter chapter;
282 if (!chObj.contains("title") || !chObj.value("title").isString()) {
283 errMsg = QStringLiteral("第 %1 个 chapter 缺少字符串字段 title").arg(i + 1);
284 return false;
285 }
286 chapter.title = chObj.value("title").toString().toUtf8().constData();
287
288 if (!chObj.contains("lessons") || !chObj.value("lessons").isArray()) {
289 errMsg = QStringLiteral("第 %1 个 chapter 缺少数组字段 lessons").arg(i + 1);
290 return false;
291 }
292 QJsonArray lessonsArr = chObj.value("lessons").toArray();
293 for (int j = 0; j < lessonsArr.size(); ++j) {
294 QJsonValue lsVal = lessonsArr.at(j);
295 if (!lsVal.isObject()) {
296 errMsg = QStringLiteral("第 %1 章 第 %2 课必须是对象").arg(i + 1).arg(j + 1);
297 return false;
298 }
299 QJsonObject lsObj = lsVal.toObject();
300 CourseLesson lesson;
301 // 四字段校验：title / code / explanation / expectedOutput
302 static const char* kRequiredFields[] = {"title", "code", "explanation", "expectedOutput"};
303 for (const char* f : kRequiredFields) {
304 if (!lsObj.contains(f) || !lsObj.value(f).isString()) {
305 errMsg = QStringLiteral("第 %1 章 第 %2 课缺少字符串字段 %3")
306 .arg(i + 1)
307 .arg(j + 1)
308 .arg(QString::fromUtf8(f));
309 return false;
310 }
311 }
312 lesson.title = lsObj.value("title").toString().toUtf8().constData();
313 lesson.code = lsObj.value("code").toString().toUtf8().constData();
314 lesson.explanation = lsObj.value("explanation").toString().toUtf8().constData();
315 lesson.expectedOutput = lsObj.value("expectedOutput").toString().toUtf8().constData();
316 chapter.lessons.push_back(std::move(lesson));
317 }
318 if (chapter.lessons.empty()) {
319 errMsg = QStringLiteral("第 %1 个 chapter 的 lessons 不能为空").arg(i + 1);
320 return false;
321 }
322 out.chapters.push_back(std::move(chapter));
323 }
324 return true;
325}

353QString CourseSystemLibrary::coursesToJsonArray(const std::vector<Course>& courses) {

364bool CourseSystemLibrary::parseCoursesArray(const QString& json, std::vector<Course>& out, QString& errMsg) {
365 out.clear();
366 QJsonParseError parseErr;
367 QJsonDocument doc = QJsonDocument::fromJson(json.toUtf8(), &parseErr);
368 if (parseErr.error != QJsonParseError::NoError) {
369 errMsg = QStringLiteral("JSON 解析失败：%1").arg(parseErr.errorString());
370 return false;
371 }
372 if (!doc.isArray()) {
373 errMsg = QStringLiteral("JSON 顶层必须是数组");
374 return false;
375 }
376 const QJsonArray arr = doc.array();
377 for (int i = 0; i < arr.size(); ++i) {
378 if (!arr[i].isObject()) {
379 errMsg = QStringLiteral("第 %1 项不是对象").arg(i + 1);
380 out.clear();
381 return false;
382 }
383 // 逐项复用 parseCourse 完整校验（字段类型/必需字段）
384 Course c;
385 QString itemErr;
386 QString itemJson = QString::fromUtf8(QJsonDocument(arr[i].toObject()).toJson(QJsonDocument::Compact));
387 if (!parseCourse(itemJson, c, itemErr)) {
388 errMsg = QStringLiteral("第 %1 项非法：%2").arg(i + 1).arg(itemErr);
389 out.clear();
390 return false;
391 }
392 out.push_back(std::move(c));
393 }
394 return true;
395}

722void CourseSystemPanel::onLoadToLibrary() {
723 if (!hasLastParsed_) {
724 previewBrowser_->append(QStringLiteral("<p style='color:red;'>请先点击「解析预览」并成功解析后再加载。</p>"));
725 return;
726 }
727 // 避免重复加载同标题课程
728 std::string newTitle = lastParsedCourse_.title;
729 for (const auto& c : customCourses_) {
730 if (c.title == newTitle) {
731 previewBrowser_->append(QStringLiteral("<p style='color:#F57F17;'>课程库中已存在同名自定义课程「%1」，"
732 "未重复添加。请修改 title 后重试。</p>")
733 .arg(esc(newTitle)));
734 return;
735 }
736 }
737 customCourses_.push_back(lastParsedCourse_);
738 refreshMergedCourses();
739 // 拓展二期：自定义课程入库后立即持久化（重启 IDE 不丢失）
740 saveCustomCoursesToSettings();
741
742 // 选中新加入的课程（末尾）
743 int newIdx = static_cast<int>(mergedCourses_.size()) - 1;
744 if (newIdx >= 0) {
745 courseCombo_->setCurrentIndex(newIdx);
746 }
747 // 切换到课程库子页，让用户看到结果
748 pageLibraryBtn_->setChecked(true);
749
750 previewBrowser_->append(QStringLiteral("<p style='color:#2E7D32;'><b>已加载到课程库：</b>%1（共 %2 章）</p>")
751 .arg(esc(newTitle))
752 .arg(static_cast<int>(lastParsedCourse_.chapters.size())));
753}
