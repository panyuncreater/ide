35// --- Clickable IR/bytecode HTML helper ---
36// Converts plain text where lines may end with "; line N" into HTML
37// with those lines wrapped in <a href="#LINE_N"> anchors.
38namespace {
39/// 将纯文本 IR 转换为可点击 HTML：识别行尾 "; line N" 注释，把该行
40/// 包裹为 <a href="#LINE_N"> 锚点，便于在浏览器中点击跳转到源码对应行。
41static QString irTextToClickableHtml(const QString& plainText) {
42 QStringList lines = plainText.split("\n");
43 QString html;
44 html += "<pre style=\"font-family:'Cascadia Code','Cascadia Mono','Consolas','JetBrains Mono','Source Code "
45 "Pro','Menlo','DejaVu Sans Mono','Courier New',monospace;\">";
46 static const QRegularExpression lineCommentRe(";\\s*line\\s+(\\d+)\\s*$");
47 for (int i = 0; i < lines.size(); ++i) {
48 QString line = lines[i];
49 auto m = lineCommentRe.match(line);
50 if (m.hasMatch()) {
51 QString lineNum = m.captured(1);
52 QString escaped = line.toHtmlEscaped();
53 html +=
54 "<a href=\"#LINE_" + lineNum + "\" style=\"color:inherit;text-decoration:none;\">" + escaped + "</a>";
55 } else {
56 html += line.toHtmlEscaped();
57 }
58 if (i < lines.size() - 1)
59 html += "\n";
60 }
61 html += "</pre>";
62 return html;
63}
64} // anonymous namespace

71/// 构造面板：组装顶部 4 个子页切换按钮（lowering / 优化对比 / 当前源码 IR /
72/// 逐步回放）与 QStackedWidget，构建各子页并连接切换信号、主题刷新与列表填充。
73void IRTransformPanel::applyTheme() {
74 // 主题切换时刷新当前页 HTML（populateLoweringDetail 中 <pre> 背景使用
75 // TeachingTheme::surface()，需重新渲染以跟随新主题）。
76 if (!stack_)
77 return;
78 switch (stack_->currentIndex()) {
79 case 0: // AST → IR lowering 页（HTML 含主题色 <pre> 背景）
80 if (loweringList_)
81 populateLoweringDetail(loweringList_->currentRow());
82 break;
83 case 1: // 优化 pass 对比页（setPlainText，无主题依赖，仍刷新以防未来扩展）

85 populateOptDetail(optList_->currentRow());

87 case 2: // 当前源码 IR 页（setPlainText，无主题依赖）
88 populateCurrentIR();
89 break;
90 case 3: // 逐步优化回放页（setPlainText，无主题依赖）
91 if (replayList_ && replayStepsList_)
92 populateReplayStep(replayList_->currentRow(), replayStepsList_->currentRow());
93 break;
94 default:
95 break;
96 }

264/// 拓展二期：提交预测——揭示优化后 IR，对照 instrAfter 判定预测。
265void IRTransformPanel::onPredictSubmit() {
266 int index = optList_->currentRow();
267 const auto& items = IRTransformLibrary::optimizationExamples();
268 if (index < 0 || index >= (int)items.size())
269 return;
270 const auto& e = items[index];
271
272 // 揭示答案
273 optAfter_->setPlainText(QString::fromUtf8(e.irAfter.c_str()));
274 optSummary_->setText(tr("%1 — %2 → %3 条指令（减少 %4 条）")
275 .arg(QString::fromUtf8(e.title.c_str()))
276 .arg(e.instrBefore)
277 .arg(e.instrAfter)
278 .arg(e.instrBefore - e.instrAfter));
279
280 int guess = predictSpin_->value();
281 if (guess == e.instrAfter) {
282 predictResultLabel_->setText(tr("✅ 预测正确！优化后确为 %1 条指令").arg(e.instrAfter));

284 predictResultLabel_->setText(
285 tr("❌ 预测 %1 条，实际 %2 条——对照右侧揭示的优化后 IR 看看哪些指令被消除了").arg(guess).arg(e.instrAfter));

414/// 从 controller 取 AST，用 AstIRBuilder 生成 IRModule，经 IRToString 转文本后
415/// 渲染为可点击 HTML，并显示基本块/指令数。异常时在状态栏报告。
416void IRTransformPanel::populateCurrentIR() {
417 if (!controller_) {
418 currentStatusLabel_->setText(tr("未绑定控制器"));
419 currentIrBrowser_->setPlainText(tr("（未绑定控制器）"));
420 return;
421 }
422 Block* ast = controller_->astRoot();
423 if (!ast) {
424 currentStatusLabel_->setText(tr("请先在主编辑器中输入并编译代码"));
425 currentIrBrowser_->setPlainText(tr("（无 AST）"));
426 return;
427 }
428
429 try {
430 AstIRBuilder builder;
431 builder.build(*ast);
432 IRModule* mod = builder.getModule();
433 if (!mod || !mod->mainFunction) {
434 currentStatusLabel_->setText(tr("IR 生成失败"));
435 return;
436 }
437 std::string irText = IRToString(*mod->mainFunction);
438 currentIrBrowser_->setHtml(irTextToClickableHtml(QString::fromUtf8(irText.c_str())));
439 int instrCount = 0;
440 for (const auto& blk : mod->mainFunction->blocks) {
441 instrCount += (int)blk.instructions.size();
442 }
443 currentStatusLabel_->setText(
444 tr("已生成 IR：%1 基本块 / %2 条指令").arg(mod->mainFunction->blocks.size()).arg(instrCount));
445 } catch (const std::exception& e) {
446 currentStatusLabel_->setText(tr("IR 生成异常"));
447 currentIrBrowser_->setPlainText(QString::fromUtf8(e.what()));
448 }
449}
