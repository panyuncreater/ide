125/// 获取 opcode 元数据（name + baseSize + isVariableLength）
126/// A3 fix: 单一数据源，替代原 opCodeName switch + instructionSize 数组两处维护
127const OpCodeInfo& getOpCodeInfo(OpCode op) {
128 auto idx = static_cast<uint8_t>(op);
129 static const OpCodeInfo kUnknown{"OP_UNKNOWN", 1, false};
130 if (idx < sizeof(kOpCodeInfo) / sizeof(kOpCodeInfo[0])) {
131 return kOpCodeInfo[idx];
132 }
133 return kUnknown;
134}

137std::string BytecodeChunk::disassemble() const {
138 std::string result;
139 if (!name.empty()) {
140 result += "== " + name + " (arity=" + std::to_string(arity) + ") ==\n";
141 }
142 size_t offset = 0;
143 while (offset < code.size()) {
144 result += disassembleInstruction(offset);
145 result += "\n";
146 }
147 return result;
148}

698void BytecodeChunk::relocateGlobalSlots(const std::vector<int>& relocationMap) {
699 size_t offset = 0;
700 while (offset < code.size()) {
701 if (offset >= code.size())
702 break;
703 OpCode op = static_cast<OpCode>(code[offset]);
704 // 仅全局槽位指令需要重定位（OP_GET_GLOBAL/OP_SET_GLOBAL/OP_DEFINE_GLOBAL/OP_DELETE_GLOBAL）
705 if (op == OpCode::OP_GET_GLOBAL || op == OpCode::OP_SET_GLOBAL || op == OpCode::OP_DEFINE_GLOBAL ||
706 op == OpCode::OP_DELETE_GLOBAL) {
707 if (offset + 2 < code.size()) {
708 uint16_t slot =
709 static_cast<uint16_t>(code[offset + 1]) | (static_cast<uint16_t>(code[offset + 2]) << 8);
710 if (slot < relocationMap.size() && relocationMap[slot] >= 0) {
711 int newSlot = relocationMap[slot];
712 code[offset + 1] = static_cast<uint8_t>(newSlot & 0xFF);
713 code[offset + 2] = static_cast<uint8_t>((newSlot >> 8) & 0xFF);
714 }
715 }
716 }
717 offset += instructionSizeAt(offset);
718 }
