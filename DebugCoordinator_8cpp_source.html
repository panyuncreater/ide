22DebugCoordinator::~DebugCoordinator() {
23 // #10 fix: 反注册 setupDebug 注册的回调，避免 worker 线程调用悬垂 this。
24 // 传空 std::function 使后续调用变为 no-op（DebugController 在锁内拷贝再调用）。

26 // 否则 worker 线程可能在锁外 cb() 调用中持已失效的 interpreter_ shared_ptr → UAF。
27 // 采用 RCU 优雅期模式：activeCallbackCount_ 原子计数，cb() 期间 >0，归零后安全析构。
28 // R54-9 fix: 先调用 stop() 唤醒可能阻塞在 pauseExecution 的 worker 线程。
29 // 原实现仅清空回调 + waitCallbacksIdle，但 worker 若阻塞在 pauseCV_.wait()，
30 // 不在 callback 中（activeCallbackCount_==0），waitCallbacksIdle 立即返回。
31 // 随后 interpreter_ 被释放，worker 醒来后访问已释放 interpreter → UAF。
32 // stop() 设 stopped_=true 并 notify_one，worker 醒来后抛 DebugStopException 终止。
33 if (debugger_) {
34 debugger_->stop();
35 debugger_->setConditionEvaluator({});
36 debugger_->setVariableCallback({});
37 debugger_->setVariableWriteCallback({}); // 拓展二期：写回调同样反注册

62 debugger_->setConditionEvaluator([interpreter = interpreter_](const std::string& condition) -> bool {
63 try {
64 // 条件断点修复(R86): 自动补充分号——用户输入的条件表达式（如 i%2==1）
65 // 通常不带分号，但 Parser::parse() 的 expressionStatement() 要求分号，
66 // 缺失分号导致解析失败 → 条件求值返回 false → 断点永不触发。
67 std::string condExpr = condition;
68 if (!condExpr.empty() && condExpr.back() != ';') {
69 condExpr += ';';
70 }
71 Lexer condLexer;
72 auto tokens = condLexer.scan(condExpr);
73 Parser condParser;
74 auto block = condParser.parse(tokens);
75 if (!condParser.getDiagnostics().hasErrors() && block && !block->statements.empty()) {
76 Value result = interpreter->evaluateCondition(block->statements[0].get());
77 return result.isTruthy();
78 }
79 } catch (const std::exception& e) {
80 // E1 fix: 条件断点求值异常记录告警，便于用户排查（条件 + 错误信息）。
81 // 上层 DebugController::shouldPauseAtBreakpoint 也会再次记录行号。
82 // AUDIT-BUG-C8 fix: 改用 LOG_* 宏，先检查级别再构造消息（懒求值）。
83 LOG_WARNING("条件断点求值异常: " + std::string(e.what()) + "（条件: " + condition + "），视为条件不满足",
84 "Debugger");
85 } catch (...) {
86 LOG_WARNING("条件断点求值发生未知异常（条件: " + condition + "），视为条件不满足", "Debugger");
87 }
88 return false;
89 });

92 debugger_->setVariableCallback([interpreter = interpreter_]() -> std::vector<VariableSnapshot> {
93 std::vector<VariableSnapshot> result;
94 // AUDIT-BUG-C4 fix: 调试回调必须用 try/catch 包裹防止崩溃（硬约束 #16）。
95 // 同文件 conditionEvaluator 已有 try/catch，原 variableCallback 遗漏。
96 // Interpreter 异常态下遍历 Environment 链/拷贝容器可能抛 bad_alloc，
97 // 未捕获会传播到 UI 线程导致 IDE 崩溃。
98 // AUDIT-P1 fix: 用 currentEnvironmentShared() 获取 shared_ptr 副本，
99 // 遍历 parent 链时通过 shared_ptr 赋值延长每个节点的生命周期，
100 // 防止 GUI 线程遍历期间 worker 线程修改 currentEnv_ 或析构 Environment → UAF。
101 try {
102 auto currentShared = interpreter->currentEnvironmentShared();
103 if (currentShared) {
104 int depth = 0;
105 while (currentShared) {
106 // R54-11 fix: 用 snapshotLocalVariables() 返回拷贝，避免遍历
107 // 期间 worker 线程修改 variables map 导致迭代器失效。
108 // shared_ptr 防止 Environment 析构，但不防止 map 内容被修改。
109 auto locals = currentShared->snapshotLocalVariables();
110 for (const auto& kv : locals) {
111 VariableSnapshot snap;
112 snap.name = kv.first;
113 snap.value = kv.second;
114 snap.scope = (currentShared->parent == nullptr) ? "全局" : (depth == 0) ? "局部" : "外层";
115 result.push_back(snap);
116 }
117 currentShared = currentShared->parent; // shared_ptr 赋值，延长 parent 生命周期
118 depth++;
119 }
120 }
121 } catch (const std::exception& e) {
122 Logger::Warning(std::string("变量快照回调异常: ") + e.what(), "Debugger");
123 } catch (...) {
124 Logger::Warning("变量快照回调发生未知异常", "Debugger");
125 }
126 return result;
127 });

129 // 拓展二期：变量写回调（setVariable）——调试暂停时 GUI 线程沿当前
130 // Environment 作用域链写入。仅在 DebugController::setVariableValue 确认

150 debugger_->setCallStackCallback([interpreter = interpreter_]() -> std::vector<CallStackEntry> {
151 std::vector<CallStackEntry> result;
152 // AUDIT-BUG-C5 fix: 调试回调必须用 try/catch 包裹防止崩溃（硬约束 #16）。
153 // AUDIT-P1 fix: 用 getCallStackSnapshot() 返回值拷贝，避免 const 引用跨线程
154 // 遍历期间 worker 线程 push_back/pop_back 导致迭代器失效。
155 // 对齐 variableCallback 的 currentEnvironmentShared() 修复模式。
156 try {
157 auto stack = interpreter->getCallStackSnapshot();
158 for (const auto& frame : stack) {
159 CallStackEntry entry;
160 entry.functionName = frame.functionName;
161 entry.line = frame.line;
162 entry.depth = frame.depth;
163 if (frame.env) {
164 // P0-2 fix: 用 snapshotLocalVariables() 获取值拷贝，避免直接遍历
165 // Environment::variables 的 const 引用。UI 线程遍历引用的 map 期间，
166 // worker 线程可能并发修改变量赋值/声明/作用域清理，导致迭代器失效或
167 // 读取撕裂值（unordered_map 并发读写是 UB）。对齐 variableCallback
168 // 的 snapshotLocalVariables() 修复模式。
169 auto locals = frame.env->snapshotLocalVariables();
170 for (const auto& kv : locals) {
171 entry.locals.emplace_back(kv.first, kv.second);
172 }
173 }
174 result.push_back(entry);
175 }
176 } catch (const std::exception& e) {
177 Logger::Warning(std::string("调用栈回调异常: ") + e.what(), "Debugger");
178 } catch (...) {
179 Logger::Warning("调用栈回调发生未知异常", "Debugger");
180 }
181 return result;
182 });
183
184 debugger_->reset();
185 debugger_->stepIn();
186}
