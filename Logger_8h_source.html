87 /// 设置日志文件输出路径（空字符串则关闭文件输出）
88 // Bug fix: fileStream_ 由 flushBuffer() 在 ioMutex_ 下访问，setOutputFile()
89 // 必须同时持有 ioMutex_ 才能避免与并发 flushBuffer() 产生数据竞争。
90 // Bug #87 fix: 统一锁顺序为 mutex_ → ioMutex_（与 log()/flush() 路径中
91 // 先持 mutex_ 再（通过 flushBuffer）持 ioMutex_ 的顺序一致），消除原实现
92 // 中 ioMutex_ → mutex_ 的反向加锁导致的潜在死锁。
93 bool setOutputFile(const std::string& path) {
94 std::lock_guard<std::mutex> lock(mutex_); // 先持 mutex_（与 log 路径一致）
95 std::lock_guard<std::mutex> ioLock(ioMutex_); // 再持 ioMutex_ 操作 fileStream_
96 if (fileStream_.is_open())
97 fileStream_.close();
98 if (path.empty())
99 return true;
100 fileStream_.open(path, std::ios::app);
101 return fileStream_.is_open();
102 }

104 /// 输出一条日志
105 // P2-2 fix: 双缓冲避免持锁 I/O。log() 仅在 mutex_ 下格式化+入队，
106 // 当队列达阈值或级别为 ERROR 时，swap 出缓冲并在 ioMutex_ 下批量写
107 // （mutex_ 持有时间从 O(行数) 降至 O(格式化单行)）。
108 void log(LogLevel level, const std::string& message, const std::string& source = "") {
109 if (level < level_.load() || level == LogLevel::NONE)
110 return;
111 std::vector<std::string> localBuffer;
112 bool shouldFlush = false;
113 {
114 std::lock_guard<std::mutex> lock(mutex_);
115 activeBuffer_.push_back(formatLine(level, message, source));
116 // ERROR 立即 flush；INFO/WARN/DEBUG 攒满阈值后批量 flush
117 if (level >= LogLevel::ERROR || activeBuffer_.size() >= FLUSH_THRESHOLD) {
118 localBuffer.swap(activeBuffer_);
119 shouldFlush = true;
120 }
121 }
122 if (shouldFlush) {
123 flushBuffer(localBuffer);
124 }
125 }
126
127 /// 强制刷新所有缓冲的日志（用于析构、关键节点）
128 void flush() {
129 std::vector<std::string> localBuffer;
130 {
131 std::lock_guard<std::mutex> lock(mutex_);
132 localBuffer.swap(activeBuffer_);
133 }
134 flushBuffer(localBuffer);
135 }

164 std::mutex mutex_;
165 // P2-2 fix: I/O 专用锁，与 mutex_ 解耦。log() 持有 mutex_ 仅做格式化+入队，
166 // 实际写文件/控制台在 flushBuffer() 中持有 ioMutex_ 进行，避免阻塞其他线程的 log() 调用
167 std::mutex ioMutex_;
168 std::atomic<LogLevel> level_{LogLevel::WARNING};
169 std::atomic<bool> consoleOutput_{true}; // P1 fix: atomic 防止数据竞争
170 std::ofstream fileStream_;
171 // P2-2 fix: 双缓冲。activeBuffer_ 由 log() 在 mutex_ 下追加；
172 // 达阈值或 ERROR 时 swap 出到本地，由 flushBuffer() 在 ioMutex_ 下批量写
173 std::vector<std::string> activeBuffer_;
174 // 缓冲触发阈值：DEBUG/INFO/WARN 攒满 32 行批量写；ERROR 立即写
175 static constexpr size_t FLUSH_THRESHOLD = 32;
176
177 /// 批量写缓冲到控制台/文件（持 ioMutex_，不持 mutex_）
178 void flushBuffer(const std::vector<std::string>& buffer) {
179 if (buffer.empty())
180 return;
181 std::lock_guard<std::mutex> ioLock(ioMutex_);
182 bool toConsole = consoleOutput_.load();
183 bool toFile = fileStream_.is_open();
184 for (const auto& line : buffer) {
185 if (toConsole) {
186 // 注意：原实现按级别选 cerr/cout，但缓冲后级别信息丢失。
187 // 改进：行内已含级别标签（[WARN]/[ERROR]），统一用 cerr 输出便于合并。
188 // 若需严格区分可改用 pair<LogLevel,string>，但当前简化为合并写 cout
189 std::cout << line << '\n';
190 }
191 if (toFile) {
192 fileStream_ << line << '\n';
193 }
194 }
195 if (toFile)
196 fileStream_.flush(); // 批量写后统一 flush
197 }

216 /// 生成带时间戳的格式化日志行
217 // P0-2 fix: std::localtime() 返回指向内部 static buffer 的指针，多线程并发调用时
218 // 产生数据竞争（项目已有多线程架构 QThread worker）。改用线程安全版本：
219 // Windows 用 localtime_s，POSIX 用 localtime_r。
220 // 同时 P1-3/Bug3 fix: 用栈缓冲 + snprintf 替代 ostringstream，避免多次堆分配
221 // （错误路径虽然低频，但 formatLine 是所有 log 的公共路径，INFO/DEBUG 级别高频）。
222 std::string formatLine(LogLevel level, const std::string& message, const std::string& source) {
223 // 时间戳
224 auto now = std::chrono::system_clock::now();
225 auto t = std::chrono::system_clock::to_time_t(now);
226 auto ms = std::chrono::duration_cast<std::chrono::milliseconds>(now.time_since_epoch()) % 1000;
227 std::tm tmBuf{};
228#ifdef _WIN32
229 localtime_s(&tmBuf, &t);
230#else
231 localtime_r(&t, &tmBuf);
232#endif
233 char timeBuf[16];
234 std::strftime(timeBuf, sizeof(timeBuf), "%H:%M:%S", &tmBuf);
235
236 // 预估容量并一次性分配：[HH:MM:SS.mmm] [LEVEL] [source] message
237 // 最坏情况：8+4+8+source.size()+1+message.size()，预留 64 字节余量
238 std::string result;
239 result.reserve(64 + source.size() + message.size());
240 result.append(timeBuf);
241 result.push_back('.');
242 // 毫秒固定 3 位补零
243 char msBuf[8];
244 std::snprintf(msBuf, sizeof(msBuf), "%03lld", static_cast<long long>(ms.count()));
245 result.append(msBuf);
246 result.append(" [");
247 result.append(levelString(level));
248 result.push_back(']');
249 if (!source.empty()) {
250 result.append(" [");
251 result.append(source);
252 result.push_back(']');
253 }
254 result.push_back(' ');
255 result.append(message);
256 return result;
257 }
