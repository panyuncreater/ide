717bool LabManualPanel::isMinorSection(const QString& headingText) {
718 // 关键字表：识别"次要章节"——主章节（目标/概念/步骤/验证/概述/简介/示例/原理）
719 // 始终展开，次章节（进阶/思考题/常见错误/延伸/拓展/参考/扩展/挑战/习题）可折叠
720 static const QStringList kMinorKeywords = {
721 QStringLiteral("进阶"), QStringLiteral("思考题"), QStringLiteral("延伸"), QStringLiteral("拓展"),
722 QStringLiteral("扩展"), QStringLiteral("参考"), QStringLiteral("挑战"), QStringLiteral("习题"),
723 QStringLiteral("常见错误"), QStringLiteral("常见问题"), QStringLiteral("FAQ"), QStringLiteral("补充"),
724 QStringLiteral("附录"), QStringLiteral("深入")};
725 for (const auto& kw : kMinorKeywords) {
726 if (headingText.contains(kw, Qt::CaseInsensitive))
727 return true;
728 }
729 return false;

732QString LabManualPanel::applyFolding(const std::string& markdown) const {
733 if (!foldMinorSections_) {
734 // 展开模式：原样返回
735 return QString::fromUtf8(markdown.c_str());
736 }
737
738 static const QRegularExpression reHeading(QStringLiteral("^(#{1,6})\\s+(.+?)\\s*$"));
739 const QString md = QString::fromUtf8(markdown.c_str());
740 const QStringList lines = md.split('\n');
741
742 QString result;
743 result.reserve(md.size());
744
745 bool inCodeBlock = false;
746 bool inFoldableSection = false;
747 int foldableLevel = 0; // 触发折叠的标题级别（仅 ## 视为可折叠，# 一级始终展开）
748
749 for (int i = 0; i < lines.size(); ++i) {
750 const QString& line = lines[i];
751 const QString trimmed = line.trimmed();
752
753 // 跟踪围栏代码块状态（``` 或 ~~~）
754 if (trimmed.startsWith(QStringLiteral("```")) || trimmed.startsWith(QStringLiteral("~~~"))) {
755 // AUDIT-P2 fix: 折叠区内不切换 inCodeBlock。折叠区内容已被跳过，
756 // 若切换 inCodeBlock 会导致状态泄漏到折叠区外——未闭合的代码块
757 // 会让 inCodeBlock 恒为 true，后续所有行（含真正标题）被当作
758 // 代码块内容跳过，文档剩余部分全部消失。
759 if (inFoldableSection) {
760 continue;
761 }
762 inCodeBlock = !inCodeBlock;
763 result += line + '\n';
764 continue;
765 }
766 if (inCodeBlock) {
767 // 代码块内的 # 不是标题，原样输出（折叠区已跳过）
768 if (!inFoldableSection)
769 result += line + '\n';
770 continue;
771 }
772
773 // 识别标题
774 QRegularExpressionMatch hm = reHeading.match(line);
775 if (hm.hasMatch()) {
776 const int level = hm.captured(1).length();
777 const QString headingText = hm.captured(2);
778
779 // 遇到任何标题，先关闭当前折叠区
780 if (inFoldableSection) {
781 // 仅当新标题级别 <= foldableLevel（同级或更高级）时才结束折叠
782 if (level <= foldableLevel) {
783 inFoldableSection = false;
784 }
785 }
786
787 if (!inFoldableSection && level == 2 && isMinorSection(headingText)) {
788 // 进入新的折叠区：保留标题行（让 TOC 链接仍可点击），
789 // 紧跟一行提示，然后跳过后续内容直到同级或更高级标题
790 result += line + '\n';
791 result += QStringLiteral("> ▶ 此节已折叠，点击右上角「展开全部章节」按钮查看完整内容\n");
792 inFoldableSection = true;
793 foldableLevel = level;
794 continue;
795 }
796
797 if (!inFoldableSection) {
798 result += line + '\n';
799 }
800 // 折叠区内的标题：跳过
801 continue;
802 }
803
804 // 普通行：折叠区内跳过，否则原样输出
805 if (!inFoldableSection) {
806 result += line + '\n';
807 }
808 }
809
810 // 移除末尾多余换行（split + join 会引入）
811 if (result.endsWith('\n'))
812 result.chop(1);
813 return result;
814}
