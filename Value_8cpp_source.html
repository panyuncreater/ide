40/// JSON 字符串转义（RFC 8259）。转义 `"`、`\`、控制字符（< 0x20）。
41std::string escapeJsonString(std::string_view s) {
42 std::string out;
43 out.reserve(s.size() + 2);
44 for (char c : s) {
45 unsigned char uc = static_cast<unsigned char>(c);
46 switch (uc) {
47 case '"':
48 out += "\\\"";
49 break;
50 case '\\':
51 out += "\\\\";
52 break;
53 case '\b':
54 out += "\\b";
55 break;
56 case '\f':
57 out += "\\f";
58 break;
59 case '\n':
60 out += "\\n";
61 break;
62 case '\r':
63 out += "\\r";
64 break;
65 case '\t':
66 out += "\\t";
67 break;
68 default:
69 if (uc < 0x20) {
70 char buf[8];
71 std::snprintf(buf, sizeof(buf), "\\u%04x", uc);
72 out += buf;
73 } else {
74 out += c;
75 }
76 break;
77 }
78 }
79 return out;
80}

85// equalsImpl — 相等比较实现（含环检测和深度保护）
86// ============================================================
87//
88// BUG-012 审计结论（2026-07-18）：visited 集合使用 RefCounted* 原始指针作为
89// 身份标识是正确且安全的，无需改用唯一 ID。原因：
90// 1. equalsImpl 是 const 成员函数，仅读取容器内容（const auto& a = aPtr->elements），
91// 不调用任何非 const 方法 → 不会触发 ensureUnique() → 不会发生 COW detach。
92// 2. visited 的生命周期是单次 equals() 调用（thread_local 局部集，调用前 clear），
93// 期间所有 RefCounted 对象的指针稳定不变。
94// 3. COW detach 只在 WRITE 操作（push/insert/字段赋值等）触发，而 equals 不写。
95// 4. 唯一理论风险是另一线程在比较期间修改容器触发 COW detach——但 MiniLang 容器
96// 非线程安全，并发访问本身就是 UB，与 visited 标识方式无关。
97// 因此原始指针作为身份标识在 equalsImpl 上下文中是不变量，无需引入额外 ID 开销。
98// toStringImpl 同理（仅读取，不触发 COW）。

100bool Value::equalsImpl(const Value& other, std::unordered_set<const void*>* visited, int depth) const {
101 // P1-7 fix: 深度保护，防止 [[[[...]]]] 线性嵌套导致栈溢出
102 if (depth >= MAX_EQUALS_DEPTH)
103 return false;
104
105 // int 和 float 之间可以比较（M3 fix: 避免大整数精度丢失）
106 if (isNumber() && other.isNumber() && getType() != other.getType()) {
107 if (isInt() && other.isFloat()) {
108 double d = other.floatVal();
109 // double 不是整数值、或超出 int64_t 范围、或 NaN → 不可能相等
110 if (std::isnan(d) || std::isinf(d))
111 return false;
112 double intPart;
113 if (std::modf(d, &intPart) != 0.0)
114 return false; // 有小数部分
115 if (OverflowCheck::doubleToIntOverflow(d))
116 return false;
117 return intVal() == static_cast<int64_t>(d);
118 }
119 if (isFloat() && other.isInt()) {
120 double d = floatVal();
121 if (std::isnan(d) || std::isinf(d))
122 return false;
123 double intPart;
124 if (std::modf(d, &intPart) != 0.0)
125 return false;
126 if (OverflowCheck::doubleToIntOverflow(d))
127 return false;
128 return static_cast<int64_t>(d) == other.intVal();
129 }
130 // P3 fix: 此行原为不可达死代码，保留以防御未来新增数值类型
131 return toDouble() == other.toDouble();
132 }
133 if (getType() != other.getType()) {
134 return false;
135 }
136 // R133-B fix: 按 ValueType 分组分派到 7 个独立 helper，主函数保留前置守卫
137 // （深度保护 + 跨类型数值比较 + 类型一致性检查），helper 共享状态全成员（box_）。
138 switch (getType()) {
139 case ValueType::VAL_INT:
140 case ValueType::VAL_FLOAT:
141 case ValueType::VAL_BOOL:
142 case ValueType::VAL_STRING:
143 case ValueType::VAL_NULL:
144 return equalsImplScalar(other);
145 case ValueType::VAL_ARRAY:
146 return equalsImplArray(other, visited, depth);
147 case ValueType::VAL_DICT:
148 return equalsImplDict(other, visited, depth);
149 case ValueType::VAL_INSTANCE:
150 return equalsImplInstance(other, visited, depth);
151 case ValueType::VAL_TUPLE:
152 return equalsImplTuple(other, visited, depth);
153 case ValueType::VAL_ENUM_VARIANT:
154 return equalsImplEnumVariant(other, visited, depth);
155 case ValueType::VAL_CLOSURE:
156 case ValueType::VAL_CHANNEL:
157 case ValueType::VAL_MUTEX:
158 case ValueType::VAL_RWLOCK:
159 case ValueType::VAL_THREAD:
160 case ValueType::VAL_COROUTINE: // R164 协程/生成器：引用相等（叶子对象）
161 return equalsImplLeafObject(other);
162 }
163 return false;
164}

365bool Value::equalsImplLeafObject(const Value& other) const {
366 // CLOSURE 与同步对象（CHANNEL/MUTEX/RWLOCK/THREAD）都是叶子对象，无递归子元素，
367 // 无需环检测。CLOSURE 按 functionId 比较，SYNC 对象按底层 Inner 指针比较。
368 switch (getType()) {
369 case ValueType::VAL_CLOSURE:
370 // L3 fix（2026-07-19）: 闭包相等性改为按 functionId 比较。
371 // 原契约按 name + env 比较，导致"同名同环境但函数体不同的闭包误判相等"（已知限制 L3）。
372 // 引入全局单调递增的 functionId（makeClosure 时分配，COW 拷贝保留），
373 // 同一闭包值的拷贝判等，独立 makeClosure 创建的闭包不判等。
374 // P2-14 fix: expired env 防护已不再需要——functionId 是值语义不受 env 生命周期影响。
375 // PERF-12: 直接通过 NaNBox 指针访问 ClosureData。
376 {
377 auto* cd1 = box_.asPtr<ClosureData>();
378 auto* cd2 = other.box_.asPtr<ClosureData>();
379 return cd1->functionId == cd2->functionId;
380 }
381 // R136 同步对象相等性：同一底层 Inner（指针相等）即为相等
382 // 语义：两个 channel Value 相等当且仅当它们指向同一通道（共享同一 Inner）
383 case ValueType::VAL_CHANNEL:
384 return box_.asPtr<ChannelData>()->inner == other.box_.asPtr<ChannelData>()->inner;
385 case ValueType::VAL_MUTEX:
386 return box_.asPtr<MutexData>()->inner == other.box_.asPtr<MutexData>()->inner;
387 case ValueType::VAL_RWLOCK:
388 return box_.asPtr<RwLockData>()->inner == other.box_.asPtr<RwLockData>()->inner;
389 case ValueType::VAL_THREAD:
390 return box_.asPtr<ThreadData>()->inner == other.box_.asPtr<ThreadData>()->inner;
391 // R164 协程/生成器：引用相等（同一 CoroutineData 指针即为相等）
392 case ValueType::VAL_COROUTINE:
393 return box_.asPtr<CoroutineData>() == other.box_.asPtr<CoroutineData>();
394 default:
395 return false;
396 }
397}

580 result += ", ";
581 if (tup[i].isString()) {
582 result += "\"";
583 result += tup[i].stringVal();
584 result += "\"";
585 } else {
586 result += tup[i].toStringImpl(visited, depth + 1);
587 }

590 if (tup.size() == 1)

592 result += ")";
593 visited.erase(ptr);

597std::string Value::toStringImplEnumVariant(std::unordered_set<const void*>& visited, int depth) const {
598 // R99 枚举与 ADT：字符串化为 EnumName.VariantName 或 EnumName.VariantName(arg1, arg2, ...)
599 // 对齐 Rust Debug 格式（Color.Red → "Color.Red"，Option.Some(42) → "Option.Some(42)"）。
600 // 字符串 fields 加引号（与 array/dict/tuple 一致的 repr 风格）。
601 auto* ptr = box_.asPtr<EnumVariantData>();
602 if (!visited.insert(ptr).second)
603 return "[cycle]";
604 std::string result;
605 result.reserve(ptr->enumName.size() + ptr->variantName.size() + ptr->fields.size() * 8 + 4);
606 result += ptr->enumName;
607 result += ".";
608 result += ptr->variantName;
609 if (!ptr->fields.empty()) {
610 result += "(";
611 for (size_t i = 0; i < ptr->fields.size(); ++i) {
612 if (i > 0)
613 result += ", ";
614 if (ptr->fields[i].isString()) {
615 result += "\"";
616 result += ptr->fields[i].stringVal();
617 result += "\"";
618 } else {
619 result += ptr->fields[i].toStringImpl(visited, depth + 1);
620 }
621 }
622 // 单字段 variant 保留尾逗号以对齐单元素元组语义（避免 Option.Some(42) 与
623 // Option.Some((42,)) 歧义——但实际两者 Value 类型不同，无需尾逗号区分；
624 // 此处不加尾逗号，对齐 Rust Debug 格式 Option.Some(42) 而非 Option.Some(42,)）
625 result += ")";
626 }
627 visited.erase(ptr);
628 return result;

656// ============================================================
657// toJsonImpl — JSON 序列化实现（R114 阶段 2）
658// ------------------------------------------------------------
659// 格式约定：
660// - 标量（int/float/bool/null）输出 JSON 原生类型
661// - 字符串输出 JSON 转义字符串
662// - 容器输出带 "_type" 标签的对象，便于回放器重建：
663// array → {"_type":"array","elements":[...]}
664// dict → {"_type":"dict","entries":[{"key":K,"value":V},...]}
665// instance→ {"_type":"instance","className":"Foo","fields":[{"name":N,"value":V}]}
666// tuple → {"_type":"tuple","elements":[...]}
667// enum → {"_type":"enum","enumName":"E","variantName":"V","fields":[...]}
668// closure → {"_type":"closure","name":"foo"}
669// - 环引用 → {"_type":"cycle"}
670// - 深度超限 → {"_type":"too_deep"}
671// ============================================================
672
673std::string Value::toJsonImpl(std::unordered_set<const void*>& visited, int depth) const {
674 if (depth >= MAX_TOSTRING_DEPTH)
675 return "{\"_type\":\"too_deep\"}";
676 switch (getType()) {
677 case ValueType::VAL_INT: {
678 char buf[32];
679 auto res = std::to_chars(buf, buf + sizeof(buf), intVal());
680 return std::string(buf, res.ptr);
681 }
682 case ValueType::VAL_FLOAT: {
683 char buf[64];
684 auto res = std::to_chars(buf, buf + sizeof(buf), box_.asFloat(), std::chars_format::general, 17);
685 if (res.ec != std::errc{})
686 return "null";
687 return std::string(buf, res.ptr);
688 }
689 case ValueType::VAL_BOOL:
690 return box_.asBool() ? "true" : "false";
691 case ValueType::VAL_NULL:
692 return "null";
693 case ValueType::VAL_STRING:
694 return "\"" + escapeJsonString(stringVal()) + "\"";
695 case ValueType::VAL_CLOSURE:
696 return "{\"_type\":\"closure\",\"name\":\"" + escapeJsonString(closureName()) + "\"}";
697 case ValueType::VAL_ARRAY: {
698 auto* ptr = box_.asPtr<ArrayData>();
699 if (!visited.insert(ptr).second)
700 return "{\"_type\":\"cycle\"}";
701 const auto& arr = ptr->elements;
702 std::string result = "{\"_type\":\"array\",\"elements\":[";
703 for (size_t i = 0; i < arr.size(); ++i) {
704 if (i > 0)
705 result += ",";
706 result += arr[i].toJsonImpl(visited, depth + 1);
707 }
708 result += "]}";
709 visited.erase(ptr);
710 return result;
711 }
712 case ValueType::VAL_DICT: {
713 auto* ptr = box_.asPtr<DictData>();
714 if (!visited.insert(ptr).second)
715 return "{\"_type\":\"cycle\"}";
716 const auto& dict = ptr->entries;
717 std::string result = "{\"_type\":\"dict\",\"entries\":[";
718 bool first = true;
719 for (const auto& kv : dict) {
720 if (!first)
721 result += ",";
722 first = false;
723 result += "{\"key\":";
724 // 键可能是 string/int/bool/float，统一用 dictKeyToValue 递归序列化
725 result += dictKeyToValue(kv.first).toJsonImpl(visited, depth + 1);
726 result += ",\"value\":";
727 result += kv.second.toJsonImpl(visited, depth + 1);
728 result += "}";
729 }
730 result += "]}";
731 visited.erase(ptr);
732 return result;
733 }
734 case ValueType::VAL_INSTANCE: {
735 auto* ptr = box_.asPtr<InstanceData>();
736 if (!visited.insert(ptr).second)
737 return "{\"_type\":\"cycle\"}";
738 std::string result = "{\"_type\":\"instance\",\"className\":\"";
739 result += escapeJsonString(ptr->className);
740 result += "\",\"fields\":[";
741 bool first = true;
742 for (const auto& kv : ptr->fields) {
743 if (!first)
744 result += ",";
745 first = false;
746 result += "{\"name\":\"";
747 result += escapeJsonString(kv.first);
748 result += "\",\"value\":";
749 result += kv.second.toJsonImpl(visited, depth + 1);
750 result += "}";
751 }
752 result += "]}";
753 visited.erase(ptr);
754 return result;
755 }
756 case ValueType::VAL_TUPLE: {
757 auto* ptr = box_.asPtr<TupleData>();
758 if (!visited.insert(ptr).second)
759 return "{\"_type\":\"cycle\"}";
760 const auto& tup = ptr->elements;
761 std::string result = "{\"_type\":\"tuple\",\"elements\":[";
762 for (size_t i = 0; i < tup.size(); ++i) {
763 if (i > 0)
764 result += ",";
765 result += tup[i].toJsonImpl(visited, depth + 1);
766 }
767 result += "]}";
768 visited.erase(ptr);
769 return result;
770 }
771 case ValueType::VAL_ENUM_VARIANT: {
772 auto* ptr = box_.asPtr<EnumVariantData>();
773 if (!visited.insert(ptr).second)
774 return "{\"_type\":\"cycle\"}";
775 std::string result = "{\"_type\":\"enum\",\"enumName\":\"";
776 result += escapeJsonString(ptr->enumName);
777 result += "\",\"variantName\":\"";
778 result += escapeJsonString(ptr->variantName);
779 result += "\",\"fields\":[";
780 for (size_t i = 0; i < ptr->fields.size(); ++i) {
781 if (i > 0)
782 result += ",";
783 result += ptr->fields[i].toJsonImpl(visited, depth + 1);
784 }
785 result += "]}";
786 visited.erase(ptr);
787 return result;
788 }
789 // R136 同步对象 JSON 序列化（叶子节点，无环检测需求）
790 case ValueType::VAL_CHANNEL:
791 return "{\"_type\":\"channel\"}";
792 case ValueType::VAL_MUTEX:
793 return "{\"_type\":\"mutex\"}";
794 case ValueType::VAL_RWLOCK:
795 return "{\"_type\":\"rwlock\"}";
796 case ValueType::VAL_THREAD:
797 return "{\"_type\":\"thread\"}";
798 // R164 协程/生成器 JSON 序列化（叶子节点）
799 case ValueType::VAL_COROUTINE:
800 return "{\"_type\":\"coroutine\"}";
801 }
802 return "null";
803}
