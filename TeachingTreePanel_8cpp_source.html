155void TeachingTreePanel::buildTree() {
156 // P2-1 fix: 从 PanelCatalog 取统一面板导航数据（原内嵌静态数组已迁移）
157 const auto& categories = PanelCatalog::categories();
158
159 // 顶部「代码编辑器」特殊项（非分类，直接点击切换回编辑器）
160 editorItem_ = new QTreeWidgetItem(tree_);
161 editorItem_->setText(0, mlTr("📝 代码编辑器"));
162 // UserRole 存储 panelId
163 editorItem_->setData(0, Qt::UserRole, QStringLiteral("editor"));
164 // 加粗显示，区别于分类
165 QFont editorFont = editorItem_->font(0);
166 editorFont.setBold(true);
167 editorItem_->setFont(0, editorFont);
168 idToItem_[QStringLiteral("editor")] = editorItem_;
169
170 // 分隔符项（视觉分隔）
171 auto* sep = new QTreeWidgetItem(tree_);
172 sep->setFlags(Qt::NoItemFlags);
173 sep->setText(0, QString());
174 sep->setSizeHint(0, QSize(-1, 9));
175 auto* sepLine = new QFrame(tree_);
176 sepLine->setFrameShape(QFrame::HLine);
177 sepLine->setFrameShadow(QFrame::Plain);
178 sepLine->setFixedHeight(1);
179 sepLine->setStyleSheet(QStringLiteral("QFrame { background: %1; border: none; max-height: 1px; }")
180 .arg(TeachingTheme::border().name()));
181 tree_->setItemWidget(sep, 0, sepLine);
182
183 // 收藏分类（动态，初始为空，loadFavoritesAndRecent 后填充）
184 favoritesCatItem_ = new QTreeWidgetItem(tree_);
185 favoritesCatItem_->setText(0, mlTr("★ 收藏"));
186 QFont favFont = favoritesCatItem_->font(0);
187 favFont.setBold(true);
188 favoritesCatItem_->setFont(0, favFont);
189 favoritesCatItem_->setData(0, Qt::UserRole, QStringLiteral("category"));
190 favoritesCatItem_->setHidden(true); // 无收藏时隐藏
191
192 // 最近访问分类（动态，初始为空）
193 recentCatItem_ = new QTreeWidgetItem(tree_);
194 recentCatItem_->setText(0, mlTr("🕑 最近访问"));
195 QFont recentFont = recentCatItem_->font(0);
196 recentFont.setBold(true);
197 recentCatItem_->setFont(0, recentFont);
198 recentCatItem_->setData(0, Qt::UserRole, QStringLiteral("category"));
199 recentCatItem_->setHidden(true);
200
201 // 6 大分类（UX-R fix: 按学习曲线递进排序，分类 tooltip 展示学习目标）
202 for (const auto& cat : categories) {
203 auto* catItem = new QTreeWidgetItem(tree_);
204 QString catText = QString::fromUtf8(cat.emoji) + " " + mlTr(cat.title);
205 catItem->setText(0, catText);
206 QFont catFont = catItem->font(0);
207 catFont.setBold(true);
208 catItem->setFont(0, catFont);
209 // 分类节点不存 panelId，标记为 category
210 catItem->setData(0, Qt::UserRole, QStringLiteral("category"));
211 // UX-R fix: 分类 tooltip = 学习目标一句话，初学者悬停即知这一类学什么
212 catItem->setToolTip(0, mlTr(cat.description));
213 // 分类文字颜色稍浅
214 QColor catColor = palette().color(QPalette::Text);
215 // 取 85% 不透明度让分类标题略灰
216 catColor.setAlphaF(0.85f);
217 catItem->setForeground(0, catColor);
218
219 for (const auto& leaf : cat.leaves) {
220 auto* leafItem = new QTreeWidgetItem(catItem);
221 QString leafText = QString::fromUtf8(leaf.emoji) + " " + mlTr(leaf.label);
222 // UX-R fix: 难度徽标 —— 进阶/高级面板加后缀提示，入门面板保持简洁；
223 // 初学者一眼就能看出哪些面板可以稍后再看，避免误入高级主题受挫
224 if (leaf.level >= 2) {
225 leafText += QStringLiteral(" ·") + mlTr(PanelCatalog::levelName(leaf.level));
226 }
227 leafItem->setText(0, leafText);
228 leafItem->setData(0, Qt::UserRole, QString::fromUtf8(leaf.id));
229 // B5/UX-R fix: 叶子节点 tooltip —— 难度分级 + 入口行为提示
230 leafItem->setToolTip(0, mlTr("难度：") + mlTr(PanelCatalog::levelName(leaf.level)) + QStringLiteral("\n") +
231 QString::fromUtf8("点击进入 ") + mlTr(leaf.label));
232 idToItem_[QString::fromUtf8(leaf.id)] = leafItem;
233 }
234 // 默认全部折叠，下面单独展开前两个分类
235 }
236
237 // B4: 恢复用户上次展开状态（QSettings 持久化），无记录时默认展开前两个分类
238 const QStringList savedExpanded =
239 QSettings().value(QStringLiteral("teachingTree/expandedCategories")).toStringList();
240 const QSet<QString> expandedSet(savedExpanded.begin(), savedExpanded.end());
241 if (expandedSet.isEmpty()) {
242 // 首次使用：默认展开前两个分类（入门导览 + 编译前端）
243 int expandedCount = 0;
244 for (int i = 0; i < tree_->topLevelItemCount(); ++i) {
245 auto* top = tree_->topLevelItem(i);
246 if (top && top->data(0, Qt::UserRole).toString() == QStringLiteral("category")) {
247 // 跳过收藏/最近访问分类（动态分类，由数据驱动展开）
248 if (top == favoritesCatItem_ || top == recentCatItem_)
249 continue;
250 tree_->expandItem(top);
251 if (++expandedCount >= 2)
252 break;
253 }
254 }
255 } else {
256 // 恢复：仅展开 savedExpanded 中记录的分类
257 for (int i = 0; i < tree_->topLevelItemCount(); ++i) {
258 auto* top = tree_->topLevelItem(i);
259 if (!top || top->data(0, Qt::UserRole).toString() != QStringLiteral("category"))
260 continue;
261 if (top == favoritesCatItem_ || top == recentCatItem_)
262 continue;
263 const QString title = categoryTitle(top);
264 if (expandedSet.contains(title)) {
265 tree_->expandItem(top);
266 } else {
267 tree_->collapseItem(top);
268 }
269 }
270 }
271
272 // 默认选中「代码编辑器」项
273 tree_->setCurrentItem(editorItem_);
274}

291void TeachingTreePanel::setCurrentPanel(const QString& panelId) {
292 auto it = idToItem_.find(panelId);
293 if (it == idToItem_.end())
294 return;
295 QTreeWidgetItem* item = it.value();
296 // AUDIT-P2 fix: 跳转前重置搜索过滤，确保目标项可见。
297 // 原实现在搜索过滤激活时直接 setCurrentItem(hidden item)，选中态被设置
298 // 但用户不可见——高亮"失效"，且树仍只显示过滤结果，与实际面板不一致。
299 // 跳转即重置过滤符合用户心智模型（跨面板导航应跳出搜索上下文）。
300 if (searchEdit_ && !searchEdit_->text().isEmpty()) {
301 QSignalBlocker blocker(searchEdit_); // 避免 clear() 重入 onSearchChanged
302 searchEdit_->clear();
303 onSearchChanged(QString()); // 显式恢复全部项可见
304 }
305 // 展开父分类（如果是叶子节点）
306 if (auto* parent = item->parent()) {
307 tree_->expandItem(parent);
308 }
309 tree_->setCurrentItem(item);
310 // 同步高亮（QTreeWidget 的当前项已有视觉高亮，这里无需额外处理）
311}

313void TeachingTreePanel::onSearchChanged(const QString& text) {
314 const QString needle = text.trimmed().toLower();
315 // 搜索开始（首次进入非空）：记录各分类折叠状态快照
316 if (!needle.isEmpty() && !searchActive_) {
317 searchActive_ = true;
318 collapsedBeforeSearch_.clear();
319 for (int i = 0; i < tree_->topLevelItemCount(); ++i) {
320 auto* top = tree_->topLevelItem(i);
321 if (!top || top->data(0, Qt::UserRole).toString() != QStringLiteral("category"))
322 continue;
323 if (!top->isExpanded()) {
324 collapsedBeforeSearch_.insert(categoryTitle(top));
325 }
326 }
327 }
328 for (int i = 0; i < tree_->topLevelItemCount(); ++i) {
329 auto* top = tree_->topLevelItem(i);
330 if (!top)
331 continue;
332 const QString role = top->data(0, Qt::UserRole).toString();
333 // 编辑器项与分隔符项不参与过滤，始终可见
334 if (role == QStringLiteral("editor") || role.isEmpty()) {
335 top->setHidden(false);
336 continue;
337 }
338 if (role != QStringLiteral("category")) {
339 top->setHidden(false);
340 continue;
341 }
342 bool anyVisible = false;
343 if (needle.isEmpty()) {
344 // 清空搜索：恢复全部叶子可见
345 for (int j = 0; j < top->childCount(); ++j) {
346 if (auto* child = top->child(j))
347 child->setHidden(false);
348 }
349 anyVisible = top->childCount() > 0;
350 // 搜索结束：恢复用户搜索前的折叠状态（而非强制全展开）
351 if (searchActive_) {
352 if (anyVisible) {
353 // collapsedBeforeSearch_ 中记录的保持折叠，其余展开
354 const QString title = categoryTitle(top);
355 if (collapsedBeforeSearch_.contains(title)) {
356 tree_->collapseItem(top);
357 } else {
358 tree_->expandItem(top);
359 }
360 }
361 } else {
362 // 非搜索触发的清空（如 setCurrentPanel 重置）：展开所有分类
363 if (anyVisible)
364 tree_->expandItem(top);
365 }
366 } else {
367 // 非空搜索：按文本包含匹配（不区分大小写）过滤叶子
368 for (int j = 0; j < top->childCount(); ++j) {
369 auto* child = top->child(j);
370 if (!child)
371 continue;
372 const bool match = child->text(0).toLower().contains(needle);
373 child->setHidden(!match);
374 if (match)
375 anyVisible = true;
376 }
377 // 有匹配则展开分类让结果可见，无匹配则折叠
378 if (anyVisible)
379 tree_->expandItem(top);
380 else
381 tree_->collapseItem(top);
382 }
383 // 无匹配时隐藏整个分类，避免空分类标题残留
384 top->setHidden(!anyVisible && !needle.isEmpty());
385 }
386 // 搜索完全清空后重置 searchActive_ 标志
387 if (needle.isEmpty() && searchActive_) {
388 searchActive_ = false;
389 collapsedBeforeSearch_.clear();
390 }
391}

393void TeachingTreePanel::onItemExpanded(QTreeWidgetItem* item) {
394 if (!item || item->data(0, Qt::UserRole).toString() != QStringLiteral("category"))
395 return;
396 // 收藏/最近访问分类不持久化（动态内容）
397 if (item == favoritesCatItem_ || item == recentCatItem_)
398 return;
399 // 收集当前所有展开的分类标题并保存
400 QStringList expanded;
401 for (int i = 0; i < tree_->topLevelItemCount(); ++i) {
402 auto* top = tree_->topLevelItem(i);
403 if (!top || top->data(0, Qt::UserRole).toString() != QStringLiteral("category"))
404 continue;
405 if (top == favoritesCatItem_ || top == recentCatItem_)
406 continue;
407 if (top->isExpanded())
408 expanded << categoryTitle(top);
409 }
410 QSettings().setValue(QStringLiteral("teachingTree/expandedCategories"), expanded);
411}

422QString TeachingTreePanel::categoryTitle(const QTreeWidgetItem* catItem) const {
423 if (!catItem)
424 return {};

462void TeachingTreePanel::rebuildFavoritesSection() {
463 if (!favoritesCatItem_)
464 return;
465 // 清空旧子项
466 while (favoritesCatItem_->childCount() > 0) {
467 auto* child = favoritesCatItem_->takeChild(0);
468 delete child;
469 }
470 // 按收藏顺序添加（保持 favorites_ 插入顺序，用 QStringList 保序）
471 const QStringList favList = QSettings().value(QStringLiteral("teaching/favorites")).toStringList();
472 for (const auto& id : favList) {
473 if (!idToItem_.contains(id))
474 continue;
475 auto* src = idToItem_[id];
476 auto* leaf = new QTreeWidgetItem(favoritesCatItem_);
477 leaf->setText(0, src->text(0));
478 leaf->setData(0, Qt::UserRole, id);
479 leaf->setToolTip(0, src->toolTip(0));
480 }
481 favoritesCatItem_->setHidden(favoritesCatItem_->childCount() == 0);
482}

503void TeachingTreePanel::toggleFavorite(const QString& panelId) {
504 if (panelId.isEmpty() || panelId == QStringLiteral("editor"))
505 return;
506 if (favorites_.contains(panelId)) {
507 favorites_.remove(panelId);
508 } else {
509 favorites_.insert(panelId);
510 // 维护保序的 QStringList（saveFavorites 用 QSet 会乱序，这里用 QStringList 保序）
511 QStringList favList = QSettings().value(QStringLiteral("teaching/favorites")).toStringList();
512 if (!favList.contains(panelId))
513 favList.append(panelId);
514 QSettings().setValue(QStringLiteral("teaching/favorites"), favList);
515 saveFavorites(); // 同步 favorites_ 集合（虽不用于顺序，保持一致）
516 rebuildFavoritesSection();
517 return;
518 }
519 // 移除收藏
520 QStringList favList = QSettings().value(QStringLiteral("teaching/favorites")).toStringList();
521 favList.removeAll(panelId);
522 QSettings().setValue(QStringLiteral("teaching/favorites"), favList);
523 rebuildFavoritesSection();
524}
