2// RegisterVM.cpp — PERF-14: 寄存器式虚拟机实现
3// ------------------------------------------------------------
4// 执行 RegBytecodeChunk。核心指令：常量加载、算术比较、控制流、
5// 全局/局部变量、函数调用、返回、容器、成员访问、闭包、类。
6//
7// 与栈式 VM(VM.cpp) 的关键差异（理解本文件的前提）：
8// · 无操作数栈。所有中间结果存于每个调用帧的固定寄存器窗口
9// RegCallFrame::registers（std::array<Value, MAX_REGISTERS=32>），
10// 用 reg(slot) 访问。因此 POP/DUP 退化为 REG_MOVE 或 no-op；
11// 函数局部变量 = 寄存器号（localCount 个寄存器从 0 开始）。
12// · 调用帧帧布局：新帧 registers[0..argCount-1] 填实参（方法调用时
13// registers[0] 是 this），其余寄存器由被调用者自由使用；
14// returnReg/returnIp 记录调用者的目标寄存器与返回地址。
15// · upvalue 全局地址编码：stackSlot = frameIdx * MAX_REGISTERS + slot，
16// 使每一帧的每个寄存器槽有全局唯一地址（C-3 fix）。closeUpvaluesFrom/
17// resolveOpenUpvalueSlot 据此跨帧定位，解决了多层嵌套闭包透传错位。
18// · 嵌套左值变异：MEMBER_SET/INDEX_SET 把接收者寄存器号记入
19// lastMutatedReceiverReg_，后续 WRITEBACK_* 整体替换回原处
20// （全局槽/局部寄存器/upvalue），与栈式 VM 的 lastMutatedReceiver_ 对应。
21// · 注册式 VM 把 hasError_ 纳入 isFinished 判定，且 execute()/stepOnce()
22// 共用成员 instructionCount_ 作为 DoS 防护预算（与栈式 VM 的局部计数不同）。

73bool RegisterVM::restoreFromSnapshot(const std::vector<Value>& registerValues,
74 const std::vector<std::pair<std::string, Value>>& globalsValues, size_t targetIp,
75 size_t targetFrameCount) {
76 if (!initialized_) {
77 // P2-12: 错误信息直接写入 diagnostics_，消除 lastError_ 字段
78 hasError_ = true;
79 diagnostics_.addError("RegisterVM 未初始化，无法回滚", 0, 0, DiagSource::RegisterVM);
80 return false;
81 }
82 if (targetFrameCount > frames_.size()) {
83 hasError_ = true;
84 diagnostics_.addError("回滚目标帧数超过当前帧数", 0, 0, DiagSource::RegisterVM);
85 return false;
86 }
87
88 // (1) 截断 frames_ 到 targetFrameCount（仅截断不扩展）
89 while (frames_.size() > targetFrameCount) {
90 frames_.pop_back();
91 }
92 // 设置栈顶帧 ip = targetIp（若仍有帧）
93 if (!frames_.empty()) {
94 currentFrame().ip = targetIp;
95 }
96
97 // (2) 将 registerValues 写入栈顶帧的 registers[]（截断到 MAX_REGISTERS=32）
98 if (!frames_.empty()) {
99 RegCallFrame& frame = currentFrame();
100 size_t copyCount = std::min(registerValues.size(), static_cast<size_t>(RegCallFrame::MAX_REGISTERS));
101 for (size_t i = 0; i < copyCount; ++i) {
102 frame.registers[i] = registerValues[i];
103 }
104 // 更新 registerCount 以覆盖所有写入的寄存器（避免 reg() 边界检查拒绝访问）
105 frame.registerCount = static_cast<uint8_t>(copyCount);
106 }
107
108 // (3) 按 name 更新 globalSlots_/globals_（已有变量覆盖，新变量忽略）
109 for (const auto& kv : globalsValues) {
110 const std::string& name = kv.first;
111 const Value& val = kv.second;
112 auto slotIt = globalNameToSlot_.find(name);
113 if (slotIt != globalNameToSlot_.end()) {
114 int slot = slotIt->second;
115 if (slot >= 0 && slot < static_cast<int>(globalSlots_.size())) {
116 globalSlots_[slot] = val;
117 }
118 } else {
119 auto it = globals_.find(name);
120 if (it != globals_.end()) {
121 it->second = val;
122 }
123 // 不创建新变量——回滚不应引入编译期未注册的变量
124 }
125 }
126
127 // (4) 清理 openUpvalues_（寄存器被覆盖，原 open upvalue 的 stackSlot 已指向新值）
128 // RegisterVM 的 openUpvalues_ 按 registerIndex 索引（每帧独立 0..31），
129 // 回滚后栈顶帧的寄存器全部被覆盖，所有 open upvalue 都应清理。
130 openUpvalues_.clear();
131
132 // (5) 清理 tryStack_（frameIndex >= targetFrameCount 的 handler 已悬垂）
133 while (!tryStack_.empty() && tryStack_.back().frameIndex >= targetFrameCount) {
134 tryStack_.pop_back();
135 }
136
137 // (6) 保守清空 pendingJumpStack_ + pendingException_（跨回滚未定义）
138 pendingJumpStack_.clear();
139 pendingException_ = Value::nullValue();
140
141 // (7) 清理错误/暂存状态
142 // P2-12: lastError_/lastErrorLine_ 已移除，错误信息统一由 diagnostics_ 承载
143 hasError_ = false;
144 diagnostics_.clear();
145 lastMutatedReceiverReg_ = 0;
146
147 return true;
148}

162// 字节码截断预扫描（对齐 StackVM 的 validateChunkInstructionBoundaries）：加载期
163// 对 chunk 线性走一遍指令边界，校验每条指令完整落在 code 范围内。与原主循环逐指令
164// 的 instructionSizeAt 检查逐点等价（对每个真实指令起点做同一计算），校验通过后
165// 主循环 release 不再逐指令展开变长指令长度。
166static bool validateRegChunkInstructionBoundaries(const RegBytecodeChunk& chunk) {
167 const size_t size = chunk.code.size();
168 size_t ip = 0;
169 while (ip < size) {
170 size_t instrSize = chunk.instructionSizeAt(ip);
171 if (ip + instrSize > size)
172 return false;
173 ip += instrSize;
174 }
175 return true;
176}

178void RegisterVM::initExecution(const RegisterCompileResult& result) {
179 resetState();
180 mainChunk_ = result.mainChunk;
181 functionChunks_ = result.functionChunks;
182
183 // 字节码截断预扫描：加载期一次性校验主 chunk 与全部函数 chunk 的指令边界。
184 // 校验失败属内部不变量破坏（编译器不会产出截断字节码），置 hasError_ 后由
185 // executeOneInstruction 入口的 hasError_ 短路返回 VM_RUNTIME_ERROR。
186 if (!validateRegChunkInstructionBoundaries(mainChunk_)) {
187 runtimeError("字节码截断: 主 chunk 指令不完整");
188 }
189 for (const auto& [funcName, funcChunk] : functionChunks_) {
190 if (!validateRegChunkInstructionBoundaries(funcChunk)) {
191 runtimeError("字节码截断: 函数 '" + funcName + "' 指令不完整");
192 break;
193 }
194 }
195
196 // 预分配到 MAX_FRAMES 上限：push_back 前已有 frames_.size() < MAX_FRAMES 检查，
197 // reserve 到上限可保证 push_back 永不 realloc / 抛 bad_alloc，消除帧推入异常安全窗口。
198 frames_.reserve(MAX_FRAMES);
199
200 // 初始化全局槽位
201 globalSlots_.resize(result.globalSlotCount);
202 for (size_t i = 0; i < result.globalSlotNames.size() && i < globalSlots_.size(); ++i) {
203 globalNameToSlot_[result.globalSlotNames[i]] = static_cast<int>(i);
204 }
205
206 // R99 enum 校验：加载 enum 元信息到 enumRegistry_ 供 REG_BUILD_ENUM_VARIANT 校验
207 for (const auto& info : result.enumInfos) {
208 enumRegistry_[info.name] = info;
209 }
210
211 // 创建 main 帧
212 RegCallFrame mainFrame;
213 mainFrame.chunk = &mainChunk_;
214 mainFrame.ip = 0;
215 // #8 fix: 定长 array，无需 resize，仅记录激活数量
216 mainFrame.registerCount =
217 static_cast<uint8_t>(mainChunk_.registerCount <= static_cast<int>(RegCallFrame::MAX_REGISTERS)
218 ? mainChunk_.registerCount
219 : static_cast<int>(RegCallFrame::MAX_REGISTERS));
220 frames_.push_back(std::move(mainFrame));
221
222 initialized_ = true;
223}

225VMResult RegisterVM::execute(const RegisterCompileResult& result) {
226 initExecution(result);
227 // AUDIT-R4 BUG-10 fix: 与 StackVM/JIT 对齐——执行期间抑制增量 GC 触发回调。
228 // 寄存器帧/全局槽上的临时 Value 不在 GcManager 根集中，回调中途触发会
229 // 误回收仅经寄存器可达的容器。详见 VM.cpp execute() 同名注释。
230 GcManager::CallbackSuppressor gcSuppressor;
231 // PERF-03 fix: 在循环入口加载一次指令预算上限到局部变量，避免主循环每条指令
232 // 都执行 RuntimeConfig::instance().maxInstructions()（含 atomic load + 单例访问）。
233 // 语义等价性：RuntimeConfig 设计为执行前配置（教学场景可调），execute() 期间
234 // 不应被修改；stepOnce() 路径仍逐次读取以支持 IDE 单步调试时实时调整预算。
235 const int64_t dynMaxInstr = RuntimeLimits::RuntimeConfig::instance().maxInstructions();
236 while (!frames_.empty() && !hasError_) {
237 VMResult r = executeOneInstruction();
238 // L14: VM_EXCEPTION_THROW 表示 runtimeError 被 try/catch 捕获，ip 已设置为 catchIp。
239 // 不视为错误，继续执行（不递增 ip，catchIp 由 throwException 写入 frame.ip）。
240 if (r == VMResult::VM_EXCEPTION_THROW)
241 continue;
242 if (r != VMResult::VM_OK)
243 return r;
244 // DoS 防护：instructionCount_ 为成员字段，execute() 与 stepOnce() 共享同一预算。
245 // 与栈式 VM 不同（其 execute 用局部计数器、stepOnce 用独立成员），RegisterVM
246 // 让全速执行与单步执行共用累计计数，保证两种模式下的指令上限语义一致。
247 // PERF-03 fix: dynMaxInstr 已在循环入口缓存，避免每条指令 atomic load。
248 if (++instructionCount_ >= dynMaxInstr) {
249 return runtimeError("指令数超出上限（可能存在死循环）");
250 }
251 }
252 return hasError_ ? VMResult::VM_RUNTIME_ERROR : VMResult::VM_OK;
253}

292VMResult RegisterVM::runtimeError(const std::string& msg, const std::string& diagCode) {
293 // BUG-016 审计结论（2026-07-18）：不在 runtimeError 中调用 stepCallback_ 是
294 // 设计决策，与 StackVM 的 runtimeError() 不调用 notifyStep() 行为一致，非 Bug。
295 // 原因：
296 // 1. stepCallback_/notifyStep 语义为"单条指令成功执行后的步进事件通知"，
297 // 用于调试器在单步模式下推进 IP 显示。错误路径不应触发步进事件——
298 // 指令并未成功执行，IP 未推进，调试器不应在错误指令处"步进"。
299 // 2. 错误传播路径：runtimeError 设置 hasError_=true → 下一次
300 // executeOneInstruction 入口短路返回 VM_RUNTIME_ERROR（见行 182-183）
301 // → execute()/stepOnce() 返回错误码 → IDE 控制器检测到错误，
302 // 通过 diagnostics_（getLastError()）显示错误信息（而非通过 stepCallback）。
303 // 3. isFinished() 将 hasError_ 纳入完成判定（行 138），单步模式下错误后
304 // 立即视为执行结束，IDE 在错误指令处停住而非继续空转。
305 // 4. 三后端一致性：StackVM VM.cpp:101 runtimeError、Interpreter::runtimeError
306 // （抛 RuntimeError 异常）、RegisterVM runtimeError 均不走步进回调路径。
307 // 若在 runtimeError 中添加 stepCallback 通知，反而会破坏三后端一致性，
308 // 且让调试器误以为指令成功执行。
309 // 因此各 execute* 函数中 `return runtimeError(...)` 路径不调用 stepCallback_
310 // 是正确的设计，BUG-016 为误报。
311 // P2-12: lastError_/lastErrorLine_ 字段已移除，错误信息直接写入 diagnostics_，
312 // getLastError()/getLastErrorLine() 从 diagnostics_ 派生（与 VM 对齐）。
313 // L14: try/catch 捕获 runtimeError — 若存在活动 try handler，将运行时错误转换为
314 // 可捕获的异常（string Value），对齐 StackVM 与 Interpreter 语义。
315 if (!tryStack_.empty() && !frames_.empty()) {
316 size_t currentFrameIdx = frames_.size() - 1;
317 if (tryStack_.back().frameIndex <= currentFrameIdx) {
318 return throwException(Value(msg));
319 }
320 }
321 hasError_ = true;
322 // 获取当前行号 + 列号
323 int errorLine = 0;
324 int col = 0;
325 if (!frames_.empty()) {
326 const auto& frame = frames_.back();
327 if (frame.chunk && frame.ip < frame.chunk->lines.size()) {
328 errorLine = frame.chunk->lines[frame.ip];
329 }
330 // BUG-IBACKEND-2: 从 chunk 读取列号（Compiler 未传入列号时默认 0）
331 if (frame.chunk && frame.ip < frame.chunk->columns.size()) {
332 col = frame.chunk->columns[frame.ip];
333 }
334 }
335 // BUG-IBACKEND-3: DiagSource 改为 RegisterVM 与 StackVM 区分；Logger 标签统一为 "RegisterVM"
336 // P2 fix (错误码优先匹配): 透传稳定诊断码到 addError，供 ErrorHintEngine 按 code 精确匹配
337 diagnostics_.addError(msg, errorLine, col, DiagSource::RegisterVM, diagCode);
338 Logger::Error("RegisterVM: " + msg, "RegisterVM");
339 return VMResult::VM_RUNTIME_ERROR;
340}

346VMResult RegisterVM::executeOneInstruction() {
347 // 单条寄存器指令执行的核心分派点，execute() 与 stepOnce() 共用。
348 // 流程：① 判空帧/hasError_ 短路；② 取 RegOp；③ (仅 debug) 用 instructionSizeAt
349 // 计算完整指令长度并做字节码截断边界检查——release 由 initExecution 预扫描覆盖；
350 // ④ 按类别 switch 到 executeXxx 方法，再由各方法按具体 RegOp 处理。
351 // 注意 REG_TYPE_CHECK / REG_SUPER_CALL 归入 executeMisc（杂项）——它们与
352 // 异常、I/O 共用同一分发桶，仅因历史归类，不影响语义。
353 if (hasError_)
354 return VMResult::VM_RUNTIME_ERROR;
355 if (frames_.empty())
356 return VMResult::VM_OK;
357
358 RegCallFrame& frame = currentFrame();
359 const RegBytecodeChunk& chunk = *frame.chunk;
360 size_t& ip = frame.ip;
361
362 if (ip >= chunk.code.size()) {
363 return runtimeError("指令指针越界");
364 }
365
366 RegOp op = static_cast<RegOp>(chunk.code[ip]);
367#ifndef NDEBUG
368 // 截断校验已上移至 initExecution 预扫描（validateRegChunkInstructionBoundaries），
369 // release 主循环不再逐指令展开变长指令长度；debug 保留作双保险。
370 size_t instrSize = chunk.instructionSizeAt(ip);
371 if (ip + instrSize > chunk.code.size()) {
372 return runtimeError("字节码截断: 指令不完整");
373 }
374#endif
375
376 // 按类别分发
377 switch (op) {
378 // 常量加载
379 case RegOp::REG_LOAD_CONST:
380 case RegOp::REG_LOAD_NULL:
381 case RegOp::REG_LOAD_TRUE:
382 case RegOp::REG_LOAD_FALSE:
383 case RegOp::REG_MOVE:
384 return executeConstants(op, ip);
385
386 // 算术
387 case RegOp::REG_ADD:
388 case RegOp::REG_SUB:
389 case RegOp::REG_MUL:
390 case RegOp::REG_DIV:
391 case RegOp::REG_MOD:
392 case RegOp::REG_NEGATE:
393 return executeArith(op, ip);
394
395 // 比较
396 case RegOp::REG_EQ:
397 case RegOp::REG_NEQ:
398 case RegOp::REG_LT:
399 case RegOp::REG_GT:
400 case RegOp::REG_LTE:
401 case RegOp::REG_GTE:
402 case RegOp::REG_NOT:
403 return executeCompare(op, ip);
404
405 // 变量
406 case RegOp::REG_LOAD_GLOBAL:
407 case RegOp::REG_STORE_GLOBAL:
408 case RegOp::REG_DEFINE_GLOBAL:
409 case RegOp::REG_DELETE_GLOBAL:
410 case RegOp::REG_LOAD_UPVALUE:
411 case RegOp::REG_STORE_UPVALUE:
412 case RegOp::REG_CLOSE_UPVALUE:
413 return executeVars(op, ip);
414
415 // 调用
416 case RegOp::REG_CALL:
417 case RegOp::REG_CALL_EXPR:
418 case RegOp::REG_TAIL_CALL: // L18 eng-tailcall
419 case RegOp::REG_RETURN:
420 case RegOp::REG_RETURN_NULL:
421 case RegOp::REG_MAKE_CLOSURE:
422 case RegOp::REG_METHOD_CALL:
423 case RegOp::REG_CLASS_NEW:
424 case RegOp::REG_DEFINE_CLASS:
425 return executeCalls(op, ip);
426
427 // 容器
428 case RegOp::REG_BUILD_ARRAY:
429 case RegOp::REG_BUILD_DICT:
430 case RegOp::REG_BUILD_TUPLE: // R98 元组与解构
431 case RegOp::REG_INDEX_GET:
432 case RegOp::REG_INDEX_SET:
433 case RegOp::REG_BUILD_ENUM_VARIANT: // R99 枚举与 ADT
434 case RegOp::REG_ENUM_VARIANT_NAME:
435 case RegOp::REG_ENUM_VARIANT_FIELD:
436 case RegOp::REG_MEMBER_GET:
437 case RegOp::REG_MEMBER_SET:
438 case RegOp::REG_SUPER_MEMBER_GET:
439 case RegOp::REG_INIT_FIELD:
440 case RegOp::REG_LEN: // R134 模式匹配扩展：容器长度
441 return executeContainers(op, ip);
442
443 // 控制流
444 case RegOp::REG_JUMP:
445 case RegOp::REG_JUMP_IF_FALSE:

483VMResult RegisterVM::throwException(Value thrownValue) {
484 // AUDIT-P1.1 fix: 异常传播中断 break/continue 续跳链，清空 pendingJumpStack_（与 StackVM 对齐）。
485 pendingJumpStack_.clear();
486 while (!tryStack_.empty()) {
487 RegTryHandler handler = tryStack_.back();
488 if (handler.frameIndex >= frames_.size()) {
489 tryStack_.pop_back();
490 continue;
491 }
492 // 跳转到 catch 块
493 while (frames_.size() > handler.frameIndex + 1) {
494 // C-3 fix: 弹帧前关闭指向该帧的 open upvalues，防止异常展开导致悬垂引用
495 size_t unwindFrameIdx = frames_.size() - 1;
496 // AUDIT-P2.5 fix: closeUpvaluesFrom 返回错误时立即终止异常展开，避免在损坏状态上继续
497 if (closeUpvaluesFrom(unwindFrameIdx * RegCallFrame::MAX_REGISTERS) != VMResult::VM_OK)
498 return VMResult::VM_RUNTIME_ERROR;
499 frames_.pop_back();
500 }
501 if (!frames_.empty()) {
502 // R7 fix: 跨帧异常展开后 handler.catchIp 可能相对于当前 chunk 越界
503 // （handler 在子帧注册，但展开到 caller 后 caller 的 chunk 不同）。
504 // defense-in-depth：入栈已校验，此处再校验当前 chunk 边界。
505 auto& curFrame = currentFrame();
506 if (handler.catchIp >= curFrame.chunk->code.size()) {
507 return runtimeError("throwException: catchIp 相对当前 chunk 越界");
508 }
509 // AUDIT-R5 R1 fix: 同帧 catch 命中时仅关闭 try 开始后开启的 upvalue
510 // （序号 >= handler.upvalueSeqFloor）。原 Bug #47 方案用帧基址 0 全量关闭，
511 // 会把 try 之前创建、catch 后仍存活的闭包 upvalue 误关为快照，
512 // 导致 catch 后闭包与变量不再共享（与 Interpreter/StackVM 不一致）。
513 // try 期间开启的 upvalue（含嵌套作用域复用低槽位的情形）仍全部关闭，
514 // 防止 catch 块复用寄存器时闭包读到错误值（保留 BUG-EXC-5 目标）。
515 // AUDIT-P2.5 fix: 关闭失败时立即终止异常处理，不跳转 catchIp
516 if (closeFrameUpvaluesSince(handler.frameIndex, handler.upvalueSeqFloor) != VMResult::VM_OK)
517 return VMResult::VM_RUNTIME_ERROR;
518 curFrame.ip = handler.catchIp;
519 // P1-4 fix: 异常值存入 pendingException_，由 REG_LOAD_EXCEPTION 读取到指定寄存器。
520 // 原方案固定写 R0 会覆盖用户变量/this（方法中 R0 是 this）。
521 pendingException_ = std::move(thrownValue);
522 }
523 tryStack_.pop_back();
524 // L14: 返回 VM_EXCEPTION_THROW 而非 VM_OK，通知调用方 ip 已被设置为 catchIp，
525 // 不应再递增 ip。dispatch 循环将此结果视为"异常已捕获，继续执行"。
526 return VMResult::VM_EXCEPTION_THROW;
527 }
528 // 未捕获的异常
529 // BUG-IBACKEND-4 fix: 三后端消息一致——统一 toString + 200 字符截断
530 // （对齐 StackVM VM.cpp:130-132 与 Interpreter）
531 std::string str = thrownValue.toString();
532 ErrorFormat::truncateForError(str);
533 return runtimeError("未捕获的异常: " + str);
534}

536VMResult RegisterVM::closeUpvaluesFrom(size_t fromSlot) {
537 // C-3 fix: stackSlot 编码为 frameIdx*32+slot，按全局地址范围 [fromSlot, ∞) 关闭。
538 // 从对应帧（由 frameIdx 解码）读取当前值并标记为已关闭，防止帧弹出后悬垂引用。
539 // AUDIT-P2.5 fix: 返回类型 void→VMResult，调用方可 fail-fast 传播错误。
540 bool hadError = false;
541 auto it = openUpvalues_.lower_bound(fromSlot);
542 while (it != openUpvalues_.end()) {
543 auto uv = it->second.uv.lock(); // AUDIT-R5 R1 fix: 条目改为携带序号的结构体
544 if (uv && !uv->isClosed) {
545 size_t frameIdx = uv->stackSlot / RegCallFrame::MAX_REGISTERS;
546 size_t slot = uv->stackSlot % RegCallFrame::MAX_REGISTERS;
547 if (frameIdx < frames_.size()) {
548 auto& targetFrame = frames_[frameIdx];
549 if (slot < targetFrame.registerCount) {
550 uv->value = targetFrame.registers[slot];
551 } else {
552 // AUDIT-P2.4 fix: slot 越界改为 runtimeError 以 fail-fast 暴露问题。
553 // 原实现仅打 Warning 继续执行，闭包静默捕获 null 难以排查根因。
554 // runtimeError 设置 hasError_=true，VM 在后续指令分发终止执行；
555 // 仍执行 isClosed=true + erase 以关闭 upvalue（保留默认 null 值），防止悬垂引用。
556 // AUDIT-P2.5 fix: 记录错误标志，函数末尾返回 VM_RUNTIME_ERROR，调用方立即 return。
557 (void)runtimeError("closeUpvaluesFrom: slot " + std::to_string(slot) + " >= registerCount " +
558 std::to_string(targetFrame.registerCount) +
559 " (frameIdx=" + std::to_string(frameIdx) + ")");
560 hadError = true;
561 }
562 } else {
563 // AUDIT-P2.4 fix: upvalue 指向已弹出的帧改为 runtimeError 以 fail-fast 暴露问题。
564 // 当前调用契约保证 frameIdx < frames_.size()（closeUpvaluesFrom 在帧弹出前调用），
565 // 此分支不可达。若触达说明调用契约被破坏，原实现仅打 Warning 会继续执行，
566 // upvalue 保留默认 null 值导致闭包读取错误值且难以排查。
567 // runtimeError 设置 hasError_=true，VM 在后续指令分发终止执行。
568 // AUDIT-P2.5 fix: 记录错误标志，函数末尾返回 VM_RUNTIME_ERROR，调用方立即 return。
569 (void)runtimeError("closeUpvaluesFrom: upvalue 指向已弹出的帧 (frameIdx=" + std::to_string(frameIdx) +
570 ", frames_.size()=" + std::to_string(frames_.size()) + ")");
571 hadError = true;
572 }
573 uv->isClosed = true;
574 }
575 ++it;
576 }
577 openUpvalues_.erase(openUpvalues_.lower_bound(fromSlot), openUpvalues_.end());
578 return hadError ? VMResult::VM_RUNTIME_ERROR : VMResult::VM_OK;
579}

581VMResult RegisterVM::closeFrameUpvaluesSince(size_t frameIdx, uint64_t seqFloor) {
582 // AUDIT-R5 R1 fix: 仅关闭 frameIdx 帧内、开启序号 >= seqFloor 的 open upvalue。
583 // 用于同帧 catch 命中：try 之前开启的 upvalue 保持 open（闭包与变量继续共享，
584 // 对齐 Interpreter/StackVM），try 期间开启的全部关闭（含嵌套作用域复用
585 // 低槽位的情形，Bug #47 场景），防止 catch 块复用寄存器时闭包读到错误值。
586 bool hadError = false;
587 const size_t frameBase = frameIdx * RegCallFrame::MAX_REGISTERS;
588 auto it = openUpvalues_.lower_bound(frameBase);
589 const auto rangeEnd = openUpvalues_.lower_bound(frameBase + RegCallFrame::MAX_REGISTERS);
590 while (it != rangeEnd) {
591 if (it->second.seq < seqFloor) {
592 ++it; // try 之前开启：保持 open
593 continue;
594 }
595 auto uv = it->second.uv.lock();
596 if (uv && !uv->isClosed) {
597 size_t fIdx = uv->stackSlot / RegCallFrame::MAX_REGISTERS;
598 size_t slot = uv->stackSlot % RegCallFrame::MAX_REGISTERS;
599 if (fIdx < frames_.size() && slot < frames_[fIdx].registerCount) {
600 uv->value = frames_[fIdx].registers[slot];
601 } else {
602 // 对齐 closeUpvaluesFrom 的 AUDIT-P2.4/P2.5 fail-fast 策略
603 (void)runtimeError("closeFrameUpvaluesSince: upvalue 槽位越界 (frameIdx=" + std::to_string(fIdx) +
604 ", slot=" + std::to_string(slot) + ")");
605 hadError = true;
606 }
607 uv->isClosed = true;
608 }
609 it = openUpvalues_.erase(it);
610 }
611 return hadError ? VMResult::VM_RUNTIME_ERROR : VMResult::VM_OK;
612}

739// R161 Watchpoint：pre-execution peek 当前 IP 指令的写入目标（不执行指令）
740WriteTarget RegisterVM::peekWriteTarget() const {
741 WriteTarget wt;
742 if (frames_.empty())
743 return wt;
744 const auto& frame = currentFrame();
745 if (!frame.chunk)
746 return wt;
747 const auto& code = frame.chunk->code;
748 size_t ip = frame.ip;
749 if (ip >= code.size())
750 return wt;
751
752 RegOp op = static_cast<RegOp>(code[ip]);
753 const auto& constants = frame.chunk->constants;
754 auto readShort = [&code](size_t offset) -> uint16_t { return code[offset] | (code[offset + 1] << 8); };
755 auto constStr = [&constants](uint16_t idx) -> std::string {
756 return (idx < constants.size()) ? constants[idx].stringVal() : std::string{};
757 };
758 auto slotToGlobalName = [this](int slot) -> std::string {
759 for (const auto& kv : globalNameToSlot_) {
760 if (kv.second == slot)
761 return kv.first;
762 }
763 return {};
764 };
765 auto regToLocalName = [&frame, ip](int reg) -> std::string { return frame.chunk->resolveSlotName(reg, ip); };
766
767 switch (op) {
768 // ---- 全局变量存储 ----
769 case RegOp::REG_STORE_GLOBAL:
770 case RegOp::REG_DEFINE_GLOBAL: {
771 if (ip + 3 >= code.size())
772 break;
773 wt.isWrite = true;
774 wt.varName = slotToGlobalName(readShort(ip + 2));
775 break;
776 }
777 // ---- upvalue 存储 ----
778 case RegOp::REG_STORE_UPVALUE: {
779 if (ip + 2 >= code.size())
780 break;
781 wt.isWrite = true;
782 uint8_t uvIdx = code[ip + 2];
783 if (uvIdx < frame.chunk->upvalues.size()) {
784 wt.varName = frame.chunk->upvalues[uvIdx].name;
785 }
786 break;
787 }
788 // ---- 字段写入 ----
789 case RegOp::REG_MEMBER_SET:
790 case RegOp::REG_INIT_FIELD: {
791 if (ip + 3 >= code.size())
792 break;
793 wt.isWrite = true;
794 wt.isFieldWrite = true;
795 // REG_MEMBER_SET: [op, obj, fieldIdx_lo, fieldIdx_hi, val] — fieldIdx at ip+2
796 // REG_INIT_FIELD: [op, fieldIdx_lo, fieldIdx_hi] — fieldIdx at ip+1
797 if (op == RegOp::REG_MEMBER_SET) {
798 wt.fieldName = constStr(readShort(ip + 2));
799 } else {
800 wt.fieldName = constStr(readShort(ip + 1));
801 }
802 break;
803 }
804 // ---- 字段写回（varIdx + fieldIdx）----
805 case RegOp::REG_WRITEBACK_MEMBER_VAR: {
806 if (ip + 4 >= code.size())
807 break;
808 wt.isWrite = true;
809 wt.isFieldWrite = true;
810 wt.varName = constStr(readShort(ip + 1));
811 wt.fieldName = constStr(readShort(ip + 3));
812 break;
813 }
814 // ---- 字段写回（localReg + fieldIdx）----
815 case RegOp::REG_WRITEBACK_MEMBER_LOCAL: {
816 if (ip + 3 >= code.size())
817 break;
818 wt.isWrite = true;
819 wt.isFieldWrite = true;
820 wt.localSlot = code[ip + 1];
821 wt.varName = regToLocalName(wt.localSlot);
822 wt.fieldName = constStr(readShort(ip + 2));
823 break;
824 }
825 // ---- 字段写回（uvIdx + fieldIdx）----
826 case RegOp::REG_WRITEBACK_MEMBER_UPVALUE: {
827 if (ip + 3 >= code.size())
828 break;
829 wt.isWrite = true;
830 wt.isFieldWrite = true;
831 uint8_t uvIdx = code[ip + 1];
832 if (uvIdx < frame.chunk->upvalues.size()) {
833 wt.varName = frame.chunk->upvalues[uvIdx].name;
834 }
835 wt.fieldName = constStr(readShort(ip + 2));
836 break;
837 }
838 // ---- 索引写入 ----
839 case RegOp::REG_INDEX_SET: {
840 wt.isWrite = true;
841 wt.isIndexWrite = true;
842 break;
843 }
844 // ---- 索引写回（varIdx）----
845 case RegOp::REG_WRITEBACK_INDEX_VAR: {
846 if (ip + 2 >= code.size())
847 break;
848 wt.isWrite = true;
849 wt.isIndexWrite = true;
850 wt.varName = constStr(readShort(ip + 1));
851 break;
852 }
853 // ---- 索引写回（localReg）----
854 case RegOp::REG_WRITEBACK_INDEX_LOCAL: {
855 if (ip + 1 >= code.size())
856 break;
857 wt.isWrite = true;
858 wt.isIndexWrite = true;
859 wt.localSlot = code[ip + 1];
860 wt.varName = regToLocalName(wt.localSlot);
861 break;
862 }
863 // ---- 索引写回（uvIdx）----
864 case RegOp::REG_WRITEBACK_INDEX_UPVALUE: {
865 if (ip + 1 >= code.size())
866 break;
867 wt.isWrite = true;
868 wt.isIndexWrite = true;
869 uint8_t uvIdx = code[ip + 1];
870 if (uvIdx < frame.chunk->upvalues.size()) {
871 wt.varName = frame.chunk->upvalues[uvIdx].name;
872 }
873 break;
874 }
875 default:
876 break;
877 }
878 return wt;
879}
