127/// 将服务层 BackendExecResult 转换为面板内部 BackendResult
128BackendComparePanel::BackendResult convertServiceResult(const ::BackendExecResult& sr) {
129 BackendComparePanel::BackendResult r;
130 r.status = sr.success ? "OK" : "ERROR";
131 r.errorMessage = sr.success ? std::string() : (sr.errorPrefix + ": " + sr.errorMsg);
132 r.elapsedMicros = sr.elapsedMs * 1000; // ms → μs
133 // 按行切分输出（保留原 BackendComparePanel 输出格式）
134 std::string s = sr.output;
135 std::istringstream iss(s);
136 std::string line;
137 while (std::getline(iss, line)) {
138 r.outputLines.push_back(line);
139 }
140 // 若有错误，附加错误信息到输出末尾（保持原行为：错误也作为输出展示）
141 if (!sr.success && !sr.errorMsg.empty()) {
142 r.outputLines.push_back(std::string("[") + sr.errorPrefix + "] " + sr.errorMsg);
143 }
144 return r;
145}

206/// 逐行比对三后端输出（按最大行数对齐，缺失行视为空），统计差异行数，
207/// 生成「三后端输出一致 / 差异 N 行」的汇总文本，供差异标签显示。
208// AUDIT-P2 fix: 三后端全错误时不能误报「一致性 PASS」——原实现只比较 outputLines，
209// 不检查 status。三后端均报相同错误时 diffCount=0 误报 PASS，用户以为代码正确。
210QString BackendComparePanel::buildDiffSummary(const BackendResult& a, const BackendResult& b, const BackendResult& c) {
211 // 先检查三后端是否全部成功
212 bool allOk = (a.status == "OK" && b.status == "OK" && c.status == "OK");
213 // 三后端状态是否一致
214 bool statusConsistent = (a.status == b.status && b.status == c.status);
215
216 // 三后端输出差异比对
217 size_t maxLines = std::max({a.outputLines.size(), b.outputLines.size(), c.outputLines.size()});
218 int diffCount = 0;
219 for (size_t i = 0; i < maxLines; ++i) {
220 std::string la = (i < a.outputLines.size()) ? a.outputLines[i] : "";
221 std::string lb = (i < b.outputLines.size()) ? b.outputLines[i] : "";
222 std::string lc = (i < c.outputLines.size()) ? c.outputLines[i] : "";
223 if (la != lb || lb != lc)
224 diffCount++;
225 }
226
227 // 三后端均失败时不应报 PASS
228 if (!allOk) {
229 if (statusConsistent && diffCount == 0) {
230 return QString::fromUtf8("三后端均失败（错误一致）— 请修复代码");
231 }
232 return QString::fromUtf8("三后端均失败且错误不一致 — 请修复代码");
233 }
234
235 if (diffCount == 0) {
236 return QString::fromUtf8("三后端输出一致（%1 行）— 一致性 PASS").arg(maxLines);
237 }
238 return QString::fromUtf8("三后端差异 %1 行 / 共 %2 行 — 一致性 FAIL").arg(diffCount).arg(maxLines);
239}
