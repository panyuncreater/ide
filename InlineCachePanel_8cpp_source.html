336void InlineCachePanel::parseTypeSequence(const std::string& text, std::vector<std::string>& out) {
337 out.clear();
338 std::string current;
339 for (char c : text) {
340 if (c == ',' || c == ',') { // 支持中英文逗号
341 if (!current.empty()) {
342 // trim
343 size_t start = current.find_first_not_of(" \t");
344 size_t end = current.find_last_not_of(" \t");
345 if (start != std::string::npos) {
346 out.push_back(current.substr(start, end - start + 1));
347 }
348 current.clear();
349 }
350 } else {
351 current += c;
352 }
353 }
354 if (!current.empty()) {
355 size_t start = current.find_first_not_of(" \t");
356 size_t end = current.find_last_not_of(" \t");
357 if (start != std::string::npos) {
358 out.push_back(current.substr(start, end - start + 1));
359 }
360 }

377/// IC 状态转换核心算法（经典 IC 理论）：
378/// - Uninitialized -> Monomorphic（首次观测）
379/// - Monomorphic -> Polymorphic（观测到新类型）
380/// - Polymorphic -> Megamorphic（缓存达 5 项）
381/// - Megamorphic 保持（永远未命中）
382IcSimStep InlineCachePanel::simulateOneStep(const std::string& observedType) {
383 IcSimStep step;
384 step.stepNumber = static_cast<int>(simSteps_.size()) + 1;
385 step.observedType = observedType;
386 step.stateBefore = simState_;
387
388 // 检查是否命中缓存
389 bool inCache = false;
390 for (const auto& t : simCache_) {
391 if (t == observedType) {
392 inCache = true;
393 break;
394 }
395 }
396 step.hit = inCache;
397 simTotalCalls_++;
398
399 switch (simState_) {
400 case IcSimState::Uninitialized:
401 // 首次观测 -> 进入单态
402 simCache_.push_back(observedType);
403 simState_ = IcSimState::Monomorphic;
404 step.note = "首次观测，进入单态（缓存该类型）";
405 break;
406
407 case IcSimState::Monomorphic:
408 if (inCache) {
409 step.note = "命中单态缓存，保持单态";
410 } else {
411 // 观测到新类型 -> 进入多态
412 simCache_.push_back(observedType);
413 simState_ = IcSimState::Polymorphic;
414 step.note = "未命中，观测到新类型，进入多态（缓存扩展为 2 项）";
415 }
416 break;
417
418 case IcSimState::Polymorphic:
419 if (inCache) {
420 step.note = "命中多态缓存，保持多态";
421 } else {
422 // 观测到新类型
423 if (static_cast<int>(simCache_.size()) < InlineCacheLibrary::kPolymorphicMax) {
424 simCache_.push_back(observedType);
425 step.note = "未命中，缓存未满，添加新类型保持多态";
426 } else {
427 // 缓存达 4 项，第 5 种类型触发溢出 -> 进入超多态
428 simCache_.clear(); // 放弃缓存
429 simState_ = IcSimState::Megamorphic;
430 step.note = "未命中，缓存达上限，放弃缓存进入超多态";
431 }
432 }
433 break;
434
435 case IcSimState::Megamorphic:
436 // 超多态：永远未命中（走通用查找）
437 step.note = "超多态，放弃缓存，走通用查找";
438 break;
439 }
440
441 if (inCache) {
442 simHits_++;
443 }
444
445 step.stateAfter = simState_;
446 step.cacheContent = formatCache(simCache_);
447 if (simState_ == IcSimState::Megamorphic) {
448 step.cacheContent = "（溢出，已放弃）";
449 }
450
451 simSteps_.push_back(step);
452 return step;
453}

455void InlineCachePanel::renderSimSteps() {

457 for (int i = 0; i < static_cast<int>(simSteps_.size()); ++i) {

542void InlineCachePanel::onRunSimulation() {
543 // 重置后完整运行
544 onResetSimulation();
545
546 std::string text = typeSeqEdit_->text().toStdString();
547 std::vector<std::string> types;
548 parseTypeSequence(text, types);
549
550 if (types.empty()) {
551 simSummary_->setHtml(tr("<p style='color:red;'>请输入类型序列（逗号分隔，如 int,int,float）</p>"));
552 return;
553 }
554
555 for (const auto& t : types) {
556 simulateOneStep(t);
557 }
558 renderSimSteps();
559 renderSimSummary();
560}

562void InlineCachePanel::onStepForward() {
563 // 若有待执行类型，执行下一步
564 if (pendingTypes_.empty()) {
565 // 从输入框重新解析
566 std::string text = typeSeqEdit_->text().toStdString();
567 parseTypeSequence(text, pendingTypes_);
568 if (pendingTypes_.empty()) {
569 simSummary_->setHtml(tr("<p style='color:red;'>请输入类型序列后单步执行</p>"));
570 return;
571 }
572 }
573 std::string next = pendingTypes_.front();
574 pendingTypes_.erase(pendingTypes_.begin());
575 simulateOneStep(next);
576 renderSimSteps();
577 renderSimSummary();
578}
