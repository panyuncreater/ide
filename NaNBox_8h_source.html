11// ============================================================
12// NaNBox — 8 字节 NaN-boxing 值编码（PERF-12 基础设施）
13// ------------------------------------------------------------
14// 将多种类型编码到单个 64 位字中，消除 shared_ptr 引用计数开销
15// 和 variant 的 24 字节内存开销。
16//
17// 编码方案（基于 IEEE 754 double 的 quiet NaN 空间）：
18//
19// 1. VAL_FLOAT: 直接存储 double 的 64 位原始位（非 NaN-boxed 时）
20// 2. VAL_INT: tag = 0x7FF8..., payload = int48 值（低 48 位）
21// 3. VAL_BOOL: tag = 0x7FF9..., payload = 0/1
22// 4. VAL_NULL: 0x7FFA000000000000
23// 5. 指针类型: tag = 0x7FFB..., payload = 48 位指针
24//
25// 布局：高 16 位（bits 63-48）为 tag field，低 48 位（bits 47-0）为 payload。
26// tag 与 payload 完全不重叠，避免 payload 满载时污染 tag。
27// 所有 tag base 设置 IEEE 754 quiet NaN 位（bit 51 = 1）+ exponent 全 1。
28//
29// 限制：
30// - int64_t 超出 int48 范围（|v| >= 2^47）时无法内联，调用者需 fallback
31// - 指针类型使用 48 位（x86-64 用户态虚拟地址空间）
32// - 指针类型不管理引用计数，调用者需确保指针有效性
33// ============================================================
34

51 static NaNBox fromFloat(double d) {
52 NaNBox box;
53 std::memcpy(&box.bits_, &d, sizeof(double));
54 // 如果 double 恰好落入 NaN-boxed tag 范围，需要规范化
55 // 罕见情况：double 是一个 quiet NaN 且高 16 位与我们的 tag 冲突
56 // （INT_TAG_BASE=0x7FF8, BOOL_TAG_BASE=0x7FF9, NULL_BITS=0x7FFA, PTR_TAG_BASE=0x7FFB）
57 // 必须将整个 64 位替换为 NAN_BOXED_FLOAT_MARKER（0x7FFC...），
58 // 丢弃原始 NaN payload——否则低 48 位可能形成合法 int/bool/null/ptr 位模式，
59 // 导致 tag() 误判。NaN != NaN（IEEE 754），payload 丢弃不影响语义正确性。
60 if (isBoxedNaN(box.bits_)) {
61 box.bits_ = NAN_BOXED_FLOAT_MARKER;
62 }
63 return box;
64 }

66 static NaNBox fromInt(int64_t i) {
67 // BUG-001 fix: 超范围值不再 abort 整个 IDE 进程，而是降级为 float 编码。
68 // 设计权衡：
69 // - 唯一合法调用方 Value(int64_t) 已在 canEncodeInt 检查后走 BoxedIntData 路径，
70 // 直接调用 fromInt 的代码若超范围属于编程错误。
71 // - 但 abort 会让 IDE 进程整体崩溃（用户代码 bug 不应导致 IDE 退出），
72 // 故采用 BUG_REPORT.md 方案 B：降级为 float，损失精度但不崩溃。
73 // - 调用方可用 isInt() 区分降级情形；正常运行路径不会触发此分支。
74 if (!canEncodeInt(i)) {
75 LOG_WARNING("NaNBox::fromInt 超范围降级为 float: value=" + std::to_string(i), "NaNBox");
76 return fromFloat(static_cast<double>(i));
77 }
78 NaNBox box;
79 // 将 int64 截断为 int48（保留符号位扩展）
80 uint64_t payload = static_cast<uint64_t>(i) & INT48_MASK;
81 box.bits_ = INT_TAG_BASE | payload;
82 return box;
83 }

97 static NaNBox fromPtr(const void* ptr) {
98 NaNBox box;
99 uint64_t ptrBits = reinterpret_cast<uint64_t>(ptr);
100 // BUG-002 fix: 高 16 位非零的指针（内核地址 / ASLR 极端布局 / 自定义分配器）
101 // 无法用 48 位 payload 编码。原实现直接 abort 让 IDE 整体崩溃，对用户不友好。
102 // 改为：先记录详细错误日志（指针值 + 来源），再 abort。
103 // - 指针无法"降级"为其他类型（不同于 fromInt 可降级为 float），故仍需 abort
104 // 防止后续 asPtr 解码出无效地址触发更难诊断的访问冲突。
105 // - 日志记录后 abort，开发者可在日志中定位首个触发点。
106 // - 正常 x86-64 用户空间地址高 16 位为 0，此分支仅在异常环境下触发。
107 //
108 // L9 fix 审计结论（2026-07-19）：
109 // 评估为"已知限制但实际不可触发"，保留 abort 作为防御性编程最后防线。
110 // 理由：(1) x86-64 用户空间虚拟地址范围 [0x0, 0x0000_7FFF_FFFF_FFFF] 由
111 // 硬件规范保证（Windows/Linux/macOS 均遵循），高 16 位始终为 0；
112 // (2) 侵入式 RefCounted 通过 new 分配，必在用户空间堆内，高 16 位为 0；
113 // (3) 改造为 BoxedPtrData 堆包装路径需新增 ValueType + 修改所有 asPtr
114 // 调用点（GcManager/Value.cpp/MemoryModelPanel 等约 20+ 处）+ 三后端同步，
115 // 改造风险高于 abort 本身（教学场景下不可触发）。
116 // 与 L10（PCH 全量重编译）的处理策略一致——硬件规范保证的不变量保留
117 // 防御性 abort，文档化说明而非改造代码。
118 if ((ptrBits & ~PTR_MASK) != 0) {
119 char buf[32];
120 std::snprintf(buf, sizeof(buf), "0x%016llx", static_cast<unsigned long long>(ptrBits));
121 LOG_ERROR("NaNBox::fromPtr 高位非零指针无法编码: " + std::string(buf) +
122 "（48 位 payload 范围内才能安全编码）",
123 "NaNBox");
124 Logger::instance().flush();
125 std::abort();
126 }
127 box.bits_ = PTR_TAG_BASE | (ptrBits & PTR_MASK);
128 return box;
129 }

143 Tag tag() const {
144 if (!isBoxedNaN(bits_))
145 return Tag::FLOAT;
146 uint64_t tagField = bits_ & TAG_FIELD_MASK;
147 if (tagField == INT_TAG_BASE)
148 return Tag::INT;
149 if (tagField == BOOL_TAG_BASE)
150 return Tag::BOOL;
151 if (bits_ == NULL_BITS)
152 return Tag::NUL;
153 if (tagField == PTR_TAG_BASE)
154 return Tag::POINTER;
155 // 不认识的 tag（规范化后的 float NaN），当作 float
156 return Tag::FLOAT;
157 }

235 // ---- 常量定义 ----
236 // IEEE 754 quiet NaN：sign(1) + exponent_all_ones(11) + quiet_nan_bit(1) = 0x7FF8
237 // 高 16 位（bits 63-48）作为 tag field，低 48 位（bits 47-0）作为 payload
238 // tag 与 payload 完全不重叠，确保 payload 满载不会污染 tag
239 //
240 // Tag 编码（高 16 位，均含 quiet NaN bit 51）：
241 // 0x7FF8 = INT (payload = int48 在低 48 位)
242 // 0x7FF9 = BOOL (payload = 0/1 在最低位)
243 // 0x7FFA = NULL (无 payload)
244 // 0x7FFB = POINTER (payload = 48 位指针)
245
246 static constexpr uint64_t TAG_FIELD_MASK = 0xFFFF000000000000ULL; // 高 16 位
247 static constexpr uint64_t INT48_MASK = 0x0000FFFFFFFFFFFFULL; // 低 48 位
248 static constexpr uint64_t INT47_SIGN_BIT = 0x0000800000000000ULL; // bit 47（int48 符号位）
249 static constexpr uint64_t PTR_MASK = 0x0000FFFFFFFFFFFFULL; // 低 48 位
250
251 // Tag 基址（高 16 位，均含 quiet NaN bit 51）
252 static constexpr uint64_t INT_TAG_BASE = 0x7FF8000000000000ULL; // int48
253 static constexpr uint64_t BOOL_TAG_BASE = 0x7FF9000000000000ULL; // bool
254 static constexpr uint64_t NULL_BITS = 0x7FFA000000000000ULL; // null
255 static constexpr uint64_t PTR_TAG_BASE = 0x7FFB000000000000ULL; // 指针
256
257 // 规范化的 float NaN 标记：当 double 的位模式落入 NaN-boxed 范围时，
258 // 用此值替代（仍是一个 quiet NaN，读取时按 double 解码）
259 static constexpr uint64_t NAN_BOXED_FLOAT_MARKER = 0x7FFC000000000000ULL;
260
261 /// 检查 64 位值是否是 NaN-boxed 值（高 16 位在我们的 tag 范围内）
262 static bool isBoxedNaN(uint64_t bits) {
263 uint64_t tagField = bits & TAG_FIELD_MASK;
264 return tagField >= INT_TAG_BASE && tagField <= PTR_TAG_BASE;
265 }
