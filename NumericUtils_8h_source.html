19/// 加法 a + b 是否溢出 int64_t
20constexpr bool addOverflow(int64_t a, int64_t b) {
21 return (b > 0 && a > INT64_MAX - b) || (b < 0 && a < INT64_MIN - b);
22}
23
24/// 减法 a - b 是否溢出 int64_t
25constexpr bool subOverflow(int64_t a, int64_t b) {
26 return (b < 0 && a > INT64_MAX + b) || (b > 0 && a < INT64_MIN + b);
27}
28
29/// 乘法 a * b 是否溢出 int64_t（内部已处理 -1 * INT64_MIN）
30constexpr bool mulOverflow(int64_t a, int64_t b) {
31 if (a == 0 || b == 0)
32 return false;
33 if (a == -1 && b == INT64_MIN)
34 return true;
35 if (b == -1 && a == INT64_MIN)
36 return true;
37 return (a > 0 && b > 0 && a > INT64_MAX / b) || (a > 0 && b < 0 && b < INT64_MIN / a) ||
38 (a < 0 && b > 0 && a < INT64_MIN / b) || (a < 0 && b < 0 && a < INT64_MAX / b);
39}
40
41/// 除法 a / b 是否为 UB（仅 INT64_MIN / -1；b == 0 由调用方单独处理）
42constexpr bool divOverflow(int64_t a, int64_t b) {
43 return a == INT64_MIN && b == -1;
44}
45
46/// 取模 a % b 是否为 UB（仅 INT64_MIN % -1；b == 0 由调用方单独处理）
47constexpr bool modOverflow(int64_t a, int64_t b) {
48 return a == INT64_MIN && b == -1;
49}
50
51/// 一元取负 -a 是否为 UB（仅 -INT64_MIN）
52constexpr bool negateOverflow(int64_t a) {
53 return a == INT64_MIN;
54}
55
56/// double 转 int64_t 是否溢出
57/// 注意：使用 >= 上界，因 double(INT64_MAX) 上取整为 2^63，
58/// 该值无法表示为 int64_t（M-新3 fix，统一修复 Interpreter.cpp 的旧版 > Bug）
59inline bool doubleToIntOverflow(double d) {
60 return d < static_cast<double>(INT64_MIN) || d >= -static_cast<double>(INT64_MIN);
61}

95/// 执行整数/浮点算术运算（不含字符串拼接，调用方预先处理）
96/// 输入：两操作数已确认为数值类型（isNumber()=true）
97inline ArithResult computeArith(ArithOp op, bool leftIsInt, int64_t leftInt, double leftFloat, bool rightIsInt,
98 int64_t rightInt, double rightFloat) {
99 ArithResult r;
100 r.isIntResult = leftIsInt && rightIsInt;
101
102 switch (op) {
103 case ArithOp::Add:
104 if (r.isIntResult) {
105 if (OverflowCheck::addOverflow(leftInt, rightInt)) {
106 r.status = ArithStatus::IntOverflow;
107 return r;
108 }
109 r.intVal = leftInt + rightInt;
110 } else {
111 r.floatVal = leftFloat + rightFloat;
112 }
113 return r;
114 case ArithOp::Sub:
115 if (r.isIntResult) {
116 if (OverflowCheck::subOverflow(leftInt, rightInt)) {
117 r.status = ArithStatus::IntOverflow;
118 return r;
119 }
120 r.intVal = leftInt - rightInt;
121 } else {
122 r.floatVal = leftFloat - rightFloat;
123 }
124 return r;
125 case ArithOp::Mul:
126 if (r.isIntResult) {
127 if (OverflowCheck::mulOverflow(leftInt, rightInt)) {
128 r.status = ArithStatus::IntOverflow;
129 return r;
130 }
131 r.intVal = leftInt * rightInt;
132 } else {
133 r.floatVal = leftFloat * rightFloat;
134 }
135 return r;
136 case ArithOp::Div:
137 if (r.isIntResult) {
138 if (rightInt == 0) {
139 r.status = ArithStatus::DivByZero;
140 return r;
141 }
142 if (OverflowCheck::divOverflow(leftInt, rightInt)) {
143 r.status = ArithStatus::IntOverflow;
144 return r;
145 }
146 r.intVal = leftInt / rightInt; // int/int → int (截断除法)
147 } else {
148 if (rightFloat == 0.0) {
149 r.status = ArithStatus::DivByZero;
150 return r;
151 }
152 r.floatVal = leftFloat / rightFloat;
153 }
154 return r;
155 case ArithOp::Mod:
156 if (r.isIntResult) {
157 if (rightInt == 0) {
158 r.status = ArithStatus::DivByZero;
159 return r;
160 }
161 if (OverflowCheck::modOverflow(leftInt, rightInt)) {
162 r.status = ArithStatus::IntOverflow;
163 return r;
164 }
165 r.intVal = leftInt % rightInt;
166 } else {
167 if (rightFloat == 0.0) {
168 r.status = ArithStatus::DivByZero;
169 return r;
170 }
171 r.floatVal = std::fmod(leftFloat, rightFloat);
172 }
173 return r;
174 }
175 r.status = ArithStatus::NotNumeric;
176 return r;
177}

196/// 执行有序比较（字符串字典序或数值比较）
197/// 输入：leftIsString/rightIsString 标识类型，leftNum/rightNum 为数值（当非字符串时）
198inline CompareResult computeCompare(CompareOp op, bool leftIsString, const std::string& leftStr, bool rightIsString,
199 const std::string& rightStr, double leftNum, double rightNum) {
200 CompareResult r;
201 // 字符串字典序比较（两侧均为字符串）
202 if (leftIsString && rightIsString) {
203 switch (op) {
204 case CompareOp::Less:
205 r.value = leftStr < rightStr;
206 break;
207 case CompareOp::Greater:
208 r.value = leftStr > rightStr;
209 break;
210 case CompareOp::LessEqual:
211 r.value = leftStr <= rightStr;
212 break;
213 case CompareOp::GreaterEqual:
214 r.value = leftStr >= rightStr;
215 break;
216 }
217 return r;
218 }
219 // 数值比较（调用方已确认两侧均为数值）
220 switch (op) {
221 case CompareOp::Less:
222 r.value = leftNum < rightNum;
223 break;
224 case CompareOp::Greater:
225 r.value = leftNum > rightNum;
226 break;
227 case CompareOp::LessEqual:
228 r.value = leftNum <= rightNum;
229 break;
230 case CompareOp::GreaterEqual:
231 r.value = leftNum >= rightNum;
232 break;
233 }
234 return r;
235}
