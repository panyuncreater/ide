38/// 返回 UTF-8 首字节对应的码位字节数（1-4），非法首字节返回 1（跳过）
39inline int byteLength(unsigned char c) {
40 return (c < 0x80) ? 1 : ((c & 0xE0) == 0xC0) ? 2 : ((c & 0xF0) == 0xE0) ? 3 : ((c & 0xF8) == 0xF0) ? 4 : 1;
41}

43/// 统计字符串中的 UTF-8 码位数
44inline int64_t codepointCount(const std::string& s) {
45 int64_t count = 0;
46 for (size_t i = 0; i < s.size();) {
47 i += byteLength(static_cast<unsigned char>(s[i]));
48 count++;
49 }
50 return count;
51}

53/// PERF: 带 ASCII 提示的重载。当调用方已通过 StringData::cachedIsAscii 确认为纯 ASCII 时，
54/// 直接返回字节长度（纯 ASCII 每字节 = 1 码位），从 O(n) 降为 O(1)。
55inline int64_t codepointCount(const std::string& s, bool knownAscii) {
56 if (knownAscii)
57 return static_cast<int64_t>(s.size());
58 return codepointCount(s);
59}

61/// 将码位索引转换为字节索引。cpIdx 超出范围时返回 s.size()。
62inline size_t codepointToByteIndex(const std::string& s, int64_t cpIdx) {
63 size_t bytePos = 0;
64 int64_t cp = 0;
65 while (bytePos < s.size() && cp < cpIdx) {
66 bytePos += byteLength(static_cast<unsigned char>(s[bytePos]));
67 cp++;
68 }
69 // AUDIT-BUG-C4 fix: 钳制到 s.size()——byteLength 对不完整 UTF-8 序列（如 4 字节
70 // leader 仅剩 1-3 字节）返回 4，可将 bytePos 推进到 s.size() 之外。
71 // 调用方（如 substr）若直接用作索引会越界抛 std::out_of_range。
72 if (bytePos > s.size())
73 bytePos = s.size();
74 return bytePos;
75}

77/// 将字节索引转换为码位索引（计算前 byteIdx 字节包含多少码位）。
78/// AUDIT-R3 P2 fix: 循环同时约束 b < s.size()，与 codepointToByteIndex 的
79/// AUDIT-BUG-C4 钳制对齐——调用方传入超过字符串长度的 byteIdx 时，
80/// 原实现以越界下标读 s（UB）。公共工具头需防御性边界。
81inline int64_t byteToCodepointIndex(const std::string& s, size_t byteIdx) {
82 int64_t charIdx = 0;
83 for (size_t b = 0; b < byteIdx && b < s.size();) {
84 b += byteLength(static_cast<unsigned char>(s[b]));
85 charIdx++;
86 }
87 return charIdx;
88}
