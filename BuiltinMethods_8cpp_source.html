198Result<Value> executeSharedStrSubstr(const Value& str, const Value* args, size_t argCount, int line, int column) {
199 if (argCount < 1 || argCount > 2) {
200 return Result<Value>::err(
201 ErrorFormat::formatStd("substr 期望 1-2 个参数(起始[, 长度])，但传入了 {} 个", argCount), line, column);
202 }
203 if (!args[0].isInt()) {
204 return Result<Value>::err("substr 起始位置必须是整数", line, column);
205 }
206 int64_t start = args[0].intVal();
207 const std::string& s = str.stringVal();
208
209 // BUG 1.1 fix: 将码位索引转换为字节索引，与 len()/indexOf() 的码位语义一致
210 // P0-4 fix: 使用 Utf8 工具函数替代重复的内联 lambda

212 int64_t totalCp = str.codepointCount(); // perf1 fix: 带缓存的码位计数
213 // AUDIT-BUG-F11 fix: 负 start 报错（与 len<0 一致），而非静默返回空串。
214 // 原 start<0 静默返回 ""，len<0 报错，错误处理不对称。用户无法区分"空串"与"负索引错误"。

216 return Result<Value>::err("substr 起始位置不能为负数", line, column);
217 }
218 if (start > totalCp) {
219 return Result<Value>::ok(Value(std::string("")));
220 }
221
222 size_t byteStart = Utf8::codepointToByteIndex(s, start);
223
224 if (argCount == 2) {
225 if (!args[1].isInt()) {
226 return Result<Value>::err("substr 长度必须是整数", line, column);
227 }
228 int64_t len = args[1].intVal();
229 if (len < 0) {
230 return Result<Value>::err("substr 长度不能为负数", line, column);
231 }
232 // #27 fix: 原 codepointToByteIndex(s, start + len) 从头重新扫描到 start+len，

234 size_t byteEnd = byteStart;
235 int64_t remaining = len;

238 --remaining;
239 }
240 if (byteEnd > s.size())
241 byteEnd = s.size();
242 return Result<Value>::ok(Value(s.substr(byteStart, byteEnd - byteStart)));
243 } else {
244 return Result<Value>::ok(Value(s.substr(byteStart)));

263// S1 fix: 共享 replace 实现 — 单遍构建 O(N)，消除 VM 侧 O(N²) Bug
264Result<Value> executeSharedStrReplace(const Value& str, const Value* args, size_t argCount, int line, int column) {
265 if (argCount != 2) {
266 return Result<Value>::err(ErrorFormat::formatStd("replace 期望 2 个参数(旧串, 新串)，但传入了 {} 个", argCount),
267 line, column);
268 }
269 const std::string& src = str.stringVal();
270 std::string from = args[0].toString();
271 std::string to = args[1].toString();
272 if (from.empty()) {
273 return Result<Value>::ok(Value(std::string(src)));
274 }
275
276 // 预扫统计匹配数以估算容量
277 size_t matchCount = 0;
278 size_t scanPos = 0;
279 while ((scanPos = src.find(from, scanPos)) != std::string::npos) {
280 ++matchCount;
281 scanPos += from.size();
282 }
283
284 // 单遍构建结果字符串
285 std::string result;
286 size_t estimated = src.size();
287 if (to.size() > from.size()) {
288 estimated += matchCount * (to.size() - from.size());
289 }
290 result.reserve(estimated);
291
292 size_t lastEnd = 0;
293 size_t pos = 0;
294 while ((pos = src.find(from, pos)) != std::string::npos) {
295 result.append(src, lastEnd, pos - lastEnd);
296 result.append(to);
297 pos += from.size();
298 lastEnd = pos;
299 }
300 result.append(src, lastEnd, std::string::npos);
301 return Result<Value>::ok(Value(std::move(result)));
302}

331Result<Value> executeSharedStrSplit(const Value& str, const Value* args, size_t argCount, int line, int column) {
332 if (auto r = checkRange("split", argCount, 0, 1, line, column); r.is_err())
333 return r;
334 const std::string& s = str.stringVal();
335 std::string sep = (argCount == 1) ? args[0].toString() : " ";
336 if (sep.empty()) {
337 return Result<Value>::err("split 的分隔符不能为空字符串", line, column);
338 }
339
340 // 预扫描统计分隔符数量以预估容量
341 size_t estCount = 1;
342 for (size_t pos = s.find(sep); pos != std::string::npos; pos = s.find(sep, pos + sep.size())) {
343 ++estCount;
344 }
345
346 std::vector<Value> parts;
347 parts.reserve(estCount);
348 size_t start = 0;
349 size_t pos = s.find(sep);
350 while (pos != std::string::npos) {
351 parts.emplace_back(Value(s.substr(start, pos - start)));
352 start = pos + sep.size();
353 pos = s.find(sep, start);
354 }
355 parts.emplace_back(Value(s.substr(start)));
356 return Result<Value>::ok(Value(std::move(parts)));
357}

516/// int(x): 将 x 转换为整数（int64）。
517/// - int 原样返回；float 向零截断（溢出检查）；bool 转 0/1；
518/// - 字符串先用 from_chars 解析（避免 locale 依赖），要求整串可解析（前后空白允许、中间不允许），否则 err；
519/// - 其他类型不支持，返回 err。溢出与格式错误均以 err 返回而非 UB。
520Result<Value> executeBuiltinInt(const Value* args, size_t argCount, int line, int column) {
521 if (auto r = checkExact("int", argCount, 1, line, column); r.is_err())
522 return r;
523 const Value& v = args[0];
524 if (v.isInt()) {
525 return Result<Value>::ok(v);
526 } else if (v.isFloat()) {
527 // BUG 9.3 fix: 浮点转整数溢出检查
528 double dv = v.floatVal();
529 if (OverflowCheck::doubleToIntOverflow(dv)) {
530 return Result<Value>::err(ErrorFormat::formatStd("int 转换溢出: {} 超出 int64_t 范围", dv), line, column);
531 }
532 // 截断小数部分（向零取整，与 C++ static_cast 一致）
533 return Result<Value>::ok(Value(static_cast<int64_t>(dv)));
534 } else if (v.isBool()) {
535 return Result<Value>::ok(Value(static_cast<int64_t>(v.boolVal() ? 1 : 0)));
536 } else if (v.isString()) {
537 // L-P2 fix: 使用 from_chars 替代 stoll，避免 locale 依赖
538 const std::string& s = v.stringVal();
539 size_t startIdx = 0;
540 while (startIdx < s.size() && std::isspace(static_cast<unsigned char>(s[startIdx])))
541 startIdx++;
542 if (startIdx >= s.size()) {
543 return Result<Value>::err("int 无法将空字符串转换为整数", line, column);
544 }
545 long long parsed = 0;
546 auto [ptr, ec] = std::from_chars(s.data() + startIdx, s.data() + s.size(), parsed);
547 if (ec != std::errc()) {
548 return Result<Value>::err("int 无法将字符串 \"" + s + "\" 转换为整数", line, column);
549 }
550 size_t consumed = static_cast<size_t>(ptr - s.data());
551 while (consumed < s.size() && std::isspace(static_cast<unsigned char>(s[consumed])))
552 consumed++;
553 if (consumed != s.size()) {
554 return Result<Value>::err("int 无法将字符串 \"" + s + "\" 转换为整数", line, column);
555 }
556 return Result<Value>::ok(Value(static_cast<int64_t>(parsed)));
557 } else {
558 return Result<Value>::err("int 不支持类型 " + v.typeName(), line, column);
559 }

585// ---- min(a, b) / min(arr): 最小值 ----
586/// min(a, b): 返回两个数值中的较小者。a、b 必须同为数值类型，否则 err。
587/// 同为 int 时做整数比较（无精度损失）；否则都转 double 后比较，返回 double。
588/// 拓展二期（SIMD 用户可见内建）：新增 min(arr) 单参数组形态——
589/// 全 int 数组且元素数 >= SIMD_MIN_ELEMENTS 且 CPU 支持 AVX2 时走
590/// simdMinInt64 内核（4 路并行 cmpgt+blend）；否则标量回退。
591/// 空数组 err（无最小值可言）。与 sum(arr) 的 SIMD 策略/回退链一致，
592/// 共享实现层保证三后端语义一致。
593Result<Value> executeBuiltinMin(const Value* args, size_t argCount, int line, int column) {
594 // 拓展二期：单参数组形态 min(arr)
595 if (argCount == 1 && args[0].isArray()) {
596 const auto& arr = args[0].arrayVal();
597 if (arr.empty()) {
598 return Result<Value>::err("min 空数组无最小值", line, column);
599 }
600 bool allInt = true;
601 for (const auto& elem : arr) {
602 if (!elem.isInt()) {
603 allInt = false;
604 break;
605 }
606 }
607 if (allInt) {
608 // SIMD 快速路径（与 sum(arr) 同策略：先批量解码到连续 int64 缓冲区）
609 if (arr.size() >= minilang::simd::SIMD_MIN_ELEMENTS && minilang::simd::hasAvx2Support()) {
610 std::vector<int64_t> buffer(arr.size());
611 for (size_t i = 0; i < arr.size(); ++i) {
612 buffer[i] = arr[i].intVal();
613 }
614 return Result<Value>::ok(Value(minilang::simd::simdMinInt64(buffer.data(), buffer.size())));
615 }
616 int64_t m = arr[0].intVal();
617 for (size_t i = 1; i < arr.size(); ++i) {
618 if (arr[i].intVal() < m)
619 m = arr[i].intVal();
620 }
621 return Result<Value>::ok(Value(m));
622 }
623 // 混合数值：转 double 比较（遇非数值 err）
624 // 七特性 MVP 阶段 5：先校验全为数值并批量解码到连续 double 缓冲，
625 // 达阈且支持 AVX2 时走 simdMinDouble 内核（与 sum(arr) 的 double SIMD 同策略）。
626 for (size_t i = 0; i < arr.size(); ++i) {
627 if (!arr[i].isNumber()) {
628 return Result<Value>::err("min 数组含非数值元素: " + arr[i].typeName(), line, column);
629 }
630 }
631 if (arr.size() >= minilang::simd::SIMD_MIN_ELEMENTS && minilang::simd::hasAvx2Support()) {
632 std::vector<double> buffer(arr.size());
633 for (size_t i = 0; i < arr.size(); ++i) {
634 buffer[i] = arr[i].toDouble();
635 }
636 return Result<Value>::ok(Value(minilang::simd::simdMinDouble(buffer.data(), buffer.size())));
637 }
638 double best = arr[0].toDouble();
639 for (size_t i = 1; i < arr.size(); ++i) {
640 double d = arr[i].toDouble();
641 if (d < best)
642 best = d;
643 }
644 return Result<Value>::ok(Value(best));
645 }
646 if (auto r = checkExact("min", argCount, 2, line, column); r.is_err())
647 return r;
648 const Value& a = args[0];
649 const Value& b = args[1];
650 if (!a.isNumber() || !b.isNumber()) {
651 return Result<Value>::err("min 期望数值参数", line, column);
652 }
653 if (a.isInt() && b.isInt()) {
654 return Result<Value>::ok((a.intVal() <= b.intVal()) ? a : b);
655 } else {
656 double da = a.toDouble(), db = b.toDouble();
657 return Result<Value>::ok(Value(da <= db ? da : db));
658 }
659}

759// ---- sum(arr): 数组元素求和 ----
760/// sum(arr): 对数组元素求和。
761/// 先判断数组是否全为 int：全 int 时做整数累加（addOverflow 溢出检查，溢出 err）；
762/// 否则转 double 累加，遇到非数值元素返回 err。返回 int 或 double。
763///
764/// R134 SIMD 快速路径：当数组元素数 >= SIMD_MIN_ELEMENTS (16) 且 CPU 支持 AVX2 时，
765/// 跳过逐元素 OverflowCheck 分支，改为分块累加（每块 256 元素，块内不可能溢出
766/// 因为 256 * (2^47-1) ≈ 2^55 < 2^63），仅块间做溢出检查。块内使用 AVX2 4 路并行
767/// 加法，吞吐量约 4x。整型路径还会用 decodeInt48Batch 批量解码 NaN-boxed int48
768/// 到连续 int64_t[]，避免每元素 isInt() + 符号扩展分支。
769Result<Value> executeBuiltinSum(const Value* args, size_t argCount, int line, int column) {
770 if (auto r = checkExact("sum", argCount, 1, line, column); r.is_err())
771 return r;
772 const Value& v = args[0];
773 if (!v.isArray()) {
774 return Result<Value>::err("sum 期望数组参数，实际为 " + v.typeName(), line, column);
775 }
776 const auto& arr = v.arrayVal();
777 bool allInt = true;
778 for (const auto& elem : arr) {
779 if (!elem.isInt()) {
780 allInt = false;
781 break;
782 }
783 }
784 if (allInt) {
785 // R134 SIMD 快速路径：分块累加 + AVX2 4 路并行
786 if (arr.size() >= minilang::simd::SIMD_MIN_ELEMENTS && minilang::simd::hasAvx2Support()) {
787 // 分块大小：256 * (2^47-1) ≈ 2^55 < 2^63，单块累加不可能溢出 int64
788 constexpr size_t CHUNK_SIZE = 256;
789 // 解码缓冲区：用 memcpy 提取 NaN-boxed bits，再批量解码
790 std::vector<int64_t> buffer(arr.size());
791 for (size_t i = 0; i < arr.size(); ++i) {
792 buffer[i] = arr[i].intVal();
793 }
794 int64_t total = 0;
795 for (size_t i = 0; i < arr.size(); i += CHUNK_SIZE) {
796 size_t chunkLen = std::min(CHUNK_SIZE, arr.size() - i);
797 int64_t chunkSum = minilang::simd::simdSumInt64(buffer.data() + i, chunkLen);
798 if (OverflowCheck::addOverflow(total, chunkSum)) {
799 return Result<Value>::err("sum 整数累加溢出", line, column);
800 }
801 total += chunkSum;
802 }
803 return Result<Value>::ok(Value(total));
804 }
805 // 标量回退路径：逐元素累加 + 溢出检查
806 int64_t total = 0;
807 for (const auto& elem : arr) {
808 if (OverflowCheck::addOverflow(total, elem.intVal())) {
809 return Result<Value>::err("sum 整数累加溢出", line, column);
810 }
811 total += elem.intVal();
812 }
813 return Result<Value>::ok(Value(total));
814 } else {
815 // R134 SIMD 快速路径：double 累加（无需溢出检查）
816 if (arr.size() >= minilang::simd::SIMD_MIN_ELEMENTS && minilang::simd::hasAvx2Support()) {
817 bool allNumber = true;
818 std::vector<double> buffer(arr.size());
819 for (size_t i = 0; i < arr.size(); ++i) {
820 if (!arr[i].isNumber()) {
821 allNumber = false;
822 break;
823 }
824 buffer[i] = arr[i].toDouble();
825 }
826 if (allNumber) {
827 double total = minilang::simd::simdSumDouble(buffer.data(), buffer.size());
828 return Result<Value>::ok(Value(total));
829 }
830 // 混合类型：回退到标量路径报告错误
831 }
832 double total = 0.0;
833 for (const auto& elem : arr) {
834 if (elem.isNumber()) {
835 total += elem.toDouble();
836 } else {
837 return Result<Value>::err("sum 数组元素包含非数值类型: " + elem.typeName(), line, column);
838 }
839 }
840 return Result<Value>::ok(Value(total));
841 }
842}

866// ---- __qmark_unwrap(r): ? 传播运算符的运行时脱糖目标（拓展二期·语言）----
867/// Parser 将 expr? 脱糖为 __qmark_unwrap(expr)（见 Parser.cpp call()）。
868/// 语义（配套 std/result 的 dict 封装协议 {"tag": ...}）：
869/// - tag=="ok"/"some" → 返回 value 字段（无 value 时返回 null）
870/// - tag=="err" → 运行时错误“? 传播 Err: <error>”（可被 try/catch 捕获）
871/// - tag=="none" → 运行时错误“? 传播 None”
872/// - 非 Result/Option dict → 运行时错误（类型约束）
873/// 七特性 MVP 阶段 1 扩展：同时支持泛型 enum 写法的 Result/Option（如
874/// `enum Result<T, E> { Ok(T), Err(E) }`）——enum variant 值按 variantName 分派：
875/// - Ok/Some → 解包 fields[0]（无 payload 时返回 null）
876/// - Err → 运行时错误“? 传播 Err: <payload>”；None → “? 传播 None”
877/// - 其他 variant 名 → 类型约束错误（与 dict 路径同样文案风格）
878/// 共享实现层单一事实源，三后端传播语义自动一致。
879Result<Value> executeBuiltinQmarkUnwrap(const Value* args, size_t argCount, int line, int column) {
880 if (auto r = checkExact("?", argCount, 1, line, column); r.is_err())
881 return r;
882 const Value& v = args[0];
883 // 泛型 enum variant 路径：按 variantName 分派（enum 名不限定，用户自定义
884 // `enum MyResult { Ok(T), Err(E) }` 同样适用，与 match 的 variantName 比较哲学一致）
885 if (v.isEnumVariant()) {
886 const std::string& vn = v.enumVariantName();
887 const std::vector<Value>& fields = v.enumVariantFields();
888 if (vn == "Ok" || vn == "Some") {
889 return Result<Value>::ok(fields.empty() ? Value::nullValue() : fields[0]);
890 }
891 if (vn == "Err") {
892 std::string msg = fields.empty() ? "unknown" : fields[0].toString();
893 return Result<Value>::err("? 传播 Err: " + msg, line, column);
894 }
895 if (vn == "None") {
896 return Result<Value>::err("? 传播 None", line, column);
897 }
898 return Result<Value>::err("? 运算符要求 Result/Option 值（Ok/Err/Some/None variant），实际为 " +
899 v.enumVariantEnumName() + "." + vn,
900 line, column);
901 }
902 if (!v.isDict()) {
903 return Result<Value>::err(
904 "? 运算符要求 Result/Option 值（std/result 的 Ok/Err/Some/None），实际为 " + v.typeName(), line, column);
905 }
906 const auto& entries = v.dictVal();
907 // PERF: 字符串字面量键走 string_view 透明查找，免构造临时 Value + DictKey
908 auto tagIt = entries.find(std::string_view("tag"));
909 if (tagIt == entries.end() || !tagIt->second.isString()) {
910 return Result<Value>::err("? 运算符要求 Result/Option 值（缺少 tag 字段）", line, column);
911 }
912 const std::string& tag = tagIt->second.stringVal();
913 if (tag == "ok" || tag == "some") {
914 auto vIt = entries.find(std::string_view("value"));
915 return Result<Value>::ok(vIt != entries.end() ? vIt->second : Value::nullValue());
916 }
917 if (tag == "err") {
918 auto eIt = entries.find(std::string_view("error"));
919 std::string msg = (eIt != entries.end()) ? eIt->second.toString() : "unknown";
920 return Result<Value>::err("? 传播 Err: " + msg, line, column);
921 }
922 if (tag == "none") {
923 return Result<Value>::err("? 传播 None", line, column);
924 }
925 return Result<Value>::err("? 运算符遇到未知 tag: " + tag, line, column);
926}

988 }

1166BuiltinMethodResult handleSyncObjectMethod(const std::string& method, Value& obj, const std::vector<Value>& args,
1167 int line, int col) {
1168 // ---- Channel 方法 ----
1169 if (obj.isChannel()) {
1170 auto inner = obj.channelInner();
1171 if (method == "send") {
1172 if (args.size() != 1) {
1173 throw RuntimeError("channel.send 期望 1 个参数，但传入了 " + std::to_string(args.size()) + " 个", line,
1174 col, DiagCodes::kArityMismatch);
1175 }
1176 std::unique_lock<std::mutex> lock(inner->mu);
1177 if (inner->closed) {
1178 throw RuntimeError("channel.send: 通道已关闭", line, col);
1179 }
1180 inner->queue.push(args[0]);
1181 inner->cv.notify_one();
1182 return BuiltinMethodResult(Value::nullValue(), false);
1183 }
1184 if (method == "recv") {
1185 if (!args.empty()) {
1186 throw RuntimeError("channel.recv 期望 0 个参数，但传入了 " + std::to_string(args.size()) + " 个", line,
1187 col, DiagCodes::kArityMismatch);
1188 }
1189 std::unique_lock<std::mutex> lock(inner->mu);
1190 inner->cv.wait(lock, [&] { return !inner->queue.empty() || inner->closed; });
1191 if (inner->queue.empty()) {
1192 // 通道已关闭且无消息
1193 return BuiltinMethodResult(Value::nullValue(), false);
1194 }
1195 Value v = std::move(inner->queue.front());
1196 inner->queue.pop();
1197 return BuiltinMethodResult(std::move(v), false);
1198 }
1199 if (method == "tryRecv") {
1200 if (!args.empty()) {
1201 throw RuntimeError("channel.tryRecv 期望 0 个参数", line, col, DiagCodes::kArityMismatch);
1202 }
1203 std::unique_lock<std::mutex> lock(inner->mu);
1204 if (inner->queue.empty()) {
1205 return BuiltinMethodResult(Value::nullValue(), false);
1206 }
1207 Value v = std::move(inner->queue.front());
1208 inner->queue.pop();
1209 return BuiltinMethodResult(std::move(v), false);
1210 }
1211 // channel.recvTimeout(ms)：等待至多 ms 毫秒；有消息返回消息，
1212 // 超时或通道已关闭且无消息返回 null（与 recv 的关闭语义对齐）。
1213 // 四后端经本共享函数分发，语义天然一致。
1214 if (method == "recvTimeout") {
1215 if (args.size() != 1) {
1216 throw RuntimeError("channel.recvTimeout 期望 1 个参数，但传入了 " + std::to_string(args.size()) + " 个",
1217 line, col, DiagCodes::kArityMismatch);
1218 }
1219 if (!args[0].isInt()) {
1220 throw RuntimeError("channel.recvTimeout 参数必须是整数（毫秒）", line, col);
1221 }
1222 int64_t ms = args[0].intVal();
1223 if (ms < 0) {
1224 throw RuntimeError("channel.recvTimeout 超时不能为负数", line, col);
1225 }
1226 // DoS 防护：spawn 为延迟执行模式，阻塞期间不会有其他线程 send，
1227 // 超长超时等价于卡死 worker，上限见 RuntimeLimits。
1228 if (ms > RuntimeLimits::MAX_CHANNEL_TIMEOUT_MS) {
1229 throw RuntimeError("channel.recvTimeout 超时上限为 " +
1230 std::to_string(RuntimeLimits::MAX_CHANNEL_TIMEOUT_MS) + " 毫秒",
1231 line, col);
1232 }
1233 std::unique_lock<std::mutex> lock(inner->mu);
1234 inner->cv.wait_for(lock, std::chrono::milliseconds(ms),
1235 [&] { return !inner->queue.empty() || inner->closed; });
1236 if (inner->queue.empty()) {
1237 // 超时或已关闭且无消息
1238 return BuiltinMethodResult(Value::nullValue(), false);
1239 }
1240 Value v = std::move(inner->queue.front());
1241 inner->queue.pop();
1242 return BuiltinMethodResult(std::move(v), false);
1243 }
1244 if (method == "close") {
1245 if (!args.empty()) {
1246 throw RuntimeError("channel.close 期望 0 个参数", line, col, DiagCodes::kArityMismatch);
1247 }
1248 std::unique_lock<std::mutex> lock(inner->mu);
1249 inner->closed = true;
1250 inner->cv.notify_all(); // 唤醒所有等待的 recv
1251 return BuiltinMethodResult(Value::nullValue(), false);
1252 }
1253 throw RuntimeError("channel 不支持方法 " + method, line, col, "undefined-function");
1254 }
1255
1256 // ---- Mutex 方法 ----
1257 if (obj.isMutex()) {
1258 auto inner = obj.mutexInner();
1259 auto owner = obj.mutexOwner();
1260 const std::thread::id self = std::this_thread::get_id();
1261 if (method == "lock") {
1262 // 自死锁检测：同一线程已持有该锁时再次 lock 会永久死锁（std::mutex 非递归），
1263 // fail-fast 抛异常而非挂起，便于教学定位。跨线程 lock 仍走 std::mutex 正常阻塞。
1264 if (owner->load() == self) {
1265 throw RuntimeError("mutex 死锁检测：当前线程已持有该锁，重复 lock 将死锁（请改用 tryLock 或先 unlock）",
1266 line, col, "deadlock-detected");
1267 }
1268 inner->lock();
1269 owner->store(self);
1270 return BuiltinMethodResult(Value::nullValue(), false);
1271 }
1272 if (method == "unlock") {
1273 owner->store(std::thread::id{});
1274 inner->unlock();
1275 return BuiltinMethodResult(Value::nullValue(), false);
1276 }
1277 if (method == "tryLock") {
1278 // 已持有时直接返回 false（避免 std::mutex 同线程 try_lock 的未定义行为），
1279 // 与既有 MutexDoubleLockGuardedByTryLock 语义一致。
1280 if (owner->load() == self) {
1281 return BuiltinMethodResult(Value(false), false);
1282 }
1283 bool acquired = inner->try_lock();
1284 if (acquired) {
1285 owner->store(self);
1286 }
1287 return BuiltinMethodResult(Value(acquired), false);
1288 }
1289 throw RuntimeError("mutex 不支持方法 " + method, line, col, "undefined-function");
1290 }
1291
1292 // ---- RwLock 方法 ----
1293 if (obj.isRwLock()) {
1294 auto inner = obj.rwlockInner();
1295 if (method == "readLock") {
1296 inner->lock_shared();
1297 return BuiltinMethodResult(Value::nullValue(), false);
1298 }
1299 if (method == "readUnlock") {
1300 inner->unlock_shared();
1301 return BuiltinMethodResult(Value::nullValue(), false);
1302 }
1303 if (method == "writeLock") {
1304 inner->lock();
1305 return BuiltinMethodResult(Value::nullValue(), false);
1306 }
1307 if (method == "writeUnlock") {
1308 inner->unlock();
1309 return BuiltinMethodResult(Value::nullValue(), false);
1310 }
1311 if (method == "tryReadLock") {
1312 bool acquired = inner->try_lock_shared();
1313 return BuiltinMethodResult(Value(acquired), false);
1314 }
1315 if (method == "tryWriteLock") {
1316 bool acquired = inner->try_lock();
1317 return BuiltinMethodResult(Value(acquired), false);
1318 }
1319 throw RuntimeError("rwlock 不支持方法 " + method, line, col, "undefined-function");
1320 }

1322 // ---- Thread 方法 ----
1323 // R136 fix: 延迟执行模式——join() 时在主线程同步执行 pending 闭包
1324 if (obj.isThread()) {
1325 auto inner = obj.threadInner();
1326 if (method == "join") {
1327 std::unique_lock<std::mutex> lock(inner->stateMu);
1328 if (inner->joined) {
1329 throw RuntimeError("thread.join: 线程已 join", line, col);
1330 }
1331 if (inner->detached) {
1332 throw RuntimeError("thread.join: 线程已 detach，无法 join", line, col);
1333 }
1334 // R136 fix: 延迟执行——主线程同步调用 invoker 执行 pending 闭包
1335 if (inner->hasPending) {
1336 auto invoker = std::static_pointer_cast<ClosureInvoker>(inner->pendingInvoker);
1337 Value closure = inner->pendingClosure;
1338 std::vector<Value> argsCopy = inner->pendingArgs; // 拷贝避免持锁访问
1339 int spLine = inner->spawnLine;
1340 int spCol = inner->spawnCol;
1341 lock.unlock(); // 释放锁，invoker 内部可能调用 VM（不再有并发）
1342 auto r = (*invoker)(closure, argsCopy.data(), argsCopy.size(), spLine, spCol);
1343 lock.lock();
1344 inner->joined = true;
1345 inner->hasPending = false;
1346 inner->pendingClosure = Value::nullValue();
1347 inner->pendingArgs.clear();
1348 inner->pendingInvoker.reset();
1349 if (r.is_err()) {
1350 throw RuntimeError(r.error().message);
1351 }
1352 } else if (inner->thr.joinable()) {
1353 // 兼容路径：若 thr 实际有线程（未来扩展），正常 join
1354 lock.unlock();
1355 inner->thr.join();
1356 lock.lock();
1357 inner->joined = true;
1358 }
1359 return BuiltinMethodResult(Value::nullValue(), false);
1360 }
1361 if (method == "detach") {
1362 std::unique_lock<std::mutex> lock(inner->stateMu);
1363 if (inner->joined) {
1364 throw RuntimeError("thread.detach: 线程已 join", line, col);
1365 }
1366 if (inner->detached) {
1367 throw RuntimeError("thread.detach: 线程已 detach", line, col);
1368 }
1369 // R136 fix: 延迟执行模式下 detach 丢弃 pending 闭包
1370 if (inner->hasPending) {
1371 inner->hasPending = false;
1372 inner->pendingClosure = Value::nullValue();
1373 inner->pendingArgs.clear();
1374 inner->pendingInvoker.reset();
1375 inner->detached = true;
1376 } else if (inner->thr.joinable()) {
1377 inner->thr.detach();
1378 inner->detached = true;
1379 }
1380 return BuiltinMethodResult(Value::nullValue(), false);
1381 }
1382 if (method == "isJoinable") {
1383 std::unique_lock<std::mutex> lock(inner->stateMu);
1384 // R136 fix: hasPending 或 thr.joinable() 都算 joinable
1385 bool joinable = (inner->hasPending || (inner->thr.joinable())) && !inner->joined && !inner->detached;
1386 return BuiltinMethodResult(Value(joinable), false);
1387 }
1388 throw RuntimeError("thread 不支持方法 " + method, line, col, "undefined-function");
1389 }
1390
1391 throw RuntimeError("类型 " + obj.typeName() + " 不支持同步对象方法 " + method, line, col, "undefined-function");
1392}

1487/// 分发数组内置方法。
1488/// 设计：变异方法（push/pop/remove）直接就地修改 obj，返回 objectModified=true
1489/// 以便调用方（Interpreter/VM）知道原对象已变更、无需再取返回值；非变异方法
1490/// （len/contains/join）通过 arraySharedMethods() 注册表委托共享纯函数实现，
1491/// 统一经 dispatchShared 将 Result<Value> 转为 BuiltinMethodResult 或抛 RuntimeError。
1492/// 未知方法名抛出 RuntimeError。方法名由 classifyBuiltinMethod 预先分类。
1493BuiltinMethodResult BuiltinMethods::handleArrayMethod(const std::string& method, Value& obj,
1494 const std::vector<Value>& args, int line, int col) {
1495 // S6 fix: 变异方法直接处理
1496 if (method == "push") {
1497 if (args.size() != 1)
1498 throw RuntimeError("push 期望 1 个参数", line, col, DiagCodes::kArityMismatch);
1499 obj.arrayVal().push_back(args[0]);
1500 return BuiltinMethodResult(Value::nullValue(), /*objectModified=*/true);
1501 }
1502
1503 if (method == "pop") {
1504 if (!args.empty())
1505 throw RuntimeError(ErrorFormat::formatStd("pop 期望 0 个参数，但传入了 {} 个", args.size()), line, col,
1506 DiagCodes::kArityMismatch);
1507 // AUDIT-BUG-C5 fix: 空数组检查用 const 访问器避免 COW detach。
1508 // 非 const arrayVal() 在 refCount>1 时会 ensureUnique 深拷贝整个数组，
1509 // 此处仅为检查 empty() 却触发不必要的克隆。
1510 if (std::as_const(obj).arrayVal().empty())
1511 throw RuntimeError("对空数组调用 pop", line, col);
1512 Value last = std::as_const(obj).arrayVal().back();
1513 obj.arrayVal().pop_back();
1514 return BuiltinMethodResult(std::move(last), /*objectModified=*/true);
1515 }
1516
1517 if (method == "remove") {
1518 if (args.size() != 1)
1519 throw RuntimeError("remove 期望 1 个参数(索引)", line, col, DiagCodes::kArityMismatch);
1520 if (!args[0].isInt())
1521 throw RuntimeError("remove 参数必须是整数索引", line, col, DiagCodes::kTypeMismatch);
1522 int64_t idx = args[0].intVal();
1523 if (idx < 0 || static_cast<size_t>(idx) >= std::as_const(obj).arrayVal().size())
1524 throw RuntimeError(ErrorFormat::formatStd("数组索引越界: {}, 有效范围 [0, {})", static_cast<long long>(idx),
1525
1526 std::as_const(obj).arrayVal().size()),
1527 line, col, DiagCodes::kIndexOutOfBounds);
1528 obj.arrayVal().erase(obj.arrayVal().begin() + static_cast<size_t>(idx));
1529 return BuiltinMethodResult(Value::nullValue(), /*objectModified=*/true);
1530 }
1531
1532 // S6 fix: 非变异方法通过注册表分发
1533 auto& registry = arraySharedMethods();
1534 auto it = registry.find(method);
1535 if (it != registry.end()) {
1536 return dispatchShared(it->second, obj, args.empty() ? nullptr : args.data(), args.size(), line, col);
1537 }
1538
1539 throw RuntimeError("数组没有方法 " + method, line, col, "undefined-function");
1540}

1550BuiltinMethodResult BuiltinMethods::handleDictMethod(const std::string& method, Value& obj,
1551 const std::vector<Value>& args, int line, int col) {

1554 if (args.size() != 1)
1555 throw RuntimeError("remove 期望 1 个参数(键)", line, col, DiagCodes::kArityMismatch);

1563 if (it != entries.end())
1564 entries.erase(it);
1565 } else {
1566 entries.erase(Value::dictKeyFromValue(args[0]).value());
1567 }
1568 return BuiltinMethodResult(Value::nullValue(), /*objectModified=*/true);
1569 }
1570 if (method == "set") {

1574 // PERF: 透明写路径，string 键命中时免字符串深拷贝
1575 if (!Value::dictKeyIsValid(args[0]))
1576 throw RuntimeError(ErrorMessages::kDictKeyInvalidType, line, col, DiagCodes::kTypeMismatch);
1577 Value::dictGetOrInsertRef(obj.dictVal(), args[0]) = args[1];

1579 }

1587 }

1591 auto it = registry.find(method);
1592 if (it != registry.end()) {
1593 return dispatchShared(it->second, obj, args.empty() ? nullptr : args.data(), args.size(), line, col);
1594 }
