39TokenPuzzlePanel::TokenPuzzlePanel(QWidget* parent) : QWidget(parent) {
40 // 初始化完成记录：全部 -1（未完成）
41 levelStars_.resize(TokenPuzzleLibrary::levelCount());
42 for (int& s : levelStars_)
43 s = -1;
44
45 // P0-2 fix (F7): 从持久化存储加载已记录的关卡星级
46 auto& store = LearnerProgressStore::instance();
47 for (int i = 0; i < TokenPuzzleLibrary::levelCount(); ++i) {
48 int s = store.getLevelStars(levelId(i).toStdString());
49 if (s >= 0)
50 levelStars_[i] = s;
51 }
52
53 buildUi();
54
55 // P0-2 fix (F7): 已完成的关卡需在 UI 上解锁下一关
56 // 第 1 关默认解锁，后续关卡仅当前一关已完成（stars >= 0）时解锁
57 auto* model = qobject_cast<QStandardItemModel*>(levelCombo_->model());
58 if (model) {
59 for (int i = 1; i < TokenPuzzleLibrary::levelCount(); ++i) {
60 if (levelStars_[i - 1] >= 0 && model->item(i)) {
61 model->item(i)->setEnabled(true);
62 }
63 }
64 }
65
66 loadLevel(0);
67 updateScoreDisplay();
68}

247void TokenPuzzlePanel::rebuildShuffledButtons() {
248 // 获取或创建网格布局
249 auto* grid = qobject_cast<QGridLayout*>(shuffledContainer_->layout());
250 if (!grid) {
251 grid = new QGridLayout(shuffledContainer_);
252 grid->setContentsMargins(6, 6, 6, 6);
253 grid->setSpacing(6);
254 }
255
256 // 清除旧按钮
257 while (grid->count() > 0) {
258 QLayoutItem* item = grid->takeAt(0);
259 if (item->widget()) {
260 delete item->widget();
261 }
262 delete item;
263 }
264 shuffledButtons_.clear();
265
266 const auto& levels = TokenPuzzleLibrary::levels();
267 if (currentLevelIndex_ < 0 || currentLevelIndex_ >= (int)levels.size())
268 return;
269 const auto& shuffled = levels[currentLevelIndex_].shuffledTokens;
270
271 const int cols = (int)shuffled.size() > 6 ? 6 : (int)shuffled.size();
272 for (int i = 0; i < (int)shuffled.size(); ++i) {
273 auto* btn = new QPushButton(QString::fromUtf8(shuffled[i].c_str()), shuffledContainer_);
274 btn->setMinimumHeight(36);
275 btn->setMinimumWidth(40);
276 btn->setSizePolicy(QSizePolicy::Preferred, QSizePolicy::Fixed);
277 const int buttonIndex = i;
278 connect(btn, &QPushButton::clicked, this, [this, buttonIndex]() { addShuffledTokenToAnswer(buttonIndex); });
279 grid->addWidget(btn, i / cols, i % cols);
280 shuffledButtons_.append(btn);
281 }
282}

330void TokenPuzzlePanel::onCheckAnswer() {
331 // AUDIT-P2 fix: 防重复守卫
332 if (busy_)
333 return;
334 const auto& levels = TokenPuzzleLibrary::levels();
335 if (currentLevelIndex_ < 0 || currentLevelIndex_ >= (int)levels.size())
336 return;
337 // AUDIT-P2 fix: 通过 early return 后才禁用按钮+设标志，函数末尾恢复
338 busy_ = true;
339 checkBtn_->setEnabled(false);
340 const auto& tokens = levels[currentLevelIndex_].tokens;
341
342 // 收集玩家答案
343 QStringList answer;
344 for (int i = 0; i < answerList_->count(); ++i) {
345 answer.append(answerList_->item(i)->text());
346 }
347
348 // 清除之前的高亮
349 for (int i = 0; i < answerList_->count(); ++i) {
350 answerList_->item(i)->setBackground(QBrush());
351 }
352
353 // 数量校验
354 if (answer.size() != (int)tokens.size()) {
355 setFeedback(mlTr("❌ 答案数量不对：你排了 %1 个 token，正确需要 %2 个").arg(answer.size()).arg(tokens.size()),
356 true);
357 // AUDIT-P1 fix: 错误路径必须恢复 busy_ 守卫，否则一次错误尝试后按钮永久禁用
358 busy_ = false;
359 checkBtn_->setEnabled(true);
360 return;
361 }
362
363 // 逐 token 比对
364 int firstWrong = -1;
365 for (int i = 0; i < (int)tokens.size(); ++i) {
366 if (answer[i] != QString::fromUtf8(tokens[i].c_str())) {
367 if (firstWrong < 0)
368 firstWrong = i;
369 }
370 }
371
372 if (firstWrong >= 0) {
373 // 高亮所有错误位置
374 for (int i = 0; i < answerList_->count(); ++i) {
375 if (answer[i] != QString::fromUtf8(tokens[i].c_str())) {
376 answerList_->item(i)->setBackground(QColor(255, 200, 200));
377 }
378 }
379 setFeedback(mlTr("❌ 第 %1 个 token 不对：你填了「%2」，应该是「%3」")
380 .arg(firstWrong + 1)
381 .arg(answer[firstWrong])
382 .arg(QString::fromUtf8(tokens[firstWrong].c_str())),
383 true);
384 // AUDIT-P1 fix: 错误路径必须恢复 busy_ 守卫，允许用户修正后重试
385 busy_ = false;
386 checkBtn_->setEnabled(true);
387 return;
388 }
389
390 // 完全正确
391 int stars = computeStars(hintUsedCount_);
392 if (stars > levelStars_[currentLevelIndex_]) {
393 levelStars_[currentLevelIndex_] = stars;
394 }
395 // P0-2 fix (F7): 持久化关卡星级到 LearnerProgressStore
396 LearnerProgressStore::instance().markLevelStars(levelId(currentLevelIndex_).toStdString(), stars);
397 saveLearnerProgressWithFeedback(this); // P2-UX fix: 失败时弹 toast 避免静默丢失
398 setFeedback(mlTr("✅ 完全正确！获得 %1").arg(starsToText(stars)));
399 unlockNextLevel();
400 updateScoreDisplay();
401 emit activityCompleted(levelId(currentLevelIndex_));
402 // AUDIT-P2 fix: 恢复按钮+标志
403 busy_ = false;
404 checkBtn_->setEnabled(true);
405}

438 unlockNextLevel();

441 // 误标记该活动为已完成）。unlockNextLevel 已基于 levelStars_ 解锁下一关，
442 // 不依赖此信号。0 星持久化由上方 markLevelStars(0) 完成。
443
444 // 自动切换到下一关
445 int next = currentLevelIndex_ + 1;
446 if (next < TokenPuzzleLibrary::levelCount()) {
447 levelCombo_->setCurrentIndex(next);
448 }
449 // AUDIT-P2 fix: 恢复按钮+标志
450 busy_ = false;
451 skipBtn_->setEnabled(true);

474 if (!model || index < 0 || index >= model->rowCount())

476
477 // 禁止切换到未解锁的关卡

565// ============================================================
566// 关卡芯片栏状态刷新
567// ------------------------------------------------------------
568// 三种状态：
569// locked — 未解锁：灰色背景 + 🔒，不可点击
570// unlocked — 已解锁未选中：浅色背景 + 边框，显示关卡号

572// dynamic property 改变后必须 style()->polish() 才能让 QSS 重新评估

575void TokenPuzzlePanel::refreshLevelChips() {
576 auto* model = qobject_cast<QStandardItemModel*>(levelCombo_->model());
577 if (!model)
578 return;
579 const auto& levels = TokenPuzzleLibrary::levels();
580 for (int i = 0; i < levelChips_.size(); ++i) {
581 QPushButton* chip = levelChips_[i];
582 if (!chip)
583 continue;
584
585 bool isLocked = false;
586 if (i < model->rowCount()) {
587 QStandardItem* it = model->item(i);
588 if (it)
589 isLocked = !it->isEnabled();
590 }
591 bool isCurrent = (i == currentLevelIndex_);
592
593 chip->setProperty("locked", isLocked);
594 chip->setProperty("current", isCurrent);
595 chip->setEnabled(!isLocked);
596
597 // 文本：locked 显示锁图标；current 且已获星显示关卡号+⭐；其余显示关卡号
598 QString text;
599 if (isLocked) {
600 // 🔒 = U+1F512 = UTF-8: F0 9F 94 92
601 text = QString::fromUtf8("\xF0\x9F\x94\x92");
602 } else if (isCurrent && i < levelStars_.size() && levelStars_[i] > 0) {
603 // ⭐ = U+2B50 = UTF-8: E2 AD 90
604 text = QString::fromUtf8("%1 \xE2\xAD\x90").arg(i + 1);
605 } else {
606 text = QString::number(i + 1);
607 }
608 chip->setText(text);
609
610 // 工具提示：显示关卡标题和教学点（目标）
611 if (i < (int)levels.size()) {
612 QString tip = mlTr("第 %1 关").arg(i + 1);
613 if (!levels[i].teachingPoint.empty()) {
614 tip += "\n" + QString::fromUtf8(levels[i].teachingPoint.c_str());
615 }
616 if (i < levelStars_.size() && levelStars_[i] > 0) {
617 tip += "\n" + mlTr("已获星：%1").arg(levelStars_[i]);
618 }
619 chip->setToolTip(tip);
620 }
621
622 // 强制 QSS 重新评估 dynamic property 选择器
623 chip->style()->unpolish(chip);
624 chip->style()->polish(chip);
625 }
