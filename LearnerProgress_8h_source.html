4// LearnerProgressStore 单例管理 LearnerProgress 数据：

8// - currentStage: 当前阶段（0-4）

10// - score: 每活动得分（0-100，P2-3 fix F9 学情画像）
11// - bestStars: 每活动历史最佳星级（与 levelStars 不同——
12// levelStars 针对子关卡，bestStars 针对活动整体）
13// - spentMinutes: 每活动累计花费分钟数（P2-3 fix F9）
14// - failCount: 每活动失败次数（P2-3 fix F9 薄弱点判定依据）
15//
16// 持久化：
17// - JSON 文件 ~/.minilang_progress.json（路径由 QStandardPaths 决定）
18// - 使用 QJsonDocument / QJsonObject / QJsonArray
19// - 加载失败/格式不匹配时回退到空进度，不崩溃
20// - 旧版本 JSON 缺失新字段时按默认值 0 处理，向后兼容
21//
22// 测试支持：
23// - resetForTesting() 清空内部状态
24// - 测试可通过设置临时文件路径或调用 setFilePathForTesting() 注入

38// LearnerProgress — 进度数据
39// ============================================================
40struct LearnerProgress {
41 std::map<std::string, bool> completed; // 已完成的活动
42 std::map<std::string, int> attemptCount; // 尝试次数
43 std::map<std::string, int64_t> lastAccessTime; // 最后访问时间（unix timestamp）
44 int currentStage = 0; // 当前阶段（0-4）
45 // P0-2 fix (F7): 细粒度关卡星级持久化。
46 // key = 关卡 ID（如 "token-puzzle-1"），value = 星级（-1=未完成，0=跳过，1-3=星级）。
47 // 游戏面板（TokenPuzzle/AstToy/VmSandbox）在构造时读取、在关卡完成时写入，
48 // 解决原 levelStars_ 仅存内存、重启即丢失的问题。
49 std::map<std::string, int> levelStars;
50 // P2-3 fix (F9): 学情画像——多维进度数据，让进度从「打卡」升级为「画像」。
51 // score: key = 活动 ID, value = 得分（0-100）。recordScore 取最大值（保留历史最佳）。
52 // bestStars: key = 活动 ID, value = 1-3 星级（与 levelStars 不同——levelStars 针对子关卡，
53 // bestStars 针对活动整体，由 markActivityScore 在完成时一并记录）。
54 // spentMinutes: key = 活动 ID, value = 累计花费分钟数。addSpentMinutes 累加。
55 // failCount: key = 活动 ID, value = 失败次数（用于 getWeakPoints 薄弱点判定）。
56 std::map<std::string, int> score;
57 std::map<std::string, int> bestStars;
58 std::map<std::string, int> spentMinutes;
59 std::map<std::string, int> failCount;
60};
