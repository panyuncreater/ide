28/// 以 id 添加带图标与文字的活动项，返回其索引。
29int ActivityBar::addItem(const QString& id, const QString& text, Fluent::IconType icon) {
30 // 唯一性校验
31 if (indexOf(id) >= 0)
32 return -1;
33
34 auto* btn = new TransparentToolButton(icon, this);
35 btn->setFixedSize(36, 36);
36 btn->setIconSize(QSize(20, 20));
37 btn->setToolTip(text);
38 btn->setCheckable(true);
39
40 int index = items_.size();
41 items_.append({btn, id, text});
42
43 // Insert before the stretch
44 layout_->insertWidget(layout_->count() - 1, btn, 0, Qt::AlignHCenter);
45
46 connect(btn, &QToolButton::clicked, this, [this, index]() { setCurrentIndex(index); });
47
48 if (currentIndex_ < 0) {
49 setCurrentIndex(index);
50 }
51
52 return index;
53}

62/// 按索引设置当前选中活动项并刷新高亮。
63void ActivityBar::setCurrentIndex(int index) {
64 if (index < 0 || index >= items_.size())
65 return;
66 if (currentIndex_ == index) {
67 // R51-4 fix: checkable QToolButton 点击会 toggle checked 状态。
68 // 若点击的是当前已选中项，early return 跳过 updateSelection 会导致
69 // 按钮停留在 unchecked 视觉态。此处仍需刷新以恢复正确的选中态。
70 updateSelection();
71 return;
72 }
73 currentIndex_ = index;
74 updateSelection();
75 emit currentChanged(index);
76 emit currentChangedById(items_[index].id);
77}
