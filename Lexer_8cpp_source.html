171std::vector<Token> Lexer::scan(const std::string& source) {
172 // P0 fix: 源码大小上限检查，防止恶意大文件导致 DoS
173 if (source.size() > MAX_SOURCE_SIZE) {
174 diagnostics_.addError("源代码过大（" + std::to_string(source.size() / 1024 / 1024) + "MB），超过上限 " +
175 std::to_string(MAX_SOURCE_SIZE / 1024 / 1024) + "MB",
176 1, 1, DiagSource::Lexer, DiagCodes::kSourceTooLarge);
177 tokens_.clear();
178 tokens_.emplace_back(TokenType::TK_EOF, "", std::monostate{}, 1, 1);
179 return tokens_;
180 }
181
182 source_ = source;
183 start_ = 0;
184 current_ = 0;
185 line_ = 1;
186 lineStart_ = 0;
187 // Perf-Finding1: 失效 columnAt 缓存（source_ 已更换，旧 byteOffset 失效）
188 cachedLineStart_ = -1;
189 cachedByteOffset_ = -1;
190 cachedColumn_ = 1;
191 tokens_.clear();
192 comments_.clear();
193 diagnostics_.clear();
194 interpDepth_ = 0; // L-P1-1: 重置插值嵌套深度
195 // PERF-19 fix: 预分配 tokens_ 容量，避免大文件场景多次 realloc。
196 // 估算：平均每 8 字符产生 1 个 token（关键字/标识符/字面量/运算符）
197 // P0 fix: 限制 reserve 在 MAX_TOKEN_COUNT 内——原 `source.size()/8 + 16` 对 10MB 源码
198 // 会预分配 1.25M 槽位（~110MB），超过 MAX_TOKEN_COUNT (1M) 25%，单次大额堆分配极易
199 // 抛 bad_alloc（特别是已运行过其他工具的内存压力下）。token 上限检查会强制 break，
200 // 预分配超过上限纯属浪费。
201 size_t reserveCap = source.size() / 8 + 16;
202 if (reserveCap > MAX_TOKEN_COUNT)

204 tokens_.reserve(reserveCap);

207 if (source_.size() >= 3 && static_cast<unsigned char>(source_[0]) == 0xEF &&
208 static_cast<unsigned char>(source_[1]) == 0xBB && static_cast<unsigned char>(source_[2]) == 0xBF) {
209 current_ = 3;
210 start_ = 3;
211 lineStart_ = 3;
212 }

214 while (!isAtEnd()) {
215 start_ = current_;
216 scanToken();
217 // P0 fix: Token 数量上限检查，防止 Token 爆炸导致 OOM
218 // AUDIT-R3 P0 fix: 比较运算符 > 改为 >=，与 scanToken 入口检查（>=）对齐。
219 // 原不一致导致 token 数恰好等于上限时：scanToken 拒绝消费输入（不推进
220 // current_）而本循环又不 break，形成死循环且每轮追加一条诊断（内存无界增长）。
221 if (tokens_.size() >= MAX_TOKEN_COUNT) {
222 diagnostics_.addError("Token 数量超过上限 " + std::to_string(MAX_TOKEN_COUNT) +
223 "，源代码可能包含过多 token",
224 line_, currentColumn(), DiagSource::Lexer, DiagCodes::kTooManyTokens);
225 break;
226 }
227 }
228
229 // 添加 EOF Token
230 tokens_.emplace_back(TokenType::TK_EOF, "", std::monostate{}, line_, currentColumn());
231
232 // 从 TK_ERROR Token 中提取诊断信息，并将注释 Token 分离到 comments_
233 // P0 fix: 不再 reserve(tokens_.size())——原代码会瞬时分配与 tokens_ 等大的内存，
234 // 峰值达 tokens_ + cleanTokens 双倍（~220MB）。改为让 cleanTokens 自然增长，
235 // 注释 Token 通常很少，cleanTokens.size() ≈ tokens_.size() - comments_.size()，
236 // move 后 tokens_ 释放，最终 cleanTokens 占用 ≈ 原大小。略多几次 realloc 但避免峰值翻倍。
237 std::vector<Token> cleanTokens;
238 for (auto& tok : tokens_) {
239 if (tok.type == TokenType::TK_ERROR) {
240 diagnostics_.addError(tok.lexeme, tok.line, tok.column, DiagSource::Lexer);
241 cleanTokens.push_back(std::move(tok));
242 } else if (tok.type == TokenType::TK_LINE_COMMENT || tok.type == TokenType::TK_BLOCK_COMMENT) {
243 comments_.push_back(std::move(tok));
244 } else {
245 cleanTokens.push_back(std::move(tok));
246 }
247 }
248 tokens_ = std::move(cleanTokens);
249
250 return std::move(tokens_);
251}

253/// 前瞻当前字符（不前进）；到达末尾返回 '哨兵，供双字符运算符/浮点判断使用。

272/// 前进一步并返回当前字符；维护行号/行首偏移，将 \r\n 与裸 \n 统一为单个 \n。
273char Lexer::advance() {
274 if (current_ >= static_cast<int>(source_.size()))
275 return '\0'; // 防御性边界检查
276 char c = source_[current_];
277 current_++;
278 if (c == '\n') {
279 line_++;
280 lineStart_ = current_;
281 } else if (c == '\r') {
282 // \r\n 视为单个换行；裸 \r（旧 Mac 格式）也触发换行
283 if (current_ < static_cast<int>(source_.size()) && source_[current_] == '\n') {
284 current_++;
285 }
286 line_++;
287 lineStart_ = current_;
288 return '\n'; // LEX-01/02/03 fix: 统一返回 '\n'，确保注释循环和字符串构建正确感知换行
289 }
290 return c;
291}

309/// 字符串/数字/标识符/注释）走对应分支，并处理双字符运算符查表与插值字符串递归。
310/// 入口处强制检查 MAX_TOKEN_COUNT，防止插值路径绕过主循环的 Token 上限防护。

313 // 递归调用 scanToken）绕过 scan() 外层循环的 MAX_TOKEN_COUNT 检查。

317 // Hard Constraint #27: Lexer must check MAX_TOKEN_COUNT at start of scanToken().
318 if (tokens_.size() >= MAX_TOKEN_COUNT) {

413bool Lexer::scanSlashToken(char c) {
414 if (c != '/')
415 return false;
416
417 // 单行注释
418 if (match('/')) {
419 // 捕获注释文本（含 // 前缀）
420 size_t commentStart = start_;
421 // P0-2 fix: 同时检查 \r 和 \n，避免 CRLF 下 advance() 消耗 \r\n 后
422 // peek() 跳过 \n 导致注释吞掉下一行内容
423 while (!isAtEnd() && peek() != '\n' && peek() != '\r')
424 advance();
425 std::string commentText(source_, commentStart, current_ - commentStart);
426 Token tok;
427 tok.type = TokenType::TK_LINE_COMMENT;
428 tok.lexeme = commentText;
429 tok.line = line_;
430 // BUG-LEX-AUDIT-3 fix: 统一用 UTF-8 感知 columnAt 计算列号，
431 // 与 addToken() 路径保持一致（D11 fix 遗漏了注释 token 路径）。
432 tok.column = columnAt(static_cast<int>(commentStart));
433 tokens_.push_back(tok);
434 return true;
435 }
436 if (match('*')) {
437 // 块注释 /* ... */（支持嵌套）
438 size_t commentStart = start_;
439 int startLine = line_;
440 // BUG-LEX-AUDIT-3 fix: 同行注释，统一用 columnAt
441 int startCol = columnAt(static_cast<int>(commentStart));
442 int depth = 1; // 嵌套深度
443 while (!isAtEnd() && depth > 0) {
444 if (peek() == '/' && peekNext() == '*') {
445 advance();
446 advance(); // 消耗 /*
447 depth++;
448 } else if (peek() == '*' && peekNext() == '/') {
449 advance();
450 advance(); // 消耗 */
451 depth--;
452 } else {
453 advance();
454 }
455 }
456 if (depth > 0) {
457 errorToken("未终止的块注释", startLine, startCol);
458 return true;
459 }
460 std::string commentText(source_, commentStart, current_ - commentStart);
461 Token tok;
462 tok.type = TokenType::TK_BLOCK_COMMENT;
463 tok.lexeme = commentText;
464 tok.line = startLine;
465 tok.column = startCol;
466 tokens_.push_back(tok);
467 return true;
468 }
469 addToken(TokenType::TK_SLASH);
470 return true;
471}

513 {'&', '&', TokenType::TK_ERROR, TokenType::TK_ERROR, "请使用 'and' 关键字代替 '&'"},
514 {'|', '|', TokenType::TK_ERROR, TokenType::TK_ERROR, "请使用 'or' 关键字代替 '|'"},
515 };
516 // #28 fix: [256] 查找表按首字符 O(1) 定位条目，替代 6 项线性扫描。
517 // 首字符仅这 6 个有条目，其余槽位为 nullptr。首次调用时初始化，后续零开销。
518 static const auto firstCharTable = []() {
519 std::array<const TwoCharOp*, 256> t{};
520 for (const auto& op : twoCharOps) {
521 t[static_cast<unsigned char>(op.first)] = &op;
522 }
523 return t;
524 }();
525 const TwoCharOp* entry = firstCharTable[static_cast<unsigned char>(c)];
526 if (!entry)
527 return false;
528 if (entry->second != '\0' && match(entry->second)) {
529 if (entry->dualType != TokenType::TK_ERROR)
530 addToken(entry->dualType);
531 else
532 errorToken(std::string("'") + c + c + "'（" + entry->hint + "）");
533 } else if (entry->soloType != TokenType::TK_ERROR) {
534 addToken(entry->soloType);
535 } else {
536 errorToken(std::string("意外字符 '") + c + "'（" + entry->hint + "）");
537 }
538 return true;
539}

541void Lexer::scanLiteralOrUnknownToken(char c) {

543 if (c == '"') {
544 string();

549 number();
550 return;

557 // 无法识别的字符
558 // BUG-LPA-06 fix: 检测 UTF-8 多字节字符首字节，消费完整码位生成单个错误。
559 // 原实现按单字节处理，3 字节 UTF-8 字符（如中文）产生 3 个错误，

574/// 识别标识符或关键字：消费 [A-Za-z0-9_]*，先拦截保留前缀 '__'，

576void Lexer::identifier() {

578 advance();

610 addToken(type, std::move(text), false); // A1 fix: variant bool
611 } else if (type == TokenType::TK_NULL) {
612 // null 关键字有字面量值
613 addToken(type, std::move(text), std::monostate{}); // A1 fix: variant monostate
614 } else {
615 addToken(type, std::move(text));
616 }
617 } else {
618 addToken(TokenType::TK_IDENTIFIER, std::move(text));
619 }
620}

622/// 解析数字字面量：处理前导点浮点、拒绝 0x/0b/0o 前缀与前导零，
623/// 识别小数与科学计数法，最后用 locale-independent 的 std::from_chars 转换为 int64/double，
624/// 溢出或格式错误时发出错误 Token 而非崩溃。
625void Lexer::number() {
626 // 前导点浮点（.123）：scanToken 已消耗 '.'，start_ 指向 '.'，跳过整数部分
627 bool isFloat = (source_[start_] == '.');
628
629 // M6 fix: 检测 0x/0b/0o 前缀，发出明确错误（而非静默分为两个 token）
630 if (!isFloat && !isAtEnd() && source_[start_] == '0' &&
631 (peek() == 'x' || peek() == 'X' || peek() == 'b' || peek() == 'B' || peek() == 'o' || peek() == 'O')) {
632 char prefix = peek();
633 advance(); // 消耗前缀字母
634 // P2-2 fix: 用 isalnum 消费整个非法字面量，避免 isxdigit 对 0b/0o 前缀的语义错误
635 // （isxdigit 会消费 a-f，但 0bff 中 ff 不是合法二进制位；此处统一消费字母数字即可）
636 while (!isAtEnd() && isAsciiAlphaNum(peek())) {
637 advance();
638 }
639 std::string text(source_, start_, current_ - start_);
640 errorToken("不支持 " + std::string(1, prefix) + " 前缀字面量: " + text + "，请使用十进制表示");
641 return;
642 }
643
644 // AUDIT-P2-CORRECT fix: 拒绝前导零（如 007, 008, 099），避免教学歧义与
645 // 未来八进制解析风险。MiniLang 作为教学语言，前导零在 C 中被误解为八进制、
646 // Python3 中报错，静默接受会误导学习者。例外：单独的 "0" 合法；
647 // "0." 开头的浮点数由下方 isFloat 分支处理（source_[start_]=='.' 为 false，
648 // 但 peek()=='.' 不会进入此分支，因 isAsciiDigit('.') 为 false）。
649 if (!isFloat && !isAtEnd() && source_[start_] == '0' && isAsciiDigit(peek())) {
650 // 前导零：0 后跟数字，报错并消费整个数字序列
651 while (!isAtEnd() && isAsciiDigit(peek())) {
652 advance();
653 }
654 std::string text(source_, start_, current_ - start_);
655 // 提供等价的合法值建议
656 int64_t suggestion = 0;
657 auto [ptr, ec] = std::from_chars(text.data() + 1, text.data() + text.size(), suggestion);
658 std::string hint = (ec == std::errc()) ? "，请使用 " + std::to_string(suggestion) : "";
659 errorToken("不允许前导零: " + text + hint);
660 return;
661 }
662
663 while (!isAtEnd() && isAsciiDigit(peek())) {
664 advance();
665 }
666
667 // 浮点数：小数部分（仅当 '.' 后紧跟数字时才视为浮点，避免 123.foo 被误分词）
668 if (!isFloat && !isAtEnd() && peek() == '.' && (static_cast<size_t>(current_) + 1 < source_.size()) &&
669 isAsciiDigit(source_[current_ + 1])) {
670 isFloat = true;
671 advance(); // 消耗 '.'
672 while (!isAtEnd() && isAsciiDigit(peek())) {
673 advance();
674 }
675 }
676
677 // M5 fix: 整数后直接跟 .e/.E 形式（如 123.e5），消耗 '.' 后进入科学计数法
678 if (!isFloat && !isAtEnd() && peek() == '.' && (static_cast<size_t>(current_) + 1 < source_.size()) &&
679 (source_[current_ + 1] == 'e' || source_[current_ + 1] == 'E')) {
680 isFloat = true;
681 advance(); // 消耗 '.'
682 }
683
684 // R53-LEX-REVERT fix: 回滚 AUDIT-P3-ROUND53 的"123. 报错"逻辑。
685 // 该 fix 让 "1." 和 "123.foo" 报错"数字字面量小数点后需有数字"，但
686 // TestLexer::Number_IntegerFollowedByDotAndIdentifier 和
687 // LexerAudit::TrailingDecimalPoint 明确期望分词为 INT + DOT（+ IDENTIFIER），
688 // 这是项目的设计契约——MiniLang 不支持方法调用语法 123.foo()，但 Lexer
689 // 应保持宽容分词，将语义判断交给 Parser。原 fix 改变了既定行为，导致回归。
690 // 现恢复"123. 后无数字/指数 → 分词为 TK_INT_LIT + TK_DOT"的行为。
691
692 // #15: 科学计数法（如 1e5, 3.14e-2, 2E+10, 123.e5）
693 if (!isAtEnd() && (peek() == 'e' || peek() == 'E')) {
694 isFloat = true;
695 advance(); // 消耗 'e'/'E'
696 if (!isAtEnd() && (peek() == '+' || peek() == '-')) {
697 advance(); // 消耗符号
698 }
699 if (isAtEnd() || !isAsciiDigit(peek())) {
700 errorToken("科学计数法格式错误: " + std::string(source_, start_, current_ - start_));
701 return;
702 }
703 while (!isAtEnd() && isAsciiDigit(peek())) {
704 advance();
705 }
706 }
707
708 std::string text(source_, start_, current_ - start_);
709
710 if (isFloat) {
711 // L-P1-3 fix: 使用 std::from_chars 替代 std::stod，locale-independent，
712 // 避免系统 locale 用 ',' 作小数点时 "1.5" 解析错误
713 double val = 0.0;
714 auto [ptr, ec] = std::from_chars(text.data(), text.data() + text.size(), val);
715 if (ec == std::errc::result_out_of_range) {
716 // AUDIT-R3 P2 fix: 区分下溢与上溢——from_chars 对两者均报 result_out_of_range。
717 // 下溢（如 1e-400、0.000...01）主流语言语义为钳制到 0.0，误报"溢出"会
718 // 拒绝合法的极小字面量。
719 // AUDIT-R4 BUG-07 fix: 原启发式"负指数或以 '0' 开头即下溢"会把 0.1e999
720 // 这类真实上溢字面量误判为下溢并静默钳制为 0.0。改为解析十进制数量级：
721 // 有效指数 = 显式指数 E + 首个非零尾数位相对小数点的幂次 p，
722 // 有效指数 < 0 → 下溢（钳制 0.0），否则 → 上溢（报错）。
723 // out_of_range 保证 |有效指数| > ~308，符号判定无歧义。
724 long long expVal = 0;
725 size_t ePos = text.find_first_of("eE");
726 if (ePos != std::string::npos) {
727 bool negExp = false;
728 size_t di = ePos + 1;
729 if (di < text.size() && (text[di] == '+' || text[di] == '-')) {
730 negExp = (text[di] == '-');
731 ++di;
732 }
733 for (; di < text.size() && isAsciiDigit(text[di]); ++di) {
734 if (expVal < 1000000000) // 钳制防指数数字串本身溢出，足够判定符号
735 expVal = expVal * 10 + (text[di] - '0');
736 }
737 if (negExp)
738 expVal = -expVal;
739 }
740 // 首个非零尾数位相对小数点的幂次 p（"123.4" 的 '1' → p=2；"0.001" 的 '1' → p=-3）
741 std::string_view mant = std::string_view(text).substr(0, ePos == std::string::npos ? text.size() : ePos);
742 size_t dotPos = mant.find('.');
743 long long p = 0;
744 size_t firstSig = std::string::npos;
745 for (size_t i = 0; i < mant.size(); ++i) {
746 if (mant[i] >= '1' && mant[i] <= '9') {
747 firstSig = i;
748 break;
749 }
750 }
751 if (firstSig != std::string::npos) {
752 if (dotPos == std::string_view::npos || firstSig < dotPos) {
753 size_t intEnd = (dotPos == std::string_view::npos) ? mant.size() : dotPos;
754 p = static_cast<long long>(intEnd - firstSig) - 1;
755 } else {
756 p = -static_cast<long long>(firstSig - dotPos);
757 }
758 }
759 if (expVal + p < 0) {
760 addToken(TokenType::TK_FLOAT_LIT, std::move(text), 0.0); // 下溢钳制到 0.0
761 } else {
762 errorToken("浮点数溢出: " + text);
763 }
764 } else if (ec != std::errc() || ptr != text.data() + text.size()) {
765 errorToken("浮点数格式错误: " + text);
766 } else {
767 addToken(TokenType::TK_FLOAT_LIT, std::move(text), val); // A1 fix: variant double
768 }
769 } else {
770 // L-P1-3 fix: 使用 std::from_chars 替代 std::stoll，locale-independent
771 int64_t val = 0;
772 auto [ptr, ec] = std::from_chars(text.data(), text.data() + text.size(), val);
773 if (ec == std::errc::result_out_of_range) {
774 // Bug #40 fix: 9223372036854775808 是 INT64_MIN 的绝对值，
775 // 存储为 INT64_MIN，后续一元负号会正确处理。
776 if (text == "9223372036854775808") {
777 addToken(TokenType::TK_INT_LIT, std::move(text), INT64_MIN);
778 } else {
779 errorToken("整数溢出: " + text);
780 }
781 } else if (ec != std::errc() || ptr != text.data() + text.size()) {
782 errorToken("整数格式错误: " + text);
783 } else {
784 addToken(TokenType::TK_INT_LIT, std::move(text), val); // A1 fix: variant int64_t
785 }
786 }

793/// 解析字符串字面量（支持 {expr} 插值）：消费转义序列，遇 { 递归扫描插值表达式并维护嵌套深度与 Token 上限。
794void Lexer::string(bool isInterp) {
795 int startLine = line_;
796 // BUG-LEX-AUDIT-3 fix: 统一用 UTF-8 感知 columnAt 计算列号，
797 // 与 addToken() 路径保持一致（D11 fix 仅改了 addToken，遗漏直接 emplace_back 路径）。
798 int startCol = columnAt(start_);
799 std::string value;
800 // P-07 fix: 预估字符串容量，避免逐字符 += 反复 realloc
801 // P0 fix: 限制 reserve 上限，防止未闭合字符串触发 GB 级内存分配
802 // BUG-LEX-AUDIT-5 fix: 上限从 1MB 降为 64KB。原 1MB 在 MAX_INTERP_DEPTH=64 嵌套场景
803 // 峰值 reserved 达 64MB 但大多未使用；64KB 上限下嵌套峰值仅 4MB，平衡 OOM 防护与内存压力。
804 size_t reserveCap =
805 static_cast<size_t>(current_) < source_.size() ? (source_.size() - static_cast<size_t>(current_)) : 0;
806 if (reserveCap > 64 * 1024)
807 reserveCap = 64 * 1024;
808 value.reserve(reserveCap);
809
810 while (!isAtEnd() && peek() != '"') {
811 // F7: 检测插值起始 {
812 if (peek() == '{') {
813 if (!handleInterpolation(startLine, startCol, value, isInterp))
814 return;
815 continue;
816 }
817
818 // 换行处理由 advance() 统一完成（line_++ 和 lineStart_ 更新），
819 // 此处不再手动递增，否则会导致行号双重递增。
820 if (peek() == '\\') {
821 if (!handleEscape(startLine, startCol, value))
822 return;
823 continue;
824 }
825
826 // PERF-ROUND53 fix: 批量扫描连续普通字符（直到遇 \、{、"、\r、\n 或 EOF）。
827 // 原实现逐字符调用 advance() + value += char，对长字符串（如 1KB 文本）
828 // 产生 1024 次 advance() 调用 + 1024 次 value += char（可能触发多次 realloc）。
829 // 批量扫描用 value.append(ptr, len) 一次性追加，减少函数调用与 realloc 次数。
830 // 注：\r/\n 必须通过 advance() 处理以维护行号（advance 将 \r/\r\n 规范化为 \n），
831 // 故批量扫描在遇到它们时停止，由下次循环迭代调用 advance()。
832 // 多字节 UTF-8 字节（0x80-0xFF）均不与特殊字符（\=0x5C、{=0x7B、"=0x22、
833 // \r=0x0D、\n=0x0A）冲突，可安全批量扫描。
834 size_t runStart = static_cast<size_t>(current_);
835 while (!isAtEnd()) {
836 char c = source_[current_];
837 if (c == '"' || c == '\\' || c == '{' || c == '\r' || c == '\n')
838 break;
839 current_++;
840 }
841 if (static_cast<size_t>(current_) > runStart) {
842 value.append(source_.data() + runStart, static_cast<size_t>(current_) - runStart);
843 } else {
844 // 首个字符即为 \r/\n（其他特殊字符已被外层 if 拦截），
845 // 调用 advance() 维护行号并将规范化后的字符（\n）加入 value
846 value += advance();
847 }
848 }
849
850 if (isAtEnd()) {
851 errorToken("未终止的字符串", startLine, startCol);
852 return;
853 }
854
855 advance(); // 消耗闭合的 '"'
856
857 // F7: 如果是插值字符串的后续片段，用 TK_STRING_PART；否则用 TK_STRING_LIT
858 TokenType finalType = isInterp ? TokenType::TK_STRING_PART : TokenType::TK_STRING_LIT;
859 // BUG-LEX-AUDIT-4 fix: 末尾片段也需检查 MAX_TOKEN_COUNT
860 if (tokens_.size() >= MAX_TOKEN_COUNT) {
861 diagnostics_.addError("Token 数量超过上限 " + std::to_string(MAX_TOKEN_COUNT) + "，源代码可能包含过多 token",
862 startLine, startCol, DiagSource::Lexer, DiagCodes::kTooManyTokens);
863 return;
864 }
865 std::string text(source_, start_, current_ - start_);
866 tokens_.emplace_back(finalType, std::move(text), std::move(value), startLine, startCol); // A1 fix: variant string
867}

869// ============================================================
870// R131 fix: handleInterpolation - 处理字符串中的 '{' 插值起始
871// 提取自 string() 主循环的 '{' 分支（117 行），含插值深度检查、
872// TK_STRING_PART/TK_INTERP_START 发射、嵌套表达式扫描、TK_INTERP_END 发射
873// 返回 true=继续外层 string 循环，false=终止 string()
874// ============================================================
875bool Lexer::handleInterpolation(int& startLine, int& startCol, std::string& value, bool& isInterp) {
876 // L-P1-1: 插值嵌套深度检查，防止栈溢出
877 if (interpDepth_ >= MAX_INTERP_DEPTH) {
878 errorToken("字符串插值嵌套过深（最大 " + std::to_string(MAX_INTERP_DEPTH) + " 层）", startLine, startCol);
879 return false;
880 }
881 interpDepth_++;
882 // AUDIT-R3 P2 fix: RAII 守卫替代手动配对递减——原实现中 4 处 MAX_TOKEN_COUNT
883 // 早退路径（return false）遗漏递减，与"错误退出时也减少深度"的契约矛盾。
884 struct InterpDepthGuard {
885 int& d;
886 ~InterpDepthGuard() { --d; }
887 } depthGuard{interpDepth_};
888 // 发出前面的文本片段（TK_STRING_PART 表示插值字符串的一部分）
889 TokenType partType = isInterp ? TokenType::TK_STRING_PART : TokenType::TK_STRING_LIT;
890 // 如果是插值字符串的第一个片段，用 TK_STRING_LIT；后续片段用 TK_STRING_PART
891 // 但为简化 Parser 逻辑，统一：插值字符串中所有文本片段都用 TK_STRING_PART，
892 // 仅当整个字符串无插值时用 TK_STRING_LIT（由下方闭合处判断）
893 // BUG-LEX-AUDIT-4 fix: 直接 emplace_back 绕过 scanToken() 的 MAX_TOKEN_COUNT 检查，
894 // 需在此显式检查，防止含大量小插值的字符串绕过 DoS 防护。
895 if (tokens_.size() >= MAX_TOKEN_COUNT) {
896 diagnostics_.addError("Token 数量超过上限 " + std::to_string(MAX_TOKEN_COUNT) + "，源代码可能包含过多 token",
897 startLine, startCol, DiagSource::Lexer, DiagCodes::kTooManyTokens);
898 return false;
899 }
900 std::string text(source_, start_, current_ - start_);
901 tokens_.emplace_back(partType, std::move(text), std::move(value), startLine,
902 startCol); // A1 fix: variant string
903
904 // 发出 TK_INTERP_START
905 // BUG-LEX-AUDIT-2 fix: 在 advance() 消耗 '{' 之前记录列号，
906 // 原实现 advance 后用 start_ 计算，但 start_ 仍指向片段起始而非 '{' 位置。
907 int braceLine = line_;
908 int braceCol = columnAt(current_);
909 // 消耗 {
910 advance();
911 if (tokens_.size() >= MAX_TOKEN_COUNT) { // BUG-LEX-AUDIT-4
912 diagnostics_.addError("Token 数量超过上限 " + std::to_string(MAX_TOKEN_COUNT) + "，源代码可能包含过多 token",
913 braceLine, braceCol, DiagSource::Lexer, DiagCodes::kTooManyTokens);
914 return false;
915 }
916 tokens_.emplace_back(TokenType::TK_INTERP_START, "{", std::monostate{}, braceLine,
