13Value NumberLiteral::getValue() const {
14 // 根据字面量标记选择构造浮点 Value 或整数 Value（A1 fix: 标量按需封装）。
15 // 之所以同时保留 intValue_ 与 floatValue_ 两个字段，是因为 MiniLang 区分
16 // 整数与浮点类型；getValue() 在读取时才决定封装成哪种 Value，
17 // 这样 AST 节点保持轻量且可被多次遍历而不会产生副作用（非破坏性）。
18 return isFloat_ ? Value(floatValue_) : Value(intValue_);
19}
