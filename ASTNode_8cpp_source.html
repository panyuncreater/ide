13Value NumberLiteral::getValue() const {

29Value StringLiteral::getValue() const {
30 // OPT-03: 缓存首次构造的 Value，循环内字符串字面量不再每轮堆分配 StringData。
31 // 共享同一 AST 的解释器可能在不同线程求值；CAS 发布指针，竞态失败方丢弃自建副本。
32 // 缓存对象随 AST 节点销毁，返回的 Value 各自持 StringData 引用（COW 隔离）。
33 Value* cached = cachedValue_.load(std::memory_order_acquire);
34 if (!cached) {
35 auto* candidate = new Value(value);
36 if (!cachedValue_.compare_exchange_strong(cached, candidate, std::memory_order_acq_rel,
37 std::memory_order_acquire)) {
38 delete candidate;
39 } else {
40 cached = candidate;
41 }
42 }
43 return *cached;
44}
