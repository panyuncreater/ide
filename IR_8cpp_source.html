132std::string formatIRInstruction(const IRInstruction& instr) {
133 std::ostringstream oss;
134 oss << irOpName(instr.op);
135 for (const auto& operand : instr.operands) {
136 const char* kindStr = "?";
137 switch (operand.kind) {
138 case IROperandKind::CONSTANT:
139 kindStr = "c";
140 break;
141 case IROperandKind::VIRTUAL:
142 kindStr = "v";
143 break;
144 case IROperandKind::LABEL:
145 kindStr = "L";
146 break;
147 case IROperandKind::GLOBAL_NAME:
148 kindStr = "g";
149 break;
150 case IROperandKind::LOCAL_SLOT:
151 kindStr = "s";
152 break;
153 case IROperandKind::UPVALUE_IDX:
154 kindStr = "u";
155 break;
156 case IROperandKind::FIELD_NAME:
157 kindStr = "f";
158 break;
159 case IROperandKind::FUNC_NAME:
160 kindStr = "fn";
161 break;
162 case IROperandKind::IMM_UINT:
163 kindStr = "#";
164 break;
165 default:
166 kindStr = "?";
167 break; // Bug-6: 缺 default 防新增枚举静默走 ?
168 }
169 oss << " " << kindStr << operand.index;
170 }
171 if (instr.line > 0)
172 oss << " ; line " << instr.line;
173 return oss.str();
174}

176std::string IRToString(const IRFunction& ir) {
177 std::ostringstream oss;
178 oss << "IRFunction \"" << ir.name << "\" {\n";
179 oss << " constants: " << ir.constants.size() << "\n";
180 oss << " globals: " << ir.globalNames.size() << "\n";
181 oss << " vregs: " << ir.nextVReg << "\n";
182 oss << " labels: " << ir.nextLabel << "\n";
183 oss << " blocks: " << ir.blocks.size() << "\n\n";
184
185 for (size_t bi = 0; bi < ir.blocks.size(); ++bi) {
186 const auto& block = ir.blocks[bi];
187 oss << " BB" << bi << " (label=" << block.labelIndex << "):\n";
188 for (const auto& instr : block.instructions) {
189 oss << " " << formatIRInstruction(instr) << "\n";
190 }
191 oss << "\n";
192 }
193 oss << "}\n";
194 return oss.str();
195}
