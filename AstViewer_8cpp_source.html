161void AstViewer::wheelEvent(QWheelEvent* event) {
162 // 第八轮：直接滚轮缩放（无需 Ctrl），对齐 VS Code Magnus/AST 视图行为
163 // Ctrl/无修饰 → 缩放；Shift → 水平滚动；Alt → 垂直滚动
164 if (event->modifiers() & Qt::AltModifier) {
165 QScrollBar* vBar = verticalScrollBar();
166 if (vBar) {
167 int delta = event->angleDelta().y();
168 vBar->setValue(vBar->value() - delta);
169 }
170 event->accept();
171 return;
172 }
173 if (event->modifiers() & Qt::ShiftModifier) {
174 QScrollBar* hBar = horizontalScrollBar();
175 if (hBar) {
176 int delta = event->angleDelta().y();
177 hBar->setValue(hBar->value() - delta);
178 }
179 event->accept();
180 return;
181 }
182 // 默认：滚轮直接缩放（GUI-13 fix: 范围限制 0.1x ~ 10x）
183 double factor = 1.15;
184 double currentScale = transform().m11();
185 if (event->angleDelta().y() > 0) {
186 if (currentScale * factor <= 10.0)
187 scale(factor, factor);
188 } else {
189 if (currentScale / factor >= 0.1)
190 scale(1.0 / factor, 1.0 / factor);
191 }
192 event->accept();
193}

401void AstViewer::layoutSubtree(RtNode* node, int depth) {
402 if (!node)
403 return;
404
405 // 折叠节点：不布局子树，轮廓仅含自身
406 if (node->collapsed) {
407 node->finalX = 0;
408 return;
409 }
410
411 // BUG-AV-1 fix: 超深度时不再递归布局子树
412 if (depth > MAX_AST_DEPTH) {
413 node->finalX = 0;
414 return;
415 }
416
417 // 先递归布局每个子树（后序）
418 for (RtNode* child : node->children) {
419 layoutSubtree(child, depth + 1);
420 buildParentContour(child); // 子树布局完成后构建其轮廓
421 }
422
423 // 无子节点：finalX=0（自身即子树中心），轮廓由 buildParentContour 填充
424 if (node->children.empty()) {
425 node->finalX = 0;
426 return;
427 }
428
429 // 第一个子节点放在 x=0（相对父中心）
430 // 后续子节点依次根据前一个子树的右轮廓与自身的左轮廓计算平移量
431 double currentRightEdge = 0; // 已布局部分的右边界（相对父中心）
432 double firstChildFinalX = 0; // 记录第一个子节点的 finalX
433 double lastChildFinalX = 0; // 记录最后一个子节点的 finalX
434 for (size_t i = 0; i < node->children.size(); ++i) {
435 RtNode* child = node->children[i];
436 if (i == 0) {
437 // 第一个子节点：finalX 即相对父中心的偏移，初始为 0
438 // 但需考虑子树自身宽度：让其中心对齐父中心
439 child->finalX = 0;
440 firstChildFinalX = child->finalX;
441 // 更新 currentRightEdge 为该子树右轮廓的最大值
442 if (!child->rightContour.empty()) {
443 currentRightEdge = *std::max_element(child->rightContour.begin(), child->rightContour.end());
444 }
445 } else {
446 // 计算与上一个子树的平移量
447 RtNode* prevChild = node->children[i - 1];
448 double shift = computeShift(prevChild, child);
449 // child 的 finalX = prevChild.finalX + shift（相对父中心）
450 // 但 computeShift 返回的是 child 需要相对自身当前 finalX 的额外平移
451 // 当前 child.finalX=0，故直接设为 prevChild.finalX + shift
452 child->finalX = prevChild->finalX + shift;
453 // Bug-9 fix: 移除 shiftSubtree(child, 0) no-op 调用。offset=0 时函数
454 // 因 `if (offset != 0)` 守卫提前返回，是纯无效调用。finalX 已直接赋值，
455 // 轮廓在 buildContour 阶段基于 child 中心计算，无需此平移。
456 // BUG-AV-5 fix: shiftSubtree 已作为死代码删除。
457
458 // 更新 currentRightEdge
459 double childRightMax = 0;
460 if (!child->rightContour.empty()) {
461 childRightMax = *std::max_element(child->rightContour.begin(), child->rightContour.end());
462 }
463 // child 右轮廓相对 child 中心，需加上 child.finalX 转换到父坐标系
464 double childAbsoluteRight = child->finalX + childRightMax;
465 if (childAbsoluteRight > currentRightEdge) {
466 currentRightEdge = childAbsoluteRight;
467 }
468 }
469 lastChildFinalX = child->finalX;
470 }
471
472 // Bug1 fix: 父节点位置应在所有子节点的中间。
473 // 当前子节点布局基于第一个子节点在 x=0，最后一个子节点在 lastChildFinalX。
474 // 父节点应居中于 [firstChildFinalX, lastChildFinalX] 的中点。
475 // 通过平移所有子节点使中点对齐到 x=0（父中心），父节点的 finalX 保持 0 即可
476 // 处于子节点群的中间位置。
477 if (!node->children.empty()) {
478 double childrenMidX = (firstChildFinalX + lastChildFinalX) / 2.0;
479 // 平移所有子节点使中点对齐到父中心（x=0）
480 if (childrenMidX != 0) {
481 for (RtNode* child : node->children) {
482 child->finalX -= childrenMidX;
483 // 同步更新子树轮廓（轮廓相对子节点中心，平移子节点不影响轮廓相对值，
484 // 但 buildParentContour 会用 child.finalX 转换到父坐标系，故无需改轮廓）
485 }
486 }
487 }
488
489 (void)currentRightEdge; // 父轮廓由 buildParentContour 统一构建
490}

492double AstViewer::computeShift(const RtNode* leftNode, const RtNode* rightNode) const {
493 // 比较 leftNode 的右轮廓 vs rightNode 的左轮廓（逐层）
494 // 所需最小间距 = max(右轮廓[i] - 左轮廓[i]) + SIBLING_SPACING
495 // 注意：轮廓相对各自节点中心，比较时需在同一坐标系。
496 // 这里 leftNode 的右轮廓相对 leftNode 中心，rightNode 的左轮廓相对 rightNode 中心。
497 // 平移 rightNode 使两者间距满足 SIBLING_SPACING。
498 // 假设 leftNode 中心在原点，rightNode 中心平移到 x，
499 // 则层 i 的间距 = (x + rightNode.leftContour[i]) - leftNode.rightContour[i]
500 // 需满足 >= SIBLING_SPACING → x >= leftNode.rightContour[i] - rightNode.leftContour[i] + SIBLING_SPACING
501 // 取所有层最大值。
502
503 const auto& lc = leftNode->rightContour;
504 const auto& rc = rightNode->leftContour;
505
506 // Bug fix: 不等深子树轮廓延伸。
507 // 原实现用 commonDepth = min(lc.size(), rc.size()) 只比较到浅侧长度，
508 // 忽略深层比较。当左子树比右子树深时，左子树深层右轮廓可能比层 0 还靠右，
509 // 但右子树（浅）只在层 0 被推开，左子树深层穿透右子树位置 → 叶子重合。
510 //
511 // 修复（经典 Reingold-Tilford 做法）：遍历 maxDepth 所有层，
512 // 浅侧轮廓用最后一个已知值延伸——等价于浅子树在深层仍占自身最后一层的宽度。
513 // 这是保守做法（可能比必要间距略大），但保证不重合。
514 auto getOrExtend = [](const std::vector<double>& contour, size_t i) -> double {
515 if (i < contour.size())
516 return contour[i];
517 if (contour.empty())
518 return 0.0;
519 return contour.back();
520 };
521
522 double minShift = 0;
523 size_t maxDepth = std::max(lc.size(), rc.size());
524 for (size_t i = 0; i < maxDepth; ++i) {
525 double lcVal = getOrExtend(lc, i);
526 double rcVal = getOrExtend(rc, i);
527 double required = lcVal - rcVal + SIBLING_SPACING;
528 if (required > minShift) {
529 minShift = required;
530 }
531 }
532 return minShift;
533}

535void AstViewer::buildParentContour(RtNode* node) {
536 // 父节点轮廓：
537 // - 层 0（自身）：leftContour[0] = -NODE_WIDTH/2, rightContour[0] = NODE_WIDTH/2
538 // - 层 i+1（子树层 i）：取所有子树轮廓层 i 的最小/最大值
539 // 子树轮廓相对子节点中心，需加上 child.finalX 转换到父坐标系
540 node->leftContour.clear();
541 node->rightContour.clear();
542
543 // 层 0：自身节点
544 node->leftContour.push_back(-NODE_WIDTH / 2.0);
545 node->rightContour.push_back(NODE_WIDTH / 2.0);
546
547 // 折叠节点：轮廓仅含自身（无子树参与）
548 if (node->collapsed) {
549 return;
550 }
551
552 // 计算所有子树的最大深度（决定父轮廓层数）
553 size_t maxChildDepth = 0;
554 for (const RtNode* child : node->children) {
555 if (child->leftContour.size() > maxChildDepth) {
556 maxChildDepth = child->leftContour.size();
557 }
558 }
559
560 // 逐层合并子树轮廓
561 for (size_t layer = 0; layer < maxChildDepth; ++layer) {
562 double layerMin = std::numeric_limits<double>::max();
563 double layerMax = std::numeric_limits<double>::lowest();
564 bool hasValue = false;
565 for (const RtNode* child : node->children) {
566 if (layer < child->leftContour.size()) {
567 double left = child->finalX + child->leftContour[layer];
568 double right = child->finalX + child->rightContour[layer];
569 if (left < layerMin)
570 layerMin = left;
571 if (right > layerMax)
572 layerMax = right;
573 hasValue = true;
574 }
575 }
576 if (hasValue) {
577 node->leftContour.push_back(layerMin);
578 node->rightContour.push_back(layerMax);
579 } else {
580 // 无子树覆盖此层（理论上不应发生，因 maxChildDepth 取自子树）
581 node->leftContour.push_back(0);
582 node->rightContour.push_back(0);
583 }
584 }
585}

587void AstViewer::computeAbsoluteCoords(RtNode* node, double parentAbsX, int depth,
588 std::vector<std::pair<double, double>>& bounds) {
589 if (!node)
590 return;
591 // 绝对 x = 父绝对 x + 本节点相对父的 finalX
592 double absX = parentAbsX + node->finalX;
593 node->finalX = absX;
594 node->finalY = depth * (NODE_HEIGHT + LEVEL_SPACING);
595
596 // 记录边界（节点矩形 ± NODE_WIDTH/2）
597 double left = absX - NODE_WIDTH / 2.0;
598 double right = absX + NODE_WIDTH / 2.0;
599 if (static_cast<size_t>(depth) >= bounds.size()) {
600 bounds.resize(depth + 1, {std::numeric_limits<double>::max(), std::numeric_limits<double>::lowest()});
601 }
602 if (left < bounds[depth].first)
603 bounds[depth].first = left;
604 if (right > bounds[depth].second)
605 bounds[depth].second = right;
606
607 // 折叠节点：不递归子节点
608 if (node->collapsed)
609 return;
610
611 // BUG-AV-1 fix: 超深度时不再递归子节点，避免栈溢出
612 if (depth > MAX_AST_DEPTH)
613 return;
614
615 for (RtNode* child : node->children) {
616 computeAbsoluteCoords(child, absX, depth + 1, bounds);
617 }
618}

620void AstViewer::drawRtNode(RtNode* node, double offsetX, int depth) {
621 if (!node || !node->astNode)
622 return;
623
624 // G-P2-1 fix: 缓存 nodeName，避免重复调用
625 const std::string nodeNameStr = node->astNode->nodeName();
626 const QString name = QString::fromStdString(nodeNameStr);
627
628 // 节点中心坐标 = (finalX + offsetX, finalY)，矩形左上角 = 中心 - (W/2, 0)
629 double centerX = node->finalX + offsetX;
630 double centerY = node->finalY;
631 double nodeX = centerX - NODE_WIDTH / 2.0;
632 double nodeY = centerY;
633
634 // 根据节点类型 + 主题选择颜色
635 QColor bgColor = nodeBgColor(name);
636 QColor borderColor = nodeBorderColor();
637 QColor txtColor = textColor();
638
639 QRectF rect(nodeX, nodeY, NODE_WIDTH, NODE_HEIGHT);
640 QGraphicsRectItem* rectItem = scene_->addRect(rect, QPen(borderColor, 1.5), QBrush(bgColor));
641 rectItem->setZValue(1);
642 // 注册到 itemToNode_ 以支持点击命中检测
643 itemToNode_[rectItem] = node;
644 // 设置 tooltip：完整节点名 + 行:列 + 子节点数
645 QString tooltip = QString::fromUtf8("节点: %1\n位置: %2:%3\n子节点: %4")
646 .arg(name)
647 .arg(node->astNode->line)
648 .arg(node->astNode->column)
649 .arg(static_cast<int>(node->children.size()));
650 rectItem->setToolTip(tooltip);
651
652 // 绘制文本（折叠节点附加 [+N] 指示）
653 QString displayText = name;
654 if (displayText.length() > 14) {
655 displayText = displayText.left(12) + "..";
656 }
657 if (node->collapsed && !node->children.empty()) {
658 displayText += QString(" [+%1]").arg(node->children.size());
659 }
660 QGraphicsTextItem* textItem = scene_->addText(displayText);
661 textItem->setPos(nodeX + 4, nodeY + 8);
662 // Dedup-4A: monospaceFont 共享缓存（原 static const QFont astFont("Consolas", 8)）
663 textItem->setFont(GuiTextUtils::monospaceFont(8));
664 textItem->setDefaultTextColor(txtColor);
665 textItem->setZValue(2);
666 // 文本也参与 tooltip（覆盖在 rect 上方时鼠标在 textItem 上）
667 textItem->setToolTip(tooltip);
668 // BUG-AV-3 fix: textItem 未注册到 itemToNode_，点击文字无法触发折叠。
669 // 让 textItem 不可选中、不接收鼠标按钮，使点击事件穿透到下层的 rectItem，
670 // 由 rectItem 的命中检测处理折叠/展开。
671 textItem->setFlag(QGraphicsItem::ItemIsSelectable, false);
672 textItem->setAcceptedMouseButtons(Qt::NoButton);
673
674 // 折叠节点：不绘制子树连线与子节点
675 if (node->collapsed)
676 return;
677
678 // BUG-AV-1 fix: 超深度时不再递归绘制子节点
679 if (depth > MAX_AST_DEPTH)
680 return;
681
682 // 父节点底部中心
683 double parentCenterX = centerX;
684 double parentBottomY = nodeY + NODE_HEIGHT;
685
686 // 递归绘制子节点 + 平滑连线（第八轮：三次贝塞尔曲线替代直线）
687 for (RtNode* child : node->children) {
688 if (!child || !child->astNode)
689 continue;
690 double childCenterX = child->finalX + offsetX;
691 double childTopY = child->finalY;
692
693 // 三次贝塞尔：控制点位于父子垂直中点，形成 S 形平滑过渡
694 double midY = (parentBottomY + childTopY) / 2.0;
695 QPainterPath path;
696 path.moveTo(parentCenterX, parentBottomY);
697 path.cubicTo(parentCenterX, midY, childCenterX, midY, childCenterX, childTopY);
698 auto* lineItem = scene_->addPath(path, QPen(lineColor(), 1.5));
699 lineItem->setZValue(0);
700
701 drawRtNode(child, offsetX, depth + 1);
702 }
