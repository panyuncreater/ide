14static void collectVarRefs(ASTNode* node, std::unordered_set<std::string>& names) {
15 if (!node)
16 return;
17 switch (node->nodeType) {
18 case NodeType::NODE_VAR_REF:
19 names.insert(static_cast<VarRef*>(node)->name);
20 break;
21 case NodeType::NODE_BINARY_OP: {
22 auto* bin = static_cast<BinaryOp*>(node);
23 collectVarRefs(bin->left.get(), names);
24 collectVarRefs(bin->right.get(), names);
25 break;
26 }
27 case NodeType::NODE_UNARY_OP:
28 collectVarRefs(static_cast<UnaryOp*>(node)->operand.get(), names);
29 break;
30 case NodeType::NODE_FUN_CALL: {
31 auto* call = static_cast<FunCall*>(node);
32 for (auto& arg : call->arguments) {
33 collectVarRefs(arg.get(), names);
34 }
35 break;
36 }
37 case NodeType::NODE_INDEX_ACCESS: {
38 auto* idx = static_cast<IndexAccess*>(node);
39 collectVarRefs(idx->object.get(), names);
40 collectVarRefs(idx->index.get(), names);
41 break;
42 }
43 case NodeType::NODE_MEMBER_ACCESS:
44 collectVarRefs(static_cast<MemberAccess*>(node)->object.get(), names);
45 break;
46 case NodeType::NODE_METHOD_CALL: {
47 auto* mc = static_cast<MethodCall*>(node);
48 collectVarRefs(mc->object.get(), names);
49 for (auto& arg : mc->arguments) {
50 collectVarRefs(arg.get(), names);
51 }
52 break;
53 }
54 case NodeType::NODE_ARRAY_LITERAL: {
55 auto* arr = static_cast<ArrayLiteral*>(node);
56 for (auto& elem : arr->elements) {
57 collectVarRefs(elem.get(), names);
58 }
59 break;
60 }
61 case NodeType::NODE_DICT_LITERAL: {
62 auto* dict = static_cast<DictLiteral*>(node);
63 for (auto& pair : dict->pairs) {
64 collectVarRefs(pair.first.get(), names);
65 collectVarRefs(pair.second.get(), names);
66 }
67 break;
68 }

133std::unique_ptr<Block> Parser::parse(const std::vector<Token>& tokens) {
134 tokens_ = &tokens; // 存储指针，避免深拷贝整个 token 流
135 current_ = 0;
136 parseDepth_ = 0; // P15 fix: 重置递归深度
137 blockDepth_ = 0; // P0-1 fix: 重置块嵌套深度
138 diagnostics_.clear();
139 knownEnums_.clear(); // R99: 重置已知 enum 名集合，防止 Parser 复用导致状态泄漏
140
141 // PERF-21 fix: 预分配 statements 向量容量，避免每个 declaration push_back 触发 realloc。
142 // 估算：平均每 4 个 token 产生 1 个声明（关键字 + 名字 + ... + 分号）
143 std::vector<std::shared_ptr<ASTNode>> statements;
144 statements.reserve(tokens_->size() / 4);
145
146 while (!isAtEnd()) {
147 // BUG-004 fix: 顶层出现 '}' 是非法 token（无 enclosing block）。
148 // 原实现 break 会丢弃后续有效声明。改为：消耗 stray '}' + 诊断，
149 // 然后继续解析后续声明。注意 synchronize() 仍在 '}' 处返回（不消费），
150 // 但本循环已先消耗 '}'，故不会无限循环。
151 // 安全性：nested block() 中的 '}' 不会被消耗到此处——block() 内部循环
152 // 同样 check(RBRACE) 退出并由 consume(TK_RBRACE) 消费闭合花括号，
153 // 不会回退到顶层 parse()。
154 if (check(TokenType::TK_RBRACE)) {
155 const Token& rbrace = peek();
156 diagnostics_.addError("多余的 '}' 在顶层（无匹配的 '{'）", rbrace.line, rbrace.column, DiagSource::Parser,
157 DiagCodes::kStrayBrace);
158 advance(); // 消耗 stray '}'，避免 declaration() → primary() 不识别 → 死循环
159 continue;
160 }
161 // BUG-PARSER-AUDIT-5 fix: 错误数量上限，防止恶意输入触发 O(N) 诊断内存膨胀。
162 // 原实现无上限，100 万个 ';' 可累积 ~100 万条 Diagnostic（~200MB）。
163 // L7 fix: 改为读取 RuntimeConfig 运行时配置（教学场景可调）
164 const int dynMaxParseErrors = RuntimeLimits::RuntimeConfig::instance().maxParseErrors();
165 if (diagnostics_.errorCount() >= dynMaxParseErrors) {
166 diagnostics_.addError("错误过多（超过 " + std::to_string(dynMaxParseErrors) + " 条），停止解析",
167 peek().line, peek().column, DiagSource::Parser, DiagCodes::kTooManyErrors);
168 break;
169 }
170 try {
171 auto decl = declaration();
172 if (decl) {
173 statements.push_back(std::move(decl));
174 }
175 } catch (const ParseError& e) {
176 // 收集错误到诊断包而非吞掉
177 diagnostics_.addError(e.what(), e.line, e.column, DiagSource::Parser, e.code);
178 // 错误恢复：同步到下一个声明边界
179 synchronize();
180 }
181 }
182
183 return std::make_unique<Block>(std::move(statements), 1, 1);
184}

188const Token& Parser::peek() const {
189 // AUDIT-P2 fix: 与 previous() 的 P0-14 fix 边界检查保持对称。
190 // 原实现直接索引 tokens_[current_]，依赖"tokens_ 末尾必有 TK_EOF"的隐式不变量。
191 // 正常路径下 Lexer::scan 总是追加 TK_EOF，但外部构造的 tokens_（测试夹具/API）
192 // 或内部 bug 可能导致 current_ 越界。添加边界检查返回 EOF 哨兵，防御性编程。
193 if (current_ < 0 || current_ >= static_cast<int>(tokens_->size())) {
194 static const Token eofSentinel(TokenType::TK_EOF, "", std::monostate{}, 0, 0);
195 return eofSentinel;
196 }
197 return (*tokens_)[current_];

239const Token& Parser::consume(TokenType type, const std::string& message, const std::string& diagCode) {
240 if (check(type))
241 return advance();
242 const Token& tok = peek();
243 // BUG-PARSER-MSG-1 fix (P2): 错误消息拼接实际得到的 token，与 primary() 风格一致。
244 // 原实现仅输出调用方传入的 message（如"期望 ';'"），用户不知道下一个 token 是什么，
245 // 难以判断问题位置。改为"期望 ';' 但得到 'var'"，提升诊断价值。
246 // P2 fix (错误码优先匹配): 调用方未显式提供 diagCode 时，按 token 类型派生默认
247 // 稳定诊断码（分号→missing-semicolon / 圆括号→unbalanced-paren / 花括号→unbalanced-brace），
248 // 覆盖全部 consume 调用点，无需逐处手填。显式传入的 diagCode 优先。
249 std::string code = diagCode;
250 if (code.empty()) {
251 switch (type) {
252 case TokenType::TK_SEMICOLON:
253 code = "missing-semicolon";
254 break;
255 case TokenType::TK_LPAREN:
256 case TokenType::TK_RPAREN:
257 code = "unbalanced-paren";
258 break;
259 case TokenType::TK_LBRACE:
260 case TokenType::TK_RBRACE:
261 code = "unbalanced-brace";
262 break;
263 default:
264 break;
265 }
266 }
267 throw ParseError(message + " 但得到 '" + tok.lexeme + "'", tok.line, tok.column, code);

304 return false;

336std::string Parser::parseTypeAnnotation() {
337 // Bug #10 fix: 递归深度保护（dict[K:V] / fun(params):ret 递归调用自身）
338 if (parseDepth_ >= MAX_PARSE_DEPTH) {
339 throw ParseError("类型注解嵌套过深（超过 " + std::to_string(MAX_PARSE_DEPTH) + " 层）", peek().line,
340 peek().column);
341 }
342 DepthGuard guard{parseDepth_};
343
344 // AUDIT-P2.7 fix: 支持 T? 可选类型、dict[K:V] 泛型字典、fun(params):ret 函数类型。
345 // Interpreter.typeMatch 已支持这三种格式的运行时类型检查，此处仅需生成对应字符串。
346 const Token& typeTok = advance(); // 消耗类型关键字或标识符
347 std::string typeAnn = typeTok.lexeme;
348
349 // AUDIT-P2.7 fix: dict[K:V] 泛型字典类型 — 当基础类型为 dict 且紧跟 '[' 时，
350 // 解析键类型 ':' 值类型 ']'，生成 "dict[keyType:valType]" 格式字符串。
351 // 递归调用 parseTypeAnnotation 解析键/值类型，天然支持嵌套（如 dict[string:int?]）。
352 // 消歧：仅当 '[' 后不是 ']' 时才进入泛型分支；dict[] 仍按数组后缀处理。
353 if (typeTok.type == TokenType::TK_DICT && check(TokenType::TK_LBRACKET) && !checkNext(TokenType::TK_RBRACKET)) {
354 advance(); // 消耗 '['
355 std::string keyType = parseTypeAnnotation();
356 consume(TokenType::TK_COLON, "期望 ':' 分隔字典的键类型与值类型");
357 std::string valType = parseTypeAnnotation();
358 consume(TokenType::TK_RBRACKET, "期望 ']' 结束字典类型注解");
359 typeAnn = "dict[" + keyType + ":" + valType + "]";
360 }
361 // AUDIT-P2.7 fix: fun(params):ret 函数类型 — 当基础类型为 fun 且紧跟 '(' 时，
362 // 解析参数类型列表 ')' ':' 返回类型，生成 "fun(paramTypes):retType" 格式字符串。
363 // 递归调用 parseTypeAnnotation 解析每个参数类型和返回类型。
364 else if (typeTok.type == TokenType::TK_FUN && check(TokenType::TK_LPAREN)) {
365 advance(); // 消耗 '('
366 std::string paramList;
367 if (!check(TokenType::TK_RPAREN)) {
368 do {
369 std::string pType = parseTypeAnnotation();
370 if (!paramList.empty())
371 paramList += ",";
372 paramList += pType;
373 } while (match(TokenType::TK_COMMA) && !check(TokenType::TK_RPAREN));
374 }
375 consume(TokenType::TK_RPAREN, "期望 ')' 结束函数类型参数列表");
376 consume(TokenType::TK_COLON, "期望 ':' 分隔函数参数列表与返回类型");
377 std::string retType = parseTypeAnnotation();
378 typeAnn = "fun(" + paramList + "):" + retType;
379 }
380
381 // AUDIT-P2-CORRECT fix: 支持多维数组类型注解（int[][], int[][][]）。
382 // AUDIT-P2.7 fix: 支持 [] 和 ? 后缀的任意顺序堆叠（如 int[]?, int?[], dict[string:int][]?）。
383 // 循环消费连续的 [] 或 ? 后缀，直到两者都不再出现。
384 // [] 后缀：仅在 [ 后紧跟 ] 时才消费，否则回退 [（保留原安全回溯语义）。
385 // ? 后缀：Lexer 将 '?' 映射到已废弃的 TK_FUNC 槽位（见 Lexer.cpp scanToken）。
386 while (true) {
387 if (check(TokenType::TK_LBRACKET)) {
388 int bracketSave = current_;
389 advance(); // 消耗 '['
390 if (check(TokenType::TK_RBRACKET)) {
391 advance(); // 消耗 ']'
392 typeAnn += "[]";
393 } else {
394 current_ = bracketSave; // 不是 [] 类型注解，回退 '['
395 break;
396 }
397 } else if (check(TokenType::TK_FUNC)) {
398 // AUDIT-P2.7 fix: T? 可选类型后缀。Interpreter.typeMatch 通过
399 // annotation.back()=='?' 识别可选类型，剥离后递归匹配。
400 advance(); // 消耗 '?'
401 typeAnn += "?";
402 } else {
403 break;
404 }
405 }
406 return typeAnn;
407}

426std::unique_ptr<ASTNode> Parser::declaration() {
427 // var 声明
428 if (check(TokenType::TK_VAR))
429 return varDecl();
430
431 // fun / function / func 声明（Lexer 已统一为 TK_FUN）
432 // R98 W3: `fun name(...)` 是函数声明，`fun(...)` 是 lambda 表达式。
433 // 通过 peekNext 区分：fun 后跟标识符 → funDecl；fun 后跟 `(` → lambda 表达式
434 // （走 statement → expressionStatement → expression → primary → lambdaExpr）。
435 // lambda 作为语句无意义（闭包值未被使用），但语法允许——用户可能写
436 // `fun(x){...}(5);` 立即调用，此时走 expressionStatement 路径。
437 if (check(TokenType::TK_FUN) && !checkNext(TokenType::TK_LPAREN))
438 return funDecl();
439
440 // L18 lang-constfun: const fun name(...) —— 编译期求值函数声明。
441 // const 仅允许修饰具名 fun 声明（非 lambda/非其它语句）。
442 if (check(TokenType::TK_CONST)) {
443 const Token& constTok = peek();
444 advance(); // 消耗 const
445 if (check(TokenType::TK_FUN) && !checkNext(TokenType::TK_LPAREN)) {
446 auto decl = funDecl();
447 if (decl && decl->nodeType == NodeType::NODE_FUN_DECL) {
448 static_cast<FunDecl*>(decl.get())->isConstFun = true;
449 }
450 return decl;
451 }
452 throw ParseError("const 仅可修饰具名函数声明（const fun name(...)）", constTok.line, constTok.column);
453 }
454
455 // class 声明
456 if (check(TokenType::TK_CLASS))
457 return classDecl();
458
459 // R99 枚举与 ADT: enum 声明
460 if (check(TokenType::TK_ENUM))
461 return enumDecl();
462
463 // 七特性 MVP 阶段 2：宏声明
464 if (check(TokenType::TK_MACRO))
465 return macroDecl();
466
467 // 七特性 MVP 阶段 3：trait 声明
468 if (check(TokenType::TK_TRAIT))
469 return traitDecl();
470
471 // 七特性 MVP 阶段 4：async fun 声明（async 仅可修饰具名 fun 声明）
472 if (check(TokenType::TK_ASYNC)) {
473 const Token& asyncTok = peek();
474 advance(); // 消耗 async
475 if (check(TokenType::TK_FUN) && !checkNext(TokenType::TK_LPAREN)) {
476 return funDecl(/*isAsync=*/true);
477 }
478 throw ParseError("async 仅可修饰具名函数声明（async fun name(...)）", asyncTok.line, asyncTok.column);
479 }
480
481 // F12: import / export 声明
482 if (check(TokenType::TK_IMPORT))
483 return importStmt();
484 if (check(TokenType::TK_EXPORT))
485 return exportStmt();
486
487 // 类型注解声明: int/float/bool/string/dict/array
488 if (isTypeKeyword()) {
489 // 可能是: 类型注解变量声明(int a=1;) 或 带返回类型的函数声明(int fib(n){})
490 // 保存当前位置以便回溯
491 int savePos = current_;
492 std::string typeAnn = parseTypeAnnotation();
493
494 if (isIdentifierOrType()) {
495 // 检查是否是带返回类型的函数声明: int fib(
496 if (checkNext(TokenType::TK_LPAREN)) {
497 return typedFunDecl(typeAnn);
498 }
499 // 普通类型注解变量声明: int a = 10;
500 if (check(TokenType::TK_IDENTIFIER)) {
501 return typedVarDecl(typeAnn);
502 }
503 // 类型关键字作函数名但后面没有( — 无法构成有效声明，回溯
504 }
505
506 // 不是类型注解，回溯
507 current_ = savePos;
508 }
509
510 // 类名类型注解声明: ClassName varName; 或 ClassName funcName() { ... }
511 // 识别模式: 标识符 标识符 (类型名 变量名/函数名)
512 if (check(TokenType::TK_IDENTIFIER)) {
513 // 预读两个 token: 第一个是类名，第二个是变量名/函数名
514 int savePos = current_;
515 std::string typeAnn = parseTypeAnnotation(); // 消耗类名 + 可选 []
516
517 if (check(TokenType::TK_IDENTIFIER)) {
518 // M7 fix: 前瞻验证 — 检查第二个标识符后是否为合法声明后续 token
519 int afterSecondPos = current_ + 1; // 跳过第二个标识符
520 // 跳过可能的 [] 数组类型后缀
521 if (afterSecondPos < static_cast<int>(tokens_->size()) &&
522 (*tokens_)[afterSecondPos].type == TokenType::TK_LBRACKET &&
523 afterSecondPos + 1 < static_cast<int>(tokens_->size()) &&
524 (*tokens_)[afterSecondPos + 1].type == TokenType::TK_RBRACKET) {
525 afterSecondPos += 2;
526 }
527 bool validDeclFollow = false;
528 if (afterSecondPos < static_cast<int>(tokens_->size())) {
529 TokenType follow = (*tokens_)[afterSecondPos].type;
530 // 合法后续: '=' 初始化, ';' 结束, '(' 函数参数, '{' 函数体
531 validDeclFollow = (follow == TokenType::TK_ASSIGN || follow == TokenType::TK_SEMICOLON ||
532 follow == TokenType::TK_LPAREN || follow == TokenType::TK_LBRACE);
533 }
534
535 if (validDeclFollow) {
536 // 检查是否是带类类型的函数声明: ClassName funcName(
537 if (checkNext(TokenType::TK_LPAREN)) {
538 return typedFunDecl(typeAnn);
539 }
540 // ClassName varName — 类类型注解变量声明
541 return typedVarDecl(typeAnn);
542 }
543 }
544
545 // 不是类类型声明，回溯
546 current_ = savePos;
547 }
548
549 return statement();
550}

552std::unique_ptr<ASTNode> Parser::varDecl() {
553 const Token& varTok = consume(TokenType::TK_VAR, "期望 'var'");
554
555 // R98 元组与解构：var (a, b, c) = expr 解构绑定
556 // 检测 var 后紧跟 '(' 的模式——避免与 var (expr) 的合法语法歧义，
557 // 解构绑定要求 '(' 后必须紧跟标识符（不能是表达式），且至少一个 ',' 分隔。
558 // L20: 支持每个变量名后可选 `: Type` 注解，前瞻需跳过 ": <Type>" 序列。
559 if (check(TokenType::TK_LPAREN)) {
560 // 前瞻：'(' IDENTIFIER (':' <TypeTokens>)? (',' IDENTIFIER (':' <TypeTokens>)?)* ')' '='
561 // 注：不强制要求逗号（var (a) = expr 也视为解构，与原实现一致）
562 int lookahead = current_ + 1;
563 bool isDestructure = false;
564 if (lookahead < static_cast<int>(tokens_->size()) && (*tokens_)[lookahead].type == TokenType::TK_IDENTIFIER) {
565 int scanPos = lookahead + 1;
566 while (scanPos < static_cast<int>(tokens_->size())) {
567 // L20: 跳过可选 ": <TypeTokens>"
568 if ((*tokens_)[scanPos].type == TokenType::TK_COLON) {
569 ++scanPos; // 消耗 ':'
570 // 跳过类型 token 序列：标识符/类型关键字/enum/'['/']'/'?'/','/'('/')'/'{'/'}'（dict[K:V],
571 // fun():ret） 简化：跳过到下一个 ',' 或 ')'，不严格解析类型语法（实际解析在 parseTypeAnnotation）
572 while (scanPos < static_cast<int>(tokens_->size())) {
573 TokenType tt = (*tokens_)[scanPos].type;
574 if (tt == TokenType::TK_COMMA || tt == TokenType::TK_RPAREN)
575 break;
576 ++scanPos;
577 }
578 if (scanPos >= static_cast<int>(tokens_->size()))
579 break;
580 }
581 if ((*tokens_)[scanPos].type == TokenType::TK_COMMA) {
582 ++scanPos;
583 if (scanPos >= static_cast<int>(tokens_->size()) ||
584 (*tokens_)[scanPos].type != TokenType::TK_IDENTIFIER) {
585 break; // 不是合法解构模式
586 }
587 ++scanPos; // 消耗 IDENTIFIER
588 continue;
589 }
590 if ((*tokens_)[scanPos].type == TokenType::TK_RPAREN) {
591 ++scanPos;
592 // 注：原实现不强制要求逗号（var (a) = expr 也视为解构），此处保持兼容
593 if (scanPos < static_cast<int>(tokens_->size()) &&
594 (*tokens_)[scanPos].type == TokenType::TK_ASSIGN) {
595 isDestructure = true;
596 }
597 break;
598 }
599 break; // 不是合法解构模式
600 }
601 }
602
603 if (isDestructure) {
604 advance(); // 消耗 '('
605 std::vector<std::string> names;
606 std::vector<std::string> nameTypeAnns; // L20: per-name 类型注解
607 bool anyNameTypeAnn = false; // L20: 是否有任一位置带注解
608 // L20: 解析第一个变量名 + 可选 `: Type`
609 const Token& firstName = consume(TokenType::TK_IDENTIFIER, "期望变量名");
610 names.push_back(firstName.lexeme);
611 {
612 std::string ann = parseOptionalNameTypeAnnotation(firstName.line, firstName.column);
613 nameTypeAnns.push_back(ann);
614 if (!ann.empty())
615 anyNameTypeAnn = true;
616 }
617 while (match(TokenType::TK_COMMA)) {
618 const Token& nextName = consume(TokenType::TK_IDENTIFIER, "期望变量名");
619 names.push_back(nextName.lexeme);
620 std::string ann = parseOptionalNameTypeAnnotation(nextName.line, nextName.column);
621 nameTypeAnns.push_back(ann);
622 if (!ann.empty())
623 anyNameTypeAnn = true;
624 }
625 consume(TokenType::TK_RPAREN, "期望 ')' 结束解构绑定");
626 consume(TokenType::TK_ASSIGN, "期望 '=' 初始化解构绑定");
627 auto init = expression();
628 consume(TokenType::TK_SEMICOLON, "期望 ';' 结束解构绑定");
629 // L20: 若任一位置带类型注解，则 nameTypeAnnotations 全量保留（空串表示该位置无注解）；
630 // 若全部无注解，保留空 vector 以维持向后兼容（hasNameTypeAnnotations() == false）
631 auto node =
632 std::make_unique<DestructureBinding>(std::move(names), std::move(init), varTok.line, varTok.column);
633 if (anyNameTypeAnn) {
634 node->nameTypeAnnotations = std::move(nameTypeAnns);
635 }
636 return node;
637 }
638 }
639
640 const Token& name = consume(TokenType::TK_IDENTIFIER, "期望变量名");
641
642 // R99 fix: 支持 var name: Type = expr; 语法（var-first 带类型注解）
643 // 原 varDecl 只支持 var name = expr;（无注解），typedVarDecl 支持 Type name = expr;（类型在前）。
644 // 但 `var c: Color = Color.Red;` 和 `var c: enum = Color.Red;` 落入 varDecl 路径，
645 // 因 `:` 后非 `=`/`;` 触发 ParseError，导致变量未定义。
646 std::string typeAnn;
647 if (match(TokenType::TK_COLON)) {
648 // 允许标识符（类名/enum 名）、类型关键字、TK_ENUM（R99 泛型 enum 注解）
649 if (!isIdentifierOrType() && !check(TokenType::TK_ENUM)) {
650 const Token& tok = peek();
651 throw ParseError("期望类型名", tok.line, tok.column);
652 }
653 typeAnn = parseTypeAnnotation();
654 }
655
656 std::unique_ptr<ASTNode> init = nullptr;
657 if (match(TokenType::TK_ASSIGN)) {
658 init = expression();
659 }
660 consume(TokenType::TK_SEMICOLON, "期望 ';' 结束变量声明");
661
662 return std::make_unique<VarDecl>(StringIntern::intern(name.lexeme), typeAnn, std::move(init), varTok.line,
663 varTok.column);
664}

679std::unique_ptr<FunDecl> Parser::funDecl(bool isAsync) {
680 // 消耗 fun 或 function 关键字
681 const Token& funTok = advance();
682
683 // R164 协程/生成器：检测 fun* 标记生成器函数（类似 JS function* / Python def+yield）
684 // 七特性 MVP 阶段 4：async fun 复用生成器机制（调用返回 Task/协程）
685 bool isGenerator = isAsync;
686 if (match(TokenType::TK_STAR)) {
687 isGenerator = true;
688 }
689
690 // 函数名：允许标识符或类型关键字（如 dict, array, int, float, string, bool）
691 const Token& name = consumeIdentifierOrType("期望函数名");
692
693 // R163 泛型函数：可选类型参数列表 <T, E, ...>
694 // 与 enumDecl 的泛型解析模式一致：函数名后、'(' 前解析 <T, E>
695 std::vector<std::string> typeParams;
696 if (match(TokenType::TK_LT)) {
697 do {
698 if (!isIdentifierOrType()) {
699 const Token& tok = peek();
700 throw ParseError("期望类型参数名", tok.line, tok.column);
701 }
702 const Token& tp = advance();
703 typeParams.push_back(tp.lexeme);
704 } while (match(TokenType::TK_COMMA));
705 consume(TokenType::TK_GT, "期望 '>' 结束类型参数列表");
706 }
707
708 consume(TokenType::TK_LPAREN, "期望 '('");
709
710 std::vector<std::string> params;
711 std::vector<std::string> paramTypes;
712 std::vector<std::shared_ptr<ASTNode>> defaultValues; // F10
713 parseParamList(params, paramTypes, defaultValues);
714 consume(TokenType::TK_RPAREN, "期望 ')'");
715
716 // 可选的返回值类型注解 : type 或 -> type
717 std::string returnType;
718 if (match(TokenType::TK_COLON)) {
719 // BUG-PARSER-AUDIT-1 fix: 在 parseTypeAnnotation 前预检 token 类型，
720 // 防止 `fun foo(): ;` 或 `fun foo(): {}` 中的 `;`/`{` 被吞作类型名，
721 // 导致误导性错误消息与块结构污染。与 parseParamList 第 479-484 行模式对齐。
722 if (!isIdentifierOrType()) {
723 const Token& tok = peek();
724 throw ParseError("期望返回类型名", tok.line, tok.column);
725 }
726 returnType = parseTypeAnnotation();
727 } else if (check(TokenType::TK_MINUS) && checkNext(TokenType::TK_GT)) {
728 advance(); // 消耗 '-'
729 advance(); // 消耗 '>'
730 if (!isIdentifierOrType()) { // BUG-PARSER-AUDIT-1
731 const Token& tok = peek();
732 throw ParseError("期望返回类型名", tok.line, tok.column);
733 }
734 returnType = parseTypeAnnotation();
735 }
736
737 // R164 协程/生成器：进入生成器函数体前重置 yieldId 计数器。
738 // isGenerator=true 时重置为 0（开始计数），否则设为 -1（标记非生成器上下文，
739 // primary() 遇到 yield 时报错）。YieldIdScope 保证异常路径下恢复外层计数器，
740 // 支持嵌套生成器函数独立计数。
741 YieldIdScope yieldScope{currentYieldId_};
742 currentYieldId_ = isGenerator ? 0 : -1;
743 // R164 协程/生成器：进入函数体前重置 yield-in-loop 追踪标志。
744 // 嵌套函数（包括嵌套 fun*）有独立的追踪上下文：外层循环不影响内层函数的
745 // yield-in-loop 判定。BoolScope 保证异常路径下恢复外层值。
746 BoolScope loopCtxScope{yieldInLoop_};
747 yieldInLoop_ = false;
748 BoolScope hasLoopYieldScope{currentFunHasYieldInLoop_};
749 currentFunHasYieldInLoop_ = false;
750 // 七特性 MVP 阶段 4：async 上下文标记（await 仅在 async fun 体内合法）。
751 // 嵌套函数独立判定：async fun 内的普通嵌套 fun 体内 await 非法。
752 BoolScope asyncCtxScope{currentFunIsAsync_};
753 currentFunIsAsync_ = isAsync;
754
755 consume(TokenType::TK_LBRACE, "期望 '{'");
756 auto body = block();
757
758 auto decl = std::make_unique<FunDecl>(StringIntern::intern(name.lexeme), std::move(params), std::move(paramTypes),
759 returnType, std::move(body), funTok.line, funTok.column);
760 decl->typeParams = std::move(typeParams);
761 // R164 协程/生成器：标记生成器函数（fun*），供 Interpreter/VM 在调用时
762 // 返回 Coroutine 值而非直接执行函数体。
763 decl->isGenerator = isGenerator;
764 // 七特性 MVP 阶段 4：async fun 标记（Formatter 区分打印 async fun vs fun*）
765 decl->isAsync = isAsync;
766 // 记录生成器内 yield 总数（供 Interpreter 重放模式判断是否已耗尽）。
767 if (isGenerator) {
768 if (isAsync || currentFunHasYieldInLoop_) {
769 // 循环内 yield：编译期节点数 ≠ 运行时执行数，用 INT_MAX 标记动态模式，
770 // done 改由函数体自然结束路径判定（避免 currentYieldId >= yieldCount 误判）。
771 // async fun 恒用动态模式：body 可能无 yield（首次 next() 直接跑完）、
772 // 也可能含 await/yield 混合，静态计数不可靠。
773 decl->yieldCount = FunDecl::kDynamicYieldCount;
774 } else {
775 decl->yieldCount = currentYieldId_;
776 }
777 }
778 // F10: 计算必需参数个数（前缀无默认值的参数数量）
779 int reqCount = 0;
780 for (size_t i = 0; i < defaultValues.size(); ++i) {
781 if (defaultValues[i] == nullptr) {
782 ++reqCount;
783 } else {
784 break; // 一旦遇到默认值，后续都有默认值
785 }
786 }
787 decl->requiredParamCount = reqCount;
788 decl->defaultValues = std::move(defaultValues);
789 return decl;
790}

869 funTok.line, funTok.column);
870 // F10: 计算必需参数个数
871 int reqCount = 0;
872 for (size_t i = 0; i < defaultValues.size(); ++i) {
873 if (defaultValues[i] == nullptr) {
874 ++reqCount;
875 } else {
876 break;
877 }
878 }

880 decl->defaultValues = std::move(defaultValues);

882}

884void Parser::parseParamList(std::vector<std::string>& params, std::vector<std::string>& paramTypes,
885 std::vector<std::shared_ptr<ASTNode>>& defaultValues) {
886 // BUG-DEF-1 fix: 默认参数值必须是字面量（数字/字符串/布尔/null/负数字面量）。
887 // 三后端一致性：Interpreter 支持任意表达式默认值（b=a+1 在闭包环境求值），
888 // 但 StackVM/RegisterVM 仅支持字面量（复杂表达式记录 0xFFFF 哨兵运行时报错）。
889 // 在 Parser 层统一拒绝复杂表达式，确保三后端行为一致。
890 // 支持的类型：NumberLiteral / StringLiteral / BoolLiteral / NullLiteral /
891 // UnaryOp(NEGATE) 嵌套包装 NumberLiteral（如 -42, --5）/
892 // InterpolatedString（字面量片段+表达式，表达式部分由 collectVarRefs 检查前向引用；
893 // VM 路径记录 0xFFFF 哨兵，运行时调用使用默认值会报错，但 Parser 层接受以保持
894 // BUG-LPA-03 回归测试语义——该测试验证 collectVarRefs 覆盖 InterpolatedString）。
895 auto isLiteralDefaultExpr = [](const ASTNode* node) -> bool {
896 if (!node)
897 return false;
898 switch (node->nodeType) {
899 case NodeType::NODE_NUMBER_LITERAL:
900 case NodeType::NODE_STRING_LITERAL:
901 case NodeType::NODE_BOOL_LITERAL:
902 case NodeType::NODE_NULL_LITERAL:
903 case NodeType::NODE_INTERPOLATED_STRING:
904 return true;
905 case NodeType::NODE_UNARY_OP: {
906 // 支持负数字面量: -42, -3.14, --5（双重否定）
907 const auto* unary = static_cast<const UnaryOp*>(node);
908 if (unary->opType != UnaryOp::UnaryOpType::UOP_NEGATE)
909 return false;
910 const ASTNode* cur = unary->operand.get();
911 int negateCount = 1;
912 while (cur && cur->nodeType == NodeType::NODE_UNARY_OP) {
913 const auto* inner = static_cast<const UnaryOp*>(cur);
914 if (inner->opType != UnaryOp::UnaryOpType::UOP_NEGATE)
915 return false;
916 ++negateCount;
917 cur = inner->operand.get();
918 }
919 return cur != nullptr && cur->nodeType == NodeType::NODE_NUMBER_LITERAL && negateCount > 0;
920 }
921 default:
922 return false;
923 }
924 };
925 if (check(TokenType::TK_RPAREN))
926 return;
927 bool seenDefault = false; // F10: 一旦出现默认参数，后续都必须有默认值
928 // Perf-Finding4 + Bug-5: 用 unordered_set 替代每参数 O(n) 线性扫描去重，
929 // 将 parseParamList 总复杂度从 O(n²) 降为 O(n)。初始化自 params 以兼容
930 // 调用方预填充场景（虽然当前 3 个调用点均传入空 vector）。
931 std::unordered_set<std::string> paramSet(params.begin(), params.end());
932 // 预留常见容量，避免 1-2 次小幅 reallocation
933 params.reserve(params.size() + 4);
934 paramTypes.reserve(paramTypes.size() + 4);
935 defaultValues.reserve(defaultValues.size() + 4);
936 do {
937 std::string pType;
938 std::string paramName;
939
940 // 支持 C 风格类型注解: int a, float b 等
941 if (isTypeKeyword()) {
942 pType = parseTypeAnnotation();
943
944 const Token& param = consumeIdentifierOrType("期望参数名");
945 paramName = param.lexeme;
946 } else if (isClassTypeDeclStart()) {
947 // PARSE-02 fix: C 风格类类型参数: ClassName paramName 或 ClassName[] paramName
948 pType = parseTypeAnnotation();
949 const Token& param = consumeIdentifierOrType("期望参数名");
950 paramName = param.lexeme;
951 } else if (isFunTypeDeclStart()) {
952 // ROUND56 fix: 函数类型参数注解 fun(params):ret paramName
953 // samples/02-types-and-operators/types_and_operators.mini 使用 fun(int):string cb 作为
954 // 高阶函数参数类型注解。原 parseParamList 仅识别 isTypeKeyword /
955 // isClassTypeDeclStart，遇到 TK_FUN 走入"参数名在前"分支抛
956 // "期望参数名 但得到 'fun'"。此处复用 parseTypeAnnotation 生成
957 // "fun(...):ret" 字符串，与 declaration() 的 dict/fun 分支语义一致。
958 pType = parseTypeAnnotation();
959 const Token& param = consumeIdentifierOrType("期望参数名");
960 paramName = param.lexeme;
961 } else {
962 // 参数名在前面: a 或 a: int
963 const Token& param = consume(TokenType::TK_IDENTIFIER, "期望参数名");
964 paramName = param.lexeme;
965
966 // 可选的参数类型注解 : type
967 if (match(TokenType::TK_COLON)) {
968 // 接受标识符或内置类型关键字（int, float, bool, string, dict, array）
969 if (isIdentifierOrType()) {
970 pType = parseTypeAnnotation();
971 } else {
972 const Token& tok = peek();
973 throw ParseError("期望参数类型名", tok.line, tok.column);
974 }
975 }
976 }
977
978 // M-新5 fix: 检测重复参数名（Perf-Finding4: O(1) hash 查找替代 O(n) 线性扫描）
979 if (!paramSet.insert(paramName).second) {
980 throw ParseError("重复的参数名 '" + paramName + "'", peek().line, peek().column);
981 }
982 params.push_back(StringIntern::intern(paramName));
983 paramTypes.push_back(pType);
984
985 // F10: 解析默认参数值 = expr
986 if (match(TokenType::TK_ASSIGN)) {
987 seenDefault = true;
988 auto defaultExpr = expression();
989 // BUG-DEF-1 fix: 三后端一致性——仅支持字面量默认值。
990 // StackVM/RegisterVM 无法在函数入口求值复杂表达式（闭包环境访问、
991 // 参数间引用等），Interpreter 虽支持但会造成三后端行为不一致。
992 // 在 Parser 层统一拒绝，给出清晰的编译期错误而非 VM 运行时错误。
993 if (!isLiteralDefaultExpr(defaultExpr.get())) {
994 throw ParseError("默认参数值必须是字面量（数字/字符串/布尔/null/负数字面量）", defaultExpr->line,
995 defaultExpr->column);
996 }
997 defaultValues.push_back(std::move(defaultExpr));
998 } else {
999 if (seenDefault) {
1000 throw ParseError("默认参数后的参数都必须有默认值: '" + paramName + "'", peek().line, peek().column);
1001 }
1002 defaultValues.push_back(nullptr);
1003 }
1004 } while (match(TokenType::TK_COMMA) && !check(TokenType::TK_RPAREN));
1005
1006 // BUG-P1 fix: 默认参数值不能引用后续参数（如 fun f(a = b, b = 1) 应报错）
1007 for (size_t i = 0; i < defaultValues.size(); ++i) {
1008 if (!defaultValues[i])
1009 continue;
1010 std::unordered_set<std::string> refNames;
1011 collectVarRefs(defaultValues[i].get(), refNames);
1012 for (const auto& ref : refNames) {
1013 for (size_t j = i + 1; j < params.size(); ++j) {
1014 if (params[j] == ref) {
1015 throw ParseError("默认参数值不能引用后续参数: '" + ref + "'", defaultValues[i]->line,
1016 defaultValues[i]->column);
1017 }
1018 }
1019 }
1020 }

1023std::unique_ptr<ClassDecl> Parser::classDecl() {
1024 const Token& classTok = consume(TokenType::TK_CLASS, "期望 'class'");
1025 const Token& name = consumeIdentifierOrType("期望类名");
1026
1027 // R163 泛型类：可选类型参数列表 <T, K, V, ...>
1028 // 与 enumDecl/funDecl 的泛型解析模式一致：类名后、extends/'{' 前解析 <T, K>
1029 std::vector<std::string> typeParams;
1030 if (match(TokenType::TK_LT)) {
1031 do {
1032 if (!isIdentifierOrType()) {
1033 const Token& tok = peek();
1034 throw ParseError("期望类型参数名", tok.line, tok.column);
1035 }
1036 const Token& tp = advance();
1037 typeParams.push_back(tp.lexeme);
1038 } while (match(TokenType::TK_COMMA));
1039 consume(TokenType::TK_GT, "期望 '>' 结束类型参数列表");
1040 }
1041
1042 // 可选的 extends SuperClassName 或 : SuperClassName
1043 std::string superClassName = parseClassExtends();
1044
1045 // 七特性 MVP 阶段 3：可选的 `with T1, T2` 混入子句（extends 之后、'{' 之前）
1046 std::vector<std::string> traits;
1047 if (match(TokenType::TK_WITH)) {
1048 do {
1049 const Token& traitName = consumeIdentifierOrType("期望 trait 名");
1050 // 重复混入检查（同一 trait 混入两次必然同名冲突，提前报更友好的错误）
1051 for (const auto& existing : traits) {
1052 if (existing == traitName.lexeme) {
1053 throw ParseError("trait 重复混入: " + traitName.lexeme, traitName.line, traitName.column);
1054 }
1055 }
1056 traits.push_back(traitName.lexeme);
1057 } while (match(TokenType::TK_COMMA));
1058 }
1059
1060 consume(TokenType::TK_LBRACE, "期望 '{'");
1061
1062 // 解析类成员
1063 std::vector<std::shared_ptr<ASTNode>> members;
1064 parseClassMembers(members);
1065
1066 consume(TokenType::TK_RBRACE, "期望 '}'");
1067
1068 // AUDIT-P2.8 fix: 记录闭合 '}' 所在行号，供 Formatter 注入类体末尾注释。
1069 int closingBraceLine = previous().line;
1070
1071 // 七特性 MVP 阶段 3：记录自身成员数后合入 trait 方法（append 到 members 尾部）。
1072 // Formatter 仅打印前 ownMemberCount 个成员，保证往返等价。
1073 int ownMemberCount = static_cast<int>(members.size());
1074 if (!traits.empty()) {
1075 mergeTraitMethods(traits, members, classTok.line, classTok.column);
1076 }
1077
1078 auto decl = std::make_unique<ClassDecl>(StringIntern::intern(name.lexeme), superClassName, std::move(members),
1079 classTok.line, classTok.column);
1080 decl->closingBraceLine = closingBraceLine;
1081 decl->typeParams = std::move(typeParams);
1082 decl->traits = std::move(traits);
1083 decl->ownMemberCount = ownMemberCount;
1084 return decl;
1085}

1091std::unique_ptr<TraitDecl> Parser::traitDecl() {
1092 // trait T { fun m() {...} ... }（MVP 仅支持方法成员）
1093 const Token& traitTok = consume(TokenType::TK_TRAIT, "期望 'trait'");
1094 const Token& name = consumeIdentifierOrType("期望 trait 名");
1095
1096 if (knownTraits_.count(name.lexeme)) {
1097 throw ParseError("trait 重复定义: " + name.lexeme, name.line, name.column);
1098 }
1099
1100 consume(TokenType::TK_LBRACE, "期望 '{' 开始 trait 体");
1101
1102 std::vector<std::shared_ptr<ASTNode>> methods;
1103 while (!check(TokenType::TK_RBRACE) && !isAtEnd()) {
1104 // 复用 parseClassMembers 的子集：仅允许 fun 方法（字段/其他成员报错，
1105 // MVP 限定 trait 为纯方法集，避免字段默认值/槽位分配的三后端合入复杂度）
1106 if (!check(TokenType::TK_FUN)) {
1107 const Token& tok = peek();
1108 throw ParseError("trait 体内仅支持方法声明（fun 开头）", tok.line, tok.column);
1109 }
1110 auto method = funDecl();
1111 // 同 trait 内方法重名检查
1112 for (const auto& existing : methods) {
1113 if (static_cast<FunDecl*>(existing.get())->name == method->name) {
1114 throw ParseError("trait " + name.lexeme + " 内方法重复定义: " + method->name, method->line,
1115 method->column);
1116 }
1117 }
1118 methods.push_back(std::shared_ptr<ASTNode>(std::move(method)));
1119 }
1120
1121 consume(TokenType::TK_RBRACE, "期望 '}' 结束 trait 体");
1122
1123 auto decl = std::make_unique<TraitDecl>(name.lexeme, std::move(methods), traitTok.line, traitTok.column);
1124 decl->closingBraceLine = previous().line;
1125 knownTraits_[decl->name] = decl.get();
1126 return decl;
1127}

1129void Parser::mergeTraitMethods(const std::vector<std::string>& traits, std::vector<std::shared_ptr<ASTNode>>& members,
1130 int line, int col) {
1131 // 收集类自身方法名（自身方法优先于 trait 方法）
1132 std::unordered_set<std::string> ownMethodNames;
1133 for (const auto& m : members) {
1134 if (m && m->nodeType == NodeType::NODE_FUN_DECL) {
1135 ownMethodNames.insert(static_cast<FunDecl*>(m.get())->name);
1136 }
1137 }
1138
1139 // 合入各 trait 方法；记录每个合入方法来自哪个 trait，用于 diamond 冲突诊断
1140 std::unordered_map<std::string, std::string> mergedFrom; // 方法名 → trait 名
1141 for (const auto& traitName : traits) {
1142 auto it = knownTraits_.find(traitName);
1143 if (it == knownTraits_.end()) {
1144 throw ParseError("未定义的 trait: " + traitName + "（trait 须先声明后使用）", line, col);
1145 }
1146 for (const auto& method : it->second->methods) {
1147 auto* fd = static_cast<FunDecl*>(method.get());
1148 // 类自身方法优先：跳过被覆盖的 trait 方法
1149 if (ownMethodNames.count(fd->name)) {
1150 continue;
1151 }
1152 // diamond 冲突：两个 trait 提供同名方法且类未覆盖
1153 auto conflictIt = mergedFrom.find(fd->name);
1154 if (conflictIt != mergedFrom.end()) {
1155 throw ParseError("trait 方法冲突（diamond）: 方法 " + fd->name + " 同时来自 trait " +
1156 conflictIt->second + " 与 " + traitName + "，类需自行覆盖以解决冲突",
1157 line, col);
1158 }
1159 mergedFrom[fd->name] = traitName;
1160 // shared_ptr 共享 FunDecl 节点：多个类混入同一 trait 共享同一方法 AST，
1161 // 三后端各自按普通类方法注册/编译（Interpreter 存 shared_ptr，
1162 // Compiler/IR 按类名生成独立 chunk，共享节点无状态冲突）。
1163 members.push_back(method);
1164 }
1165 }

1179
1180void Parser::parseClassMembers(std::vector<std::shared_ptr<ASTNode>>& members) {
1181 // 调用前应已 consume '{'，循环直到 '}' 或 EOF
1182 while (!check(TokenType::TK_RBRACE) && !isAtEnd()) {
1183 // BUG-PARSER-AUDIT-2 fix: classDecl 成员循环需 try/catch 错误恢复，
1184 // 与 block() 模式一致。原实现无恢复，单个坏成员抛异常会穿透到外层 block，
1185 // synchronize 在 class 的 '}' 处返回，外层 block 消费 class 的 '}' 作为
1186 // 自己的闭合，导致整个类丢失 + 外层块结构错乱。
1187 try {
1188 // 类成员可以是：
1189 // - var 声明（字段）
1190 // - fun/function 声明（方法）
1191 // - 带类型注解的声明
1192 // - 裸方法名定义: methodName() {} （不带 fun 关键字）
1193 if (check(TokenType::TK_VAR)) {
1194 members.push_back(varDecl());
1195 } else if (check(TokenType::TK_FUN)) { // Lexer 已统一 function/func → TK_FUN
1196 members.push_back(funDecl());
1197 } else if (isTypeKeyword()) {
1198 // 带类型注解的字段声明（如 int count = 0;）或带返回类型的方法声明（如 int getValue() {}）
1199 int savePos = current_;
1200 std::string typeAnn = parseTypeAnnotation();
1201
1202 if (isIdentifierOrType()) {
1203 // 检查是否是带返回类型的方法声明: int getValue(
1204 if (checkNext(TokenType::TK_LPAREN)) {
1205 members.push_back(typedFunDecl(typeAnn));
1206 } else if (check(TokenType::TK_IDENTIFIER)) {
1207 members.push_back(typedVarDecl(typeAnn));
1208 } else {
1209 current_ = savePos;
1210 break;
1211 }
1212 } else {
1213 // 回溯
1214 current_ = savePos;
1215 break;
1216 }
1217 } else if (check(TokenType::TK_IDENTIFIER)) {
1218 // 裸方法定义: methodName(params) { body }
1219 // 或类类型字段: ClassName fieldName; / ClassName[] fieldName;
1220 // 或类类型方法: ClassName methodName() { body }
1221 int savePos = current_;
1222 const Token& firstTok = advance(); // 方法名或类名
1223
1224 // 检查是否是方法定义: name(
1225 if (check(TokenType::TK_LPAREN)) {
1226 // 这是一个裸方法定义
1227 consume(TokenType::TK_LPAREN, "期望 '('");
1228
1229 std::vector<std::string> params;
1230 std::vector<std::string> paramTypes;
1231 std::vector<std::shared_ptr<ASTNode>> defaultValues; // F10
1232 parseParamList(params, paramTypes, defaultValues);
1233 consume(TokenType::TK_RPAREN, "期望 ')'");
1234
1235 // PARSE-06 fix: 可选的返回类型注解（支持 : type 和 -> type）
1236 std::string returnType;
1237 if (match(TokenType::TK_COLON)) {
1238 // BUG-PARSER-AUDIT-1 fix: 裸方法 `:` 后预检类型 token
1239 if (!isIdentifierOrType()) {
1240 const Token& tok = peek();
1241 throw ParseError("期望返回类型名", tok.line, tok.column);
1242 }
1243 returnType = parseTypeAnnotation();
1244 } else if (check(TokenType::TK_MINUS) && checkNext(TokenType::TK_GT)) {
1245 advance(); // 消耗 '-'
1246 advance(); // 消耗 '>'
1247 if (!isIdentifierOrType()) { // BUG-PARSER-AUDIT-1
1248 const Token& tok = peek();
1249 throw ParseError("期望返回类型名", tok.line, tok.column);
1250 }
1251 returnType = parseTypeAnnotation();
1252 }
1253
1254 consume(TokenType::TK_LBRACE, "期望 '{'");
1255 auto body = block();
1256
1257 auto decl = std::make_unique<FunDecl>(StringIntern::intern(firstTok.lexeme), std::move(params),
1258 std::move(paramTypes), returnType, std::move(body),
1259 firstTok.line, firstTok.column);
1260 // F10: 计算必需参数个数
1261 int reqCount = 0;
1262 for (size_t i = 0; i < defaultValues.size(); ++i) {
1263 if (defaultValues[i] == nullptr) {
1264 ++reqCount;
1265 } else {
1266 break;
1267 }
1268 }
1269 decl->requiredParamCount = reqCount;
1270 decl->defaultValues = std::move(defaultValues);
1271 members.push_back(std::move(decl));
1272 } else if (check(TokenType::TK_LBRACKET)) {
1273 // BUG-LPA-04 fix: ClassName[] fieldName; 数组类型字段
1274 // 原实现消耗 ClassName 后遇 [ 直接回溯，不支持类类型数组字段。
1275 // 与参数列表/for循环/顶层声明行为对齐：消费 [] 后缀构造类型注解。
1276 // AUDIT-P2 fix: 支持多维数组字段（ClassName[][] field），
1277 // 对齐 parseTypeAnnotation 的 while 循环模式。原实现仅消费单对 []。
1278 int bracketSave = current_;
1279 std::string typeAnn = firstTok.lexeme;
1280 while (check(TokenType::TK_LBRACKET)) {
1281 advance(); // 消耗 '['
1282 if (!check(TokenType::TK_RBRACKET)) {
1283 current_ = bracketSave;
1284 break;
1285 }
1286 advance(); // 消耗 ']'
1287 typeAnn += "[]";
1288 }
1289 if (check(TokenType::TK_IDENTIFIER)) {
1290 // ClassName[] fieldName — 类类型数组字段
1291 // （ClassName[] methodName() 不合法，不支持类数组返回类型方法）
1292 members.push_back(typedVarDecl(typeAnn));
1293 } else {
1294 // 无法识别，回溯
1295 current_ = savePos;
1296 break;
1297 }
1298 } else if (check(TokenType::TK_IDENTIFIER)) {
1299 // 可能是类类型字段: ClassName fieldName; 或类类型方法: ClassName methodName()
1300 if (checkNext(TokenType::TK_LPAREN)) {
1301 // ClassName methodName() — 带类类型的方法声明
1302 members.push_back(typedFunDecl(firstTok.lexeme));
1303 } else {
1304 // ClassName fieldName — 类类型字段声明
1305 // current_ 已在 advance() 后指向 fieldName，无需回溯
1306 members.push_back(typedVarDecl(firstTok.lexeme));
1307 }
1308 } else {
1309 // 无法识别，回溯
1310 current_ = savePos;
1311 break;
1312 }
1313 } else {
1314 break;
1315 }
1316 } catch (const ParseError& e) {
1317 // BUG-PARSER-AUDIT-2 fix: 成员解析错误恢复
1318 diagnostics_.addError(e.what(), e.line, e.column, DiagSource::Parser, e.code);
1319 synchronize();
1320 // synchronize 在 '}' 处返回（不消费），循环条件 check(RBRACE) 退出，
1321 // 由下方 consume(TK_RBRACE) 消费 class 的闭合花括号。
1322 }
1323 }
1324}

1330std::unique_ptr<EnumDecl> Parser::enumDecl() {

1333 const Token& name = consumeIdentifierOrType("期望 enum 名称");
1334
1335 // 可选泛型类型参数列表 <T, U, ...>
1336 std::vector<std::string> typeParams;
1337 if (match(TokenType::TK_LT)) {
1338 do {
1339 if (!isIdentifierOrType()) {
1340 const Token& tok = peek();
1341 throw ParseError("期望类型参数名", tok.line, tok.column);
1342 }
1343 const Token& tp = advance();
1344 typeParams.push_back(tp.lexeme);
1345 } while (match(TokenType::TK_COMMA));
1346 consume(TokenType::TK_GT, "期望 '>' 结束类型参数列表");
1347 }
1348
1349 consume(TokenType::TK_LBRACE, "期望 '{' 开始 enum 体");
1350
1351 // 解析 variant 列表
1352 std::vector<EnumVariant> variants;
1353 while (!check(TokenType::TK_RBRACE) && !isAtEnd()) {
1354 try {
1355 if (!isIdentifierOrType()) {
1356 const Token& tok = peek();
1357 throw ParseError("期望 variant 名称", tok.line, tok.column);
1358 }
1359 const Token& varName = advance();
1360 EnumVariant variant;
1361 variant.name = varName.lexeme;
1362
1363 // 可选的参数类型列表 (T1, T2, ...)
1364 if (match(TokenType::TK_LPAREN)) {
1365 if (!check(TokenType::TK_RPAREN)) {
1366 do {
1367 // variant 参数类型可以是任意类型注解
1368 std::string paramType = parseTypeAnnotation();

1370 } while (match(TokenType::TK_COMMA) && !check(TokenType::TK_RPAREN));
1371 }
1372 consume(TokenType::TK_RPAREN, "期望 ')' 结束 variant 参数列表");
1373 }
1374
1375 variants.push_back(std::move(variant));
1376
1377 // variant 之间用 ',' 分隔，尾逗号可选
1378 if (!match(TokenType::TK_COMMA)) {
1379 break;
1380 }
1381 } catch (const ParseError& e) {
1382 // 错误恢复：与 classDecl 一致
1383 diagnostics_.addError(e.what(), e.line, e.column, DiagSource::Parser, e.code);
1384 synchronize();
1385 // synchronize 在 '}' 处返回（不消费），循环条件 check(RBRACE) 退出
1386 }
1387 }
1388
1389 consume(TokenType::TK_RBRACE, "期望 '}' 结束 enum 体");
1390
1391 int closingBraceLine = previous().line;
1392 auto decl = std::make_unique<EnumDecl>(name.lexeme, std::move(typeParams), std::move(variants), enumTok.line,
1393 enumTok.column);
1394 decl->closingBraceLine = closingBraceLine;
1395
1396 // 注册 enum 名到 knownEnums_，供 call() 区分 EnumVariantExpr 与 MemberAccess
1397 knownEnums_.insert(name.lexeme);

1400}

1406std::unique_ptr<MacroDecl> Parser::macroDecl() {
1407 // macro name(p1, p2) { <expr> }
1408 const Token& macroTok = consume(TokenType::TK_MACRO, "期望 'macro'");
1409 const Token& name = consumeIdentifierOrType("期望宏名");
1410 std::string macroName = name.lexeme;
1411
1412 // 重定义检查（宏表是 parse 期全局单一命名空间，重名会静默改变已解析调用的语义）
1413 if (knownMacros_.count(macroName)) {
1414 throw ParseError("宏重复定义: " + macroName, name.line, name.column);
1415 }
1416
1417 consume(TokenType::TK_LPAREN, "期望 '('");
1418 std::vector<std::string> params;
1419 if (!check(TokenType::TK_RPAREN)) {
1420 do {
1421 if (!isIdentifierOrType()) {
1422 const Token& tok = peek();
1423 throw ParseError("期望宏参数名", tok.line, tok.column);
1424 }
1425 const Token& p = advance();
1426 // 重复参数名检查（重名参数会使替换表静默丢失前一个实参）
1427 for (const auto& existing : params) {
1428 if (existing == p.lexeme) {
1429 throw ParseError("宏参数名重复: " + p.lexeme, p.line, p.column);
1430 }
1431 }
1432 params.push_back(p.lexeme);
1433 } while (match(TokenType::TK_COMMA));
1434 }
1435 consume(TokenType::TK_RPAREN, "期望 ')'");
1436
1437 consume(TokenType::TK_LBRACE, "期望 '{' 开始宏体");
1438 // 七特性宏升级：区分语句宏与表达式宏。
1439 // 若首 token 为语句起始关键字（var/if/while/for/return/print/break/continue/throw/try，
1440 // 均不是合法表达式起始，无歧义）→ 解析为语句块（Block）；否则解析为单表达式。
1441 TokenType bt = peek().type;
1442 bool isStmtBody =
1443 (bt == TokenType::TK_VAR || bt == TokenType::TK_IF || bt == TokenType::TK_WHILE || bt == TokenType::TK_FOR ||
1444 bt == TokenType::TK_RETURN || bt == TokenType::TK_PRINT || bt == TokenType::TK_BREAK ||
1445 bt == TokenType::TK_CONTINUE || bt == TokenType::TK_THROW || bt == TokenType::TK_TRY);
1446 std::shared_ptr<ASTNode> body;
1447 if (isStmtBody) {
1448 // 语句宏：复用 block()（'{' 已消耗，previous() 为 '{'）解析语句列表并消耗 '}'。
1449 body = std::shared_ptr<ASTNode>(block());
1450 } else {
1451 // 表达式宏：body 限单个表达式；尾部分号可选
1452 auto expr = expression();
1453 match(TokenType::TK_SEMICOLON);
1454 consume(TokenType::TK_RBRACE, "期望 '}' 结束宏体（表达式宏限单个表达式）");
1455 body = std::shared_ptr<ASTNode>(std::move(expr));
1456 }
1457
1458 auto decl = std::make_unique<MacroDecl>(std::move(macroName), std::move(params), std::move(body), macroTok.line,
1459 macroTok.column);
1460 decl->isStatementMacro = isStmtBody;
1461 // body 解析完成后才登记 → 递归宏（自引用）在 body 解析时报"未定义的宏"，
1462 // 从根上消除无限展开。
1463 knownMacros_[decl->name] = decl.get();
1464 return decl;
1465}

1467std::unique_ptr<ASTNode> Parser::macroCall(const Token& nameTok) {
1468 // 调用前提：nameTok 已消耗，当前位置 '!' '('（调用方已 check 确认）
1469 advance(); // 消耗 '!'（TK_NOT）
1470 advance(); // 消耗 '('
1471
1472 auto it = knownMacros_.find(nameTok.lexeme);
1473 if (it == knownMacros_.end()) {
1474 throw ParseError("未定义的宏: " + nameTok.lexeme + "!（宏须先声明后使用）", nameTok.line, nameTok.column);
1475 }
1476 MacroDecl* decl = it->second;
1477
1478 // 解析实参列表
1479 std::vector<std::shared_ptr<ASTNode>> args;
1480 if (!check(TokenType::TK_RPAREN)) {
1481 do {
1482 args.push_back(expression());
1483 } while (match(TokenType::TK_COMMA) && !check(TokenType::TK_RPAREN));
1484 }
1485 consume(TokenType::TK_RPAREN, "期望 ')' 结束宏调用参数列表");
1486
1487 if (args.size() != decl->params.size()) {
1488 throw ParseError("宏 " + decl->name + "! 期望 " + std::to_string(decl->params.size()) + " 个参数，实际 " +
1489 std::to_string(args.size()) + " 个",
1490 nameTok.line, nameTok.column);
1491 }
1492
1493 // 构建替换表并克隆 body 模板（实参子树 shared_ptr 共享，不克隆实参本身）
1494 MacroExpander::SubstMap subst;
1495 for (size_t i = 0; i < decl->params.size(); ++i) {
1496 subst[decl->params[i]] = args[i];
1497 }
1498 std::string unsupported;
1499 auto expanded = MacroExpander::cloneWithSubstitution(decl->bodyExpr.get(), subst, 0, &unsupported);
1500 if (!expanded) {
1501 if (unsupported == "<depth-limit>") {
1502 throw ParseError("宏展开嵌套过深（超过 " + std::to_string(RuntimeLimits::MAX_MACRO_EXPANSION_DEPTH) +
1503 " 层）",
1504 nameTok.line, nameTok.column);
1505 }
1506 throw ParseError("宏 " + decl->name + "! 的 body 含不支持的构造: " + unsupported +
1507 "（表达式/语句模板宏仅支持表达式与基础语句子集）",
1508 nameTok.line, nameTok.column);
1509 }
1510
1511 auto callNode = std::make_unique<MacroCallExpr>(nameTok.lexeme, std::move(args), std::move(expanded), nameTok.line,
1512 nameTok.column);
1513 // 七特性宏升级：语句宏展开为 Block（不产生表达式值），后端执行后补 null
1514 // 作为表达式值保证栈平衡；表达式宏 producesValue=true。
1515 callNode->producesValue = !decl->isStatementMacro;
1516 return callNode;
1517}

1519std::unique_ptr<MatchExpr> Parser::matchExpr() {
1520 // match scrutinee { case Pattern [if guard] => body; ... default => body; }
1521 const Token& matchTok = consume(TokenType::TK_MATCH, "期望 'match'");
1522 DepthGuard guard{parseDepth_};
1523
1524 // scrutinee 必须用括号包裹以避免与 { case ... } 的歧义
1525 consume(TokenType::TK_LPAREN, "期望 '(' 开始 match 表达式");
1526 auto scrutinee = expression();
1527 // R134 fix: scrutinee 支持元组字面量 (e1, e2, ...)。
1528 // expression() 不会跨 ',' 解析元组（元组字面量由 primary 处理），
1529 // 因此当首个 expression 后紧跟 ',' 时需切换到元组解析模式。
1530 // 否则 `match (3, 4) { case (a, b) => ... }` 会因 consume(RPAREN) 看到 ',' 而抛错。
1531 if (check(TokenType::TK_COMMA)) {
1532 std::vector<std::shared_ptr<ASTNode>> elements;
1533 elements.push_back(std::move(scrutinee));
1534 while (match(TokenType::TK_COMMA) && !check(TokenType::TK_RPAREN)) {
1535 elements.push_back(expression());
1536 }
1537 scrutinee = std::make_unique<TupleLiteral>(std::move(elements), matchTok.line, matchTok.column);
1538 }
1539 consume(TokenType::TK_RPAREN, "期望 ')' 结束 match 表达式");
1540
1541 consume(TokenType::TK_LBRACE, "期望 '{' 开始 match 体");
1542
1543 std::vector<MatchCase> cases;
1544 bool hasDefault = false;
1545
1546 while (!check(TokenType::TK_RBRACE) && !isAtEnd()) {
1547 try {
1548 MatchCase mc;
1549
1550 if (match(TokenType::TK_DEFAULT)) {
1551 // default => body
1552 if (hasDefault) {
1553 const Token& tok = previous();
1554 throw ParseError("match 表达式不能有多个 default 分支", tok.line, tok.column);
1555 }
1556 hasDefault = true;
1557 mc.isDefault = true;
1558 mc.pattern = nullptr;
1559 } else if (match(TokenType::TK_CASE)) {
1560 // case Pattern => body (R134: Pattern 支持 OR/TUPLE/VARIABLE/嵌套)
1561 mc.pattern = matchPattern();
1562 mc.isDefault = false;
1563 } else {
1564 // 兼容裸 Pattern => body（无 case 关键字）
1565 mc.pattern = matchPattern();
1566 mc.isDefault = false;
1567 }
1568
1569 // R134: 可选 guard 表达式 `if cond`
1570 if (match(TokenType::TK_IF)) {
1571 mc.guard = expression();
1572 }
1573
1574 consume(TokenType::TK_ARROW, "期望 '=>' 分隔 match 模式与体");
1575
1576 // case 体：单表达式（以 ';' 结束）或块语句
1577 if (check(TokenType::TK_LBRACE)) {
1578 advance(); // 消耗 '{'
1579 mc.body = block();
1580 } else {
1581 auto expr = expression();
1582 // 可选 ';'（match case 体可能不强制分号）
1583 match(TokenType::TK_SEMICOLON);
1584 mc.body = std::move(expr);
1585 }
1586
1587 cases.push_back(std::move(mc));
1588 } catch (const ParseError& e) {
1589 diagnostics_.addError(e.what(), e.line, e.column, DiagSource::Parser, e.code);
1590 // AUDIT-R6 F1 fix (P0): matchExpr 错误恢复死循环。synchronize() 在同步关键字
1591 // （var/fun/class 等，BUG-PARSER-SYNC-1）处不消耗 token 直接返回，但本循环无法
1592 // 处理这些关键字（既非 case/default 也非 '}'），裸 pattern 分支再次抛错→
1593 // synchronize 仍不前进→死循环挂死（如 `match (1) { case n: { var f = 1; } }`，
1594 // 用户写错 `case X:` 而非 `case X =>` 即触发）。修复：同步后无进展则强制
1595 // 前进一个 token，保证循环收敛（'}'/EOF 由循环条件终止）。
1596 int beforeSync = current_;
1597 synchronize();
1598 if (current_ == beforeSync && !isAtEnd() && !check(TokenType::TK_RBRACE)) {
1599 advance();
1600 }
1601 }
1602 }
1603
1604 consume(TokenType::TK_RBRACE, "期望 '}' 结束 match 体");
1605
1606 return std::make_unique<MatchExpr>(std::move(scrutinee), std::move(cases), matchTok.line, matchTok.column);
1607}

1609// R134 模式匹配扩展：matchPattern 改为递归下降解析器。
1610// 解析优先级（从低到高）：
1611// orPattern: andPattern ('|' andPattern)* -- OR pattern
1612// andPattern: primaryPattern -- 当前无 and pattern，留接口
1613// primaryPattern:
1614// | '_' -- WILDCARD
1615// | INT_LIT/FLOAT_LIT/STRING_LIT/... -- LITERAL
1616// | '(' pattern (',' pattern)* ')' -- TUPLE pattern (或裸 (pattern))
1617// | Identifier -- VARIABLE pattern（绑定整个 scrutinee）
1618// | Identifier '.' Identifier '(' ... ')' -- VARIANT pattern（嵌套子 pattern）
1619//
1620// 注意：VARIANT 的子 pattern 不再是字符串变量名，而是完整的 pattern，
1621// 支持 `Some(Some(x))` / `Pair(Some(a), None)` 等嵌套形式。
1622std::shared_ptr<MatchPattern> Parser::matchPattern() {
1623 auto left = matchPrimaryPattern();
1624 // OR pattern: left or right or ...（MiniLang 用 'or' 关键字而非 '|'）
1625 while (match(TokenType::TK_OR)) {
1626 auto right = matchPrimaryPattern();
1627 // 合并到 OR pattern：若 left 已是 OR，则追加；否则新建 OR
1628 if (left->kind == MatchPatternKind::OR) {
1629 left->subPatterns.push_back(std::move(right));
1630 } else {
1631 auto orPattern = std::make_shared<MatchPattern>(MatchPatternKind::OR, left->line, left->column);
1632 orPattern->subPatterns.push_back(std::move(left));
1633 orPattern->subPatterns.push_back(std::move(right));
1634 left = orPattern;
1635 }
1636 }
1637 return left;
1638}

1640std::shared_ptr<MatchPattern> Parser::matchPrimaryPattern() {
1641 // Bug #9 fix: 互递归深度保护（matchPattern <-> matchPrimaryPattern）
1642 if (parseDepth_ >= MAX_PARSE_DEPTH) {
1643 throw ParseError("模式嵌套过深（超过 " + std::to_string(MAX_PARSE_DEPTH) + " 层）", peek().line, peek().column);
1644 }
1645 DepthGuard guard{parseDepth_};
1646
1647 const Token& tok = peek();
1648
1649 // 1. 通配符 _
1650 if (check(TokenType::TK_IDENTIFIER) && tok.lexeme == "_") {
1651 advance();
1652 return std::make_shared<MatchPattern>(MatchPatternKind::WILDCARD, tok.line, tok.column);
1653 }
1654
1655 // 2. 字面量模式（INT_LIT / FLOAT_LIT / STRING_LIT / TRUE / FALSE / NULL）
1656 if (check(TokenType::TK_INT_LIT) || check(TokenType::TK_FLOAT_LIT) || check(TokenType::TK_STRING_LIT) ||
1657 check(TokenType::TK_TRUE) || check(TokenType::TK_FALSE) || check(TokenType::TK_NULL)) {
1658 auto literalExpr = primary();
1659 auto p = std::make_shared<MatchPattern>(MatchPatternKind::LITERAL, tok.line, tok.column);
1660 p->literal = std::move(literalExpr);
1661 return p;
1662 }
1663
1664 // 3. 元组模式：(pattern, pattern, ...) 或裸 (pattern)
1665 if (check(TokenType::TK_LPAREN)) {
1666 advance(); // 消耗 '('
1667 // 空 () 视为 0 元组（不支持，要求至少 1 个元素）
1668 if (check(TokenType::TK_RPAREN)) {
1669 throw ParseError("match 元组模式要求至少 1 个元素", tok.line, tok.column);
1670 }
1671 auto p = std::make_shared<MatchPattern>(MatchPatternKind::TUPLE, tok.line, tok.column);
1672 do {
1673 p->subPatterns.push_back(matchPattern());
1674 } while (match(TokenType::TK_COMMA) && !check(TokenType::TK_RPAREN));
1675 consume(TokenType::TK_RPAREN, "期望 ')' 结束元组模式");
1676 // 单元素 (x) 不是真正的元组，回退为裸 pattern
1677 if (p->subPatterns.size() == 1) {
1678 return std::move(p->subPatterns[0]);
1679 }
1680 return p;
1681 }
1682
1683 // 4. 标识符：可能是 VARIABLE pattern 或 VARIANT pattern
1684 if (check(TokenType::TK_IDENTIFIER)) {
1685 const std::string& name = tok.lexeme;
1686 // 4a. VARIANT pattern: EnumName.VariantName(...)
1687 if (knownEnums_.find(name) != knownEnums_.end()) {
1688 advance(); // 消耗 enum 名
1689 consume(TokenType::TK_DOT, "期望 '.' 分隔 enum 名与 variant 名");
1690 if (!isIdentifierOrType()) {
1691 const Token& v = peek();
1692 throw ParseError("期望 variant 名", v.line, v.column);
1693 }
1694 const Token& varNameTok = advance();
1695
1696 auto p = std::make_shared<MatchPattern>(MatchPatternKind::VARIANT, tok.line, tok.column);
1697 p->enumName = name;
1698 p->variantName = varNameTok.lexeme;
1699
1700 // 可选的子 pattern 列表 (subPattern1, subPattern2, ...)
1701 // 子 pattern 是完整 pattern（递归 matchPattern），支持嵌套
1702 if (match(TokenType::TK_LPAREN)) {
1703 if (!check(TokenType::TK_RPAREN)) {
1704 do {
1705 p->subPatterns.push_back(matchPattern());
1706 } while (match(TokenType::TK_COMMA) && !check(TokenType::TK_RPAREN));
1707 }
1708 consume(TokenType::TK_RPAREN, "期望 ')' 结束 variant 子 pattern 列表");
1709 }
1710 return p;
1711 }
1712 // 4b. VARIABLE pattern: x（绑定整个 scrutinee 到变量 x）
1713 // 注意：变量名不能是关键字（match/case/default/if 等）
1714 advance(); // 消耗变量名
1715 auto p = std::make_shared<MatchPattern>(MatchPatternKind::VARIABLE, tok.line, tok.column);
1716 p->variableName = name;
1717 return p;
1718 }
1719
1720 throw ParseError("期望 match 模式（'_' / 字面量 / Identifier / EnumName.VariantName(...) / (p1, p2, ...)）",
1721 tok.line, tok.column);
1722}

1724std::unique_ptr<ASTNode> Parser::statement() {
1725 // P1-1 fix: 无花括号的 if/while/for 嵌套语句也需深度保护，防止栈溢出 DoS
1726 // 覆盖 "if (a) if (b) if (c) ..." 这类无花括号嵌套场景
1727 if (parseDepth_ >= MAX_PARSE_DEPTH) {
1728 throw ParseError("语句嵌套过深（超过 " + std::to_string(MAX_PARSE_DEPTH) + " 层）", peek().line, peek().column);
1729 }
1730 DepthGuard guard{parseDepth_}; // C4 fix: 自动 ++/-- parseDepth_
1731
1732 // BUG-LPA-05 fix: import/export 在无花括号单语句体中显式拒绝。
1733 // 原实现 statement() 不识别 TK_IMPORT/TK_EXPORT，落入 expressionStatement()
1734 // → primary() 抛"意外的 Token"，错误消息误导用户。
1735 // blockDepth_ 检查仅覆盖块内场景，无花括号体（if/while/for 无 {}）走
1736 // statement() 路径不递增 blockDepth_，需在此显式拒绝。
1737 if (check(TokenType::TK_IMPORT)) {
1738 const Token& tok = peek();
1739 throw ParseError("import 语句只能在顶层使用", tok.line, tok.column);
1740 }
1741 if (check(TokenType::TK_EXPORT)) {
1742 const Token& tok = peek();
1743 throw ParseError("export 语句只能在顶层使用", tok.line, tok.column);
1744 }
1745
1746 if (check(TokenType::TK_IF))
1747 return ifStmt();
1748 if (check(TokenType::TK_WHILE))
1749 return whileStmt();
1750 if (check(TokenType::TK_FOR))
1751 return forStmt();
1752 if (check(TokenType::TK_RETURN))
1753 return returnStmt();
1754 if (check(TokenType::TK_BREAK))
1755 return breakStmt();
1756 if (check(TokenType::TK_CONTINUE))
1757 return continueStmt();
1758 if (check(TokenType::TK_TRY))
1759 return tryStmt();
1760 if (check(TokenType::TK_THROW))
1761 return throwStmt();
1762 if (check(TokenType::TK_PRINT))
1763 return printStmt();
1764 if (check(TokenType::TK_LBRACE)) {
1765 advance();
1766 return block();
1767 }
1768 return expressionStatement();
1769}

1831std::unique_ptr<ForStmt> Parser::forStmt() {
1832 const Token& forTok = consume(TokenType::TK_FOR, "期望 'for'");
1833 consume(TokenType::TK_LPAREN, "期望 '('");
1834
1835 // 初始化部分
1836 std::unique_ptr<ASTNode> init = nullptr;
1837 if (check(TokenType::TK_VAR)) {
1838 init = varDecl();
1839 // varDecl 已经消耗了分号
1840 } else if (isTypeKeyword()) {
1841 // 类型注解声明，如 int j = 0;
1842 int savePos = current_;
1843 std::string typeAnn = parseTypeAnnotation();
1844
1845 if (check(TokenType::TK_IDENTIFIER)) {
1846 init = typedVarDecl(typeAnn);
1847 // typedVarDecl 已经消耗了分号
1848 } else {
1849 // 回溯，当做表达式处理
1850 current_ = savePos;
1851 init = expression();
1852 consume(TokenType::TK_SEMICOLON, "期望 ';'");
1853 }
1854 } else if (isClassTypeDeclStart()) {
1855 // H2 fix: 类名类型注解声明，如 Point p = create(); 或 Point[] arr = build();
1856 int savePos = current_;
1857 std::string typeAnn = parseTypeAnnotation(); // 消耗类名 + 可选 []
1858 if (check(TokenType::TK_IDENTIFIER) && checkNext(TokenType::TK_LPAREN)) {
1859 // ClassName funcName( — 函数声明不应出现在 for 初始化中，回溯当表达式处理
1860 current_ = savePos;

2181// 每一层函数只处理"本级及更低优先级"的运算符，遇到更高优先级就下沉到下一层函数。
2182// 层级从松到紧依次为：
2183// assignment(右结合) → or_ → and_ → equality → comparison → term → factor

2185// 其中 or_(1) < and_(2) < equality(3) < comparison(4) < term(+|-,5)

2188// ((a-b)-c)——循环不断把"右侧同级子表达式"挂到已构建的左树上。
2189// 右结合（赋值）则在 assignment() 中通过"递归调用 assignment() 而非循环"实现：
2190// a = b = c 解析为 a = (b = c)。
2191// 每层入口均做 parseDepth_ 深度保护（配合 DepthGuard 自动回退），
2192// 防止极端嵌套表达式触发 C++ 递归栈溢出导致 DoS。

2747 // 不跳过块边界/EOF——advance() 前先检查当前 token 是否已是同步点。
2748 // 原 advance() 无条件跳过当前 token，若当前 token 是 }，
2749 // 会被跳过而 previous() 检查仅识别 TK_SEMICOLON，导致块边界丢失产生级联错误。
2750 //
2751 // FIX-1: 分号必须消耗（advance），否则主循环再次调用 declaration() 时
2752 // primary() 不识别 ';' 抛异常 → synchronize() 又看到 ';' 直接 return → 死循环。
2753 // FIX-2: 插值字符串 token（TK_INTERP_END/TK_STRING_PART/TK_INTERP_START）
2754 // 只在插值字符串上下文中有意义，顶层 parse 循环中无法处理。
2755 // 若不消耗而直接 return，primary() 不识别这些 token → 抛异常 → synchronize() 又
2756 // 看到同样 token → 死循环。必须消耗所有插值相关 token 以跳过断裂的字符串上下文。
2757 // '}' 是结构边界，不能消耗；EOF 无需消耗（isAtEnd 会终止循环）。
2758 if (peek().type == TokenType::TK_RBRACE || peek().type == TokenType::TK_EOF) {
2759 return;
2760 }
2761 if (peek().type == TokenType::TK_SEMICOLON) {
2762 advance(); // 消耗分号以确保向前推进
2763 return;
2764 }
2765 // 消耗所有插值字符串相关 token，跳过断裂的字符串上下文
2766 while (peek().type == TokenType::TK_INTERP_END || peek().type == TokenType::TK_STRING_PART ||
2767 peek().type == TokenType::TK_INTERP_START) {
2768 advance();
2769 }
2770 if (isAtEnd())
2771 return;
2772 // 如果消耗插值 token 后到达了同步点，停止
2773 if (peek().type == TokenType::TK_RBRACE || peek().type == TokenType::TK_EOF ||
2774 peek().type == TokenType::TK_SEMICOLON) {
2775 if (peek().type == TokenType::TK_SEMICOLON)
2776 advance();
2777 return;
