73/// P13 fix: 小缓冲区优化的参数容器，避免方法调用时的堆分配
74/// 对于 argCount <= N 使用栈上内联存储，超出时回退到 vector
75template <typename T, size_t N = 8> class SmallArgs {
76 T inline_[N];
77 std::vector<T> heap_;
78 size_t sz_ = 0;
79
80public:
81 // P0 fix: 显式 value-initialize inline_ 数组——原 `= default` 在某些场景下
82 // 跳过元素默认构造，导致栈上 NaNBox 是随机位模式（包括 0xFFFFFFFFFFFFFFFF），
83 // 后续被 NaNBox::tag() 误判为合法 FLOAT（NaN）掩盖底层 UB。
84 // PERF: 使用 std::fill_n 替代手动循环，编译器可向量化为 SIMD 批量写入。
85 SmallArgs() : sz_(0) { std::fill_n(inline_, N, T()); }
86 explicit SmallArgs(size_t count) : sz_(count) {
87 std::fill_n(inline_, N, T());
88 if (count > N)
89 heap_.resize(count);
90 }
91 size_t size() const { return sz_; }
92 bool empty() const { return sz_ == 0; }
93 // P0 fix: assert 在 Release 被剥离，越界访问会读到栈垃圾（可能形成 0xFFFFFFFFFFFFFFFF），
94 // 被当作合法 Value 使用。改为运行时 abort，与 VMStack 一致风格。
95 T& operator[](size_t i) {
96 if (i >= sz_) [[unlikely]] {
97 std::abort();
98 }
99 return (sz_ <= N) ? inline_[i] : heap_[i];
100 }
101 const T& operator[](size_t i) const {
102 if (i >= sz_) [[unlikely]] {
103 std::abort();
104 }
105 return (sz_ <= N) ? inline_[i] : heap_[i];
106 }
107 T* begin() { return (sz_ <= N) ? inline_ : heap_.data(); }
108 T* end() { return begin() + sz_; }
109 const T* begin() const { return (sz_ <= N) ? inline_ : heap_.data(); }
110 const T* end() const { return begin() + sz_; }
111 T* data() { return begin(); }
112 const T* data() const { return begin(); }
113 void push_back(const T& v) {
114 if (sz_ < N) {
115 inline_[sz_++] = v;
116 } else {
117 if (sz_ == N) {
118 // PERF: 预分配 2N 容量，避免 assign 后立即 push_back 触发二次重分配
119 heap_.reserve(N * 2);
120 heap_.assign(inline_, inline_ + N);
121 }
122 heap_.push_back(v);
123 sz_++;
124 }
125 }
126};

180// ============================================================
181// PERF-13: VMStack — 定长数组 + 栈顶指针，替代 std::vector<Value>
182// ------------------------------------------------------------
183// Value 从 24 字节降至 8 字节后，1024 元素仅 8KB，可直接内联到 VM 对象。
184// 消除 push_back 的容量检查和潜在的堆分配开销。
185// API 与 std::vector<Value> 兼容（size/empty/clear/push_back/pop_back/
186// back/operator[]/resize/emplace_back/reserve），实现直接替换。
187// ============================================================
188class VMStack {
189public:
190 VMStack() : top_(0) {
191 // 显式 value-initialize 所有元素为 nullValue。
192 // 原 `Value data_[CAPACITY];` 在某些编译器/场景下可能跳过元素默认构造，
193 // 导致栈上 NaNBox 是随机位模式（包括 0xFFFFFFFFFFFFFFFF），后续读取
194 // 会被 NaNBox::tag() 误判为合法 FLOAT（NaN），掩盖底层 UB。
195 // PERF: std::fill_n 替代手动循环，MSVC 可向量化为 rep stosd/AVX 批量写入。
196 std::fill_n(data_, CAPACITY, Value::nullValue());
197 }
198
199 // ---- 容量查询 ----
200 size_t size() const { return top_; }
201 bool empty() const { return top_ == 0; }
202 static constexpr size_t capacity() { return CAPACITY; }
203
204 // ---- 元素访问 ----
205 // B6/P0 fix: assert 在 Release 构建被剥离，越界读会读到未初始化栈内存
206 // （可能形成 0xFFFFFFFFFFFFFFFF 位模式，被 NaNBox 误判为合法 NaN float）。
207 // 改为运行时 abort，与 RegisterVM::reg() 的 B3 fix 风格一致——显式失败优于静默继续。
208 Value& operator[](size_t i) {
209 if (i >= top_) [[unlikely]] {
210 std::abort();
211 }
212 return data_[i];
213 }
214 const Value& operator[](size_t i) const {
215 if (i >= top_) [[unlikely]] {
216 std::abort();
217 }
218 return data_[i];
219 }
220 Value& back() {
221 if (top_ == 0) [[unlikely]] {
222 std::abort();
223 }
224 return data_[top_ - 1];
225 }
226 const Value& back() const {
227 if (top_ == 0) [[unlikely]] {
228 std::abort();
229 }
230 return data_[top_ - 1];
231 }
232
233 // ---- 栈操作 ----
234 void push_back(const Value& v) {
235 if (top_ >= CAPACITY) [[unlikely]] {
236 std::abort();
237 }
238 data_[top_++] = v;
239 }
240 void push_back(Value&& v) {
241 if (top_ >= CAPACITY) [[unlikely]] {
242 std::abort();
243 }
244 data_[top_++] = std::move(v);
245 }
246 template <typename... Args> void emplace_back(Args&&... args) {
247 if (top_ >= CAPACITY) [[unlikely]] {
248 std::abort();
249 }
250 data_[top_++] = Value(std::forward<Args>(args)...);
251 }
252 void pop_back() {
253 if (top_ == 0) [[unlikely]] {
254 std::abort();
255 }
256 --top_;
257 // PERF/MEM: 立即释放弹出槽位的引用，避免 RefCounted 对象延迟释放。
258 // 若该槽位是最后一个引用，则立即触发析构/回收，降低内存峰值。
259 // 开销：一次 Value 赋值（NaN-boxing 仅写 8 字节 + 条件 release）。
260 data_[top_] = Value::nullValue();
261 }
262
263 // ---- 批量操作 ----
264 void clear() { top_ = 0; }
265 // B6/P0 fix: resize 仅能缩小，原 assert Release 被剥离可能导致 top_ 虚增
266 // 读到未初始化槽位（栈垃圾），改为运行时 abort。
267 void resize(size_t n) {
268 if (n > top_) [[unlikely]] {
269 std::abort();
270 }
271 // PERF/MEM: 批量缩小时释放被弹出槽位的引用，与 pop_back 语义一致。
272 for (size_t i = n; i < top_; ++i) {
273 data_[i] = Value::nullValue();
274 }
275 top_ = n;
276 }
277 /// no-op：定长数组无需预分配
278 void reserve(size_t) {}
279
280 /// 转换为 vector（用于 GUI 调试视图，按值返回）
281 std::vector<Value> toVector() const { return std::vector<Value>(data_, data_ + top_); }
282
283private:
284 static constexpr size_t CAPACITY = RuntimeLimits::MAX_STACK_SIZE;
285 Value data_[CAPACITY];
286 size_t top_;
287};
