19/// 计算两个字符串的 Levenshtein 编辑距离（插入/删除/替换各计1次）
20/// 使用单行滚动数组优化，空间复杂度 O(min(m,n))
21inline int editDistance(std::string_view a, std::string_view b) {
22 const int m = static_cast<int>(a.size());
23 const int n = static_cast<int>(b.size());
24 if (m == 0)
25 return n;
26 if (n == 0)
27 return m;
28
29 // 确保 a 是较短的串，节省空间（滚动数组只需长度+1 的一维向量）
30 if (m > n)
31 return editDistance(b, a);
32
33 std::vector<int> prev(n + 1);
34 std::vector<int> curr(n + 1);
35
36 // 边界初始化：prev[j] 表示将空串 a 变为 b[0..j] 需 j 次插入
37 for (int j = 0; j <= n; ++j)
38 prev[j] = j;
39
40 for (int i = 1; i <= m; ++i) {
41 // curr[0] 表示将 a[0..i] 变为空串需 i 次删除
42 curr[0] = i;
43 for (int j = 1; j <= n; ++j) {
44 // cost：本字符相同则无需替换（0），否则计一次替换（1）
45 int cost = (a[i - 1] == b[j - 1]) ? 0 : 1;
46 // 经典递推：取 删除(prev[j]+1) / 插入(curr[j-1]+1) / 替换(prev[j-1]+cost)
47 // 三者最小值。prev 为上一行结果，curr[j-1] 为当前行已算的左邻。
48 curr[j] = std::min({prev[j] + 1, // 删除
49 curr[j - 1] + 1, // 插入
50 prev[j - 1] + cost}); // 替换
51 }
52 // 滚动窗口：交换后 curr 成为下一轮的 prev，旧 prev 复用为下一轮 curr，
53 // 仅需两行向量即可完成整张 DP 表，空间 O(n)。
54 std::swap(prev, curr);
55 }
56 return prev[n];
57}

59/// 在候选词集合中查找与 misspelled 最接近的词
60/// maxDistance: 最大允许编辑距离（默认2），超过则不推荐
61/// 返回空字符串表示无合适建议
62inline std::string findClosest(std::string_view misspelled, const std::vector<std::string>& candidates,
63 int maxDistance = 2) {
64 if (misspelled.empty() || candidates.empty())
65 return {};
66
67 // 特殊处理：空串或单字符不纠错
68 if (misspelled.size() <= 1)
69 return {};
70
71 std::string best;
72 int bestDist = maxDistance + 1;
73
74 for (const auto& cand : candidates) {
75 // 大小写不敏感比较时，编辑距离可能更小
76 // 但为简单起见，直接比较（MiniLang 关键字全小写）
77 int dist = editDistance(misspelled, cand);
78
79 // 额外规则：候选词长度差异过大时施加惩罚。
80 // 原因：纯编辑距离对"长度悬殊但少数字符不同"的词不敏感
81 // （如 it→function 只有 1 次替换+6 次插入=7，但实际显然不是拼写纠错目标）。
82 // 当 |长度差|>2 时按超出部分累加惩罚，压低长候选的优先级，
83 // 使建议更贴近用户的真实意图（短词纠短词）。
84 int lenDiff = static_cast<int>(cand.size()) - static_cast<int>(misspelled.size());
85 if (std::abs(lenDiff) > 2)
86 dist += std::abs(lenDiff) - 2;
87
88 // 距离更优直接采纳；距离相等时偏向长度相同的候选（更可能是同一词的不同写法）
89 if (dist < bestDist || (dist == bestDist && cand.size() == misspelled.size())) {
90 bestDist = dist;
91 best = cand;
92 }
93 }
94
95 // 如果最佳距离仍然太大，不推荐
96 if (bestDist > maxDistance)
97 return {};
98 return best;
99}

101/// 构建建议消息：如果找到建议，返回 "，你是不是想写"xxx"？"，否则返回空串
102inline std::string suggestSuffix(std::string_view misspelled, const std::vector<std::string>& candidates,
103 int maxDistance = 2) {
104 auto suggestion = findClosest(misspelled, candidates, maxDistance);
105 if (suggestion.empty() || suggestion == misspelled)
106 return {};
107 return "，你是不是想写\"" + suggestion + "\"？";
108}
