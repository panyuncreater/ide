187/// 构造异常流转面板：初始化场景页与阶段页。
188ExceptionFlowPanel::ExceptionFlowPanel(QWidget* parent) : QWidget(parent) {
189 auto* mainLayout = new QVBoxLayout(this);
190 mainLayout->setContentsMargins(0, 0, 0, 0);
191 mainLayout->setSpacing(0);
192
193 // 顶部页面切换按钮
194 auto* topBar = new QHBoxLayout();
195 topBar->setContentsMargins(4, 4, 4, 4);
196 topBar->setSpacing(8);
197 pageScenarioBtn_ = new QPushButton("教学场景库", this);
198 pagePhaseBtn_ = new QPushButton("传播图解", this);
199 pageScenarioBtn_->setCheckable(true);
200 pagePhaseBtn_->setCheckable(true);
201 pageScenarioBtn_->setChecked(true);
202 topBar->addWidget(pageScenarioBtn_);
203 topBar->addWidget(pagePhaseBtn_);
204 topBar->addStretch();
205 mainLayout->addLayout(topBar);
206
207 stack_ = new QStackedWidget(this);
208 mainLayout->addWidget(stack_);
209
210 // 子页 1：教学场景库
211 auto* scenarioPage = new QWidget(stack_);
212 buildScenarioPage(scenarioPage);
213 stack_->addWidget(scenarioPage);
214
215 // 子页 2：传播图解
216 auto* phasePage = new QWidget(stack_);
217 buildPhasePage(phasePage);
218 stack_->addWidget(phasePage);
219
220 // 顶部按钮切换
221 connect(pageScenarioBtn_, &QPushButton::clicked, this, [this]() {
222 stack_->setCurrentWidget(stack_->widget(0));
223 pageScenarioBtn_->setChecked(true);
224 pagePhaseBtn_->setChecked(false);
225 PanelAnimator::slideInWidget(stack_->currentWidget());
226 });
227 connect(pagePhaseBtn_, &QPushButton::clicked, this, [this]() {
228 stack_->setCurrentWidget(stack_->widget(1));
229 pageScenarioBtn_->setChecked(false);
230 pagePhaseBtn_->setChecked(true);
231 PanelAnimator::slideInWidget(stack_->currentWidget());
232 });
233
234 // 默认选中第一个
235 if (!ExceptionFlowLibrary::scenarios().empty()) {
236 scenarioList_->setCurrentRow(0);
237 }
238 if (!ExceptionPhaseLibrary::phases().empty()) {
239 phaseList_->setCurrentRow(0);
240 }
241}
