23// ============================================================
24// NaN-boxing 常量（R142 阶段 3a：与 interpreter/NaNBox.h 保持一致）
25// ============================================================
26// JIT 代码通过硬编码这些常量生成机器码，无法直接引用 NaNBox.h 的 private 常量。
27// 修改 NaNBox.h 时必须同步更新此处。static_assert 验证一致性（见 JIT.cpp 文件末尾）。
28inline constexpr uint64_t JIT_INT_TAG_BASE = 0x7FF8000000000000ULL; ///< INT tag 基址
29inline constexpr uint64_t JIT_BOOL_TAG_BASE = 0x7FF9000000000000ULL; ///< BOOL tag 基址
30inline constexpr uint64_t JIT_NULL_BITS = 0x7FFA000000000000ULL; ///< NULL 完整 64 位表示
31inline constexpr uint64_t JIT_PTR_TAG_BASE = 0x7FFB000000000000ULL; ///< POINTER tag 基址
32inline constexpr uint64_t JIT_TAG_FIELD_MASK = 0xFFFF000000000000ULL; ///< 高 16 位 tag 提取掩码
33inline constexpr uint64_t JIT_INT48_MASK = 0x0000FFFFFFFFFFFFULL; ///< 低 48 位 payload 提取掩码
34inline constexpr uint64_t JIT_INT47_SIGN_BIT = 0x0000800000000000ULL; ///< int48 符号位（bit 47）
35// R143 阶段 3b：浮点 NaN 规范化标记（与 NaNBox.h NAN_BOXED_FLOAT_MARKER 一致）
36// 当 double 的位模式落入 boxed NaN 范围（tag field 在 0x7FF8..0x7FFB）时，
37// 用此值替代以避免与 INT/BOOL/NULL/POINTER tag 冲突。
38inline constexpr uint64_t JIT_NAN_BOXED_FLOAT_MARKER = 0x7FFC000000000000ULL;
39
40// ============================================================
41// NaN-boxing 编码工具函数
42// ============================================================
43
44/// 编译期将 int64_t 编码为 NaN-boxing INT（仅支持 int48 范围）
45/// @return 编码后的 64 位原始位；超 int48 范围返回 false（调用方报错）
46inline bool encodeNanBoxInt(int64_t value, uint64_t& outEncoded) {
47 // int48 范围：-(2^47) <= value < 2^47
48 if (value < -(1LL << 47) || value >= (1LL << 47)) {
49 return false;
50 }
51 outEncoded = JIT_INT_TAG_BASE | (static_cast<uint64_t>(value) & JIT_INT48_MASK);
52 return true;
53}
54
55/// R143 阶段 3b：编译期将 double 编码为 NaN-boxing FLOAT
56/// 直接存储 double 的 64 位原始位，若落入 boxed NaN 范围则替换为 NAN_BOXED_FLOAT_MARKER
57/// @return 编码后的 64 位原始位
58inline uint64_t encodeNanBoxFloat(double value) {
59 uint64_t bits;
60 std::memcpy(&bits, &value, sizeof(double));
61 // 检查是否落入 boxed NaN 范围（tag field 在 0x7FF8..0x7FFB）
62 uint64_t tagField = bits & JIT_TAG_FIELD_MASK;
63 if (tagField >= JIT_INT_TAG_BASE && tagField <= JIT_PTR_TAG_BASE) {
64 return JIT_NAN_BOXED_FLOAT_MARKER;
65 }
66 return bits;
67}
68
69/// R143 阶段 3b：检查 raw bits 是否是 INT 类型（tag field == INT_TAG_BASE）
70inline bool isIntBits(uint64_t bits) {
71 return (bits & JIT_TAG_FIELD_MASK) == JIT_INT_TAG_BASE;
72}

74// ============================================================
75// Value 位转换工具
76// ============================================================
77// R143 修复：Value 不是 POD（有用户定义的拷贝/析构管理引用计数），不能直接 memcpy。
78// 必须使用 Value::fromBitsBorrowed / Value::detach 这对"借用-转移"语义接口：
79// - bitsToValue（fromBitsBorrowed）：从 raw bits 构造 Value，对指针类型 addRef，
80// 使新 Value 在 RAII 下正确管理引用计数（析构会 release）。
81// - valueToBits（detach）：提取 Value 的 raw bits 并转移所有权（将 Value 置 null 避免析构 release）。
82// 调用方拿到的 raw bits 成为唯一引用，必须最终通过 fromBitsBorrowed 恢复为 Value（或由 JIT 栈持有）。
83// 这避免了两个陷阱：
84// 1. memcpy 构造 Value 不 addRef → 析构 release 导致引用计数下溢（UAF）
85// 2. memcpy 提取 bits 后 Value 析构 release → 返回的 raw bits 是悬垂指针（UAF）
86// 详见 Value.h 中 fromBitsBorrowed/detach 的注释。
87
88inline Value bitsToValue(uint64_t bits) {
89 return Value::fromBitsBorrowed(bits);
90}
91
92inline uint64_t valueToBits(Value& v) {
93 return Value::detach(v);
94}
95
96/// R143 修复：从表达式构造 Value 并转移所有权到 raw bits。
97/// 用于辅助函数返回堆类型（STRING/ARRAY/...）时，避免 Value 析构释放对象。
98/// 用法：`return jitReturn(Value(std::move(concat)));`
99/// 标量类型（INT/FLOAT/BOOL/NULL）也安全：detach 只是复制 bits 并置 null，无引用计数副作用。
100template <typename T> inline uint64_t jitReturn(T&& v) {
101 Value tmp(std::forward<T>(v));
102 return valueToBits(tmp);
103}
