186/// 执行一次查找（forward 指定方向），返回是否找到。
187bool FindReplacePanel::findText(bool forward) {
188 QString text = findEdit_->text();
189 if (text.isEmpty())
190 return false;
191
192 QTextDocument::FindFlags flags;
193 if (caseSensitiveCheck_->isChecked())
194 flags |= QTextDocument::FindCaseSensitively;
195 if (wholeWordCheck_->isChecked())
196 flags |= QTextDocument::FindWholeWords;
197 if (!forward)
198 flags |= QTextDocument::FindBackward;
199
200 QTextCursor cursor = editor_->textCursor();
201 QTextCursor found = editor_->document()->find(text, cursor, flags);
202
203 if (found.isNull()) {
204 // 到达末尾/开头，回绕查找
205 if (forward) {
206 cursor.movePosition(QTextCursor::Start);
207 } else {
208 cursor.movePosition(QTextCursor::End);
209 }
210 found = editor_->document()->find(text, cursor, flags);
211 }
212
213 if (!found.isNull()) {
214 editor_->setTextCursor(found);
215 statusLabel_->setText("");
216 return true;
217 } else {
218 statusLabel_->setText("未找到");
219 statusLabel_->setStyleSheet(QString("color: %1;").arg(TeachingTheme::error().name()));
220 return false;
221 }
222}

234/// 执行单次替换当前匹配项。
235void FindReplacePanel::onReplace() {
236 QString findTextStr = findEdit_->text();
237 QString replaceTextStr = replaceEdit_->text();
238 if (findTextStr.isEmpty())
239 return;
240
241 QTextCursor cursor = editor_->textCursor();
242 // 如果当前有选中文本且匹配查找内容，替换它
243 QString selected = cursor.selectedText();
244 // P2 fix: 正确处理大小写敏感和全字匹配选项
245 bool match = false;
246 if (caseSensitiveCheck_->isChecked()) {
247 match = (selected == findTextStr);
248 } else {
249 match = (selected.compare(findTextStr, Qt::CaseInsensitive) == 0);
250 }
251 // 全字匹配检查：选中文本前后字符不能是单词字符
252 if (match && wholeWordCheck_ && wholeWordCheck_->isChecked()) {
253 int selStart = cursor.selectionStart();
254 int selEnd = cursor.selectionEnd();
255 QTextDocument* doc = editor_->document();
256 // 检查前一个字符
257 if (selStart > 0) {
258 QChar prevChar = doc->characterAt(selStart - 1);
259 if (prevChar.isLetterOrNumber() || prevChar == '_') {
260 match = false;
261 }
262 }
263 // 检查后一个字符
264 if (match && selEnd < doc->characterCount()) {
265 QChar nextChar = doc->characterAt(selEnd);
266 if (nextChar.isLetterOrNumber() || nextChar == '_') {
267 match = false;
268 }
269 }
270 }
271
272 if (match) {
273 // BUG-FR-4 fix: 用 beginEditBlock/endEditBlock 包裹替换操作。
274 // 原实现直接 insertText 会产生独立的撤销步骤，替换 + 查找下一个
275 // 无法作为单步撤销。包裹后替换为一个原子编辑操作，Ctrl+Z 一次撤销。
276 cursor.beginEditBlock();
277 cursor.insertText(replaceTextStr);
278 cursor.endEditBlock();
279 }
280 // 查找下一个
281 findText(true);
282}

284/// 替换全部匹配项并刷新。
285void FindReplacePanel::onReplaceAll() {
286 QString findTextStr = findEdit_->text();
287 QString replaceTextStr = replaceEdit_->text();
288 if (findTextStr.isEmpty())
289 return;
290
291 QTextDocument::FindFlags flags;
292 if (caseSensitiveCheck_->isChecked())
293 flags |= QTextDocument::FindCaseSensitively;
294 if (wholeWordCheck_->isChecked())
295 flags |= QTextDocument::FindWholeWords;
296
297 // AUDIT-BUG-E7 fix: beginEditBlock/endEditBlock 必须在同一 cursor 上调用。
298 // 原实现 cursor=found 后 endEditBlock 在新 cursor 上调用，违反 Qt API 契约。
299 // 修复：用单独的 editCursor 管理 edit block，搜索用 found cursor。
300 QTextCursor editCursor(editor_->document());
301 editCursor.beginEditBlock();
302
303 int replaceCount = 0;
304 QTextCursor searchCursor(editor_->document());
305 searchCursor.movePosition(QTextCursor::Start);
306 // P2 fix: 限制替换数量上限，防止超大文档阻塞 UI
307 // D14 fix: 上限统一引用 RuntimeLimits::MAX_REPLACE_ALL
308 QTextCursor found; // BUG-FR-2: 提升到循环外，用于退出后判断是否达上限
309 while (replaceCount < RuntimeLimits::MAX_REPLACE_ALL) {
310 found = editor_->document()->find(findTextStr, searchCursor, flags);
311 if (found.isNull())
312 break;
313 searchCursor = found;
314 searchCursor.insertText(replaceTextStr);
315 replaceCount++;
316 }
317 editCursor.endEditBlock();
318
319 // BUG-FR-2 fix: 循环退出后检查 found 是否仍非 null。
320 // 若 found 非 null，说明因达到 MAX_REPLACE_ALL 上限退出（而非无更多匹配），
321 // 此时文档中可能仍有剩余匹配未替换，需提示用户。
322 if (!found.isNull()) {
323 statusLabel_->setText(QString("已替换 %1+ 处（达到上限，仍有剩余匹配）").arg(replaceCount));
324 statusLabel_->setStyleSheet(QString("color: %1;").arg(TeachingTheme::warning().name()));
325 } else {
326 statusLabel_->setText(QString("已替换 %1 处").arg(replaceCount));
327 statusLabel_->setStyleSheet(QString("color: %1;").arg(TeachingTheme::success().name()));
328 }
329 clearHighlights();
330}

332/// 高亮所有匹配项以便视觉定位。
333void FindReplacePanel::highlightMatches(const QString& text) {
334 QList<QTextEdit::ExtraSelection> selections;
335
336 QTextDocument::FindFlags flags;
337 if (caseSensitiveCheck_->isChecked())
338 flags |= QTextDocument::FindCaseSensitively;
339 if (wholeWordCheck_->isChecked())
340 flags |= QTextDocument::FindWholeWords;
341
342 QTextCursor cursor(editor_->document());
343 cursor.movePosition(QTextCursor::Start);
344
345 int count = 0;
346 while (true) {
347 QTextCursor found = editor_->document()->find(text, cursor, flags);
348 if (found.isNull())
349 break;
350 cursor = found;
351
352 QTextEdit::ExtraSelection sel;
353 sel.cursor = found;
354 sel.format.setBackground(QColor(255, 255, 0, 100)); // 半透明黄色
355 selections.append(sel);
356 count++;
357 // P2 fix: 限制高亮匹配数量，防止大文档卡顿
358 // D14 fix: 上限统一引用 RuntimeLimits::MAX_FIND_HIGHLIGHTS
359 if (count >= RuntimeLimits::MAX_FIND_HIGHLIGHTS)
360 break;
361 }
362
363 editor_->setFindSelections(selections);
364
365 if (count >= RuntimeLimits::MAX_FIND_HIGHLIGHTS) {
366 statusLabel_->setText(QString("匹配 %1+ 处").arg(count));
367 statusLabel_->setStyleSheet(QString("color: %1;").arg(TeachingTheme::textSecondary().name()));
368 } else if (count > 0) {
369 statusLabel_->setText(QString("匹配 %1 处").arg(count));
370 statusLabel_->setStyleSheet(QString("color: %1;").arg(TeachingTheme::textSecondary().name()));
371 } else {
372 statusLabel_->setText("未找到");
373 statusLabel_->setStyleSheet(QString("color: %1;").arg(TeachingTheme::error().name()));
374 }
375}
