23// 各优化 pass 共同依赖的 IR 表示约定（理解下方算法的前提）：
24// · IR 以基本块（IRBlock::instructions）为单位，函数 IRFunction 持有若干基本块；
25// 当前 pass 多为"块内"局部分析（不跨块传播），仅保守地在块边界重置摘要信息。
26// · 指令 IRInstruction 形如 op + 操作数列表 operands。操作数约定：
27// - 操作数[0]（首操作数）通常是"目标"dest vreg（纯计算/加载类指令）；
28// STORE_*/JUMP 等指令首操作数可能是 slot/label 而非 vreg。
29// - 其余操作数是源：VIRTUAL=SSA 风格虚拟寄存器(vreg，allocVReg 单调递增不复用)，
30// CONSTANT=常量池索引，LOCAL_SLOT=局部槽，LABEL=跳转标签，GLOBAL_NAME/GLOBAL_SLOT=全局。
31// · vreg 的 SSA 不变量：每个 vreg 应只被"定义"一次（见 loopUnrollingPass 的 vreg 重命名修复）。
32// 多数 pass 通过"建立 vreg→定义/值"映射并替换引用"来化简，而非修改指令语义。
33// · 折叠/化简必须保守：带运行时副作用的指令（除零、溢出、LOAD_GLOBAL 未定义、字符串拼接、
34// ADD 的 COW detach）不可被 DCE 随意删除，否则会抑制本应抛出的运行时错误（见 isPureCompute）。

38/// 辅助：判断指令是否为"常量加载"（LOAD_CONST/LOAD_NULL/LOAD_TRUE/LOAD_FALSE）
39/// 若是，返回 true 并输出对应的常量值
40bool isConstLoad(const IRInstruction& instr, const IRFunction& ir, Value& outVal) {
41 switch (instr.op) {
42 case IROp::LOAD_CONST:
43 if (instr.operands.size() < 2)
44 return false;
45 if (instr.operands[1].index >= ir.constants.size())
46 return false;
47 outVal = ir.constants[instr.operands[1].index];
48 return true;
49 case IROp::LOAD_NULL:
50 outVal = Value::nullValue();
51 return true;
52 case IROp::LOAD_TRUE:
53 outVal = Value(true);
54 return true;
55 case IROp::LOAD_FALSE:
56 outVal = Value(false);
57 return true;
58 default:
59 return false;
60 }
61}

63// 辅助：判断指令是否为纯计算（无副作用，仅定义 dest vreg）
64/// Phase-5 fix: 算术指令 ADD/SUB/MUL/DIV/MOD/NEGATE 现在也列为纯计算。
65/// 原 BUG-IR-DCE-1 的担忧是 `1/0;` 表达式语句会被 DCE 误删而抑制除零错误，
66/// 但 Phase-5 让 IROp::POP 携带所消费的 vreg 操作数（对齐 IR spec `POP: [src_vreg]`），
67/// DCE 看到 POP 引用该 vreg → 算术指令不再是"死代码" → 不会被删除。
68/// 仅当 CSE 使某算术指令的 dest 完全无引用（所有消费者都被重定向到首次计算的 dest）
69/// 时 DCE 才删除该冗余指令——此时首次计算已保留相同副作用，删除安全。
70bool isPureCompute(IROp op) {
71 switch (op) {
72 case IROp::LOAD_CONST:
73 case IROp::LOAD_NULL:
74 case IROp::LOAD_TRUE:
75 case IROp::LOAD_FALSE:
76 case IROp::LOAD_LOCAL:
77 // #23 fix: LOAD_GLOBAL/LOAD_UPVALUE 不列为纯计算——它们在运行时会检查
78 // 变量是否定义并可能抛"未定义的变量"错误。若 DCE 删除 dest 未引用的加载，
79 // 会抑制该错误（如 `undefinedVar;` 表达式语句本应报错却被静默删除）。
80 // LOAD_LOCAL 可保留：局部变量由编译期静态绑定，不存在运行时未定义。
81 case IROp::ADD:
82 case IROp::SUB:
83 case IROp::MUL:
84 // BUGFIX-P2 #57: DIV/MOD 可能触发运行时除零错误（副作用），
85 // 不应被DCE删除。仅在有明确证据证明除数非零时才可删除。
86 // case IROp::DIV:
87 // case IROp::MOD:
88 case IROp::NEGATE:
89 case IROp::NOT:
90 case IROp::EQ:
91 case IROp::NEQ:
92 case IROp::LT:
93 case IROp::GT:
94 case IROp::LTE:
95 case IROp::GTE:
96 // AUDIT-R6 F2 fix: DUP 从纯计算列表移除——栈式后端的 DUP 是位置性压栈指令
97 // （lowering 仅 emit OP_DUP 复制栈顶，不读 src vreg），DCE 删除后消费者
98 // （INDEX_GET 等）会弹走原始值导致栈下溢。实证：irOptimize + 两元素解构
99 // 即触发“栈下溢”（CSE 将同源 DUP 重定向后 DCE 删除第二条 DUP）。
100 return true;
101 default:
102 return false;
103 }

106/// 辅助：对两个 Value 执行算术运算，成功返回 true
107bool foldArith(IROp op, const Value& lhs, const Value& rhs, Value& result) {
108 // 整数算术
109 if (lhs.isInt() && rhs.isInt()) {
110 int64_t a = lhs.intVal(), b = rhs.intVal();
111 switch (op) {
112 // #14 fix: 整数 ADD/SUB/MUL 折叠需检查溢出（与 RegisterVM::executeArith 一致），
113 // 溢出时不折叠（返回 false），留待运行时按 OverflowCheck 路径报错。
114 // 原实现直接 a+b/a-b/a*b 在溢出时是 UB（int64_t 有符号溢出）。
115 case IROp::ADD:
116 if (OverflowCheck::addOverflow(a, b))
117 return false;
118 result = Value(a + b);
119 return true;
120 case IROp::SUB:
121 if (OverflowCheck::subOverflow(a, b))
122 return false;
123 result = Value(a - b);
124 return true;
125 case IROp::MUL:
126 if (OverflowCheck::mulOverflow(a, b))
127 return false;
128 result = Value(a * b);
129 return true;
130 case IROp::DIV:
131 if (b == 0)
132 return false; // 除零不折叠，留待运行时报错
133 if (OverflowCheck::divOverflow(a, b))
134 return false; // INT64_MIN / -1
135 result = Value(a / b);
136 return true;
137 case IROp::MOD:
138 if (b == 0)
139 return false;
140 if (OverflowCheck::modOverflow(a, b))
141 return false; // INT64_MIN % -1
142 result = Value(a % b);
143 return true;
144 default:
145 return false;
146 }
147 }
148 // 浮点算术（至少一方为 float）
149 if (lhs.isNumber() && rhs.isNumber()) {
150 double a = lhs.toDouble(), b = rhs.toDouble();
151 switch (op) {
152 case IROp::ADD:
153 result = Value(a + b);
154 return true;
155 case IROp::SUB:
156 result = Value(a - b);
157 return true;
158 case IROp::MUL:
159 result = Value(a * b);
160 return true;
161 case IROp::DIV:
162 if (b == 0.0)
163 return false;
164 result = Value(a / b);
165 return true;
166 case IROp::MOD:
167 if (b == 0.0)
168 return false;
169 // C++ fmod
170 result = Value(std::fmod(a, b));
171 return true;
172 default:
173 return false;
174 }
175 }
176 // R97 #4 fix: 字符串拼接折叠（对齐 Compiler::tryFoldBinary 字符串 BIN_ADD 分支）
177 // 字符串拼接本身无副作用（不抛异常），可安全折叠。
178 // 注意：isPureCompute 仍将 ADD 列为非纯计算（因 int/float ADD 可能溢出/除零），
179 // 但 foldArith 在常量传播阶段被调用时仅在两操作数均为常量时尝试，安全。
180 if (lhs.isString() && rhs.isString() && op == IROp::ADD) {
181 result = Value(lhs.stringVal() + rhs.stringVal());
182 return true;
183 }
184 return false;
185}

297// ---- 常量折叠 ----
298bool constantFoldingPass(IRFunction& ir) {
299 bool modified = false;
300 // 建立 vreg → 定义指令的映射（仅纯计算指令，且 dest 为首操作数）
301 // vregDefs[v] = {blockIdx, instrIdx}
302 std::unordered_map<uint32_t, std::pair<size_t, size_t>> vregDefs;
303 for (size_t bi = 0; bi < ir.blocks.size(); ++bi) {
304 for (size_t ii = 0; ii < ir.blocks[bi].instructions.size(); ++ii) {
305 const auto& instr = ir.blocks[bi].instructions[ii];
306 if (instr.operands.empty())
307 continue;
308 if (instr.operands[0].kind == IROperandKind::VIRTUAL) {
309 vregDefs[instr.operands[0].index] = {bi, ii};
310 }
311 }
312 }
313
314 // 遍历所有算术/比较/一元指令，若操作数均为常量加载则折叠
315 for (size_t bi = 0; bi < ir.blocks.size(); ++bi) {
316 for (size_t ii = 0; ii < ir.blocks[bi].instructions.size(); ++ii) {
317 auto& instr = ir.blocks[bi].instructions[ii];
318 // 二元运算：ADD/SUB/MUL/DIV/MOD/EQ/NEQ/LT/GT/LTE/GTE
319 if (instr.operands.size() == 3) {
320 IROp op = instr.op;
321 bool isArith =
322 (op == IROp::ADD || op == IROp::SUB || op == IROp::MUL || op == IROp::DIV || op == IROp::MOD);
323 bool isCompare = (op == IROp::EQ || op == IROp::NEQ || op == IROp::LT || op == IROp::GT ||
324 op == IROp::LTE || op == IROp::GTE);
325 if (!isArith && !isCompare)
326 continue;
327
328 // dest, src1, src2
329 IROperand dest = instr.operands[0];
330 IROperand src1 = instr.operands[1];
331 IROperand src2 = instr.operands[2];
332 if (src1.kind != IROperandKind::VIRTUAL || src2.kind != IROperandKind::VIRTUAL)
333 continue;
334
335 // 查找 src1/src2 的定义
336 auto it1 = vregDefs.find(src1.index);
337 auto it2 = vregDefs.find(src2.index);
338 if (it1 == vregDefs.end() || it2 == vregDefs.end())
339 continue;
340
341 // 获取定义指令
342 const auto& def1 = ir.blocks[it1->second.first].instructions[it1->second.second];
343 const auto& def2 = ir.blocks[it2->second.first].instructions[it2->second.second];
344
345 Value v1, v2;
346 if (!isConstLoad(def1, ir, v1) || !isConstLoad(def2, ir, v2))
347 continue;
348
349 // 尝试折叠
350 Value folded;
351 bool ok = isArith ? foldArith(op, v1, v2, folded) : foldCompare(op, v1, v2, folded);
352 if (!ok)
353 continue;
354
355 // 替换：原指令改为 LOAD_CONST dest, constIdx
356 uint32_t constIdx = ir.addConstant(folded);
357 instr.op = IROp::LOAD_CONST;
358 instr.operands = {dest, IROperand::constant(constIdx)};
359 // 更新 vregDefs 中 dest 的定义
360 vregDefs[dest.index] = {bi, ii};
361 modified = true;
362 }
363 // 一元运算：NEGATE/NOT
364 else if (instr.operands.size() == 2) {
365 IROp op = instr.op;
366 if (op != IROp::NEGATE && op != IROp::NOT)
367 continue;
368
369 IROperand dest = instr.operands[0];
370 IROperand src = instr.operands[1];
371 if (src.kind != IROperandKind::VIRTUAL)
372 continue;
373
374 auto it = vregDefs.find(src.index);
375 if (it == vregDefs.end())
376 continue;
377 const auto& def = ir.blocks[it->second.first].instructions[it->second.second];
378
379 Value v;
380 if (!isConstLoad(def, ir, v))
381 continue;
382
383 Value folded;
384 if (!foldUnary(op, v, folded))
385 continue;
386
387 uint32_t constIdx = ir.addConstant(folded);
388 instr.op = IROp::LOAD_CONST;
389 instr.operands = {dest, IROperand::constant(constIdx)};
390 vregDefs[dest.index] = {bi, ii};
391 modified = true;
392 }
393 }
394 }
395 return modified;
396}

398// ---- 死代码消除 ----
399bool deadCodeEliminationPass(IRFunction& ir) {
400 // 第一遍：收集所有被引用的 vreg
401 std::unordered_set<uint32_t> usedVRegs;
402 for (const auto& block : ir.blocks) {
403 for (const auto& instr : block.instructions) {
404 // dest（首操作数）不算"被引用"，只有后续指令引用才算
405 for (size_t i = 0; i < instr.operands.size(); ++i) {
406 // 对于纯计算指令，首操作数是 dest，跳过
407 // 对于 STORE_* 等指令，首操作数可能是 slot/uv_idx 而非 vreg
408 if (i == 0 && isPureCompute(instr.op))
409 continue;
410 if (instr.operands[i].kind == IROperandKind::VIRTUAL) {
411 usedVRegs.insert(instr.operands[i].index);
412 }
413 }
414 }
415 }
416
417 // 第二遍：删除 dest 未被引用的纯计算指令
418 bool modified = false;
419 for (auto& block : ir.blocks) {
420 std::vector<IRInstruction> kept;
421 kept.reserve(block.instructions.size());
422 for (auto& instr : block.instructions) {
423 if (isPureCompute(instr.op) && !instr.operands.empty() &&
424 instr.operands[0].kind == IROperandKind::VIRTUAL) {
425 uint32_t destVReg = instr.operands[0].index;
426 if (usedVRegs.find(destVReg) == usedVRegs.end()) {
427 // dest 从未被引用，删除此指令
428 modified = true;
429 continue;
430 }
431 }
432 kept.push_back(std::move(instr));
433 }
434 block.instructions = std::move(kept);
435 }
436 return modified;
437}

439// ---- 分支折叠 ----
440// 设计与安全性论证见 IR.h branchFoldingPass 声明处。实现为 3 个独立 sweep
441// （常量分支折叠 / 跳转线程化与紧邻消除 / 不可达代码删除），每轮 sweep 重建
442// 位置表（删除指令会使同块后续位置失效，分 sweep 隔离失效范围），迭代至不动点。
443bool branchFoldingPass(IRFunction& ir) {
444 bool modified = false;
445
446 // 位置表重建 helper：labelIdx → (blockIdx, instrIdx)（LABEL 指令位置）
447 auto buildLabelPos = [&ir](std::unordered_map<uint32_t, std::pair<size_t, size_t>>& labelPos) {
448 for (size_t bi = 0; bi < ir.blocks.size(); ++bi) {
449 const auto& instrs = ir.blocks[bi].instructions;
450 for (size_t ii = 0; ii < instrs.size(); ++ii) {
451 if (instrs[ii].op == IROp::LABEL && instrs[ii].operands.size() == 1 &&
452 instrs[ii].operands[0].kind == IROperandKind::LABEL) {
453 labelPos[instrs[ii].operands[0].index] = {bi, ii};
454 }
455 }
456 }
457 };
458
459 for (int iter = 0; iter < 4; ++iter) {
460 bool iterModified = false;
461
462 // Sweep 1: 常量条件分支折叠
463 {
464 std::unordered_map<uint32_t, std::pair<size_t, size_t>> labelPos;
465 buildLabelPos(labelPos);
466 // 条件 vreg 定义表：仅接受"恰好一个 VIRTUAL dest 定义"的 vreg（多定义保守跳过）
467 std::unordered_map<uint32_t, std::pair<size_t, size_t>> vregDefs;
468 std::unordered_set<uint32_t> multiDef;
469 for (size_t bi = 0; bi < ir.blocks.size(); ++bi) {
470 const auto& instrs = ir.blocks[bi].instructions;
471 for (size_t ii = 0; ii < instrs.size(); ++ii) {
472 const auto& instr = instrs[ii];
473 if (instr.op == IROp::JUMP_IF_FALSE || instr.op == IROp::JUMP || instr.op == IROp::LABEL)
474 continue; // 跳转/标签的 VIRTUAL 首操作数不是定义（JUMP_IF_FALSE 首操作数是 src）
475 if (!instr.operands.empty() && instr.operands[0].kind == IROperandKind::VIRTUAL) {
476 auto ins = vregDefs.emplace(instr.operands[0].index, std::make_pair(bi, ii));
477 if (!ins.second)
478 multiDef.insert(instr.operands[0].index);
479 }
480 }
481 }
482
483 for (size_t bi = 0; bi < ir.blocks.size(); ++bi) {
484 auto& instrs = ir.blocks[bi].instructions;
485 for (size_t ii = 0; ii < instrs.size(); ++ii) {
486 auto& instr = instrs[ii];
487 if (instr.op != IROp::JUMP_IF_FALSE || instr.operands.size() != 2 ||
488 instr.operands[0].kind != IROperandKind::VIRTUAL ||
489 instr.operands[1].kind != IROperandKind::LABEL)
490 continue;
491 uint32_t condVreg = instr.operands[0].index;
492 if (multiDef.count(condVreg))
493 continue;
494 auto defIt = vregDefs.find(condVreg);
495 if (defIt == vregDefs.end())
496 continue;
497 const auto& defInstr = ir.blocks[defIt->second.first].instructions[defIt->second.second];
498 Value condVal;
499 if (!isConstLoad(defInstr, ir, condVal) || !condVal.isBool())
500 continue;
501 auto labelIt = labelPos.find(instr.operands[1].index);
502 if (labelIt == labelPos.end())
503 continue; // 目标标签缺失（防御），保守不折叠
504
505 if (condVal.boolVal()) {
506 // 条件恒真：分支永不执行。JUMP_IF_FALSE 是 peek（零栈效应），
507 // 直接删除——fall-through 路径本身已消费条件值（POP/STORE）。
508 instrs.erase(instrs.begin() + static_cast<std::ptrdiff_t>(ii));
509 --ii;
510 } else {
511 // 条件恒假：必跳转。替换为 JUMP（零栈效应，目标路径期望
512 // 条件值留在栈上，如 LABEL 后的 POP cond / STORE cond）。
513 instr.op = IROp::JUMP;
514 instr.operands = {IROperand::label(instr.operands[1].index)};
515 }
516 iterModified = true;
517 }
518 }
519 }

521 // Sweep 2: 跳转线程化 + 跳转到紧邻 LABEL 消除
522 {
523 std::unordered_map<uint32_t, std::pair<size_t, size_t>> labelPos;
524 buildLabelPos(labelPos);
525
526 for (size_t bi = 0; bi < ir.blocks.size(); ++bi) {
527 auto& instrs = ir.blocks[bi].instructions;
528 for (size_t ii = 0; ii < instrs.size(); ++ii) {
529 auto& instr = instrs[ii];
530 if (instr.op != IROp::JUMP || instr.operands.size() != 1 ||
531 instr.operands[0].kind != IROperandKind::LABEL)
532 continue;
533 uint32_t target = instr.operands[0].index;
534 // 线程化：目标 LABEL 紧跟另一条 JUMP 时重定向（链式跟随，防环）
535 std::unordered_set<uint32_t> visited{target};
536 bool hopped = false;
537 while (true) {
538 auto t = labelPos.find(target);
539 if (t == labelPos.end())
540 break;
541 const auto& [tb, ti] = t->second;
542 const auto& tInstrs = ir.blocks[tb].instructions;
543 if (ti + 1 >= tInstrs.size())
544 break;
545 const auto& nxt = tInstrs[ti + 1];
546 if (nxt.op != IROp::JUMP || nxt.operands.size() != 1 ||
547 nxt.operands[0].kind != IROperandKind::LABEL)
548 break;
549 uint32_t l2 = nxt.operands[0].index;
550 if (l2 == target || !visited.insert(l2).second)
551 break;
552 target = l2;
553 hopped = true;
554 }
555 if (hopped) {
556 instr.operands[0] = IROperand::label(target);
557 iterModified = true;
558 }
559 // 跳转到紧邻 LABEL：fall-through 等价，删除 JUMP
560 auto t = labelPos.find(instr.operands[0].index);
561 if (t != labelPos.end() && t->second.first == bi && t->second.second == ii + 1) {
562 instrs.erase(instrs.begin() + static_cast<std::ptrdiff_t>(ii));
563 --ii;
564 iterModified = true;
565 }
566 }
567 }
568 }
569
570 // Sweep 3: 不可达代码删除（无条件转移后到下一 LABEL 之间）
571 for (auto& block : ir.blocks) {
572 auto& instrs = block.instructions;
573 size_t ii = 0;
574 while (ii < instrs.size()) {
575 IROp op = instrs[ii].op;
576 if (op == IROp::JUMP || op == IROp::RETURN || op == IROp::RETURN_NULL || op == IROp::THROW) {
577 size_t j = ii + 1;
578 while (j < instrs.size() && instrs[j].op != IROp::LABEL)
579 ++j;
580 if (j > ii + 1) {
581 instrs.erase(instrs.begin() + static_cast<std::ptrdiff_t>(ii + 1),
582 instrs.begin() + static_cast<std::ptrdiff_t>(j));
583 iterModified = true;
584 }
585 }
586 ++ii;
587 }
588 }
589
590 modified = modified || iterModified;
591 if (!iterModified)
592 break;
593 }
594 return modified;
595}

597// ---- 复制传播 ----
598// PERF-15: 寄存器式 lowering 已启用，复制传播可以安全使用。
599// 规则：
600// 1. 记录每个 vreg 的"定义"：LOAD_CONST c → 该 vreg 等价于 constant c
601// 2. 后续指令引用该 vreg 时，直接用 constant 替换
602// 3. 跨 STORE_LOCAL/STORE_GLOBAL/STORE_UPVALUE 后不再传播（变量可能被修改）
603// 4. 在基本块边界重置（保守策略，避免跨块分析复杂性）
604// 注意：复制传播替换操作数后，原 LOAD_CONST 指令的 dest 变为无引用，
605// 由后续的死代码消除 pass 删除。
606bool copyPropagationPass(IRFunction& ir) {
607 // BUGFIX-P1 (#24): 此 pass 的替换逻辑会产生非法 CONSTANT 操作数（大多数 IR 指令不支持
608 // IROperandKind::CONSTANT 操作数），保持禁用状态（所有调用点 enableCopyPropagation=false）。
609 // 正确实现应使用 vreg→vreg 传播而非直接替换为 CONSTANT。
610 bool modified = false;
611 // vreg → (常量索引, 类型) 映射
612 // 类型标记区分 LOAD_CONST / LOAD_NULL / LOAD_TRUE / LOAD_FALSE
613 enum class ConstKind { NONE, CONST, NULL_, TRUE_, FALSE_ };
614 struct ConstDef {
615 ConstKind kind = ConstKind::NONE;
616 uint32_t constIdx = 0;
617 };
618 std::unordered_map<uint32_t, ConstDef> vregConst;
619
620 for (auto& block : ir.blocks) {
621 vregConst.clear(); // 基本块边界重置
622 for (auto& instr : block.instructions) {
623 // 替换操作数中的 vreg 为等价常量
624 for (size_t i = 0; i < instr.operands.size(); ++i) {
625 if (instr.operands[i].kind == IROperandKind::VIRTUAL) {
626 auto it = vregConst.find(instr.operands[i].index);
627 if (it != vregConst.end() && it->second.kind != ConstKind::NONE) {
628 // 替换为等价常量
629 switch (it->second.kind) {
630 case ConstKind::CONST:
631 instr.operands[i] = IROperand::constant(it->second.constIdx);
632 modified = true;
633 break;
634 case ConstKind::NULL_:
635 // 不替换为 LOAD_NULL 的操作数（需要新增指令）
636 // 保持 vreg 引用，由 DCE 删除 LOAD_NULL
637 break;
638 case ConstKind::TRUE_:
639 case ConstKind::FALSE_:
640 // 同上，不替换
641 break;
642 case ConstKind::NONE:
643 break;
644 // Bug-6 同型修复：枚举扩展时静默跳过替换，留下不一致状态
645 default:
646 // P1-2 fix: assert 在 Release 构建中被剥离，改为同时 Logger::Error 留痕。
647 Logger::Error("copyPropagationPass: 未处理的 ConstKind 枚举值", "IR");
648 assert(false && "copyPropagationPass: 未处理的 ConstKind 枚举值");
649 break;
650 }
651 }
652 }
653 }
654
655 // 记录本指令的 dest 定义
656 if (!instr.operands.empty() && instr.operands[0].kind == IROperandKind::VIRTUAL) {
657 uint32_t destVReg = instr.operands[0].index;
658 switch (instr.op) {
659 case IROp::LOAD_CONST:
660 if (instr.operands.size() >= 2) {
661 vregConst[destVReg] = {ConstKind::CONST, instr.operands[1].index};
662 }
663 break;
664 case IROp::LOAD_NULL:
665 vregConst[destVReg] = {ConstKind::NULL_, 0};
666 break;
667 case IROp::LOAD_TRUE:
668 vregConst[destVReg] = {ConstKind::TRUE_, 0};
669 break;
670 case IROp::LOAD_FALSE:
671 vregConst[destVReg] = {ConstKind::FALSE_, 0};
672 break;
673 default:
674 // 其他指令定义 dest，清除等价关系
675 vregConst.erase(destVReg);
676 break;
677 }
678 }
679
680 // STORE_* 指令可能修改外部状态，清除所有等价关系（保守策略）
681 // 实际上只需清除被修改变量的等价关系，但简化处理
682 if (instr.op == IROp::STORE_LOCAL || instr.op == IROp::STORE_GLOBAL || instr.op == IROp::STORE_UPVALUE ||
683 instr.op == IROp::DEFINE_GLOBAL) {
684 // 清除局部变量的等价关系（简化：不清除，因为 vreg 是 SSA 风格）
685 // 实际上 vreg 不会被 STORE 修改，只有 LOCAL_SLOT 概念会
686 }
687 }
688 }
689 return modified;
690}

692// ---- 公共子表达式消除（CSE，局部 — 基本块内）----
693// C4: 在每个基本块内对纯计算指令做值编号，相同表达式 dest 复用。
694// 与 BUG-IR-DCE-1 的关键差异：本 pass 不删除指令，仅替换后续引用。
695// 算术指令（含副作用：除零/溢出）即使 dest 已被替换，仍保留原指令执行。
696bool commonSubexpressionEliminationPass(IRFunction& ir) {
697 bool modified = false;
698 // 值编号 key：将 (op, operands) 编码为字符串
699 auto encodeKey = [](const IRInstruction& instr) -> std::string {
700 std::string key;
701 key.push_back(static_cast<char>(instr.op));
702 key.push_back('|');
703 for (size_t idx = 0; idx < instr.operands.size(); ++idx) {
704 const auto& op = instr.operands[idx];
705 // BUGFIX-P1 (#23): 跳过 dest vreg（operands[0]）。SSA 下每条指令 dest 唯一，
706 // 若编入 key 会导致相同表达式永远无法匹配，CSE 完全失效。
707 // 与 IRSSA.cpp gvnPass 的 encodeValueKey 保持一致。
708 if (idx == 0 && op.kind == IROperandKind::VIRTUAL)
709 continue;
710 key.push_back(static_cast<char>(op.kind));
711 key.push_back(':');
712 key.append(std::to_string(op.index));
713 key.push_back(',');
714 }
715 return key;
716 };
717 // 判断指令是否可参与 CSE（纯计算 + 有 dest vreg）
718 auto isCSEable = [](const IRInstruction& instr) -> bool {
719 if (instr.operands.empty())
720 return false;
721 if (instr.operands[0].kind != IROperandKind::VIRTUAL)
722 return false;
723 // 包含算术（保留原指令执行副作用，仅替换 dest 引用）
724 switch (instr.op) {
725 case IROp::ADD:
726 case IROp::SUB:
727 case IROp::MUL:
728 case IROp::DIV:
729 case IROp::MOD:
730 case IROp::NEGATE:
731 case IROp::EQ:
732 case IROp::NEQ:
733 case IROp::LT:
734 case IROp::GT:
735 case IROp::LTE:
736 case IROp::GTE:
737 case IROp::NOT:
738 // AUDIT-R6 F2 fix: DUP 从 CSE 可参与列表移除。解构绑定为每个变量发射一条
739 // DUP，其源操作数均为同一 tuple vreg，encodeKey 跳过 dest 后两条 DUP 键完全
740 // 相同→CSE 重定向第二条 dest→DCE 删除第二条 DUP（压栈指令）→栈失衡下溢。
741 // DUP 是唯一“源 vreg 可重复”的原 CSEable 指令，移除后 Phase-5 安全前提恢复。
742 return true;
743 default:
744 return false;
745 }
746 };
747
748 for (auto& block : ir.blocks) {
749 // 块内值编号表：key → 已记录的 dest vreg
750 std::unordered_map<std::string, uint32_t> valueMap;
751 // 替换映射：被替换的 dest vreg → 替代 vreg
752 std::unordered_map<uint32_t, uint32_t> replacements;
753 // 待标记为"dest 已替换"的指令索引（保留指令以触发副作用，但 dest 不再被引用）
754 // 实际操作：在第二轮遍历中替换后续指令对该 dest 的引用
755
756 // 第一遍：构建替换映射
757 for (auto& instr : block.instructions) {
758 // 先按替换映射更新本指令的 operands（处理级联替换）
759 // BUGFIX-P2 #56: 对 CSE-able 指令（dest 在 operands[0]），从 i=1 开始
760 // 跳过 dest 操作数。dest 是定义点而非使用点，替换 dest 会破坏指令的
761 // 定义语义（冗余指令写入错误 vreg）。非 CSE-able 指令（如 JUMP_IF_FALSE/
762 // RETURN/POP）的 operands[0] 是源操作数，仍需替换。
763 size_t startIdx = isCSEable(instr) ? 1 : 0;
764 for (size_t opi = startIdx; opi < instr.operands.size(); ++opi) {
765 auto& op = instr.operands[opi];
766 if (op.kind == IROperandKind::VIRTUAL) {
767 auto it = replacements.find(op.index);
768 if (it != replacements.end()) {
769 op.index = it->second;
770 modified = true;
771 }
772 }
773 }
774
775 if (!isCSEable(instr))
776 continue;
777 // 二元/一元运算的 dest vreg
778 uint32_t destVReg = instr.operands[0].index;
779 std::string key = encodeKey(instr);
780 auto it = valueMap.find(key);
781 if (it != valueMap.end()) {
782 // 已存在相同表达式：本指令 dest 引用替换为已存在的 vreg
783 replacements[destVReg] = it->second;
784 // 不删除本指令（保留副作用），后续对该 dest 的引用会在循环开头被替换
785 } else {
786 valueMap[key] = destVReg;
787 }
788 }
789 }
790 return modified;
791}

793// ---- 循环展开（保守策略 — 仅展开常量边界的小循环）----
794// C4: 检测 `for (var i = 0; i < N; i = i + 1) { body }` 模式，N 为小整常量时展开 N 次。
795// 实现：扫描 IR 指令序列，识别循环模式（LABEL start; cond; JUMP_IF_FALSE exit; POP; body;
796// counter update; JUMP start; LABEL exit; POP），将 body 复制 N 份替换原循环。
797
798namespace {
799/// 循环展开模式匹配结果（analyzeLoopUnrollPattern 填充）
800struct LoopUnrollPattern {
801 size_t startIndex; // 模式起点（LABEL L1）的指令索引
802 size_t bodyStart; // body 起始指令索引（i+6）
803 size_t bodyEnd; // body 结束指令索引（不含，即尾部 LOAD_LOCAL 起点）
804 uint32_t counterSlot; // 计数器局部变量槽位
805 int64_t unrollCount; // 展开次数（N）
806 uint32_t startLabel; // 循环起始 label
807 uint32_t exitLabel; // 循环退出 label
808};
809
810/// 在 instrs 中从 startIndex 开始分析循环展开模式。
811/// 模式：
812/// [i+0] LABEL L1
813/// [i+1] LOAD_LOCAL slot (i)
814/// [i+2] LOAD_CONST N
815/// [i+3] LT dest, i, N
816/// [i+4] JUMP_IF_FALSE dest, L_exit
817/// [i+5] POP
818/// [i+6..i+6+bodySize-1] body
819/// [i+6+bodySize] LOAD_LOCAL slot (i)
820/// [i+6+bodySize+1] LOAD_CONST 1
821/// [i+6+bodySize+2] ADD dest, i, 1
822/// [i+6+bodySize+3] STORE_LOCAL slot, dest
823/// [i+6+bodySize+4] JUMP L1
824/// [i+6+bodySize+5] LABEL L_exit
825/// [i+6+bodySize+6] POP
826/// 匹配成功返回 true 并填充 outPattern；否则返回 false。
827bool analyzeLoopUnrollPattern(const std::vector<IRInstruction>& instrs, size_t startIndex, const IRFunction& ir,
828 LoopUnrollPattern& outPattern) {
829 constexpr int kMaxUnrollCount = 4; // 最大展开次数（避免代码膨胀）
830 constexpr size_t kMaxUnrollBodySize = 20; // body 指令数上限
831
832 const auto& labelStart = instrs[startIndex];
833 if (labelStart.op != IROp::LABEL)
834 return false;
835 uint32_t startLabel = labelStart.operands[0].index;
836
837 const auto& loadI1 = instrs[startIndex + 1];
838 if (loadI1.op != IROp::LOAD_LOCAL || loadI1.operands.size() < 2)
839 return false;
840 if (loadI1.operands[1].kind != IROperandKind::LOCAL_SLOT)
841 return false;
842 uint32_t counterSlot = loadI1.operands[1].index;
843
844 // AUDIT-R3 P1-9 fix: 回溯验证计数器初值确为常量 0——emitUnrolledLoopBody 硬编码
845 // "LOAD_CONST 0; STORE_LOCAL slot" 归零初始化，若循环实际初值非 0（如 i 从 5
846 // 开始），展开后执行次数错误（语义破坏）。向前扫描 LABEL 之前最近的
847 // "LOAD_CONST(int 0) → STORE_LOCAL(counterSlot)" 序列（允许中间隔 POP，
848 // 因 VarDecl 初始化 emit STORE_LOCAL+POP 模式），不匹配则拒绝展开。
849 {
850 size_t k = startIndex;
851 while (k > 0 && instrs[k - 1].op == IROp::POP)
852 --k;
853 if (k < 2)
854 return false;
855 const auto& initStore = instrs[k - 1];
856 const auto& initLoad = instrs[k - 2];
857 if (initStore.op != IROp::STORE_LOCAL || initStore.operands.size() < 2 ||
858 initStore.operands[0].kind != IROperandKind::LOCAL_SLOT || initStore.operands[0].index != counterSlot)
859 return false;
860 if (initLoad.op != IROp::LOAD_CONST || initLoad.operands.size() < 2 ||
861 initLoad.operands[1].kind != IROperandKind::CONSTANT)
862 return false;
863 uint32_t initConstIdx = initLoad.operands[1].index;
864 if (initConstIdx >= ir.constants.size() || !ir.constants[initConstIdx].isInt() ||
865 ir.constants[initConstIdx].intVal() != 0)
866 return false;
867 }
868
869 const auto& loadN = instrs[startIndex + 2];
870 if (loadN.op != IROp::LOAD_CONST || loadN.operands.size() < 2)
871 return false;
872 if (loadN.operands[1].kind != IROperandKind::CONSTANT)
873 return false;
874 uint32_t nConstIdx = loadN.operands[1].index;
875 if (nConstIdx >= ir.constants.size())
876 return false;
877 const Value& nVal = ir.constants[nConstIdx];
878 if (!nVal.isInt())
879 return false;
880 int64_t nInt = nVal.intVal();
881 if (nInt < 1 || nInt > kMaxUnrollCount)
882 return false;
883
884 const auto& ltInstr = instrs[startIndex + 3];
885 if (ltInstr.op != IROp::LT || ltInstr.operands.size() < 3)
886 return false;
887
888 const auto& jumpFalse = instrs[startIndex + 4];
889 if (jumpFalse.op != IROp::JUMP_IF_FALSE || jumpFalse.operands.size() < 2)
890 return false;
891 uint32_t exitLabel = jumpFalse.operands[1].index;
892
893 const auto& pop1 = instrs[startIndex + 5];
894 if (pop1.op != IROp::POP)
895 return false;
896
897 // 在剩余指令中找模式尾部：LOAD_LOCAL slot; LOAD_CONST 1; ADD; STORE_LOCAL slot; JUMP L1; LABEL L_exit; POP
898 size_t tailStart = startIndex + 6;
899 size_t bodyEnd = tailStart;
900 bool foundTail = false;
901 for (size_t probe = 0; probe < kMaxUnrollBodySize && bodyEnd + 7 <= instrs.size(); ++probe, ++bodyEnd) {
902 const auto& t0 = instrs[bodyEnd];
903 const auto& t1 = instrs[bodyEnd + 1];
904 const auto& t2 = instrs[bodyEnd + 2];
905 const auto& t3 = instrs[bodyEnd + 3];
906 const auto& t4 = instrs[bodyEnd + 4];
907 const auto& t5 = instrs[bodyEnd + 5];
908 const auto& t6 = instrs[bodyEnd + 6];
909 if (t0.op != IROp::LOAD_LOCAL || t0.operands.size() < 2)
910 continue;
911 if (t0.operands[1].kind != IROperandKind::LOCAL_SLOT)
912 continue;
913 if (t0.operands[1].index != counterSlot)
914 continue;
915 if (t1.op != IROp::LOAD_CONST)
916 continue;
917 if (t2.op != IROp::ADD)
918 continue;
919 if (t3.op != IROp::STORE_LOCAL || t3.operands.size() < 2)
920 continue;
921 if (t3.operands[0].kind != IROperandKind::LOCAL_SLOT)
922 continue;
923 if (t3.operands[0].index != counterSlot)
924 continue;
925 if (t4.op != IROp::JUMP || t4.operands.size() < 1)
926 continue;
927 if (t4.operands[0].kind != IROperandKind::LABEL)
928 continue;
929 if (t4.operands[0].index != startLabel)
930 continue;
931 if (t5.op != IROp::LABEL || t5.operands.size() < 1)
932 continue;
933 if (t5.operands[0].index != exitLabel)
934 continue;
935 if (t6.op != IROp::POP)
936 continue;
937 foundTail = true;
938 break;
939 }
940 if (!foundTail)
941 return false;
942
943 // body 必须不包含 break/continue/return/throw
944 // AUDIT-R3 P1-9 fix: 排除清单扩充——体内嵌套 LABEL/条件跳转（嵌套控制流）
945 // 与对计数器槽位的再写入（体内修改 i）同样会被错误展开，一并拒绝。
946 for (size_t j = tailStart; j < bodyEnd; ++j) {
947 const auto& bodyInstr = instrs[j];
948 IROp op = bodyInstr.op;
949 if (op == IROp::RETURN || op == IROp::RETURN_NULL || op == IROp::THROW || op == IROp::JUMP ||
950 op == IROp::LABEL || op == IROp::JUMP_IF_FALSE) {
951 return false;
952 }
953 if (op == IROp::STORE_LOCAL && !bodyInstr.operands.empty() &&
954 bodyInstr.operands[0].kind == IROperandKind::LOCAL_SLOT && bodyInstr.operands[0].index == counterSlot) {
955 return false; // 体内写计数器，展开不安全
956 }
957 }
958
959 outPattern.startIndex = startIndex;
960 outPattern.bodyStart = tailStart;
961 outPattern.bodyEnd = bodyEnd;
962 outPattern.counterSlot = counterSlot;
963 outPattern.unrollCount = nInt;
964 outPattern.startLabel = startLabel;
965 outPattern.exitLabel = exitLabel;
966 return true;
967}

969/// 发射展开后的循环体到 newInstrs。
970/// 展开：生成 N 份 body 副本 + 计数器初始化（LOAD_CONST 0; STORE_LOCAL slot）+ 每份迭代后的计数器递增。
971///
972/// BUG-IR-OPT-AUDIT-5 fix: vreg 重命名。
973/// 原实现直接复制 body 指令（含 operands）N 次，导致同一 vreg 在多个迭代中
974/// 被重复定义（每个迭代的 ADD dest 都用同一个 vreg），违反 IR 的 SSA-like
975/// 不变量（每个 vreg 应只被赋值一次）。后续 CSE/DCE 等优化 pass 会因重复
976/// 定义而误判（如 DCE 看到 dest 被多个指令定义时无法正确删除无引用指令，
977/// CSE 的 valueMap 会因 dest vreg 已被覆盖而误命中）。
978/// 修复：对 iter >= 1 的副本，收集 body 内定义的 vreg（作为纯计算指令的 dest
979/// 或 LOAD_LOCAL/LOAD_CONST 的 dest），分配 fresh vreg 并重命名副本中所有引用
980/// （包括 dest 和 src），保证每个迭代的 vreg 唯一。
981void emitUnrolledLoopBody(std::vector<IRInstruction>& newInstrs, const std::vector<IRInstruction>& instrs,
982 const LoopUnrollPattern& pattern, IRFunction& ir) {
983 const size_t tailStart = pattern.bodyStart;
984 const size_t bodyEnd = pattern.bodyEnd;
985 const size_t bodySize = bodyEnd - tailStart;

995 newInstrs.emplace_back(IROp::STORE_LOCAL,

997 instrs[pattern.startIndex].line);

1003 continue;
1004 if (instr.operands[0].kind == IROperandKind::VIRTUAL) {
1005 bodyDefinedVRegs.insert(instr.operands[0].index);
1006 }
1007 }

1010 uint32_t oneConst = ir.addConstant(Value(static_cast<int64_t>(1)));
1011 // 生成 nInt 份 body
1012 for (int64_t iter = 0; iter < nInt; ++iter) {
1013 // BUG-IR-OPT-AUDIT-5: 为本次迭代构建 vreg 重命名表
1014 // iter 0 用原 vreg（与原 body 一致，保持 SSA 单赋值），
1015 // iter >= 1 分配 fresh vreg 替换 body 内定义的 vreg，避免重复赋值。
1016 std::unordered_map<uint32_t, uint32_t> vregRemap;
1017 if (iter > 0) {
1018 for (uint32_t v : bodyDefinedVRegs) {
1019 vregRemap[v] = ir.nextVReg++;
1020 }
1021 }
1022 for (size_t j = tailStart; j < bodyEnd; ++j) {
1023 IRInstruction copy = instrs[j]; // 复制指令（含 operands）
1024 // 重命名所有 VIRTUAL 操作数（dest 和 src 都需重命名）
1025 for (auto& op : copy.operands) {
1026 if (op.kind == IROperandKind::VIRTUAL) {
1027 auto it = vregRemap.find(op.index);
1028 if (it != vregRemap.end()) {
1029 op.index = it->second;
1030 }
1031 }
1032 }
1033 newInstrs.push_back(std::move(copy));
1034 }
1035 // 计数器递增：LOAD_LOCAL slot; LOAD_CONST 1; ADD; STORE_LOCAL slot
1036 // Bug #85 fix: 复用循环外注册的 oneConst
1037 uint32_t iReg = ir.nextVReg++;
1038 uint32_t oneReg = ir.nextVReg++;
1039 uint32_t sumReg = ir.nextVReg++;
1040 newInstrs.emplace_back(IROp::LOAD_LOCAL, IROperandList{IROperand::vreg(iReg), IROperand::local(counterSlot)},
1041 instrs[bodyEnd].line);
1042 newInstrs.emplace_back(IROp::LOAD_CONST, IROperandList{IROperand::vreg(oneReg), IROperand::constant(oneConst)},
1043 instrs[bodyEnd].line);
1044 newInstrs.emplace_back(IROp::ADD,
1045 IROperandList{IROperand::vreg(sumReg), IROperand::vreg(iReg), IROperand::vreg(oneReg)},
1046 instrs[bodyEnd].line);
1047 newInstrs.emplace_back(IROp::STORE_LOCAL, IROperandList{IROperand::local(counterSlot), IROperand::vreg(sumReg)},
1048 instrs[bodyEnd].line);
1049 }
