19std::string TraceSnapshot::toJson() const {
20 std::ostringstream os;
21 os << "{";
22 os << "\"step\":" << step << ",";
23 os << "\"backend\":\"";
24 switch (backend) {
25 case TraceBackend::Interpreter:
26 os << "interpreter";
27 break;
28 case TraceBackend::StackVM:
29 os << "stackvm";
30 break;
31 case TraceBackend::RegisterVM:
32 os << "registervm";
33 break;
34 }
35 os << "\",";
36 os << "\"line\":" << line << ",";
37 os << "\"column\":" << column << ",";
38 os << "\"opOrNodeName\":\"" << opOrNodeName << "\",";
39 os << "\"astNodePath\":\"" << astNodePath << "\",";
40 os << "\"ip\":" << ip << ",";
41 os << "\"frameCount\":" << frameCount << ",";
42
43 // 栈快照（字符串数组）
44 os << "\"stack\":[";
45 for (size_t i = 0; i < stackStrings.size(); ++i) {
46 if (i > 0)
47 os << ",";
48 os << "\"" << stackStrings[i] << "\"";
49 }
50 os << "],";
51
52 // 寄存器快照
53 os << "\"registers\":[";
54 for (size_t i = 0; i < registerStrings.size(); ++i) {
55 if (i > 0)
56 os << ",";
57 os << "\"" << registerStrings[i] << "\"";
58 }
59 os << "],";
60
61 // 全局变量
62 os << "\"globals\":[";
63 for (size_t i = 0; i < globalsStrings.size(); ++i) {
64 if (i > 0)
65 os << ",";
66 os << "{\"name\":\"" << globalsStrings[i].first << "\",\"value\":\"" << globalsStrings[i].second << "\"}";
67 }
68 os << "],";
69
70 // 可见变量
71 os << "\"visibleVars\":[";
72 for (size_t i = 0; i < visibleVarsStrings.size(); ++i) {
73 if (i > 0)
74 os << ",";
75 os << "{\"name\":\"" << visibleVarsStrings[i].first << "\",\"value\":\"" << visibleVarsStrings[i].second
76 << "\"}";
77 }
78 os << "],";
79
80 // 调用栈
81 os << "\"callStack\":[";
82 for (size_t i = 0; i < callStack.size(); ++i) {
83 if (i > 0)
84 os << ",";
85 os << "{\"function\":\"" << callStack[i].functionName << "\",";
86 os << "\"line\":" << callStack[i].line << ",";
87 os << "\"depth\":" << callStack[i].depth << ",";
88 os << "\"locals\":[";
89 for (size_t j = 0; j < callStack[i].locals.size(); ++j) {
90 if (j > 0)
91 os << ",";
92 os << "{\"name\":\"" << callStack[i].locals[j].first << "\",\"value\":\""
93 << callStack[i].locals[j].second.toString() << "\"}";
94 }
95 os << "]}";
96 }
97 os << "]";
98
99 os << "}";
100 return os.str();
101}

107void ExecutionTraceRecorder::pushSnapshot(TraceSnapshot&& snap) {
108 // P0-3 fix: 双重检查消除 TOCTOU。
109 // 锁外 fast-path 短路避免无 push 时无谓加锁；锁内再次检查 enabled_ 防止
110 // fast-path 与获取锁之间被 setEnabled(false) + clear() 翻转的窗口。
111 // 场景：worker 线程 capture 完成（fast-path enabled_=true）→ UI 线程
112 // setEnabled(false) + clear()（清空 snapshots_/stepCounter_）→ worker
113 // 获取锁 push 一个 stale 快照到已清空的列表，破坏 clear() 语义。
114 // 锁内再次检查 enabled_=false → 丢弃，维护 clear() 不变量。
115 if (!enabled_.load(std::memory_order_relaxed))
116 return;
117 std::lock_guard<std::mutex> lock(mutex_);
118 if (!enabled_.load(std::memory_order_relaxed))
119 return;
120 snap.step = stepCounter_++;
121 if (snap.step >= maxSteps_) {
122 // ring buffer 策略：移除最旧的快照
123 if (!snapshots_.empty())
124 snapshots_.erase(snapshots_.begin());
125 // stepCounter_ 保持单调递增；UI 通过 stepAt(idx) 索引访问，
126 // 步号字段（snap.step）会超过 maxSteps_，但 vector 索引始终从 0 开始
127 }
128 snapshots_.push_back(std::move(snap));
129}

144void ExecutionTraceRecorder::captureVmStep(VmStepper& stepper, TraceBackend backend) {
145 if (!enabled_.load(std::memory_order_relaxed))
146 return;
147
148 TraceSnapshot snap;
149 snap.backend = backend;
150 snap.line = stepper.getCurrentLine();
151 snap.ip = stepper.getCurrentIP();
152 snap.opOrNodeName = stepper.getCurrentOpCodeName();
153 snap.frameCount = stepper.getFrameCount();
154
155 // 操作数栈 / 寄存器窗口
156 auto stack = stepper.getStack();
157 snap.stackStrings.reserve(stack.size());
158 for (const auto& v : stack) {
159 snap.stackStrings.push_back(valueToStringSafe(v));
160 }
161 if (backend == TraceBackend::RegisterVM) {
162 snap.registerStrings = snap.stackStrings; // RegisterVM 的 getStack 返回寄存器窗口
163 }
164
165 // 调用栈（含每帧 locals）
166 snap.callStack = stepper.getCallStack();
167
168 // 全局变量
169 auto globals = stepper.getGlobals();
170 snap.globalsStrings.reserve(globals.size());
171 for (const auto& kv : globals) {
172 snap.globalsStrings.emplace_back(kv.first, valueToStringSafe(kv.second));
173 }
174
175 // 当前帧局部变量作为"可见变量"
176 auto currentLocals = stepper.getCurrentFrameLocals();
177 snap.visibleVarsStrings.reserve(currentLocals.size());
178 for (const auto& kv : currentLocals) {
179 snap.visibleVarsStrings.emplace_back(kv.first, valueToStringSafe(kv.second));
180 }
181
182 // 阶段 3：结构化深拷贝（FullState 模式）
183 if (recordingMode_ == RecordingMode::FullState) {
184 snap.stackValues = std::move(stack);
185 if (backend == TraceBackend::RegisterVM) {
186 snap.registerValues = snap.stackValues; // 寄存器窗口与栈相同
187 }
188 for (const auto& kv : globals) {
189 snap.globalsValues.emplace_back(kv.first, kv.second);
190 }
191 for (const auto& kv : currentLocals) {
192 snap.visibleVarsValues.emplace_back(kv.first, kv.second);
193 }
194 }
195
196 pushSnapshot(std::move(snap));
197}

199void ExecutionTraceRecorder::captureInterpreterStep(Interpreter& interp, int nodeLine, int nodeColumn,
200 const std::string& nodeName, const std::string& astNodePath) {
201 if (!enabled_.load(std::memory_order_relaxed))
202 return;
203
204 TraceSnapshot snap;
205 snap.backend = TraceBackend::Interpreter;
206 snap.line = nodeLine;
207 snap.column = nodeColumn;
208 snap.opOrNodeName = nodeName;
209 snap.astNodePath = astNodePath;
210 snap.ip = 0; // Interpreter 模式无 IP 概念
211
212 // 调用栈快照（跨线程值拷贝）——先取调用栈以确定 frameCount
213 auto frames = interp.getCallStackSnapshot();
214 snap.frameCount = frames.size();
215
216 // 跨线程安全：currentEnvironmentShared 返回 shared_ptr 副本
217 auto env = interp.currentEnvironmentShared();
218 if (env) {
219 auto vars = env->allVariablesMap();
220 snap.visibleVarsStrings.reserve(vars.size());
221 for (const auto& kv : vars) {
222 snap.visibleVarsStrings.emplace_back(kv.first, valueToStringSafe(kv.second));
223 }
224 if (recordingMode_ == RecordingMode::FullState) {
225 for (const auto& kv : vars) {
226 snap.visibleVarsValues.emplace_back(kv.first, kv.second);
227 }
228 }
229 }
230 snap.callStack.reserve(frames.size());
231 for (const auto& f : frames) {
232 CallStackEntry entry;
233 entry.functionName = f.functionName;
234 entry.line = f.line;
235 entry.depth = f.depth;
236 if (f.env) {
237 auto frameVars = f.env->allVariablesMap();
238 for (const auto& kv : frameVars) {
239 entry.locals.emplace_back(kv.first, kv.second);
240 }
241 }
242 snap.callStack.push_back(std::move(entry));
243 }
244
245 // 全局环境
246 auto globalEnv = interp.getGlobalEnvironment();
247 if (globalEnv) {
248 auto globals = globalEnv->allVariablesMap();
249 snap.globalsStrings.reserve(globals.size());
250 for (const auto& kv : globals) {
251 snap.globalsStrings.emplace_back(kv.first, valueToStringSafe(kv.second));
252 }
253 if (recordingMode_ == RecordingMode::FullState) {
254 for (const auto& kv : globals) {
255 snap.globalsValues.emplace_back(kv.first, kv.second);
256 }
257 }
258 }
259
260 // L18: FullState 模式下捕获完整 Interpreter 状态快照（用于回溯调试）
261 // 类型擦除为 shared_ptr<void>，回滚时由 Interpreter::restoreFromSnapshot 消费
262 if (recordingMode_ == RecordingMode::FullState) {
263 snap.interpreterState = interp.captureStateSnapshot();
264 }
265
266 pushSnapshot(std::move(snap));
267}
