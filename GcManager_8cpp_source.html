12// SHUTDOWN-UAF fix: 进程退出阶段的静态析构顺序保护。
13// GcManager 是 Meyer's Singleton（函数局部 static），其析构发生在 main() 返回后的
14// 静态析构阶段。然而 thread_local Value（如 VM.cpp 的 nullSentinel、Value.h 的
15// tlsCloned）以及文件作用域 static 容器可能在 GcManager 之后析构，触发
16// ~RefCounted() → GcManager::instance().onDestroyed(this)——此时单例已销毁，
17// 访问其 mutex_/slots_ 成员导致"读取访问权限冲突"（this 为已释放地址）。
18//
19// 修复：用 trivially-destructible 的 atomic<bool> 标记单例存活状态。
20// std::atomic<bool> 无用户定义析构，不参与静态析构排序，进程退出时其存储
21// 始终有效（与 int 全局变量相同语义）。
22static std::atomic<bool> g_gcManagerAlive{false};
23
24// SHUTDOWN-UAF fix: 构造时标记存活，析构时清除。
25// 析构器在 main() 返回后的静态析构阶段执行（Meyer's Singleton 语义），
26// 清除标志后，后续 RefCounted 析构将跳过 onDestroyed 调用。
27GcManager::GcManager() {
28 g_gcManagerAlive.store(true, std::memory_order_release);
29}

32 g_gcManagerAlive.store(false, std::memory_order_release);

46void GcManager::registerTracked(RefCounted* obj) {
47 if (!obj)
48 return;
49 // P0-1 fix: 全程持锁，保护 slots_/freeSlots_/计数器/触发状态。
50 // checkIncrementalGc 可能触发 gcTriggerCallback_→collectCycle，collectCycle 会重入
51 // 同一把锁（recursive_mutex 安全），且 collectCycle 内 GcOnly delete 也会重入 onDestroyed。
52 std::lock_guard<std::recursive_mutex> lock(mutex_);
53 // R135 GC 模式分流：RefCountOnly 模式跳过登记，纯引用计数管理生命周期。
54 // 循环引用会泄漏（已知限制，用于基线对比与教学演示）。
55 if (gcMode_ == GcMode::RefCountOnly) {
56 return;
57 }
58 // 2026-10-06 slot-index 重构：优先复用死亡槽位（freeSlots_ 命中时零堆分配、
59 // 无哈希节点分配）；freeSlots_ 耗尽才扩容 slots_。
60 // AUDIT-P2-CORRECT 的回滚问题随之消失：唯一可抛异常点（push_back 扩容）位于
61 // 任何状态变更之前，抛出时登记表未被修改，不变量自然保持。
62 size_t slot;
63 if (!freeSlots_.empty()) {
64 slot = freeSlots_.back();
65 freeSlots_.pop_back();
66 slots_[slot] = obj;
67 } else {
68 slots_.push_back(obj);
69 slot = slots_.size() - 1;
70 }
71 obj->gcSlot_ = slot;
72 obj->gcTracked_ = true;
73 ++liveTrackedCount_;
74 // BUG-003 fix: 增量分配计数 + 阈值触发，避免长时间运行函数中循环引用累积
75 // 导致的内存峰值。
76 // P0 fix: 不再直接调用 checkIncrementalGc()——容器构造函数体内调用
77 // registerTracked 时新容器尚未加入 roots，若此时触发 GC 会被误回收。
78 // 改为设置 pendingIncrementalGc_ 标志，由 Interpreter 在语句边界
79 // （构造完成后）调用 checkPendingGc() 安全触发。
80 ++allocationsSinceLastGc_;
81 if (!gcInProgress_ && allocationsSinceLastGc_ >= gcAllocationThreshold_ && gcTriggerCallback_) {
82 pendingIncrementalGc_ = true;
83 }
84}

86void GcManager::checkIncrementalGc() {
87 // BUG-003 fix: 增量 GC 触发逻辑
88 // - 阈值未达：直接返回（O(1) 检查）
89 // - 回调未注册：直接返回（无 Interpreter 场景，如单元测试直接构造 Value）
90 // - GC 进行中：直接返回（防止回调内 collectCycle → 析构 → registerTracked → 递归）
91 // - 触发：调用回调，回调内 Interpreter 收集 roots 并调用 collectCycle
92 if (gcInProgress_)
93 return;
94 if (allocationsSinceLastGc_ < gcAllocationThreshold_)
95 return;
96 if (!gcTriggerCallback_)
97 return;
98 gcInProgress_ = true;
99 try {
100 gcTriggerCallback_();
101 } catch (...) {
102 // 回调内不应抛异常（collectCycle 不抛）；防御性捕获避免异常逃逸到构造函数
103 gcInProgress_ = false;
104 throw;
105 }
106 gcInProgress_ = false;
107}

109void GcManager::checkPendingGc() {
110 // P0 fix: 供 Interpreter 在语句边界（容器构造完成后）调用。
111 // 若 registerTracked 期间累积了待触发的增量 GC 请求，此处安全执行。
112 // PERF-GC: 无锁快速路径——本方法每条语句调用一次，pending==false 占绝对多数。
113 // relaxed 读足够：false 直接返回无误读风险；true 时进入锁内复查（取锁的
114 // acquire 语义保证看到 registerTracked 锁内写入的完整状态）。
115 if (!pendingIncrementalGc_.load(std::memory_order_relaxed))
116 return;
117 std::lock_guard<std::recursive_mutex> lock(mutex_);
118 if (pendingIncrementalGc_.load(std::memory_order_relaxed)) {
119 pendingIncrementalGc_.store(false, std::memory_order_relaxed);
120 checkIncrementalGc();
121 }

124void GcManager::onDestroyed(RefCounted* obj) {
125 // P0-1 fix: 加锁保护槽位回收。任何线程上的 RefCounted 析构都会触及此路径，
126 // 与 registerTracked/collectCycle 并发时无锁写 slots_/freeSlots_ 会破坏登记表
127 // 内部结构。collectCycle 持锁期间 delete 触发本方法时通过 recursive_mutex 安全重入。
128 std::lock_guard<std::recursive_mutex> lock(mutex_);
129 // 指针身份卫兵：仅当槽位仍被本对象占用时才回收。覆盖三类陈旧场景
130 // （等价于旧 aliveSet_.erase 对已注销指针的 no-op 语义，且天然 ABA 免疫）：
131 // 1. 本轮 sweep/Phase 3 已回收本对象的槽位（sweep 清空子元素后 refCount
132 // 归零晚于 Phase 3 压缩）；
133 // 2. reset() 已清空登记表（gcSlot_ 越界）；
134 // 3. 槽位已被新对象复用（slots_[gcSlot_] 指向他人，身份不符）。
135 if (obj->gcSlot_ < slots_.size() && slots_[obj->gcSlot_] == obj) {
136 slots_[obj->gcSlot_] = nullptr;
137 freeSlots_.push_back(obj->gcSlot_);
138 --liveTrackedCount_;
139 }
140}

142void GcManager::markValue(const Value& v, std::unordered_set<const void*>& marked,
143 std::vector<const Value*>& worklist) {
144 // AUDIT-P1-ROUND53 fix: 原 markValue 对 ARRAY/DICT/INSTANCE/CLOSURE 四种堆类型
145 // 纯递归遍历子元素。MiniLang 的 MAX_LOOP_ITERATIONS=10000000 允许 while 循环
146 // 构建极深嵌套的线性容器链（如 100000 层 [[[[...]]]]）。这条链不是循环引用
147 // （每层 refCount=1，全部从根可达），但 collectCycle Phase 1 mark 阶段仍需
148 // 从 roots 遍历全部可达对象。markValue 递归深度等于嵌套深度，100000 层远超
149 // Windows 默认 1MB 栈容量（每帧约 100-200 字节，~5000-10000 层即溢出）。
150 // 改为显式 worklist 迭代式：栈深度恒定，消除栈溢出风险，且 worklist 连续
151 // 内存访问对 cache 友好（同时解决 PERF-53-4 递归调用开销）。
152 // PERF-GC: worklist 由调用方持有（整个 collectCycle 仅 1 次 vector 分配），
153 // 原实现每个根元素各建一个 worklist，在 stop-the-world 暂停内产生 R 次分配。
154 worklist.push_back(&v);
155
156 while (!worklist.empty()) {
157 const Value* cur = worklist.back();
158 worklist.pop_back();
159
160 // 仅指针类型的 Value 需 mark（int/float/bool/null 直接跳过）
161 if (!cur->box_.isPointer())
162 continue;
163 const void* ptr = cur->box_.asPtr<void>();
164 if (!marked.insert(ptr).second)
165 continue; // 已标记，跳过避免循环
166
167 // 将子元素压入 worklist 而非递归
168 switch (cur->box_.asPtr<RefCounted>()->type) {
169 case ValueType::VAL_ARRAY: {
170 const auto& elements = cur->box_.asPtr<Value::ArrayData>()->elements;
171 for (const auto& elem : elements) {
172 worklist.push_back(&elem);
173 }
174 break;
175 }
176 case ValueType::VAL_DICT: {
177 const auto& entries = cur->box_.asPtr<Value::DictData>()->entries;
178 for (const auto& kv : entries) {
179 worklist.push_back(&kv.second);
180 }
181 break;
182 }
183 case ValueType::VAL_INSTANCE: {
184 const auto& fields = cur->box_.asPtr<Value::InstanceData>()->fields;
185 for (const auto& kv : fields) {
186 worklist.push_back(&kv.second);
187 }
188 break;
189 }
190 case ValueType::VAL_CLOSURE: {
191 // 闭包的 capturedVars 可能持有容器引用
192 const auto* closure = cur->box_.asPtr<Value::ClosureData>();
193 const auto& captured = closure->capturedVars;
194 for (const auto& kv : captured) {
195 worklist.push_back(&kv.second);
196 }
197 // AUDIT-P2-CORRECT fix: 遍历 VM 闭包的 upvalues，标记已关闭 upvalue 的 value。
198 // VMUpvalue::value 在 isClosed=true 时持有值（可能为容器引用），形成循环引用。
199 // open 状态时值在栈上（GC roots），value 字段不持有有效值，markValue 安全跳过。
200 // 原实现仅 mark capturedVars，遗漏 vmClosure->upvalues，可能导致仅通过
201 // VM 闭包 upvalues 可达的循环容器被误判为不可达孤岛而误回收。
202 if (closure->vmClosure) {
203 for (const auto& upval : closure->vmClosure->upvalues) {
204 if (upval && upval->isClosed) {
205 worklist.push_back(&upval->value);
206 }
207 }
208 }
209 // env 是 weak_ptr，不 mark（避免重新引入循环）
210 break;
211 }
212 case ValueType::VAL_TUPLE: {
213 // R98 元组与解构：元组元素可能持有可变容器形成间接环，需 mark 元素。
214 const auto& elements = cur->box_.asPtr<Value::TupleData>()->elements;
215 for (const auto& elem : elements) {
216 worklist.push_back(&elem);
217 }
218 break;
219 }
220 case ValueType::VAL_ENUM_VARIANT: {
221 // R99 枚举与 ADT：enum variant 的 fields 可能持有可变容器形成间接环，需 mark。
222 const auto& fields = cur->box_.asPtr<Value::EnumVariantData>()->fields;
223 for (const auto& f : fields) {
224 worklist.push_back(&f);
225 }
226 break;
227 }
228 case ValueType::VAL_COROUTINE: {
229 // AUDIT-R3 P1-6 fix: 协程持有 args（调用参数）/currentValueBox（最近
230 // yield 值）/vmClosureBox（闭包值），均可能引用被跟踪容器。原实现
231 // 未遍历——仅通过挂起协程可达的循环容器在 mark 阶段不可达，
232 // sweep 会清空其元素，协程恢复后拿到被掉空的容器。
233 const auto* cd = cur->box_.asPtr<Value::CoroutineData>();
234 for (const auto& a : cd->args)
235 worklist.push_back(&a);
236 for (const auto& cv : cd->currentValueBox)
237 worklist.push_back(&cv);
238 for (const auto& cb : cd->vmClosureBox)
239 worklist.push_back(&cb);
240 break;
241 }
242 case ValueType::VAL_STRING:
243 case ValueType::VAL_INT:
244 // 叶子节点，无子引用
245 break;
246 default:
247 break;
248 }
249 }
250}

252void GcManager::collectCycle(const std::vector<const void*>& roots) {
253 // P0-1 fix: 全程持锁，互斥其他线程的 registerTracked/onDestroyed/reset/统计读取。
254 // GcOnly 模式 delete 对象触发 ~RefCounted→onDestroyed 通过 recursive_mutex 重入安全。
255 std::lock_guard<std::recursive_mutex> lock(mutex_);
256 // R135 GC 模式分流：RefCountOnly 模式无 tracked 对象，直接返回。
257 if (gcMode_ == GcMode::RefCountOnly) {
258 return;
259 }
260 // 快速退出：无存活被跟踪对象时无事可做（登记表可能仍保有墓碑槽位容量）。
261 if (liveTrackedCount_ == 0)
262 return;
263
264 // R113 C 项：Phase 1: Mark - 从 roots 出发标记所有可达的容器节点
265 currentPhase_ = GcPhase::Marking;
266 std::unordered_set<const void*> marked;
267 marked.reserve(liveTrackedCount_ * 2);
268 // PERF-GC: 整个 mark 阶段共享一个 worklist（1 次分配），传引用给所有 markValue 调用。
269 std::vector<const Value*> markWorklist;
270 for (const void* rootPtr : roots) {
271 if (marked.insert(rootPtr).second) {
272 // 根据 type 遍历根指针的子元素
273 const RefCounted* rc = static_cast<const RefCounted*>(rootPtr);
274 switch (rc->type) {
275 case ValueType::VAL_ARRAY: {
276 const auto& elements = static_cast<const Value::ArrayData*>(rootPtr)->elements;
277 for (const auto& elem : elements) {
278 markValue(elem, marked, markWorklist);
279 }
280 break;
281 }
282 case ValueType::VAL_DICT: {
283 const auto& entries = static_cast<const Value::DictData*>(rootPtr)->entries;
284 for (const auto& kv : entries) {
285 markValue(kv.second, marked, markWorklist);
286 }
287 break;
288 }
289 case ValueType::VAL_INSTANCE: {
290 const auto& fields = static_cast<const Value::InstanceData*>(rootPtr)->fields;
291 for (const auto& kv : fields) {
292 markValue(kv.second, marked, markWorklist);
293 }
294 break;
295 }
296 case ValueType::VAL_CLOSURE: {
297 const auto& captured = static_cast<const Value::ClosureData*>(rootPtr)->capturedVars;
298 for (const auto& kv : captured) {
299 markValue(kv.second, marked, markWorklist);
300 }
301 break;
302 }
303 case ValueType::VAL_TUPLE: {
304 // R98 元组与解构：元组作为根时，标记其元素
305 const auto& elements = static_cast<const Value::TupleData*>(rootPtr)->elements;
306 for (const auto& elem : elements) {
307 markValue(elem, marked, markWorklist);
308 }
309 break;
310 }
311 case ValueType::VAL_ENUM_VARIANT: {
312 // R99 枚举与 ADT：enum variant 作为根时，标记其 fields
313 const auto& fields = static_cast<const Value::EnumVariantData*>(rootPtr)->fields;
314 for (const auto& f : fields) {
315 markValue(f, marked, markWorklist);
316 }
317 break;
318 }
319 case ValueType::VAL_COROUTINE: {
320 // AUDIT-R3 P1-6 fix: 协程作为根时，标记其持有的参数/yield 值/闭包值
321 const auto* cd = static_cast<const Value::CoroutineData*>(rootPtr);
322 for (const auto& a : cd->args)
323 markValue(a, marked, markWorklist);
324 for (const auto& v : cd->currentValueBox)
325 markValue(v, marked, markWorklist);
326 for (const auto& v : cd->vmClosureBox)
327 markValue(v, marked, markWorklist);
328 break;
329 }
330 default:
331 break;
332 }
333 }
334 }
335 // R113 C 项：Phase 1 结束，记录可达节点数
336 lastMarkedCount_ = marked.size();
337
338 // Phase 2: Sweep - 遍历 slots_ 登记表，对槽位被占用（非 null 墓碑）但 marked
339 // 中不存在的对象（不可达的循环孤岛）执行回收。
340 //
341 // 2026-10-06 slot-index 重构：原实现逐对象 aliveSet_.find（哈希查找）区分
342 // 存活与已析构；现登记表本身即真相源——非空槽位在锁内必为存活对象，
343 // nullptr 墓碑直接跳过。迭代安全性与旧设计等价并更强：
344 // - 清空孤岛子元素触发的级联析构（~RefCounted→onDestroyed）只会把
345 // 其他槽位置空 + push freeSlots_，从不改变 slots_.size()，
346 // 因此按下标迭代 slots_ 不受迭代器/扩容失效影响（旧实现依赖
347 // "级联析构不向 tracked_ push_back"这一偶然性质保证 range-for 安全）。
348 //
349 // R135 GC 模式分流：
350 // RefCountWithCycleGc (默认): 清空子元素打破循环，让 refCount 降至 0 自然释放
351 // GcOnly: 收集到 toDelete 列表，迭代结束后统一 delete
352 //
353 // 安全性说明：
354 // - 非 null 槽位即存活（全局锁内不变量），无需哈希查找即可安全 dereference。
355 // - RefCountWithCycleGc 清空 obj 的子元素会触发级联析构，其他槽位随之置空，
356 // 故后续迭代到那些槽位时 nullptr → 安全跳过。
357 // - GcOnly 模式不在迭代中 delete（避免级联析构破坏迭代不变量），
358 // 而是收集到 toDelete，迭代结束后统一 delete。delete 触发 ~RefCounted →
359 // onDestroyed → 槽位回收，安全。
360 // R113 C 项：Phase 2 开始
361 currentPhase_ = GcPhase::Sweeping;
362 size_t collectedCount = 0;
363 std::vector<RefCounted*> toDelete; // R135 GcOnly 模式：延迟 delete 列表
364 for (size_t i = 0; i < slots_.size(); ++i) {
365 RefCounted* obj = slots_[i];
366 if (!obj)
367 continue; // 已析构墓碑（旧 aliveSet_.find 失败的等价路径）
368 // 跳过从根集可达的对象
369 if (marked.find(obj) != marked.end())
370 continue;
371
372 // 不可达的循环孤岛处理：先清空子元素打破循环。
373 // R135 GcOnly 修复：原实现 GcOnly 分支直接 push 到 toDelete 跳过清空，
374 // 导致后续 delete obj 触发 ~ArrayData → ~vector<Value> → 释放元素（即 obj 自身）
375 // → release() → refCount 降为 0 → 再次 delete obj，构成 use-after-free（SEH 0xc0000005）。
376 // 修复：GcOnly 与 RefCountWithCycleGc 共享相同的 std::move 清空策略打破循环。
377 // 区别仅在收尾：RefCountWithCycleGc 依赖 refCount 自然释放（不清 toDelete），
378 // GcOnly 收集存活对象到 toDelete 在迭代结束后显式 delete（处理 refCount>1 的强制回收语义）。
379 // AUDIT-BUG-I3 fix: 先 move 出子元素到局部变量再清空，防止自引用容器
380 // （如 a.append(a)）在 clear() 期间级联析构重入 vector 析构器导致 use-after-free。
381 // move 后 obj->elements/entries/fields 为空，级联析构 obj 时其析构器看到空容器，安全。
382 // R135 GcOnly 路径：std::move 析构 tmp 时若触发级联 delete obj（仅当 refCount 来自循环引用），
383 // onDestroyed 会回收 obj 的槽位（置空），后续通过槽位检查跳过 toDelete.push_back
384 // 避免 double-free。
385 switch (obj->type) {
386 case ValueType::VAL_ARRAY: {
387 auto tmp = std::move(static_cast<Value::ArrayData*>(obj)->elements);
388 (void)tmp; // tmp 析构时若级联释放 obj，obj->elements 已空
389 break;
390 }
391 case ValueType::VAL_DICT: {
392 auto tmp = std::move(static_cast<Value::DictData*>(obj)->entries);
393 (void)tmp;
394 break;
395 }
396 case ValueType::VAL_INSTANCE: {
397 auto tmp = std::move(static_cast<Value::InstanceData*>(obj)->fields);
398 (void)tmp;
399 break;
400 }
401 case ValueType::VAL_TUPLE: {
402 // R98 元组与解构：清空元组元素打破循环（元组元素可能持有容器形成间接环）
403 auto tmp = std::move(static_cast<Value::TupleData*>(obj)->elements);
404 (void)tmp;
405 break;
406 }
407 case ValueType::VAL_ENUM_VARIANT: {
408 // R99 枚举与 ADT：清空 fields 打破循环（fields 可能持有容器形成间接环）
409 auto tmp = std::move(static_cast<Value::EnumVariantData*>(obj)->fields);
410 (void)tmp;
411 break;
412 }
413 default:
414 break;
415 }
416
417 if (gcMode_ == GcMode::GcOnly) {
418 // R135 GcOnly 模式：std::move 清空子元素后，若 obj 仅被循环引用持有，
419 // 级联析构已 delete obj 并回收其槽位（置空），跳过 toDelete.push_back 避免 double-free。
420 // 若 obj 还被循环外引用（refCount > 0），槽位仍被 obj 占用，需 push 到 toDelete
421 // 在迭代结束后显式 delete（GcOnly 语义：GC 主导生命周期，强制回收不可达对象）。
422 if (slots_[i] == obj) {
423 toDelete.push_back(obj);
424 }
425 }
426 ++collectedCount;
427 }
428
429 // R135 GcOnly 模式：迭代结束后统一 delete 不可达对象。
430 // toDelete 仅包含 std::move 清空子元素后仍存活的对象（refCount > 0，
431 // 即被循环外引用持有）。仅被循环引用持有的对象已在 std::move tmp 析构时
432 // 通过 RefCounted release 机制自然 delete，不进入 toDelete。
433 // delete 触发 ~RefCounted → onDestroyed → 槽位回收（置空 + 进 freeSlots_）。
434 // Phase 3 压缩登记表时会收缩掉死亡槽位。
435 if (gcMode_ == GcMode::GcOnly && !toDelete.empty()) {
436 for (RefCounted* obj : toDelete) {
437 // 安全性：obj 在 push 时确认槽位仍被其占用（未被级联析构释放）。
438 // 且所有 toDelete 条目的子元素已在 sweep 中 move 清空，delete 不会级联
439 // 析构其他条目（与旧设计相同的 R135 保证），无悬挂风险。
440 // GcOnly 语义：GC 主导生命周期，强制回收不可达对象，无视 refCount > 0。
441 // 调用方需保证无其他强引用（GcOnly 不与 COW 共享引用共存）。
442 delete obj;
443 }
444 }
445 // R113 C 项：Phase 2 结束，记录回收孤岛数
446 lastCollectedCount_ = collectedCount;
447
448 // Phase 3: 压缩登记表（原地）。
449 // BUG-INTR-AUDIT-1 fix（语义保留）：原实现无条件 tracked_.clear()，导致上一轮
450 // marked 为可达而存活的循环引用容器（如 a.push(a)）从跟踪结构中移除，且不会在
451 // 下一轮 execute 中重新 registerTracked（registerTracked 仅在容器构造时调用）。
452 // 当这些容器后来变为不可达时，sweep 阶段不会检查它们，无法打破循环，导致永久泄漏。
453 // 修复语义：仅保留本轮被标记为可达的存活容器，清除已析构墓碑与已回收的孤岛。
454 // 2026-10-06 slot-index 重构：以单趟原地压缩取代旧「survivors 重建 tracked_ +
455 // 清空重灌 aliveSet_」的双结构哈希风暴——存活对象前移并回写 gcSlot_，死亡槽位
456 // 置空并回收进 freeSlots_（仅回收本轮仍被占用的槽位；早已是墓碑的槽位下标
457 // 已在 freeSlots_ 中，不重复入栈，避免复用冲突）。
458 // R113 C 项：Phase 3 开始
459 currentPhase_ = GcPhase::Finalizing;
460 size_t w = 0;
461 for (size_t i = 0; i < slots_.size(); ++i) {
462 RefCounted* obj = slots_[i];
463 if (!obj)
464 continue; // 墓碑槽位：压缩时直接收缩掉
465 // 仅保留本轮被标记为可达的容器（下一轮可能变为不可达，需要再次检查）
466 if (marked.find(obj) != marked.end()) {
467 if (w != i) {
468 slots_[w] = obj;
469 obj->gcSlot_ = w; // 回写新下标（对象存活且持锁，安全）
470 }
471 ++w;
472 }
473 }
474 // 尾部死亡槽位：置空 + 回收（跳过已是墓碑的槽位，其下标已在 freeSlots_ 中）
475 for (size_t i = w; i < slots_.size(); ++i) {
476 if (slots_[i]) {
477 slots_[i] = nullptr;
478 freeSlots_.push_back(i);
479 }
480 }
481 liveTrackedCount_ = w;
482 // 容量策略：slots_.size() 保持历史峰值不缩容，死亡槽位经 freeSlots_ 复用，
483 // 与旧 tracked_ 的高水位行为一致（旧 Phase 3 由 survivors 重置低水位，但
484 // 下次增长重新扩容；新设计以 O(1) 复用取代重复扩容）。
485
486 if (collectedCount > 0) {
487 LOG_INFO("GcManager: 回收 " + std::to_string(collectedCount) + " 个循环引用孤岛", "GC");
488 }
489 // BUG-003 fix: 重置分配计数，下一次增量触发需累计到阈值
490 allocationsSinceLastGc_ = 0;
491 // R113 C 项：collectCycle 结束，回归 Idle 并累计 GC 次数
492 ++totalGcCount_;
493 currentPhase_ = GcPhase::Idle;
494}
