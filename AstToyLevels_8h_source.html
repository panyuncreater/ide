3// ------------------------------------------------------------
4// 通过拖拽/点击组装 AST 节点，让学习者理解「AST 是代码的结构化表示」。

6// 仅括号不同就导致完全不同的树结构（优先级 vs 括号改变结构）。
7//
8// 设计原则：
9// - 本文件仅含数据结构与静态数据接口声明，不依赖 IdeController.h
10// 或任何引擎层文件——与 BugHuntLibrary.cpp / LearningPathData.cpp
11// 的设计模式一致，便于测试目标独立链接。
12// - AstToyLevels.cpp 仅依赖 Qt6::Core（QString），不依赖 Qt6::Widgets，
13// 可安全加入 minilang_tests 测试目标（测试目标不链接 app/ 下文件链）。
14// - 静态数据嵌入 .cpp，与 BugHuntLibrary.cpp 模式一致。
15//
16// 题目设计：
17// 1. 42 — 最简：只有一个叶子节点
18// 2. 1 + 2 — 二元运算：根 + 两个叶子
19// 3. 1 + 2 * 3 — 优先级：* 在 + 下面
20// 4. (1 + 2) * 3 — 括号改变结构：+ 在 * 下面
21// 5. print(x) — 函数调用树
22// 6. var x = 1 + 2;— 完整语句（VarDecl 包含 initializer）
23//
24// 标准答案 TreeNode 标签约定：
25// - 二元运算： "BinaryOp:+" / "BinaryOp:-" / "BinaryOp:*" / "BinaryOp:/"
26// - 整数字面量："Number:<value>"（如 "Number:42"）
27// - 标识符引用："Identifier:<name>"（如 "Identifier:x"）
28// - print 调用： "Print"
29// - 变量声明： "VarDecl:<name>"（如 "VarDecl:x"）

47 // --------------------------------------------------------
48 // TreeNode — 标准答案的树结构表示
49 // --------------------------------------------------------
50 // 第一个节点是根，children 是按代码顺序排列的子节点。
51 // 叶子节点 children 为空 vector。
52 // label 命名约定见文件头注释。
53 struct TreeNode {
54 std::string label; // 节点标签
55 std::vector<TreeNode> children; // 子节点（按代码顺序）
56
57 // 默认构造：空节点
58 TreeNode() = default;
59 // 便捷构造：叶子节点
60 explicit TreeNode(std::string l) : label(std::move(l)) {}
61 // 完整构造
62 TreeNode(std::string l, std::vector<TreeNode> c) : label(std::move(l)), children(std::move(c)) {}
63 };
64
65 TreeNode answer; // 标准答案
66};
