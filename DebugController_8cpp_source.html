148bool DebugController::shouldPauseAtBreakpoint(int line, const QSet<int>& localBreakpoints,
149 const QMap<int, BreakpointInfo>& localBreakpointInfos,
150 int snapMinBreakpointLine) {
151 // C3 fix + DBG-03: 单行循环断点重触发——用 crossedLine_ 检测是否跨过不同行
152 if ((line != lastSeenLine_.load() || crossedLine_.load()) && line >= snapMinBreakpointLine &&
153 localBreakpoints.contains(line)) {
154 // AUDIT-BUG-F4 fix: crossedLine_ 重置必须推迟到断点真正命中（return true）之前。
155 // 原实现在条件求值之前重置，条件不满足时 crossedLine_ 已被清 false，
156 // 单行循环中行号不变 → crossedLine_ 永不再变 true → 条件断点首次不满足后永不再触发。
157 // 检查是否为条件断点

159 // 拓展二期：依赖断点链——依赖行断点从未命中过则本断点未激活
160 // （不计数不暂停，对 Logpoint/条件/无条件断点统一生效）。
161 // 从快照读依赖行 hitCount：同一次 checkBreak 内快照自洽，
162 // 跨语句的更新在下一次 checkBreak 快照中可见。
163 if (infoIt != localBreakpointInfos.end() && infoIt->hasDependency()) {
164 auto depIt = localBreakpointInfos.find(infoIt->dependsOnLine);
165 if (depIt == localBreakpointInfos.end() || depIt->hitCount <= 0) {
166 return false;

169 // R104 Logpoint：命中不暂停，仅输出日志并递增 hitCount
170 // Logpoint 仍可附加条件（条件为真时才输出日志）
171 if (infoIt != localBreakpointInfos.end() && infoIt->isLogpoint()) {

173 if (infoIt->isConditional()) {
174 if (evaluator_ && evaluator_->hasCallback()) {
175 shouldLog = evaluator_->evaluate(infoIt->condition, line);
176 } else {
177 shouldLog = false; // 无求值器，条件 Logpoint 视为不命中
178 }

194 }
195 if (cb) {

205 crossedLine_.store(false);
206 }
207 // Logpoint 永远不暂停
208 return false;
209 }
210 if (infoIt != localBreakpointInfos.end() && infoIt->isConditional()) {
211 // A5 fix: 委托给 DebugEvaluator 求值（异常处理 + 日志已封装）
212 if (evaluator_ && evaluator_->hasCallback()) {
213 if (evaluator_->evaluate(infoIt->condition, line)) {
214 int newCount = 0;
215 {
216 std::lock_guard<std::mutex> lock(pauseMutex_);
217 auto realIt = breakpointInfos_.find(line);
218 if (realIt != breakpointInfos_.end()) {
219 realIt->hitCount++;
220 newCount = realIt->hitCount;
221 }
222 }
223 // 拓展二期：命中条件——已计数但不满足则不暂停。
224 // crossedLine_ 仍重置：本次"到达"已消费，同行后续子表达式
225 // 不应重复计数（与命中暂停路径一致）。
226 crossedLine_.store(false);

228 return false;
229 }
230 return true;
231 }
232 // 条件不满足：保留 crossedLine_ 状态，允许下次迭代重新求值

234 } else {

237 if (infoIt != localBreakpointInfos.end()) {
238 std::lock_guard<std::mutex> lock(pauseMutex_);
239 auto realIt = breakpointInfos_.find(line);
240 if (realIt != breakpointInfos_.end()) {
241 realIt->hitCount++;

243 }
244 }
245 crossedLine_.store(false); // 命中后重置，同行后续子表达式不再触发

248 return false; // 已计数但命中条件不满足：不暂停
249 }
250 return true;
251 }
252 }

256bool DebugController::shouldPauseForStepping(StepMode snapMode, int line, int snapCurrentDepth, int snapStepOverDepth,
257 int snapStepOutDepth, int snapLastPausedLine, int snapLastPausedDepth,
258 bool snapCrossedDeeper) {
259 (void)line; // STEP_OUT 不使用 line
260 switch (snapMode) {
261 case StepMode::MODE_STEP_IN:
262 // 行号变化时暂停，或调用深度变化时暂停（递归函数同行不同深度）
263 return (line != snapLastPausedLine || snapCurrentDepth != snapLastPausedDepth);
264 case StepMode::MODE_STEP_OVER:
265 // 只暂停同一或更浅调用深度，且行号变化或从深层返回
266 if (snapCurrentDepth <= snapStepOverDepth && (line != snapLastPausedLine || snapCrossedDeeper)) {
267 return true;
268 }
269 return false;
270 case StepMode::MODE_STEP_OUT:
271 // H6 fix: 仅检查调用深度，深度变浅即已从函数返回
272 return (snapCurrentDepth < snapStepOutDepth);
273 default:
274 return false;
275 }

668bool DebugController::checkFunctionBreakpoint(const std::string& functionName, int line) {
669 // 快速路径：无函数断点时立即返回（避免每次函数调用都加锁）
670 // 注：functionBreakpoints_ 由 pauseMutex_ 保护，但读取 emptiness 需要锁。
671 // 此处先取锁内快照判断，避免漏检。
672 FunctionBreakpointInfo snapshot;
673 {
674 std::lock_guard<std::mutex> lock(pauseMutex_);
675 if (!running_ || stopped_)
676 return false;
677 auto it = functionBreakpoints_.find(functionName);
678 if (it == functionBreakpoints_.end())
679 return false;
680 snapshot = it.value();
681 }
682
683 // 条件求值（锁外，避免持锁回调死锁）
684 if (snapshot.isConditional()) {
685 if (evaluator_ && evaluator_->hasCallback()) {
686 if (!evaluator_->evaluate(snapshot.condition, line)) {
687 return false; // 条件不满足
688 }
689 } else {
690 return false; // 无求值器，条件断点视为不命中
691 }
692 }
693
694 // 命中：递增 hitCount 并暂停
695 // Bug #88 fix: 条件求值在锁外执行，期间用户可能移除该断点。
696 // 在 doPause 前必须重新检查断点是否仍存在，否则会对已删除的断点
697 // 触发暂停（语义错误，且 hitCount 写入到无效条目）。
698 bool stillExists = false;
699 {
700 std::lock_guard<std::mutex> lock(pauseMutex_);
701 auto it = functionBreakpoints_.find(functionName);
702 if (it != functionBreakpoints_.end()) {
703 it->hitCount++;
704 stillExists = true;
705 }
706 }
707 if (!stillExists) {
708 return false; // 条件求值期间断点已被移除
709 }
710 doPause(line, currentDepth_.load());
711 return true;
712}

793bool DebugController::checkWatchpointHit(const std::string& varName, bool isFieldWrite, const std::string& fieldName,
794 int line) {
795 // 快速路径：无 watchpoint 时立即返回（避免每次赋值都加锁）
796 if (!hasWatchpoints_.load(std::memory_order_relaxed))
797 return false;
798
799 // 与 VmStepper::checkWatchpointHit 匹配语义对齐：
800 // - Variable 类型：varName 完全匹配（索引写入视为修改变量本身，也匹配）
801 // - Field 类型：要求 isFieldWrite，fieldName 匹配，varName 空时通配
802 WatchpointInfo snapshot;
803 bool matched = false;
804 {
805 std::lock_guard<std::mutex> lock(pauseMutex_);
806 if (!running_ || stopped_)
807 return false;
808 for (const auto& wp : watchpoints_) {
809 if (wp.kind == WatchpointTargetKind::Variable && !isFieldWrite) {
810 // L19 audit fix: 添加 !varName.empty() 防御性保护（与 VmStepper 对齐），
811 // 避免空 varName 误匹配 varName="" 的 Variable watchpoint。
812 if (!varName.empty() && wp.varName == varName) {
813 snapshot = wp;
814 matched = true;
815 break;
816 }
817 } else if (wp.kind == WatchpointTargetKind::Field && isFieldWrite) {
818 if (wp.fieldName == fieldName && (wp.varName.empty() || wp.varName == varName)) {
819 snapshot = wp;
820 matched = true;
821 break;
822 }
823 }
824 }
825 }
826
827 if (!matched)
828 return false;
829
830 // 条件求值（锁外，避免持锁回调死锁）
831 if (snapshot.isConditional()) {
832 if (evaluator_ && evaluator_->hasCallback()) {
833 if (!evaluator_->evaluate(snapshot.condition, line)) {
834 return false; // 条件不满足
835 }
836 } else {
837 return false; // 无求值器，条件 watchpoint 视为不命中
838 }
839 }
840
841 // 命中：递增 hitCount 并暂停（重新查找避免锁外快照过期）
842 {
843 std::lock_guard<std::mutex> lock(pauseMutex_);
844 for (auto& wp : watchpoints_) {
845 if (wp.kind == snapshot.kind && wp.varName == snapshot.varName && wp.fieldName == snapshot.fieldName &&
846 wp.condition == snapshot.condition) {
847 wp.hitCount++;
848 break;
849 }
850 }
851 }
852 doPause(line, currentDepth_.load());
853 return true;
854}

856void DebugController::stepIn() {
857 LOG_DEBUG("Step In", "Debugger");
858 // P1-8 fix: 消除 TOCTOU。原实现先无锁检查 !running_ 决定走快速路径（无 CV notify），
859 // 再加锁走慢速路径。检查与加锁之间存在窗口，且快速路径不 notify 在 worker
860 // 极端时序下可能丢唤醒。改为统一加锁路径，notify 对无 waiter 是 no-op。
861 {
862 std::lock_guard<std::mutex> lock(pauseMutex_);
863 // AUDIT-P2 fix: 不清除 stopped_。若 stop() 已设置 stopped_=true，step 应无效
864 // （worker 将在 checkBreak 抛 DebugStopException 终止）。原实现清除 stopped_=false
865 // 可在 worker 响应 stop 的窗口内取消正在进行的 stop，导致状态机不一致。
866 // stopped_ 的清除应由 reset()（新调试会话）负责。
867 if (stopped_)
868 return;
869 mode_.store(static_cast<int>(StepMode::MODE_STEP_IN));
870 running_ = true;
871 paused_ = false;
872 }
873 pauseCV_.notify_one();
874}

1089 // paused_=true 后 stopped_ 才被设为 true，则短暂返回 true → 一帧视觉抖动）。
1090 // 改用 try_lock 获取一致快照：
1091 // - 锁获取成功：pauseExecution 未持有 mutex，可一致读取 paused_ 和 stopped_
1092 // - 锁获取失败：pauseExecution 正在 wait（worker 持有 mutex），说明调试器已暂停
1093 // （paused_ 在调用 pauseExecution 前已设为 true），仅检查 stopped_ 即可。
1094 // try_lock 不会阻塞 UI 线程——pauseExecution 持锁期间正是调试器暂停期间。

1099 // P2 #60 fix: try_lock 失败时检查 paused_ 原子变量，而非仅依赖 stopped_。
1100 // pauseExecution 持有锁时 paused_ 已为 true，但显式读取更精确：
1101 // 若 stop() 已设 stopped_=true 或 paused_ 被重置，返回 false。

1105void DebugController::reset() {
1106 // D-P2-10 fix: 所有状态重置统一在锁内进行，避免锁内外重置的一致性间隙
1107 //
1108 // terminate 防御：WorkerManager::forceStop 在 worker 死循环时调用 terminate()，
1109 // 若 worker 恰在 pauseExecution() 的 wait() 唤醒后重新获取 mutex 的极小窗口内
1110 // 被杀死，pauseMutex_ 会被死线程持有，lock() 会永久阻塞导致主线程死锁。
1111 // 改用 try_lock：成功则完整重置；失败则重置全部 atomic 字段。
1112 // AUDIT-R4 BUG-16 fix: stepOverDepth_/stepOutDepth_/tempBreakpointLine_ 已
1113 // atomic 化，无条件重置（原实现获锁失败时跳过，残留旧值使下次调试
1114 // 会话首次单步多走/少走一层）。仅 breakpointInfos_ 的 hitCount 重置
1115 // 仍需锁保护（QMap 非线程安全），获锁失败时跳过——hitCount 仅影响
1116 // 条件断点的 hitCondition 计数显示，非关键状态。
1117 bool locked = false;
1118 {
1119 std::unique_lock<std::mutex> lock(pauseMutex_, std::try_to_lock);
1120 locked = lock.owns_lock();
1121 mode_.store(static_cast<int>(StepMode::MODE_RUN));
1122 running_ = false;
1123 stopped_ = false;
1124 paused_ = false;
1125 currentDepth_.store(0);
1126 lastPausedLine_.store(-1);
1127 lastPausedDepth_.store(-1);
1128 lastSeenLine_.store(-1);
1129 crossedLine_.store(false);
1130 crossedDeeper_.store(false);
1131 // AUDIT-R4 BUG-16 fix: atomic 化后无条件重置（不再依赖 locked）
1132 stepOverDepth_.store(0);
1133 stepOutDepth_.store(0);
1134 // R98 runToCursor: reset() 清除临时断点。新调试会话不应继承上一次会话的

1148 "Debugger");
1149 }
