1/**
2 * @file common/MemoryInspectionAPI.h
3 * @brief 内存检视只读快照 API（ARCH-10）。
4 *
5 * 为 GUI 教学面板（MemoryModelPanel / GcVisualizerPanel / VariableInspectorPanel
6 * 等）提供对 NaN-boxing / RefCounted / GcManager 内部状态的只读快照访问，
7 * 消除面板对 interpreter/NaNBox.h / interpreter/RefCounted.h /
8 * interpreter/GcManager.h 内部类型的直接依赖。
9 *
10 * 设计原则：
11 * - 只暴露教学面板需要的只读快照（纯数据结构，无内部类型透出）
12 * - GcMode / GcPhase 枚举定义在此处（单一真相源），GcManager.h 通过
13 * #include 此头文件复用，避免 common/ → interpreter/ 反向依赖
14 * - 静态方法形式，无实例状态（GcManager 单例访问封装在 .cpp 内）
15 * - 返回值均为值拷贝或 std::string，调用方无需持有锁或担心生命周期
16 *
17 * 与 IBackend 的关系：
18 * - IBackend 抽象执行后端（Compiler/VM/RegisterVM）的回调与诊断接口
19 * - MemoryInspectionAPI 抽象内存子系统（NaNBox/RefCounted/GcManager）的只读检视
20 * - 两者互补，共同构成 GUI 层与引擎层之间的服务接口
21 *
22 * @since ARCH-10
23 */

47/// R113 C 项：GC 阶段状态枚举——用于 MemoryModelPanel 实时展示当前 mark-sweep 进度。
48/// collectCycle 是同步阻塞操作，currentPhase_ 在阶段切换时短暂变更，
49/// collectCycle 结束后回归 Idle。UI 在 animTimer_ 周期内能观察到上次 GC 经历的阶段。
50enum class GcPhase : uint8_t {
51 Idle, // 未在 GC 中（或上次 GC 已结束）
52 Marking, // Phase 1: 从 roots 出发 mark 所有可达容器节点
53 Sweeping, // Phase 2: 清扫 marked 未命中且槽位仍被占用的循环孤岛（slots_ 登记表）
54 Finalizing, // Phase 3: 原地压缩 slots_ 登记表并回收死亡槽位，重置 marked 标志
55};
56
57/// R135 GC 模式枚举——三种内存管理策略可切换，用于教学对比与性能基准。
58/// 模式语义：
59/// RefCountOnly - 纯引用计数，不注册 GcManager，循环引用会泄漏（基线对比用）
60/// RefCountWithCycleGc - 引用计数主导 + GC 仅回收循环孤岛（默认，项目历史模式）
61/// GcOnly - GC 主导：sweep 阶段直接 delete 不可达对象（实验性）
62/// 切换时机：仅在 Interpreter/VM 完全重置后切换（避免运行中切换导致状态不一致）
63enum class GcMode : uint8_t {
64 RefCountOnly = 0,
65 RefCountWithCycleGc = 1,
66 GcOnly = 2,
67};

73/// NaN-box 位模式快照（替代面板直接调用 NaNBox::fromInt(...).rawBits()）
74/// 所有字段为值类型，调用方无需依赖 NaNBox 类。
75struct NaNBoxSnapshot {
76 uint64_t bits = 0; // 原始 64 位位模式
77 std::string bitsHex; // "0x" + 16 位十六进制（带前导零）
78 std::string bitsBinary; // 64 字符 "0/1" 字符串（最高位在索引 0）
79 std::string tagName; // "FLOAT" / "INT" / "BOOL" / "NUL" / "POINTER"
80 int tagValue = -1; // NaNBox::Tag 枚举值（0-4）；-1 表示未知
81};

83/// 堆对象元信息快照（替代面板直接访问 RefCounted* p = v.asPointer(); p->type）
84struct HeapObjectSnapshot {
85 bool isPointer = false; // 是否为堆指针类型
86 std::string structName; // C++ 结构名（"ArrayData" / "DictData" / "InstanceData" 等）
87 std::string addressHex; // "0x" + 16 位十六进制指针
88 size_t fieldCount = 0; // 元素数/字段数（数组=元素数，字典/实例=字段数，闭包=capturedVars）
89 std::string valueTypeName; // "VAL_INT" / "VAL_STRING" / "VAL_ARRAY" / ...
90 int valueType = -1; // ValueType 枚举值（与 Value::getType() 一致）
91 size_t refCount = 0; // 引用计数（RefCounted::useCount()），仅 isPointer 时有效
92};

94/// GC 统计快照（替代面板直接调用 GcManager::instance().trackedCount() 等）
95struct GcStatsSnapshot {
96 size_t trackedCount = 0; // 当前 tracked 节点数
97 size_t totalGcCount = 0; // 累计 GC 次数
98 size_t lastMarkedCount = 0; // 上次 GC 标记的可达节点数
99 size_t lastCollectedCount = 0; // 上次 GC 回收的孤岛数
100 GcMode mode = GcMode::RefCountWithCycleGc; // 当前 GcMode
101 GcPhase phase = GcPhase::Idle; // 当前 GcPhase
102 size_t allocationsSinceLastGc = 0; // 自上次 GC 后累计分配数
103 size_t allocationThreshold = 0; // 增量触发阈值
104};

106// ============================================================
107// MemoryInspectionAPI — 内存检视只读 API
108// ============================================================
109// 所有方法均为静态，无实例状态。GcManager 单例访问、NaNBox 构造、RefCounted
110// 指针解引用等内部实现细节封装在 .cpp 中，调用方仅依赖此头文件的纯数据结构。
111//
112// 线程安全：
113// - 读接口（inspectValue / inspectHeap / getGcStats）通过 GcManager 内部锁
114// 保护，与 worker 线程的 collectCycle 互斥。
115// - 写接口（resetGc / setGcMode / collectCycle）同上，调用方需确保 Interpreter/VM
116// 已完全重置（不在执行中），避免运行中切换导致状态不一致。
