45// ============================================================
46// matchTrigger — 检测光标前的触发词
47// ------------------------------------------------------------
48// 算法：
49// 1) 从 textBeforeCursor 末尾向前扫描，跳过尾随空白（如有）
50// 2) 继续向前扫描连续的非空白字符，得到候选触发词 [start, end)
51// 3) 检查触发词前方是否为空白或位于文档开头（start == 0 或
52// textBeforeCursor[start-1] 是空白）
53// —— 由于扫描在空白处停止，若 start > 0 则
54// textBeforeCursor[start-1] 必为空白
55// 4) 将候选触发词与 snippet.trigger 精确匹配，返回首个匹配项
56//
57// 不命中场景：
58// - textBeforeCursor 为空 → 返回 nullptr
59// - 光标前全是空白 → 候选触发词为空，返回 nullptr
60// - 触发词未在 snippet 库中 → 返回 nullptr
61// ============================================================
62const CodeSnippet* CodeSnippetEngine::matchTrigger(const QString& textBeforeCursor) {
63 if (textBeforeCursor.isEmpty())
64 return nullptr;
65
66 const int total = textBeforeCursor.size();
67 int end = total;
68
69 // 跳过尾随空白（光标可能位于空白之后）
70 while (end > 0 && textBeforeCursor[end - 1].isSpace()) {
71 --end;
72 }
73 if (end == 0)
74 return nullptr; // 全是空白
75
76 // 向前扫描连续的非空白字符
77 int start = end;
78 while (start > 0 && !textBeforeCursor[start - 1].isSpace()) {
79 --start;
80 }
81 // 此时若 start > 0，则 textBeforeCursor[start-1] 必为空白（循环退出条件）
82 // 若 start == 0，则触发词位于文档开头，同样满足"前方是空白或行首"
83
84 const QString candidate = textBeforeCursor.mid(start, end - start);
85 if (candidate.isEmpty())
86 return nullptr;
87
88 const auto& all = snippets();
89 for (const auto& snip : all) {
90 if (candidate.toStdString() == snip.trigger) {
91 return &snip;
92 }
93 }
94 return nullptr;
95}

97// ============================================================
98// expand — 展开模板
99// ------------------------------------------------------------
100// 算法：
101// 1) 逐字符扫描 templateText，识别 ${N:default} 模式
102// 2) 模式识别：'$' '{' [digits] ':' [default] '}'
103// 3) 将 default 文本写入输出，记录 [start, end) 范围
104// 4) 相同 N 的多次出现生成多个独立 range
105// 5) primaryCursorPos = 第一个占位符的 start（用于编辑器初始定位）
106// 若无占位符则为 -1（编辑器将光标置于展开文本末尾）
107//
108// 边界处理：
109// - "${" 后无 '}'：按字面量输出 "$" 和后续字符
110// - "${N}" 无冒号：default 为空，range 长度为 0
111// - "${:default}" 无 N：仍然识别为占位符（N 默认为 0）
112// - 嵌套 "${...${...}...}"：当前实现不支持嵌套，第一个 '}' 即结束
113// ============================================================
114CodeSnippetEngine::Expansion CodeSnippetEngine::expand(const CodeSnippet& snip) {
115 Expansion result;
116 const QString tpl = QString::fromStdString(snip.templateText);
117 QString& output = result.text;
118
119 int i = 0;
120 const int n = tpl.size();
121 while (i < n) {
122 // 检测 "${" 起始
123 if (i + 1 < n && tpl[i] == QChar('$') && tpl[i + 1] == QChar('{')) {
124 // 查找匹配的 '}'
125 int j = i + 2;
126 while (j < n && tpl[j] != QChar('}')) {
127 ++j;
128 }
129 if (j < n) {
130 // 找到 "${...}"，解析内容
131 const QString inner = tpl.mid(i + 2, j - i - 2);
132 const int colonPos = inner.indexOf(QChar(':'));
133 QString defaultText;
134 if (colonPos >= 0) {
135 defaultText = inner.mid(colonPos + 1);
136 } else {
137 // "${N}" 形式，无默认值
138 defaultText.clear();
139 }
140 const int rangeStart = output.size();
141 output += defaultText;
142 const int rangeEnd = output.size();
143 result.placeholderRanges.push_back({rangeStart, rangeEnd});
144 i = j + 1;
145 continue;
146 }
147 // 未找到 '}'，按字面量输出 '$' 并继续
148 }
149 output += tpl[i];
150 ++i;
151 }
152
153 // 设置 primaryCursorPos：第一个占位符的起点，或 -1 表示末尾
154 if (!result.placeholderRanges.empty()) {
155 result.primaryCursorPos = result.placeholderRanges.front().first;
156 } else {
157 result.primaryCursorPos = -1;
158 }
159
160 return result;
161}
