47/// Convert IR text (lines ending with "; line N") to clickable HTML
48static QString irTextToClickableHtml(const QString& plainText) {
49 QStringList lines = plainText.split("\n");
50 QString html;
51 html += "<pre style=\"font-family:'Cascadia Code','Cascadia Mono','Consolas','JetBrains Mono','Source Code "
52 "Pro','Menlo','DejaVu Sans Mono','Courier New',monospace;\">";
53 static const QRegularExpression lineCommentRe(";\\s*line\\s+(\\d+)\\s*$");
54 for (int i = 0; i < lines.size(); ++i) {
55 QString line = lines[i];
56 auto m = lineCommentRe.match(line);
57 if (m.hasMatch()) {
58 QString lineNum = m.captured(1);
59 QString escaped = line.toHtmlEscaped();
60 html +=
61 "<a href=\"#LINE_" + lineNum + "\" style=\"color:inherit;text-decoration:none;\">" + escaped + "</a>";
62 } else {
63 html += line.toHtmlEscaped();
64 }
65 if (i < lines.size() - 1)
66 html += "\n";
67 }
68 html += "</pre>";
69 return html;
70}

72/// Convert bytecode disassembly (format: "<offset> L<line> <OP> ...") to clickable HTML
73static QString bytecodeTextToClickableHtml(const QString& plainText) {
74 QStringList lines = plainText.split("\n");
75 QString html;
76 html += "<pre style=\"font-family:'Cascadia Code','Cascadia Mono','Consolas','JetBrains Mono','Source Code "
77 "Pro','Menlo','DejaVu Sans Mono','Courier New',monospace;\">";
78 // Match lines starting with digits, space, L followed by digits
79 static const QRegularExpression bcLineRe("^(\\d+)\\s+L(\\d+)\\s");
80 for (int i = 0; i < lines.size(); ++i) {
81 QString line = lines[i];
82 auto m = bcLineRe.match(line);
83 if (m.hasMatch()) {
84 QString lineNum = m.captured(2);
85 QString escaped = line.toHtmlEscaped();
86 html +=
87 "<a href=\"#LINE_" + lineNum + "\" style=\"color:inherit;text-decoration:none;\">" + escaped + "</a>";
88 } else {
89 html += line.toHtmlEscaped();
90 }
91 if (i < lines.size() - 1)
92 html += "\n";
93 }
94 html += "</pre>";
95 return html;
96}

472/// 切换到指定阶段索引并刷新对应的分步页面内容。
473void PipelineViewer::switchToStep(int step) {
474 if (step < 0 || step >= 5)
475 return;
476 // AUDIT-P2 fix: 同一步骤重复点击不重新触发动画（原实现 page->pos() 依赖
477 // 前一个动画的中间值，导致 finalPos 错误，页面卡在偏移位置）。
478 if (step == currentStep_) {
479 reloadCurrentStep(true);
480 return;
481 }
482 const int oldStep = currentStep_;
483 currentStep_ = step;
484 stack_->setCurrentIndex(step);
485
486 stepSourceBtn_->setChecked(step == 0);
487 stepTokenBtn_->setChecked(step == 1);
488 stepAstBtn_->setChecked(step == 2);
489 stepIrBtn_->setChecked(step == 3);
490 stepBytecodeBtn_->setChecked(step == 4);
491
492 reloadCurrentStep(true);
493
494 // === 切换动画：仅水平滑动 ===
495 // 注：移除 fadeInWidget —— QGraphicsOpacityEffect 对含子 widget 的页面
496 // 会卡 opacity=0 导致切换后空白，仅保留水平滑动动画即可。
497 // 水平滑动：前进方向（step > oldStep）从右侧滑入，后退方向从左侧滑入
498 // QStackedLayout 在 setCurrentIndex 后已固定子控件 geometry，
499 // 之后调用 move() 不会被布局覆盖直到下一次几何变化
500 QWidget* page = stack_->currentWidget();
501 // AUDIT-P2 fix: 停止 page 上正在运行的 pos 动画，避免多个 QPropertyAnimation
502 // 叠加修改 pos 属性导致页面最终位置错误。
503 const auto oldAnims = page->findChildren<QPropertyAnimation*>();
504 for (auto* a : oldAnims) {
505 if (a->targetObject() == page && a->propertyName() == "pos") {
506 a->stop();
507 a->deleteLater();
508 }
509 }
510 const int offset = 24;
511 const int dx = (step >= oldStep) ? offset : -offset;
512 const QPoint finalPos = page->pos();
513 page->move(finalPos.x() + dx, finalPos.y());
514 auto* slideAnim = new QPropertyAnimation(page, "pos", page);
515 slideAnim->setDuration(220);
516 slideAnim->setStartValue(page->pos());
517 slideAnim->setEndValue(finalPos);
518 slideAnim->setEasingCurve(QEasingCurve::OutCubic);
519 slideAnim->start(QAbstractAnimation::DeleteWhenStopped);
520}

522/// 在不切换阶段的前提下，重新加载当前阶段页面（数据变更后调用）。
523/// force=true 时绕过指纹缓存，强制重新填充（switchToStep 等用户显式操作应传 true）。
524void PipelineViewer::reloadCurrentStep(bool force) {
525 forceRefresh_ = force;
526 switch (currentStep_) {
527 case 0:
528 populateSource();

581/// 填充「源码」页：原样展示当前编辑区源代码。
582void PipelineViewer::populateSource() {
583 if (!controller_) {
584 sourceBrowser_->setPlainText(QString::fromUtf8("（未绑定控制器）"));
585 return;
586 }
587 // 显示当前 token 流推导出的源码（从 controller->lastTokens 反推）
588 // 简化方案：直接显示 token 字面拼接
589 const auto& tokens = controller_->lastTokens();
590 if (tokens.empty()) {
591 sourceBrowser_->setPlainText(QString::fromUtf8("（尚未编译，请在主编辑器输入代码并触发编译分析）"));
592 return;
593 }
594 std::ostringstream os;
595 int lastLine = 1;
596 // AUDIT-P2 fix: 原实现每次循环调用 os.str().back() 获取最后字符，os.str()
597 // 返回 std::string 值拷贝（O(n)），n 个 Token 总复杂度 O(n²)。改用 lastChar
598 // 变量跟踪最后写入字符，降为 O(n)。
599 char lastChar = '\0';
600 for (const auto& tk : tokens) {
601 while (lastLine < tk.line) {
602 os << "\n";
603 lastChar = '\n';
604 lastLine++;
605 }
606 if (tk.column > 1 && lastChar != '\0' && lastChar != '\n') {
607 os << " ";
608 lastChar = ' ';
609 }
610 os << tk.lexeme;
611 if (!tk.lexeme.empty())
612 lastChar = tk.lexeme.back();
613 }
614 sourceBrowser_->setPlainText(QString::fromUtf8(os.str().c_str()));
615}

617/// 填充「词法」页：展示词法分析得到的 token 序列。
618void PipelineViewer::populateTokens() {
619 if (!controller_) {
620 tokenTable_->setRowCount(0);
621 return;
622 }
623 const auto& tokens = controller_->lastTokens();
624 // AUDIT-P1 fix: 指纹缓存——vmStateChanged 高频触发时若 tokens 指针未变则跳过重建。
625 // forceRefresh_=true（用户显式切换页面）时绕过缓存。
626 if (!forceRefresh_ && &tokens == lastTokensPtr_)
627 return;
628 lastTokensPtr_ = &tokens;
629 tokenTable_->setRowCount((int)tokens.size());
630 for (int i = 0; i < (int)tokens.size(); ++i) {
631 const auto& tk = tokens[i];
632 tokenTable_->setItem(i, 0, new QTableWidgetItem(QString::number(i)));
633 tokenTable_->setItem(i, 1, new QTableWidgetItem(QString::fromStdString(Token::typeToString(tk.type))));
634 tokenTable_->setItem(i, 2, new QTableWidgetItem(QString::fromUtf8(tk.lexeme.c_str())));
635 tokenTable_->setItem(i, 3, new QTableWidgetItem(QString::number(tk.line)));
636 tokenTable_->setItem(i, 4, new QTableWidgetItem(QString::number(tk.column)));
637 }

641void PipelineViewer::dumpAst(std::ostringstream& os, ASTNode* node, int depth, int maxDepth) {
