108IRCFG::IRCFG(const IRFunction& ir) {
109 // 阶段 1：遍历所有 IRBasicBlock，按 LABEL/终结指令切分为严格基本块
110 // 每个 IRBasicBlock 通常以 LABEL 起始（AstIRBuilder::newBlock 保证）
111 for (size_t bi = 0; bi < ir.blocks.size(); ++bi) {
112 const auto& blk = ir.blocks[bi];
113 if (blk.instructions.empty()) {
114 CFGNode node;
115 node.id = static_cast<uint32_t>(nodes_.size());
116 node.sourceBlockIdx = bi;
117 node.startInstr = 0;
118 node.endInstr = 0;
119 node.labelIndex = blk.labelIndex;
120 node.hasLabel = false;
121 labelToNode_[blk.labelIndex] = node.id;
122 nodes_.push_back(node);
123 continue;
124 }
125
126 size_t i = 0;
127 bool firstChunk = true;
128 while (i < blk.instructions.size()) {
129 CFGNode node;
130 node.id = static_cast<uint32_t>(nodes_.size());
131 node.sourceBlockIdx = bi;
132 node.startInstr = i;
133
134 // 若以 LABEL 起始，记录标签
135 if (blk.instructions[i].op == IROp::LABEL) {
136 node.hasLabel = true;
137 node.labelIndex = blk.instructions[i].operands[0].index;
138 labelToNode_[node.labelIndex] = node.id;
139 ++i;
140 } else {
141 node.hasLabel = false;
142 // 继承所在 IRBasicBlock 的 labelIndex 作为 fallback
143 if (firstChunk) {
144 node.labelIndex = blk.labelIndex;
145 labelToNode_[blk.labelIndex] = node.id;
146 } else {
147 // 无 LABEL 的非首块：用前一个块的 label 作为 fallback
148 node.labelIndex = blk.labelIndex;
149 }
150 }
151 firstChunk = false;
152
153 // 扫描到下一条 LABEL 或终结指令之后
154 while (i < blk.instructions.size()) {
155 IROp op = blk.instructions[i].op;
156 if (op == IROp::LABEL) {
157 break; // 新严格块起始
158 }
159 ++i;
160 if (isTerminatorOp(op)) {
161 break; // 终结指令结束本块（含）
162 }
163 }
164 node.endInstr = i; // [startInstr, endInstr)
165 nodes_.push_back(node);
166 }
167 }
168
169 // 若函数为空（无 blocks），创建单入口空块
170 if (nodes_.empty()) {
171 CFGNode entry;
172 entry.id = 0;
173 entry.sourceBlockIdx = 0;
174 entry.startInstr = 0;
175 entry.endInstr = 0;
176 entry.labelIndex = CFGNode::kEntryLabel;
177 entry.hasLabel = false;
178 nodes_.push_back(entry);
179 return;
180 }
181
182 // 阶段 2：构建后继边
183 for (size_t ni = 0; ni < nodes_.size(); ++ni) {
184 auto& node = nodes_[ni];
185 const auto& blk = ir.blocks[node.sourceBlockIdx];
186
187 if (node.startInstr >= node.endInstr) {
188 // 空块：fall-through 到下一个节点
189 if (ni + 1 < nodes_.size()) {
190 node.succs.push_back(static_cast<uint32_t>(ni + 1));
191 }
192 continue;
193 }
194
195 size_t lastIdx = node.endInstr - 1;
196 const auto& lastInstr = blk.instructions[lastIdx];
197
198 switch (lastInstr.op) {
199 case IROp::JUMP: {
200 uint32_t targetLabel = lastInstr.operands[0].index;
201 auto it = labelToNode_.find(targetLabel);
202 if (it != labelToNode_.end()) {
203 node.succs.push_back(it->second);
204 }
205 break;
206 }
207 case IROp::JUMP_IF_FALSE: {
208 // fall-through + 目标
209 if (ni + 1 < nodes_.size()) {
210 node.succs.push_back(static_cast<uint32_t>(ni + 1));
211 }
212 if (lastInstr.operands.size() >= 2) {
213 uint32_t targetLabel = lastInstr.operands[1].index;
214 auto it = labelToNode_.find(targetLabel);
215 if (it != labelToNode_.end()) {
216 node.succs.push_back(it->second);
217 }
218 }
219 break;
220 }
221 case IROp::RETURN:
222 case IROp::RETURN_NULL:
223 case IROp::THROW:
224 // 无后继
225 break;
226 default:
227 // 隐式 fall-through
228 if (ni + 1 < nodes_.size()) {
229 node.succs.push_back(static_cast<uint32_t>(ni + 1));
230 }
231 break;
232 }
233
234 // BUGFIX-P2 #55: TRY_BEGIN 隐式异常边——TRY_BEGIN 不是终结指令（不改变块分割），
235 // 但它声明了 try 块的 catch handler。若块内任一指令抛异常，控制流跳转到 catch label。
236 // 扫描本节点所有指令，为每个 TRY_BEGIN 添加到 catch handler 块的隐式边。
237 for (size_t ii = node.startInstr; ii < node.endInstr; ++ii) {
238 const auto& scanInstr = blk.instructions[ii];
239 if (scanInstr.op == IROp::TRY_BEGIN && !scanInstr.operands.empty()) {
240 uint32_t catchLabel = scanInstr.operands[0].index;
241 auto cit = labelToNode_.find(catchLabel);
242 if (cit != labelToNode_.end()) {
243 // 避免重复边
244 bool alreadyExists = false;
245 for (uint32_t s : node.succs) {
246 if (s == cit->second) {
247 alreadyExists = true;
248 break;
249 }
250 }
251 if (!alreadyExists) {
252 node.succs.push_back(cit->second);
253 }
254 }
255 }
256 }
257 }
258
259 // 阶段 3：从后继边构建前驱边
260 for (auto& node : nodes_) {
261 for (uint32_t s : node.succs) {
262 if (s < nodes_.size()) {
263 nodes_[s].preds.push_back(node.id);
264 }
265 }
266 }
267}

284DominatorTree::DominatorTree(const IRCFG& cfg) : cfg_(cfg) {
285 size_t n = cfg.size();
286 idom_.assign(n, kInvalid); // kInvalid 表示 undefined
287 domChildren_.assign(n, {});
288 postOrderNum_.assign(n, kInvalid);
289 postorder_.clear();
290 rpo_.clear();
291
292 computeRPO();
293 computeIDom();
294
295 // 构建 domChildren_
296 for (size_t i = 0; i < n; ++i) {
297 uint32_t d = idom_[i];
298 if (d != kInvalid && d != i) {
299 domChildren_[d].push_back(static_cast<uint32_t>(i));
300 }
301 }
302}

304void DominatorTree::computeRPO() {
305 // DFS 后序遍历（仅访问从 entry 可达的节点）
306 std::vector<bool> visited(cfg_.size(), false);
307 std::vector<std::pair<uint32_t, size_t>> stack; // (node, next succ idx)
308
309 uint32_t entry = cfg_.entryId();
310 visited[entry] = true;
311 stack.push_back({entry, 0});
312
313 while (!stack.empty()) {
314 auto& [node, idx] = stack.back();
315 const auto& succs = cfg_.node(node).succs;
316 if (idx < succs.size()) {
317 uint32_t s = succs[idx++];
318 if (!visited[s]) {
319 visited[s] = true;
320 stack.push_back({s, 0});
321 }
322 } else {
323 postorder_.push_back(node);
324 stack.pop_back();
325 }
326 }
327
328 // RPO = reverse postorder
329 rpo_.assign(postorder_.rbegin(), postorder_.rend());
330}

332void DominatorTree::computeIDom() {
333 if (postorder_.empty())
334 return;
335
336 // postorder 编号：用于 intersect 比较（已由 computeRPO 填充到 postOrderNum_）
337 for (size_t i = 0; i < postorder_.size(); ++i) {
338 postOrderNum_[postorder_[i]] = static_cast<uint32_t>(i);
339 }
340
341 uint32_t entry = cfg_.entryId();
342 idom_[entry] = entry;
343
344 bool changed = true;
345 while (changed) {
346 changed = false;
347 // 按 RPO 遍历（跳过 entry）
348 for (uint32_t b : rpo_) {
349 if (b == entry)
350 continue;
351
352 uint32_t newIdom = kInvalid;
353 for (uint32_t p : cfg_.node(b).preds) {
354 if (idom_[p] == kInvalid)
355 continue;
356 if (newIdom == kInvalid) {
357 newIdom = p;
358 } else {
359 newIdom = intersect(p, newIdom);
360 }
361 }
362
363 if (newIdom != kInvalid && idom_[b] != newIdom) {
364 idom_[b] = newIdom;
365 changed = true;
366 }
367 }
368 }

371uint32_t DominatorTree::intersect(uint32_t b1, uint32_t b2) const {
372 // CHK 算法核心：沿 postorder 更深者上溯直到相遇
373 while (b1 != b2) {
374 while (postOrderNum_[b1] < postOrderNum_[b2]) {
375 b1 = idom_[b1];
376 }
377 while (postOrderNum_[b2] < postOrderNum_[b1]) {
378 b2 = idom_[b2];
379 }
380 }
381 return b1;
382}

384bool DominatorTree::dominates(uint32_t a, uint32_t b) const {
385 // a 支配 b 当且仅当从 b 沿 idom 链上溯能到达 a
386 if (a >= idom_.size() || b >= idom_.size())
387 return false;
388 if (idom_[b] == kInvalid)
389 return a == b;
390 uint32_t cur = b;
391 while (true) {
392 if (cur == a)
393 return true;
394 if (cur == idom_[cur])
395 return false; // 到达 entry 的自环
396 cur = idom_[cur];
397 if (cur == kInvalid)
398 return false;
399 }
400}

402uint32_t DominatorTree::nearestCommonDominator(uint32_t a, uint32_t b) const {
403 if (a >= idom_.size() || b >= idom_.size())
404 return cfg_.entryId();
405 if (idom_[a] == kInvalid || idom_[b] == kInvalid)
406 return cfg_.entryId();
407
408 // 简化实现：用路径集合法
409 std::unordered_set<uint32_t> pathA;
410 uint32_t cur = a;
411 while (true) {
412 pathA.insert(cur);
413 if (cur == idom_[cur])
414 break;
415 cur = idom_[cur];
416 if (cur == kInvalid)
417 break;
418 }
419 cur = b;
420 while (pathA.find(cur) == pathA.end()) {
421 if (cur == idom_[cur])
422 return cfg_.entryId();
423 cur = idom_[cur];
424 if (cur == kInvalid)
425 return cfg_.entryId();
426 }
427 return cur;
428}

434DominanceFrontier::DominanceFrontier(const IRCFG& cfg, const DominatorTree& domTree) {
435 df_.assign(cfg.size(), {});
436
437 for (size_t b = 0; b < cfg.size(); ++b) {
438 const auto& node = cfg.node(static_cast<uint32_t>(b));
439 if (node.preds.size() >= 2) {
440 uint32_t idomB = domTree.idom(static_cast<uint32_t>(b));
441 for (uint32_t p : node.preds) {
442 uint32_t runner = p;
443 // runner 上溯到 idom[b]，沿途加入 b 到 DF
444 while (runner != DominatorTree::kInvalid && runner != idomB) {
445 df_[runner].push_back(static_cast<uint32_t>(b));
446 uint32_t next = domTree.idom(runner);
447 if (next == runner || next == DominatorTree::kInvalid)
448 break; // 到达 entry 自环或不可达
449 runner = next;
450 }
451 }
452 }
453 }
454
455 // 去重
456 for (auto& list : df_) {
457 std::sort(list.begin(), list.end());
458 list.erase(std::unique(list.begin(), list.end()), list.end());
459 }
460}

466NaturalLoopInfo::NaturalLoopInfo(const IRCFG& cfg, const DominatorTree& domTree) : cfg_(cfg) {
467 detectBackEdges(domTree);
468
469 // 对每条回边收集自然循环体
470 for (const auto& edge : backEdges_) {
471 Loop loop;
472 loop.header = edge.second;
473 loop.backEdges.push_back(edge.first);
474 collectLoopBody(edge.second, edge.first);
475 // collectLoopBody 填充 loopBody_，转移到 loop
476 // P2-10 fix: 原代码误用 loop.body.end() 作为结束迭代器（loop.body 此时为空，
477 // 导致赋值空范围，loop.body 恒仅含 header）。修正为 loopBody_.end()。
478 loop.body.assign(loopBody_.begin(), loopBody_.end());
479 std::sort(loop.body.begin(), loop.body.end());
480
481 size_t loopIdx = loops_.size();
482 loops_.push_back(std::move(loop));
483
484 // 更新 loopsOf_ 和 inLoop_
485 for (uint32_t n : loops_.back().body) {
486 loopsOf_[n].push_back(loopIdx);
487 inLoop_.insert(n);
488 }
489 }

492void NaturalLoopInfo::detectBackEdges(const DominatorTree& domTree) {
493 backEdges_.clear();
494 for (size_t n = 0; n < cfg_.size(); ++n) {
495 const auto& node = cfg_.node(static_cast<uint32_t>(n));
496 for (uint32_t s : node.succs) {
497 // 边 n→s 是回边当且仅当 s 支配 n
498 if (domTree.dominates(s, static_cast<uint32_t>(n))) {
499 backEdges_.push_back({static_cast<uint32_t>(n), s});
500 }
501 }
502 }

505void NaturalLoopInfo::collectLoopBody(uint32_t header, uint32_t tail) {
506 loopBody_.clear();
507 loopBody_.insert(header);
508 if (tail == header)
509 return;
510
511 loopBody_.insert(tail);
512 std::stack<uint32_t> stack;
513 stack.push(tail);
514 while (!stack.empty()) {
515 uint32_t n = stack.top();
516 stack.pop();
517 for (uint32_t p : cfg_.node(n).preds) {
518 if (p != header && loopBody_.find(p) == loopBody_.end()) {
519 loopBody_.insert(p);
520 stack.push(p);
521 }
522 }
523 }

530bool ssaConstructPass(IRFunction& ir) {
531 // 仅在含控制流的函数上构造（≥2 个严格块）
532 bool hasControlFlow = false;
533 for (const auto& blk : ir.blocks) {
534 for (const auto& instr : blk.instructions) {
535 if (instr.op == IROp::JUMP_IF_FALSE || instr.op == IROp::JUMP) {
536 hasControlFlow = true;
537 break;
538 }
539 // L17 fix: 含 try/catch 的函数跳过 SSA 构造——TRY_BEGIN 的隐式异常边
540 // 可从块内任意指令跳入 catch handler（中断点在块中间），而
541 // ssaDestruct 只能在前驱块尾部插入 PHI 写回 STORE_LOCAL，异常路径
542 // 上写回永远不执行，catch 内读到陈旧/未定义值（复现：类继承
543 // super.init 链 + try/catch 下 this 读到 null）。异常控制流下的
544 // SSA 需要 landing-pad 级建模，当前保守跳过（该函数仍享受
545 // 折叠+DCE 基础优化，仅跳过 GVN/LICM）。
546 if (instr.op == IROp::TRY_BEGIN)
547 return false;
548 }
549 }
550 if (!hasControlFlow)
551 return false;
552
553 IRCFG cfg(ir);
554 if (cfg.size() < 2)
555 return false;
556
557 DominatorTree domTree(cfg);
558 DominanceFrontier df(cfg, domTree);
559
560 // 收集每个 local slot 的定义块（含 STORE_LOCAL 的 CFG 节点）
561 std::map<uint32_t, std::set<uint32_t>> slotDefBlocks; // slot → block ids
562 for (size_t ni = 0; ni < cfg.size(); ++ni) {
563 const auto& node = cfg.node(static_cast<uint32_t>(ni));
564 const auto& blk = ir.blocks[node.sourceBlockIdx];
565 for (size_t i = node.startInstr; i < node.endInstr; ++i) {
566 if (blk.instructions[i].op == IROp::STORE_LOCAL) {
567 if (!blk.instructions[i].operands.empty()) {
568 uint32_t slot = blk.instructions[i].operands[0].index;
569 slotDefBlocks[slot].insert(static_cast<uint32_t>(ni));
570 }
571 }
572 }
573 }
574
575 if (slotDefBlocks.empty())
576 return false;
577
578 // 阶段 1：迭代支配边界 PHI 放置（Cytron 算法）
579 // phiPlacement[blockId] = {slots needing PHI in this block}
580 std::map<uint32_t, std::set<uint32_t>> phiPlacement;
581 bool placedAny = false;
582
583 for (auto& [slot, defBlocks] : slotDefBlocks) {
584 std::set<uint32_t> worklist = defBlocks;
585 std::set<uint32_t> hasPhi;
586 while (!worklist.empty()) {
587 uint32_t b = *worklist.begin();
588 worklist.erase(worklist.begin());
589 for (uint32_t d : df.df(b)) {
590 if (hasPhi.find(d) == hasPhi.end()) {
591 hasPhi.insert(d);
592 phiPlacement[d].insert(slot);
593 placedAny = true;
594 // 若 d 不在 defBlocks，加入 worklist
595 if (defBlocks.find(d) == defBlocks.end()) {
596 worklist.insert(d);
597 }
598 }
599 }
600 }
601 }
602
603 if (!placedAny)
604 return false;
605
606 // 阶段 2：插入 PHI 指令到 IRFunction
607 // PHI 格式: [dest_vreg, slot(LOCAL_SLOT), pred1_nodeid(IMM_UINT), vreg1(VIRTUAL), ...]
608 // 初始 pred/vreg 对为空，重命名阶段填充
609 // P2-10 fix3: pred 标识符从 labelIndex（LABEL）改为 CFG node ID（IMM_UINT），
610 // 避免 fall-through 块 labelIndex 不稳定导致的 PHI 前驱歧义
611 // phiInstrMap[(blockId, slot)] = (blockIdx, localInstrIdx, destVReg)
612 struct PhiEntry {
613 uint32_t blockId;
614 uint32_t slot;
615 size_t blockIdx;
616 size_t localIdx; // 插入位置
617 uint32_t destVReg;
618 };
619 std::vector<PhiEntry> phiEntries;
620
621 // P2-10 fix4: PHI 插入顺序——按 (sourceBlockIdx, startInstr) 降序插入。
622 // 原实现按 blockId 升序（= startInstr 升序）遍历 phiPlacement，先在低位置插入
623 // PHI 后在高位置插入，但低位置 insert 会使高位置的 startInstr 偏移，导致后续
624 // PHI 插入到错误位置（如 L5 的 PHI 被插入到前驱块 node 2 中部而非 L5 块首）。
625 // 降序插入：高位置先插入不影响低位置的 startInstr，确保每个 PHI 都插入到
626 // 正确的块首（LABEL 之后）。同块内多个 PHI 仍按 slot 升序 + offset 递增插入。
627 std::vector<std::pair<uint32_t, std::set<uint32_t>>> sortedPhiPlacement(phiPlacement.begin(), phiPlacement.end());
628 std::sort(sortedPhiPlacement.begin(), sortedPhiPlacement.end(), [&](const auto& a, const auto& b) {
629 const auto& nodeA = cfg.node(a.first);
630 const auto& nodeB = cfg.node(b.first);
631 if (nodeA.sourceBlockIdx != nodeB.sourceBlockIdx)
632 return nodeA.sourceBlockIdx > nodeB.sourceBlockIdx; // 降序
633 return nodeA.startInstr > nodeB.startInstr; // 降序
634 });
635
636 for (auto& [blockId, slots] : sortedPhiPlacement) {
637 const auto& node = cfg.node(blockId);
638 size_t bi = node.sourceBlockIdx;
639 auto& blk = ir.blocks[bi];
640 // 插入位置：LABEL 之后（若有），否则块起始
641 size_t insertPos = node.startInstr;
642 if (node.hasLabel) {
643 insertPos = node.startInstr + 1; // 跳过 LABEL
644 }
645
646 size_t offset = 0;
647 for (uint32_t slot : slots) {
648 uint32_t destVReg = ir.nextVReg++;
649 IROperandList ops;
650 ops.push_back(IROperand::vreg(destVReg));
651 ops.push_back(IROperand::local(slot));
652 // pred/vreg 对在重命名阶段填充
653
654 IRInstruction phiInstr(IROp::PHI, std::move(ops), 0);
655 blk.instructions.insert(blk.instructions.begin() + insertPos + offset, std::move(phiInstr));
656
657 phiEntries.push_back({blockId, slot, bi, insertPos + offset, destVReg});
658 ++offset;
659 }
660 }
661
662 // P2-10 fix: PHI 插入改变了 IR 指令序列的索引，原 CFG 的 startInstr/endInstr
663 // 已失效。重命名阶段遍历指令依赖准确的索引，必须重建 CFG。
664 // 不变量：PHI 插入不改变控制流（LABEL/JUMP/RETURN 位置关系不变），重建后的
665 // CFG 拓扑与原 CFG 一致，仅指令索引偏移。phiPlacement/phiDestMap 基于原
666 // blockId，重建后 blockId 不变（节点数和顺序不变）。
667 cfg = IRCFG(ir);
668 // 重建支配树（DominatorTree 持 cfg_ 引用，cfg 被重新赋值后需重建以更新引用）
669 // P2-10 fix: 原 std::swap(domTree, newDomTree) 因 DominatorTree 含引用成员
670 // 不可交换而编译失败；改为直接用 newDomTree 替代后续 domTree 引用。
671 DominatorTree newDomTree(cfg);
672
673 // 阶段 3：重命名（DFS 支配树）
674 // 构建 (blockId, slot) → destVReg 映射
675 std::map<std::pair<uint32_t, uint32_t>, uint32_t> phiDestMap;
676 for (const auto& pe : phiEntries) {
677 phiDestMap[{pe.blockId, pe.slot}] = pe.destVReg;
678 }
679
680 // slotStacks[slot] = 当前 SSA 值栈
681 std::unordered_map<uint32_t, std::vector<uint32_t>> slotStacks;
682 // vregRepl[vreg] = 替代 vreg（LOAD_LOCAL dest → 当前 slot 值）
683 std::unordered_map<uint32_t, uint32_t> vregRepl;
684
685 // L17 fix: 删除原"函数参数假 vreg 初始化"——原实现为 slot 0..arity-1
686 // 凭空分配初始 vreg 压栈，但该 vreg 无任何定义指令，重命名后参数的
687 // LOAD_LOCAL 被替换为未定义 vreg（如 LT v3 v8 v9 中 v9 无定义），
688 // lowering 后读未初始化寄存器 → 脏值/类型错误（复现：irOptimize 默认
689 // 启用后任意带 while(i<n) 的函数失败）。现在参数 slot 栈初始为空，
690 // LOAD_LOCAL 在栈空时不替换（保留真实帧槽读取），语义正确。
691
692 // 递归重命名
693 std::function<void(uint32_t)> rename = [&](uint32_t blockId) {
694 const auto& node = cfg.node(blockId);
695 auto& blk = ir.blocks[node.sourceBlockIdx];
696
697 // 记录本块压栈的 (slot) 列表，退出时弹栈
698 std::vector<uint32_t> pushedSlots;
699
700 // 处理本块的 PHI（在指令范围起始）
701 auto it = phiPlacement.find(blockId);
702 if (it != phiPlacement.end()) {
703 for (uint32_t slot : it->second) {
704 uint32_t destVReg = phiDestMap[{blockId, slot}];
705 slotStacks[slot].push_back(destVReg);
706 pushedSlots.push_back(slot);
707 }
708 }
709
710 // 处理常规指令
711 for (size_t i = node.startInstr; i < node.endInstr; ++i) {
712 auto& instr = blk.instructions[i];
713
714 // 跳过 PHI 指令
715 if (instr.op == IROp::PHI)
716 continue;
717
718 // 先应用 vregRepl 到 VIRTUAL 操作数
719 for (auto& op : instr.operands) {
720 if (op.kind == IROperandKind::VIRTUAL) {
721 auto rit = vregRepl.find(op.index);
722 if (rit != vregRepl.end()) {
723 op.index = rit->second;
724 }
725 }
726 }
727
728 if (instr.op == IROp::STORE_LOCAL && instr.operands.size() >= 2) {
729 uint32_t slot = instr.operands[0].index;
730 // Bug #86 fix: 上方循环已对所有 VIRTUAL 操作数（含 operands[1]）
731 // 统一应用 vregRepl，此处不再重复查找，避免双重替换语义错误。
732 uint32_t srcVReg = instr.operands[1].index;
733 slotStacks[slot].push_back(srcVReg);
734 pushedSlots.push_back(slot);
735 } else if (instr.op == IROp::LOAD_LOCAL && instr.operands.size() >= 2) {
736 uint32_t destVReg = instr.operands[0].index;
737 uint32_t slot = instr.operands[1].index;
738 auto& stack = slotStacks[slot];
739 if (!stack.empty()) {
740 vregRepl[destVReg] = stack.back();
741 }
742 }
743 }
744
745 // 填充后继块的 PHI 操作数
746 // P2-10 fix3: 使用 CFG node ID（IMM_UINT）作为前驱标识符，而非 labelIndex。
747 // 原方案用 labelIndex（LABEL）标识前驱，但 fall-through 块无 LABEL 时继承
748 // 所在 IRBasicBlock 的 labelIndex，导致同一 IRBasicBlock 内多个 CFG 节点共享
749 // labelIndex，PHI 前驱标识歧义。ssaDestructPass 据此 node ID 通过 CFG 精确
750 // 定位前驱块的终结指令位置（仅在 [startInstr, endInstr) 范围搜索，不跨越块边界）。
751 for (uint32_t succId : node.succs) {
752 auto succIt = phiPlacement.find(succId);
753 if (succIt == phiPlacement.end())
754 continue;
755 uint32_t predNodeId = node.id;
756
757 const auto& succNode = cfg.node(succId);
758 auto& succBlk = ir.blocks[succNode.sourceBlockIdx];
759 for (uint32_t slot : succIt->second) {
760 // 找到 succId 块中此 slot 的 PHI 指令
761 uint32_t phiDest = phiDestMap[{succId, slot}];
762 // L17 fix: 栈空时跳过——该前驱路径上 slot 的当前值仍在帧槽中
763 // （参数初值由调用协议写入，或更早的 STORE_LOCAL 已写入），
764 // ssaDestruct 无需在该前驱插入 STORE_LOCAL 写回。原实现栈空时
765 // 填 vreg=0，会把无关的 v0 写入参数槽位，破坏参数值。
766 auto& stack = slotStacks[slot];
767 if (stack.empty())
768 continue;
769 uint32_t vreg = stack.back();
770 // 在 succBlk 中查找 PHI
771 for (auto& instr : succBlk.instructions) {
772 if (instr.op == IROp::PHI && !instr.operands.empty() && instr.operands[0].index == phiDest) {
773 // 追加 (predNodeId, vreg) 对——predNodeId 以 IMM_UINT 存储
774 instr.operands.push_back(IROperand::imm(predNodeId));
775 instr.operands.push_back(IROperand::vreg(vreg));
776 break;
777 }
778 }
779 }
780 }
781
782 // 递归到支配树子节点（P2-10 fix: 用 newDomTree 而非已失效的 domTree）
783 for (uint32_t child : newDomTree.children(blockId)) {
784 rename(child);
785 }
786
787 // 弹栈
788 for (auto popIt = pushedSlots.rbegin(); popIt != pushedSlots.rend(); ++popIt) {
789 slotStacks[*popIt].pop_back();
790 }
791 };
792
793 rename(cfg.entryId());
794
795 LOG_DEBUG("ssaConstructPass: 插入 " + std::to_string(phiEntries.size()) + " 个 PHI 节点", "IR-SSA");
796 return true;
797}

799bool ssaDestructPass(IRFunction& ir) {
800 // 收集所有 PHI 指令，拆解为前驱块尾部的 STORE_LOCAL + 替换为 LOAD_LOCAL
801 bool hasPhi = false;
802 for (const auto& blk : ir.blocks) {
803 for (const auto& instr : blk.instructions) {
804 if (instr.op == IROp::PHI) {
805 hasPhi = true;
806 break;
807 }
808 }
809 if (hasPhi)
810 break;
811 }
812 if (!hasPhi)
813 return false;
814
815 // P2-10 fix3: 构建 CFG，使用 node ID 精确定位前驱块的终结指令位置。
816 // 原 labelPos 方案在 fall-through 块中失效：fall-through 块无终结指令，
817 // 从 LABEL 位置向前搜索终结指令会跨越 CFG 节点边界，找到下一个 CFG 节点
818 // （如循环条件块）的 JUMP_IF_FALSE，导致 STORE_LOCAL 插入到错误位置
819 // （如 PHI 所在块内部而非前驱块尾部），破坏循环变量更新语义。
820 //
821 // 新方案：ssaConstructPass 以 IMM_UINT 存储 CFG node ID 作为前驱标识符，
822 // ssaDestructPass 通过 cfg.node(predNodeId) 取得前驱节点的 sourceBlockIdx
823 // 与 [startInstr, endInstr) 指令范围，仅在此范围内搜索终结指令，
824 // 确保 STORE_LOCAL 插入在前驱块的正确位置（终结指令之前或 fall-through 块末尾）。
825 IRCFG cfg(ir);
826
827 // BUGFIX-P1 (#25): 两阶段处理，避免使用过期 CFG 索引。
828 // 原实现在处理每个 PHI 块时边收集边向前驱块插入 STORE_LOCAL，改变了指令
829 // 数量，但函数入口构建的 CFG 的 startInstr/endInstr 未更新；后续 PHI 若引用
830 // 已被修改的前驱块，会使用过期索引定位终结指令，导致 STORE_LOCAL 插入到
831 // 错误位置。修复：
832 // 阶段 1：在任何修改发生前，遍历所有 PHI，使用新鲜 CFG 收集待插入的
833 // STORE_LOCAL 信息（目标块索引、终结指令位置、slot、源 vreg）。
834 // 阶段 2：按 (blockIdx, position) 降序插入，靠后的插入不会使靠前位置失效。
835 bool modified = false;
836
837 // 阶段 1：收集所有待插入的 STORE_LOCAL（此时 CFG 仍与当前 IR 一致）
838 struct PendingStore {
839 size_t blockIdx;
840 size_t position;
841 std::vector<std::pair<uint32_t, uint32_t>> stores; // (slot, vreg)
842 };
843 std::map<std::pair<size_t, size_t>, std::vector<std::pair<uint32_t, uint32_t>>> predStores;
844
845 for (const auto& blk : ir.blocks) {
846 for (const auto& instr : blk.instructions) {
847 if (instr.op != IROp::PHI)
848 continue;
849 if (instr.operands.size() < 2)
850 continue;
851 uint32_t slot = instr.operands[1].index;
852
853 // 遍历 (pred_node_id, vreg) 对——pred_node_id 以 IMM_UINT 存储
854 for (size_t k = 2; k + 1 < instr.operands.size(); k += 2) {
855 uint32_t predNodeId = instr.operands[k].index;
856 uint32_t vreg = instr.operands[k + 1].index;
857
858 if (predNodeId >= cfg.size())
859 continue;
860
861 const auto& predNode = cfg.node(predNodeId);
862 size_t predBlockIdx = predNode.sourceBlockIdx;
863 const auto& predBlk = ir.blocks[predBlockIdx];
864
865 // 仅在 predNode 的指令范围 [startInstr, endInstr) 内搜索终结指令。
866 // fall-through 块（无终结指令）的 insertBefore = endInstr（节点末尾）。
867 size_t insertBefore = predNode.endInstr;
868 for (size_t i = predNode.startInstr; i < predNode.endInstr; ++i) {
869 if (isTerminatorOp(predBlk.instructions[i].op)) {
870 insertBefore = i;
871 break;
872 }
873 }
874 predStores[{predBlockIdx, insertBefore}].push_back({slot, vreg});
875 }
876 }
877 }
878
879 // 阶段 2：按 (blockIdx, position) 降序插入 STORE_LOCAL。
880 // 降序保证靠后/靠高位置的插入不会使靠前/靠低位置的索引失效。
881 {
882 std::vector<PendingStore> sortedStores;
883 sortedStores.reserve(predStores.size());
884 for (auto& [key, stores] : predStores) {
885 sortedStores.push_back(PendingStore{key.first, key.second, std::move(stores)});
886 }
887 std::sort(sortedStores.begin(), sortedStores.end(), [](const PendingStore& a, const PendingStore& b) {
888 if (a.blockIdx != b.blockIdx)
889 return a.blockIdx > b.blockIdx; // blockIdx 降序（不同块互不影响）
890 return a.position > b.position; // position 降序
891 });
892 for (auto& ps : sortedStores) {
893 auto& targetBlock = ir.blocks[ps.blockIdx];
894 size_t offset = 0;
895 for (auto& [slot, vreg] : ps.stores) {
896 IRInstruction storeInstr(IROp::STORE_LOCAL, {IROperand::local(slot), IROperand::vreg(vreg)}, 0);
897 targetBlock.instructions.insert(targetBlock.instructions.begin() + ps.position + offset,
898 std::move(storeInstr));
899 ++offset;
900 }
901 }
902 }
903
904 // 阶段 3：将所有 PHI 替换为 LOAD_LOCAL（不改变指令数量，不影响已插入位置）
905 for (auto& blk : ir.blocks) {
906 std::vector<IRInstruction> newInstrs;
907 newInstrs.reserve(blk.instructions.size());
908 for (const auto& instr : blk.instructions) {
909 if (instr.op == IROp::PHI) {
910 if (instr.operands.size() >= 2) {
911 uint32_t destVReg = instr.operands[0].index;
912 uint32_t slot = instr.operands[1].index;
913 newInstrs.emplace_back(
914 IROp::LOAD_LOCAL, IROperandList{IROperand::vreg(destVReg), IROperand::local(slot)}, instr.line);
915 modified = true;
916 }
917 } else {
918 newInstrs.push_back(instr);
919 }
920 }
921 blk.instructions = std::move(newInstrs);
922 }
923
924 LOG_DEBUG("ssaDestructPass: PHI 消除完成", "IR-SSA");
925 return modified;
926}

928// ============================================================
929// GVN（全局值编号）实现
930// ============================================================
931
932bool gvnPass(IRFunction& ir) {
933 IRCFG cfg(ir);
934 if (cfg.size() < 2)
935 return false; // 单块用 CSE 即可
936
937 DominatorTree domTree(cfg);
938
939 bool modified = false;
940
941 // 支配树 DFS，维护值编号表（继承自父块）
942 // valueTable: (op, operands) → dest vreg
943 // replacements: 被替换的 dest vreg → 替代 vreg（沿支配树向下传播）
944 std::function<void(uint32_t, std::unordered_map<std::string, uint32_t>&,
945 const std::unordered_map<uint32_t, uint32_t>&)>
946 dfs = [&](uint32_t blockId, std::unordered_map<std::string, uint32_t>& parentTable,
947 const std::unordered_map<uint32_t, uint32_t>& parentReplacements) {
948 const auto& node = cfg.node(blockId);
949 auto& blk = ir.blocks[node.sourceBlockIdx]; // 非 const：GVN 需修改操作数 vreg 引用
950
951 // 拷贝父表（scoped）
952 std::unordered_map<std::string, uint32_t> table = parentTable;
953 // 继承父块替换映射并添加本块新替换
954 std::unordered_map<uint32_t, uint32_t> replacements = parentReplacements;
955
956 for (size_t i = node.startInstr; i < node.endInstr; ++i) {
957 auto& instr = blk.instructions[i];
958
959 // PHI 指令：完全跳过，不应用替换到操作数，不加入值表。
960 // PHI 的 (pred_node_id, vreg) 对有特定语义：vreg 引用前驱块末尾的具体
961 // SSA 值。即使两个 vreg 计算相同值（如两个 LOAD_CONST 0），也不能在
962 // PHI 中替换——ssaDestructPass 据此 vreg 在前驱块尾部插入 STORE_LOCAL，
963 // 替换会破坏前驱块 SSA 值与 PHI 操作数的引用关系，导致循环变量初始化
964 // 错误（如 NestedLoopsPassIsolationWithLocals 测试中 i/j 被 GVN 合并到
965 // v0 后，destruct 插入的 STORE_LOCAL 引用错误 vreg）。
966 if (instr.op == IROp::PHI)
967 continue;
968
969 // 应用替换到 VIRTUAL 操作数
970 for (auto& op : instr.operands) {
971 if (op.kind == IROperandKind::VIRTUAL) {
972 auto rit = replacements.find(op.index);
973 if (rit != replacements.end()) {
974 op.index = rit->second;
975 modified = true;
976 }
977 }
978 }
979
980 // 跳过非纯计算
981 if (!isPureComputeForSSA(instr.op))
982 continue;
983 if (!hasDestVReg(instr))
984 continue;
985
986 uint32_t destVReg = getDestVReg(instr);
987 std::string key = encodeValueKey(instr);
988 auto it = table.find(key);
989 if (it != table.end()) {
990 // 相同表达式已存在：替换 dest 引用
991 replacements[destVReg] = it->second;
992 } else {
993 table[key] = destVReg;
994 }
995 }
996
997 // 递归到支配树子节点（传播替换映射）
998 for (uint32_t child : domTree.children(blockId)) {
999 dfs(child, table, replacements);
1000 }
1001 };
1002
1003 std::unordered_map<std::string, uint32_t> emptyTable;
1004 std::unordered_map<uint32_t, uint32_t> emptyReplacements;
1005 dfs(cfg.entryId(), emptyTable, emptyReplacements);
1006
1007 if (modified) {
1008 LOG_DEBUG("gvnPass: 全局值编号完成", "IR-SSA");
1009 }
1010 return modified;
1011}

1013// ============================================================
1014// LICM（循环不变代码外提）实现
1015// ============================================================
1016
1017bool licmPass(IRFunction& ir) {
1018 // P2-10 fix: 嵌套循环索引失效保护。原实现在入口处一次性构建 CFG/loopInfo，
1019 // 处理一个循环并修改 IR（删除+插入）后，后续循环的 CFG 节点范围
1020 // (startInstr/endInstr) 已偏移，继续处理会删错指令。
1021 // 正确做法：每次修改后重建 CFG 重跑，直到无更多可外提指令。
1022 bool modified = false;
1023
1024 // L17 fix: 不动点迭代燃料上限。多循环函数（如 ackermann 的短路/分支 CFG
1025 // 被 NaturalLoopInfo 识别出多个自然循环）中，外提到循环 A 的 preheader
1026 // 可能落入循环 B 的 body，下一轮又被当作 B 的不变候选外提回 A 方向，
1027 // 候选在循环间乒乓迁移，不动点永不收敛 → 编译期挂死（复现：
1028 // irOptimize+irSSAOptimize 编译 ackermann）。LICM 是纯优化，提前停止
1029 // 不影响语义，故加燃料上限保证终止：正常收敛只需 O(可外提指令数) 轮。
1030 size_t totalInstrs = 0;
1031 for (const auto& blk : ir.blocks)
1032 totalInstrs += blk.instructions.size();
1033 const size_t kMaxIterations = totalInstrs + 16;
1034 size_t iterations = 0;
1035
1036 // 结构分析复用（第四轮优化）：LICM 只增删纯计算指令（isPureComputeForSSA
1037 // 不含 LABEL/终结指令），从不增删/移动块图定界符，因此节点分区唯一可能的
1038 // 变化是"非 LABEL 起始的指令块（块首无 LABEL gap / 终结指令后 gap）的指令
1039 // 被全部外提 → 该节点消失"，且外提指令的插入（preheader 终结指令前或尾部
1040 // gap 末尾）不产生新节点——节点数严格单调不增，任一分区变化必使节点数减 1。
1041 // 故"重建 IRCFG 后节点数不变"⇔ 分区与基准一致 ⇒ 节点 id 语义未变，
1042 // DominatorTree（CHK 不动点）/NaturalLoopInfo（回边扫描）可跨不动点批次
1043 // 复用；每轮仅重建 IRCFG 取新鲜指令区间 (startInstr/endInstr)（P2-10 语义
1044 // 保留）。节点数变化（分区塌缩）时重建三者，退化为原每批全量重建行为。
1045 // 注意 dominates/loops 查询只读已存的 idom_/loops_ 数据，不解引用 cfg_
1046 // 引用；analysisCfg 原地赋值保持地址稳定，重建后引用不悬垂。
1047 IRCFG analysisCfg(ir);
1048 if (analysisCfg.size() < 2)
1049 return false;
1050 auto domTree = std::make_unique<DominatorTree>(analysisCfg);
1051 auto loopInfo = std::make_unique<NaturalLoopInfo>(analysisCfg, *domTree);
1052 if (loopInfo->loops().empty())
1053 return false;
1054 size_t analysisNodeCount = analysisCfg.size();
1055
1056 while (true) {
1057 if (++iterations > kMaxIterations) {
1058 LOG_DEBUG("licmPass: 达到迭代上限 " + std::to_string(kMaxIterations) + "，强制收敛（函数 " + ir.name + "）",
1059 "IR-SSA");
1060 break;
1061 }
1062 IRCFG cfg(ir); // 每轮重建：仅取外提增删后的新鲜指令区间与 preds
1063 if (cfg.size() != analysisNodeCount) {
1064 analysisCfg = IRCFG(ir);
1065 domTree = std::make_unique<DominatorTree>(analysisCfg);
1066 loopInfo = std::make_unique<NaturalLoopInfo>(analysisCfg, *domTree);
1067 analysisNodeCount = analysisCfg.size();
1068 }
1069
1070 if (loopInfo->loops().empty())
1071 break;
1072
1073 bool changedThisIter = false;
1074
1075 for (const auto& loop : loopInfo->loops()) {
1076 uint32_t header = loop.header;
1077
1078 // 寻找 preheader：header 的前驱中不在循环体内的节点
1079 uint32_t preheader = IRCFG::kInvalid;
1080 int preheaderCount = 0;
1081 for (uint32_t p : cfg.node(header).preds) {
1082 bool inLoop = false;
1083 for (uint32_t b : loop.body) {
1084 if (b == p) {
1085 inLoop = true;
1086 break;
1087 }
1088 }
1089 if (!inLoop) {
1090 preheader = p;
1091 ++preheaderCount;
1092 }
1093 }
1094
1095 if (preheaderCount != 1 || preheader == IRCFG::kInvalid)
1096 continue;
1097
1098 // L17 fix: preheader 必须支配 header（教科书 LICM 前提条件）。
1099 // AstIRBuilder 的尾递归→循环转换会生成回到函数入口的 JUMP，
1100 // 产生 header=entry 的自然循环；多处尾调用时存在多个共享 entry
1101 // header 的循环，彼此的回边源块互为对方的"非循环前驱"。若不校验
1102 // 支配性，指令会被外提到不支配 header 的回边源块：
1103 // (1) 首次进入函数不经过该块 → 读未初始化寄存器（语义错）；
1104 // (2) 候选在两个循环的回边源块间乒乓迁移，不动点永不收敛
1105 // → 编译期挂死（复现：irOptimize 编译 ackermann）。
1106 if (!domTree->dominates(preheader, header))
1107 continue;
1108
1109 // 收集循环体内定义的所有 vreg
1110 std::unordered_set<uint32_t> loopDefinedVRegs;
1111 for (uint32_t b : loop.body) {
1112 const auto& node = cfg.node(b);
1113 const auto& blk = ir.blocks[node.sourceBlockIdx];
1114 for (size_t i = node.startInstr; i < node.endInstr; ++i) {
1115 const auto& instr = blk.instructions[i];
1116 if (hasDestVReg(instr)) {
1117 loopDefinedVRegs.insert(getDestVReg(instr));
1118 }
1119 }
1120 }
1121
1122 // 寻找可外提的指令（纯计算 + 所有 vreg 操作数在循环外定义 + 不涉及 local slot）
1123 struct HoistCandidate {
1124 uint32_t blockId;
1125 size_t blockIdx;
1126 size_t localIdx;
1127 IRInstruction instr;
1128 };
1129 std::vector<HoistCandidate> candidates;
1130
1131 for (uint32_t b : loop.body) {
1132 if (b == header)
1133 continue;
1134 const auto& node = cfg.node(b);
1135 const auto& blk = ir.blocks[node.sourceBlockIdx];
1136 for (size_t i = node.startInstr; i < node.endInstr; ++i) {
1137 const auto& instr = blk.instructions[i];
1138 if (!isPureComputeForSSA(instr.op))
1139 continue;
1140 if (!hasDestVReg(instr))
1141 continue;
1142
1143 bool invariant = true;
1144 for (size_t k = 1; k < instr.operands.size(); ++k) {
1145 if (instr.operands[k].kind == IROperandKind::VIRTUAL) {
1146 if (loopDefinedVRegs.count(instr.operands[k].index)) {
1147 invariant = false;
1148 break;
1149 }
1150 } else if (instr.operands[k].kind == IROperandKind::LOCAL_SLOT) {
1151 invariant = false;
1152 break;
1153 }
1154 }
1155 if (invariant) {
1156 candidates.push_back({b, node.sourceBlockIdx, i, instr});
1157 }
1158 }
1159 }
1160
1161 if (candidates.empty())
1162 continue;
1163
1164 // 步骤 1：从原块删除外提指令（按 blockIdx/localIdx 降序）
1165 std::sort(candidates.begin(), candidates.end(), [](const HoistCandidate& a, const HoistCandidate& b) {
1166 if (a.blockIdx != b.blockIdx)
1167 return a.blockIdx > b.blockIdx;
1168 return a.localIdx > b.localIdx;
1169 });
1170 for (const auto& cand : candidates) {
1171 ir.blocks[cand.blockIdx].instructions.erase(ir.blocks[cand.blockIdx].instructions.begin() +
1172 cand.localIdx);
1173 }
1174
1175 // 步骤 2：在 preheader 的终结指令前插入外提指令
1176 const auto& phNode = cfg.node(preheader);
1177 auto& phBlk = ir.blocks[phNode.sourceBlockIdx];
1178 size_t insertBefore = phNode.endInstr;
1179 if (insertBefore > phBlk.instructions.size())
1180 insertBefore = phBlk.instructions.size();
1181 for (size_t i = phNode.startInstr; i < insertBefore; ++i) {
1182 if (isTerminatorOp(phBlk.instructions[i].op)) {
1183 insertBefore = i;
1184 break;
1185 }
1186 }
1187 size_t offset = 0;
1188 for (const auto& cand : candidates) {
1189 phBlk.instructions.insert(phBlk.instructions.begin() + insertBefore + offset, cand.instr);
1190 ++offset;
1191 }
1192
1193 modified = true;
1194 changedThisIter = true;
1195 break; // 重建 CFG 后重跑
1196 }
1197
1198 if (!changedThisIter)
1199 break;
1200 }
1201
1202 if (modified) {
1203 LOG_DEBUG("licmPass: 循环不变代码外提完成", "IR-SSA");
1204 }
1205 return modified;
1206}

1209// 函数内联实现
1210// ============================================================
1211
1212bool inlinePass(IRModule& module) {
1213 bool modified = false;
1214 constexpr size_t kInlineThreshold = 15; // 指令数阈值
1215 constexpr int kMaxTotalInlineOps = 1000; // 总指令预算（防代码膨胀）
1216
1217 int totalInlinedOps = 0;
1218
1219 // 收集所有可内联的函数
1220 auto isInlineable = [&](const IRFunction& fn) -> bool {
1221 if (fn.isGenerator)
1222 return false;
1223 if (fn.blocks.size() != 1)
1224 return false; // 仅内联单块函数
1225 const auto& instrs = fn.blocks[0].instructions;
1226 size_t count = 0;
1227 for (const auto& instr : instrs) {
1228 // 跳过 LABEL
1229 if (instr.op == IROp::LABEL)
1230 continue;
1231 // 不内联含控制流的函数（LABEL 除外）
1232 if (instr.op == IROp::JUMP || instr.op == IROp::JUMP_IF_FALSE)
1233 return false;
1234 // L18 eng-tailcall: 不内联含 TAIL_CALL 的函数——TAIL_CALL 的帧复用
1235 // 语义依赖自己的帧，内联到 caller 后不成立。
1236 if (instr.op == IROp::TAIL_CALL)
1237 return false;
1238 // 不内联含 try/throw 的函数
1239 if (instr.op == IROp::TRY_BEGIN || instr.op == IROp::TRY_END || instr.op == IROp::THROW)
1240 return false;
1241 // 不内联含 yield 的函数（生成器）
1242 if (instr.op == IROp::YIELD)
1243 return false;
1244 // 七特性 MVP 阶段 4：不内联含 await 的函数（同步 drain 可中断，与生成器同类）
1245 if (instr.op == IROp::AWAIT)
1246 return false;
1247 // L17 fix: 不内联含闭包/upvalue 指令的函数——MAKE_CLOSURE 的捕获
1248 // 列表（isLocal+slot）与 LOAD/STORE/CLOSE_UPVALUE 的 uv 索引均指向
1249 // callee 自己的帧/捕获表，内联到 caller 后捕获错误变量（复现：
1250 // 内联 outerFn 后闭包误捕 caller 槽位，返回闭包对象而非调用结果）。
1251 if (instr.op == IROp::MAKE_CLOSURE || instr.op == IROp::LOAD_UPVALUE || instr.op == IROp::STORE_UPVALUE ||
1252 instr.op == IROp::CLOSE_UPVALUE)
1253 return false;
1254 ++count;
1255 }
1256 // BUGFIX-P2 #58: 若 callee 含 STORE_LOCAL 写入参数槽位（slot < arity），
1257 // 说明参数被重新赋值。内联后参数映射（LOAD_LOCAL → 实参 vreg）无法正确
1258 // 处理后续对同一参数的读取（会读到 caller 的槽位而非更新后的值），拒绝内联。
1259 // L17 fix: 扩展为拒绝一切 STORE_LOCAL 与 slot>=arity 的 LOCAL_SLOT 引用——
1260 // callee 自有局部变量（slot>=arity）的读写在内联后会直接踩踏 caller 的
1261 // 同号槽位（无槽位重映射机制），语义错误。可内联集合收窄为
1262 // "纯参数表达式函数"（如 add(a,b)=a+b），这正是小函数内联的目标场景。
1263 // STORE_LOCAL operands: [slot, src_vreg]，slot 在 operands[0]。
1264 for (const auto& instr2 : instrs) {
1265 if (instr2.op == IROp::STORE_LOCAL)
1266 return false; // 写局部变量/参数，不可内联
1267 for (const auto& opnd : instr2.operands) {
1268 if (opnd.kind == IROperandKind::LOCAL_SLOT && opnd.index >= static_cast<uint32_t>(fn.arity)) {
1269 return false; // 引用 callee 自有局部槽位，内联后踩 caller 帧
1270 }
1271 }
1272 }
1273 return count <= kInlineThreshold;
1274 };
1275
1276 // 遍历所有函数（main + 子函数），寻找可内联的 CALL 指令
1277 std::vector<IRFunction*> allFns;
1278 if (module.mainFunction)
1279 allFns.push_back(module.mainFunction.get());
1280 for (auto& fn : module.functions) {
1281 if (fn)
1282 allFns.push_back(fn.get());
1283 }
1284
1285 // PERF-INLINE: 迭代到不动点（最多 3 轮）——单轮内联会暴露新的 CALL（a→b→c
1286 // 传递内联），且被内联体复制进 caller 后原 CALL 不再被重扫。总指令预算跨轮
1287 // 共享，防代码膨胀。每轮内无修改即收敛。
1288 for (int round = 0; round < 3; ++round) {
1289 bool roundModified = false;
1290 for (auto* caller : allFns) {
1291 if (!caller || caller->blocks.empty())
1292 continue;
1293
1294 // PERF-INLINE: 扫描 caller 全部块——原实现仅扫 blocks[0]，任何含控制流
1295 // 的 caller（≥2 块）中位于其他块的 CALL 永远不会被内联。
1296 for (size_t blockIdx = 0; blockIdx < caller->blocks.size(); ++blockIdx) {
1297 auto& instrs = caller->blocks[blockIdx].instructions;
1298 std::vector<IRInstruction> newInstrs;
1299 newInstrs.reserve(instrs.size());
1300 bool blockModified = false;
1301
1302 for (size_t i = 0; i < instrs.size(); ++i) {
1303 const auto& instr = instrs[i];
1304
1305 // 查找 CALL 指令: dest = call name(args)
1306 // operands: [dest, name_idx, arg_count, arg1, arg2, ...]
1307 if (instr.op == IROp::CALL && instr.operands.size() >= 3) {
1308 uint32_t nameIdx = instr.operands[1].index;
1309 if (nameIdx < caller->globalNames.size()) {
1310 const std::string& calleeName = caller->globalNames[nameIdx];
1311 IRFunction* callee = module.findFunction(calleeName);
1312
1313 // 防递归内联
1314 // L17 fix3: 实参数量必须等于 callee arity——否则运行时应报
1315 // "参数数量不匹配"错误，内联会吞掉该错误破坏三后端一致性
1316 // （复现：foo(a,b) 调用 foo(1,2,3) 内联后静默成功）。
1317 uint32_t callArgCount = instr.operands[2].index;
1318 if (callee && callee != caller && callArgCount == static_cast<uint32_t>(callee->arity) &&
1319 isInlineable(*callee) && totalInlinedOps < kMaxTotalInlineOps) {
1320 // 内联：复制 callee 指令，重映射 vreg
1321 uint32_t vregBase = caller->nextVReg;
1322 uint32_t calleeVRegCount = callee->nextVReg;
1323 caller->nextVReg += calleeVRegCount;
1324
1325 auto remapVReg = [&](uint32_t v) -> uint32_t { return vregBase + v; };
1326
1327 // P2-10 fix: 常量池/名称池重映射——callee 的 CONSTANT/FUNC_NAME/
1328 // GLOBAL_NAME/FIELD_NAME 操作数索引指向 callee 自己的 constants/
1329 // globalNames 数组，内联到 caller 后必须重映射到 caller 的对应数组，
1330 // 否则 LOAD_CONST 读到错误常量、CALL 调用错误函数名。
1331 std::unordered_map<uint32_t, uint32_t> constRemap;
1332 std::unordered_map<uint32_t, uint32_t> nameRemap;
1333 auto remapConst = [&](uint32_t idx) -> uint32_t {
1334 auto it = constRemap.find(idx);
1335 if (it != constRemap.end())
1336 return it->second;
1337 if (idx < callee->constants.size()) {
1338 uint32_t newIdx = caller->addConstant(callee->constants[idx]);
1339 constRemap[idx] = newIdx;
1340 return newIdx;
1341 }
1342 return idx;
1343 };
1344 auto remapName = [&](uint32_t idx) -> uint32_t {
1345 auto it = nameRemap.find(idx);
1346 if (it != nameRemap.end())
1347 return it->second;
1348 if (idx < callee->globalNames.size()) {
1349 uint32_t newIdx = caller->addGlobal(callee->globalNames[idx]);
1350 nameRemap[idx] = newIdx;
1351 return newIdx;
1352 }
1353 return idx;
1354 };
1355
1356 // P2-10 fix: 参数映射——callee 的 LOAD_LOCAL slot_p (p < arity) 读取的是
1357 // callee 自己的参数槽位，但内联到 caller 后这些槽位属于 caller 的局部变量，
1358 // 语义错误（会读到无关变量）。正确做法：将 callee 参数的 LOAD_LOCAL 替换为
1359 // 对 CALL 指令中对应实参 vreg 的引用（DUP 语义）。
1360 // CALL operands: [dest, name_idx, arg_count, arg1, arg2, ...]
1361 // arg_{p+1} 对应 callee 的 slot_p（参数 p）。
1362 std::unordered_map<uint32_t, uint32_t> paramSlotToArgVReg;
1363 if (instr.operands.size() >= 3 + static_cast<size_t>(callee->arity)) {
1364 for (int p = 0; p < callee->arity; ++p) {
1365 uint32_t argVReg = instr.operands[3 + p].index;
1366 paramSlotToArgVReg[static_cast<uint32_t>(p)] = argVReg;
1367 }
1368 }
1369 // paramDestRemap: callee 内 LOAD_LOCAL slot_p 的 dest vreg（已重映射）
1370 // → 对应实参 vreg。后续引用此 dest 的指令直接用实参 vreg。
1371 // 局限性：若 callee 重新赋值参数（STORE_LOCAL slot_p），后续 LOAD_LOCAL
1372 // 仍会读取 caller 槽位（错误）。当前仅支持参数只读的常见场景。
1373 std::unordered_map<uint32_t, uint32_t> paramDestRemap;
1374 auto resolveVReg = [&](uint32_t v) -> uint32_t {
1375 auto rit = paramDestRemap.find(v);
1376 return rit != paramDestRemap.end() ? rit->second : v;
1377 };
1378
1379 size_t calleeOps = 0;
1380 for (const auto& cInstr : callee->blocks[0].instructions) {
1381 if (cInstr.op == IROp::LABEL)
1382 continue;
1383 if (cInstr.op == IROp::RETURN_NULL) {
1384 // L17 fix: 隐式/显式 `return;` → CALL 的 dest 必须收到 null。
1385 // 原实现直接跳过，dest vreg 无定义，lowering 后读脏寄存器
1386 // （复现：`fun foo(){return;} var x=foo();` 内联后 x 为闭包脏值）。
1387 uint32_t destVReg = instr.operands[0].index;
1388 newInstrs.emplace_back(IROp::LOAD_NULL,
1389 IROperandList{IROperand::vreg(destVReg)}, cInstr.line);
1390 // L17 fix2: 首个终结指令后的代码不可达，停止复制——
1391 // 否则 `return 7;` 后的尾随隐式 RETURN_NULL 会把 dest 覆盖为 null。
1392 break;
1393 }
1394 if (cInstr.op == IROp::RETURN) {
1395 // return src → 将 src 赋值给 CALL 的 dest
1396 if (!cInstr.operands.empty() &&
1397 cInstr.operands[0].kind == IROperandKind::VIRTUAL) {
1398 uint32_t retVal = resolveVReg(remapVReg(cInstr.operands[0].index));
1399 uint32_t destVReg = instr.operands[0].index;
1400 newInstrs.emplace_back(
1401 IROp::DUP,
1402 IROperandList{IROperand::vreg(destVReg), IROperand::vreg(retVal)},
1403 cInstr.line);
1404 }
1405 // L17 fix2: 同上，首个 RETURN 之后的指令不可达，停止复制。
1406 break;
1407 }
1408
1409 // 参数 LOAD_LOCAL：跳过，记录 dest → 实参 vreg 映射
1410 if (cInstr.op == IROp::LOAD_LOCAL && cInstr.operands.size() >= 2 &&
1411 cInstr.operands[1].kind == IROperandKind::LOCAL_SLOT) {
1412 uint32_t slot = cInstr.operands[1].index;
1413 if (slot < static_cast<uint32_t>(callee->arity)) {
1414 auto pit = paramSlotToArgVReg.find(slot);
1415 if (pit != paramSlotToArgVReg.end()) {
1416 uint32_t remappedDest = remapVReg(cInstr.operands[0].index);
1417 paramDestRemap[remappedDest] = pit->second;
1418 continue; // 跳过 LOAD_LOCAL，后续引用改用实参 vreg
1419 }
1420 }
1421 }
1422
1423 // 复制指令，重映射 vreg + 常量池/名称池 + 参数 dest 替换
1424 IROperandList newOps;
1425 newOps.reserve(cInstr.operands.size());
1426
1427 // D9 fix: DEFINE_CLASS 类元数据操作数重映射（IMM_UINT kind 承载的池索引）。
1428 // 通用重映射只处理 CONSTANT/FUNC_NAME/GLOBAL_NAME/FIELD_NAME kind，而
1429 // DEFINE_CLASS 的字段默认值常量索引（[3+i*3+1]）与父类名索引（[1]）是 IMM_UINT
1430 // kind，原样复制会指向 callee 自己的池 → 内联后索引越界/错位
1431 // （复现：`fun make(){ class L { var x=42; } return L; }` 内联后
1432 // RegisterBytecodeBackend 报"DEFINE_CLASS 字段默认值常量索引越界"）。
1433 // 操作数布局：[0]className(FUNC_NAME) [1]parentName(IMM_UINT/UINT32_MAX)
1434 // [2]fieldCount [3+i*3]fieldName(FIELD_NAME)
1435 // [3+i*3+1]defaultConstIdx(IMM_UINT/UINT32_MAX)
1436 // [3+i*3+2]exprLocalSlot(IMM_UINT/UINT32_MAX，含非字面量默认值的函数
1437 // 因 STORE_LOCAL 已被 isInlineable 拒绝内联，此处防御性保留)
1438 // [3+3F]methodCount [方法对...]
1439 const uint32_t d9FieldCount =
1440 (cInstr.op == IROp::DEFINE_CLASS && cInstr.operands.size() >= 3)
1441 ? cInstr.operands[2].index
1442 : 0;
1443 for (size_t oi = 0; oi < cInstr.operands.size(); ++oi) {
1444 const auto& op = cInstr.operands[oi];
1445 if (op.kind == IROperandKind::VIRTUAL) {
1446 newOps.push_back(IROperand::vreg(resolveVReg(remapVReg(op.index))));
1447 } else if (op.kind == IROperandKind::CONSTANT) {
1448 newOps.push_back(IROperand::constant(remapConst(op.index)));
1449 } else if (op.kind == IROperandKind::FUNC_NAME ||
1450 op.kind == IROperandKind::GLOBAL_NAME ||
1451 op.kind == IROperandKind::FIELD_NAME) {
1452 newOps.push_back(IROperand{op.kind, remapName(op.index)});
1453 } else if (op.kind == IROperandKind::IMM_UINT &&
1454 cInstr.op == IROp::DEFINE_CLASS) {
1455 // DEFINE_CLASS 的 IMM_UINT 池索引特判（D9 fix）
1456 if (oi == 1 && op.index != UINT32_MAX) {
1457 newOps.push_back(IROperand::imm(remapName(op.index))); // 父类名
1458 } else if (oi >= 3 && oi < 3 + d9FieldCount * 3 && (oi - 3) % 3 == 1 &&
1459 op.index != UINT32_MAX) {
1460 newOps.push_back(IROperand::imm(remapConst(op.index))); // 字段默认值
1461 } else {
1462 newOps.push_back(op);
1463 }
1464 } else {
1465 newOps.push_back(op);
1466 }
1467 }
1468 newInstrs.emplace_back(cInstr.op, std::move(newOps), cInstr.line);
1469 ++calleeOps;
1470 }
1471 totalInlinedOps += static_cast<int>(calleeOps);
1472 blockModified = true;
1473 continue; // 跳过原 CALL 指令
1474 }
1475 }
1476 }
1477
1478 newInstrs.push_back(instr);
1479 }
1480
1481 // 仅在本块发生内联时回写（原实现用全局 modified 判断，未修改的块
1482 // 也会被无谓搬移一遍）
1483 if (blockModified) {
1484 instrs = std::move(newInstrs);
1485 roundModified = true;
1486 }
1487 } // blockIdx
1488 } // caller
1489
1490 modified = modified || roundModified;
1491 if (!roundModified)
1492 break; // 收敛
1493 } // round
1494
1495 if (modified) {
1496 LOG_DEBUG("inlinePass: 函数内联完成", "IR-SSA");
1497 }
1498 return modified;
1499}
