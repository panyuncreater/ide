283 /// 统一通知，避免双重通知。

285 /// 获取所选帧的局部变量（用于 VariableInspector 显示 + Watch 求值注入）。
286 /// selectedFrame_ == -1 时返回当前帧（栈顶）locals。
287 /// VM 模式 / Interpreter 调试暂停模式均支持。其他状态返回空。
288 /// 越界时 fallback 到栈顶（与 GDB `frame N` 行为一致）。
289 /// @note R121-build fix: 改为 inline 以避免 VariableInspectorPanel.cpp 依赖
290 /// IdeController.cpp 的非 inline 符号（minilang_tests 不链接 IdeController.cpp）。
291 /// 所依赖的 vmStepper_ / debugCoord_ 方法均已 inline。
292 std::vector<std::pair<std::string, Value>> getSelectedFrameLocals() const {
293 std::vector<std::pair<std::string, Value>> result;
294 bool vmMode = vmStepper_.isRunning() || vmStepper_.isInitialized();
295 bool interpPaused = debugCoord_.isPaused();
296 if (!vmMode && !interpPaused) {
297 return result;
298 }
299 if (vmMode) {
300 size_t frameCount = vmStepper_.getFrameCount();
301 if (frameCount == 0)
302 return result;
303 size_t targetIdx = static_cast<size_t>(selectedFrame_);
304 // selectedFrame_ == -1 或越界 → fallback 到栈顶（frameCount - 1）
305 if (selectedFrame_ < 0 || targetIdx >= frameCount) {
306 targetIdx = frameCount - 1;
307 }
308 auto locals = vmStepper_.getFrameLocalsAt(targetIdx);
309 for (const auto& kv : locals) {
310 result.emplace_back(kv.first, kv.second);
311 }
312 } else {
313 // Interpreter 调试暂停模式：通过 debugCoord_ 直接访问 CallFrame.env
314 // 越界由 debugCoord_.getFrameLocalsAt 内部处理（返回空），此处需手动 fallback 到栈顶
315 if (selectedFrame_ < 0) {
316 // -1 表示栈顶：用 getDebugVariableSnapshot() 拿当前帧 locals（保持 R117 行为）
317 for (const auto& snap : debugCoord_.getDebugVariableSnapshot()) {
318 result.emplace_back(snap.name, snap.value);
319 }
320 } else {
321 auto locals = debugCoord_.getFrameLocalsAt(selectedFrame_);
322 if (locals.empty()) {
323 // 越界 fallback：尝试栈顶
324 for (const auto& snap : debugCoord_.getDebugVariableSnapshot()) {
325 result.emplace_back(snap.name, snap.value);
326 }
327 } else {
328 result = std::move(locals);
329 }
330 }
331 }
332 return result;
333 }
