25/// 共享的编译流程：Lexer → Parser → Compiler，返回 CompileResult。
26/// 失败时设置 error 并返回空 optional。
27struct CompileOutput {
28 bool ok = false;
29 std::string error;
30 // CompileResult 通过移动语义返回（调用方持有所有权）
31};

33/// 编译源码到 CompileResult。成功返回 true。
34bool compileSource(const std::string& src, /*out*/ CompileResult& cr, /*out*/ std::string& error) {
35 Lexer lx;
36 auto tk = lx.scan(src);
37 Parser p;
38 auto ast = p.parse(tk);
39 if (!ast) {
40 error = "<parse-fail>";
41 return false;
42 }
43 Compiler c;
44 cr = c.compile(*ast);
45 if (c.getDiagnostics().hasErrors()) {
46 error = "<compile:" + c.getLastError() + ">";
47 return false;
48 }
49 return true;
50}

54TypeFeedbackResult runTypeFeedback(const std::string& src, const std::string& methodChunkName, uint64_t threshold) {
55 TypeFeedbackResult r;
56 CompileResult cr;
57 std::string compileErr;
58 if (!compileSource(src, cr, compileErr)) {
59 r.error = compileErr;
60 return r;
61 }
62 JITBackend jit;
63 if (!methodChunkName.empty()) {
64 jit.setHotThreshold(methodChunkName, threshold);
65 }
66 jit.setOutputCallback([&](const std::string& s) { r.output += s; });
67 JitResult res = jit.execute(cr);
68 if (res != JitResult::OK) {
69 r.error = jit.getLastError();
70 return r;
71 }
72 r.ok = true;
73
74 auto tf = jit.getTypeFeedback(); // vector<pair<string,TypeFeedback>>
75 auto specialized = jit.getSpecializedChunks(); // vector<string>（拷贝）
76 r.specializedChunks = specialized;
77
78 std::set<std::string> specSet(specialized.begin(), specialized.end());
79 r.rows.reserve(tf.size());
80 for (const auto& [name, fb] : tf) {
81 TypeFeedbackRow row;
82 row.chunkName = name;
83 row.intCount = fb.intCount;
84 row.floatCount = fb.floatCount;
85 row.otherCount = fb.otherCount;
86 if (specSet.count(name)) {
87 // 推断特化类型：floatCount>0 → FLOAT，否则 INT
88 row.decision = (fb.floatCount > 0 && fb.otherCount == 0) ? "FLOAT 特化" : "INT 特化";
89 } else {
90 row.decision = "不特化";
91 }
92 r.rows.push_back(std::move(row));
93 }
94 return r;
95}

97HotspotResult runHotspot(const std::string& src, const std::string& methodChunkName, uint64_t threshold) {
98 HotspotResult r;
99 CompileResult cr;
100 std::string compileErr;
101 if (!compileSource(src, cr, compileErr)) {
102 r.error = compileErr;
103 return r;
104 }
105 JITBackend jit;
106 if (!methodChunkName.empty()) {
107 jit.setHotThreshold(methodChunkName, threshold);
108 }
109 jit.setOutputCallback([&](const std::string& s) { r.output += s; });
110 JitResult res = jit.execute(cr);
111 if (res != JitResult::OK) {
112 r.error = jit.getLastError();
113 return r;
114 }
115 r.ok = true;
116
117 auto callCounts = jit.getHotChunkStats(); // vector<pair<string,uint64_t>>
118 auto recompileFlags = jit.getRecompileStats(); // vector<pair<string,uint64_t>>
119
120 // 合并 callCount 与 recompileFlag（按 chunkName 对齐）
121 r.rows.reserve(callCounts.size());
122 for (const auto& [name, cnt] : callCounts) {
123 HotspotRow row;
124 row.chunkName = name;
125 row.callCount = cnt;
126 // 查找对应的 recompileFlag
127 for (const auto& [rname, flag] : recompileFlags) {
128 if (rname == name) {
129 row.recompiledFlag = flag;
130 break;
131 }
132 }
133 r.rows.push_back(std::move(row));
134 }
135 return r;
136}

138TierMetricsResult runTieredCompilation(const std::string& src, const std::string& osrChunkName, uint64_t osrThreshold) {
139 TierMetricsResult r;
140 CompileResult cr;
141 std::string compileErr;
142 if (!compileSource(src, cr, compileErr)) {
143 r.error = compileErr;
144 return r;
145 }
146 JITBackend jit;
147 // 启用分层编译（隐含 lazy compilation）+ OSR 迁移
148 jit.setTieredCompilation(true);
149 jit.setOsrMigrationMode(true);
150 if (!osrChunkName.empty() && osrThreshold > 0) {
151 jit.setOsrThreshold(osrChunkName, osrThreshold);
152 }
153 jit.setOutputCallback([&](const std::string& s) { r.output += s; });
154 JitResult res = jit.execute(cr);
155 if (res != JitResult::OK) {
156 r.error = jit.getLastError();
157 return r;
158 }
159 r.ok = true;
160 r.tieredEnabled = true;
161
162 // 采集 per-chunk tier 状态
163 auto tiers = jit.getChunkTiers(); // vector<pair<string, JitTier>>
164 auto osrLoops = jit.getOsrLoopStats();
165 auto osrRecompiles = jit.getOsrRecompileStats();
166 auto lazyChunks = jit.getLazyCompiledChunks();
167 auto osrMigrated = jit.getOsrMigratedChunks();
168 auto deoptChunks = jit.getDeoptimizedChunks();
169 r.totalDeoptCount = jit.getDeoptCount();
170 r.lazyCompiledChunks = lazyChunks;
171 r.osrMigratedChunks = osrMigrated;
172 r.deoptimizedChunks = deoptChunks;
173
174 std::set<std::string> lazySet(lazyChunks.begin(), lazyChunks.end());
175 std::set<std::string> osrMigratedSet(osrMigrated.begin(), osrMigrated.end());
176 std::set<std::string> deoptSet(deoptChunks.begin(), deoptChunks.end());
177
178 r.rows.reserve(tiers.size());
179 for (const auto& [name, tier] : tiers) {
180 TierMetricsRow row;
181 row.chunkName = name;
182 switch (tier) {
183 case minilang::JitTier::Interpreter:
184 row.tier = "Tier 0 (Interpreter)";
185 break;
186 case minilang::JitTier::Baseline:
187 row.tier = "Tier 1 (Baseline)";
188 break;
189 case minilang::JitTier::Specialized:
190 row.tier = "Tier 2 (Specialized)";
191 break;
192 }
193 row.lazyCompiled = lazySet.count(name) > 0;
194 row.osrMigrated = osrMigratedSet.count(name) > 0;
195 row.deoptimized = deoptSet.count(name) > 0;
196 // 查找 OSR 回边计数和重编译标志
197 for (const auto& [oname, cnt] : osrLoops) {
198 if (oname == name) {
199 row.osrLoopCount = cnt;
200 break;
201 }
202 }
203 for (const auto& [oname, flag] : osrRecompiles) {
204 if (oname == name) {
205 row.osrRecompiled = flag;
206 break;
207 }
208 }
209 r.rows.push_back(std::move(row));
210 }
211 return r;
212}

214AsmDumpResult runAsmDump(const std::string& src) {
215 AsmDumpResult r;
216 CompileResult cr;
217 std::string compileErr;
218 if (!compileSource(src, cr, compileErr)) {
219 r.error = compileErr;
220 return r;
221 }
222
223 // 字节码反汇编：mainChunk + 全部函数/方法 chunk（复用 disassemble）
224 r.bytecodeText = "== mainChunk ==\n" + cr.mainChunk.disassemble();
225 for (const auto& [name, chunk] : cr.functionChunks) {
226 r.bytecodeText += "\n== " + name + " ==\n" + chunk.disassemble();
227 }
228
229 // JIT 编译+执行，捕获发射的汇编文本
230 JITBackend jit;
231 jit.setAsmCapture(true);
232 jit.setOutputCallback([&](const std::string& s) { r.output += s; });
233 JitResult res = jit.execute(cr);
234 if (res != JitResult::OK) {
235 r.error = jit.getLastError();
236 return r;
237 }
238 r.asmText = jit.getCapturedAsm();
239 r.ok = true;
240 return r;
241}
