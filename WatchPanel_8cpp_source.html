219void WatchPanel::onAddExpression() {
220 QString text = exprInput_->text().trimmed();
221 if (text.isEmpty())
222 return;
223 // 防重复：若已存在相同表达式则不重复添加
224 for (int r = 0; r < watchTable_->rowCount(); ++r) {
225 if (auto* item = watchTable_->item(r, 0)) {
226 if (item->text() == text) {
227 exprInput_->clear();
228 return;
229 }
230 }
231 }
232 editingGuard_ = true;
233 int row = watchTable_->rowCount();
234 watchTable_->insertRow(row);
235 auto* exprItem = new QTableWidgetItem(text);
236 exprItem->setFlags(exprItem->flags() | Qt::ItemIsEditable);
237 auto* typeItem = new QTableWidgetItem(QString::fromUtf8("—"));
238 typeItem->setFlags(typeItem->flags() & ~Qt::ItemIsEditable);
239 auto* valueItem = new QTableWidgetItem(QString::fromUtf8("—"));
240 valueItem->setFlags(valueItem->flags() & ~Qt::ItemIsEditable);
241 auto* statusItem = new QTableWidgetItem(QString::fromUtf8("待求值"));
242 statusItem->setFlags(statusItem->flags() & ~Qt::ItemIsEditable);
243 watchTable_->setItem(row, 0, exprItem);
244 watchTable_->setItem(row, 1, typeItem);
245 watchTable_->setItem(row, 2, valueItem);
246 watchTable_->setItem(row, 3, statusItem);
247 editingGuard_ = false;
248 exprInput_->clear();
249 saveExpressions();
250 // 立即求值一次
251 refreshAll();
252}

275void WatchPanel::onCellChanged(int row, int col) {
276 // 仅表达式列（col==0）可编辑，且忽略程序化 setItem 触发
277 if (col != 0 || editingGuard_)
278 return;
279 if (auto* item = watchTable_->item(row, 0)) {
280 // 去除首尾空白
281 QString trimmed = item->text().trimmed();
282 if (trimmed != item->text()) {
283 editingGuard_ = true;
284 item->setText(trimmed);
285 editingGuard_ = false;
286 }
287 // 空表达式：状态标记
288 if (trimmed.isEmpty()) {
289 if (auto* st = watchTable_->item(row, 3))
290 st->setText(QString::fromUtf8("空表达式"));
291 saveExpressions();
292 return;
293 }
294 }
295 saveExpressions();
296 // 用户修改了表达式后立即求值
297 refreshAll();
298}

332void WatchPanel::refreshAll() {
333 if (!controller_) {
334 liveStatusLabel_->setText(QString::fromUtf8("未绑定控制器"));
335 return;
336 }
337
338 // 状态描述
339 bool vmMode = controller_->isVmRunning() || controller_->isVmInitialized();
340 bool interpPaused = controller_->isDebugPaused();
341 // R121: 追加选中帧提示（用户在 CallStackPanel 选了某帧时显示）
342 int selFrame = controller_->getSelectedFrame();
343 QString frameHint;
344 if ((vmMode || interpPaused) && selFrame >= 0) {
345 frameHint = QString::fromUtf8(" | 已选帧: %1").arg(selFrame);
346 }
347 if (vmMode) {
348 liveStatusLabel_->setText(QString::fromUtf8("VM 模式暂停中") + frameHint);
349 } else if (interpPaused) {
350 liveStatusLabel_->setText(QString::fromUtf8("Interpreter 调试暂停中") + frameHint);
351 } else {
352 liveStatusLabel_->setText(QString::fromUtf8("未在调试暂停状态（表达式将在暂停时求值）"));
353 }
354
355 // 逐行求值
356 editingGuard_ = true;
357 for (int r = 0; r < watchTable_->rowCount(); ++r) {
358 auto* exprItem = watchTable_->item(r, 0);
359 auto* typeItem = watchTable_->item(r, 1);
360 auto* valueItem = watchTable_->item(r, 2);
361 auto* statusItem = watchTable_->item(r, 3);
362 if (!exprItem || !typeItem || !valueItem || !statusItem)
363 continue;
364
365 QString expr = exprItem->text().trimmed();
366 if (expr.isEmpty()) {
367 typeItem->setText(QString::fromUtf8("—"));
368 valueItem->setText(QString::fromUtf8("—"));
369 statusItem->setText(QString::fromUtf8("空表达式"));
370 continue;
371 }
372
373 auto result = controller_->evaluateWatchExpression(expr.toStdString());
374 if (result.ok) {
375 typeItem->setText(QString::fromUtf8(result.typeName.c_str()));
376 valueItem->setText(QString::fromUtf8(result.valueRepr.c_str()));
377 statusItem->setText(QString::fromUtf8("OK"));
378 // 类型着色：标量用绿色，容器用蓝色，实例用紫色，闭包/函数用橙色
379 QString color = TeachingTheme::textPrimary().name();
380 if (result.typeName == "int" || result.typeName == "float" || result.typeName == "bool" ||
381 result.typeName == "null") {
382 color = "#859900"; // Solarized green
383 } else if (result.typeName == "string") {
384 color = "#B58900"; // Solarized yellow
385 } else if (result.typeName == "array" || result.typeName == "dict") {
386 color = "#268BD2"; // Solarized blue
387 } else if (result.typeName == "instance") {
388 color = "#6C71C4"; // Solarized violet
389 } else if (result.typeName == "closure" || result.typeName == "function" ||
390 result.typeName == "bound-method") {
391 color = "#CB4B16"; // Solarized orange
392 }
393 typeItem->setForeground(QColor(color));
394 } else {
395 typeItem->setText(QString::fromUtf8("—"));
396 valueItem->setText(QString::fromUtf8("—"));
397 statusItem->setText(QString::fromUtf8(result.error.c_str()));
398 statusItem->setForeground(QColor("#DC322F")); // Solarized red
399 }
400 }
401 editingGuard_ = false;
402}
