273 QString html = QStringLiteral(
274 "<h2>MiniLang 模块系统原理</h2>"
275 "<p>模块系统通过 <b>import / export</b> 支持跨文件代码组织。一个模块从源码到可用"
276 "需经历：<b>Lexer → Parser → 预扫描 → 模块缓存 → 循环导入延迟加载 → 执行</b>。"
277 "模块系统是 MiniLang 的高频 Bug 模式（路径安全、循环导入、预扫描遗漏 FunDecl/ExportStmt、"
278 "错误传播），理解每一阶段的不变量是排查此类 Bug 的关键。</p>"
279
280 "<h3>一、加载管线</h3>"
281 "<ol>"
282 "<li><b>词法/语法分析</b>：import / export 生成 ImportStmt / ExportStmt AST 节点。</li>"
283 "<li><b>路径解析</b>：import 路径基于当前模块目录，自动补全 .ml；路径安全禁止逃逸项目根。</li>"
284 "<li><b>循环导入延迟加载</b>：目标模块入「加载栈」前检查是否已在栈中，是则返回部分 env（P2-14）。</li>"
285 "<li><b>预扫描</b>：遍历顶层 AST，收集 FunDecl / ExportStmt 并预注册到模块作用域，"
286 "支持前向引用。VarDecl/ClassDecl/EnumDecl 不预扫描。</li>"
287 "<li><b>模块缓存</b>：以规范路径为 key 缓存已加载模块，二次 import 命中缓存跳过执行。</li>"
288 "<li><b>执行</b>：后序执行模块顶层语句；import 端绑定 export 的符号到当前作用域。</li>"
289 "</ol>"

291 "<h3>二、import / export 语义</h3>"
292 "<ul>"
293 "<li><b>import \"foo\"</b>：加载 foo.ml，执行后将其 export 的符号绑定到当前作用域。</li>"
294 "<li><b>export fun name() {...}</b>：声明并导出函数，预扫描阶段即被收集。</li>"
295 "<li><b>export var x = ...</b>：导出变量，其值在模块执行后确定。</li>"
296 "<li>未 export 的顶层声明对 import 端不可见（模块隔离）。</li>"
297 "</ul>"

299 "<h3>三、模块缓存与循环导入延迟加载</h3>"

301 "循环导入通过「加载栈」检测：进入 import 递归前压栈，完成出栈。"
302 "若目标已在栈中即 a→b→a，P2-14 后不再报错——返回已创建的部分 env，"
303 "模块环境立即入缓存，内容按执行顺序填充。访问未定义导出名时由 Environment::get 抛\"未定义变量\"错误。"
304 "语义参考 ES Modules + Python。注意缓存命中（已加载完成）"
305 "与「正在加载中」（栈中）是两种不同状态，前者返回完整 env 后者返回部分 env。</p>"
306
307 "<h3>四、预扫描的必要性</h3>"
308 "<p>预扫描收集 FunDecl/ExportStmt 使<b>函数定义顺序无关</b>——"
309 "先调用后定义也能工作（前向引用）。若预扫描遗漏某节点类型（历史 Bug 模式），"
310 "则调用方在运行时找不到该符号，报「未定义」错误。VarDecl "
311 "不预扫描因为初始化表达式有副作用，必须按顺序执行。</p>"
312
313 "<h3>五、错误传播链</h3>"

460 s.step = ++stepNo;
461 s.module = mod;
462 s.operation = op;

583void ModuleSystemVisualizerPanel::renderScenarioResult(const ModuleSimResult& result) {
584 // 1. 填充时间轴 QTableWidget
585 timelineTable_->setRowCount(static_cast<int>(result.steps.size()));
586 for (int i = 0; i < static_cast<int>(result.steps.size()); ++i) {
587 const auto& s = result.steps[i];
588 auto* stepItem = new QTableWidgetItem(QString::number(s.step));
589 auto* modItem = new QTableWidgetItem(QString::fromUtf8(s.module.c_str()));
590 auto* opItem = new QTableWidgetItem(QString::fromUtf8(s.operation.c_str()));
591 auto* statusItem = new QTableWidgetItem(QString::fromUtf8(s.status.c_str()));
592 stepItem->setTextAlignment(Qt::AlignCenter);
593 statusItem->setTextAlignment(Qt::AlignCenter);
594 statusItem->setForeground(QColor(statusColor(s.status)));
595 timelineTable_->setItem(i, 0, stepItem);
596 timelineTable_->setItem(i, 1, modItem);
597 timelineTable_->setItem(i, 2, opItem);
598 timelineTable_->setItem(i, 3, statusItem);
599 timelineTable_->item(i, 2)->setToolTip(QString::fromUtf8(s.operation.c_str()));
600 }
601
602 // 2. 渲染模拟过程详情 HTML（依赖图 + 加载日志 + 代码）
603 QString monoFamily = GuiTextUtils::monoFontFamilyQss();
604 QString html;
605 html += QStringLiteral("<html><body style='font-family:%1; font-size:14px;'>").arg(monoFamily);
606 html += QStringLiteral("<h2>%1</h2>").arg(htmlEscape(result.scenarioTitle));
607
608 if (!result.summary.empty()) {
609 QString color = result.success ? QStringLiteral("#0a7a28") : QStringLiteral("#c0392b");
610 html += QStringLiteral("<p style='color:%1;font-weight:bold;'>%2</p>").arg(color, htmlEscape(result.summary));
611 }
612
613 // 依赖图
614 if (!result.dependencyGraphHtml.empty()) {
615 html += QStringLiteral("<h3>模块依赖图</h3>");
616 html += QString::fromUtf8(result.dependencyGraphHtml.c_str());
617 }
618
619 // 加载日志（步骤表）
620 if (!result.steps.empty()) {
621 html += QStringLiteral("<h3>加载日志</h3>");
622 html += QStringLiteral("<table border='1' cellpadding='5' cellspacing='0' "
623 "style='font-family:%1;font-size:12px;border-collapse:collapse;'>")
624 .arg(monoFamily);
625 html += QStringLiteral("<tr style='background:#f0f0f0;'><th>步骤</th><th>模块</th>"
626 "<th>操作</th><th>状态</th></tr>");
627 for (const auto& s : result.steps) {
628 html += QStringLiteral("<tr><td style='text-align:center;'>%1</td><td>%2</td><td>%3</td>"
629 "<td style='text-align:center;color:%4;'>%5</td></tr>")
630 .arg(QString::number(s.step), htmlEscape(s.module), htmlEscape(s.operation),
631 statusColor(s.status), htmlEscape(s.status));
632 }
633 html += QStringLiteral("</table>");
634 }
635
636 // 模拟代码片段
637 if (!result.scenarioCode.empty()) {
638 html += QStringLiteral("<h3>场景代码</h3>");
639 html += QStringLiteral("<pre style='background:#f7f7f7;padding:8px;border:1px solid #ddd;"
640 "font-family:%1;font-size:12px;white-space:pre-wrap;'>")
641 .arg(monoFamily);
642 html += htmlEscape(result.scenarioCode);
643 html += QStringLiteral("</pre>");
644 }
645
646 html += QStringLiteral("</body></html>");
647 simBrowser_->setHtml(html);
648}
