55 /// C9 fix: 前端管线结果
56 /// AUDIT-R5 R7 fix (BUG-5): diagnostics 原为指向 lexer_/parser_ 内部 DiagnosticBag
57 /// 的裸指针，源码级缓存命中返回旧 PipelineResult 时，后续新一次前端管线运行
58 /// 会改写 lexer_/parser_ 内部状态，使外部持有的诊断内容被静默改写（版本错位）。
59 /// 改为自持快照 diagnosticsStorage（值拷贝），diagnostics 指向该内部副本；
60 /// 拷贝/赋值时重指向自身副本，保持调用方 `result.diagnostics->...` API 不变，
61 /// 且快照免疫后续 lexer_/parser_ 改写与 PipelineResult 拷贝。
62 struct PipelineResult {
63 PipelineStatus status = PipelineStatus::OK;
64 std::string errorMessage; ///< 异常消息（status != OK 时有效）
65 DiagnosticBag diagnosticsStorage; ///< 自持诊断快照（AUDIT-R5 R7 fix）
66 const DiagnosticBag* diagnostics = nullptr; ///< 指向 diagnosticsStorage 或 nullptr
67
68 PipelineResult() = default;
69 PipelineResult(const PipelineResult& o)
70 : status(o.status), errorMessage(o.errorMessage), diagnosticsStorage(o.diagnosticsStorage),
71 diagnostics(o.diagnostics ? &diagnosticsStorage : nullptr) {}
72 PipelineResult& operator=(const PipelineResult& o) {
73 if (this != &o) {
74 status = o.status;
75 errorMessage = o.errorMessage;
76 diagnosticsStorage = o.diagnosticsStorage;
77 diagnostics = o.diagnostics ? &diagnosticsStorage : nullptr;
78 }
79 return *this;
80 }

82 void setDiagnostics(const DiagnosticBag& bag) {
83 diagnosticsStorage = bag;
84 diagnostics = &diagnosticsStorage;
85 }
86 };

133 /// REPL 专用：直接设置词法/语法分析结果（供 magic 命令无参查询）
134 /// REPL 传入源码，内部静默执行 Lex+Parse（不发 diagnosticsReady 信号），
135 /// 更新 lastTokens_/astRoot_。注意：不清空 lastCompileResult_，
136 /// 因为用户可能仍想查看最近一次 Run(F5) 的字节码。
137 void setReplPipelineState(const std::string& source) {
138 invalidatePipelineCache();
139 lastTokens_ = lexer_.scan(source);
140 bool hasError = false;
141 for (const auto& tok : lastTokens_) {
142 if (tok.type == TokenType::TK_ERROR) {
143 hasError = true;
144 break;
145 }
146 }
147 if (!hasError) {
148 try {
149 astRoot_ = parser_.parse(lastTokens_);
150 } catch (...) {
151 astRoot_.reset();
152 }
153 } else {
154 astRoot_.reset();
155 }
156 }
157
158 /// 清空管线状态（%reset 时调用）
159 void resetPipelineState() {
160 invalidatePipelineCache();
161 lastTokens_.clear();
162 astRoot_.reset();
163 lastCompileResult_ = CompileResult{};
164 lastSource_.clear(); // P1-7: 清空源码记录
165 }
