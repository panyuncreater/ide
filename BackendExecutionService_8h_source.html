1/**
2 * @file common/BackendExecutionService.h
3 * @brief 后端执行服务中间层（ARCH-10）。
4 *
5 * 为 GUI 教学面板（BackendParallelPanel / PerformanceRacePanel /
6 * ProfileDashboardPanel 等）提供对编译器+执行后端的高层封装，
7 * 消除面板对 compiler/core/Compiler.h / compiler/backend-stack/VM.h / compiler/backend-reg/RegisterVM.h /
8 * compiler/ir/IR.h / interpreter/Interpreter.h / lexer/Lexer.h / parser/Parser.h
9 * 等内部头文件的直接依赖。
10 *
11 * 设计原则：
12 * - 面板通过 BackendType 枚举选择后端，服务负责完整的
13 * Lexer → Parser → Compiler → Backend 执行流程
14 * - 返回 BackendExecResult 结构体，包含输出/错误/耗时/指令数等只读数据
15 * - 错误处理统一：词法/语法/编译/运行时错误均通过 errorMsg + success 字段返回
16 * - 不持有状态：每次调用独立执行，面板无需管理对象生命周期
17 *
18 * 与 IBackend 的关系：
19 * - IBackend 抽象执行后端的调试/诊断接口（断点/单步/栈检查）
20 * - BackendExecutionService 抽象"源码 → 执行结果"的完整流程
21 * - 两者互补：面板用 BackendExecutionService 跑代码，用 IBackend 调试
22 *
23 * @since ARCH-10
24 */

59/// 单次后端执行的完整结果（只读快照）。
60/// 所有字段为值类型或 std::string，调用方无需依赖任何编译器/VM 内部类型。
61struct BackendExecResult {
62 bool success = false; // 是否成功（无词法/语法/编译/运行时错误）
63 BackendType backend = BackendType::Interpreter; // 实际使用的后端
64 std::string output; // 标准输出（print 累积）
65 std::string errorMsg; // 错误信息（success=false 时有效，含错误前缀）
66 std::string errorPrefix; // 错误前缀（"词法错误" / "语法错误" / "编译错误" / "运行时错误"）
67 int64_t elapsedMs = 0; // 执行耗时（毫秒，= elapsedUs/1000，保留兼容旧调用方）
68 // AUDIT-R2 P2-2 fix: 新增微秒精度耗时——教学题库代码多在 1ms 内完成，
69 // 毫秒粒度下三后端耗时恒显 0，面板对比失去意义。
70 // 注：VM/RegisterVM 路径的耗时仅覆盖执行阶段（不含编译），与 Interpreter
71 // 路径口径一致（AUDIT-R2 P2-3 fix，此前 VM 路径含编译时间，对比不对称）。
72 int64_t elapsedUs = 0; // 执行耗时（微秒）
73 int64_t instrCount = -1; // 指令数（字节码字节数；-1 表示 N/A，如 Interpreter 无字节码）
74 std::vector<std::string> diagnostics; // 编译诊断信息（warning/error 文本列表）
75

98// ============================================================
99// BackendExecDetail — 扩展执行结果
100// ============================================================
101// 在 BackendExecResult 基础上增加：
102// - generatorChunks：生成器 chunk 信息（仅 StackVM/RegisterVM 路径有效）
103// - opcodeCounts：256 槽 opcode 频次计数（仅 StackVM/RegisterVM 路径有效；
104// Interpreter 路径无 opcode 概念，全为 0）
105// - peakTrackedCount：执行期间 GcManager 跟踪对象相对本次执行前基线的
106// 净增峰值（AUDIT-R2 P1-5 fix：不再 reset 全局单例，避免破坏并发执行的
107// worker 线程 GC 跟踪状态）
108struct BackendExecDetail : BackendExecResult {
109 std::vector<GeneratorChunkInfo> generatorChunks;
110 std::array<uint64_t, 256> opcodeCounts{};
111 size_t peakTrackedCount = 0;
112};

114// ============================================================
115// BackendExecutionService — 后端执行服务
116// ============================================================
117// 所有方法均为静态，无实例状态。Lexer/Parser/Compiler/VM/Interpreter 的
118// 对象创建与执行封装在 .cpp 中，调用方仅依赖此头文件的纯数据结构。
119//
120// 线程安全：
121// - 单次 execute 调用在调用线程内同步执行（不创建新线程）
122// - 不同线程并发调用 execute 是安全的（每次调用创建独立的局部对象）
123// - 但执行期间会访问 GcManager 全局单例（与 worker 线程 collectCycle 互斥）
124class BackendExecutionService {
125public:
126 /// 完整执行入口：Lexer → Parser → Compiler → 指定后端执行。
127 /// @param src MiniLang 源码
128 /// @param backend 后端类型（Interpreter / StackVM_IR / RegisterVM_IR）
129 /// @param detail 可选出参（PERF-DETAIL，2026-10-04）：非空时在本次执行内
130 /// 单遍收集 generatorChunks / opcodeCounts / peakTrackedCount——
131 /// 原设计 executeWithDetail 需全流程执行两遍（第二遍重新
132 /// Lexer+Parser+compile+execute）。注意 detail 模式的耗时含
133 /// step 采样开销（每条指令一次回调），与无 detail 的计时口径
134 /// 不完全可比。
135 /// @return 执行结果（含输出/错误/耗时/指令数）
136 static BackendExecResult execute(const std::string& src, BackendType backend, BackendExecDetail* detail = nullptr);
137
138 /// 扩展执行入口：返回详细执行信息（含生成器 chunk / opcode 计数 / GC 峰值）。
139 /// 用于 CoroutineVisualizerPanel（生成器 chunk 信息）和
140 /// ProfileDashboardPanel（opcode profile + GC peak tracked）。
141 /// @param src MiniLang 源码
142 /// @param backend 后端类型
143 /// @return 执行详情（基类字段同 BackendExecResult，扩展字段见 BackendExecDetail）
144 static BackendExecDetail executeWithDetail(const std::string& src, BackendType backend);
145
146 /// ARCH-10 IVmBackend 工厂方法：创建指定类型的后端实例，返回多态指针。
147 /// 用于教学面板需要通过 IVmBackend* 多态操作 VM 的场景（如 BytecodeTracePanel
148 /// 需要步进执行 + 读取栈/寄存器/调用栈，通过统一接口避免 if(backend==StackVM)
149 /// 静态分派）。面板持有 unique_ptr 自动管理生命周期，无需依赖具体 VM 类。
150 /// @param backend 后端类型（仅 StackVM_IR / RegisterVM_IR 有效；Interpreter
151 /// 不实现 IVmBackend，传入会返回 nullptr）
152 /// @return 后端实例的 unique_ptr，或 nullptr（backend == Interpreter）。
153 /// 本工厂不执行任何编译，不存在"编译失败"返回路径。
154 /// 注意：调用方负责在创建后调用 IVmBackend 的子类特有方法（如 VM::execute
155 /// 或 RegisterVM::execute）开始执行；本工厂仅创建对象，不执行代码。
156 static std::unique_ptr<IVmBackend> createBackend(BackendType backend);
157};
