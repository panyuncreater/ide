174 painter.setPen(QPen(arrowColor, 1.4, Qt::SolidLine, Qt::RoundCap, Qt::RoundJoin));

192 bottom = top + static_cast<int>(codeEditor->blockBoundingRect(block).height());

202 // F8: 检查是否点击了折叠区域（右侧 14px）

205 // 覆盖 [width()-13, width()-5)。命中区统一为 [width()-14, width()-4)。
206 if (clickX >= width() - 14 && clickX < width() - 4) {
207 // 映射 Y 坐标到块号
208 QTextCursor cursor = codeEditor->cursorForPosition(QPoint(0, static_cast<int>(event->position().y())));
209 int blockNumber = cursor.blockNumber();
210 QTextBlock block = codeEditor->document()->findBlockByNumber(blockNumber);
211 if (block.isValid() && codeEditor->isFoldable(block)) {
212 codeEditor->toggleFold(blockNumber);
213 return;
214 }
215 }
216
217 // 将 Y 坐标映射到行号
218 QTextCursor cursor = codeEditor->cursorForPosition(QPoint(0, static_cast<int>(event->position().y())));
219 int lineNumber = cursor.blockNumber() + 1;
220
221 // AUDIT-BUG-F13 fix: 检查点击是否在文档内容区域内。
222 // cursorForPosition 对最后一行下方的空白区域返回文档末尾光标（最后一个块号），
223 // 导致空白区域点击在最后一行设置断点。检查 Y 坐标是否超出最后一个块的下边界。
224 QTextBlock lastBlock = codeEditor->document()->lastBlock();
225 qreal lastBlockBottom = static_cast<qreal>(
226 codeEditor->blockBoundingGeometry(lastBlock).translated(codeEditor->contentOffset()).bottom());
227 if (static_cast<qreal>(event->position().y()) >= lastBlockBottom) {
228 return; // 点击在最后一行下方的空白区域，不设置断点
229 }
230
231 // DB-2 fix: 不允许在空行/纯注释行设置断点
232 QTextBlock block = codeEditor->document()->findBlockByNumber(lineNumber - 1);
233 if (block.isValid()) {
234 QString text = block.text().trimmed();
235 // BUG-CE-7 fix: 不仅识别行注释 //，也识别块注释行（userState >= 100
236 // 或向后兼容编码 2 表示块注释跨行上下文）
237 int s = block.userState();
238 bool isCommentLine = text.startsWith("//") || s == 2 || s >= 100;
239 if (text.isEmpty() || isCommentLine) {
240 return; // 跳过不可执行行
241 }
242 }
243
244 if (codeEditor->breakpoints_.contains(lineNumber)) {
245 codeEditor->breakpoints_.remove(lineNumber);
246 codeEditor->breakpointConditions_.remove(lineNumber);
247 codeEditor->breakpointKinds_.remove(lineNumber);
248 codeEditor->logpointMessages_.remove(lineNumber);
249 } else {
250 codeEditor->breakpoints_.insert(lineNumber);
251 }
252 update();
253
254 // 更新断点显示
255 codeEditor->viewport()->update();
256
257 // AUDIT-P2-CORRECT fix: 发射 breakpointsChanged 信号，通知上层（Ide）
258 // 在调试暂停期间同步断点到 DebugController，避免新断点不生效/已删除断点仍触发。
259 emit codeEditor->breakpointsChanged();
260}

882void CodeEditor::onContentsChange(int position, int charsRemoved, int charsAdded) {
883 Q_UNUSED(charsRemoved);
884 // AUDIT-P1-CORRECT fix: charsAdded 用于多行删除场景推导实际删除行数，
885 // 不再 Q_UNUSED 丢弃。
886
887 // BUG-CE-2/CE-3 fix: 文档内容变化后，断点（1-based 行号）、断点条件、
888 // 折叠块（0-based blockNumber）以行号为 key，文档修改导致行号变化时需同步偏移。
889 //
890 // 实现说明：contentsChange 在变更已应用后发射，被删除文本不可访问，无法直接
891 // 统计其换行符数量。改为通过文档块数变化计算 delta（lastBlockCount_ 在上次
892 // 变更后/构造时更新），保证纯插入/纯删除/替换三类场景均正确。
893 int newBlockCount = document()->blockCount();
894 int delta = newBlockCount - lastBlockCount_;
895 lastBlockCount_ = newBlockCount;
896 if (delta == 0)
897 return;
898
899 // 计算 position 所在行（1-based），仅调整 >= startLine 的行号
900 QTextBlock block = document()->findBlock(position);
901 if (!block.isValid())
902 return;
903 int startLine = block.blockNumber() + 1; // 1-based，变更起始行
904
905 // BUG-GUI-AUDIT-2 fix: 区分"行末插入换行符"与"行首插入换行符"两种场景。
906 // - 行首插入换行符（光标在行首按 Enter）：新行创建在当前行之前，原行内容下移，
907 // 断点应 +1（当前实现已正确）。
908 // - 行末插入换行符（光标在行末按 Enter）：新行创建在当前行之后，原行内容不变，
909 // 断点应保持原行号（不应 +1）。原实现错误地对原行断点也 +1。
910 // - 删除整行（含换行符）：原行内容消失，下移内容上移。VS Code 语义是断点保持
911 // 原行号（指向新移入的内容），原实现错误地 -1。
912 // AUDIT-P1.3 fix: 用 startsAtBoundary = (position == block.position()) 统一检测
913 // 行首边界，替代原 block.length() > 1 检测。原检测在空行（length==1）场景
914 // 无法区分"行首"与"行末"（空行 position 恒等于 block.position()），且对
915 // 文档末尾无换行符的块（length==text.length()）计算偏差。startsAtBoundary
916 // 直接语义化"删除/插入是否从行首开始"，更精确。
917 bool startsAtBoundary = (position == block.position());
918 if (delta > 0) {
919 // 插入场景：若插入发生在块末尾换行符位置（非行首），原行内容不变，断点不应偏移。
920 // 将 startLine +1 使原行 line < startLine，不参与偏移。
921 // 行首插入（startsAtBoundary）：原行内容下移，断点应 +1（不 ++startLine）。
922 if (!startsAtBoundary && position >= block.position() + block.length() - 1) {
923 ++startLine;
924 }
925 } else if (delta < 0) {
926 // AUDIT-P1-CORRECT fix: 多行删除需正确处理被删除行区间内的断点。
927 // 原实现用 delta（NET 行数变化）统一偏移所有 >= startLine 的断点，
928 // 仅通过 startLine +1 保护一行。对于多行删除（delta = -N），中间被删除行
929 // 的断点被错误平移而非删除，导致断点指向错误行号。
930 //
931 // 利用 charsAdded 推导实际删除的行数：linesRemoved = linesAdded - delta
932 // （delta 为负，linesAdded 通常为 0 表示纯删除；替换场景 linesAdded > 0）
933 int linesAdded = 0;
934 if (charsAdded > 0) {
935 for (int i = position; i < position + charsAdded && i < document()->characterCount(); ++i) {
936 if (document()->characterAt(i) == '\n')
937 ++linesAdded;
938 }
939 }
940 int linesRemoved = linesAdded - delta; // 实际删除的行数（delta 为负）
941
942 if (linesRemoved > 0) {
943 // 多行删除：需要删除被删除行区间内的断点
944 // AUDIT-P1.3 fix: lastDeletedLine 需按 startsAtBoundary 区分计算。
945 // - 行首删除（startsAtBoundary）：原行内容被删除，新内容上移。被删除的
946 // 行区间是 [startLine+1, startLine+linesRemoved-1]（原行断点保持，
947 // 指向新移入的内容）。lastDeletedLine = startLine + linesRemoved - 1。
948 // - 非行首删除（行末/行中间删除）：原行保留，下一行合并进来。被删除的
949 // 行区间是 [startLine+1, startLine+linesRemoved]（下一行到下N行被删除）。
950 // lastDeletedLine = startLine + linesRemoved。
951 // 原实现统一用 startLine + linesRemoved - 1，非行首删除时少算一行，
952 // 导致最后一个被删除行的断点被 shift（+delta）而非删除，错误地创建到原行。
953 int firstDeletedLine = startLine; // keep 边界：line <= firstDeletedLine 保持
954 int lastDeletedLine = startsAtBoundary ? (startLine + linesRemoved - 1) : (startLine + linesRemoved);
955 // 断点 line <= firstDeletedLine：保持（VS Code 语义，指向合并后内容）
956 // 断点 firstDeletedLine < line <= lastDeletedLine：删除（不插入）
957 // 断点 line > lastDeletedLine：平移 delta

1391QString CodeEditor::hoverExpressionAt(const QPoint& viewportPos) const {

1399 if (word.isEmpty() || !(word[0].isLetter() || word[0] == QLatin1Char('_')))
1400 return {};

1403 const QString blockText = probe.block().text();
1404 int start = probe.selectionStart() - probe.block().position();
1405 QString prefix;
1406 int i = start - 1;
1407 while (i >= 1 && blockText[i] == QLatin1Char('.')) {
1408 int j = i - 1;
1409 while (j >= 0 && (blockText[j].isLetterOrNumber() || blockText[j] == QLatin1Char('_')))
1410 --j;
1411 // 链段必须是合法标识符（首字符非数字）
1412 if (j + 1 >= i || blockText[j + 1].isDigit())
1413 break;
1414 prefix = blockText.mid(j + 1, i - j) + prefix; // 含末尾 '.'
1415 i = j;
1416 }
1417 return prefix + word;
1418}

1740bool CodeEditor::handleLineMoveKeys(QKeyEvent* event) {
1741 // R53-UX9 fix: Alt+Up / Alt+Down 移动当前行（或选区）上/下。
1742 // 对齐 VSCode 常用编辑器快捷键，提升代码重组效率。
1743 // 不与现有快捷键冲突（Alt+Up/Down 此前未绑定）。
1744 if (event->modifiers() != Qt::AltModifier || (event->key() != Qt::Key_Up && event->key() != Qt::Key_Down)) {
1745 return false;
1746 }
1747 bool moveUp = (event->key() == Qt::Key_Up);
1748 QTextCursor tc = textCursor();
1749 QTextDocument* doc = document();
1750
1751 // 确定要移动的行范围 [startLine, endLine]
1752 int selStart = tc.selectionStart();
1753 int selEnd = tc.selectionEnd();
1754 int startLine = doc->findBlock(selStart).blockNumber();
1755 int endLine = doc->findBlock(selEnd).blockNumber();
1756 // 若选区末尾恰在行首（非整行选中末尾），endLine 应回退一行
1757 if (selStart != selEnd && doc->findBlock(selEnd).position() == selEnd && endLine > startLine) {
1758 --endLine;
1759 }
1760 int totalBlocks = doc->blockCount();
1761
1762 if (moveUp) {
1763 if (startLine <= 0)
1764 return true; // 已是第一行，无法上移
1765 QTextBlock prevBlock = doc->findBlockByNumber(startLine - 1);
1766 QString prevText = prevBlock.text();
1767 tc.beginEditBlock();

1769 QTextCursor del = tc;
1770 del.setPosition(prevBlock.position());
1771 del.movePosition(QTextCursor::NextBlock, QTextCursor::KeepAnchor);
1772 del.removeSelectedText();
1773 // 在移动块末尾插入被删的行（删除 1 行后，原 endLine 现为 endLine-1）
1774 QTextBlock endBlock = doc->findBlockByNumber(endLine - 1);
1775 tc.setPosition(endBlock.position() + endBlock.text().length());
1776 tc.insertText('\n' + prevText);
1777 tc.endEditBlock();
1778 // 重新选中移动后的块
1779 QTextBlock newStart = doc->findBlockByNumber(startLine - 1);
1780 QTextBlock newEnd = doc->findBlockByNumber(endLine - 1);
1781 QTextCursor sel = tc;
1782 sel.setPosition(newStart.position());
1783 sel.setPosition(newEnd.position() + newEnd.text().length(), QTextCursor::KeepAnchor);
1784 setTextCursor(sel);
1785 } else {
1786 if (endLine >= totalBlocks - 1)
1787 return true; // 已是最后一行，无法下移
1788 QTextBlock nextBlock = doc->findBlockByNumber(endLine + 1);
1789 QString nextText = nextBlock.text();
1790 tc.beginEditBlock();
1791 // 删除下一行（仅选中该块到下一块开头，包含换行符；末行特殊处理）
1792 QTextCursor del = tc;
1793 if (nextBlock.blockNumber() < totalBlocks - 1) {
1794 del.setPosition(nextBlock.position());
1795 del.movePosition(QTextCursor::NextBlock, QTextCursor::KeepAnchor);
1796 } else {
1797 // 最后一行：选中前置换行符 + 该行文本
1798 del.setPosition(nextBlock.position() - 1);
1799 del.setPosition(nextBlock.position() + nextBlock.text().length(), QTextCursor::KeepAnchor);
1800 }
1801 del.removeSelectedText();

1803 QTextBlock startBlock = doc->findBlockByNumber(startLine);
1804 tc.setPosition(startBlock.position());
1805 tc.insertText(nextText + '\n');
1806 tc.endEditBlock();
1807 // 重新选中移动后的块
1808 QTextBlock newStart = doc->findBlockByNumber(startLine + 1);
1809 QTextBlock newEnd = doc->findBlockByNumber(endLine + 1);
1810 QTextCursor sel = tc;
1811 sel.setPosition(newStart.position());
1812 sel.setPosition(newEnd.position() + newEnd.text().length(), QTextCursor::KeepAnchor);
1813 setTextCursor(sel);
1814 }
1815 return true;
1816}

1941bool CodeEditor::tryExpandSnippet() {
1942 // 获取光标前全部文本
1943 QTextCursor tc = textCursor();
1944 const int cursorPos = tc.position();
1945 QTextCursor scan = tc;
1946 scan.movePosition(QTextCursor::Start, QTextCursor::KeepAnchor);
1947 const QString textBefore = scan.selectedText();
1948
1949 const CodeSnippet* snip = CodeSnippetEngine::matchTrigger(textBefore);
1950 if (!snip)
1951 return false;
1952
1953 // 计算触发词在文档中的范围并删除
1954 const QString trigger = QString::fromStdString(snip->trigger);
1955 const int triggerStart = cursorPos - trigger.length();
1956 if (triggerStart < 0)
1957 return false; // 防御性：触发词长度超过文档长度
1958
1959 QTextCursor del = textCursor();
1960 del.setPosition(triggerStart);
1961 del.setPosition(cursorPos, QTextCursor::KeepAnchor);
1962 del.removeSelectedText();
1963
1964 // 展开模板并插入
1965 const auto expansion = CodeSnippetEngine::expand(*snip);
1966 const int insertPos = triggerStart; // 插入起点（触发词已被删除）
1967
1968 QTextCursor ins = textCursor();
1969 ins.setPosition(insertPos);
1970 ins.insertText(expansion.text);
1971
1972 // 构造占位符的绝对位置列表（在文档中的字符偏移）
1973 currentPlaceholders_.clear();
1974 currentPlaceholders_.reserve(expansion.placeholderRanges.size());
1975 for (const auto& [relStart, relEnd] : expansion.placeholderRanges) {
1976 currentPlaceholders_.push_back({insertPos + relStart, insertPos + relEnd});
1977 }
1978
1979 if (currentPlaceholders_.empty()) {
1980 // 无占位符：光标置于展开文本末尾，不进入导航模式
1981 currentPlaceholderIdx_ = -1;
1982 QTextCursor end = textCursor();
1983 end.setPosition(insertPos + expansion.text.length());
1984 isSnippetNavigating_ = true;
1985 setTextCursor(end);
1986 isSnippetNavigating_ = false;
1987 } else {
1988 // 有占位符：选中第一个占位符，进入导航模式
1989 currentPlaceholderIdx_ = 0;
1990 selectCurrentPlaceholder();
1991 }
1992 return true;
