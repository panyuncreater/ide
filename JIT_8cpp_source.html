1/**
2 * @file compiler/jit/JIT.cpp
3 * @brief JIT 后端实现：基于 asmjit 的字节码→本地代码编译器（R138-R141 演进）。
4 *
5 * 实现策略：
6 * 1. 两遍扫描 BytecodeChunk.code 字节流：
7 * - 第一遍：收集所有跳转目标位置（OP_JUMP/OP_JUMP_IF_FALSE/OP_LOOP 的操作数），
8 * 为每个目标位置创建 asmjit::Label
9 * - 第二遍：为每条 OpCode 生成对应的 x86-64 机器码，在每个字节码位置绑定对应 Label
10 * 2. 编译期提取 OP_INT 常量并内联为立即数（movabs rax, imm64）
11 * 3. 操作数栈通过 r15 寄存器维护（向低地址增长），与 StackVM 的 stack_ 等价
12 * 4. OP_PRINT 通过调用 C++ 运行时辅助函数 jitPrintInt 实现
13 * 5. R139: 全局变量 slot 通过 JitContext.globalSlots 指针访问（offset 24）
14 * 6. R139: 比较指令使用 cmp + setcc + movzx 模式，结果为 0/1
15 * 7. R141: 所有 chunk（mainChunk + functionChunks）编译到一个 CodeHolder，
16 * 每个 functionChunk 对应入口 Label，OP_CALL 用 jmp targetLabel
17 * 8. R141: 函数调用通过 JitContext.frames 帧栈管理（offset 32/40），
18 * r13 = basePointer 指向当前帧 slot 0
19 *
20 * 调用约定（Windows x64）：
21 * - JIT 入口函数签名: int64_t jitEntry(JitContext* ctx)
22 * - ctx 通过 rcx 传入，prologue 中保存到 r12
23 * - 调用 jitPrintInt 时：rcx=ctx, rdx=value，sub rsp, 32 提供 shadow space
24 *
25 * 调用约定（System V x86-64）：
26 * - ctx 通过 rdi 传入，prologue 中保存到 r12
27 * - 调用 jitPrintInt 时：rdi=ctx, rsi=value，sub rsp, 8 维持 16 字节对齐
28 *
29 * @see JIT.h Bytecode.h IBackend.h
30 * @since R138
31 */

54JITBackend::~JITBackend() {
55 // R151/ADR-008: 析构时释放 JIT 已分配的可执行内存，避免内存泄漏
56 // （asmjit::JitRuntime 不会自动释放通过 runtime_.add 分配的内存块）。
57 // per-chunk 架构下所有权权威为 codeBlocks_（main 块也登记其中，
58 // currentEntry_ 仅镜像入口地址，不再单独释放以防双重释放）。
59 // owned=false 的缓存共享块生命周期归进程级块缓存，实例不释放。
60 for (const auto& blk : codeBlocks_) {
61 if (blk.owned && blk.entry != nullptr) {
62 runtime_.release(blk.entry);
63 }
64 }
65 codeBlocks_.clear();
66 currentEntry_ = nullptr;
67 // R152: 释放特化版本 CodeHolder 内存
68 for (JitEntryFn entry : ownedSpecializedEntries_) {
69 if (entry != nullptr) {
70 runtime_.release(entry);
71 }
72 }
73 ownedSpecializedEntries_.clear();
74 // AUDIT-R4 BUG-12 fix: 释放尚未到达 execute() 安全点的退休特化块
75 releaseRetiredSpecializedEntries();
76 // R157: 释放 lazy compilation 分配的 CodeHolder 内存
77 for (JitEntryFn entry : ownedLazyEntries_) {
78 if (entry != nullptr) {
79 runtime_.release(entry);
80 }
81 }
82 ownedLazyEntries_.clear();
83}

122// ============================================================
123// Safepoint GC — JIT 代码在 OP_LOOP 回边处调用
124// ============================================================
125// 收集 JIT 操作数栈和全局槽位中的存活容器对象作为 GC roots，
126// 然后调用 GcManager::collectCycle 回收循环引用孤岛。
127// JIT 代码在调用前已同步 r15 到 ctx->stackTop。
128extern "C" void jitSafepointGc(JitContext* ctx) {
129 if (!ctx || !ctx->gcNeededFlag)
130 return;
131 // 重置标志
132 ctx->gcNeededFlag->store(0, std::memory_order_relaxed);
133
134 // 收集 roots：操作数栈上的指针类型 Value + 全局槽位
135 std::vector<const void*> roots;
136
137 // 1. 扫描全局槽位
138 if (ctx->globalSlots) {
139 auto* backend = static_cast<JITBackend*>(ctx->backendPtr);
140 if (backend) {
141 for (int64_t bits : backend->globalSlotsRef()) {
142 uint64_t ubits = static_cast<uint64_t>(bits);
143 uint64_t tag = ubits & jit_internal::JIT_TAG_FIELD_MASK;
144 if (tag == jit_internal::JIT_PTR_TAG_BASE) {
145 // 指针类型（堆对象）：提取低 48 位作为 RefCounted*
146 const void* ptr = reinterpret_cast<const void*>(ubits & jit_internal::JIT_INT48_MASK);
147 if (ptr)
148 roots.push_back(ptr);
149 }
150 }
151 }
152 }
153
154 // 2. 扫描操作数栈（从 stackTop 到栈底，栈向下增长，所以 stackTop 是最低地址）
155 // 注：JitContext 中没有 stackBottom 字段（无法精确知道栈底，取决于 prologue
156 // 分配大小），因此使用保守上限 1024 个槽位。依据：当前帧大小约 8240 字节
157 // （8240/8 = 1030 槽），1024 是覆盖单帧的保守上限。
158 // Bug #49 fix: 增加防御性上限钳制——即使未来调高预期扫描量，
159 // 也不会越过栈分配区域误扫邻接内存。
160 if (ctx->stackTop) {
161 int64_t* sp = ctx->stackTop;
162 // 扫描当前栈上最多 1024 个槽位（操作数栈容量上限）
163 int slotsToScan = 1024;
164 if (slotsToScan > 1024)
165 slotsToScan = 1024; // 防御性上限钳制
166 for (int i = 0; i < slotsToScan && sp; ++i, ++sp) {
167 uint64_t ubits = static_cast<uint64_t>(*sp);
168 uint64_t tag = ubits & jit_internal::JIT_TAG_FIELD_MASK;
169 if (tag == jit_internal::JIT_PTR_TAG_BASE) {
170 const void* ptr = reinterpret_cast<const void*>(ubits & jit_internal::JIT_INT48_MASK);
171 if (ptr)
172 roots.push_back(ptr);
173 }
174 }
175 }
176
177 // 执行 GC
178 if (!roots.empty()) {
179 GcManager::instance().collectCycle(roots);
180 }
181}

203JitResult JITBackend::execute(const CompileResult& result) {
204 // 重置状态
205 // P2-12: lastError_ 字段保留作为兜底（getLastError 末尾返回），diagnostics_ 为权威来源
206 hasError_ = false;
207 lastError_.clear();
208 diagnostics_.clear();

210#ifndef NDEBUG
211 // D4 fix（审计报告 2026-08-03 D4 加固）：debug 构建执行前校验 JIT 硬编码
212 // NaN-boxing tag 常量与 NaNBox.h 编码一致。原 verifyNanBoxConstants() 无任何
213 // 调用点（原注释声称"首次 execute 时校验"但从未接线）；TestJIT 全量用例在
214 // debug 构建下执行 JIT 时即隐式覆盖。Release 下 assert 展开为空，零开销。
215 verifyNanBoxConstants();
216#endif

218 // R152: 保存当前 CompileResult 引用，供 compileChunkSpecialized 访问 chunk 数据
219 currentResult_ = &result;
220
221 // R139: 初始化全局变量 slot 存储
222 // R142 阶段 3a：初值填充 NaN-boxing NULL (0x7FFA000000000000) 而非 0
223 // 大小由 CompileResult.globalSlotCount 决定
224 globalSlots_.assign(result.globalSlotCount > 0 ? result.globalSlotCount : 0, static_cast<int64_t>(JIT_NULL_BITS));
225 jitContext_.globalSlots = globalSlots_.data();
226
227 // R147: 构建全局变量名→slot 映射，供编译期解析 OP_INDEX_SET_VAR 的 nameIdx→slot
228 // StackVM 运行时通过 resolveMutableGlobal(varName) 查找，JIT 改为编译期解析避免运行时开销
229 globalNameToSlot_.clear();
230 for (int i = 0; i < static_cast<int>(result.globalSlotNames.size()); ++i) {
231 globalNameToSlot_[result.globalSlotNames[i]] = i;
232 }
233
234 // R148: 初始化类注册表和字段顺序收集器
235 // classInfo_ 在 OP_DEFINE_CLASS 执行时填充，供 OP_CLASS_NEW/OP_MEMBER_GET/SET 查询
236 // pendingFieldOrder_ 在 OP_INIT_FIELD 执行时收集字段声明顺序，供 OP_DEFINE_CLASS 提取
237 classInfo_.clear();
238 pendingFieldOrder_.clear();
239 jitContext_.classInfoPtr = &classInfo_;
240 jitContext_.pendingFieldOrderPtr = &pendingFieldOrder_;
241
242 // enum 元信息注册表：从 CompileResult.enumInfos 加载（与 VM initExecution 对齐），
243 // jitBuildEnumVariant 经 ctx->enumRegistryPtr 校验 variant 名/arity/字段类型
244 enumRegistry_.clear();
245 for (const auto& info : result.enumInfos) {
246 enumRegistry_[info.name] = info;
247 }
248 jitContext_.enumRegistryPtr = &enumRegistry_;

250 // 并发原语：同步对象方法内联处理邮箱 + 闭包跳板帧地板栈重置
251 jitContext_.syncMethodHandled = 0;
252 trampolineFrameFloors_.clear();
253 closureTrampoline_ = nullptr; // compileAllChunks 成功后重新解析
254
255 // R149: methodEntries_ 在 compileAllChunks 末尾填充，jitMethodCall 沿继承链查找用
256 // callerBp 是运行时邮箱（JIT 代码调用 jitMethodCall 前写入 r13），无需初始化
257 jitContext_.methodEntriesPtr = &methodEntries_;
258 // R155: funcEntries_ 在 compileAllChunks 末尾填充，jitCallExpr 闭包值调用查找用
259 jitContext_.funcEntriesPtr = &funcEntries_;
260 jitContext_.callerBp = nullptr;

262 // R141: 初始化函数调用帧栈
263 // 预分配 MAX_FRAMES 帧（与 StackVM/RegisterVM MAX_FRAMES 一致），frameCount_ 重置为 0
264 // frameStack_.data() 在 vector 不发生 reallocation 时保持有效，
265 // 故 execute 期间不应修改 frameStack_ 的大小
266 frameStack_.assign(RuntimeLimits::MAX_FRAMES, JitFrame{});
267 frameCount_ = 0;
268 jitContext_.frames = frameStack_.data();
269 jitContext_.frameCount = &frameCount_;

271 // R156: 初始化 upvalue 管理结构
272 // frameUpvaluesStack_ 与 frameStack_ 并行索引，记录每个帧的闭包 upvalues
273 // openUpvalues_ 是升序 multimap，记录所有 open 状态的 upvalue（按栈槽地址索引）
274 // functionClosures_ 是闭包注册表（与 StackVM VM::functionClosures_ 对齐），每次 execute 重置
275 frameUpvaluesStack_.clear();
276 openUpvalues_.clear();
277 functionClosures_.clear();
278 jitContext_.currentBp = nullptr;
279
280 // R162: 清空异常处理栈（支持多次 execute 调用，避免上次残留 handler 干扰）
281 // tryStack_ 和 pendingJumpStack_ 在 OP_TRY_BEGIN/OP_PUSH_JUMP_TARGET 时 push，
282 // 正常路径 OP_TRY_END/OP_FINALLY_END 时 pop，但异常中断时可能残留。
283 tryStack_.clear();
284 pendingJumpStack_.clear();
285
286 // R162: 清空名称变量表（catch 变量等无 slot 的名称变量走此路径）

289 jitContext_.globalsPtr = &globals_;

292 // osrSavedBp/osrSavedSp: OSR 触发时 JIT 代码写入，OSR 入口点读取恢复
293 // osrEntryPoint: triggerOsrMigration 写入，JIT 代码 jmp 此地址实现栈帧迁移
294 // deoptEntryPoint/deoptChunkIdx: triggerDeoptimize 写入（教学版简化，下次 execute 生效）
295 jitContext_.osrSavedBp = nullptr;
296 jitContext_.osrSavedSp = nullptr;
297 jitContext_.osrEntryPoint = nullptr;
298 jitContext_.deoptEntryPoint = nullptr;

302 // memberGetICPtr 在 compileAllChunks 之后设置（此时 memberGetIC_ 已 resize 完毕）
303 icHitCount_ = 0;
304 icMissCount_ = 0;

307 jitContext_.hasError = &hasError_;
308 jitContext_.errorBuffer = &lastError_;

310 // R151/ADR-008: 释放上一次 execute() 分配的全部 JIT 代码块（支持多次 execute 调用）
311 // asmjit::JitRuntime::release 释放 runtime_.add 返回的内存块，避免重复 execute 泄漏。
312 // per-chunk 架构下每 chunk 独立可执行内存，按块粒度释放；currentEntry_ 仅镜像。
313 // owned=false 的缓存共享块跳过（生命周期归进程级块缓存）。
314 for (const auto& blk : codeBlocks_) {
315 if (blk.owned && blk.entry != nullptr) {
316 runtime_.release(blk.entry);

320 currentEntry_ = nullptr;

322 // 跨代合并共存，见 compileAllChunks 注记），新一次 execute 从 0 重新编号
323 callImportSlots_.clear();
324 callImportPatches_.clear();

326 // 特化块的安全点（旧块在上一轮执行中可能仍在本机调用栈上，
327 // 不能在替换点立即释放）。
328 releaseRetiredSpecializedEntries();

330 // 多次 execute() 时，compileAllChunks 会重新 nullify funcEntries_/methodEntries_，
331 // 上一次 lazy 编译的入口不再被引用，需释放避免泄漏
332 for (JitEntryFn entry : ownedLazyEntries_) {
333 if (entry != nullptr) {
334 runtime_.release(entry);
335 }
336 }

338 lazyCompiledChunksList_.clear();

341 // 门控 = 纯 eager 教学路径（!lazyMode_ && !tieredMode_ && 无自定义阈值）：
342 // lazy/tiering/自定义阈值路径的生成代码依赖运行模式与阈值，绕过缓存按需重编译。
343 const bool cacheable = !lazyMode_ && !tieredMode_ && customOsrThresholds_.empty() && customThresholds_.empty();
344 uint64_t cacheKey = 0;
345 bool cacheHit = false;
346 if (cacheable) {
347 cacheKey = computeBlockCacheKey(result);
348 cacheHit = tryExecuteFromCache(cacheKey);
349 }
350 if (!cacheHit) {
351 if (cacheable) {

353 // 持有的副本，命中实例直接引用同一批块与副本，零重绑定（ADR-008）。

367 activeRuntime_ = &runtime_;

370 }
371 }

401 // 调用 JIT 编译后的本地代码
402 // Safepoint GC 方案：将 GcManager 的增量触发回调重定向为设置 gcNeededFlag_，
403 // JIT 代码在 OP_LOOP 回边处检查此标志并调用 jitSafepointGc 收集 roots。
404 // 相比原 CallbackSuppressor 完全抑制方案，本方案允许 GC 在长循环中执行，
405 // 避免内存峰值无限增长。回调仅设置标志（O(1)原子操作），实际 GC 在 safepoint 处执行。
406 gcNeededFlag_.store(0, std::memory_order_relaxed);
407 jitContext_.gcNeededFlag = &gcNeededFlag_;
408 // R166 fix: 循环迭代上限（与 VM dynMaxInstr 语义一致：execute 前配置，执行期间固定）
409 jitContext_.maxLoopIterations = RuntimeLimits::RuntimeConfig::instance().maxLoopIterations();
410 // 设置 GC 触发回调为设置标志（而非直接调用 collectCycle）
411 {
412 // 保存原回调，替换为设置标志的轻量回调
413 GcManager::CallbackSuppressor gcSuppressor;
414 // Suppressor 抑制回调后，我们重新设置一个轻量回调仅设置标志
415 GcManager::instance().setGcTriggerCallback([this]() { gcNeededFlag_.store(1, std::memory_order_release); });
416 int64_t ret = currentEntry_(&jitContext_);
417 (void)ret;
418 }
419 // gcSuppressor 析构恢复原回调
420
421 if (hasError_) {
422 // P2-12: JIT 代码通过 ctx->errorBuffer 写入错误消息到 lastError_，
423 // 在此同步到 diagnostics_（统一错误查询接口，getLastError 从 diagnostics_ 派生）。
424 // DiagSource::JIT 与 VM/Compiler 区分，便于错误来源定位。
425 if (!lastError_.empty()) {
426 diagnostics_.addError(lastError_, 0, 0, DiagSource::JIT);
427 }
428 return JitResult::RuntimeError;
429 }
430
431 // R152/R153: execute() 返回后检查 recompiledFlags，对触发的 chunk 进行特化重编译
432 // 仅对 recompiledFlags_[i]==1 且尚未特化的 chunk 调用特化重编译函数。
433 // 根据类型反馈选择特化策略：
434 // - R152 INT 特化：otherCount==0 && floatCount==0 → compileChunkSpecialized
435 // - R153 FLOAT 特化：floatCount>0 && otherCount==0 → compileChunkSpecializedFloat
436 // - 其他（otherCount>0）：不特化（保持原版本）
437 // 特化版本入口更新到 methodEntries_，下次 execute() 时 emitMethodCall 自动走特化版本
438 for (size_t i = 0; i < recompiledFlags_.size() && i < chunkNames_.size(); ++i) {
439 if (recompiledFlags_[i] == 0) {
440 continue;
441 }
442 // 检查是否已特化（避免重复特化）
443 const std::string& chunkName = chunkNames_[i];
444 bool alreadySpecialized = false;
445 for (const auto& s : specializedChunks_) {
446 if (s == chunkName) {
447 alreadySpecialized = true;
448 break;
449 }
450 }
451 if (alreadySpecialized) {
452 continue;
453 }
454 // 根据类型反馈选择特化策略
455 const auto& tf = (i < typeFeedback_.size()) ? typeFeedback_[i] : minilang::TypeFeedback{};
456 JitEntryFn specEntry = nullptr;
457 if (tf.otherCount == 0 && tf.floatCount == 0) {
458 // R152: 纯 INT 算术 → INT 特化（跳过 emitCheckInt）
459 specEntry = compileChunkSpecialized(result, i);
460 } else if (tf.floatCount > 0 && tf.otherCount == 0) {
461 // R153: 纯 FLOAT 算术 → FLOAT 特化（跳过 INT 原生路径 + 跳过类型反馈收集）
462 specEntry = compileChunkSpecializedFloat(result, i);
463 }
464 // otherCount > 0：不特化（保持类型分派）
465 (void)specEntry; // 入口已更新到 methodEntries_，无需额外处理
466 }
467
468 return JitResult::OK;
469}

610uint64_t JITBackend::computeBlockCacheKey(const CompileResult& result) {
611 uint64_t h = 1469598103934665603ull; // FNV-1a 64 offset basis
612 auto mix = [&h](const void* data, size_t n) {
613 const auto* p = static_cast<const uint8_t*>(data);
614 for (size_t i = 0; i < n; ++i) {
615 h ^= p[i];
616 h *= 1099511628211ull;
617 }
618 };
619 auto mixU64 = [&mix](uint64_t v) { mix(&v, sizeof(v)); };
620 auto mixStr = [&mix, &mixU64](const std::string& s) {
621 mixU64(s.size());
622 mix(s.data(), s.size());
623 };
624 auto mixChunk = [&](const BytecodeChunk& c) {
625 mixStr(c.name);
626 mixU64(static_cast<uint64_t>(c.arity));
627 mixU64(static_cast<uint64_t>(c.requiredArity));
628 mixU64(static_cast<uint64_t>(c.localCount));
629 mixU64(c.code.size());
630 mix(c.code.data(), c.code.size());
631 mixU64(c.constants.size());
632 for (const auto& v : c.constants) {
633 if (v.isString()) {
634 // 字符串常量按内容指纹：raw bits 是堆指针（每次编译分配新地址，
635 // 会导致同源码键不稳定——见 BlockCache 回归测试）
636 mixU64(0x535452); // "STR" 类型标记
637 mixStr(v.stringVal());
638 } else {
639 mixU64(Value::bitsOf(v)); // 标量按 NaN-boxing 原位编码（保守：内容不同必不同键）
640 }
641 }
642 mixU64(c.defaultConstIndices.size());
643 for (int idx : c.defaultConstIndices) {
644 mixU64(static_cast<uint64_t>(idx));
645 }
646 mixU64(c.fieldOrder.size());
647 for (const auto& f : c.fieldOrder) {
648 mixStr(f);
649 }
650 mixU64(c.upvalues.size());
651 for (const auto& u : c.upvalues) {
652 mix(&u, sizeof(u)); // UpvalueDesc 为 POD，按内存布局指纹
653 }
654 mixU64(c.isGenerator ? 1 : 0);
655 mixU64(static_cast<uint64_t>(c.yieldCount));
656 };
657 mixChunk(result.mainChunk);
658 mixU64(result.functionChunks.size());
659 for (const auto& [name, c] : result.functionChunks) {
660 (void)name; // 名称随 chunk.name 入键
661 mixChunk(c);
662 }
663 mixU64(result.globalSlotCount);
664 mixU64(result.globalSlotNames.size());
665 for (const auto& n : result.globalSlotNames) {
666 mixStr(n);
667 }
668 mixU64(result.enumInfos.size());
669 for (const auto& e : result.enumInfos) {
670 mixStr(e.name);
671 }
672 mixU64(kJitCodegenVersion);
673 return h;
674}

676void JITBackend::initPerChunkState() {
677 const size_t n = chunkNames_.size();
678 // R150: per-chunk 调用计数数组（热点检测用）
679 chunkCallCounts_.assign(n, 0);
680 jitContext_.chunkCallCounts = chunkCallCounts_.data();
681 // R151: 热点阈值——默认仅方法 chunk（name 含 '.'）启用；customThresholds_ 覆盖
682 hotThresholds_.assign(n, 0);
683 for (size_t i = 0; i < n; ++i) {
684 if (chunkNames_[i].find('.') != std::string::npos) {
685 hotThresholds_[i] = minilang::kDefaultHotThreshold;
686 }
687 auto it = customThresholds_.find(chunkNames_[i]);
688 if (it != customThresholds_.end()) {
689 hotThresholds_[i] = it->second;
690 }
691 }
692 recompiledFlags_.assign(n, 0);
693 jitContext_.hotThresholds = hotThresholds_.data();
694 jitContext_.recompiledFlags = recompiledFlags_.data();
695 // R152: 类型反馈计数器数组
696 typeFeedback_.assign(n, minilang::TypeFeedback{});
697 jitContext_.typeFeedback = typeFeedback_.data();
698 jitContext_.backendPtr = this;
699 // R154: lastMutatedReceiver_（嵌套左值赋值链中转邮箱）
700 lastMutatedReceiver_ = static_cast<int64_t>(JIT_NULL_BITS);
701 jitContext_.lastMutatedReceiverPtr = &lastMutatedReceiver_;
702 // R157: OSR 回边计数/阈值/标志数组
703 osrLoopCounts_.assign(n, 0);
704 osrLoopThresholds_.assign(n, 0);
705 for (size_t i = 0; i < n; ++i) {
706 auto osrIt = customOsrThresholds_.find(chunkNames_[i]);
707 if (osrIt != customOsrThresholds_.end()) {
708 osrLoopThresholds_[i] = osrIt->second;
709 }
710 }
711 osrRecompiledFlags_.assign(n, 0);
712 jitContext_.osrLoopCountsPtr = osrLoopCounts_.data();
713 jitContext_.osrLoopThresholdsPtr = osrLoopThresholds_.data();
714 jitContext_.osrRecompiledFlagsPtr = osrRecompiledFlags_.data();
715 // R166 fix: 循环迭代上限数组（无限循环防护）
716 loopIterations_.assign(n, 0);
717 jitContext_.loopIterationsPtr = loopIterations_.data();
718}

720bool JITBackend::tryExecuteFromCache(uint64_t key) {
721 std::lock_guard<std::mutex> lock(blockCacheMutex());
722 auto& map = blockCacheMap();
723 auto it = map.find(key);
724 if (it == map.end()) {
725 return false;
726 }
727 JitBlockCacheEntry& e = *it->second;
728
729 // 共享块引用（owned=false：生命周期归缓存，实例的释放循环跳过）
730 for (const auto& sb : e.blocks) {
731 JitCodeBlock blk;
732 blk.entry = sb.entry;
733 blk.chunkName = sb.chunkName;
734 blk.isMain = sb.isMain;
735 blk.entryOffset = sb.entryOffset;
736 blk.epilogueOffset = sb.epilogueOffset;
737 blk.trampolineOffset = sb.trampolineOffset;
738 blk.owned = false;
739 codeBlocks_.push_back(std::move(blk));
740 }
741 currentEntry_ = e.blocks.empty() ? nullptr : e.blocks.front().entry;
742 methodEntries_.clear();
743 for (const auto& [n, info] : e.methodEntries) {
744 methodEntries_.emplace(n, info);
745 }
746 funcEntries_.clear();
747 for (const auto& [n, info] : e.funcEntries) {
748 funcEntries_.emplace(n, info);
749 }
750 chunkNames_ = e.chunkNames;
751 initPerChunkState();
752 // IC 数组：缓存代码的 callSiteId 编号 [0,N) 与新实例分配严格对齐
753 // （缓存组覆盖全部 chunk，编号即全量编号）
754 memberGetIC_.assign(e.memberIcCount > 0 ? e.memberIcCount : 1, MemberGetInlineCacheEntry{});
755 methodCallIC_.assign(e.methodIcCount, MethodCallICEntry{});
756 nextCallSiteId_ = e.memberIcCount;
757 nextMethodCallSiteId_ = e.methodIcCount;
758 // 导入槽：同索引重填（槽位在缓存代码中按相同编号发射）
759 callImportSlots_.assign(e.importPatches.size(), nullptr);
760 callImportPatches_ = e.importPatches;
761 for (const auto& patchInfo : callImportPatches_) {
762 void* target = nullptr;
763 if (patchInfo.isMethod) {
764 auto mit = methodEntries_.find(patchInfo.calleeName);
765 if (mit != methodEntries_.end()) {
766 target = mit->second.entryPtr;
767 }
768 } else {
769 auto fit = funcEntries_.find(patchInfo.calleeName);
770 if (fit != funcEntries_.end()) {
771 target = fit->second.entryPtr;
772 }
773 }
774 if (target == nullptr) {
775 // 缓存元数据损坏（不应发生）：按未命中处理
776 codeBlocks_.clear();
777 methodEntries_.clear();
778 funcEntries_.clear();
779 callImportSlots_.clear();
780 callImportPatches_.clear();
781 return false;
782 }
783 callImportSlots_[patchInfo.slotIdx] = target;
784 }
785 jitContext_.callImportSlots = callImportSlots_.data();
786 // main 块派生地址（错误退出桩目标 + 闭包同步跳板）
787 if (!e.blocks.empty() && e.blocks.front().isMain) {
788 const auto& main = e.blocks.front();
789 jitContext_.errorExit = reinterpret_cast<void*>(reinterpret_cast<uintptr_t>(main.entry) + main.epilogueOffset);
790 closureTrampoline_ =
791 reinterpret_cast<JitTrampolineFn>(reinterpret_cast<uintptr_t>(main.entry) + main.trampolineOffset);
792 }
793 // 教学汇编捕获：仅实例显式启用时才呈现缓存文本（getCapturedAsm 契约 =
794 // setAsmCapture(true) + execute 后有效，未启用时保持为空）
795 capturedAsm_ = asmCaptureEnabled_ ? e.capturedAsm : std::string();
796 cachedResult_ = e.result;
797 currentResult_ = cachedResult_.get();
798 // LRU 触碰（移到队首）
799 auto& lru = blockCacheLru();
800 for (auto lit = lru.begin(); lit != lru.end(); ++lit) {
801 if (lit->first == key) {
802 lru.splice(lru.begin(), lru, lit);
803 break;
804 }
805 }
806 cacheHitsCounter().fetch_add(1, std::memory_order_relaxed);
807 return true;
808}

810void JITBackend::storeBlocksToCache(uint64_t key) {
811 auto entry = std::make_shared<JitBlockCacheEntry>();
812 for (const auto& blk : codeBlocks_) {
813 JitBlockCacheEntry::SharedBlock sb;
814 sb.entry = blk.entry;
815 sb.chunkName = blk.chunkName;
816 sb.isMain = blk.isMain;
817 sb.entryOffset = blk.entryOffset;
818 sb.epilogueOffset = blk.epilogueOffset;
819 sb.trampolineOffset = blk.trampolineOffset;
820 entry->blocks.push_back(std::move(sb));
821 }
822 for (const auto& [n, info] : methodEntries_) {
823 entry->methodEntries.emplace_back(n, info);
824 }
825 for (const auto& [n, info] : funcEntries_) {
826 entry->funcEntries.emplace_back(n, info);
827 }
828 entry->importPatches = callImportPatches_;
829 entry->chunkNames = chunkNames_;
830 entry->memberIcCount = memberGetIC_.size();
831 entry->methodIcCount = methodCallIC_.size();
832 entry->capturedAsm = capturedAsm_;
833 entry->result = cachedResult_; // 快照所有权移交缓存（块内嵌指针宿主）
834
835 std::lock_guard<std::mutex> lock(blockCacheMutex());
836 auto& map = blockCacheMap();
837 auto& lru = blockCacheLru();
838 // 淘汰：FIFO（容量 kBlockCacheCapacity），被淘汰条目的共享块经共享 runtime 释放
839 while (lru.size() >= kBlockCacheCapacity) {
840 auto victim = lru.back();
841 map.erase(victim.first);
842 for (const auto& sb : victim.second->blocks) {
843 if (sb.entry != nullptr) {
844 sharedCacheRuntime().release(sb.entry);
845 }
846 }
847 lru.pop_back();
848 }
849 lru.emplace_front(key, entry);
850 map[key] = entry;
851 cacheStoresCounter().fetch_add(1, std::memory_order_relaxed);
852}
