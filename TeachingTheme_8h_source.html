2// TeachingTheme.h — 教学面板统一主题色板
3// ------------------------------------------------------------
4// 解决问题：教学相关组件散布硬编码颜色（#0078d4 / #5a5a5a / #666 等），
5// 暗色主题下对比度不足、与 QFluentKit 主题色脱节。
6//
7// 提供：
8// - TeachingTheme::primary() 主色（跟随 Fluent Theme::themeColor）
9// - TeachingTheme::primaryHover() hover 态
10// - TeachingTheme::primaryPressed() pressed 态
11// - TeachingTheme::textPrimary() 主文本色
12// - TeachingTheme::textSecondary() 次要文本色
13// - TeachingTheme::textHint() 提示文本色
14// - TeachingTheme::surface() 卡片/面板背景
15// - TeachingTheme::border() 边框色
16//
17// 使用方式：
18// label->setStyleSheet(QString("color: %1;").arg(TeachingTheme::textSecondary().name()));
19// btn->setStyleSheet(QString("background: %1; color: white;").arg(TeachingTheme::primary().name()));
20//
21// 依赖：QFluentKit Theme（主题色变更自动响应）
22//
23// 注：深色主题已移除（2026-07-06），所有颜色固定为亮色配色。
24// 保留 Theme::isDark() 调用以保持 API 兼容，但因 setThemeMode 强制 LIGHT，
25// isDark() 永远返回 false，三元分支永远走 light 路径。
26//
27// R74 fix (2026-07-10)：背景色从 Solarized 米黄（#FDF6E3/#EEE8D5）回退为
28// 中性白/浅灰（#FFFFFF/#F5F5F5），解决跨机器渲染不一致问题。语义强调色
29// （success/warning/error/info）保留原 Solarized 配色以保证可读性。
30// ============================================================
