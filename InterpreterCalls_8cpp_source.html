65 checkBreak(&node);

67 // P1 重构：按调用类型分派到辅助方法
68 if (node.callee) {
69 lastValue_ = callClosureValue(node);

71 }
72 if (node.name == "dict" || node.name == "array") {
73 lastValue_ = callBuiltinConstructor(node);

75 }
76 if (classRegistry_.find(node.name) != classRegistry_.end()) {
77 lastValue_ = constructClassInstance(node);

81 // E3 fix: 改用共享层 executeSharedInput，统一与 VM 的 input() 语义。

83 // 捕获并附上调用点行号/列号上抛 RuntimeError，调用方得到明确的错误信息
84 // 而非静默返回空串继续执行。

118 // 必须在 isBuiltinFunction 之后、callNamedFunction 之前拦截，

340Value Interpreter::callBuiltinConstructor(FunCall& node) {
341 // 内置函数: dict() 和 array()
342 if (node.name == "dict") {
343 // dict() 或 dict({"a": 1, ...})
344 if (node.arguments.empty()) {

346 }

353 return args[0];
354 }
355 runtimeError("dict() 期望无参数或一个字典参数", node.line, node.column);

358 // array() 或 array(1, 2, 3)
359 if (node.arguments.empty()) {
360 return Value(std::vector<Value>());
361 }
362 std::vector<Value> args;
363 for (auto& arg : node.arguments) {
364 args.push_back(evaluate(arg.get()));
365 }
366 return Value(args);
367 }
368 return Value::nullValue();
369}
