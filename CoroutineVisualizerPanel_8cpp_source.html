75/// 描述生成器 chunk 信息（ARCH-10 重构：直接接收 GeneratorChunkInfo 列表，
76/// 不再依赖 BytecodeChunk/RegBytecodeChunk 内部类型）
77QString describeGeneratorChunks(const std::vector<GeneratorChunkInfo>& chunks) {
78 QStringList found;
79 for (const auto& g : chunks) {
80 QString yc = g.isDynamic ? QStringLiteral("动态(INT_MAX)") : QString::number(g.yieldCount);
81 found << QStringLiteral("%1 (yieldCount=%2)").arg(QString::fromUtf8(g.name.c_str()), yc);
82 }
83 if (found.isEmpty()) {
84 return QStringLiteral("未发现生成器 chunk（源码中无 fun* 声明？）");
85 }
86 return found.join(QStringLiteral("<br>"));
87}

91 const QString& instrCount, qint64 elapsedMs, bool hasRuntimeError) {
92 QString stats = QStringLiteral("<b>指令数:</b> %1 | <b>耗时:</b> %2 ms | <b>运行时错误:</b> %3")
93 .arg(instrCount)

95 .arg(hasRuntimeError ? QStringLiteral("是") : QStringLiteral("否"));
96 return QStringLiteral("<h3>编译阶段</h3><p>%1</p>"

100 .arg(compileStage)
101 .arg(execStage)
102 .arg(coroutineSummary)

104}

262/// 将服务层 BackendExecDetail 转换为面板内部 BackendExecResult（含 traceHtml）
263CoroutineVisualizerPanel::BackendExecResult convertDetailToPanelResult(const ::BackendExecDetail& sd,
264 const QString& coroutineState,
265 const QString& compileStageTemplate) {
266 CoroutineVisualizerPanel::BackendExecResult r;
267 r.output = QString::fromUtf8(sd.output.c_str());
268 r.elapsedMs = static_cast<qint64>(sd.elapsedMs);
269 r.instrCount = QString::fromUtf8(sd.instrCountText().c_str());
270 r.coroutineState = coroutineState;
271
272 QString compileStage;
273 if (!sd.success && (sd.errorPrefix == "词法错误" || sd.errorPrefix == "语法错误" || sd.errorPrefix == "编译错误")) {
274 // 编译阶段错误
275 QString errHtml = QString::fromUtf8(sd.errorMsg.c_str()).toHtmlEscaped();
276 compileStage = QStringLiteral("❌ %1: %2").arg(QString::fromUtf8(sd.errorPrefix.c_str()), errHtml);
277 r.status = mlTr("❌ ") + QString::fromUtf8(sd.errorPrefix.c_str()) + QString::fromUtf8(": ") +
278 QString::fromUtf8(sd.errorMsg.c_str());
279 r.success = false;
280 r.traceHtml =
281 buildTraceHtml(compileStage, QStringLiteral(""), r.coroutineState, r.instrCount, r.elapsedMs, false);
282 return r;
283 }
284
285 // 编译成功
286 QString genInfo = describeGeneratorChunks(sd.generatorChunks);
287 compileStage = compileStageTemplate.arg(r.instrCount, genInfo);
288
289 bool hasRuntimeError = (!sd.success && sd.errorPrefix == "运行时错误");
290 QString execStage = QString::fromUtf8(sd.output.c_str()).toHtmlEscaped();
291 if (hasRuntimeError) {
292 QString err = QString::fromUtf8(sd.errorMsg.c_str()).toHtmlEscaped();
293 execStage += QStringLiteral("\n❌ 运行时错误: %1").arg(err);
294 r.status = mlTr("❌ 运行时错误: ") + err;
295 r.success = false;
296 } else {
297 r.status = mlTr("✅ 成功");
298 r.success = true;
299 }
300 r.traceHtml = buildTraceHtml(compileStage, execStage, r.coroutineState, r.instrCount, r.elapsedMs, hasRuntimeError);
301 return r;
302}

335/// 解析源码与输出，提取协程状态摘要：
336/// - 统计源码中 .next() 与 .done() 调用次数
337/// - 按 print 输出行顺序，前 nextCount 行作为 yield 返回值，
338/// 后 doneCount 行作为 done 返回值
339/// - 摘要格式："next×N, done=<last value>"
340QString CoroutineVisualizerPanel::extractCoroutineState(const std::string& src, const QString& output) {
341 const std::string nextTok = ".next()";
342 const std::string doneTok = ".done()";
343 int nextCount = 0;
344 int doneCount = 0;
345 for (size_t pos = 0; (pos = src.find(nextTok, pos)) != std::string::npos; pos += nextTok.size()) {
346 ++nextCount;
347 }
348 for (size_t pos = 0; (pos = src.find(doneTok, pos)) != std::string::npos; pos += doneTok.size()) {
349 ++doneCount;
350 }
351
352 if (nextCount == 0 && doneCount == 0) {
353 return mlTr("（源码未发现 .next()/.done() 调用，非协程代码）");
354 }
355
356 // 切分输出（按换行，忽略空行）
357 QStringList lines = output.split('\n', Qt::SkipEmptyParts);
358
359 // 取最后一个 done 的值作为最终态
360 QString lastDoneValue;
361 int idx = 0;
362 for (int i = 0; i < nextCount && idx < lines.size(); ++i, ++idx) {
363 // 跳过 yield 返回值行
364 }
365 for (int i = 0; i < doneCount && idx < lines.size(); ++i, ++idx) {
366 lastDoneValue = lines[idx].trimmed();
367 }
368
369 if (doneCount > 0) {
370 if (lastDoneValue.isEmpty()) {
371 return QStringLiteral("next×%1, done=<空>").arg(nextCount);
372 }
373 return QStringLiteral("next×%1, done=%2").arg(nextCount).arg(lastDoneValue);
374 }
375 return QStringLiteral("next×%1, done=未查询").arg(nextCount);
376}

408void CoroutineVisualizerPanel::renderConsistency(const BackendExecResult& interp, const BackendExecResult& stackvm,
409 const BackendExecResult& regvm) {
410 bool allSuccess = interp.success && stackvm.success && regvm.success;
411 bool outputEqual = (interp.output == stackvm.output) && (stackvm.output == regvm.output);
412 if (allSuccess && outputEqual) {
413 consistencyLabel_->setText(mlTr("✅ 三后端协程输出一致，yield 重放语义等价性验证通过"));
414 consistencyLabel_->setStyleSheet(QStringLiteral("color:#27AE60; font-weight:bold;"));
