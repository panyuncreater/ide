30void CrashReportDialog::buildUi(const CrashReport& report) {
31 auto* layout = new QVBoxLayout(this);
32
33 // 标题
34 titleLabel_ = new QLabel(mlTr("检测到上次会话异常退出"), this);
35 QFont titleFont = titleLabel_->font();
36 titleFont.setBold(true);
37 titleFont.setPointSize(titleFont.pointSize() + 2);
38 titleLabel_->setFont(titleFont);
39 layout->addWidget(titleLabel_);
40
41 // 信息标签
42 QString infoText;
43 if (report.valid) {
44 infoText = mlTr("信号/异常：") + QString::fromStdString(report.signalName) + "\n" + mlTr("时间：") +
45 QString::fromStdString(report.timestamp) + "\n" + mlTr("崩溃文件：") +
46 QString::fromStdString(report.dumpFilePath);
47 } else {
48 infoText = mlTr("（无有效崩溃报告）");
49 }
50 infoLabel_ = new QLabel(infoText, this);
51 infoLabel_->setWordWrap(true);
52 infoLabel_->setTextInteractionFlags(Qt::TextSelectableByMouse);
53 layout->addWidget(infoLabel_);
54
55 // 栈回溯
56 layout->addWidget(new QLabel(mlTr("栈回溯："), this));
57 traceView_ = new QPlainTextEdit(this);
58 traceView_->setReadOnly(true);
59 if (!report.stackTrace.empty()) {
60 traceView_->setPlainText(QString::fromStdString(report.stackTrace));
61 } else {
62#ifdef _WIN32
63 traceView_->setPlainText(
64 mlTr("（Windows minidump 为二进制格式，请用 Visual Studio 或 WinDbg 打开 .dmp 文件查看调用栈）"));
65#else
66 traceView_->setPlainText(mlTr("（无文本栈回溯）"));
67#endif
68 }
69 layout->addWidget(traceView_, 1);
70
71 // 复选框：不再提示
72 dontShowAgain_ = new QCheckBox(mlTr("忽略此报告，下次启动不再提示"), this);
73 layout->addWidget(dontShowAgain_);
74
75 // 按钮行
76 auto* btnLayout = new QHBoxLayout();
77 openDirButton_ = new QPushButton(mlTr("打开所在目录"), this);
78 copyButton_ = new QPushButton(mlTr("复制信息"), this);
79 deleteButton_ = new QPushButton(mlTr("删除报告"), this);
80 ignoreButton_ = new QPushButton(mlTr("关闭"), this);
81 ignoreButton_->setDefault(true);
82
83 btnLayout->addWidget(openDirButton_);
84 btnLayout->addWidget(copyButton_);
85 btnLayout->addStretch();
86 btnLayout->addWidget(deleteButton_);
87 btnLayout->addWidget(ignoreButton_);
88 layout->addLayout(btnLayout);
89
90 connect(openDirButton_, &QPushButton::clicked, this, &CrashReportDialog::onOpenDirectory);
91 connect(copyButton_, &QPushButton::clicked, this, &CrashReportDialog::onCopyInfo);
92 connect(deleteButton_, &QPushButton::clicked, this, &CrashReportDialog::onDeleteReport);
93 connect(ignoreButton_, &QPushButton::clicked, this, &CrashReportDialog::onIgnore);
94}

96void CrashReportDialog::onOpenDirectory() {
97 if (report_.dumpFilePath.empty())
98 return;
99 QFileInfo fi(QString::fromStdString(report_.dumpFilePath));
100 QString dir = fi.absolutePath();
101 QDesktopServices::openUrl(QUrl::fromLocalFile(dir));

121void CrashReportDialog::onDeleteReport() {
122 auto ret = QMessageBox::question(this, mlTr("确认删除"), mlTr("确定要删除此崩溃报告吗？此操作不可撤销。"),
123 QMessageBox::Yes | QMessageBox::No);
124 if (ret != QMessageBox::Yes)
125 return;
126
127 // 删除磁盘上的报告文件
128 if (!report_.dumpFilePath.empty()) {
129 namespace fs = std::filesystem;
130 std::error_code ec;
131 fs::remove(report_.dumpFilePath, ec);
132 }
133 reportDeleted_ = true;
134 accept();
135}
