25// 容器与成员操作类指令
26// ------------------------------------------------------------
27// 本文件三类指令共享两条贯穿始终的设计主线：
28// 1) COW（写时拷贝）安全：数组/字典/实例的值以 shared_ptr 持有底层数据。
29// 只读访问（pop 出的 const Value）用 const 重载读取，不触发深拷贝；
30// 原地修改（arr[i]=v / dict[k]=v / obj.f=v）会经 arrayVal()/dictVal()/
31// fields() 的 ensureUnique 自动 detach 出新副本。因此"变异后的新容器"
32// 与"调用者栈/全局变量中的旧引用"不再是同一份——必须把新容器写回原处，
33// 否则修改对外不可见。这正是 lastMutatedReceiver_ 与 OP_WRITEBACK_* 的由来。
34// 2) 嵌套左值写回链：对 `arr[i][j]=v`、`obj.f.g=v` 这类嵌套赋值，IR 路径把
35// 每一级变异后的容器存入 lastMutatedReceiver_，由 OP_WRITEBACK_* 整体替换
36// 回基对象（全局变量/栈槽/upvalue）。注意语义是"整体替换"而非"写子字段"，
37// 否则会产生意外嵌套（见各 BUG-NEW 注释）。
38// 成员/索引"读"指令(OP_*_GET)从栈弹出 obj 与下标/字段名，结果压栈；
39// "写"指令(OP_*_SET)弹出 val 与 obj，就地修改后存入 lastMutatedReceiver_，
40// 由后续写回指令完成对外可见的更新。
41// ============================================================

47VMResult VM::executeContainerOps(OpCode op, size_t& ip) {
48 switch (op) {
49 case OpCode::OP_BUILD_ARRAY:
50 case OpCode::OP_BUILD_DICT:
51 case OpCode::OP_BUILD_TUPLE:
52 case OpCode::OP_BUILD_ENUM_VARIANT:
53 return executeContainerBuildOps(op, ip);
54 case OpCode::OP_INDEX_GET:
55 case OpCode::OP_INDEX_SET:
56 case OpCode::OP_INDEX_SET_VAR:
57 case OpCode::OP_INDEX_SET_LOCAL:
58 return executeIndexOps(op, ip);
59 case OpCode::OP_SUPER_MEMBER_GET:
60 case OpCode::OP_MEMBER_GET:
61 case OpCode::OP_MEMBER_SET:
62 case OpCode::OP_MEMBER_SET_VAR:
63 case OpCode::OP_MEMBER_SET_LOCAL:
64 return executeMemberOps(op, ip);
65 case OpCode::OP_ENUM_VARIANT_NAME:
66 case OpCode::OP_ENUM_VARIANT_FIELD:
67 return executeEnumOps(op, ip);
68 default:
69 return runtimeError(ErrorFormat::formatStd("未知操作码: {}", static_cast<int>(op)));
70 }
71}

74// executeContainerBuildOps - 容器构造指令
75// OP_BUILD_ARRAY / OP_BUILD_DICT / OP_BUILD_TUPLE / OP_BUILD_ENUM_VARIANT
76// ============================================================
77VMResult VM::executeContainerBuildOps(OpCode op, size_t& ip) {
78 VMCallFrame& frame = currentFrame();
79 const BytecodeChunk& chunk = *frame.chunk;
80
81 switch (op) {
82 case OpCode::OP_BUILD_ARRAY: {
83 uint8_t count = chunk.code[ip + 1];
84 if (stack_.size() < count)
85 return runtimeError("栈下溢: OP_BUILD_ARRAY");
86 std::vector<Value> elements(count);
87 // 逆序弹出直接填入预分配槽位，无需 reverse
88 for (int i = count - 1; i >= 0; --i) {
89 elements[i] = pop();
90 }
91 push(Value(std::move(elements)));
92 notifyStep(ip, op);
93 ip += 2;
94 break;
95 }
96
97 case OpCode::OP_BUILD_DICT: {
98 uint8_t pairCount = chunk.code[ip + 1];
99 if (stack_.size() < static_cast<size_t>(pairCount) * 2)
100 return runtimeError("栈下溢: OP_BUILD_DICT");
101 // BUG-017 fix (2026-07-18): 记录入口栈深度，循环结束后断言
102 // 栈净变化 = -2*pairCount + 1（弹出全部键值对，压入一个 dict）。
103 // [[maybe_unused]]：Release（NDEBUG）下 assert 展开为 ((void)0)，
104 // stackDepthBefore 不再被引用，触发 MSVC C4189（/WX 视为错误）。
105 [[maybe_unused]] const size_t stackDepthBefore = stack_.size();
106 // L4 fix: 字典键支持 string/int/bool/float，使用 DictKey map
107 // R97 #2 fix: 改用 Value::DictMap（含 DictKeyEqual 透明比较器）
108 Value::DictMap dict;
109 dict.reserve(pairCount);
110 // 先入后出：倒序弹出键值对
111 for (uint8_t i = 0; i < pairCount; ++i) {
112 Value val = pop();
113 Value key = pop();
114 // L4 fix: 键必须是 string/int/bool/float
115 auto dk = Value::dictKeyFromValue(key);
116 if (!dk) {
117 // BUG-VM-04 fix: 错误返回前清理栈上剩余未处理的键值对
118 popN(static_cast<size_t>(pairCount - i - 1) * 2);
119 return runtimeError(ErrorMessages::kDictKeyInvalidType);
120 }
121 dict.emplace(std::move(*dk), std::move(val));
122 }
123 push(Value(std::move(dict)));
124 // BUG-017 fix: 防御性断言——成功路径栈净变化必须为 -2*pairCount + 1。
125 assert(stack_.size() == stackDepthBefore - static_cast<size_t>(pairCount) * 2 + 1);
126 notifyStep(ip, op);
127 ip += 2;
128 break;
129 }
130
131 case OpCode::OP_BUILD_TUPLE: {
132 // R98 元组与解构：构建 immutable 元组
133 uint8_t count = chunk.code[ip + 1];
134 if (stack_.size() < count)
135 return runtimeError("栈下溢: OP_BUILD_TUPLE");
136 std::vector<Value> elements(count);
137 for (int i = count - 1; i >= 0; --i) {
138 elements[i] = pop();
139 }
140 push(Value::makeTuple(std::move(elements)));
141 notifyStep(ip, op);
142 ip += 2;
143 break;
144 }
145
146 case OpCode::OP_BUILD_ENUM_VARIANT: {
147 // 操作数: enumNameConstIdx(2B) + variantNameConstIdx(2B) + argCount(1B)
148 // 栈布局（编译器按声明顺序 push 参数，最后一个参数在栈顶）：
149 // [..., arg0, arg1, ..., argN-1] (argCount = N)
150 // 消费 argCount 个参数，构造 EnumVariantData(enumName, variantName, fields) push。
151 uint16_t enumIdx = chunk.code[ip + 1] | (chunk.code[ip + 2] << 8);
152 uint16_t varIdx = chunk.code[ip + 3] | (chunk.code[ip + 4] << 8);
153 uint8_t argCount = chunk.code[ip + 5];
154 if (enumIdx >= chunk.constants.size() || varIdx >= chunk.constants.size()) {
155 return runtimeError("OP_BUILD_ENUM_VARIANT: 常量池索引越界");
156 }
157 if (stack_.size() < argCount) {
158 return runtimeError("栈下溢: OP_BUILD_ENUM_VARIANT");
159 }
160 const std::string& enumName = chunk.constants[enumIdx].stringVal();
161 const std::string& variantName = chunk.constants[varIdx].stringVal();
162
163 // R99 enum 校验：运行时检查 enum 已声明、variant 存在、参数 arity 一致。
164 // 对齐 Interpreter::visitEnumVariantExpr 的校验逻辑，保证三后端一致。
165 auto enumIt = enumRegistry_.find(enumName);
166 if (enumIt == enumRegistry_.end()) {
167 return runtimeError(ErrorFormat::formatStd("未定义的 enum: {}", enumName), "undefined-function");
168 }
169 const VMEnumInfo& info = enumIt->second;
170 const VMEnumVariantInfo* varInfo = nullptr;
171 for (const auto& v : info.variants) {
172 if (v.name == variantName) {
173 varInfo = &v;
174 break;
175 }
176 }
177 if (varInfo == nullptr) {
178 return runtimeError(ErrorFormat::formatStd("enum '{}' 没有 variant '{}'", enumName, variantName),
179 "undefined-function");
180 }
181 if (static_cast<int>(argCount) != varInfo->arity) {
182 return runtimeError(ErrorFormat::formatStd("enum variant '{}.{}' 期望 {} 个参数，得到 {} 个", enumName,
183
184 variantName, varInfo->arity, static_cast<int>(argCount)),
185 DiagCodes::kArityMismatch);
186 }
187
188 std::vector<Value> fields(argCount);
189 for (int i = argCount - 1; i >= 0; --i) {
190 fields[i] = pop();
191 }
192 // AUDIT-R6 F3 fix: 字段类型注解校验（对齐 Interpreter::visitEnumVariantExpr）。
193 // 原实现仅校验 arity，E.V("s") 对 V(int) 静默构造成功——与 Interpreter 报错不一致。
194 // 泛型类型参数运行时擦除（isTypeParameter 豁免）；实例继承链 fallback 对齐
195 // executeMiscTypeCheck 的 classInfo_ 查找模式。
196 for (size_t fi = 0; fi < fields.size() && fi < varInfo->paramTypes.size(); ++fi) {
197 const std::string& expectedType = varInfo->paramTypes[fi];
198 if (expectedType.empty() || minilang::isTypeParameter(expectedType, info.typeParams))
199 continue;
200 if (!minilang::typeMatchValue(fields[fi], expectedType)) {
201 bool inheritOk = false;
202 if (fields[fi].isInstance()) {
203 auto classIt = classInfo_.find(fields[fi].className());
204 int depth = 0;
205 while (classIt != classInfo_.end() && depth < 64) {
206 if (classIt->second.name == expectedType) {
207 inheritOk = true;
208 break;
209 }
210 if (classIt->second.superClassName.empty())
211 break;
212 classIt = classInfo_.find(classIt->second.superClassName);
213 ++depth;
214 }
215 }
216 if (!inheritOk) {
217 return runtimeError(
218 ErrorFormat::formatStd("enum variant '{}.{}' 第 {} 个参数类型不匹配：期望 {}，得到 {}",
219 enumName, variantName, fi + 1, expectedType, fields[fi].typeName()),
220 DiagCodes::kTypeMismatch);
221 }
222 }
223 }
224 push(Value::makeEnumVariant(enumName, variantName, std::move(fields)));
225 notifyStep(ip, op);
226 ip += 6;
227 break;
228 }
229
230 default:
231 return runtimeError(ErrorFormat::formatStd("未知操作码: {}", static_cast<int>(op)));
232 }
233
234 return VMResult::VM_OK;
235}

262// ============================================================
263// executeIndexGet - 索引读取（OP_INDEX_GET，跨 array/dict/string/tuple 多态）
264// 含 ASCII 快速路径 + UTF-8 慢路径（与 R97 #3 fix 一致）
265// ============================================================
266VMResult VM::executeIndexGet(size_t& ip) {
267 if (stack_.size() < 2)
268 return runtimeError("栈下溢: OP_INDEX_GET");
269 Value idx = pop();
270 // V-P2-15 fix: obj 只读不写，改为 const 避免 arrayVal/dictVal/stringVal 触发 COW 深拷贝
271 const Value obj = pop();
272 if (obj.isArray() && idx.isInt()) {
273 int64_t i = idx.intVal();
274 if (BoundsCheck::inBounds(i, obj.arrayVal().size())) {
275 push(obj.arrayVal()[static_cast<size_t>(i)]);
276 } else {
277 return runtimeError(ErrorFormat::formatStd("数组索引越界: {}, 有效范围 [0, {})", static_cast<long long>(i),
278
279 obj.arrayVal().size()),
280 DiagCodes::kIndexOutOfBounds);
281 }
282 } else if (obj.isDict()) {
283 // L4 fix: 字典键支持 string/int/bool/float
284 // PERF: 透明查找，string 键免 DictKey 构造的字符串深拷贝
285 if (!Value::dictKeyIsValid(idx))
286 return runtimeError(ErrorMessages::kDictKeyInvalidType, DiagCodes::kTypeMismatch);
287 const Value* found = Value::dictFindValue(obj.dictVal(), idx);
288 if (found) {
289 push(*found);
290 } else {
291 push(Value::nullValue()); // 字典访问不存在的键返回 null（与解释器一致）
292 }
293 } else if (obj.isArray()) {
294 return runtimeError(ErrorMessages::kArrayIndexMustBeInt);
295 } else if (obj.isString() && idx.isInt()) {
296 // V3 fix: 基于 UTF-8 码位索引，与解释器 M6 fix 一致
297 int64_t i = idx.intVal();
298 const std::string& s = obj.stringVal();
299
300 // P7 fix: ASCII 快速路径 — 纯 ASCII 字符串直接按字节索引 O(1)
301 // R97 #3 fix: 用 StringData 内缓存的 isAscii 标志替代 lastAsciiStr* 4 字段缓存。
302 // isAscii 与字符串生命周期绑定，O(1) 读取，无堆地址复用误命中风险。
303 if (BoundsCheck::inBounds(i, s.size()) && obj.isAsciiString()) {
304 push(Value(s.substr(static_cast<size_t>(i), 1)));
305 notifyStep(ip, OpCode::OP_INDEX_GET);
306 ip += 1;
307 return VMResult::VM_OK; // 跳过慢路径，直接完成 OP_INDEX_GET
308 }
309
310 size_t charCount = 0;
311 size_t bytePos = 0;
312 size_t targetBytePos = 0;
313 size_t targetByteLen = 0;
314 bool found = false;
315 while (bytePos < s.size()) {
316 size_t charLen = Utf8::byteLength(static_cast<unsigned char>(s[bytePos]));
317 if (static_cast<size_t>(i) == charCount) {
318 targetBytePos = bytePos;
319 targetByteLen = charLen;
320 found = true;
321 }
322 bytePos += charLen;
323 charCount++;
324 }
325 if (i < 0 || !found) {
326 return runtimeError(ErrorFormat::formatStd("字符串索引越界: {}, 有效范围 [0, {})",
327
328 static_cast<long long>(i), static_cast<long long>(charCount)),
329 DiagCodes::kIndexOutOfBounds);
330 }
331 push(Value(s.substr(targetBytePos, targetByteLen)));
332 } else if (obj.isString()) {
333 return runtimeError(ErrorMessages::kStringIndexMustBeInt, DiagCodes::kTypeMismatch);
334 } else if (obj.isDict()) {
335 // R97 #1 fix: 对齐 RegisterVM/Interpreter——dict 非 string 索引原落入通用
336 // ErrorMessages::kTypeNotIndexable 分支，现显式报字典键类型错误。
337 // L4 fix 遗漏的旧字符串 "字典键必须是字符串" 已修正为统一消息。
338 return runtimeError(ErrorMessages::kDictKeyInvalidType, DiagCodes::kTypeMismatch);
339 } else if (obj.isTuple() && idx.isInt()) {
340 // R98 元组与解构：元组索引访问（immutable，与数组语义一致）
341 int64_t i = idx.intVal();
342 if (BoundsCheck::inBounds(i, obj.tupleVal().size())) {
343 push(obj.tupleVal()[static_cast<size_t>(i)]);
344 } else {
345 return runtimeError(ErrorFormat::formatStd("元组索引越界: {}, 有效范围 [0, {})", static_cast<long long>(i),
346
347 obj.tupleVal().size()),
348 DiagCodes::kIndexOutOfBounds);
349 }
350 } else if (obj.isTuple()) {
351 return runtimeError("元组索引必须是整数", DiagCodes::kTypeMismatch);
352 } else {
353 return runtimeError(ErrorMessages::kTypeNotIndexable);
354 }
355 notifyStep(ip, OpCode::OP_INDEX_GET);
356 ip += 1;
357 return VMResult::VM_OK;
358}

360// ============================================================
361// executeIndexSet - 嵌套索引赋值（OP_INDEX_SET，栈顶 obj+idx+val）
362// M1 fix: 用于 arr[i][j] = val 形态，pop obj 后存入 lastMutatedReceiver_
363// ============================================================
364VMResult VM::executeIndexSet(size_t& ip) {
365 // M1 fix: 嵌套索引赋值（如 arr[i][j] = val）
366 // 栈序: [..., obj, outerIdx, innerIdx, val]
367 // 弹出 val, innerIdx, obj → 修改 obj[innerIdx] → 存入 lastMutatedReceiver_
368 if (stack_.size() < 3)
369 return runtimeError("栈下溢: OP_INDEX_SET");
370 Value val = pop();
371 Value innerIdx = pop();
372 Value obj = pop();
373 if (obj.isArray() && innerIdx.isInt()) {
374 int64_t i = innerIdx.intVal();
375 // Perf-Finding: 越界错误路径用 std::as_const 避免 COW detach（pop 出的 obj 与原栈槽共享 ArrayData）
376 if (BoundsCheck::inBounds(i, std::as_const(obj).arrayVal().size())) {
377 obj.arrayVal()[static_cast<size_t>(i)] = val;
378 } else {
379 return runtimeError(ErrorFormat::formatStd("数组索引越界: {}, 有效范围 [0, {})", static_cast<long long>(i),
380
381 std::as_const(obj).arrayVal().size()),
382 DiagCodes::kIndexOutOfBounds);
383 }
384 } else if (obj.isDict()) {
385 // L4 fix: 字典键支持 string/int/bool/float
386 // PERF: 透明写路径，string 键命中时免字符串深拷贝
387 if (!Value::dictKeyIsValid(innerIdx))
388 return runtimeError(ErrorMessages::kDictKeyInvalidType, DiagCodes::kTypeMismatch);
389 Value::dictGetOrInsertRef(obj.dictVal(), innerIdx) = val;
390 } else if (obj.isArray()) {
391 return runtimeError(ErrorMessages::kArrayIndexMustBeInt, DiagCodes::kTypeMismatch);
392 } else {
393 return runtimeError(ErrorMessages::kTypeNotIndexAssignable);
394 }
395 lastMutatedReceiver_ = std::move(obj);
396 notifyStep(ip, OpCode::OP_INDEX_SET);
397 ip += 1;
398 return VMResult::VM_OK;
399}

449// executeIndexSetLocal - 局部变量索引赋值（OP_INDEX_SET_LOCAL）
450// 修改 stack_[bp+slot] 中的数组/字典元素，含 fieldsModified 标记同步（BUG-INH-AUDIT-4 fix）
451// ============================================================
452VMResult VM::executeIndexSetLocal(size_t& ip) {
453 const BytecodeChunk& chunk = *currentFrame().chunk;
454 // 直接修改 stack_[bp+slot] 中的数组/字典元素（用于方法内 this.arr[i] = val）
455 uint8_t slot = chunk.code[ip + 1];
456 if (stack_.size() < 2)
457 return runtimeError("栈下溢: OP_INDEX_SET_LOCAL");
458 Value val = pop();
459 Value index = pop();
460 size_t bp = currentFrame().basePointer;
461 if (bp + slot >= stack_.size()) {
462 return runtimeError("内部错误: 局部变量槽越界");
463 }
464 Value& obj = stack_[bp + slot]; // 栈引用，直接修改
465 // BUG-INH-AUDIT-4 fix: slot==0 谓词对 OP_INDEX_SET_LOCAL 是死代码——slot 0 是
466 // this 实例（非数组/字典），数组/字典分支永不命中。正确谓词应与 OP_SET_LOCAL
467 // 对齐：slot 在字段范围 [1, fieldOrder.size()] 时标记 fieldsModified，确保
468 // this.arr[i]=val 的 COW detach 后字段同步回 this.fields()。
469 bool isFieldSlot = (slot > 0 && currentFrame().chunk && slot <= currentFrame().chunk->fieldOrder.size());
470 if (obj.isArray() && index.isInt()) {
471 int64_t i = index.intVal();
472 // Perf-Finding: 越界错误路径用 std::as_const 避免 COW detach（栈槽 obj 来自 this.arr 时 refCount 常 >1）
473 if (BoundsCheck::inBounds(i, std::as_const(obj).arrayVal().size())) {
474 obj.arrayVal()[static_cast<size_t>(i)] = val;
475 if (isFieldSlot)
476 currentFrame().fieldsModified = true;
477 } else {
478 return runtimeError(ErrorFormat::formatStd("数组索引越界: {}, 有效范围 [0, {})", static_cast<long long>(i),
479
480 std::as_const(obj).arrayVal().size()),
481 DiagCodes::kIndexOutOfBounds);
482 }
483 } else if (obj.isDict()) {
484 // L4 fix: 字典键支持 string/int/bool/float
485 // PERF: 透明写路径，string 键命中时免字符串深拷贝
486 if (!Value::dictKeyIsValid(index))
487 return runtimeError(ErrorMessages::kDictKeyInvalidType, DiagCodes::kTypeMismatch);
488 Value::dictGetOrInsertRef(obj.dictVal(), index) = val;
489 if (isFieldSlot)
490 currentFrame().fieldsModified = true;
491 } else if (obj.isArray()) {
492 return runtimeError(ErrorMessages::kArrayIndexMustBeInt, DiagCodes::kTypeMismatch);
493 } else {
494 return runtimeError(ErrorMessages::kTypeNotIndexAssignable);
495 }
496 notifyStep(ip, OpCode::OP_INDEX_SET_LOCAL);
497 ip += 2;
498 return VMResult::VM_OK;
499}

501// ============================================================
502// executeMemberOps - 成员访问指令
503// OP_SUPER_MEMBER_GET / OP_MEMBER_GET / OP_MEMBER_SET / OP_MEMBER_SET_VAR / OP_MEMBER_SET_LOCAL
504// ============================================================
505VMResult VM::executeMemberOps(OpCode op, size_t& ip) {
506 VMCallFrame& frame = currentFrame();
507 const BytecodeChunk& chunk = *frame.chunk;
508
509 switch (op) {
510 case OpCode::OP_SUPER_MEMBER_GET:
511 case OpCode::OP_MEMBER_GET: {
512 uint16_t idx = chunk.code[ip + 1] | (chunk.code[ip + 2] << 8);
513 if (idx >= chunk.constants.size())
514 return runtimeError("常量池索引越界");
515 const std::string& fieldName = chunk.constants[idx].stringVal();
516 // V-P2-16 fix: obj 只读不写，改为 const 避免 fields/dictVal 触发 COW 深拷贝
517 const Value obj = pop();
518 if (obj.isInstance()) {
519 auto it = obj.fields().find(fieldName);
520 if (it != obj.fields().end()) {
521 push(it->second);
522 } else {
523 // V-P2 fix: 字段未找到时查找方法（与 Interpreter visitMemberAccess 一致），
524 // 返回 "method:ClassName.methodName" 标记字符串，而非直接报错。
525 // 对于 super.field，字段在构造时已合并到实例，方法查找也从当前类开始即可
526 // （super.method() 调用走 OP_SUPER_CALL 直接指定起始类，不经过此路径）
527 const BytecodeChunk* methodChunk = findMethodChunk(obj.className(), fieldName);
528 if (methodChunk != nullptr) {
529 // PERF-05 fix: 驻留 "method:Class.field" 字符串，避免每次成员访问重复构造。
530 // 同一类+方法的成员访问频繁发生（如 obj.x 多次读取），驻留后池中复用同一 std::string。
531 push(Value(StringIntern::internConcat(StringIntern::internConcat("method:", obj.className()),
532 std::string(".") + fieldName)));
533 } else {
534 return runtimeError("类 " + obj.className() + " 没有字段或方法 '" + fieldName + "'");
535 }
536 }
537 } else if (obj.isDict()) {
538 // L4 fix: member access on dict wraps fieldName as DictKey{string}
539 auto it = obj.dictVal().find(Value::DictKey{fieldName});
540 if (it != obj.dictVal().end()) {
541 push(it->second);
542 } else {
543 push(Value::nullValue());
544 }
545 } else if (obj.isNull()) {
546 // P2 fix (null-access): null 值成员访问给出明确的 null-access 诊断码
547 return runtimeError("不能在 null 值上访问属性或调用方法", DiagCodes::kNullAccess);
548 } else {
549 return runtimeError(ErrorMessages::kTypeNotMemberAccessible);
550 }
551 notifyStep(ip, op);
552 ip += 3;
553 break;
554 }
555
556 case OpCode::OP_MEMBER_SET: {
557 // M1 fix: 嵌套成员赋值（如 obj.field.subfield = val 或 arr[i].field = val）
558 // 栈序: [..., obj, val]
559 // 弹出 val, obj → 修改 obj.field → 存入 lastMutatedReceiver_
560 if (stack_.size() < 2)
561 return runtimeError("栈下溢: OP_MEMBER_SET");
562 uint16_t idx = chunk.code[ip + 1] | (chunk.code[ip + 2] << 8);
563 if (idx >= chunk.constants.size())
564 return runtimeError("常量池索引越界");
565 const std::string& fieldName = chunk.constants[idx].stringVal();
566 Value val = pop();
567 Value obj = pop();
568 if (obj.isInstance()) {
569 obj.fields()[fieldName] = val;
570 } else if (obj.isDict()) {
571 // L4 fix: 字典 member-set 用 DictKey{string}
572 obj.dictVal()[Value::DictKey{fieldName}] = val;
573 } else {
574 return runtimeError(ErrorMessages::kTypeNotMemberAssignable);
575 }
576 lastMutatedReceiver_ = std::move(obj);
577 notifyStep(ip, op);
578 ip += 3;
579 break;
580 }
581
582 case OpCode::OP_MEMBER_SET_VAR: {
583 // 直接修改全局变量.fields()[fieldName]
584 uint16_t varIdx = chunk.code[ip + 1] | (chunk.code[ip + 2] << 8);
585 uint16_t fieldIdx = chunk.code[ip + 3] | (chunk.code[ip + 4] << 8);
586 if (varIdx >= chunk.constants.size() || fieldIdx >= chunk.constants.size())
587 return runtimeError("常量池索引越界");
588 const std::string& varName = chunk.constants[varIdx].stringVal();
589 const std::string& fieldName = chunk.constants[fieldIdx].stringVal();
590 Value val = pop();
591 // A2: Dedup-7B resolveMutableGlobal 统一全局变量解析
592 Value* objPtr = resolveMutableGlobal(varName);
593 if (!objPtr)
594 return runtimeError("未定义的变量: " + varName, DiagCodes::kUndefinedVariable);
595 Value& obj = *objPtr; // 引用，直接修改
596 if (obj.isInstance()) {
597 obj.fields()[fieldName] = val;
598 } else if (obj.isDict()) {
599 // L4 fix: 字典 member-set-var 用 DictKey{string}
600 obj.dictVal()[Value::DictKey{fieldName}] = val;
601 } else {
602 return runtimeError(ErrorMessages::kTypeNotMemberAssignable);
603 }
604 notifyStep(ip, op);
605 ip += 5;
606 break;
607 }
608
609 case OpCode::OP_MEMBER_SET_LOCAL: {
610 // 直接修改 stack_[bp+slot].fields()[fieldName]（用于方法内 this.field = val）
611 uint8_t slot = chunk.code[ip + 1];
612 uint16_t fieldIdx = chunk.code[ip + 2] | (chunk.code[ip + 3] << 8);
613 if (fieldIdx >= chunk.constants.size())
614 return runtimeError("常量池索引越界");
615 const std::string& fieldName = chunk.constants[fieldIdx].stringVal();
616 Value val = pop();
617 size_t bp = currentFrame().basePointer;
618 if (bp + slot >= stack_.size()) {
619 return runtimeError("内部错误: 局部变量槽越界");
620 }
621 Value& obj = stack_[bp + slot]; // 栈引用，直接修改
622 if (obj.isInstance()) {
623 obj.fields()[fieldName] = val;
624 // VM fix: 标记字段已修改，OP_RETURN 可跳过只读方法的字段同步
625 if (slot == 0)
626 currentFrame().fieldsModified = true;
627 // 当 slot==0（写 this.field）时，也需同步更新对应的字段槽
628 // 否则 OP_RETURN 会用字段槽的旧值覆盖 this.fields()，导致 this.field 赋值丢失
629 if (slot == 0 && currentFrame().chunk && !currentFrame().chunk->fieldOrder.empty()) {
630 // #11 fix: 用 BytecodeChunk::fieldSlotIndex O(1) 查找替代线性扫描
631 size_t fi = currentFrame().chunk->fieldSlotIndex(fieldName);
632 if (fi != SIZE_MAX) {
633 size_t slotPos = bp + 1 + fi;
634 if (slotPos < stack_.size()) {
635 stack_[slotPos] = val;
636 }
637 }
638 }
639 } else if (obj.isDict()) {
640 // L4 fix: 字典 member-set-local 用 DictKey{string}
641 obj.dictVal()[Value::DictKey{fieldName}] = val;
642 } else {
643 return runtimeError(ErrorMessages::kTypeNotMemberAssignable);
644 }
645 notifyStep(ip, op);
646 ip += 4;
647 break;
648 }
649
650 default:
651 return runtimeError(ErrorFormat::formatStd("未知操作码: {}", static_cast<int>(op)));
652 }
653
654 return VMResult::VM_OK;
655}

657// ============================================================
658// executeEnumOps - 枚举查询指令
659// OP_ENUM_VARIANT_NAME / OP_ENUM_VARIANT_FIELD
660// ============================================================
661VMResult VM::executeEnumOps(OpCode op, size_t& ip) {
662 VMCallFrame& frame = currentFrame();
663 const BytecodeChunk& chunk = *frame.chunk;
664
665 switch (op) {
666 case OpCode::OP_ENUM_VARIANT_NAME: {
667 // 操作数: enumNameConstIdx(2B) + variantNameConstIdx(2B)
668 // 栈: [..., scrutinee] → pop scrutinee，检查是否为 enum variant 且 enum/variant 名匹配，
669 // push bool。不匹配（包括非 enum variant 类型）push false。
670 uint16_t enumIdx = chunk.code[ip + 1] | (chunk.code[ip + 2] << 8);
671 uint16_t varIdx = chunk.code[ip + 3] | (chunk.code[ip + 4] << 8);
672 if (enumIdx >= chunk.constants.size() || varIdx >= chunk.constants.size()) {
673 return runtimeError("OP_ENUM_VARIANT_NAME: 常量池索引越界");
674 }
675 if (stack_.empty()) {
676 return runtimeError("栈下溢: OP_ENUM_VARIANT_NAME");
677 }
678 Value scrut = pop();
679 const std::string& enumName = chunk.constants[enumIdx].stringVal();
680 const std::string& variantName = chunk.constants[varIdx].stringVal();
681 bool matched = false;
682 if (scrut.isEnumVariant()) {
683 matched = (scrut.enumVariantEnumName() == enumName) && (scrut.enumVariantName() == variantName);
684 }
685 push(Value(matched));
686 notifyStep(ip, op);
687 ip += 5;
688 break;
689 }
690
691 case OpCode::OP_ENUM_VARIANT_FIELD: {
692 // 无操作数。栈: [..., scrutinee, index] → pop index, pop scrutinee,
693 // push scrutinee.fields[index]。scrutinee 非 enum variant 或 index 越界 → runtimeError。
694 if (stack_.size() < 2) {
695 return runtimeError("栈下溢: OP_ENUM_VARIANT_FIELD");
696 }
697 Value idx = pop();
698 Value scrut = pop();
699 if (!scrut.isEnumVariant()) {
700 return runtimeError("OP_ENUM_VARIANT_FIELD: 栈顶不是 enum variant");
701 }
702 if (!idx.isInt()) {
703 return runtimeError("OP_ENUM_VARIANT_FIELD: 索引必须是整数");
704 }
705 int64_t i = idx.intVal();
706 const auto& fields = scrut.enumVariantFields();
707 if (i < 0 || static_cast<size_t>(i) >= fields.size()) {
708 return runtimeError(ErrorFormat::formatStd("OP_ENUM_VARIANT_FIELD: 索引越界 {}, 有效范围 [0, {})",
709
710 static_cast<long long>(i), fields.size()),
711 DiagCodes::kIndexOutOfBounds);
712 }
713 push(fields[static_cast<size_t>(i)]);
714 notifyStep(ip, op);
715 ip += 1;
716 break;
717 }
718
719 default:
720 return runtimeError(ErrorFormat::formatStd("未知操作码: {}", static_cast<int>(op)));
721 }
722
723 return VMResult::VM_OK;
724}

754VMResult VM::writebackToGlobalVar(OpCode op, size_t& ip) {
755 VMCallFrame& frame = currentFrame();
756 const BytecodeChunk& chunk = *frame.chunk;
757 bool isMember = (op == OpCode::OP_WRITEBACK_MEMBER_VAR);
758
759 // 操作数: varIdx(2B) [+ fieldIdx(2B) 仅 MEMBER]
760 uint16_t varIdx = chunk.code[ip + 1] | (chunk.code[ip + 2] << 8);
761
762 if (isMember) {
763 // BUG-NEW fix: 语义对齐 RegisterVM 的 REG_WRITEBACK_MEMBER_VAR ——
764 // 将 lastMutatedReceiver_（MEMBER_SET 产生的变异后整个容器）整体替换全局变量，
765 // 而非写入 obj.field。原实现 obj.field = lastMutatedReceiver_ 导致
766 // d.x = 42 把整个变异后 d 赋给 d["x"]，产生嵌套字典。
767 uint16_t fieldIdx = chunk.code[ip + 3] | (chunk.code[ip + 4] << 8);
768 if (varIdx >= chunk.constants.size() || fieldIdx >= chunk.constants.size())
769 return runtimeError("常量池索引越界");
770 const std::string& varName = chunk.constants[varIdx].stringVal();
771 // fieldIdx 仅用于反汇编/调试，运行时不需要（整体替换语义）
772 (void)chunk.constants[fieldIdx];
773 // A2: Dedup-7B resolveMutableGlobal 统一全局变量解析
774 Value* objPtr = resolveMutableGlobal(varName);
775 if (!objPtr) {
776 lastMutatedReceiver_ = Value::nullValue();
777 return runtimeError("未定义的变量: " + varName, DiagCodes::kUndefinedVariable);
778 }
779 *objPtr = std::move(lastMutatedReceiver_);
780 lastMutatedReceiver_ = Value::nullValue();
781 notifyStep(ip, op);
782 ip += 5;
783 return VMResult::VM_OK;
784 }
785
786 // OP_WRITEBACK_INDEX_VAR
787 // BUG-NEW fix: 语义对齐 RegisterVM 的 REG_WRITEBACK_INDEX_VAR ——
788 // 整体替换全局变量为 lastMutatedReceiver_，不 pop 索引。
789 // 原实现 pop 索引后做 obj[index] = mutated，但 IR 路径不向栈推入索引，
790 // 导致栈下溢；且语义应为整体替换而非写入 obj[index]。
791 if (varIdx >= chunk.constants.size()) {
792 lastMutatedReceiver_ = Value::nullValue();
793 return runtimeError("常量池索引越界");
794 }
795 const std::string& varName = chunk.constants[varIdx].stringVal();
796 // A2: Dedup-7B resolveMutableGlobal 统一全局变量解析
797 Value* objPtr = resolveMutableGlobal(varName);
798 if (!objPtr) {
799 lastMutatedReceiver_ = Value::nullValue();
800 return runtimeError("未定义的变量: " + varName, DiagCodes::kUndefinedVariable);
801 }
802 *objPtr = std::move(lastMutatedReceiver_);
803 lastMutatedReceiver_ = Value::nullValue();
804 notifyStep(ip, op);
805 ip += 3;
806 return VMResult::VM_OK;
807}

809VMResult VM::writebackToStackSlot(OpCode op, size_t& ip) {
810 VMCallFrame& frame = currentFrame();
811 const BytecodeChunk& chunk = *frame.chunk;
812 bool isMember = (op == OpCode::OP_WRITEBACK_MEMBER_LOCAL);
813
814 // 操作数: slot(1B) [+ fieldIdx(2B) 仅 MEMBER]
815 uint8_t slot = chunk.code[ip + 1];
816
817 if (isMember) {
818 // BUG-NEW fix: 语义对齐 RegisterVM —— 整体替换栈槽为 lastMutatedReceiver_，
819 // 而非写入 obj.field。原实现 obj.field = mutated 导致局部 d.x = 42 把整个
820 // 变异后 d 赋给 d["x"]。IR 路径不使用字段槽（fieldSlot），故移除 slot==0
821 // 的字段槽同步逻辑（该逻辑仅服务于 Compiler.cpp 非 IR 路径，但本指令仅由
822 // IR 路径发射，不会冲突）。
823 //
824 // BUG-INH-AUDIT-4 fix: 若 slot 是字段槽（this.field 的字段位置），
825 // 标记 fieldsModified=true，确保 OP_RETURN 时字段同步回 this 实例。
826 // 否则 this.obj.field = val 的 COW detach 后字段不会写回 this.fields()。
827 uint16_t fieldIdx = chunk.code[ip + 2] | (chunk.code[ip + 3] << 8);
828 if (fieldIdx >= chunk.constants.size())
829 return runtimeError("常量池索引越界");
830 // fieldIdx 仅用于反汇编/调试，运行时不需要（整体替换语义）
831 (void)chunk.constants[fieldIdx];
832 size_t bp = currentFrame().basePointer;
833 if (bp + slot >= stack_.size()) {
834 lastMutatedReceiver_ = Value::nullValue();
835 return runtimeError("OP_WRITEBACK_MEMBER_LOCAL: 栈槽越界");
836 }
837 stack_[bp + slot] = std::move(lastMutatedReceiver_);
838 lastMutatedReceiver_ = Value::nullValue();
839 // BUG-INH-AUDIT-4 fix: 字段槽写回需标记 fieldsModified
840 if (slot > 0 && currentFrame().chunk && slot <= currentFrame().chunk->fieldOrder.size()) {
841 currentFrame().fieldsModified = true;
842 }
843 notifyStep(ip, op);
844 ip += 4;
845 return VMResult::VM_OK;
846 }
847
848 // OP_WRITEBACK_INDEX_LOCAL
849 // BUG-NEW fix: 同 OP_WRITEBACK_INDEX_VAR，整体替换栈槽，不 pop 索引。
850 // BUG-INH-AUDIT-4 fix: 若 slot 是字段槽（this.arr[i]=val 的字段位置），
851 // 标记 fieldsModified=true，确保 OP_RETURN 时字段同步回 this 实例。
852 // 否则 OP_INDEX_SET 的 COW detach 后新数组虽写回 stack_[bp+slot]，
853 // 但 this.fields() 仍指向旧数组，OP_RETURN 不会同步。
854 size_t bp = currentFrame().basePointer;
855 if (bp + slot >= stack_.size()) {
856 lastMutatedReceiver_ = Value::nullValue();
857 return runtimeError("OP_WRITEBACK_INDEX_LOCAL: 栈槽越界");
858 }
859 stack_[bp + slot] = std::move(lastMutatedReceiver_);
860 lastMutatedReceiver_ = Value::nullValue();
861 // BUG-INH-AUDIT-4 fix: 字段槽写回需标记 fieldsModified
862 if (slot > 0 && currentFrame().chunk && slot <= currentFrame().chunk->fieldOrder.size()) {
863 currentFrame().fieldsModified = true;
864 }
865 notifyStep(ip, op);
866 ip += 2;
867 return VMResult::VM_OK;
868}

870VMResult VM::writebackToUpvalue(OpCode op, size_t& ip) {
871 VMCallFrame& frame = currentFrame();
872 const BytecodeChunk& chunk = *frame.chunk;
873 bool isMember = (op == OpCode::OP_WRITEBACK_MEMBER_UPVALUE);
874
875 // 操作数: uvIdx(1B) [+ fieldIdx(2B) 仅 MEMBER，运行时不读取]
876 uint8_t uvIdx = chunk.code[ip + 1];
877
878 // 注意：原 MEMBER_UPVALUE 实现不检查 fieldIdx 越界（fieldIdx 仅用于反汇编），
879 // 与 MEMBER_VAR/MEMBER_LOCAL 行为不同。此处保留原行为。
880 if (static_cast<size_t>(uvIdx) >= frame.upvalues.size()) {
881 lastMutatedReceiver_ = Value::nullValue();
882 return runtimeError(isMember ? "OP_WRITEBACK_MEMBER_UPVALUE: upvalue 索引越界"
883 : "OP_WRITEBACK_INDEX_UPVALUE: upvalue 索引越界");
884 }
885 auto& uv = frame.upvalues[uvIdx];
886 if (uv->isClosed) {
887 uv->value = std::move(lastMutatedReceiver_);
888 } else {
889 if (uv->stackSlot >= stack_.size()) {
890 lastMutatedReceiver_ = Value::nullValue();
891 return runtimeError(isMember ? "OP_WRITEBACK_MEMBER_UPVALUE: upvalue 栈槽越界"
892 : "OP_WRITEBACK_INDEX_UPVALUE: upvalue 栈槽越界");
893 }
894 stack_[uv->stackSlot] = std::move(lastMutatedReceiver_);
895 // P0-3 fix: 与 OP_SET_UPVALUE 对齐——若修改的是某外层帧的字段槽，
896 // 标记该帧 fieldsModified，确保 OP_RETURN 时字段同步回 this 实例。
897 // 闭包内 `capturedField.subfield = v`（MEMBER）或 `capturedArr[i] = v`（INDEX）
898 // 的变异必须传播回 this 实例。
899 if (uv->owningFrameIdx < frames_.size()) {
900 VMCallFrame& of = frames_[uv->owningFrameIdx];
901 if (of.chunk && !of.chunk->fieldOrder.empty()) {
902 size_t fieldStart = of.basePointer + 1;
903 size_t fieldEnd = fieldStart + of.chunk->fieldOrder.size();
904 if (uv->stackSlot >= fieldStart && uv->stackSlot < fieldEnd) {
905 of.fieldsModified = true;
906 }
907 }
908 }
909 }
910 lastMutatedReceiver_ = Value::nullValue();
911 notifyStep(ip, op);
912 ip += isMember ? 4 : 2;
913 return VMResult::VM_OK;
914}

958// ============================================================
959// executeMiscStackOps - 栈操作 + 字段初始化指令（7 个 case）
960// 围绕栈顶元素操作：消费（POP/PRINT）、复制（DUP/DUP_N）、交换（SWAP）、
961// 读取变异接收者（LOAD_MUTATED）、字段写入栈顶实例（INIT_FIELD）
962// ============================================================
963VMResult VM::executeMiscStackOps(OpCode op, size_t& ip) {
964 VMCallFrame& frame = currentFrame();
965 const BytecodeChunk& chunk = *frame.chunk;
966 switch (op) {
967 case OpCode::OP_PRINT: {
968 Value val = pop();
969 // S5 fix: 字符串类型直接用 stringVal() 引用，避免 toString() 拷贝
970 outputCallback_(val.isString() ? val.stringVal() : val.toString());
971 notifyStep(ip, op);
972 ip += 1;
973 break;
974 }
975
976 case OpCode::OP_POP:
977 pop();
978 notifyStep(ip, op);
979 ip += 1;
980 break;
981
982 case OpCode::OP_DUP:
983 push(peek(0));
984 notifyStep(ip, op);
985 ip += 1;
986 break;
987
988 case OpCode::OP_SWAP: {
989 // R99 match 表达式：交换栈顶两个值
990 if (stack_.size() < 2)
991 return runtimeError("栈下溢: OP_SWAP");
992 // 仅交换 top 和 top-1（通过 std::swap_refs 或手动）
993 Value tmp = std::move(stack_[stack_.size() - 1]);
994 stack_[stack_.size() - 1] = std::move(stack_[stack_.size() - 2]);
995 stack_[stack_.size() - 2] = std::move(tmp);
996 notifyStep(ip, op);
997 ip += 1;
998 break;
999 }
1000
1001 case OpCode::OP_LEN: {
1002 // R134 模式匹配扩展：获取容器长度（array/dict/string/tuple）
1003 if (stack_.empty())
1004 return runtimeError("栈下溢: OP_LEN");
1005 const Value& v = peek(0);
1006 int64_t len = 0;
1007 if (v.isArray()) {
1008 len = static_cast<int64_t>(v.arrayVal().size());
1009 } else if (v.isDict()) {
1010 len = static_cast<int64_t>(v.dictVal().size());
1011 } else if (v.isString()) {
1012 // 字符串长度按 codepoint 计算（与 RegisterVM REG_LEN / print/字符串索引语义一致）
1013 len = static_cast<int64_t>(v.codepointCount());
1014 } else if (v.isTuple()) {
1015 len = static_cast<int64_t>(v.tupleVal().size());
1016 } else {
1017 return runtimeError("OP_LEN: 操作数必须是 array/dict/string/tuple，实际为 " + v.typeName());
1018 }
1019 pop();
1020 push(Value(len));
1021 notifyStep(ip, op);
1022 ip += 1;
1023 break;
1024 }
1025
1026 case OpCode::OP_LOAD_MUTATED: {
1027 // MEDIUM-1/2 fix: 读取 lastMutatedReceiver_ 到栈顶（不清除）。
1028 // 嵌套左值写回链中，上一级 MEMBER_SET/INDEX_SET 将变异后容器存入
1029 // lastMutatedReceiver_，此处读取供下一级 SET 作为 val 使用。
1030 push(lastMutatedReceiver_);
1031 notifyStep(ip, op);
1032 ip += 1;
1033 break;
1034 }
1035
1036 case OpCode::OP_DUP_N: {
1037 uint8_t depth = chunk.code[ip + 1];
1038 if (depth >= stack_.size()) {
1039 return runtimeError("OP_DUP_N: 栈深度不足");
1040 }
1041 push(peek(depth));
1042 notifyStep(ip, op);
1043 ip += 2;
1044 break;
1045 }
1046
1047 case OpCode::OP_INIT_FIELD: {
1048 uint16_t idx = chunk.code[ip + 1] | (chunk.code[ip + 2] << 8);
1049 if (idx >= chunk.constants.size())
1050 return runtimeError("常量池索引越界");
1051 const std::string& fieldName = chunk.constants[idx].stringVal();
1052 Value val = pop();
1053 // 栈顶是实例（OP_CLASS_NEW 推入的），直接修改
1054 if (!stack_.empty() && stack_.back().isInstance()) {
1055 stack_.back().fields()[fieldName] = val;
1056 } else {
1057 return runtimeError("OP_INIT_FIELD: 栈顶不是实例");
1058 }
1059 // M3 fix: 记录字段声明顺序（OP_INIT_FIELD 按 AST 声明顺序执行）
1060 pendingFieldOrder_.push_back(fieldName);
1061 notifyStep(ip, op);
1062 ip += 3;
1063 break;
1064 }
1065
1066 default:
1067 return runtimeError(ErrorFormat::formatStd("executeMiscStackOps 未知操作码: {}", static_cast<int>(op)));
1068 }
1069
1070 return VMResult::VM_OK;
1071}

1208// executeMiscExceptionOps - 异常处理 + finally 跳转栈指令（5 个 case）
1209// 围绕 tryStack_（try/catch handler 栈）和 pendingJumpStack_（finally 末尾跳转目标栈）管理
1210// ============================================================
1211VMResult VM::executeMiscExceptionOps(OpCode op, size_t& ip) {
1212 VMCallFrame& frame = currentFrame();
1213 const BytecodeChunk& chunk = *frame.chunk;
1214 switch (op) {
1215 case OpCode::OP_TRY_BEGIN: {
1216 // F11: push try handler，记录 catch 目标和当前栈深度
1217 uint16_t catchOffset = chunk.code[ip + 1] | (chunk.code[ip + 2] << 8);
1218 size_t catchIp = ip + 3 + catchOffset;
1219 if (catchIp >= chunk.code.size())
1220 return runtimeError("OP_TRY_BEGIN: catch 目标越界");
1221 tryStack_.push_back({catchIp, stack_.size(), frames_.size() - 1});
1222 notifyStep(ip, op);
1223 ip += 3;
1224 break;
1225 }
1226
1227 case OpCode::OP_TRY_END: {
1228 // F11: try 块正常结束，弹出 try 处理器
1229 if (!tryStack_.empty() && tryStack_.back().frameIndex == frames_.size() - 1) {
1230 tryStack_.pop_back();
1231 }
1232 notifyStep(ip, op);
1233 ip += 1;
1234 break;
1235 }
1236
1237 case OpCode::OP_THROW: {
1238 // F11: 弹出栈顶值作为异常，搜索 try 处理器
1239 // Bug1 fix: 空栈检查 — pop() 在空栈时设置 hasError_ 并返回 null，
1240 // 若继续 throwException(null) 会使 catch handler 以错误数据执行
1241 if (stack_.empty()) {
1242 return runtimeError("throw 语句在空栈上执行");
1243 }
1244 Value thrownValue = pop();
1245 // P2-2 fix: 调试模式下通知步进（throwException 可能改变 ip）
1246 notifyStep(ip, op);
1247 return throwException(std::move(thrownValue));
1248 }
1249
1250 case OpCode::OP_PUSH_JUMP_TARGET: {
1251 // AUDIT-P1.1 fix: push 跳转目标到 pendingJumpStack_，供 finally 末尾的 OP_FINALLY_END 取出。
1252 // AUDIT-R7 F1 fix: 附带当前帧索引，使 FINALLY_END 能识别并丢弃跨帧残留。
1253 uint16_t target = chunk.code[ip + 1] | (chunk.code[ip + 2] << 8);
1254 if (target >= chunk.code.size())
1255 return runtimeError("OP_PUSH_JUMP_TARGET: 跳转目标越界");
1256 pendingJumpStack_.push_back({static_cast<size_t>(target), frames_.size() - 1});
1257 notifyStep(ip, op);
1258 ip += 3;
1259 break;
1260 }
1261
1262 case OpCode::OP_FINALLY_END: {
1263 // AUDIT-P1.1 fix: finally 块正常路径末尾。若 pendingJumpStack_ 非空，
1264 // 说明是 break/continue 触发的 finally，pop 目标并跳转；否则继续执行（正常完成）。
1265 // AUDIT-R7 F1 fix: 只消费本帧条目——先惰性丢弃已返回深帧的残留（finally 体内
1266 // return/TCO 跳过了其 FINALLY_END）；若栈顶属于更浅帧（调用方在途续跳，本帧
1267 // 是 finally 体内被调函数）则不消费，视为正常完成继续执行。
1268 notifyStep(ip, op);
1269 size_t curFrameIdx = frames_.size() - 1;
1270 while (!pendingJumpStack_.empty() && pendingJumpStack_.back().frameIndex > curFrameIdx) {
1271 pendingJumpStack_.pop_back();
1272 }
1273 if (!pendingJumpStack_.empty() && pendingJumpStack_.back().frameIndex == curFrameIdx) {
1274 size_t target = pendingJumpStack_.back().target;
1275 pendingJumpStack_.pop_back();
1276 if (target >= chunk.code.size())
1277 return runtimeError("OP_FINALLY_END: 跳转目标越界");
1278 ip = target;
1279 } else {
1280 ip += 1;
1281 }
1282 break;
1283 }
1284
1285 default:
1286 return runtimeError(ErrorFormat::formatStd("executeMiscExceptionOps 未知操作码: {}", static_cast<int>(op)));
1287 }
1288
1289 return VMResult::VM_OK;
