79/// 标记当前里程碑/活动已完成并刷新。
80// AUDIT-P1 fix: 完成态持久化与信号解耦——markCompleted 内直接调用
81// LearnerProgressStore::markCompleted + save，避免依赖外部 connect 信号到
82// learningPathPanel_->markActivityCompleted（懒加载未构造时信号无接收方，

84void CodeJourneyInfoPanel::markCompleted() {
85 if (journeyCompleted_)
86 return;
87 journeyCompleted_ = true;
88 // 直接持久化到 LearnerProgressStore，不依赖外部信号连接
89 LearnerProgressStore::instance().markCompleted("code-journey");
90 saveLearnerProgressWithFeedback(this); // P2-UX fix: 失败时弹 toast 避免静默丢失
91 refreshProgress();
92 emit journeyCompleted();
93}
