4// 简化方案：不使用全屏遮罩层，而是给目标 widget 临时设置

163/// 显示指定索引的步骤，定位气泡到目标控件。
164void GuidedTour::showStep(int index) {
165 if (index < 0 || index >= steps_.size())
166 return;
167 if (!bubble_)
168 return;
169
170 currentIndex_ = index;
171 const Step& step = steps_.at(index);
172
173 // ---- 恢复上一步被高亮 widget 的原始样式 ----
174 if (highlightedTarget_) {
175 highlightedTarget_->setStyleSheet(savedStyleSheet_);
176 highlightedTarget_.clear();
177 savedStyleSheet_.clear();
178 }
179
180 // ---- 高亮当前目标 widget（target 为 nullptr 时跳过，气泡将居中显示）----
181 if (step.target) {
182 savedStyleSheet_ = step.target->styleSheet();
183 step.target->setStyleSheet(
184 QString("border: 2px solid %1; border-radius: 4px;").arg(TeachingTheme::primary().name()));
185 highlightedTarget_ = step.target;
186 }
187
188 // ---- 填充气泡内容 ----
189 bubbleTitle_->setText(step.title);
190 bubbleDesc_->setText(step.description);
191 stepIndicator_->setText(mlTr("步骤 %1 / %2").arg(index + 1).arg(steps_.size()));
192 primaryBtn_->setText(step.primaryBtnText);
193 secondaryBtn_->setText(step.secondaryBtnText);
194 // 上一步按钮：首步禁用
195 if (prevBtn_) {
196 prevBtn_->setEnabled(index > 0);
197 }
198
199 // ---- 计算目标矩形并初步定位气泡 ----
200 // OPT-2 fix: 移除 show() 之前的 adjustSize()——首次显示前 Qt 布局尚未完成，
201 // sizeHint 可能不精确，导致 positionBubble 用的 bubble_->size() 偏差。
202 // 先用当前 size 初步定位，show() 之后再通过 QTimer::singleShot(0, ...) 在
203 // 事件循环空闲时 adjustSize() 并重新定位，确保首次气泡几何已就绪。
204 // R51-3 fix: singleShot lambda 不再捕获 targetRect 值——show() 后布局可能
205 // 导致 target 控件移动/调整大小，捕获旧值会定位到错误位置。改为在 lambda
206 // 内重新从 step.target 计算 targetRect。
207 // ROUND-89 P1 fix: 改用 QPointer 捕获目标 widget，防止 closeEvent 期间
208 // 目标 widget 被析构后 singleShot lambda 访问悬垂指针 → UAF。
209 QPointer<QWidget> targetWidget = step.target;
210 QRect targetRect;
211 if (targetWidget) {
212 // 目标在 host_ 坐标系中的矩形
213 targetRect = QRect(targetWidget->mapTo(host_, QPoint(0, 0)), targetWidget->size());
214 } else {
215 // 无目标：在 host_ 中央显示（零尺寸矩形触发居中逻辑）
216 targetRect = QRect(host_->rect().center(), QSize(0, 0));
217 }
218 positionBubble(targetRect);
219
220 bubble_->show();
221 bubble_->raise();
222
223 // 延迟到事件循环空闲：此时布局已计算完成，adjustSize 得到精确 sizeHint，
224 // 再用新尺寸重新定位气泡，修正首次显示定位偏差。
225 QTimer::singleShot(0, this, [this, targetWidget]() {
226 if (!bubble_)
227 return;
228 bubble_->adjustSize();
229 // R51-3 fix: 重新计算 targetRect，反映 show() 后布局变化
230 QRect rect;
231 if (targetWidget) {
232 rect = QRect(targetWidget->mapTo(host_, QPoint(0, 0)), targetWidget->size());
233 } else {
234 rect = QRect(host_->rect().center(), QSize(0, 0));
235 }
236 positionBubble(rect);
237 });
238
239 emit stepChanged(index);
240}

242// ============================================================
243// positionBubble — 根据剩余空间决定气泡位置
244// 优先级：下方 > 上方 > 右侧 > 左侧 > 兜底（host 底部居中）
245// targetRect 为 host_ 坐标系；最终钳制到 host_ 边界内。
246// ============================================================
247
248/// 依据目标控件矩形计算并定位引导气泡位置。
249void GuidedTour::positionBubble(const QRect& targetRect) {
250 if (!bubble_ || !host_)
251 return;
252
253 const QSize bubbleSize = bubble_->size();
254 const QRect hostRect = host_->rect();
255 const int gap = 8;
256
257 // 无目标（target == nullptr，targetRect 为零尺寸）：host_ 中央显示
258 if (targetRect.isNull()) {
259 int x = hostRect.center().x() - bubbleSize.width() / 2;
260 int y = hostRect.center().y() - bubbleSize.height() / 2;
261 x = qBound(0, x, qMax(0, hostRect.width() - bubbleSize.width()));
262 y = qBound(0, y, qMax(0, hostRect.height() - bubbleSize.height()));
263 bubble_->move(x, y);
264 return;
265 }
266
267 // 默认水平居中于目标，置于目标下方
268 int x = targetRect.center().x() - bubbleSize.width() / 2;
269 int y = targetRect.bottom() + gap;
270 bool placed = false;
271
272 // 1. 下方
273 if (y + bubbleSize.height() <= hostRect.height()) {
274 placed = true;
275 } else {
276 // 2. 上方
277 y = targetRect.top() - gap - bubbleSize.height();
278 if (y >= 0) {
279 placed = true;
280 } else {
281 // 3. 右侧
282 y = targetRect.center().y() - bubbleSize.height() / 2;
283 x = targetRect.right() + gap;
284 if (x + bubbleSize.width() <= hostRect.width()) {
285 placed = true;
286 } else {
287 // 4. 左侧
288 x = targetRect.left() - gap - bubbleSize.width();
289 if (x >= 0) {
290 placed = true;
291 }
292 }
293 }
294 }
295
296 if (!placed) {
297 // 兜底：host 底部居中
298 x = hostRect.center().x() - bubbleSize.width() / 2;
299 y = hostRect.height() - bubbleSize.height() - gap;
300 }
301
302 // 钳制到 host 边界内
303 x = qBound(0, x, qMax(0, hostRect.width() - bubbleSize.width()));
304 y = qBound(0, y, qMax(0, hostRect.height() - bubbleSize.height()));
305
306 bubble_->move(x, y);
307}

350/// 隐藏遮罩与气泡，结束引导态。
351void GuidedTour::hideOverlay() {
352 if (highlightedTarget_) {
353 highlightedTarget_->setStyleSheet(savedStyleSheet_);
354 highlightedTarget_.clear();
355 savedStyleSheet_.clear();
356 }
357 // AUDIT-P0 fix: bubble_ 改为 deleteLater 释放，避免反复 start/hideOverlay 累积泄漏。
358 // 原 hideOverlay 仅 hide() 不 delete，多次触发主引导或面板引导会泄漏 bubble_。
359 // QPointer 保证若 bubble_ 已被 host_ 析构链销毁则此处不再访问。
360 if (bubble_) {
361 bubble_->hide();
362 bubble_->deleteLater();
363 bubble_ = nullptr;
364 }
365 currentIndex_ = -1;
