137// ProfileBarChart — QPainter 自绘柱状图子控件（R111 维度切换修复）
138// ------------------------------------------------------------
139// 原实现在 ProfileDashboardPanel::paintEvent 中直接向面板背景自绘柱状图，
140// 但面板可见区被右侧 QTabWidget 等不透明子控件完全遮挡，柱状图从不可见，
141// 导致「切换维度」看不到任何变化。改为独立子控件占据 timingTab 真实布局
142// 空间，setBars() 更新数据并 update() 重绘，维度切换即时可见。
143// 纯绘制控件，无 Q_OBJECT（不需要信号 / 槽）。
144// ============================================================
145class ProfileBarChart : public QWidget {
146public:
147 struct Bar {
148 QString name; // 后端名
149 double value; // 当前维度数值
150 QString label; // 柱顶数值标签
151 bool success; // 失败后端以 0 高度显示
152 };
153 explicit ProfileBarChart(QWidget* parent = nullptr) : QWidget(parent) { setMinimumHeight(200); }
154 void setBars(std::vector<Bar> bars, const QString& axisTitle) {
155 bars_ = std::move(bars);
156 axisTitle_ = axisTitle;
157 update();
158 }
159
160protected:
161 void paintEvent(QPaintEvent* event) override {
162 (void)event;
163 QPainter p(this);
164 p.setRenderHint(QPainter::Antialiasing);
165
166 const int margin = 20;
167 // 顶部预留 12px 放维度标题
168 QRect chartRect(margin, margin + 12, width() - 2 * margin, height() - 2 * margin - 12);
169 if (chartRect.height() < 50 || chartRect.width() < 40)
170 return;
171 p.setPen(QColor(200, 200, 200));
172 p.drawRect(chartRect);
173
174 // 维度标题
175 p.setPen(QColor(80, 80, 80));
176 p.drawText(chartRect.left(), chartRect.top() - 5, axisTitle_);
177
178 if (bars_.empty())
179 return;
180 double maxVal = 0;
181 for (const auto& b : bars_) {
182 if (b.success && b.value > maxVal)
183 maxVal = b.value;
184 }
185 if (maxVal <= 0)
186 return;
187
188 static const QColor kColors[3] = {
189 QColor(102, 153, 204), // Interpreter 蓝
190 QColor(204, 153, 102), // StackVM 橙
191 QColor(153, 204, 102), // RegisterVM 绿
192 };
193 const int n = static_cast<int>(bars_.size());
194 int slot = chartRect.width() / (n * 2);
195 if (slot < 1)
196 slot = 1;
197 for (int i = 0; i < n; ++i) {
198 const auto& b = bars_[i];
199 double v = b.success ? b.value : 0.0;
200 int barHeight = static_cast<int>(v / maxVal * (chartRect.height() - 30));
201 QRect bar(chartRect.left() + slot / 2 + i * slot * 2, chartRect.bottom() - barHeight - 20, slot, barHeight);
202 p.setBrush(QBrush(kColors[i % 3]));
203 p.setPen(QColor(160, 160, 160));
204 p.drawRect(bar);
205 p.setPen(QColor(0, 0, 0));
206 p.drawText(bar.left(), chartRect.bottom() - 5, b.name);
207 if (b.success && !b.label.isEmpty())
208 p.drawText(bar.left(), bar.top() - 5, b.label);
209 }
210 }
211
212private:
213 std::vector<Bar> bars_;
214 QString axisTitle_;
215};
216#endif

415 // AUDIT-R2 P1-5 fix: 不再 reset 全局 GcManager（会破坏并发执行的 worker

423 double micros = static_cast<double>(detail.elapsedUs);

425}

465 std::vector<double> samples;
466 samples.reserve(iterations);
467 size_t maxTracked = 0;

474 // Bug #75 fix: 让出 UI 事件循环，避免内层测量循环阻塞界面

477 t.avgMicros = mean(samples);
478 t.stddevMicros = stddev(samples);
479 t.peakTrackedCount = maxTracked;
480 t.success = true;
481 } catch (const std::exception& e) {
482 t.success = false;
483 t.errorMessage = e.what();
484 }
485 return t;
486}

515/// 对指定场景在三后端上分别测速并收集计时结果。
516void ProfileDashboardPanel::runProfile(int scenarioIndex) {
517 const auto& items = ProfileLibrary::scenarios();
518 if (scenarioIndex < 0 || scenarioIndex >= (int)items.size())
519 return;
520 // AUDIT-P1 fix: 重入守卫——processEvents(ExcludeUserInputEvents) 期间定时器/信号槽
521 // 可能触发间接调用 runProfile，导致 lastResults_ 被覆盖、状态混乱。
522 if (profiling_)
523 return;
524 profiling_ = true;
525 const auto& scenario = items[scenarioIndex];
526
527 runProfileBtn_->setEnabled(false);
528 // 问题 6: 醒目的"运行中"状态 — 橙色背景 + 动画圆点 + 进度
529 startStatusAnimation(QString::fromUtf8("运行中 [1/3] Interpreter"));
530 statusLabel_->setStyleSheet("QLabel { background: #CB4B16; color: white; border-radius: 4px;"
531 " padding: 4px 12px; font-weight: bold; }");
532 // BUG-GUI-AUDIT-1 fix attempt: Qt 6 已移除通用 ExcludeTimers flag，保持 ExcludeUserInputEvents。
533 QApplication::processEvents(QEventLoop::ExcludeUserInputEvents);
534 // ROUND-76 fix: ExcludeUserInputEvents 不排除 QCloseEvent！
535 // 若用户在 processEvents 期间关闭窗口，closeEvent 已在此同步执行并 setClosing(true)。
536 // 必须立即退出，避免后续访问正在关闭的 widget。
537 if (closing_) {
538 stopStatusAnimation();
539 profiling_ = false;
540 return;
541 }
542
543 // 先 lex + parse 源码（仅用于早期错误检查；测量走 BackendExecutionService 重新执行）
544 Lexer lexer;
545 auto tokens = lexer.scan(scenario.sourceCode);
546 Parser parser;
547 auto ast = parser.parse(tokens);
548 if (parser.getDiagnostics().hasErrors()) {
549 stopStatusAnimation();
550 statusLabel_->setStyleSheet("");
551 statusLabel_->setText(
552 QString::fromUtf8("解析失败：%1").arg(QString::fromUtf8(parser.getDiagnostics().summary().c_str())));
553 runProfileBtn_->setEnabled(true);
554 profiling_ = false; // AUDIT-P1 fix: 错误返回点恢复守卫
555 return;
556 }
557 (void)ast; // ARCH-10: ast 仅用于 parse 错误检查，测量阶段直接传 sourceCode
558
559 // 多次测量三后端
560 std::vector<BackendTiming> results;
561 results.push_back(measureBackend(
562 "Interpreter", [this](const std::string& s) { return measureInterpreterOnce(s); }, scenario.sourceCode,
563 scenario.iterations));
564
565 // 问题 7: 在后端之间处理事件，避免长时间阻塞 UI
566 statusRunningBase_ = QString::fromUtf8("运行中 [2/3] StackVM");
567 QApplication::processEvents(QEventLoop::ExcludeUserInputEvents);
568 // ROUND-76 fix: 检查 closing_ 避免在窗口关闭后继续执行
569 if (closing_) {
570 stopStatusAnimation();
571 profiling_ = false;
572 return;
573 }
574
575 // P1-1: StackVM / RegisterVM 走 measureXxxWithProfile，同时累加 opcode 计数
576 // R77 fix: 指令计数取 N 次平均（与时间统计 mean(samples) 一致），否则显示值为
577 // 单次实际值的 N 倍（iterations=3 时 OP_CALL 等热点被放大 3 倍），与 OpCode
578 // 性能文档库的参考量级（如"fib(20) OP_CALL ~21891 次"）矛盾。
579 // R111: 同时累加 GC tracked 峰值（取 max）用于内存维度
580 std::array<uint64_t, 256> stackVmCounts{};
581 std::array<uint64_t, 256> registerVmCounts{};
582 {
583 BackendTiming t;
584 t.name = "StackVM";
585 std::vector<double> samples;
586 samples.reserve(scenario.iterations);
587 size_t maxTracked = 0;
588 try {
589 for (int i = 0; i < scenario.iterations; ++i) {
590 auto [us, counts, tracked] = measureStackVMWithProfile(scenario.sourceCode);
591 samples.push_back(us);
592 for (size_t j = 0; j < 256; ++j)
593 stackVmCounts[j] += counts[j];
594 if (tracked > maxTracked)
595 maxTracked = tracked;
596 }
597 // 取平均：指令计数是确定值，取平均消除潜在非确定性，与 avgMicros 口径一致
598 if (scenario.iterations > 0) {
599 for (size_t j = 0; j < 256; ++j)
600 stackVmCounts[j] /= static_cast<uint64_t>(scenario.iterations);
601 }
602 t.avgMicros = mean(samples);
603 t.stddevMicros = stddev(samples);
604 t.peakTrackedCount = maxTracked;
605 // R111: 计算指令总数（sum of opcode counts）
606 uint64_t total = 0;
607 for (size_t j = 0; j < 256; ++j)
608 total += stackVmCounts[j];
609 t.totalInstructions = total;
610 t.success = true;
611 } catch (const std::exception& e) {
612 t.success = false;
613 t.errorMessage = e.what();
614 }
615 results.push_back(t);
616 }
617
618 statusRunningBase_ = QString::fromUtf8("运行中 [3/3] RegisterVM");
619 QApplication::processEvents(QEventLoop::ExcludeUserInputEvents);
620 // ROUND-76 fix: 检查 closing_ 避免在窗口关闭后继续执行
621 if (closing_) {
622 stopStatusAnimation();
623 profiling_ = false;
624 return;
625 }
626
627 {
628 BackendTiming t;
629 t.name = "RegisterVM";
630 std::vector<double> samples;
631 samples.reserve(scenario.iterations);
632 size_t maxTracked = 0;
633 try {
634 for (int i = 0; i < scenario.iterations; ++i) {
635 auto [us, counts, tracked] = measureRegisterVMWithProfile(scenario.sourceCode);
636 samples.push_back(us);
637 for (size_t j = 0; j < 256; ++j)
638 registerVmCounts[j] += counts[j];
639 if (tracked > maxTracked)
640 maxTracked = tracked;
641 }
642 // R77 fix: 取平均，与 StackVM 口径一致
643 if (scenario.iterations > 0) {
644 for (size_t j = 0; j < 256; ++j)
645 registerVmCounts[j] /= static_cast<uint64_t>(scenario.iterations);
646 }
647 t.avgMicros = mean(samples);
648 t.stddevMicros = stddev(samples);
649 t.peakTrackedCount = maxTracked;
650 // R111: 计算指令总数
651 uint64_t total = 0;
652 for (size_t j = 0; j < 256; ++j)
653 total += registerVmCounts[j];
654 t.totalInstructions = total;
655 t.success = true;
656 } catch (const std::exception& e) {
657 t.success = false;
658 t.errorMessage = e.what();
659 }
660 results.push_back(t);
661 }
662
663 lastResults_ = results;
664
665 // P1-1: 聚合 opcode 计数为 Top N 热点列表（按计数降序）
666 auto aggregateTop10 = [](const std::array<uint64_t, 256>& counts,
667 bool isRegisterVm) -> std::vector<OpCodeProfileEntry> {
668 std::vector<OpCodeProfileEntry> all;
669 uint64_t total = 0;
670 for (size_t i = 0; i < 256; ++i) {
671 if (counts[i] > 0) {
672 std::string name;
673 if (isRegisterVm) {
674 name = regOpName(static_cast<RegOp>(i));
675 } else {
676 name = opCodeName(static_cast<OpCode>(i));
677 }
678 all.push_back({name, counts[i], 0.0});
679 total += counts[i];
680 }
681 }
682 std::sort(all.begin(), all.end(),
683 [](const OpCodeProfileEntry& a, const OpCodeProfileEntry& b) { return a.count > b.count; });
684 if (all.size() > 10)
685 all.resize(10);
686 if (total > 0) {
687 for (auto& e : all)
688 e.ratio = static_cast<double>(e.count) / static_cast<double>(total);
689 }
690 return all;
691 };
692 lastStackVMOpProfile_ = aggregateTop10(stackVmCounts, false);
693 lastRegisterVMOpProfile_ = aggregateTop10(registerVmCounts, true);
694
695 renderResults(results, scenario);
696 renderOpCodeProfile(lastStackVMOpProfile_, lastRegisterVMOpProfile_);
697#ifdef MINILANG_HAVE_QTCHARTS
698 renderChart(results); // C2: QtCharts 模式刷新柱状图
699#else
700 updateBarChart(); // R111 fix: 刷新独立柱状图子控件
701#endif
702 runProfileBtn_->setEnabled(true);
703 stopStatusAnimation();
704 statusLabel_->setStyleSheet("");
705 statusLabel_->setText(QString::fromUtf8("剖析完成"));
706 profiling_ = false; // AUDIT-P1 fix: 正常结束点恢复守卫
707}

772/// 根据测速结果生成中文性能分析结论文本。
773QString ProfileDashboardPanel::buildAnalysis(const std::vector<BackendTiming>& results,
774 const ProfileScenario& scenario) {
775 std::ostringstream os;
776 os << "<h3>" << scenario.title << "</h3>";
777 os << "<p><i>[" << scenario.category << "]</i> " << scenario.iterations << " 次迭代平均</p>";
778 os << "<p>" << scenario.description << "</p>";
779 os << "<hr>";
780
781 // 找出最快后端
782 const BackendTiming* fastest = nullptr;
783 for (const auto& r : results) {
784 if (r.success && (!fastest || r.avgMicros < fastest->avgMicros)) {
785 fastest = &r;
786 }
787 }
788 if (!fastest) {
789 os << "<p style='color:#cc0000;'><b>所有后端均失败</b></p>";
790 } else {
791 os << "<p><b>最快后端：</b>" << fastest->name << " (" << fastest->avgMicros << " μs)</p>";
792 // 计算与最慢的比值
793 const BackendTiming* slowest = nullptr;
794 for (const auto& r : results) {
795 if (r.success && (!slowest || r.avgMicros > slowest->avgMicros)) {
796 slowest = &r;
797 }
798 }
799 if (slowest && slowest != fastest) {
800 double speedup = slowest->avgMicros / fastest->avgMicros;
801 os << "<p><b>最慢后端：</b>" << slowest->name << " (" << slowest->avgMicros << " μs)</p>";
802 os << "<p><b>加速比：</b>" << speedup << "x</p>";
803 }
804 }
805 // 问题 7: 列出失败后端及其错误原因
806 for (const auto& r : results) {
807 if (!r.success) {
808 // R52-7 fix: errorMessage 来自 e.what()，可能含 < > & 字符（如 "expected <expression>"）需转义
809 os << "<p style='color:#cc0000;'><b>" << r.name << " 失败：</b>"
810 << QString::fromUtf8(r.errorMessage.c_str()).toHtmlEscaped().toStdString() << "</p>";
811 }
812 }
813 os << "<hr>";
814 os << "<p><small>注：标准差反映多次运行的稳定性。"
815 << "若标准差 > 10% 平均值，可能是 GC 周期或系统调度影响。</small></p>";
816 return QString::fromUtf8(os.str().c_str());
817}

973/// 计算样本标准差，衡量后端耗时稳定性。
974double ProfileDashboardPanel::stddev(const std::vector<double>& xs) {
975 if (xs.size() < 2)
976 return 0.0;
977 double m = mean(xs);
978 double sq = 0;
979 for (double x : xs)
980 sq += (x - m) * (x - m);
981 return std::sqrt(sq / (xs.size() - 1));
