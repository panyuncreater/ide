64UnrollSimResult LoopUnrollingLibrary::simulate(int loopCount, int factor, int bodyInstrCount) {
65 UnrollSimResult r;
66 r.factor = factor;
67 // 边界保护
68 if (loopCount <= 0 || factor <= 0 || bodyInstrCount < 0) {
69 r.iterations = 0;
70 r.jumps = 0;
71 r.controlInstrs = 0;
72 r.bodyInstrs = 0;
73 r.totalInstrs = 0;
74 r.speedup = 0.0;
75 return r;
76 }
77 // 展开后迭代次数 = ceil(N/K)
78 r.iterations = (loopCount + factor - 1) / factor;
79 r.jumps = r.iterations;
80 // 循环控制指令 = LOOP + JUMP_IF_FALSE，每次迭代 2 条
81 r.controlInstrs = r.iterations * 2;
82 // 循环体指令 = N * bodyInstrCount（展开不改变循环体总指令数）
83 r.bodyInstrs = loopCount * bodyInstrCount;
84 r.totalInstrs = r.controlInstrs + r.bodyInstrs;
85 // 原始总指令 = N * (2 + bodyInstrCount)
86 int originalTotal = loopCount * (2 + bodyInstrCount);
87 r.speedup = static_cast<double>(originalTotal) / static_cast<double>(r.totalInstrs);
88 return r;
89}

91std::string LoopUnrollingLibrary::generateUnrolledCode(int loopCount, int factor, const std::string& body) {
92 std::ostringstream oss;
93 // 边界保护
94 if (loopCount <= 0 || factor <= 0) {
95 oss << "// 无效参数：loopCount=" << loopCount << " factor=" << factor << "\n";
96 return oss.str();
97 }
98 int iters = (loopCount + factor - 1) / factor;
99 int remainder = loopCount % factor;
100 oss << "// 展开因子 K=" << factor << "，循环次数 N=" << loopCount << "\n";
101 oss << "// 展开后迭代次数 = ceil(" << loopCount << "/" << factor << ") = " << iters << "\n";
102 if (remainder != 0) {
103 oss << "// 注意：N 不被 K 整除，尾部剩余 " << remainder << " 次迭代需单独处理\n";
104 oss << "// （此处省略尾部处理，实际编译器需补足剩余循环或守卫分支）\n";
105 }
106 oss << "var i = 0;\n";
107 oss << "var sum = 0;\n";
108 oss << "while (i < " << loopCount << ") {\n";
109 for (int k = 0; k < factor; ++k) {
110 oss << " " << body << " i = i + 1; // 第 " << (k + 1) << " 次循环体\n";
111 }
112 oss << "}\n";
113 oss << "print(sum);\n";
114 return oss.str();
115}
