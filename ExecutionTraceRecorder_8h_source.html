3// ============================================================
4// ExecutionTraceRecorder — 可回放执行时间轴核心抽象（R114 阶段 1+2+3）
5// ------------------------------------------------------------
6// 三后端统一的执行轨迹记录器。订阅 VmStepper 的步进事件与
7// Interpreter 的 checkBreak 钩子，每步采集 TraceSnapshot 写入
8// 环形缓冲区。GUI 时间轴 widget 通过 stepAt(i) 随机访问任意步。
9//
10// 三阶段能力：
11// 阶段 1：单步快照采集（栈/locals/globals/callStack 字符串化）
12// 阶段 2：三后端统一 + AST 节点路径 + Value JSON 序列化
13// 阶段 3：状态深拷贝（Environment/VM 帧/Value 副本）支持回滚
14//
15// 线程模型：
16// - VM 路径在主线程调用（VmStepper 的 QTimer + UI 槽）
17// - Interpreter 路径在 worker 线程调用（checkBreak 内）
18// - 内部 mutex 保护 deque，跨线程 push 主线程 read
19//
20// 容量策略：
21// - 默认上限 5000 步（教学场景足够，避免内存膨胀）
22// - 超限后采用 ring buffer 策略：pop_front + push_back
23// - 用户可调整容量（如长循环场景可调至 20000）
24// ============================================================

51/// 单步快照——三后端统一格式
52struct TraceSnapshot {

59 size_t ip = 0; // VM 模式 IP；Interpreter 模式 0
60 size_t frameCount = 0; // 帧深度
61

63 std::vector<std::string> stackStrings; // 操作数栈顶→底
64 std::vector<std::string> registerStrings; // 寄存器窗口（仅 RegisterVM）
65 std::vector<CallStackEntry> callStack; // 含每帧 locals
66 std::vector<std::pair<std::string, std::string>> globalsStrings; // 全局变量
67 std::vector<std::pair<std::string, std::string>> visibleVarsStrings; // 当前可见变量
68
69 // ---- 结构化快照（阶段 3：状态回滚）----
70 // 仅在 recordingMode_ == FullState 时填充，否则为空
71 std::vector<Value> stackValues; // 操作数栈值深拷贝
72 std::vector<Value> registerValues; // 寄存器值深拷贝
73 std::vector<std::pair<std::string, Value>> globalsValues; // 全局变量值
74 std::vector<std::pair<std::string, Value>> visibleVarsValues; // 可见变量值
75 // L18: Interpreter 后端完整状态快照（类型擦除，避免头文件循环依赖）。
76 // 实际类型为 std::shared_ptr<Interpreter::StateSnapshot>。
77 // 仅 Interpreter 后端 + FullState 模式填充；VM 后端为空。
78 // 回滚时 VmStepper/DebugController 通过 Interpreter::restoreFromSnapshot 消费。
79 std::shared_ptr<void> interpreterState;
80 // 注：Environment 链与 VM 帧的深拷贝由 PlaybackController 在回滚时按需重建，

82 // L18 更新：Interpreter 后端例外——interpreterState 持有完整环境链快照，
83 // 因为 Interpreter 无 IP/帧概念，必须依赖完整状态快照回滚（而非重建）
84
85 /// JSON 序列化（阶段 2）——用于持久化/跨语言交换
86 std::string toJson() const;
87};

156 /// 获取最后一步步号（用于 UI 显示"当前步 / 总步数"）
157 size_t lastStep() const {
158 std::lock_guard<std::mutex> lock(mutex_);
159 return snapshots_.empty() ? 0 : snapshots_.back().step;
160 }
161
162 // ---- 控制 ----
163 /// 拓展二期：丢弃 idx 及之后的全部快照（回溯调试 stepBack 用：
164 /// 回滚到历史步后丢弃"未来"步，使连续 stepBack 可持续后退，
165 /// 且后续重新执行的录制不与旧轨迹混杂）。
166 void truncateFrom(size_t idx) {
167 std::lock_guard<std::mutex> lock(mutex_);
168 if (idx < snapshots_.size()) {
169 snapshots_.erase(snapshots_.begin() + static_cast<std::ptrdiff_t>(idx), snapshots_.end());
170 }
171 }
172
173 /// 清空所有快照（重置录制会话）
174 void clear() {
175 std::lock_guard<std::mutex> lock(mutex_);
176 snapshots_.clear();
177 stepCounter_ = 0;
178 }
179
180 /// 开始新录制会话（清空 + 重置步号）
181 void startSession() {
182 clear();
183 enabled_ = true;
184 }
185
186 /// 结束录制会话（禁用录制但保留快照）
187 void endSession() { enabled_ = false; }
188

192 size_t stepCounter_ = 0;
193 size_t maxSteps_ = 5000;
194 RecordingMode recordingMode_ = RecordingMode::StringsOnly;
195 std::atomic<bool> enabled_{false};
196
