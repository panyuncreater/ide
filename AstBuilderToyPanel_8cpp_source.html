321void AstBuilderToyPanel::addNodeWithLabel(const QString& label) {
322 QTreeWidgetItem* current = tree_->currentItem();
323 if (!current) {
324 // 无选中项
325 if (tree_->topLevelItemCount() == 0) {
326 // 树空 → 添加为根
327 auto* item = new QTreeWidgetItem(tree_);
328 item->setText(0, label);
329 tree_->addTopLevelItem(item);
330 tree_->setCurrentItem(item);
331 feedback_->setText(QString::fromUtf8("✓ 已添加根节点：%1").arg(label));
332 } else {
333 feedback_->setText(mlTr("⚠ 请先选中一个节点作为父节点，或点击「清空」重新开始"));
334 }
335 return;
336 }
337 // 有选中项 → 添加为子节点
338 auto* child = new QTreeWidgetItem(current);
339 child->setText(0, label);
340 current->addChild(child);
341 tree_->expandItem(current);
342 // 叶子节点（Number:* / Identifier:*）添加后保持父节点选中，
343 // 避免连续添加叶子时形成错误嵌套（下一个叶子变成上一个叶子的子节点）。
344 // 非叶子节点（BinaryOp:* / Print / VarDecl:*）添加后切换到新节点，
345 // 方便为其继续添加子节点。
346 bool isLeaf = label.startsWith(QString::fromUtf8("Number:")) || label.startsWith(QString::fromUtf8("Identifier:"));
347 if (isLeaf) {
348 tree_->setCurrentItem(current);
349 } else {
350 tree_->setCurrentItem(child);
351 }
352 feedback_->setText(QString::fromUtf8("✓ 已添加子节点：%1（父节点：%2）").arg(label).arg(current->text(0)));
353}

370void AstBuilderToyPanel::onCheck() {
371 if (tree_->topLevelItemCount() == 0) {
372 feedback_->setText(mlTr("❌ 画布为空，请先搭建 AST"));
373 return;
374 }
375 if (tree_->topLevelItemCount() > 1) {
376 feedback_->setText(mlTr("❌ 画布有多个根节点，AST 只能有 1 个根"));
377 return;
378 }
379
380 AstToyLevel::TreeNode userTree;
381 if (!treeToTreeNode(userTree)) {
382 feedback_->setText(mlTr("❌ 无法读取你搭建的树"));
383 return;
384 }
385
386 const auto& ls = AstToyLibrary::levels();
387 if (currentIndex_ < 0 || currentIndex_ >= (int)ls.size())
388 return;
389 const auto& ans = ls[currentIndex_].answer;
390
391 if (treeEquals(userTree, ans)) {
392 completed_ = true;
393 feedback_->setText(mlTr("🎉 完全正确！拓扑结构与标准答案一致。"));
394 markCurrentCompleted();
395 } else {
396 QString userStr, ansStr;
397 treeNodeToString(userTree, userStr);
398 treeNodeToString(ans, ansStr);
399 feedback_->setText(QString::fromUtf8("❌ 不匹配。\n\n你的树：\n%1\n\n标准答案：\n%2").arg(userStr).arg(ansStr));
400 }

403void AstBuilderToyPanel::markCurrentCompleted() {
404 const auto& ls = AstToyLibrary::levels();
405 if (currentIndex_ < 0 || currentIndex_ >= (int)ls.size())
406 return;
407 int lvl = ls[currentIndex_].level;
408 // AUDIT-P2 fix: 幂等守卫完整覆盖——原实现只保护 completedLevels_.push_back，
409 // 未保护 markLevelStars/save/emit。重复检查会重复发射信号 + 重复磁盘写入。
410 bool alreadyDone = std::find(completedLevels_.begin(), completedLevels_.end(), lvl) != completedLevels_.end();
411 if (!alreadyDone) {
412 completedLevels_.push_back(lvl);
413 refreshProgress();
414 // 仅首次完成时持久化 + 发射信号
415 std::string id = "ast-toy-level-" + std::to_string(lvl);
416 LearnerProgressStore::instance().markLevelStars(id, 3);
417 saveLearnerProgressWithFeedback(this); // P2-UX fix: 失败时弹 toast 避免静默丢失
418 QString levelId = QString::fromUtf8("ast-toy-level-%1").arg(lvl);
419 emit activityCompleted(levelId);
420 }
421 refreshLevelChips(); // 完成后刷新芯片（显示已完成标记）
422}

424// ============================================================
425// 关卡芯片栏状态刷新
426// ------------------------------------------------------------
427// 所有题目均解锁（无解锁机制），芯片仅区分 current / unlocked
428// 已完成的题目在芯片上显示 ⭐ 标记
429// dynamic property 改变后必须 style()->polish() 才能让 QSS 重新评估
430// ============================================================
431
432void AstBuilderToyPanel::refreshLevelChips() {
433 const auto& levels = AstToyLibrary::levels();
434 for (int i = 0; i < levelChips_.size(); ++i) {
435 QPushButton* chip = levelChips_[i];
436 if (!chip)
437 continue;
438
439 bool isCurrent = (i == currentIndex_);
440 bool isCompleted = false;
441 if (i < (int)levels.size()) {
442 int lvl = levels[i].level;
443 isCompleted = std::find(completedLevels_.begin(), completedLevels_.end(), lvl) != completedLevels_.end();
444 }
445
446 // 本面板所有题目均解锁，locked 恒为 false
447 chip->setProperty("locked", false);
448 chip->setProperty("current", isCurrent);
449 chip->setEnabled(true);
450
451 // 文本：当前选中且已完成显示关卡号+⭐；其余显示关卡号
452 QString text;
453 if (isCurrent && isCompleted) {
454 // ⭐ = U+2B50 = UTF-8: E2 AD 90
455 text = QString::fromUtf8("%1 \xE2\xAD\x90").arg(i + 1);
456 } else {
457 text = QString::number(i + 1);
458 }
459 chip->setText(text);
460
461 // 工具提示：显示题目编号、目标表达式和教学点
462 if (i < (int)levels.size()) {
463 const auto& lv = levels[i];
464 QString tip = mlTr("题目 %1").arg(lv.level);
465 tip += "\n" + mlTr("目标：") + QString::fromUtf8(lv.targetExpression.c_str());
466 if (!lv.teachingPoint.empty()) {
467 tip += "\n" + mlTr("教学点：") + QString::fromUtf8(lv.teachingPoint.c_str());
468 }
469 if (isCompleted) {
470 tip += "\n" + mlTr("已完成");
471 }
472 chip->setToolTip(tip);
473 }
474
475 // 强制 QSS 重新评估 dynamic property 选择器
476 chip->style()->unpolish(chip);
477 chip->style()->polish(chip);
478 }
479}

482// P1-3 fix (F14): 用真实 Lexer + Parser 验证
483// 解析 targetExpression，把真实 AST 树 dump 到 feedback 区
484// 供学员对比"自己搭建的树"与"真实编译器生成的树"
485// ============================================================
486void AstBuilderToyPanel::onVerifyWithRealParser() {
487 const auto& ls = AstToyLibrary::levels();
488 if (currentIndex_ < 0 || currentIndex_ >= (int)ls.size()) {
489 feedback_->setText(mlTr("❌ 无有效题目"));
490 return;
491 }
492 const auto& lvl = ls[currentIndex_];
493 // 包装成完整语句：纯表达式加 var __toy_tmp = 前缀，已是语句的保持原样
494 std::string expr = lvl.targetExpression;
495 std::string source;
496 if (expr.empty()) {
497 feedback_->setText(mlTr("❌ 目标表达式为空"));
498 return;
499 }
500 // 检测是否已是完整语句（含 ; 或 var/print 关键字开头）
501 bool isStatement = (expr.back() == ';') || expr.find("var ") == 0 || expr.find("print(") == 0;
502 if (isStatement) {
503 source = expr;
504 if (source.back() != ';')
505 source += ";";
506 } else {
507 // 包装成 var _toy_tmp = <expr>; 让 Parser 接受
508 // ROUND56 fix: 原使用 __toy_tmp 触发 Lexer 的保留前缀 '__' 拒绝
509 // （#10 fix: __ 前缀保留给编译器内部使用，如 __blk_save_*、__wb_idx_*）。
510 // 改用单下划线前缀 _toy_tmp，既不与用户变量冲突也不触发保留检查。
511 source = "var _toy_tmp = " + expr + ";";
512 }
513
514 // 调用真实 Lexer + Parser
515 Lexer lexer;
516 auto tokens = lexer.scan(source);
517 if (lexer.getDiagnostics().hasErrors()) {
518 std::string errs;
519 for (const auto& d : lexer.getDiagnostics().all()) {
520 if (d.isError())
521 errs += d.format() + "\n";
522 }
523 feedback_->setText(QString::fromUtf8("❌ ") + mlTr("词法错误：\n") + QString::fromUtf8(errs.c_str()));
524 return;
525 }
526
527 Parser parser;
528 auto ast = parser.parse(tokens);
529 if (parser.hasErrors()) {
530 std::string errs;
531 for (const auto& d : parser.getDiagnostics().all()) {
532 if (d.isError())
533 errs += d.format() + "\n";
534 }
535 feedback_->setText(QString::fromUtf8("❌ ") + mlTr("解析错误：\n") + QString::fromUtf8(errs.c_str()));
536 return;
537 }
538 if (!ast) {
539 feedback_->setText(mlTr("❌ Parser 返回空 AST"));
540 return;
541 }
542
543 // dump AST 树形结构（复用 PipelineViewer 的 dumpAst 思路）
544 std::ostringstream os;
545 os << "🛠 " << mlTr("真实 Parser 生成的 AST：").toStdString() << "\n";
546 os << "源码: " << source << "\n";
547 // AUDIT-P2 fix: 纯表达式被包装为 var _toy_tmp = <expr>; 才能被语句级 Parser
548 // 接受。此时真实 AST 根节点是 Block → VarDecl，而学员搭建的是裸表达式树。
549 // 明确标注包装行为，避免学员误以为自己搭建的树错误。
550 if (!isStatement) {
551 os << "💡 "
552 << mlTr("注：纯表达式已包装为 var _toy_tmp = <expr>; 才能解析。"
553 "下方 AST 根节点 Block/VarDecl 是包装语句，"
554 "VarDecl 的子树才是目标表达式")
555 .toStdString()
556 << "\n";
557 }
558 os << "--- AST 树形结构 ---\n";
559 // 递归 dump
560 std::function<void(ASTNode*, int)> dumpRec = [&](ASTNode* node, int depth) {
561 if (!node)
562 return;
563 for (int i = 0; i < depth; ++i)
564 os << " ";
565 os << node->nodeName();
566 if (node->line > 0)
567 os << " [line " << node->line << "]";
568 os << "\n";
569 for (auto* child : node->children()) {
570 dumpRec(child, depth + 1);
571 }
572 };
573 dumpRec(ast.get(), 0);
574 os << "\n💡 "
575 << mlTr("对比你搭建的树与上方真实 AST，理解优先级与括号如何"
576 "影响树形结构")
577 .toStdString();
578 feedback_->setText(QString::fromUtf8(os.str().c_str()));
579}
