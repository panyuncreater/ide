44// 判断是否为 IR opcode（大写字母+下划线，至少 2 字符）
45static bool isIROpcode(const std::string& tok) {
46 if (tok.size() < 2)
47 return false;
48 for (char c : tok) {
49 if (!((c >= 'A' && c <= 'Z') || c == '_'))
50 return false;
51 }
52 return true;
53}
54
55// 判断是否为寄存器/局部变量（v 数字 或 t 数字）
56static bool isRegister(const std::string& tok) {
57 if (tok.size() < 2)
58 return false;
59 if (tok[0] != 'v' && tok[0] != 't')
60 return false;
61 for (size_t i = 1; i < tok.size(); ++i) {
62 if (tok[i] < '0' || tok[i] > '9')
63 return false;
64 }
65 return true;
66}
67
68// 判断是否为数字常量
69// BUG-IRV-5 fix: 收紧规则——仅允许首字符为 '-'，避免 "1-2" 等非法字面量被误识别。
70static bool isNumericConst(const std::string& tok) {
71 if (tok.empty())
72 return false;
73 bool hasDigit = false;
74 for (size_t i = 0; i < tok.size(); ++i) {
75 char c = tok[i];
76 if (c >= '0' && c <= '9') {
77 hasDigit = true;
78 continue;
79 }
80 // 仅允许首字符为 '-'（负号），其余位置的 '-' 视为非法
81 if (i == 0 && c == '-')
82 continue;
83 if (c == '.')
84 continue;
85 return false;
86 }
87 return hasDigit;
88}

90// 判断是否为字符串字面量（含引号）
91static bool isStringLiteral(const std::string& tok) {
92 return !tok.empty() && tok.front() == '"' && tok.back() == '"';
93}

95/// 将单行 IR 指令按 token 类型着色为 HTML 片段。
96QString IrViewer::formatIRLineHtml(const std::string& text) const {
97 // 分离注释（; 开头到行尾）
98 std::string code = text;
99 std::string comment;
100 size_t semiPos = code.find(';');
101 if (semiPos != std::string::npos) {
102 comment = code.substr(semiPos);
103 code = code.substr(0, semiPos);
104 }
105
106 // 按空白拆分 token
107 std::vector<std::string> tokens;
108 std::istringstream iss(code);
109 std::string tok;
110 while (iss >> tok) {
111 tokens.push_back(tok);
112 }
113
114 QString html;
115 bool firstToken = true;
116 for (const auto& t : tokens) {
117 if (!firstToken)
118 html += "&nbsp;";
119 firstToken = false;
120
121 // 去除尾随逗号
122 std::string core = t;
123 std::string suffix;
124 if (!core.empty() && core.back() == ',') {
125 suffix = ",";
126 core.pop_back();
127 }
128
129 QString esc = htmlEscape(core);
130 if (isIROpcode(core)) {
131 html += "<span style=\"color:#268BD2;font-weight:bold;\">" + esc + "</span>";
132 } else if (isRegister(core)) {
133 html += "<span style=\"color:#2AA198;\">" + esc + "</span>";
134 } else if (isStringLiteral(core)) {
135 html += "<span style=\"color:#CB4B16;\">" + esc + "</span>";
136 } else if (isNumericConst(core)) {
137 html += "<span style=\"color:#CB4B16;\">" + esc + "</span>";
138 } else if (core == "=" || core == "->" || core == "|" || core == "&") {
139 html += "<span style=\"color:#8C8C8C;\">" + esc + "</span>";
140 } else {
141 // 标识符（函数名、标签等）默认色
142 html += "<span style=\"color:#1E1E1E;\">" + esc + "</span>";
143 }
144 if (!suffix.empty()) {
145 html += "<span style=\"color:#8C8C8C;\">" + QString::fromStdString(suffix).toHtmlEscaped() + "</span>";
146 }
147 }
148
149 if (!comment.empty()) {
150 if (!html.isEmpty())
151 html += "&nbsp;";
152 html += "<span style=\"color:#5A5A5A;font-style:italic;\">" + htmlEscape(comment) + "</span>";
153 }
154
155 return html;
156}
