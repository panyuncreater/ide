94ReplPanel::~ReplPanel() {
95 if (pollTimer_)
96 pollTimer_->stop();
97 // 深度防御：closeEvent 路径已通过 waitReplFuture() 处理，此处为 no-op。
98 // 非 closeEvent 路径（如直接 delete）controller_ 可能已析构，不能调用
99 // requestReplStop()（C++ 异常不能捕获 UAF），直接 wait() 阻塞至完成
100 // （由 MAX_LOOP_ITERATIONS 兜底）。
101 // BUG-REPL-G5 fix (P2): 原实现直接 wait() 无超时，若 closeEvent 路径异常
102 // 未调用 waitReplFuture() 且异步任务陷入死循环（且未触发 MAX_LOOP_ITERATIONS），
103 // 析构会无限阻塞。改为 wait_for(5s) + 日志告警，超时后仍回退阻塞 wait()
104 // 作为最后手段（进程即将退出，由 OS 兜底）。controller_ 已可能析构，不调用
105 // requestReplStop()，仅等待异步任务自然完成或被外部中止。
106 if (replFuture_.valid()) {
107 auto status = replFuture_.wait_for(std::chrono::seconds(5));
108 if (status != std::future_status::ready) {
109 // 超时未完成：记录告警后回退阻塞等待（避免 future 析构时 anyway 阻塞）
110 LOG_WARNING("REPL 析构等待异步任务超时（5s），回退阻塞等待", "REPL");
111 replFuture_.wait();
112 }
113 }
114}

116/// 阻塞等待当前 REPL 异步任务结束，避免重复执行。
117void ReplPanel::waitReplFuture() {
118 if (!replFuture_.valid())
119 return;
120 // REPL-TIMEOUT fix: 协作中止 + 超时等待，避免 closeEvent 永久阻塞。
121 // 原 wait() 无超时，死循环场景下 closeEvent 卡死。现改为：
122 // 1. 设置 stopRequested_ 标志（checkBreak 在每个语句节点检查并抛异常）
123 // 2. wait_for(5s) 等待协作中止生效
124 // 3. 超时则强制进程退出（IDE-CLOSE-01 fix: 原 replFuture_.wait() 无超时阻塞
125 // 会导致 closeEvent 永远不返回，IDE 无法正常关闭。改为 std::_Exit(1)
126 // 强制进程退出，与 forceStop 的 std::_Exit(0) 语义一致）
127 if (controller_) {
128 controller_->requestReplStop();
129 auto status = replFuture_.wait_for(std::chrono::seconds(5));
130 if (status == std::future_status::ready)
131 return;
132 // AUDIT-BUG-C8 fix: 改用 LOG_* 宏，先检查级别再构造消息（懒求值）。
133 LOG_WARNING("REPL 异步任务未在 5 秒内响应中止请求，强制退出进程", "REPL");
134 }
135 // IDE-CLOSE-01 fix: 超时后强制进程退出，避免 closeEvent 卡死。
136 // replFuture_ 析构时若 future 未 ready 会阻塞 wait()，无法安全返回。
137 std::_Exit(1);
138}

180/// 回车回调：按输入完整性决定立即执行或继续多行输入。
181void ReplPanel::onReturnPressed() {
182 // AUDIT-REPL-7 fix: 在任何状态变更前检查异步执行状态。
183 // 原实现 executeLine 内部检查后仍清空 pendingInput_，导致续行累积输入丢失。
184 // 改为入口拒绝并保留 inputLine_ 内容，用户可调整输入或等待。
185 if (replRunning_) {
186 appendError("上一次执行尚未完成，请稍候...");
187 return;
188 }
189
190 QString line = inputLine_->toPlainText();
191 QString trimmedLine = line.trimmed();
192
193 // AUDIT-REPL-3 fix: 续行模式下空行中止续行（等价 Ctrl+C 中断）。
194 // 用户误触续行（如多打 `{`）时无需闭合，直接回车即可丢弃并重启。
195 if (inContinuation_ && trimmedLine.isEmpty()) {
196 appendOutput("[续行已中止]");
197 pendingInput_.clear();
198 inContinuation_ = false;
199 inputLine_->clear();
200 return;
201 }
202
203 // R4: 续行模式 — 累积输入
204 if (inContinuation_) {
205 pendingInput_ += "\n" + line;
206 } else {
207 // G-P2-8 fix: 空输入时清空输入框（用户可能输入了纯空白），避免残留显示
208 if (trimmedLine.isEmpty()) {
209 inputLine_->clear();
210 return;
211 }
212 pendingInput_ = trimmedLine;
213 }
214
215 // 保存到历史（仅首次行）
216 if (!inContinuation_) {
217 // AUDIT-P3-ROUND53 fix: 历史去重——若与最后一条相同则不追加，
218 // 对齐主流 REPL（Python/Node/bash）行为。避免重复命令多次出现，
219 // 按 Up 键需翻越多次才能到达上一条不同命令。
220 if (history_.empty() || history_.back() != trimmedLine) {
221 history_.push_back(trimmedLine);
222 saveHistoryToSettings(); // 拓展二期：追加即持久化（崩溃不丢历史）
223 }
224 historyIndex_ = history_.size();
225 }
226
227 // 显示输入（使用纯文本+颜色格式，避免HTML注入）
228 {
229 QTextCursor cursor(outputArea_->document());
230 cursor.movePosition(QTextCursor::End);
231 cursor.insertText("\n");
232 QTextCharFormat fmt;
233 fmt.setForeground(QColor("#859900"));
234 cursor.setCharFormat(fmt);
235 QString prompt = inContinuation_ ? "... " : ">>> ";
236 cursor.insertText(prompt + (inContinuation_ ? line : trimmedLine));
237 cursor.setCharFormat(QTextCharFormat());
238 outputArea_->setTextCursor(cursor);
239 outputArea_->ensureCursorVisible();
240 }
241
242 // PANEL-05 fix: clear 在续行模式下也可退出续行
243 if (trimmedLine == "clear") {
244 outputArea_->clear();
245 pendingInput_.clear();
246 inContinuation_ = false;
247 inputLine_->clear();
248 return;
249 }
250 // 特殊命令（仅在非续行模式）
251 if (!inContinuation_) {
252 if (trimmedLine == "help") {
253 outputArea_->append("MiniLang 支持的语法:\n"
254 " var x = 10; 变量声明\n"
255 " int a = 5; 类型注解变量\n"
256 " fun f(x) { ... } 函数声明\n"
257 " if (cond) { ... } 条件语句\n"
258 " while (cond) { ... } 循环语句\n"
259 " for (init; cond; upd) { ... } for循环\n"
260 " print(expr); 输出\n"
261 " [1, 2, 3] 数组字面量\n"
262 " {\"key\": val} 字典字面量\n"
263 " null 空值\n"
264 " class Name { ... } 类声明\n"
265 " import \"mod\" { f }; 模块导入\n"
266 " 多行输入: 未闭合的 { ( [ 或未闭合字符串会自动续行\n"
267 " 续行中按回车(空行)可中止续行\n"
268 "REPL 行为说明:\n"
269 " - 表达式语句(如 '1 + 2;')自动求值并打印结果\n"
270 " - 'clear' 仅清空输出区与续行缓冲,不重置已定义变量/函数/类\n"
271 " - 'reload \"mod\"' 清除模块缓存,下次 import 重新加载源码\n"
272 " - 'reload all' 清除所有模块缓存\n"
273 " - '%reset' 重置 REPL 环境(清除所有变量/函数/类/模块缓存)\n"
274 "─── Magic 命令 ─────────────────────────────\n"
275 "输入 %help 查看完整 magic 命令列表与详细说明。\n"
276 "常用 magic 命令速查:\n"
277 " %ast <expr> 查看 AST\n"
278 " %ir <expr> 查看 IR\n"
279 " %disassemble 反汇编当前字节码\n"
280 " %compare <expr> 三后端对比\n"
281 " %profile <expr> 性能剖析\n"
282 " %memory 内存模型\n"
283 " %tokens <expr> 词法分析结果\n"
284 " %reset 重置 REPL 环境\n"
285 " %version 查看 MiniLang 版本\n"
286 "────────────────────────────────────────────\n");
287 pendingInput_.clear();
288 inputLine_->clear();
289 return;
290 }
291 // 模块缓存刷新命令：reload "mod" 或 reload all
292 // 场景：用户修改了模块源文件后，需在 REPL 中获取最新版本
293 // AUDIT-BUG-F2 fix: 用精确匹配 + 空格前缀，避免误匹配 reloadable/reloadX 等标识符，
294 // 原 startsWith("reload") 会命中这些标识符并清空用户输入。
295 if (trimmedLine == "reload" || trimmedLine.startsWith("reload ")) {
296 // BUG-REPL-G2 fix (P2): reload 命令绕过 isRunning() 检查。
297 // reload 会清空模块缓存，若 worker 线程正在执行 import，缓存失效会导致
298 // worker 后续访问已加载模块时崩溃或行为不一致。executeLine 入口虽已检查
299 // isRunning()，但 reload 在 onReturnPressed 中提前 return，绕过该检查。
300 // 此处显式拦截，与 executeLine 的检查语义一致。
301 if (controller_ && controller_->isRunning()) {
302 appendError("程序正在运行，请先停止后再使用 reload");
303 pendingInput_.clear();
304 inputLine_->clear();
305 return;
306 }
307 QString arg = trimmedLine == "reload" ? QString() : trimmedLine.mid(7).trimmed();
308 if (arg.isEmpty()) {
309 appendError("用法: reload \"模块路径\" 或 reload all");
310 } else if (arg == "all") {
311 controller_->clearAllModuleCache();
312 appendOutput("[已清除所有模块缓存，下次 import 将重新加载源码]");
313 } else if (arg.startsWith("\"") && arg.endsWith("\"") && arg.length() >= 2) {
314 std::string path = arg.mid(1, arg.length() - 2).toStdString();
315 controller_->clearModuleCache(path);
316 appendOutput(QString::fromStdString("[已清除模块 " + path + " 缓存，下次 import 将重新加载]"));
317 } else {
318 appendError("用法: reload \"模块路径\" 或 reload all");
319 }
320 pendingInput_.clear();
321 inputLine_->clear();
322 return;
323 }
324 // ROUND-69 P1-1 fix: magic 命令在 isInputComplete 之前拦截。
325 // 原代码在 executeLine 中检测 % 前缀，但 isInputComplete() 先被调用，
326 // "%ast fun f() {" 因 { 未闭合被误判为"不完整"进入续行模式，用户被迫
327 // 输入 } 闭合后才能触发 magic 命令。magic 命令是单行命令，不应走续行逻辑。
328 if (trimmedLine.startsWith('%')) {
329 if (replRunning_) {
330 appendError("上一次执行尚未完成，请稍候...");
331 pendingInput_.clear();
332 inputLine_->clear();
333 return;
334 }
335 if (controller_ && controller_->isRunning()) {
336 appendError("程序正在运行，请先停止后再使用 REPL magic 命令");
337 pendingInput_.clear();
338 inputLine_->clear();
339 return;
340 }
341 if (controller_ && controller_->isVmRunning()) {
342 appendError("VM RUN 模式正在执行，请先停止 VM 再使用 REPL magic 命令");
343 pendingInput_.clear();
344 inputLine_->clear();
345 return;
346 }
347 std::string result = MagicCommands::handle(trimmedLine.toStdString(), controller_);
348 if (!result.empty()) {
349 appendOutput(QString::fromStdString(result));
350 }
351 pendingInput_.clear();
352 inputLine_->clear();
353 return;
354 }
355 }
356
357 // R4: 检查输入是否完整
358 if (!isInputComplete(pendingInput_)) {
359 inContinuation_ = true;
360 inputLine_->clear();
361 return;
362 }
363
364 // 输入完整 — 执行并重置续行状态
365 inContinuation_ = false;
366 // AUDIT-P2-ROUND53 fix: 多行输入历史仅保存首行，Up 键无法恢复完整多行输入。
367 // 执行前用完整 pendingInput_（含 \n 拼接的续行）替换之前存入的首行。
368 if (!history_.empty() && historyIndex_ == static_cast<int>(history_.size())) {
369 history_.back() = pendingInput_;
370 }
371 executeLine(pendingInput_);
372 pendingInput_.clear();
373
374 inputLine_->clear();
375}

377/// 提交一行/一段 MiniLang 代码到后端异步执行。
378void ReplPanel::executeLine(const QString& line) {
379 // ROUND-70 fix: 移除 executeLine 中的 magic 命令检测（死代码）。
380 // onReturnPressed 已在 isInputComplete 之前拦截 magic 命令（行 303-329），
381 // executeLine 永远不会收到以 % 开头的输入。保留检测会让读者误以为
382 // executeLine 可能被其他路径直接调用并收到 magic 命令，增加维护负担。
383
384 if (!controller_) {
385 appendError("解释器未初始化");
386 return;
387 }
388
389 // QT-R-06 fix: 若上一次异步执行尚未完成，拒绝新输入
390 if (replRunning_) {
391 appendError("上一次执行尚未完成，请稍候...");
392 return;
393 }
394
395 // R161: 调试器 REPL 模式——调试暂停时在当前上下文执行任意表达式/语句。
396 // 复用 ReplPanel 现有 UI（输入框/历史/续行），走主线程同步求值（不走 std::async），
397 // 因为 Interpreter 暂停期 worker 阻塞在 pauseCV_、VM 暂停期主线程空闲，无竞争。
398 // 阶段1 只读语义：临时 Interpreter 独立实例，副作用不写回主程序。
399 if (controller_->isDebugPaused() || (controller_->isVmInitialized() && !controller_->isVmRunning())) {
400 std::string capturedOutput;
401 auto watchResult = controller_->evaluateDebuggerRepl(line.toStdString(), &capturedOutput);
402 if (!capturedOutput.empty()) {
403 appendOutput(QString::fromStdString(capturedOutput));
404 }
405 if (watchResult.ok) {
406 // 裸表达式显示求值结果；print/var/赋值语句返回 null 不显示
407 if (!watchResult.valueRepr.empty() && watchResult.valueRepr != "null" && watchResult.typeName != "null") {
408 appendOutput(QString::fromStdString(watchResult.valueRepr));
409 }
410 } else {
411 appendError(QString::fromStdString(watchResult.error));
412 }
413 return;
414 }
415
416 // 2026-06-29 审计修复 R1/R2: 第二道防线——显式检查 worker 是否正在运行。
417 // 原互斥是单向的（onRun 检查 isReplRunning，但 ReplPanel 不检查 isRunning），
418 // 仅依赖 setInputEnabled(false) 禁用输入框。若 UI 禁用因异常路径未生效，
419 // 或未来重构绕过 UI 直接调用 executeLine，会导致 worker 线程与 REPL 异步任务
420 // 并发访问同一 Interpreter 的 globalEnv_/classRegistry_ 等共享状态 → 数据竞争。
421 // 此处显式检查 isRunning() 作为第二道防线，不依赖 UI 状态。
422 if (controller_->isRunning()) {
423 appendError("程序正在运行，请先停止后再使用 REPL");
424 return;
425 }
426 // BUG-REPL-AUDIT-14 fix: VM RUN 模式异步执行期间 isRunning() 仅反映 workerMgr 状态，
427 // VM RUN 仍可能活跃。prepareRun 已检查 vmStepper_.isRunning() 并拒绝，REPL 路径
428 // 需对称检查，避免 VM RUN 期间 REPL 启动并发访问 Compiler/CompileResult 共享状态。
429 if (controller_->isVmRunning()) {
430 appendError("VM RUN 模式正在执行，请先停止 VM 再使用 REPL");
431 return;
432 }
433
434 // AUDIT-REPL-5 fix: 移除误导性死代码注释"确保语句以分号结尾（简单表达式除外）"。
435 // 实际无任何分号补全逻辑——Parser::expressionStatement 强制 consume(TK_SEMICOLON)，
436 // 故裸表达式必须以 ';' 结尾否则报"期望 ';'"。用户需显式输入分号。
437 std::string source = line.toStdString();
438
439 // 词法分析（主线程，轻量）
440 Lexer lexer;
441 std::vector<Token> tokens;
442 try {
443 tokens = lexer.scan(source);
444 } catch (const std::exception& e) {
445 appendError(QString("词法错误: %1")
446 .arg(QString::fromStdString(ErrorHintEngine::enrichErrorMessage(e.what(), "parse", {}))));
447 return;
448 }
449
450 // 检查词法错误
451 for (const auto& tok : tokens) {
452 if (tok.type == TokenType::TK_ERROR) {
453 appendError(QString("词法错误 (行 %1, 列 %2): %3")
454 .arg(tok.line)
455 .arg(tok.column)
456 .arg(QString::fromStdString(ErrorHintEngine::enrichErrorMessage(tok.lexeme, "parse", {}))));
457 return;
458 }
459 }
460
461 // 语法分析（主线程，轻量）
462 Parser parser;
463 std::unique_ptr<Block> ast;
464 try {
465 ast = parser.parse(tokens);
466 } catch (const ParseError& e) {
467 appendError(QString("语法错误 (行 %1, 列 %2): %3")
468 .arg(e.line)
469 .arg(e.column)
470 .arg(QString::fromStdString(ErrorHintEngine::enrichErrorMessage(e.what(), "parse", {}))));
471 return;
472 }
473
474 if (!ast)
475 return;
476
477 // REPL-MAGIC fix: 成功解析后更新管线状态（tokens/AST），使无参 magic 命令
478 // （%ast/%tokens）能显示最近 REPL 输入的结果，而非上次 Run(F5) 的陈旧数据。
479 // setReplPipelineState 内部静默执行 Lex+Parse（不发 diagnostics 信号），
480 // 编译结果清空（REPL 不走字节码编译）。
481 controller_->setReplPipelineState(source);
482
483 // QT-R-06 fix: 异步执行解释器，避免主线程阻塞。
484 // 词法/语法分析在主线程（轻量，<1ms），解释器执行可能耗时（如 while 循环）放后台。
485 // 用 std::async + QTimer 轮询替代 QtConcurrent（Qt6::Concurrent 未安装）。
486 // 异常在异步任务中捕获，通过 QMetaObject::invokeMethod 投递到主线程信号链。
487 // PANEL-02 fix: 先保留 AST 再执行，确保异常时 classRegistry_/闭包 body 指针不悬空
488 Block* rawAst = ast.get();
489 controller_->retainReplAst(std::move(ast));
490
491 // 标记执行中，禁用输入
492 // Bug fix: std::async 可能在资源耗尽或线程数限制时抛 std::system_error，
493 // 此时若已设置 replRunning_=true / 禁用输入但未启动轮询定时器，
494 // REPL 将永久卡死（replRunning_ 早期返回守卫阻止后续输入）。
495 // 改为：先成功启动 async，再切换状态。AST 已在 replAsts_ 中保留，
496 // async 失败时该 AST 不会被使用（executeRepl 未执行），仅造成轻微内存占用。
497 IdeController* ctrl = controller_; // 显式捕获
498 // AUDIT fix: 重置错误标志，避免上次执行的残余状态影响本次输出判断
499 hadReplError_.store(false);
500 std::atomic<bool>* errFlag = &hadReplError_;
501 // BUG-REPL-G6 fix (P2): 捕获 this 以便将错误同时投递到 REPL 输出区。
502 // 原实现仅 emit ctrl->runtimeError/genericError 信号（由 Ide 连接到主错误面板），
503 // REPL 用户看不到错误反馈。
504 // AUDIT-P1 fix: 原用裸 ReplPanel* self，~ReplPanel 的 wait() 只保证 future 完成，
505 // 但 invokeMethod(QueuedConnection) 投递的 lambda 在主线程事件队列中，析构期间
506 // 主线程阻塞在 wait() 不会 dispatch，析构后回到事件循环才 dispatch → UAF。
507 // 改用 QPointer + qApp 作为 invokeMethod context：qApp 永远存活，lambda 在主线程
508 // 执行时检查 self 是否为 null，避免悬垂访问。
509 QPointer<ReplPanel> self(this);
510 std::future<Value> newFuture;
511 try {
512 newFuture = std::async(std::launch::async, [ctrl, rawAst, errFlag, self]() -> Value {
513 try {
514 return ctrl->executeRepl(*rawAst);
515 } catch (const RuntimeError& e) {
516 errFlag->store(true);
517 // BUG-REPL-G6 fix (P2): 同时投递到 REPL 输出区，让用户在 REPL 中
518 // 直接看到错误反馈（原有 emit ctrl->runtimeError 仍保留，由 Ide
519 // 连接到主错误面板）
520 // P0-3 fix (F11): 在 worker 线程中收集 scopeVars（此时 Interpreter
521 // 状态仍可访问，主线程在 pollReplFuture 阻塞等待），用于拼写建议
522 std::vector<std::string> scopeVars;
523 if (ctrl) {
524 scopeVars = ctrl->getReplScopeVariableNames();
525 }
526 std::string enriched = ErrorHintEngine::enrichErrorMessage(e.what(), "runtime", scopeVars);
527 std::string msg = enriched;
528 QMetaObject::invokeMethod(
529 qApp,
530 [self, msg]() {
531 if (self)
532 self->appendError(QString::fromStdString(msg));
533 },
534 Qt::QueuedConnection);
535 QMetaObject::invokeMethod(
536 ctrl,
537 [ctrl, msg = std::string(e.what()), line = e.line, col = e.column]() {
538 emit ctrl->runtimeError(QString::fromStdString(msg), line, col);
539 },
540 Qt::QueuedConnection);
541 return Value::nullValue();
542 } catch (const DebugStopException&) {
543 // RA-C fix: REPL 中止（closeEvent 超时 / 用户停止）——静默退出，不报错
544 return Value::nullValue();
545 } catch (const std::exception& e) {
546 errFlag->store(true);
547 // BUG-REPL-G6 fix (P2): 同理投递到 REPL 输出区
548 // P0-3 fix (F11): 通用异常也通过 ErrorHintEngine 增强
549 std::vector<std::string> scopeVars;
550 if (ctrl) {
551 scopeVars = ctrl->getReplScopeVariableNames();
552 }
553 std::string enriched = ErrorHintEngine::enrichErrorMessage(e.what(), "runtime", scopeVars);
554 std::string msg = enriched;
555 QMetaObject::invokeMethod(
556 qApp,
557 [self, msg]() {
558 if (self)
559 self->appendError(QString::fromStdString(msg));
560 },
561 Qt::QueuedConnection);
562 QMetaObject::invokeMethod(
563 ctrl,
564 [ctrl, msg = std::string(e.what())]() { emit ctrl->genericError(QString::fromStdString(msg)); },
565 Qt::QueuedConnection);
566 return Value::nullValue();
567 }
568 });
569 } catch (const std::exception& e) {
570 // async 启动失败：报告错误，REPL 保持可用状态（不切换 replRunning_）
571 appendError(QString("无法启动异步执行: %1").arg(e.what()));
572 return;
573 }
574 // async 启动成功，切换状态并启动轮询
575 replFuture_ = std::move(newFuture);
576 replRunning_ = true;
577 setInputEnabled(false);
578 pollTimer_->start();
579}

581/// 轮询异步执行结果，完成后回写输出并恢复输入。
582void ReplPanel::pollReplFuture() {
583 // QT-R-06 fix: 轮询 std::future 状态，完成则显示结果并恢复输入
584 if (!replFuture_.valid())
585 return;
586
587 // 检查是否完成（非阻塞）
588 auto status = replFuture_.wait_for(std::chrono::seconds(0));
589 if (status != std::future_status::ready)
590 return; // 仍在执行
591
592 pollTimer_->stop();
593
594 // AUDIT-REPL-8 fix: get() 可能抛未捕获异常（如 std::bad_alloc 不属 std::exception 派生，
595 // 或异步 lambda 内漏捕获的异常类型）。外层 try/catch 兜底防止传播到 Qt 事件循环致崩溃。
596 Value result;
597 try {
598 result = replFuture_.get();
599 } catch (const std::exception& e) {
600 appendError(QString("REPL 执行异常: %1").arg(e.what()));
601 replRunning_ = false;
602 setInputEnabled(true);
603 inputLine_->setFocus();
604 return;
605 } catch (...) {
606 appendError("REPL 执行未知异常");
607 replRunning_ = false;
608 setInputEnabled(true);
609 inputLine_->setFocus();
610 return;
611 }
612
613 // ROUND-70 fix: 跳过 null 结果的打印。
614 // 原行为（PANEL-03 fix）：null 结果也打印，导致 print/var/fun/class 等语句
615 // 执行后多输出一行 "null"，与 Python/Node REPL 行为不一致。
616 // 改为仅打印非 null 结果——表达式语句（如 1+2;）的结果正常显示，
617 // 声明语句（var/fun/class）和 print 语句返回 null 不再产生多余输出。
618 // AUDIT fix: 若异步执行已通过信号显示错误，跳过结果输出避免重复显示
619 if (!hadReplError_.load() && !result.isNull()) {
620 appendOutput(QString::fromStdString(result.toString()));
621 }
622
623 // 恢复输入
624 replRunning_ = false;
625 setInputEnabled(true);
626 inputLine_->setFocus();
627}

629/// 判断当前输入是否为完整可执行的程序（括号/引号配平等）。
630bool ReplPanel::isInputComplete(const QString& input) {
631 int braceDepth = 0; // {}
632 int parenDepth = 0; // ()
633 int bracketDepth = 0; // []
634 bool inString = false;
635 int blockCommentDepth = 0; // AUDIT-BUG-R2 fix: 跟踪嵌套块注释深度（与 Lexer 一致）
636 bool inLineComment = false;
637 int tryCount = 0; // BUG-R1 fix: 跟踪 try/catch 配对
638 int catchCount = 0;
639 int finallyCount = 0; // P2-C fix: 跟踪 finally 块
640
641 // AUDIT-REPL-1 fix: 字符串插值栈。MiniLang 字符串支持 "...{expr}..." 插值，
642 // { 在字符串内开启表达式上下文（可能含嵌套字符串/字典/数组），} 闭合插值回到字符串模式。
643 // 栈元素 true = 该 { 由插值开启（闭合时回到字符串模式），false = 普通代码 {。
644 // 原 bug: inString 状态下跳过所有字符，不识别 { 的插值语义，导致
645 // `var x = "{";` 被判定为完整（实际 { 开启插值，" 开启嵌套字符串，EOF 未闭合）。
646 std::vector<bool> interpOpens;
647
648 for (int i = 0; i < input.length(); ++i) {
649 QChar c = input[i];
650
651 // 处理块注释（跳过内部字符，支持嵌套）
652 if (blockCommentDepth > 0) {
653 if (c == '*' && i + 1 < input.length() && input[i + 1] == '/') {
654 --blockCommentDepth;
655 ++i;
656 } else if (c == '/' && i + 1 < input.length() && input[i + 1] == '*') {
657 ++blockCommentDepth;
658 ++i;
659 }
660 continue;
661 }
662
663 // 处理行注释（跳过内部字符）
664 if (inLineComment) {
665 // AUDIT-BUG-R3 fix: \r 也标志行注释结束（Windows \r\n / 旧 Mac \r）
666 if (c == '\n' || c == '\r') {
667 inLineComment = false;
668 }
669 continue;
670 }
671
672 // 处理字符串字面量（跳过内部字符，识别插值 { ）
673 if (inString) {
674 if (c == '\\' && i + 1 < input.length()) {
675 ++i; // 跳过转义字符
676 continue;
677 }
678 if (c == '"') {
679 inString = false;
680 continue;
681 }
682 // AUDIT-REPL-1 fix: { 在字符串内开启插值表达式上下文
683 if (c == '{') {
684 interpOpens.push_back(true); // 标记此 { 为插值开启
685 ++braceDepth;
686 inString = false; // 退出字符串模式，进入表达式扫描
687 continue;
688 }
689 continue; // 字符串内其他字符跳过
690 }
691
692 if (c == '"') {
693 inString = true;
694 continue;
695 }
696
697 // 注释起始检测
698 if (c == '/' && i + 1 < input.length()) {
699 if (input[i + 1] == '/') {
700 inLineComment = true;
701 ++i;
702 continue;
703 }
704 if (input[i + 1] == '*') {
705 ++blockCommentDepth;
706 ++i;
707 continue;
708 }
709 }
710
711 // BUG-R1 fix: 识别 try/catch 关键字以检测缺失的 catch
712 if (c.isLetter() || c == '_') {
713 int start = i;
714 while (i < input.length() && (input[i].isLetterOrNumber() || input[i] == '_')) {
715 ++i;
716 }
717 int len = i - start;
718 // 仅匹配完整单词 "try" / "catch" / "finally"，避免匹配 "trying" / "catcher"
719 if (len == 3 && input[start] == 't' && input[start + 1] == 'r' && input[start + 2] == 'y') {
720 ++tryCount;
721 } else if (len == 5 && input[start] == 'c' && input[start + 1] == 'a' && input[start + 2] == 't' &&
722 input[start + 3] == 'c' && input[start + 4] == 'h') {
723 ++catchCount;
724 } else if (len == 7 && input[start] == 'f' && input[start + 1] == 'i' && input[start + 2] == 'n' &&
725 input[start + 3] == 'a' && input[start + 4] == 'l' && input[start + 5] == 'l' &&
726 input[start + 6] == 'y') {
727 ++finallyCount;
728 }
729 --i; // 补偿 for 循环的 ++i
730 continue;
731 }
732
733 switch (c.toLatin1()) {
734 case '{':
735 interpOpens.push_back(false); // 普通代码 {
736 ++braceDepth;
737 break;
738 case '}':
739 --braceDepth;
740 // BUG-REPL-G3 fix (P2): 负 braceDepth 表示括号不匹配（如多余 }），
741 // 视为输入不完整避免误判完整后送入 Parser 触发无法定位的语法错误。
742 // 原 isInputComplete 仅在末尾检查 braceDepth!=0，负深度会被末尾的
743 // "!= 0" 判断捕获，但中途负深度可能因后续 { 重新归零而漏判
744 // （如 "}{" 末尾 braceDepth=0 误判完整）。此处提前返回更准确。
745 if (braceDepth < 0)
746 return false;
747 // AUDIT-REPL-1 fix: 若此 } 闭合的是插值 {，回到字符串模式
748 if (!interpOpens.empty()) {
749 bool wasInterp = interpOpens.back();
750 interpOpens.pop_back();
751 if (wasInterp) {
752 inString = true;
753 }
754 }
755 break;
756 case '(':
757 ++parenDepth;
758 break;
759 case ')':
760 --parenDepth;
761 // BUG-REPL-G3 fix (P2): 同理负 parenDepth 视为不完整
762 if (parenDepth < 0)
763 return false;
764 break;
765 case '[':
766 ++bracketDepth;
767 break;
768 case ']':
769 --bracketDepth;
770 // BUG-REPL-G3 fix (P2): 同理负 bracketDepth 视为不完整
771 if (bracketDepth < 0)
772 return false;
773 break;
774 }
775 }
776
777 // AUDIT-BUG-R2 fix: 未闭合的嵌套块注释视为输入不完整
778 if (blockCommentDepth > 0)
779 return false;
780 // 未闭合的字符串（含插值内未闭合的嵌套字符串）
781 if (inString)
782 return false;
783 // 括号不匹配
784 if (braceDepth != 0 || parenDepth != 0 || bracketDepth != 0)
785 return false;
786 // BUG-R1 fix: try 缺少 catch 或 finally 视为输入不完整
787 // P2-C fix: try-finally（无 catch）也是合法结构，catch 或 finally 至少其一即可配对
788 if (tryCount > catchCount + finallyCount)
789 return false;
790
791 return true;
792}

794/// 事件过滤器：拦截输入框特殊按键（如上下历史）。
795bool ReplPanel::eventFilter(QObject* obj, QEvent* event) {
796 if (obj == inputLine_ && event->type() == QEvent::KeyPress) {
797 QKeyEvent* keyEvent = static_cast<QKeyEvent*>(event);
798 // L5 fix: QPlainTextEdit 无 returnPressed 信号，在此拦截 Return/Enter 键
799 // 触发 onReturnPressed。Shift+Enter 允许在多行历史中插入换行（不触发执行）。
800 if ((keyEvent->key() == Qt::Key_Return || keyEvent->key() == Qt::Key_Enter) &&
801 !(keyEvent->modifiers() & Qt::ShiftModifier)) {
802 onReturnPressed();
803 return true;
804 }
805 if (keyEvent->key() == Qt::Key_Up) {
806 // 上一条历史
807 if (historyIndex_ > 0) {
808 historyIndex_--;
809 inputLine_->setPlainText(history_[historyIndex_]);
810 }
811 return true;
812 }
813 if (keyEvent->key() == Qt::Key_Down) {
814 // 下一条历史
815 if (historyIndex_ < static_cast<int>(history_.size()) - 1) {
816 historyIndex_++;
817 inputLine_->setPlainText(history_[historyIndex_]);
818 } else {
819 historyIndex_ = history_.size();
820 inputLine_->clear();
821 }
822 return true;
823 }
824 // R53-UX8 fix: Esc 中止续行。原续行模式下用户只能继续输入完整代码，
825 // 无法中止——多行结构（如未闭合的 fun 定义）一旦开始就必须完成。
826 // Esc 在续行模式下中止当前 pendingInput_，回到单行模式。
827 // 非续行模式下不拦截 Esc（保持 QLineEdit 默认行为，允许清空输入框）。
828 if (keyEvent->key() == Qt::Key_Escape && inContinuation_) {
829 inContinuation_ = false;
830 pendingInput_.clear();
831 inputLine_->clear();
832 appendOutput(QString::fromUtf8("[续行已中止]"));
833 return true;
834 }
835 // R53-UX8 fix: Ctrl+L 清屏（清空输出区，保留 pendingInput_/history/变量状态）。
836 // 与 'clear' 命令行为一致，但快捷键更便捷。运行中拒绝以避免与 worker 输出竞争。
837 if ((keyEvent->key() == Qt::Key_L) && (keyEvent->modifiers() & Qt::ControlModifier)) {
838 if (controller_ && controller_->isRunning()) {
839 appendError("程序正在运行，无法清屏");
840 return true;
841 }
842 outputArea_->clear();
843 return true;
844 }
845 // 拓展二期：Tab 前缀补全（关键字 + 历史词）。
846 // 唯一匹配直接补全；多候选补全至公共前缀并在输出区列出候选；
847 // 无匹配/空前缀时放行（保留默认 Tab 缩进）。
848 if (keyEvent->key() == Qt::Key_Tab && !(keyEvent->modifiers() & Qt::ShiftModifier)) {
849 if (tryTabComplete())
850 return true;
851 }
852 }
853 return QWidget::eventFilter(obj, event);
854}

856// ============================================================
857// 拓展二期：Tab 补全 + 历史持久化
858// ============================================================
859
860bool ReplPanel::tryTabComplete() {
861 // 取光标前的当前词（标识符字符 [A-Za-z0-9_]）
862 QTextCursor cursor = inputLine_->textCursor();
863 QString text = inputLine_->toPlainText();
864 int pos = cursor.position();
865 int start = pos;
866 while (start > 0) {
867 QChar c = text.at(start - 1);
868 if (c.isLetterOrNumber() || c == '_') {
869 --start;
870 } else {
871 break;
872 }
873 }
874 QString prefix = text.mid(start, pos - start);
875 if (prefix.isEmpty())
876 return false; // 无前缀：保留默认 Tab 行为
877
878 // 候选集：MiniLang 关键字/内建 + 历史输入中的标识符词
879 static const QStringList kKeywords = {
880 "var", "fun", "if", "else", "while", "for", "return", "break", "continue", "print", "true",
881 "false", "null", "and", "or", "not", "class", "extends", "super", "import", "from", "export",
882 "try", "catch", "finally", "throw", "enum", "match", "case", "default", "yield", "len", "type",
883 "str", "int", "abs", "min", "max", "range", "sum", "channel", "mutex", "rwlock", "spawn"};
884 QSet<QString> candidateSet;
885 for (const QString& kw : kKeywords) {
886 if (kw.startsWith(prefix) && kw != prefix)
887 candidateSet.insert(kw);
888 }
889 // 历史词：拆历史条目中的标识符（用户自己的变量/函数名高频出现在历史里）
890 static const QRegularExpression kWordRe(QStringLiteral("[A-Za-z_][A-Za-z0-9_]*"));
891 for (const QString& h : history_) {
892 auto it = kWordRe.globalMatch(h);
893 while (it.hasNext()) {
894 QString w = it.next().captured(0);
895 if (w.startsWith(prefix) && w != prefix)
896 candidateSet.insert(w);
897 }
898 }
899 if (candidateSet.isEmpty())
900 return false;
901
902 QStringList candidates = candidateSet.values();
903 candidates.sort();
904 if (candidates.size() == 1) {
905 // 唯一匹配：直接补全
906 cursor.setPosition(start);
907 cursor.setPosition(pos, QTextCursor::KeepAnchor);
908 cursor.insertText(candidates.first());
909 return true;
910 }
911 // 多候选：补全至最长公共前缀 + 列出候选
