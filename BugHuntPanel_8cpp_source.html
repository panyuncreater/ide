316void BugHuntPanel::refreshBugChips() {
317 // issue 5: 刷新芯片状态（current 高亮 + 难度筛选可见性）
318 // R51-2 fix: 可见性同时考虑难度筛选与搜索关键字，避免两者互相覆盖
319 const auto& items = BugHuntLibrary::items();
320 QString loweredSearch = lastSearchText_.toLower();
321 int firstVisible = -1;
322 for (int i = 0; i < bugChips_.size() && i < (int)items.size(); ++i) {
323 QPushButton* chip = bugChips_[i];
324 if (!chip)
325 continue;
326 int diffId = static_cast<int>(items[i].difficulty);
327 bool visibleByDiff = (currentDifficultyFilter_ == -1 || diffId == currentDifficultyFilter_);
328 // R51-2 fix: 同时检查搜索关键字
329 bool visibleBySearch = true;
330 if (!loweredSearch.isEmpty()) {
331 QString haystack = QString::fromUtf8(items[i].title.c_str()).toLower() + " " +
332 QString::fromUtf8(items[i].id.c_str()).toLower() + " " +
333 QString::fromUtf8(items[i].category.c_str()).toLower();
334 visibleBySearch = haystack.contains(loweredSearch);
335 }
336 bool visible = visibleByDiff && visibleBySearch;
337 chip->setVisible(visible);
338 chip->setProperty("current", (i == currentItemIndex_));
339 chip->style()->unpolish(chip);
340 chip->style()->polish(chip);
341 if (visible && firstVisible < 0)
342 firstVisible = i;
343 }
344 // 若当前题目被筛选隐藏，自动选中第一个可见芯片
345 if (currentItemIndex_ < 0 && firstVisible >= 0) {
346 onItemSelected(firstVisible);
347 } else if (currentItemIndex_ >= 0) {
348 // 注意：不能用 QWidget::isVisible() 判定难度筛选可见性——
349 // 构造期间父 widget 未 show()，isVisible() 恒返回 false，会导致
350 // refreshBugChips → onItemSelected → refreshBugChips 无限递归栈溢出。
351 // 这里用难度筛选+搜索条件判定，与上面 firstVisible 口径一致。
352 bool curVisible = false;
353 if (currentItemIndex_ < (int)items.size()) {
354 int curDiffId = static_cast<int>(items[currentItemIndex_].difficulty);
355 curVisible = (currentDifficultyFilter_ == -1 || curDiffId == currentDifficultyFilter_);
356 if (curVisible && !loweredSearch.isEmpty()) {
357 QString haystack = QString::fromUtf8(items[currentItemIndex_].title.c_str()).toLower() + " " +
358 QString::fromUtf8(items[currentItemIndex_].id.c_str()).toLower() + " " +
359 QString::fromUtf8(items[currentItemIndex_].category.c_str()).toLower();
360 curVisible = haystack.contains(loweredSearch);
361 }
362 }
363 if (!curVisible && firstVisible >= 0 && firstVisible != currentItemIndex_) {
364 onItemSelected(firstVisible);
365 }
366 }

645// 单后端运行结果（供 onTripleVerify 使用）
646struct BackendRunResult {
647 std::string output; // 标准输出
648 std::string exceptionName; // 异常名（空 = 无异常）
649 long long micros = 0; // 耗时（微秒）
650};

652// ARCH-10: 通过 BackendExecutionService 触发三后端，面板不再直接依赖
653// Lexer/Parser/Compiler/VM/RegisterVM/Interpreter 等内部头文件。
654// 服务返回的 BackendExecResult 已包含 output/success/errorPrefix/errorMsg/elapsedMs，
655// 这里转换为面板内部使用的 BackendRunResult 结构。
656
657// Interpreter 路径：树遍历解释器
658BackendRunResult runInterpreter(const std::string& source) {
659 BackendRunResult r;
660 auto sr = BackendExecutionService::execute(source, BackendType::Interpreter);
661 r.output = sr.output;
662 r.micros = sr.elapsedUs; // AUDIT-R2 P2-2 fix: 直接使用微秒精度字段，不再 ms×1000 冒充 μs
663 if (!sr.success) {
664 r.exceptionName = sr.errorPrefix + ": " + sr.errorMsg;
665 }
666 return r;
667}
668
669// StackVM 路径：栈式字节码虚拟机
670BackendRunResult runStackVM(const std::string& source) {
671 BackendRunResult r;
672 auto sr = BackendExecutionService::execute(source, BackendType::StackVM_IR);
673 r.output = sr.output;
674 r.micros = sr.elapsedUs; // AUDIT-R2 P2-2 fix: 微秒精度
675 if (!sr.success) {
676 r.exceptionName = sr.errorPrefix + ": " + sr.errorMsg;
677 }
678 return r;
679}

681// RegisterVM 路径：寄存器式虚拟机
682BackendRunResult runRegisterVM(const std::string& source) {
683 BackendRunResult r;
684 auto sr = BackendExecutionService::execute(source, BackendType::RegisterVM_IR);
685 r.output = sr.output;
686 r.micros = sr.elapsedUs; // AUDIT-R2 P2-2 fix: 微秒精度
687 if (!sr.success) {
688 r.exceptionName = sr.errorPrefix + ": " + sr.errorMsg;
689 }
690 return r;

694void BugHuntPanel::onTripleVerify() {
695 // AUDIT-P2 fix: 防重复守卫——三后端验证执行 3 条串行链，快速双击触发 6 次执行
696 if (tripleVerifying_)
697 return;
698 // 变体模式下验证当前变体；标准模式下验证当前题目
699 if (variantMode_) {
700 if (currentVariantIndex_ < 0) {
701 statusLabel_->setText(mlTr("请先选择变体"));
702 return;
703 }
704 } else {
705 if (currentItemIndex_ < 0) {
706 statusLabel_->setText(mlTr("请先选择题目"));
707 return;
708 }
709 }
710
711 std::string source = codeEditor_->toPlainText().toStdString();
712 if (source.empty()) {
713 outputEdit_->setPlainText(mlTr("（空代码）"));
714 return;
715 }
716
717 // AUDIT-R2 P1-4 fix: RAII 守卫——异常抛出时也能恢复标志与按钮状态
718 VerifyScopeGuard guard(tripleVerifying_, tripleVerifyBtn_);
719
720 // 三条路径独立执行
721 auto ir = runInterpreter(source);
722 auto sv = runStackVM(source);
723 auto rv = runRegisterVM(source);
724
725 // 一致性判定
726 bool outputConsistent = (ir.output == sv.output) && (sv.output == rv.output);
727 // AUDIT-R2 P2-1 fix: 原判定要求三后端均有异常才可能"一致"，导致最常见的
728 // 成功场景（三后端均无异常）被误报"异常行为 [不一致]"。正确语义：
729 // 三后端异常名完全相同（含"均为空"）即一致。
730 bool exceptionConsistent = (ir.exceptionName == sv.exceptionName) && (sv.exceptionName == rv.exceptionName);
731
732 std::ostringstream out;
733 out << "============= 三后端对比验证 =============\n";
734 out << "[Interpreter] 输出: " << (ir.output.empty() ? "（无）" : ir.output);
735 out << " 异常: " << (ir.exceptionName.empty() ? "无" : ir.exceptionName) << "\n";
736 out << " 耗时: " << ir.micros << " μs\n\n";
737 out << "[StackVM] 输出: " << (sv.output.empty() ? "（无）" : sv.output);
738 out << " 异常: " << (sv.exceptionName.empty() ? "无" : sv.exceptionName) << "\n";
739 out << " 耗时: " << sv.micros << " μs\n\n";
740 out << "[RegisterVM] 输出: " << (rv.output.empty() ? "（无）" : rv.output);
741 out << " 异常: " << (rv.exceptionName.empty() ? "无" : rv.exceptionName) << "\n";
742 out << " 耗时: " << rv.micros << " μs\n\n";
743 out << "============= 一致性结论 =============\n";
744 out << "三后端输出 [" << (outputConsistent ? "一致" : "不一致") << "]\n";
745 out << "异常行为 [" << (exceptionConsistent ? "一致" : "不一致") << "]\n";
746
747 // 教学注解
748 out << "教学注解: ";
749 if (outputConsistent && ir.exceptionName.empty() && sv.exceptionName.empty() && rv.exceptionName.empty()) {
750 out << "三后端输出完全一致且无异常——该代码路径在三套引擎上语义等价。";
751 } else if (outputConsistent && exceptionConsistent) {
752 out << "三后端均抛出相同异常且输出一致——异常路径语义等价，关注异常本身是否为预期行为。";
753 } else if (!outputConsistent) {
754 out << "三后端输出不一致——存在语义差异，需逐行对比 Interpreter/StackVM/RegisterVM 实现。";
755 if (!ir.exceptionName.empty() && sv.exceptionName.empty() && rv.exceptionName.empty()) {
756 out << "（Interpreter 抛异常但 VM 路径未抛——可能 Interpreter 更严格的检查）";
757 } else if (ir.exceptionName.empty() && (!sv.exceptionName.empty() || !rv.exceptionName.empty())) {
758 out << "（VM 路径抛异常但 Interpreter 未抛——可能 VM 编译/执行有缺陷）";
759 } else if (!sv.exceptionName.empty() && rv.exceptionName.empty() && ir.exceptionName.empty()) {
760 out << "（仅 StackVM 抛异常——栈式 VM 路径存在 bug）";
761 } else if (!rv.exceptionName.empty() && sv.exceptionName.empty() && ir.exceptionName.empty()) {
762 out << "（仅 RegisterVM 抛异常——寄存器式 VM 路径存在 bug）";
763 }
764 } else if (!exceptionConsistent) {
765 out << "异常行为不一致——三后端对异常的处理存在差异（异常类型/消息/是否抛出）。";
766 }
767 out << "\n";
768
769 outputEdit_->setPlainText(QString::fromUtf8(out.str().c_str()));
770 statusLabel_->setText(mlTr("三后端对比验证完成"));
771 // 调试流程：标记「验证」步骤完成
772 if (!stepVerified_) {
773 stepVerified_ = true;
774 refreshDebugSteps();
775 }
776 if (variantMode_) {
777 variantStatusLabel_->setText(
778 mlTr("变体验证完成 — 输出%1").arg(outputConsistent ? mlTr("一致") : mlTr("不一致")));
779 }
780
781 // P0-B fix: 三后端输出一致且均无异常 → 视为"狩猎"成功，发射 challengeSolved。
782 // 学员修复 Bug 后三后端应产生一致输出且无异常——这与原 Bug 代码三后端不一致
783 // 或抛异常的状态形成对照，构成判分依据。仅标准模式（非变体）发射，因变体本身
784 // 是探索性挑战而非可机检修复。
785 if (!variantMode_ && currentItemIndex_ >= 0) {
786 const auto& items = BugHuntLibrary::items();
787 if (currentItemIndex_ < (int)items.size()) {
788 bool allClean = ir.exceptionName.empty() && sv.exceptionName.empty() && rv.exceptionName.empty();
789 // AUDIT-P2 fix: 排除"三后端均无输出"的退化解——学员删除所有 print
790 // 即可让 outputConsistent 平凡为 true，从而作弊通关。要求至少一个
791 // 后端有非空输出，确保学员确实修复了 Bug 并产生预期输出。
792 bool hasOutput = !ir.output.empty() || !sv.output.empty() || !rv.output.empty();
793 if (outputConsistent && allClean && hasOutput) {
794 int diffInt = static_cast<int>(items[currentItemIndex_].difficulty);
795 // AUDIT-P1 fix: 跟踪已解决题目，仅当当前难度所有题目全部解决时
796 // 才发射 challengeSolved。原实现单题通过即发射，与游戏面板
797 // "全部子关卡完成"语义不一致（TokenPuzzle 要求 5 关全通、
798 // AstToy 要求 6 关全通）。BEGINNER 5 道、INTERMEDIATE 4 道、EXPERT 6 道。
799 solvedItemIndices_.insert(currentItemIndex_);
800 // AUDIT-P1 fix: 持久化已解决题目到 LearnerProgressStore，解决跨会话完成判定丢失。
801 std::string solvedKey = "bughunt-solved-" + items[currentItemIndex_].id;
802 LearnerProgressStore::instance().markLevelStars(solvedKey, 1);
803 saveLearnerProgressWithFeedback(this); // P2-UX fix: 失败时弹 toast 避免静默丢失
804 int totalInDiff = 0;
805 int solvedInDiff = 0;
806 for (int i = 0; i < (int)items.size(); ++i) {
807 if (static_cast<int>(items[i].difficulty) == diffInt) {
808 ++totalInDiff;
809 if (solvedItemIndices_.contains(i))
810 ++solvedInDiff;
811 }
812 }
813 if (solvedInDiff == totalInDiff) {
814 emit challengeSolved(diffInt);
815 out << "\n✅ 狩猎成功！本档全部 " << totalInDiff << " 道题已解决，活动标记完成。\n";
816 } else {
817 out << "\n✅ 本题狩猎成功！本档已解决 " << solvedInDiff << "/" << totalInDiff << " 道。\n";
818 }
819 // 重新刷新输出（追加了成功提示）
820 outputEdit_->setPlainText(QString::fromUtf8(out.str().c_str()));
821 }
822 }
823 }

825}
