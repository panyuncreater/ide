67/// 从文件加载进度数据；失败返回 false 并保留内存状态。
68// R51-4 fix: 字段类型不匹配时拒绝加载（返回 false），保留内存中原数据。
69// 原实现构建全新 loaded 对象，遇到类型不匹配的字段静默跳过（该字段在 loaded
70// 中保持默认空值），最后 data_ = std::move(loaded) 整体覆盖——导致内存中
71// 该字段的有效数据被默认空值覆盖丢失。修复：若某顶层字段存在但类型与预期
72// 不符（如 completed 是数组而非对象），视为文件损坏，整体拒绝加载。
73// 缺失字段（向后兼容旧文件）仍按默认值处理，不影响加载。
74bool LearnerProgressStore::load() {
75 QString path = filePath();
76 QFile file(path);
77 if (!file.exists() || !file.open(QIODevice::ReadOnly)) {
78 // 文件不存在或无法打开——回退到空进度，不视为错误
79 // P2-UX fix: 记录日志便于诊断权限/路径问题（文件不存在不记录，仅记录打开失败）
80 if (file.exists()) {
81 LOG_WARNING("Progress file exists but cannot be opened: " + path.toStdString(), "Progress");
82 }
83 return false;
84 }
85
86 QByteArray raw = file.readAll();
87 file.close();
88
89 QJsonParseError parseError;
90 QJsonDocument doc = QJsonDocument::fromJson(raw, &parseError);
91 if (parseError.error != QJsonParseError::NoError || !doc.isObject()) {
92 // JSON 格式不匹配——回退到空进度，不崩溃
93 // P2-UX fix: 记录解析错误便于诊断文件损坏
94 LOG_WARNING("Progress file JSON parse error: " + parseError.errorString().toStdString(), "Progress");
95 return false;
96 }
97
98 LearnerProgress loaded;
99 QJsonObject root = doc.object();
100
101 // AUDIT-P2 fix: 校验 schemaVersion。缺失字段默认为 1（向后兼容旧文件），
102 // 高于当前支持版本(1)则拒绝加载，避免未来 schema 变更后误读旧结构。
103 // R51-4 fix: schemaVersion 若存在但非数字也视为损坏，拒绝加载。
104 if (root.contains(QString::fromUtf8("schemaVersion"))) {
105 QJsonValue svVal = root.value(QString::fromUtf8("schemaVersion"));
106 if (!svVal.isDouble())
107 return false;
108 }
109 int sv = root.value(QString::fromUtf8("schemaVersion")).toInt(1);
110 if (sv > 1)
111 return false;
112
113 // R51-4 fix: 顶层字段类型校验辅助——字段存在但类型不符则拒绝加载
114 auto requireObjectType = [&](const char* key) -> bool {
115 QString qk = QString::fromUtf8(key);
116 if (root.contains(qk) && !root.value(qk).isObject())
117 return false;
118 return true;
119 };
120 if (!requireObjectType("completed") || !requireObjectType("attemptCount") || !requireObjectType("lastAccessTime") ||
121 !requireObjectType("levelStars") || !requireObjectType("score") || !requireObjectType("bestStars") ||
122 !requireObjectType("spentMinutes") || !requireObjectType("failCount")) {
123 return false;
124 }
125 // currentStage 若存在必须是数字
126 if (root.contains(QString::fromUtf8("currentStage")) && !root.value(QString::fromUtf8("currentStage")).isDouble()) {
127 return false;
128 }
129
130 // completed: { "id": true, ... }
131 QJsonObject completedObj = root.value(QString::fromUtf8("completed")).toObject();
132 for (auto it = completedObj.begin(); it != completedObj.end(); ++it) {
133 if (it.value().isBool()) {
134 loaded.completed[it.key().toStdString()] = it.value().toBool();
135 }
136 }
137
138 // attemptCount: { "id": 3, ... }
139 QJsonObject attemptObj = root.value(QString::fromUtf8("attemptCount")).toObject();
140 for (auto it = attemptObj.begin(); it != attemptObj.end(); ++it) {
141 if (it.value().isDouble()) {
142 // AUDIT-P2 fix: 防御性校验——手动篡改的 JSON 可能含负数，
143 // 负数 attemptCount 会破坏 nextRecommended 排序语义。
144 int v = it.value().toInt();
145 if (v >= 0) {
146 loaded.attemptCount[it.key().toStdString()] = v;
147 }
148 }
149 }
150
151 // lastAccessTime: { "id": 1700000000, ... }
152 QJsonObject accessObj = root.value(QString::fromUtf8("lastAccessTime")).toObject();
153 for (auto it = accessObj.begin(); it != accessObj.end(); ++it) {
154 if (it.value().isDouble()) {
155 // AUDIT-P2 fix: 负数时间戳兜底为 0，与其他数值字段的范围校验保持一致，
156 // 防止手动篡改的 JSON 破坏 nextRecommended 的时间排序语义。
157 int64_t ts = static_cast<int64_t>(it.value().toDouble());
158 if (ts < 0)
159 ts = 0;
160 loaded.lastAccessTime[it.key().toStdString()] = ts;
161 }
162 }
163
164 // currentStage: int
165 // AUDIT-P1 fix: 校验上限改为 stageCount()（5），允许"全部 100% 完成"的合法状态。
166 // 原校验 s <= 4 拒绝 recomputeStage 在全通关时设置的 currentStage=5，
167 // 导致通关用户跨会话恢复时进度被重置为 0，"🎉 通关"提示丢失。
168 if (root.contains(QString::fromUtf8("currentStage"))) {
169 QJsonValue v = root.value(QString::fromUtf8("currentStage"));
170 if (v.isDouble()) {
171 int s = v.toInt();
172 if (s >= 0 && s <= LearningPathData::stageCount()) {
173 loaded.currentStage = s;
174 }
175 }
176 }
177
178 // P0-2 fix (F7): levelStars: { "token-puzzle-1": 3, ... }
179 QJsonObject starsObj = root.value(QString::fromUtf8("levelStars")).toObject();
180 for (auto it = starsObj.begin(); it != starsObj.end(); ++it) {
181 if (it.value().isDouble()) {
182 // AUDIT-P2 fix: levelStars 合法范围 [-1, 3]（-1=未完成,0=跳过,1-3=星级）
183 int v = it.value().toInt();
184 if (v >= -1 && v <= 3) {
185 loaded.levelStars[it.key().toStdString()] = v;
186 }
187 }
188 }
189
190 // P2-3 fix (F9): 学情画像多维进度数据（向后兼容——缺失字段按空 map 处理）
191 // score: { "lab-01": 85, ... }
192 QJsonObject scoreObj = root.value(QString::fromUtf8("score")).toObject();
193 for (auto it = scoreObj.begin(); it != scoreObj.end(); ++it) {
194 if (it.value().isDouble()) {
195 // AUDIT-P2 fix: score 合法范围 [0, 100]，与 recordScore 截断逻辑一致
196 int v = it.value().toInt();
197 if (v >= 0 && v <= 100) {
198 loaded.score[it.key().toStdString()] = v;
199 }
200 }
201 }
202 // bestStars: { "lab-01": 3, ... }
203 QJsonObject bestStarsObj = root.value(QString::fromUtf8("bestStars")).toObject();
204 for (auto it = bestStarsObj.begin(); it != bestStarsObj.end(); ++it) {
205 if (it.value().isDouble()) {
206 // AUDIT-P2 fix: bestStars 合法范围 [0, 3]
207 int v = it.value().toInt();
208 if (v >= 0 && v <= 3) {
209 loaded.bestStars[it.key().toStdString()] = v;
210 }
211 }
212 }
213 // spentMinutes: { "lab-01": 12, ... }
214 QJsonObject spentObj = root.value(QString::fromUtf8("spentMinutes")).toObject();
215 for (auto it = spentObj.begin(); it != spentObj.end(); ++it) {
216 if (it.value().isDouble()) {
217 // AUDIT-P2 fix: spentMinutes 不能为负，与 addSpentMinutes 拒绝负数一致
218 int v = it.value().toInt();
219 if (v >= 0) {
220 loaded.spentMinutes[it.key().toStdString()] = v;
221 }
222 }
223 }
224 // failCount: { "lab-02": 2, ... }
225 QJsonObject failObj = root.value(QString::fromUtf8("failCount")).toObject();
226 for (auto it = failObj.begin(); it != failObj.end(); ++it) {
227 if (it.value().isDouble()) {
228 // AUDIT-P2 fix: failCount 不能为负，负数会破坏 getWeakPoints 排序
229 int v = it.value().toInt();
230 if (v >= 0) {
231 loaded.failCount[it.key().toStdString()] = v;
232 }
233 }
234 }
235
236 data_ = std::move(loaded);
237 return true;

242// ============================================================
243/// 将内存中的进度数据写回文件。
244// AUDIT-P1 fix: 原子写入——写临时文件 → flush → close → remove 旧 → rename，
245// 避免崩溃/断电/磁盘满导致目标文件被截断为部分内容或空文件，全部进度丢失。
246// AUDIT-P2 fix: 写入 schemaVersion 字段，支持未来 schema 迁移。
247// R51-3 fix: 改用 QSaveFile 替代手写 remove→rename 序列。原实现先 remove 旧文件
248// 再 rename，在两步之间崩溃会导致目标文件丢失（旧文件已删、新文件未就位）。
249// QSaveFile::commit() 在 Windows 上使用 MOVEFILE_REPLACE_EXISTING 原子替换，
250// 在 Linux 上用 rename(2) 原子覆盖，消除数据丢失窗口。
251bool LearnerProgressStore::save() const {
252 QString path = filePath();
253 QFileInfo fi(path);
254 QDir().mkpath(fi.absolutePath());
255
256 QJsonObject root;
257
258 // AUDIT-P2 fix: schema 版本号，支持未来字段重命名/类型变更/语义变更的迁移
259 root.insert(QString::fromUtf8("schemaVersion"), 1);
260
261 QJsonObject completedObj;
262 for (const auto& [id, val] : data_.completed) {
263 completedObj.insert(QString::fromStdString(id), val);
264 }
265 root.insert(QString::fromUtf8("completed"), completedObj);
266
267 QJsonObject attemptObj;
268 for (const auto& [id, val] : data_.attemptCount) {
269 attemptObj.insert(QString::fromStdString(id), val);
270 }
271 root.insert(QString::fromUtf8("attemptCount"), attemptObj);
272
273 QJsonObject accessObj;
274 for (const auto& [id, val] : data_.lastAccessTime) {
275 // int64_t 转为 double 存储（JSON 数字精度）
276 accessObj.insert(QString::fromStdString(id), static_cast<double>(val));
277 }
278 root.insert(QString::fromUtf8("lastAccessTime"), accessObj);
279
280 root.insert(QString::fromUtf8("currentStage"), data_.currentStage);
281
282 // P0-2 fix (F7): 持久化细粒度关卡星级
283 QJsonObject starsObj;
284 for (const auto& [id, stars] : data_.levelStars) {
285 starsObj.insert(QString::fromStdString(id), stars);
286 }
287 root.insert(QString::fromUtf8("levelStars"), starsObj);
288
289 // P2-3 fix (F9): 持久化学情画像多维进度数据
290 QJsonObject scoreObj;
291 for (const auto& [id, sc] : data_.score) {
292 scoreObj.insert(QString::fromStdString(id), sc);
293 }
294 root.insert(QString::fromUtf8("score"), scoreObj);
295
296 QJsonObject bestStarsObj;
297 for (const auto& [id, stars] : data_.bestStars) {
298 bestStarsObj.insert(QString::fromStdString(id), stars);
299 }
300 root.insert(QString::fromUtf8("bestStars"), bestStarsObj);
301
302 QJsonObject spentObj;
303 for (const auto& [id, mins] : data_.spentMinutes) {
304 spentObj.insert(QString::fromStdString(id), mins);
305 }
306 root.insert(QString::fromUtf8("spentMinutes"), spentObj);
307
308 QJsonObject failObj;
309 for (const auto& [id, fails] : data_.failCount) {
310 failObj.insert(QString::fromStdString(id), fails);
311 }
312 root.insert(QString::fromUtf8("failCount"), failObj);
313
314 QJsonDocument doc(root);
315
316 // R51-3 fix: 使用 QSaveFile 实现真正的原子写入。
317 // QSaveFile 在 commit() 前写入临时文件，commit() 时原子替换目标文件。
318 // 若 commit 前崩溃/断电，目标文件保持原样不受影响。
319 // directWriteFallback=false：禁止 QSaveFile 在无法原子替换时直接写目标文件
320 // （直接写会在崩溃时留下截断文件）。
321 QSaveFile sf(path);
322 sf.setDirectWriteFallback(false);
323 if (!sf.open(QIODevice::WriteOnly)) {
324 // P2-UX fix: 记录写入失败原因，进度可能丢失——调用方应检查返回值并通知用户
325 LOG_WARNING("Progress save failed: cannot open file for writing: " + path.toStdString() +
326 " (error: " + sf.errorString().toStdString() + ")",
327 "Progress");
328 return false;
329 }
330 QByteArray payload = doc.toJson(QJsonDocument::Indented);
331 qint64 written = sf.write(payload);
332 if (written != static_cast<qint64>(payload.size())) {
333 sf.cancelWriting();
334 // P2-UX fix: 记录写入不完整原因（磁盘满/权限等）
335 LOG_WARNING("Progress save failed: incomplete write to " + path.toStdString() + " (wrote " +
336 std::to_string(written) + " of " + std::to_string(payload.size()) + " bytes)",
337 "Progress");
338 return false;
339 }
340 if (!sf.commit()) {
341 // P2-UX fix: 记录原子替换失败原因（目标文件被占用/权限等）
342 LOG_WARNING("Progress save failed: QSaveFile commit failed for " + path.toStdString() +
343 " (error: " + sf.errorString().toStdString() + ")",
344 "Progress");
345 return false;
346 }
347 return true;
348}

353/// 标记某活动已完成并记录时间戳。
354void LearnerProgressStore::markCompleted(const std::string& activityId) {
355 if (activityId.empty())
356 return;
357 data_.completed[activityId] = true;
358 auto now = static_cast<int64_t>(
359 std::chrono::duration_cast<std::chrono::seconds>(std::chrono::system_clock::now().time_since_epoch()).count());
360 data_.lastAccessTime[activityId] = now;
361 // P1-2 fix (F8): 完成状态变化后立即重算阶段，避免 currentStage 永远为 0。
362 // 此处调用 recomputeStage 是幂等的——多次调用只覆盖 currentStage，无副作用。
363 recomputeStage(LearningPathData::activities());
364}

366// ============================================================
367// P1-2 fix (F8): 重算 currentStage
368// 规则：
369// - 遍历 stage 0..N，找出"完成比例 >= 50%"的最高阶段 highestCompletedStage
370// - 若所有阶段都 >= 50%，currentStage = stageCount（已通关所有阶段）
371// - 否则 currentStage = highestCompletedStage + 1（下一未完成阶段）
372// - 若所有阶段都 < 50%，currentStage = 0（仍是新手阶段）
373// 注：阈值 50% 的取值平衡"完成感"与"鼓励性"——过半即解锁下一阶段提示。
374// ============================================================
375/// 依据全部活动重算各阶段完成状态。
376void LearnerProgressStore::recomputeStage(const std::vector<LearningActivity>& all) {
377 if (all.empty()) {
378 data_.currentStage = 0;
379 return;
380 }
381 int highestQualified = -1; // 完成比例 >= 50% 的最高阶段
382 for (int stage = 0; stage < LearningPathData::stageCount(); ++stage) {
383 int sp = stageProgress(stage, all);
384 if (sp >= 50) {
385 highestQualified = stage;
386 }
387 }
388 if (highestQualified < 0) {
389 // 没有任何阶段过半，仍在阶段 0
390 data_.currentStage = 0;
391 } else if (highestQualified == LearningPathData::stageCount() - 1) {
392 // 最后一阶段也过半——若已 100% 则标记为 stageCount（通关），否则停留在最后阶段
393 int lastStageProgress = stageProgress(highestQualified, all);
394 data_.currentStage = (lastStageProgress >= 100) ? LearningPathData::stageCount() : highestQualified;
395 } else {
396 // 下一未完成阶段
397 data_.currentStage = highestQualified + 1;
398 }
399}

429/// 为某关卡记录星级（0-3）。
430void LearnerProgressStore::markLevelStars(const std::string& levelId, int stars) {
431 if (levelId.empty())
432 return;
433 // 仅当新星级 >= 已记录星级时才覆盖（保留历史最佳成绩）
434 auto it = data_.levelStars.find(levelId);

436 data_.levelStars[levelId] = stars;
437 }
438 // stars >= 0 表示关卡至少被完成或跳过，同步记录访问时间
439 if (stars >= 0) {

441 std::chrono::duration_cast<std::chrono::seconds>(std::chrono::system_clock::now().time_since_epoch())
442 .count());
443 data_.lastAccessTime[levelId] = now;

471/// 记录某活动得分与星级。
472void LearnerProgressStore::recordScore(const std::string& activityId, int score, int stars) {
473 if (activityId.empty())
474 return;
475 // score 截断到 [0, 100]
476 if (score < 0)
477 score = 0;
478 if (score > 100)
479 score = 100;
480 // 保留历史最佳得分（仅当新分 >= 旧分时覆盖）
481 auto it = data_.score.find(activityId);
482 if (it == data_.score.end() || score > it->second) {
483 data_.score[activityId] = score;
484 }
485 // 保留历史最佳星级（仅当新星级 >= 旧星级时覆盖）
486 if (stars > 0) {
487 auto sit = data_.bestStars.find(activityId);
488 if (sit == data_.bestStars.end() || stars > sit->second) {
489 data_.bestStars[activityId] = stars;
490 }
491 }
492 // 同步更新访问时间
493 auto now = static_cast<int64_t>(
494 std::chrono::duration_cast<std::chrono::seconds>(std::chrono::system_clock::now().time_since_epoch()).count());
495 data_.lastAccessTime[activityId] = now;
496}

542// ============================================================
543// P2-3 fix (F9): 薄弱点分析
544// 算法：
545// 1. 遍历所有活动，筛选"未完成 + (attemptCount >= minAttempts 或 failCount >= 1)"
546// 2. 按 failCount 降序 → attemptCount 降序 → id 字典序排序
547// 3. 截取前 maxCount 个返回（maxCount=0 表示不限制）
548// ============================================================
549/// 基于失败/低星识别薄弱点列表。
550std::vector<WeakPoint> LearnerProgressStore::getWeakPoints(const std::vector<LearningActivity>& all, int minAttempts,
551 std::size_t maxCount) const {
552
553 std::vector<WeakPoint> candidates;
554 for (const auto& a : all) {
555 // 已完成则跳过——薄弱点仅针对未通过活动
556 auto cit = data_.completed.find(a.id);
557 if (cit != data_.completed.end() && cit->second)
558 continue;
559
560 int attempts = 0;
561 auto ait = data_.attemptCount.find(a.id);
562 if (ait != data_.attemptCount.end())
563 attempts = ait->second;
564
565 int fails = 0;
566 auto fit = data_.failCount.find(a.id);
567 if (fit != data_.failCount.end())
568 fails = fit->second;
569
570 int sc = 0;
571 auto sit = data_.score.find(a.id);
572 if (sit != data_.score.end())
573 sc = sit->second;
574
575 // 筛选条件：尝试次数 >= 阈值 或 至少失败过一次
576 if (attempts >= minAttempts || fails >= 1) {
577 candidates.push_back(WeakPoint{a.id, attempts, fails, sc});
578 }
579 }
580
581 // 排序：失败次数降序 → 尝试次数降序 → id 字典序
582 std::sort(candidates.begin(), candidates.end(), [](const WeakPoint& a, const WeakPoint& b) {
583 if (a.fails != b.fails)
584 return a.fails > b.fails;
585 if (a.attempts != b.attempts)
586 return a.attempts > b.attempts;
587 return a.activityId < b.activityId;
588 });
589
590 // 截取前 maxCount 个（maxCount=0 表示不限制）
591 if (maxCount > 0 && candidates.size() > maxCount) {
592 candidates.resize(maxCount);
593 }
594 return candidates;
595}

598// P2-3 fix (F9): 预计剩余时间
599// 算法：sum(未完成且已解锁活动的 estimatedMinutes)
600// 注：未解锁活动不计入——因为它们当前无法开始，不应计入"剩余"。
601// ============================================================
602/// 估算完成剩余活动所需分钟数。
603int LearnerProgressStore::estimatedRemainingMinutes(const std::vector<LearningActivity>& all) const {
604 int total = 0;
605 for (const auto& a : all) {
606 // 已完成则跳过
607 auto cit = data_.completed.find(a.id);
608 if (cit != data_.completed.end() && cit->second)
609 continue;
610
611 // 未解锁则跳过（不计入剩余时间）
612 if (!isUnlocked(a.id, all))
613 continue;
614
615 total += a.estimatedMinutes;
616 }
617 return total;
618}

676/// 返回下一个推荐学习活动的 id/描述。
677std::string LearnerProgressStore::nextRecommended(const std::vector<LearningActivity>& all) const {
678
679 // 1. 筛选候选：未完成 + 已解锁
680 // 学习限制已移除：isUnlocked 统一返回 true（除无效 id），所有未完成活动均为候选。
681 std::vector<const LearningActivity*> candidates;
682 for (const auto& a : all) {
683 // 已完成则跳过
684 auto cit = data_.completed.find(a.id);
685 if (cit != data_.completed.end() && cit->second)
686 continue;
687
688 if (isUnlocked(a.id, all))
689 candidates.push_back(&a);
690 }
691
692 if (candidates.empty())
693 return std::string();
694
695 // 2. 排序：stage 升序 → attemptCount 升序 → estimatedMinutes 升序 → id 字典序
696 auto attemptOf = [this](const std::string& id) -> int {
697 auto it = data_.attemptCount.find(id);
698 return it != data_.attemptCount.end() ? it->second : 0;
699 };
700
701 std::sort(candidates.begin(), candidates.end(), [&attemptOf](const LearningActivity* a, const LearningActivity* b) {
702 if (a->stage != b->stage)
703 return a->stage < b->stage;
704 int aa = attemptOf(a->id);
705 int ab = attemptOf(b->id);
706 if (aa != ab)
707 return aa < ab;
708 if (a->estimatedMinutes != b->estimatedMinutes)
709 return a->estimatedMinutes < b->estimatedMinutes;
710 return a->id < b->id;
711 });
712
713 return candidates.front()->id;
714}
