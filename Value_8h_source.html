144 /// DictKey 的 hash 函数：按 variant index 区分类型，避免 bool(true) 与 int(1) 冲突。
145 /// float 用位模式 hash，避免 std::hash<double> 在不同平台 NaN/-0.0 行为不一致。
146 /// R97 #2 fix: 改为透明 hash（C++20 is_transparent），支持 unordered_map::find<string_view>
147 /// 等异构查找，避免查找路径构造完整 DictKey（含 string 拷贝）的开销。
148 /// 关键不变量：透明路径（如 string_view）与 DictKey 路径（DictKey{string}）必须产生
149 /// 相同 hash，否则 find<string_view> 找不到桶。本实现用"类型前缀 XOR value hash"模式：
150 /// DictKey{string} 的 hash = kStringTag ^ hash(string)，与 find(string_view) 一致。
151 struct DictKeyHash {
152 using is_transparent = void; // 启用 C++20 透明查找
153
154 // 类型标签（避免不同类型相同值的 hash 碰撞，如 bool(true) vs int(1)）
155 // 用斐波那契哈希常数的不同位移区分类型，确保 4 类键的 hash 空间基本不重叠
156 static constexpr size_t kStringTag = 0x9E3779B9; // 黄金分割常数
157 static constexpr size_t kIntTag = 0x6A09E667; // sqrt(2) 的高位
158 static constexpr size_t kBoolTag = 0xBB67AE85; // sqrt(3) 的高位
159 static constexpr size_t kDoubleTag = 0x3C6EF372; // sqrt(5) 的高位
160
161 // 透明路径
162 size_t operator()(const std::string& s) const { return kStringTag ^ std::hash<std::string>{}(s); }
163 size_t operator()(const std::string_view& sv) const { return kStringTag ^ std::hash<std::string_view>{}(sv); }
164 size_t operator()(int64_t v) const { return kIntTag ^ std::hash<int64_t>{}(v); }
165 size_t operator()(bool v) const { return kBoolTag ^ std::hash<bool>{}(v); }
166 size_t operator()(double v) const {
167 if (v == 0.0)
168 v = 0.0; // P0 fix: 规范化 -0.0 → +0.0，保证 hash 不变量
169 uint64_t bits;
170 std::memcpy(&bits, &v, sizeof(double));
171 return kDoubleTag ^ std::hash<uint64_t>{}(bits);
172 }
173 // DictKey 路径（必须与透明路径产生相同 hash）
174 size_t operator()(const DictKey& k) const {
175 return std::visit(
176 [&](const auto& v) -> size_t {
177 using T = std::decay_t<decltype(v)>;
178 if constexpr (std::is_same_v<T, std::string>)
179 return kStringTag ^ std::hash<std::string>{}(v);
180 else if constexpr (std::is_same_v<T, int64_t>)
181 return kIntTag ^ std::hash<int64_t>{}(v);
182 else if constexpr (std::is_same_v<T, bool>)
183 return kBoolTag ^ std::hash<bool>{}(v);
184 else {
185 double dv = v;
186 if (dv == 0.0)
187 dv = 0.0; // P0 fix: 规范化 -0.0 → +0.0
188 uint64_t bits;
189 std::memcpy(&bits, &dv, sizeof(double));
190 return kDoubleTag ^ std::hash<uint64_t>{}(bits);
191 }
192 },
193 k);
194 }
195 };

515 template <typename T> T* ensureUnique() {
516 T* ptr = box_.asPtr<T>();
517 if (!ptr->isUnique()) {
518 // 引用计数 > 1：深拷贝一份新数据
519 // P0 fix: 用 unique_ptr 包裹 cloned，防止 new T(*ptr) 抛 bad_alloc 时
520 // 部分已克隆的子树泄漏（如 ArrayData/DictData/InstanceData 的 cloneImpl
521 // 递归克隆嵌套 Value，中途抛异常会泄漏已分配的子对象）。
522 auto cloned = std::make_unique<T>(*ptr); // 拷贝构造（RefCounted 拷贝 ctor 重置 refCount=1）
523 // AUDIT-BUG-C1 fix: 拷贝构造不会调用 GcManager::registerTracked（仅显式构造函数调用）。
524 // COW 克隆的容器必须注册到 GcManager，否则循环引用（如 b.push(b) 后 COW detach）
525 // 不会被 collectCycle 回收，导致永久内存泄漏。
526 // AUDIT-P2-CORRECT fix: 在 release 旧引用前注册到 GcManager，
527 // 若 registerTracked 抛 bad_alloc，unique_ptr 自动 delete cloned，
528 // 旧引用计数不变（ptr->release() 未执行），仍由调用方持有，无双重释放风险。
529 // 原实现先 release/transfer 所有权再 registerTracked，若 registerTracked 抛异常，
530 // raw 泄漏（已脱离 unique_ptr）且 box_ 仍指向旧 ptr（release 已执行→双重释放）。
531 if constexpr (std::is_same_v<T, ArrayData> || std::is_same_v<T, DictData> ||
532 std::is_same_v<T, InstanceData>) {
533 GcManager::instance().registerTracked(cloned.get());
534 }
535 ptr->release(); // 释放旧引用
536 T* raw = cloned.release();
537 box_ = NaNBox::fromPtr(static_cast<const void*>(raw));
538 return raw;
539 }
540 return ptr;
541 }

686 // 使新 Value 在 RAII 下正确管理引用计数（析构会 release）。

688 // - bitsOf：提取 Value 的 raw bits，不修改引用计数。
689 // 原 Value 仍正常管理自己的引用计数，调用方拿到的 bits 可后续用

714 /// 递归深拷贝实现。
715 /// cloned 映射：旧 RefCounted* → 已克隆的新 Value，用于打断环形 back-edge
716 /// 和避免 DAG 重复克隆。遇到已克隆节点直接返回新引用（addRef）。
717 Value cloneImpl(int depth, std::unordered_map<const void*, Value>& cloned) const {
718 if (!box_.isPointer()) {
719 Value result;
720 result.box_ = box_;
721 return result;
722 }
723 auto* rc = box_.asPtr<RefCounted>();
724 // 深度保护：超限时返回浅拷贝（保留原引用），避免栈溢出
725 if (depth >= MAX_CLONE_DEPTH) {
726 return *this; // 拷贝构造 + addRef
727 }
728 // 环检测：遇到已克隆节点直接复用（addRef 后返回）
729 auto it = cloned.find(rc);
730 if (it != cloned.end()) {
731 return it->second; // 拷贝构造 + addRef
732 }
733
734 switch (rc->type) {
735 case ValueType::VAL_INT: {
736 auto* p = static_cast<BoxedIntData*>(rc);
737 return fromHeapPtr(new BoxedIntData(p->value));
738 }
739 case ValueType::VAL_STRING: {
740 auto* p = static_cast<StringData*>(rc);
741 return fromHeapPtr(new StringData(p->value));
742 }
743 case ValueType::VAL_ARRAY: {
744 auto* p = static_cast<ArrayData*>(rc);
745 auto* newPtr = new ArrayData();
746 Value result = fromHeapPtr(newPtr);
747 // 先登记到 cloned，再递归克隆元素——这样环 back-edge
748 // 在递归遇到此节点时能从 cloned 取到新引用。
749 cloned.emplace(rc, result);
750 newPtr->elements.reserve(p->elements.size());
751 for (const auto& elem : p->elements) {
752 newPtr->elements.push_back(elem.cloneImpl(depth + 1, cloned));
753 }
754 return result;
755 }
756 case ValueType::VAL_DICT: {
757 auto* p = static_cast<DictData*>(rc);
758 auto* newPtr = new DictData();
759 Value result = fromHeapPtr(newPtr);
760 cloned.emplace(rc, result);
761 newPtr->entries.reserve(p->entries.size());
762 for (const auto& kv : p->entries) {
763 newPtr->entries.emplace(kv.first, kv.second.cloneImpl(depth + 1, cloned));
764 }
765 return result;
766 }
767 case ValueType::VAL_INSTANCE: {
768 auto* p = static_cast<InstanceData*>(rc);
769 auto* newPtr = new InstanceData(p->className);
770 Value result = fromHeapPtr(newPtr);
771 cloned.emplace(rc, result);
772 newPtr->fields.reserve(p->fields.size());
773 for (const auto& kv : p->fields) {
774 newPtr->fields.emplace(kv.first, kv.second.cloneImpl(depth + 1, cloned));
775 }
776 return result;
777 }
778 case ValueType::VAL_CLOSURE: {
779 auto* p = static_cast<ClosureData*>(rc);
780 auto* newPtr = new ClosureData(*p); // 浅拷贝 env/params/body/vmClosure
781 Value result = fromHeapPtr(newPtr);
782 cloned.emplace(rc, result);
783 newPtr->capturedVars.clear();
784 for (const auto& kv : p->capturedVars) {
785 newPtr->capturedVars.emplace(kv.first, kv.second.cloneImpl(depth + 1, cloned));
786 }
787 return result;
788 }
789 case ValueType::VAL_TUPLE: {
790 // R98 元组与解构：immutable 元组深拷贝
791 auto* p = static_cast<TupleData*>(rc);
792 auto* newPtr = new TupleData();
793 Value result = fromHeapPtr(newPtr);
794 cloned.emplace(rc, result);
795 newPtr->elements.reserve(p->elements.size());
796 for (const auto& elem : p->elements) {
797 newPtr->elements.push_back(elem.cloneImpl(depth + 1, cloned));
798 }
799 return result;
800 }
801 case ValueType::VAL_ENUM_VARIANT: {
802 // R99 枚举与 ADT：immutable enum variant 深拷贝
803 auto* p = static_cast<EnumVariantData*>(rc);
804 auto* newPtr = new EnumVariantData();
805 newPtr->enumName = p->enumName;
806 newPtr->variantName = p->variantName;
807 Value result = fromHeapPtr(newPtr);
808 cloned.emplace(rc, result);
809 newPtr->fields.reserve(p->fields.size());
810 for (const auto& f : p->fields) {
811 newPtr->fields.push_back(f.cloneImpl(depth + 1, cloned));
812 }
813 return result;
