89/// 获取 RegOp 元数据
90/// A3 fix: 单一数据源，替代原 regOpName switch + instructionSize switch 两处维护
91const RegOpInfo& getRegOpInfo(RegOp op) {
92 auto idx = static_cast<uint8_t>(op);
93 static const RegOpInfo kUnknown{"UNKNOWN", 1, false};
94 if (idx < sizeof(kRegOpInfo) / sizeof(kRegOpInfo[0])) {
95 return kRegOpInfo[idx];
96 }
97 return kUnknown;
98}

102uint16_t RegBytecodeChunk::addConstant(const Value& value) {
103 // 简单线性扫描去重（常量池通常较小）
104 // BUG-CP-1 fix: 类型严格匹配，避免 Value::equals() 跨类型数值相等性
105 // （如 Value(0).equals(Value(0.0)) == true）导致 int/float 常量被错误去重。
106 for (size_t i = 0; i < constants.size(); ++i) {
107 if (constants[i].getType() == value.getType() && constants[i].equals(value)) {
108 return static_cast<uint16_t>(i);
109 }
110 }
111 if (constants.size() >= 65535) {
112 // 超出索引上限，抛异常（与 BytecodeChunk 行为一致）
113 throw std::runtime_error("常量池索引超出 65535 上限");
114 }
115 constants.push_back(value);
116 return static_cast<uint16_t>(constants.size() - 1);
117}

160size_t RegBytecodeChunk::instructionSizeAt(size_t offset) const {
161 if (offset >= code.size())
162 return 1;
163 RegOp op = static_cast<RegOp>(code[offset]);
164 size_t baseSize = instructionSize(op);
165
166 // 变长指令：根据操作数计算实际长度
167 switch (op) {
168 case RegOp::REG_CALL: {
169 // op + dst + nameIdx(2B) + argCount + args
170 if (offset + 4 < code.size()) {
171 uint8_t argCount = code[offset + 4];
172 return static_cast<size_t>(5 + argCount);
173 }
174 return baseSize;
175 }
176 case RegOp::REG_TAIL_CALL: {
177 // L18 eng-tailcall: 布局同 REG_CALL
178 if (offset + 4 < code.size()) {
179 uint8_t argCount = code[offset + 4];
180 return static_cast<size_t>(5 + argCount);
181 }
182 return baseSize;
183 }
184 case RegOp::REG_CALL_EXPR: {
185 // op + dst + callee + argCount + args
186 if (offset + 3 < code.size()) {
187 uint8_t argCount = code[offset + 3];
188 return static_cast<size_t>(4 + argCount);
189 }
190 return baseSize;
191 }
192 case RegOp::REG_METHOD_CALL: {
193 // op + dst + obj + methodIdx(2B) + argCount + args
194 if (offset + 5 < code.size()) {
195 uint8_t argCount = code[offset + 5];
196 return static_cast<size_t>(6 + argCount);
197 }
198 return baseSize;
199 }
200 case RegOp::REG_BUILD_ARRAY: {
201 // op + dst + count + elems
202 if (offset + 2 < code.size()) {
203 uint8_t count = code[offset + 2];
204 return static_cast<size_t>(3 + count);
205 }
206 return baseSize;
207 }
208 case RegOp::REG_BUILD_DICT: {
209 // op + dst + pairCount + (k, v)×pairCount
210 if (offset + 2 < code.size()) {
211 uint8_t pairCount = code[offset + 2];
212 return static_cast<size_t>(3 + static_cast<size_t>(pairCount) * 2);
213 }
214 return baseSize;
215 }
216 case RegOp::REG_BUILD_TUPLE: {
217 // R98 元组与解构：op + dst + count + elems（与 REG_BUILD_ARRAY 结构相同）
218 if (offset + 2 < code.size()) {
219 uint8_t count = code[offset + 2];
220 return static_cast<size_t>(3 + count);
221 }
222 return baseSize;
223 }
224 case RegOp::REG_BUILD_ENUM_VARIANT: {
225 // R99 枚举与 ADT：op + dst + enumNameConstIdx(2B) + variantNameConstIdx(2B) + argCount + args
226 // 布局: [op, dst, eLo, eHi, vLo, vHi, argCount, arg1, arg2, ...]
227 if (offset + 6 < code.size()) {
228 uint8_t argCount = code[offset + 6];
229 return static_cast<size_t>(7 + argCount);
230 }
231 return baseSize;
232 }
233 case RegOp::REG_MAKE_CLOSURE: {
234 // op + dst + nameIdx(2B) + uvCount + [isLocal, idx]×uvCount
235 if (offset + 4 < code.size()) {
236 uint8_t uvCount = code[offset + 4];
237 return static_cast<size_t>(5 + static_cast<size_t>(uvCount) * 2);
238 }
239 return baseSize;
240 }
241 case RegOp::REG_CLASS_NEW: {
242 // op + dst + nameIdx(2B) + argCount + args
243 if (offset + 4 < code.size()) {
244 uint8_t argCount = code[offset + 4];
245 return static_cast<size_t>(5 + argCount);
246 }
247 return baseSize;
248 }
249 case RegOp::REG_SUPER_CALL: {
250 // op + dst + nameIdx(2B) + argCount + recvReg + classIdx(2B) + args
251 if (offset + 7 < code.size()) {
252 uint8_t argCount = code[offset + 4];
253 return static_cast<size_t>(8 + argCount);
254 }
255 return baseSize;
256 }
257 case RegOp::REG_DEFINE_CLASS: {
258 // C-9 fix: op + nameIdx(2B) + parentIdx(2B) + fieldCount(1B) + [fieldIdx(2B)×F]
259 // + methodCount(1B) + [methodIdx(2B)+funIdx(2B)]×M
260 // BUG-INH-1 fix: 每个字段新增 defaultConstIdx(2B)，故字段段为 [fieldIdx(2B)+defaultIdx(2B)]×F
261 // BUG-INH-IR-1 fix: 每个字段再新增 exprReg(1B)，故字段段为 [fieldIdx(2B)+defaultIdx(2B)+exprReg(1B)]×F
262 // 新布局：op + nameIdx(2B) + parentIdx(2B) + fieldCount(1B)
263 // + [fieldIdx(2B)+defaultIdx(2B)+exprReg(1B)]×F
264 // + methodCount(1B) + [methodIdx(2B)+funIdx(2B)]×M
265 // 最小 7B；需读取 fieldCount（offset+5）和 methodCount（offset+6+F*5）
266 if (offset + 6 < code.size()) {
267 uint8_t fieldCount = code[offset + 5];
268 size_t methodCountPos = offset + 6 + static_cast<size_t>(fieldCount) * 5;
269 if (methodCountPos < code.size()) {
270 uint8_t methodCount = code[methodCountPos];
271 return static_cast<size_t>(7 + static_cast<size_t>(fieldCount) * 5 +
272 static_cast<size_t>(methodCount) * 4);
273 }
274 }
275 return baseSize;
276 }
277 default:
278 return baseSize;
279 }

294// L11 预编译模块：全局槽位重定位
295// 指令格式：
296// REG_LOAD_GLOBAL: op(1B) + dst(1B) + slot(2B LE) → slot at offset+2..offset+3
297// REG_STORE_GLOBAL: op(1B) + src(1B) + slot(2B LE) → slot at offset+2..offset+3
298// REG_DEFINE_GLOBAL: op(1B) + src(1B) + slot(2B LE) → slot at offset+2..offset+3
299// REG_DELETE_GLOBAL: op(1B) + nameConstIdx(2B LE) → 常量池索引，不是 slot，跳过
300void RegBytecodeChunk::relocateGlobalSlots(const std::vector<int>& relocationMap) {
301 size_t offset = 0;
302 while (offset < code.size()) {
303 if (offset >= code.size())
304 break;
305 RegOp op = static_cast<RegOp>(code[offset]);
306 // 仅全局槽位指令需要重定位（REG_DELETE_GLOBAL 用 nameConstIdx，不是 slot）
307 if (op == RegOp::REG_LOAD_GLOBAL || op == RegOp::REG_STORE_GLOBAL || op == RegOp::REG_DEFINE_GLOBAL) {
308 // op(1B) + reg(1B) + slot(2B LE) → slot 在 offset+2..offset+3
309 if (offset + 3 < code.size()) {
310 uint16_t slot =
311 static_cast<uint16_t>(code[offset + 2]) | (static_cast<uint16_t>(code[offset + 3]) << 8);
312 if (slot < relocationMap.size() && relocationMap[slot] >= 0) {
313 int newSlot = relocationMap[slot];
314 code[offset + 2] = static_cast<uint8_t>(newSlot & 0xFF);
315 code[offset + 3] = static_cast<uint8_t>((newSlot >> 8) & 0xFF);
316 }
317 }
318 }
319 offset += instructionSizeAt(offset);
320 }
