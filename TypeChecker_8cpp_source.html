21// 辅助：递归遍历 AST，检查 VarDecl 和 Assignment 的字面量类型匹配
22/// 字面量类型检查遍历器。在编译期为带类型注解的变量声明/赋值做"字面量 vs 注解"
23/// 的一致性检查：仅当初始化器是字面量（NumberLiteral/StringLiteral/BoolLiteral/NullLiteral）
24/// 时才能静态判定类型，非字面量表达式交由运行时 OP_TYPE_CHECK 处理。
25/// 通过 varTypes 表在作用域间传播注解；visit* 各语句对块/分支/循环/函数体
26/// 使用"保存-恢复"模式隔离作用域，避免内层注解污染外层（AUDIT-BUG-E1）。
27class LiteralTypeWalker : public DefaultVisitor {
28public:
29 DiagnosticBag diagnostics;
30 std::unordered_map<std::string, std::string> varTypes; // 变量名→类型注解
31 std::unordered_map<std::string, TypeInfo> varTypeInfos; // 变量名→推断类型
32 std::unordered_map<std::string, std::string> funcReturnTypes; // 函数名→返回类型注解
33
34 /// 从表达式 AST 节点推断类型
35 TypeInfo inferExprType(ASTNode* expr) {
36 if (!expr)
37 return TypeInfo{TypeKind::UNKNOWN};
38 // 字面量
39 if (auto* num = dynamic_cast<NumberLiteral*>(expr))
40 return TypeInfo{num->isFloat_ ? TypeKind::FLOAT : TypeKind::INT};
41 if (dynamic_cast<StringLiteral*>(expr))
42 return TypeInfo{TypeKind::STRING};
43 if (dynamic_cast<BoolLiteral*>(expr))
44 return TypeInfo{TypeKind::BOOL};
45 if (dynamic_cast<NullLiteral*>(expr))
46 return TypeInfo{TypeKind::NULL_T};
47 if (dynamic_cast<TupleLiteral*>(expr))
48 return TypeInfo{TypeKind::ARRAY}; // tuple 视为 array 子类型
49 // 变量引用：查符号表
50 if (auto* id = dynamic_cast<VarRef*>(expr)) {
51 auto it = varTypeInfos.find(id->name);
52 if (it != varTypeInfos.end())
53 return it->second;
54 return TypeInfo{TypeKind::UNKNOWN};
55 }
56 // 二元运算
57 if (auto* bin = dynamic_cast<BinaryOp*>(expr)) {
58 TypeInfo lhs = inferExprType(bin->left.get());
59 TypeInfo rhs = inferExprType(bin->right.get());
60 // 字符串拼接
61 if (bin->opType == BinOpType::BIN_ADD && (lhs.kind == TypeKind::STRING || rhs.kind == TypeKind::STRING))
62 return TypeInfo{TypeKind::STRING};
63 // 数值运算：有 float 则结果为 float
64 if (lhs.isNumeric() && rhs.isNumeric()) {
65 if (lhs.kind == TypeKind::FLOAT || rhs.kind == TypeKind::FLOAT)
66 return TypeInfo{TypeKind::FLOAT};
67 return TypeInfo{TypeKind::INT};
68 }
69 // 比较运算符返回 bool
70 if (bin->opType == BinOpType::BIN_EQ || bin->opType == BinOpType::BIN_NEQ ||
71 bin->opType == BinOpType::BIN_LT || bin->opType == BinOpType::BIN_GT ||
72 bin->opType == BinOpType::BIN_LTE || bin->opType == BinOpType::BIN_GTE)
73 return TypeInfo{TypeKind::BOOL};
74 // 逻辑运算符返回操作数类型（MiniLang 短路返回原值）
75 if (bin->opType == BinOpType::BIN_AND || bin->opType == BinOpType::BIN_OR)
76 return lhs;
77 return TypeInfo{TypeKind::UNKNOWN};
78 }
79 // 一元运算
80 if (auto* unary = dynamic_cast<UnaryOp*>(expr)) {
81 TypeInfo operand = inferExprType(unary->operand.get());
82 if (unary->opType == UnaryOp::UnaryOpType::UOP_NEGATE)
83 return operand; // 保持数值类型
84 if (unary->opType == UnaryOp::UnaryOpType::UOP_NOT)
85 return TypeInfo{TypeKind::BOOL};
86 return operand;
87 }
88 // 函数调用：查函数返回类型注解
89 if (auto* call = dynamic_cast<FunCall*>(expr)) {
90 auto it = funcReturnTypes.find(call->name);
91 if (it != funcReturnTypes.end())
92 return TypeInfo::fromAnnotation(it->second);
93 return TypeInfo{TypeKind::UNKNOWN};
94 }
95 return TypeInfo{TypeKind::UNKNOWN};
96 }

98 /// 检查单个变量声明的字面量类型匹配。
99 /// 无注解或无初始化器时仅登记注解供后续 Assignment 复用；
100 /// 否则用 dynamic_cast 判定初始化器字面量类型，与注解比对，不匹配则发警告。
101 /// NullLiteral 兼容任何注解（视为通过）。最终把注解写入 varTypes。
102 void checkVarDecl(VarDecl& node) {
103 if (node.typeAnnotation.empty() || !node.initializer) {
104 // 无注解或无初始化器：仅记录注解供 Assignment 用
105 if (!node.typeAnnotation.empty()) {
106 varTypes[node.name] = node.typeAnnotation;
107 varTypeInfos[node.name] = TypeInfo::fromAnnotation(node.typeAnnotation);
108 } else if (node.initializer) {
109 // 无注解但有初始化器：从初始化表达式推断类型
110 TypeInfo inferred = inferExprType(node.initializer.get());
111 if (inferred.kind != TypeKind::UNKNOWN)
112 varTypeInfos[node.name] = inferred;
113 }
114 return;
115 }
116 // 有注解+有初始化器：用 inferExprType 推断右值类型并检查兼容性
117 TypeInfo declaredType = TypeInfo::fromAnnotation(node.typeAnnotation);
118 TypeInfo actualType = inferExprType(node.initializer.get());
119 if (actualType.kind != TypeKind::UNKNOWN && actualType.kind != TypeKind::NULL_T) {
120 if (!declaredType.isCompatible(actualType)) {
121 diagnostics.addWarning("变量 " + node.name + " 类型注解为 " + node.typeAnnotation + "，但初始化值为 " +
122 actualType.toString(),
123 node.line, node.column, DiagSource::TypeChecker);
124 }
125 }
126 varTypes[node.name] = node.typeAnnotation;
127 varTypeInfos[node.name] = declaredType;
128 }
129
130 /// 访问变量声明：先执行字面量类型检查，再递归遍历初始化表达式
131 /// （初始化器可能是嵌套声明/块，需一并检查）。
132 void visitVarDecl(VarDecl& node) override {
133 checkVarDecl(node);
134 // 递归检查初始化表达式（可能含嵌套声明）
135 if (node.initializer)
136 node.initializer->accept(*this);
137 }
138
139 /// 访问赋值语句：若左值变量有已知注解，检查右侧字面量是否兼容。
140 /// 非字面量右侧跳过（运行时检查）。随后递归遍历右侧表达式。
141 void visitAssignment(Assignment& node) override {
142 auto it = varTypes.find(node.name);
143 if (it != varTypes.end()) {
144 // 有类型注解：用 inferExprType 推断右值类型并检查兼容性
145 TypeInfo declaredType = TypeInfo::fromAnnotation(it->second);
146 TypeInfo actualType = inferExprType(node.value.get());
147 if (actualType.kind != TypeKind::UNKNOWN && actualType.kind != TypeKind::NULL_T) {
148 if (!declaredType.isCompatible(actualType)) {
149 diagnostics.addWarning("赋值给 " + node.name + " 类型注解为 " + it->second + "，但赋值值为 " +
150 actualType.toString(),
151 node.line, node.column, DiagSource::TypeChecker);
152 }
153 }
154 }
155 if (node.value)
156 node.value->accept(*this);
157 }

159 void visitBlock(Block& node) override {
160 // AUDIT-BUG-E1 fix: 块作用域隔离——保存/恢复 varTypes，
161 // 防止内层块（if/while/for body）的变量类型注解污染外层。
162 // visitFunDecl 已有此模式，此处对齐。
163 auto saved = varTypes;
164 auto savedInfos = varTypeInfos; // P1 #27 fix: 同时保存/恢复 varTypeInfos，防止块作用域推断类型泄漏
165 for (auto& stmt : node.statements) {
166 if (stmt)
167 stmt->accept(*this);
168 }
169 varTypes = std::move(saved);
170 varTypeInfos = std::move(savedInfos);
171 }
172
173 void visitIfStmt(IfStmt& node) override {
174 if (node.condition)
175 node.condition->accept(*this);
176 // AUDIT-BUG-E1 fix: then/else 分支是独立作用域
177 auto saved = varTypes;
178 auto savedInfos = varTypeInfos; // P2 #63 fix: 对齐 visitBlock，同时保存/恢复 varTypeInfos
179 if (node.thenBranch)
180 node.thenBranch->accept(*this);
181 varTypes = saved;
182 varTypeInfos = savedInfos; // P2 #63 fix: 恢复 then 分支前的类型推断状态
183 if (node.elseBranch)
184 node.elseBranch->accept(*this);
185 varTypes = std::move(saved);
186 varTypeInfos = std::move(savedInfos); // P2 #63 fix: 恢复外层作用域
187 }

189 void visitWhileStmt(WhileStmt& node) override {
190 if (node.condition)
191 node.condition->accept(*this);
192 // AUDIT-BUG-E1 fix: 循环体是独立作用域
193 auto saved = varTypes;
194 if (node.body)
195 node.body->accept(*this);
196 varTypes = std::move(saved);
197 }

199 void visitForStmt(ForStmt& node) override {
200 // AUDIT-P3-CORRECT fix: for 循环 initializer/condition/update/body 共享一个作用域。
201 // 原实现（AUDIT-BUG-E1 fix）错误地将各部分隔离，导致 initializer 声明的变量
202 // （如 `var i: int = 0`）在 condition/update/body 中不可见，类型检查被跳过。
203 // MiniLang（与 C/Java/JS 一致）中 for 循环四部分共享同一作用域，
204 // 循环结束后该作用域销毁（i 不在外层可见）。
205 auto saved = varTypes;
206 if (node.initializer)
207 node.initializer->accept(*this);
208 // 不恢复 varTypes — initializer 声明的变量在 condition/update/body 中可见
209 if (node.condition)
210 node.condition->accept(*this);
211 if (node.update)
212 node.update->accept(*this);
213 if (node.body)
214 node.body->accept(*this);
215 varTypes = std::move(saved); // 循环结束后恢复，i 不在外层可见
216 }
217
218 void visitFunDecl(FunDecl& node) override {
219 // 记录函数返回类型注解（供 inferExprType 查询）
220 if (!node.returnType.empty()) {
221 funcReturnTypes[node.name] = node.returnType;
222 }
223 // 保存外层 varTypes/varTypeInfos，函数体内独立作用域
224 auto savedVarTypes = varTypes;
225 auto savedVarTypeInfos = varTypeInfos;
226 // 注册参数类型到符号表
227 for (size_t i = 0; i < node.params.size() && i < node.paramTypes.size(); ++i) {
228 if (!node.paramTypes[i].empty()) {
229 varTypes[node.params[i]] = node.paramTypes[i];
230 varTypeInfos[node.params[i]] = TypeInfo::fromAnnotation(node.paramTypes[i]);
231 }
232 }
233 // 返回类型检查：如果有 returnType 注解，检查函数体中 ReturnStmt 的返回值类型
234 std::string currentReturnType;
235 if (!node.returnType.empty()) {
236 currentReturnType = node.returnType;
237 }
238 // 设置当前函数返回类型上下文（用于 visitReturnStmt）
239 auto savedReturnType = currentCheckingReturnType_;
240 currentCheckingReturnType_ = currentReturnType;
241 if (node.body)
242 node.body->accept(*this);
243 currentCheckingReturnType_ = savedReturnType;
244 varTypes = std::move(savedVarTypes);
245 varTypeInfos = std::move(savedVarTypeInfos);
246 }
247
248 void visitReturnStmt(ReturnStmt& node) override {
249 // 检查返回值类型与当前函数的 returnType 注解是否兼容
250 if (!currentCheckingReturnType_.empty() && node.value) {
251 TypeInfo declaredReturn = TypeInfo::fromAnnotation(currentCheckingReturnType_);
252 TypeInfo actualReturn = inferExprType(node.value.get());
253 if (actualReturn.kind != TypeKind::UNKNOWN && actualReturn.kind != TypeKind::NULL_T) {
254 if (!declaredReturn.isCompatible(actualReturn)) {
255 diagnostics.addWarning("函数返回类型注解为 " + currentCheckingReturnType_ + "，但返回值为 " +
256 actualReturn.toString(),
257 node.line, node.column, DiagSource::TypeChecker);
258 }
259 }
260 }
261 if (node.value)
262 node.value->accept(*this);
263 }

265 void visitClassDecl(ClassDecl& node) override {
266 for (auto& m : node.members) {
267 if (m)
268 m->accept(*this);
269 }
270 }

272 // BUG-FE-AUDIT-3 fix: try/catch/finally 三个块均为独立作用域，需递归检查
273 // 其中 VarDecl 的字面量类型注解冲突。原实现未 override，走 DefaultVisitor
274 // no-op，导致 try 块成为类型检查"黑洞"。
275 void visitTryStmt(TryStmt& node) override {
276 auto saved = varTypes;
277 if (node.tryBlock)
278 node.tryBlock->accept(*this);
279 varTypes = saved;
280 // catch 块引入新作用域，catchVarName 是局部变量（无类型注解，不需登记）
281 if (node.catchBlock)
282 node.catchBlock->accept(*this);
283 varTypes = saved;
284 if (node.finallyBlock)
285 node.finallyBlock->accept(*this);
286 varTypes = std::move(saved);
287 }
288
289 // BUG-FE-AUDIT-4 fix: export 包装的声明需递归检查，否则 export var x: int = "str"
290 // 这类明显的类型注解冲突不会在编译期被捕获。
291 void visitExportStmt(ExportStmt& node) override {
292 if (node.declaration)
293 node.declaration->accept(*this);
294 }

296 void visitThrowStmt(ThrowStmt& node) override {
297 if (node.expression)
298 node.expression->accept(*this);
299 }

301private:
302 std::string currentCheckingReturnType_; // 当前正在检查的函数返回类型
303
304 // 字面量类型兼容性检查（编译期，仅字面量）
305 /// 判断字面量实际类型 actual 是否兼容类型注解 annotation。
306 /// 规则：无注解→兼容；int 注解只接受 int；float 注解接受 float 与 int（int 可提升）；
307 /// bool/string 严格匹配；array/dict/类名等无法在字面量层面判定→保守返回兼容
308 /// （避免误报，交由运行时检查）。返回 false 时调用方发警告。
309 static bool typeMatchLiteral(const std::string& actual, const std::string& annotation) {
310 if (annotation.empty())
311 return true;
312 if (annotation == TypeName::INT)
313 return actual == TypeName::INT;
314 if (annotation == TypeName::FLOAT)
315 return actual == TypeName::FLOAT || actual == TypeName::INT;
316 if (annotation == TypeName::BOOL)
317 return actual == TypeName::BOOL;
318 if (annotation == TypeName::STRING)
319 return actual == TypeName::STRING;
320 // R98 元组与解构：tuple 注解只接受 tuple 字面量
321 if (annotation == TypeName::TUPLE)
322 return actual == TypeName::TUPLE;
323 return true; // array/dict/类名等无法在字面量层面检查
324 }
325};

327/// 入口：对整棵 AST（顶层 Block）运行字面量类型检查。
328/// 构造 LiteralTypeWalker 并派发遍历；walker 不修改 AST（故对 program 做 const_cast
329/// 以满足 DefaultVisitor 的 non-const 接口），仅收集 diagnostic 警告。
330/// 返回 DiagnosticBag，调用方据此向用户展示类型注解不匹配的警告。
331DiagnosticBag MiniLangTypeChecker::check(const Block& program) {
332 lastWalkerTypeInfos_.clear();
333 LiteralTypeWalker walker;
334 // const_cast: DefaultVisitor 需要 non-const 引用，但 walker 不修改 AST
335 const_cast<Block&>(program).accept(walker);
336 lastWalkerTypeInfos_ = std::move(walker.varTypeInfos);
337 return std::move(walker.diagnostics);
338}

340/// 按变量名推断类型（基于上次 check() 运行的符号表）。
341TypeInfo MiniLangTypeChecker::inferType(const std::string& name) const {
342 auto it = lastWalkerTypeInfos_.find(name);
343 if (it != lastWalkerTypeInfos_.end())
344 return it->second;
345 return TypeInfo{};
346}
