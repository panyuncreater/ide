8// ============================================================
9// InterpreterCoroutine.cpp — 协程/生成器实现（重放 + 真挂起双模式）
10// ------------------------------------------------------------
11// 拆分自 Interpreter.cpp，包含协程相关的所有方法：
12// - makeCoroutineValue: 构造 Coroutine 值
13// - callCoroutineNext: .next() 核心方法（挂起/重放双模式）
14// - coroutineSuspend / coroutineFiberBody: 真挂起路径（MINILANG_CORO_FIBER=1）
15// - handleCoroutineMethod: 协程方法分发（.next/.done）
16//
17// 模式选择（MINILANG_CORO_FIBER，见 interpreter/CoroutineFiber.h）：
18// - =1（Windows fiber）：真挂起。yield 保存执行位置切回调用方，.next() 从
19// 挂起点恢复——前缀副作用只执行一次，n 次 .next() 总代价 O(n)。递归
20// 树遍历解释器的 C++ 栈即延续，fiber 切换零语义损失（含 yield 位于
21// 嵌套函数调用/表达式中间的场景）。
22// - =0（其他平台）：重放模式。每次 .next() 从头重新执行函数体，用运行时
23// yield 执行计数器跳过已消费的 yield（见 callCoroutineNext 重放分支）。
24//
25// 历史注记：本文件曾注释"StackVM/RegisterVM 用真挂起模式（D.5 实现）"——
26// 该描述与代码不符（四后端原均为重放模式，VM 的 OP_YIELD/REG_YIELD 同样
27// 走计数器跳过），第四轮优化起三后端在本宏启用的平台上同步落地真挂起，
28// 未启用平台上保留四后端一致的重放语义（ADR-002：不允许单后端分叉）。
29//
30// yield 表达式值语义（三端统一）：被跳过（重放）或恢复后（挂起）的 yield
31// 表达式求值为其自身 yield 值——重放由"push(yieldValue)/lastValue_=yieldValue"
32// 实现，挂起由恢复路径的同一赋值实现，两种模式的可观察行为逐点一致，
33// 唯一差异是挂起模式副作用只执行一次。
34// ============================================================

117void Interpreter::coroutineFiberBody() {
118 auto* fs = static_cast<CoroutineFiberState*>(activeCoroutineFiber_);
119 auto* cd = fs->cd;
120 try {
121 // S2 对齐：fiber 生命周期内持有递归深度一层（恶意嵌套生成器耗尽栈防护）
122 if (recursionDepth_ + 1 >= MAX_RECURSION_DEPTH) {
123 runtimeError(ErrorFormat::formatStd(ErrorMessages::kRecursionDepthExceededFmtStd, MAX_RECURSION_DEPTH), 0,
124 0, DiagCodes::kRecursionDepth);
125 }
126 RecursionGuard recursionGuard{recursionDepth_};
127 // 生成器环境：父级为定义时闭包环境（若有），否则为 .next() 调用点环境（对齐重放建帧）
128 auto funEnv = std::make_shared<Environment>(cd->closureEnv ? cd->closureEnv : currentEnv_);
129 fs->funEnv = funEnv;
130 for (size_t i = 0; i < cd->generatorDecl->params.size() && i < cd->args.size(); ++i) {
131 funEnv->define(cd->generatorDecl->params[i], cd->args[i]);
132 }
133 callStack_.emplace_back(cd->generatorDecl->name, funEnv, cd->generatorDecl->line, recursionDepth_);
134 currentEnv_ = funEnv;
135 currentFunctionReturnType_ = cd->generatorDecl->returnType;
136 // B1 TCO: 生成器执行非蹦床，禁用尾调用上下文（信号不可穿透 .next() 边界）
137 TcoScopeGuard tcoGuard{*this, nullptr, std::string(), /*isMethod=*/false, /*enabled=*/false};
138 executeFunctionBody(static_cast<Block&>(*cd->generatorDecl->body));
139 // 函数体自然结束（无 return、无未耗尽 yield）：协程耗尽，最终值为 null（对齐重放）
140 cd->done = true;
141 cd->currentValueBox.clear();
142 cd->currentValueBox.push_back(Value::nullValue());
143 funEnv->closeCapturedVariables();
144 } catch (const ReturnException& e) {
145 // 生成器函数显式 return：协程耗尽，返回 return 值
146 cd->done = true;
147 cd->currentValueBox.clear();
148 cd->currentValueBox.push_back(std::move(e.returnValue));
149 if (fs->funEnv)
150 fs->funEnv->closeCapturedVariables();
151 } catch (...) {
152 // 错误逃逸（RuntimeError/ThrowException/DebugStopException）：封送后由
153 // callCoroutineNext 重抛——调用方上下文由其守卫恢复，done 不置位（对齐重放）
154 fs->failed = true;
155 fs->pendingError = std::current_exception();
156 if (fs->funEnv)
157 fs->funEnv->closeCapturedVariables();
158 }
159 fs->completed = true;
160 minilang_fiber::switchToFiber(fs->resumer);
161 // 不可达：fiber 终结后仅会被 deleteFiber 回收（CoroutineData 析构或 .next() 内 reset）
162}

164void Interpreter::coroutineSuspend(Value yieldValue) {
165 auto* fs = static_cast<CoroutineFiberState*>(activeCoroutineFiber_);
166 auto* cd = fs->cd;
167 fs->handoffValue = yieldValue;
168 // currentValueBox / currentYieldId / done 判定逐点对齐重放 catch 分支：
169 // 最终 yield（currentYieldId 达到 yieldCount）后协程耗尽，尾部语句不执行
170 cd->currentValueBox.clear();
171 cd->currentValueBox.push_back(yieldValue);
172 cd->currentYieldId++;
173 if (cd->currentYieldId >= cd->yieldCount)
174 cd->done = true;
175 // 保存上下文并截断调用栈到调用方基线（调用方在两次 .next() 之间继续执行，
176 // 不得看到生成器的陈旧帧；帧随快照保活并持 GC 根）
177 fs->suspendedEnv = currentEnv_;
178 fs->suspendedCallStack.assign(std::make_move_iterator(callStack_.begin() + fs->resumerStackDepth),
179 std::make_move_iterator(callStack_.end()));
180 callStack_.resize(fs->resumerStackDepth);
181 fs->suspendedReturnType = currentFunctionReturnType_;
182 fs->suspendedRecursionDepth = recursionDepth_;
183 minilang_fiber::switchToFiber(fs->resumer);
184 // —— 恢复点：下一次 .next() 切回。此刻解释器成员为调用方状态，先还原 ——
185 currentEnv_ = std::move(fs->suspendedEnv);
186 currentFunctionReturnType_ = std::move(fs->suspendedReturnType);
187 recursionDepth_ = fs->suspendedRecursionDepth;
188 if (!fs->suspendedCallStack.empty()) {
189 callStack_.insert(callStack_.end(), std::make_move_iterator(fs->suspendedCallStack.begin()),
190 std::make_move_iterator(fs->suspendedCallStack.end()));
191 fs->suspendedCallStack.clear();
192 }
193 // yield 表达式值 = 自身 yield 值（对齐重放 skip 语义：lastValue_ = yieldValue）
194 lastValue_ = fs->handoffValue;
195}
196#endif // MINILANG_CORO_FIBER

198Value Interpreter::callCoroutineNext(Value& coroVal) {
199 // CoroutineData 是 Value 的私有嵌套类型，使用 auto* 推导避免显式声明类型名
200 auto* cd = coroVal.coroutineData();
201
202 // 已耗尽：返回 currentValue（最后一次 yield/return 的值），语义与 Python 一致
203 if (cd->done) {
204#if MINILANG_CORO_FIBER
205 // 最终 yield 触发的 done：fiber 停在 yield 点永不再恢复，弃用之
206 // （完成路径的 done 此处 reset 为 no-op）
207 cd->interpreterFiber.reset();
208#endif
209 return cd->currentValueBox.empty() ? Value::nullValue() : cd->currentValueBox.front();
210 }
211
212 // 保存调用方上下文（currentEnv_ / currentCoroutineTargetYieldId_ /
213 // currentYieldExecutionCount_ / currentFunctionReturnType_）
214 auto prevEnv = currentEnv_;
215 int prevTargetYieldId = currentCoroutineTargetYieldId_;
216 int prevYieldExecCount = currentYieldExecutionCount_;
217 std::string prevReturnType = currentFunctionReturnType_;
218
219 // RAII 守卫：异常路径同样恢复上下文
220 // AUDIT-R3 P2-8 fix: 调用栈收缩并入守卫——原实现的弹帧循环位于 try/catch
221 // 之后的普通代码路径，try 只捕获 YieldSignal/ReturnException；生成器体抛
222 // RuntimeError/ThrowException/DebugStopException 时异常直接逃出函数，
223 // callStack_ 残留陈旧帧（脚本 catch 后继续执行时调试栈面板错乱、
224 // 反复失败的 next 调用持续累积帧）。
225 struct CoroutineContextGuard {
226 Interpreter& interp;
227 std::shared_ptr<Environment> prevEnv;
228 int prevTargetYieldId;
229 int prevYieldExecCount;
230 std::string prevReturnType;
231 std::shared_ptr<Environment> funEnv;
232 size_t savedStackDepth; // AUDIT-R3 P2-8 fix: 析构时统一收缩调用栈
233 ~CoroutineContextGuard() {
234 if (funEnv)
235 funEnv->closeCapturedVariables();
236 interp.currentEnv_ = prevEnv;
237 interp.currentCoroutineTargetYieldId_ = prevTargetYieldId;
238 interp.currentYieldExecutionCount_ = prevYieldExecCount;
239 interp.currentFunctionReturnType_ = std::move(prevReturnType);
240 // AUDIT-R3 P2-8 fix: 异常路径也恢复调用栈深度
241 while (interp.callStack_.size() > savedStackDepth) {
242 interp.callStack_.pop_back();
243 }
244 }
245 } guard{*this, prevEnv, prevTargetYieldId, prevYieldExecCount, prevReturnType, nullptr, callStack_.size()};
246
247#if MINILANG_CORO_FIBER
248 // —— 真挂起路径（Windows fiber）：yield 保存执行位置切回，副作用只执行一次。
249 // 成功返回/失败重抛均经守卫析构恢复调用方上下文（prevEnv/target/计数/returnType
250 // + 调用栈收缩；挂起路径的调用栈已由 coroutineSuspend 截断，收缩为 no-op）。
251 {
252 auto* fs = static_cast<CoroutineFiberState*>(cd->interpreterFiber.get());
253 if (!fs) {
254 minilang_fiber::convertThreadToFiber(); // 幂等：已转换线程复用当前 fiber
255 fs = new CoroutineFiberState();
256 fs->interp = this;
257 fs->cd = cd;
258 cd->interpreterFiber = {fs, [](void* p) { delete static_cast<CoroutineFiberState*>(p); }};
259 fs->fiber = minilang_fiber::createFiber(&coroutineFiberTrampoline, fs);
260 }
261 if (activeCoroutineFiber_ == fs) {
262 runtimeError("生成器不能在自身执行体内调用 .next()（真挂起模式不支持自恢复）", 0, 0);
263 }
264 int prevRecursion = recursionDepth_;
265 fs->resumerStackDepth = callStack_.size();
266 CoroutineFiberState* prevActive = activeCoroutineFiber_;
267 activeCoroutineFiber_ = fs; // visitYieldExpr 据此进入挂起分支（含生成器体内嵌套函数的 yield）
268 fs->resumer = minilang_fiber::currentFiber();
269 minilang_fiber::switchToFiber(fs->fiber);
270 // 返回点：fiber 挂起（yield）或终结（自然结束/return/错误逃逸）
271 activeCoroutineFiber_ = prevActive;
272 recursionDepth_ = prevRecursion;
273 if (fs->failed) {
274 auto err = fs->pendingError;
275 cd->interpreterFiber.reset(); // 死 fiber 弃用：下次 .next() 全新重建（对齐重放重新执行）
276 std::rethrow_exception(err);
277 }
278 if (fs->completed)
279 cd->interpreterFiber.reset(); // fiber 生命周期终结（deleter 删平台资源）
280 // yield 值 / return 值 / 自然耗尽 null 均已写入 currentValueBox
281 return cd->currentValueBox.empty() ? Value::nullValue() : cd->currentValueBox.front();
282 }
283#else
284 // —— 重放路径（MINILANG_CORO_FIBER=0 平台）：见文件头"模式选择"——
285 // Windows 下此路径不编译（fiber 分支全路径 return，重放代码不可达）。
286
287 // S2 fix: 递归深度保护（重放也算递归调用，避免恶意嵌套生成器耗尽栈）
288 if (recursionDepth_ + 1 >= MAX_RECURSION_DEPTH) {
289 runtimeError(ErrorFormat::formatStd(ErrorMessages::kRecursionDepthExceededFmtStd, MAX_RECURSION_DEPTH), 0, 0,
290 DiagCodes::kRecursionDepth);
291 }
292 RecursionGuard recursionGuard{recursionDepth_};
293
294 Value result = Value::nullValue();
295 bool reachedYield = false;
296
297 // AUDIT-R3 P2-8 fix: savedStackDepth 已入 CoroutineContextGuard（构造时记录），
298 // 任何退出路径（含 RuntimeError/ThrowException/DebugStopException 逃逸）
299 // 均由守卫析构统一收缩调用栈，不再依赖 try/catch 之后的手动弹帧循环。
300
301 try {
302 // 创建生成器函数环境：父级为定义时闭包环境（若有），否则为当前环境
303 std::shared_ptr<Environment> funEnv =
304 std::make_shared<Environment>(cd->closureEnv ? cd->closureEnv : currentEnv_);
305 guard.funEnv = funEnv; // 守卫持有以便异常路径 closeCapturedVariables
306
307 // 绑定调用参数（重放时每次重新绑定，参数值在创建协程时已固定）
308 for (size_t i = 0; i < cd->generatorDecl->params.size() && i < cd->args.size(); ++i) {
309 funEnv->define(cd->generatorDecl->params[i], cd->args[i]);
310 }
311
312 // 压入调用栈（用于调试器显示调用层次）
313 callStack_.emplace_back(cd->generatorDecl->name, funEnv, cd->generatorDecl->line, recursionDepth_);
314
315 currentEnv_ = funEnv;
316 currentCoroutineTargetYieldId_ = cd->currentYieldId;
317 // R164 fix: 每次重放开始时重置运行时 yield 执行计数器
318 currentYieldExecutionCount_ = 0;
319 currentFunctionReturnType_ = cd->generatorDecl->returnType;
320
321 // 重放执行函数体（从头开始）
322 // B1 TCO: 生成器重放非蹦床，禁用尾调用上下文（信号不可穿透 .next() 边界）
323 TcoScopeGuard tcoGuard{*this, nullptr, std::string(), /*isMethod=*/false, /*enabled=*/false};
324 executeFunctionBody(static_cast<Block&>(*cd->generatorDecl->body));
325 // 函数体自然结束（无 return，无未耗尽的 yield）：协程耗尽
326 cd->done = true;
327 cd->currentValueBox.clear();
328 cd->currentValueBox.push_back(result);
329 } catch (const YieldSignal& e) {
330 // 命中目标 yield：返回 yield 值，递增 currentYieldId
331 result = e.yieldValue;
332 reachedYield = true;
333 cd->currentValueBox.clear();
334 cd->currentValueBox.push_back(result);
335 cd->currentYieldId++;
336 if (cd->currentYieldId >= cd->yieldCount) {
337 cd->done = true;
338 }
339 } catch (const ReturnException& e) {
340 // 生成器函数显式 return：协程耗尽，返回 return 值
341 result = std::move(e.returnValue);
342 cd->currentValueBox.clear();
343 cd->currentValueBox.push_back(result);
344 cd->done = true;
345 }
346
347 // AUDIT-R3 P2-8 fix: 弹帧由 CoroutineContextGuard 析构统一处理（含异常路径）
348
349 (void)reachedYield; // 防止未使用警告
350 return result;
351#endif // MINILANG_CORO_FIBER
352}
