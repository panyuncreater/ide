9// ============================================================
10// InterpreterCoroutine.cpp — 协程/生成器实现（重放 + 真挂起双模式）
11// ------------------------------------------------------------
12// 拆分自 Interpreter.cpp，包含协程相关的所有方法：
13// - makeCoroutineValue: 构造 Coroutine 值
14// - callCoroutineNext: .next() 核心方法（挂起/重放双模式）
15// - coroutineSuspend / coroutineFiberBody: 真挂起路径（MINILANG_CORO_FIBER=1）
16// - handleCoroutineMethod: 协程方法分发（.next/.done）
17//
18// 模式选择（MINILANG_CORO_FIBER，见 interpreter/CoroutineFiber.h）：
19// - =1（Windows fiber）：真挂起。yield 保存执行位置切回调用方，.next() 从
20// 挂起点恢复——前缀副作用只执行一次，n 次 .next() 总代价 O(n)。递归
21// 树遍历解释器的 C++ 栈即延续，fiber 切换零语义损失（含 yield 位于
22// 嵌套函数调用/表达式中间的场景）。
23// - =0（其他平台）：重放模式。每次 .next() 从头重新执行函数体，用运行时
24// yield 执行计数器跳过已消费的 yield（见 callCoroutineNext 重放分支）。
25//
26// 历史注记：本文件曾注释"StackVM/RegisterVM 用真挂起模式（D.5 实现）"——
27// 该描述与代码不符（四后端原均为重放模式，VM 的 OP_YIELD/REG_YIELD 同样
28// 走计数器跳过），第四轮优化起三后端在本宏启用的平台上同步落地真挂起，
29// 未启用平台上保留四后端一致的重放语义（ADR-002：不允许单后端分叉）。
30//
31// yield 表达式值语义（三端统一）：被跳过（重放）或恢复后（挂起）的 yield
32// 表达式求值为其自身 yield 值——重放由"push(yieldValue)/lastValue_=yieldValue"
33// 实现，挂起由恢复路径的同一赋值实现，两种模式的可观察行为逐点一致，
34// 唯一差异是挂起模式副作用只执行一次。
35// ============================================================

128// 首次进入协程 fiber：完成 ASan 到栈标注（fake stack 从无到有，无需保存），
129// 登记当前 fiber，转入用户过程。proc 按协议不返回（终结前 switchToFiber
130// 回 resumer，见 coroutineFiberBody 尾部注释）。
131void fiberEntry(int lo, int hi) {
132#if MINILANG_FIBER_ASAN
133 __sanitizer_finish_switch_fiber(nullptr, nullptr, nullptr);
134#endif
135 uintptr_t addr = (static_cast<uintptr_t>(static_cast<uint32_t>(hi)) << 32) | static_cast<uint32_t>(lo);
136 auto* self = reinterpret_cast<FiberContext*>(addr);
137 t_currentFiber = self;
138 self->proc(self->arg);
139 // 不可达（对齐 Windows fiber proc 返回语义的防御收紧）：协议破坏时
140 // 显式终止而非静默线程退出/未定义回落（uc_link 为 nullptr）
141 std::abort();

153// 栈内存新分配（惰性页分配，未触碰页不占 RSS）：对齐 Windows 4MB 余量
154inline FiberHandle createFiber(void (*proc)(void*), void* arg) {
155 auto* fc = new FiberContext();
156 fc->proc = proc;
157 fc->arg = arg;
158 fc->stack = new unsigned char[kFiberStackSize];
159 fc->stackSize = kFiberStackSize;
160 getcontext(&fc->ctx);
161 fc->ctx.uc_stack.ss_sp = fc->stack;
162 fc->ctx.uc_stack.ss_size = fc->stackSize;
163 fc->ctx.uc_link = nullptr; // 终结经 switchToFiber(resumer) 离开，不回落
164 // makecontext 仅能传 int 宽参数：控制块指针拆高低 32 位（x86-64/Linux）
165 uintptr_t addr = reinterpret_cast<uintptr_t>(fc);
166 makecontext(&fc->ctx, reinterpret_cast<void (*)()>(&fiberEntry), 2, static_cast<int>(addr & 0xFFFFFFFFu),
167 static_cast<int>(addr >> 32));
168 return fc;
169}
170inline void switchToFiber(FiberHandle target) {
171 FiberContext* self = t_currentFiber;
172#if MINILANG_FIBER_ASAN
173 void* fakeStackSave = nullptr;
174 __sanitizer_start_switch_fiber(&fakeStackSave, target->stack, target->stackSize);
175#endif
176 swapcontext(&self->ctx, &target->ctx);
177#if MINILANG_FIBER_ASAN
178 __sanitizer_finish_switch_fiber(fakeStackSave, nullptr, nullptr);
179#endif
180 t_currentFiber = self;
181}

266void Interpreter::coroutineFiberBody() {
267 auto* fs = static_cast<CoroutineFiberState*>(activeCoroutineFiber_);
268 auto* cd = fs->cd;
269 try {
270 // S2 对齐：fiber 生命周期内持有递归深度一层（恶意嵌套生成器耗尽栈防护）
271 if (recursionDepth_ + 1 >= MAX_RECURSION_DEPTH) {
272 runtimeError(ErrorFormat::formatStd(ErrorMessages::kRecursionDepthExceededFmtStd, MAX_RECURSION_DEPTH), 0,
273 0, DiagCodes::kRecursionDepth);
274 }
275 RecursionGuard recursionGuard{recursionDepth_};
276 // 生成器环境：父级为定义时闭包环境（若有），否则为 .next() 调用点环境（对齐重放建帧）
277 auto funEnv = std::make_shared<Environment>(cd->closureEnv ? cd->closureEnv : currentEnv_);
278 fs->funEnv = funEnv;
279 for (size_t i = 0; i < cd->generatorDecl->params.size() && i < cd->args.size(); ++i) {
280 funEnv->define(cd->generatorDecl->params[i], cd->args[i]);
281 }
282 callStack_.emplace_back(cd->generatorDecl->name, funEnv, cd->generatorDecl->line, recursionDepth_);
283 currentEnv_ = funEnv;
284 currentFunctionReturnType_ = cd->generatorDecl->returnType;
285 // B1 TCO: 生成器执行非蹦床，禁用尾调用上下文（信号不可穿透 .next() 边界）
286 TcoScopeGuard tcoGuard{*this, nullptr, std::string(), /*isMethod=*/false, /*enabled=*/false};
287 executeFunctionBody(static_cast<Block&>(*cd->generatorDecl->body));
288 // 函数体自然结束（无 return、无未耗尽 yield）：协程耗尽，最终值为 null（对齐重放）
289 cd->done = true;
290 cd->currentValueBox.clear();
291 cd->currentValueBox.push_back(Value::nullValue());
292 funEnv->closeCapturedVariables();
293 } catch (const ReturnException& e) {
294 // 生成器函数显式 return：协程耗尽，返回 return 值
295 cd->done = true;
296 cd->currentValueBox.clear();
297 cd->currentValueBox.push_back(std::move(e.returnValue));
298 if (fs->funEnv)
299 fs->funEnv->closeCapturedVariables();
300 } catch (const ThrowException& e) {
301 // 脚本 throw：额外保存值的独立副本，确保多次 .next() 重抛同一错误值。
302 fs->failed = true;
303 fs->pendingError = std::current_exception();
304 fs->pendingThrownValue = e.thrownValue;
305 fs->hasPendingThrownValue = true;
306 if (fs->funEnv)
307 fs->funEnv->closeCapturedVariables();
308 } catch (...) {
309 // 错误逃逸（RuntimeError/DebugStopException 等）：封送后由
310 // callCoroutineNext 重抛——调用方上下文由其守卫恢复，done 不置位（对齐重放）
311 fs->failed = true;
312 fs->pendingError = std::current_exception();
313 fs->hasPendingThrownValue = false;
314 if (fs->funEnv)
315 fs->funEnv->closeCapturedVariables();
316 }
317 fs->completed = true;
318 minilang_fiber::switchToFiber(fs->resumer);
319 // 不可达：fiber 终结后仅会被 deleteFiber 回收（CoroutineData 析构或 .next() 内 reset）
320}

322void Interpreter::coroutineSuspend(Value yieldValue) {
323 auto* fs = static_cast<CoroutineFiberState*>(activeCoroutineFiber_);
324 auto* cd = fs->cd;
325 fs->handoffValue = yieldValue;
326 // currentValueBox / currentYieldId / done 判定逐点对齐重放 catch 分支：
327 // 最终 yield（currentYieldId 达到 yieldCount）后协程耗尽，尾部语句不执行
328 cd->currentValueBox.clear();
329 cd->currentValueBox.push_back(yieldValue);
330 cd->currentYieldId++;
331 if (cd->currentYieldId >= cd->yieldCount)
332 cd->done = true;
333 // 保存上下文并截断调用栈到调用方基线（调用方在两次 .next() 之间继续执行，
334 // 不得看到生成器的陈旧帧；帧随快照保活并持 GC 根）
335 fs->suspendedEnv = currentEnv_;
336 fs->suspendedCallStack.assign(std::make_move_iterator(callStack_.begin() + fs->resumerStackDepth),
337 std::make_move_iterator(callStack_.end()));
338 callStack_.resize(fs->resumerStackDepth);
339 fs->suspendedReturnType = currentFunctionReturnType_;
340 fs->suspendedRecursionDepth = recursionDepth_;
341 // P1 fix（2026-10-22）：TCO 五成员随快照保存，恢复点对称还原——生成器体恢复执行
342 // 时回到挂起前的 TCO 语境（生成器体为 disabled 态），不再读到调用方真值。
343 fs->suspendedTcoDecl = tcoDecl_;
344 fs->suspendedTcoName = tcoName_;
345 fs->suspendedTcoIsMethod = tcoIsMethod_;
346 fs->suspendedTcoEnabled = tcoEnabled_;
347 fs->suspendedTcoTryDepth = tcoTryDepth_;
348 minilang_fiber::switchToFiber(fs->resumer);
349 // —— 恢复点：下一次 .next() 切回。此刻解释器成员为调用方状态，先还原 ——
350 currentEnv_ = std::move(fs->suspendedEnv);
351 currentFunctionReturnType_ = std::move(fs->suspendedReturnType);
352 recursionDepth_ = fs->suspendedRecursionDepth;
353 tcoDecl_ = fs->suspendedTcoDecl;
354 tcoName_ = std::move(fs->suspendedTcoName);
355 tcoIsMethod_ = fs->suspendedTcoIsMethod;
356 tcoEnabled_ = fs->suspendedTcoEnabled;
357 tcoTryDepth_ = fs->suspendedTcoTryDepth;
358 if (!fs->suspendedCallStack.empty()) {
359 callStack_.insert(callStack_.end(), std::make_move_iterator(fs->suspendedCallStack.begin()),
360 std::make_move_iterator(fs->suspendedCallStack.end()));
361 fs->suspendedCallStack.clear();
362 }
363 // yield 表达式值 = 自身 yield 值（对齐重放 skip 语义：lastValue_ = yieldValue）
364 lastValue_ = fs->handoffValue;
365}

367void Interpreter::collectCoroutineFiberGcRoots(const CoroutineFiberState& fs, std::vector<const void*>& roots) {
368 // P0 fix（2026-10-21）：挂起协程快照根集——suspendedEnv/funEnv/suspendedCallStack
369 // 各帧环境链中的变量（tracked 容器）全部贡献为 GC 根。链间去重交给 mark 阶段的
370 // marked 集（重复根无害，见 triggerIncrementalGc 同类注释）。
371 auto collectChain = [&roots](const Environment* env) {
372 int depth = 0;
373 while (env && depth++ < 1024) {
374 collectEnvLocalGcRoots(env, roots);
375 env = env->parent.get();
376 }
377 };
378 if (fs.suspendedEnv)
379 collectChain(fs.suspendedEnv.get());
380 if (fs.funEnv)
381 collectChain(fs.funEnv.get());
382 for (const auto& frame : fs.suspendedCallStack) {
383 if (frame.env)
384 collectChain(frame.env.get());
385 }
386 if (fs.hasPendingThrownValue) {
387 if (const void* ptr = fs.pendingThrownValue.gcRootPtr())

393Value Interpreter::callCoroutineNext(Value& coroVal) {
394 // CoroutineData 是 Value 的私有嵌套类型，使用 auto* 推导避免显式声明类型名
395 auto* cd = coroVal.coroutineData();
396
397 // 已耗尽：返回 currentValue（最后一次 yield/return 的值），语义与 Python 一致
398 if (cd->done) {
399#if MINILANG_CORO_FIBER
400 // 最终 yield 触发的 done：fiber 停在 yield 点永不再恢复，弃用之
401 // （完成路径的 done 此处 reset 为 no-op）
402 cd->gcRootContributor = nullptr; // P0 fix：丢弃快照同步置空根集贡献回调
403 cd->interpreterFiber.reset();
404#endif
405 return cd->currentValueBox.empty() ? Value::nullValue() : cd->currentValueBox.front();
406 }
407
408 // 保存调用方上下文（currentEnv_ / currentCoroutineTargetYieldId_ /
409 // currentYieldExecutionCount_ / currentFunctionReturnType_）
410 auto prevEnv = currentEnv_;
411 int prevTargetYieldId = currentCoroutineTargetYieldId_;
412 int prevYieldExecCount = currentYieldExecutionCount_;
413 std::string prevReturnType = currentFunctionReturnType_;
414
415 // RAII 守卫：异常路径同样恢复上下文
416 // AUDIT-R3 P2-8 fix: 调用栈收缩并入守卫——原实现的弹帧循环位于 try/catch
417 // 之后的普通代码路径，try 只捕获 YieldSignal/ReturnException；生成器体抛
418 // RuntimeError/ThrowException/DebugStopException 时异常直接逃出函数，
419 // callStack_ 残留陈旧帧（脚本 catch 后继续执行时调试栈面板错乱、
420 // 反复失败的 next 调用持续累积帧）。
421 struct CoroutineContextGuard {
422 Interpreter& interp;
423 std::shared_ptr<Environment> prevEnv;
424 int prevTargetYieldId;
425 int prevYieldExecCount;
426 std::string prevReturnType;
427 std::shared_ptr<Environment> funEnv;
428 size_t savedStackDepth; // AUDIT-R3 P2-8 fix: 析构时统一收缩调用栈
429 // P1 fix（2026-10-22）：TCO 五成员纳入 .next() 退出恢复——fiber 体守卫在终结
430 // 路径的析构写回的是首次进入时的陈旧蹦床快照，本恢复在其后以调用方真值覆盖，
431 // 消除蹦床误派发（终结上下文的尾调用被误分类为 SelfFunction 无 target →
432 // 函数体携错误实参重跑）；挂起窗口的 disabled 滞留由 coroutineSuspend 的
433 // 对称保存/恢复处理。
434 FunDecl* prevTcoDecl;
435 std::string prevTcoName;
436 bool prevTcoIsMethod;
437 bool prevTcoEnabled;
438 int prevTcoTryDepth;
439 ~CoroutineContextGuard() {
440 if (funEnv)
441 funEnv->closeCapturedVariables();
442 interp.currentEnv_ = prevEnv;
443 interp.currentCoroutineTargetYieldId_ = prevTargetYieldId;
444 interp.currentYieldExecutionCount_ = prevYieldExecCount;
445 interp.currentFunctionReturnType_ = std::move(prevReturnType);
446 interp.tcoDecl_ = prevTcoDecl;
447 interp.tcoName_ = std::move(prevTcoName);
448 interp.tcoIsMethod_ = prevTcoIsMethod;
449 interp.tcoEnabled_ = prevTcoEnabled;
450 interp.tcoTryDepth_ = prevTcoTryDepth;
451 // AUDIT-R3 P2-8 fix: 异常路径也恢复调用栈深度
452 while (interp.callStack_.size() > savedStackDepth) {
453 interp.callStack_.pop_back();
454 }
455 }
456 } guard{*this, prevEnv, prevTargetYieldId, prevYieldExecCount, prevReturnType, nullptr, callStack_.size(),
457 tcoDecl_, tcoName_, tcoIsMethod_, tcoEnabled_, tcoTryDepth_};
458
459#if MINILANG_CORO_FIBER
460 // —— 真挂起路径（Windows fiber）：yield 保存执行位置切回，副作用只执行一次。
461 // 成功返回/失败重抛均经守卫析构恢复调用方上下文（prevEnv/target/计数/returnType
462 // + 调用栈收缩；挂起路径的调用栈已由 coroutineSuspend 截断，收缩为 no-op）。
463 {
464 auto* fs = static_cast<CoroutineFiberState*>(cd->interpreterFiber.get());
465 if (!fs) {
466 minilang_fiber::convertThreadToFiber(); // 幂等：已转换线程复用当前 fiber
467 fs = new CoroutineFiberState();
468 fs->interp = this;
469 fs->cd = cd;
470 fs->ownerThread = std::this_thread::get_id(); // §3.8：登记创建线程
471 cd->interpreterFiber = {fs, [](void* p) { delete static_cast<CoroutineFiberState*>(p); }};
472 fs->fiber = minilang_fiber::createFiber(&coroutineFiberTrampoline, fs);
473 // P0 fix（2026-10-21）：注册挂起快照 GC 根集贡献回调。捕获裸 fs：fs 由
474 // cd->interpreterFiber 持有，mark 到达本 cd 时必存活；两处丢弃 fiber 的
475 // reset 均同步置空回调（见下）。
476 cd->gcRootContributor = [fs](std::vector<const void*>& roots) {
477 fs->interp->collectCoroutineFiberGcRoots(*fs, roots);
478 };
479 }
480 // §3.8 跨线程恢复防御（2026-10-22）：fiber 只能由创建线程恢复——非创建线程
481 // 切入时 Windows SwitchToFiber 为跨线程未定义行为、Linux ucontext 层
482 // t_currentFiber 未登记（空指针解引用）；且挂起快照的解释器状态只与创建
483 // 线程的调用栈/上下文绑定。显式走 runtimeError 通道报错，而非进入未定义
484 // 路径（重放模式 =0 无 fiber 面，不经过本分支，天然不受影响）。
485 if (fs->ownerThread != std::this_thread::get_id()) {
486 runtimeError("协程不可跨线程恢复", 0, 0);
487 }
488 if (fs->failed) {
489 // P1 fix（2026-10-21）：失败 fiber 不再弃用重建。原实现 reset 后下次
490 // .next() 从头重跑函数体，coroutineSuspend 无条件递增 currentYieldId，
491 // 与重放模式"跳过已消费 yield 后同点再抛"逐点分歧（首查即提前置 done、
492 // 返回错误前首个 yield 值、前缀副作用重复）。保留失败 fiber：后续
493 // .next() 持续重抛同一错误——值/错误序列与重放逐点一致，前缀副作用
494 // 只执行一次（ADR-007 挂起 vs 重放的唯一既定差异口径）。
495 // 实现要点：重抛必须在切入前（此处）——fiber 体已终结，switch 进去即重入
496 // 死栈 = UB；首次失败在 switch 返回后重抛（见下方返回点检查）。
497 if (fs->hasPendingThrownValue)
498 throw ThrowException(fs->pendingThrownValue);
499 std::rethrow_exception(fs->pendingError);
500 }
501 if (activeCoroutineFiber_ == fs) {
502 runtimeError("生成器不能在自身执行体内调用 .next()（真挂起模式不支持自恢复）", 0, 0);
503 }
504 int prevRecursion = recursionDepth_;
505 fs->resumerStackDepth = callStack_.size();
506 CoroutineFiberState* prevActive = activeCoroutineFiber_;
507 activeCoroutineFiber_ = fs; // visitYieldExpr 据此进入挂起分支（含生成器体内嵌套函数的 yield）
508 fs->resumer = minilang_fiber::currentFiber();
509 minilang_fiber::switchToFiber(fs->fiber);
510 // 返回点：fiber 挂起（yield）或终结（自然结束/return/错误逃逸）
511 activeCoroutineFiber_ = prevActive;
512 recursionDepth_ = prevRecursion;
513 if (fs->failed) {
514 // 首次失败到达此处（fiber 体内设 failed 后切回）：重抛并保留 fiber
515 // （后续 .next() 走上方前置检查直接重抛，不再重建）。
516 if (fs->hasPendingThrownValue)
517 throw ThrowException(fs->pendingThrownValue);
518 std::rethrow_exception(fs->pendingError);
519 }
520 if (fs->completed) {
521 cd->gcRootContributor = nullptr; // P0 fix：丢弃快照同步置空根集贡献回调
522 cd->interpreterFiber.reset(); // fiber 生命周期终结（deleter 删平台资源）
523 }
524 // yield 值 / return 值 / 自然耗尽 null 均已写入 currentValueBox
525 return cd->currentValueBox.empty() ? Value::nullValue() : cd->currentValueBox.front();
526 }
527#else
528 // —— 重放路径（MINILANG_CORO_FIBER=0 平台）：见文件头"模式选择"——
529 // Windows 下此路径不编译（fiber 分支全路径 return，重放代码不可达）。
530
531 // S2 fix: 递归深度保护（重放也算递归调用，避免恶意嵌套生成器耗尽栈）
532 if (recursionDepth_ + 1 >= MAX_RECURSION_DEPTH) {
533 runtimeError(ErrorFormat::formatStd(ErrorMessages::kRecursionDepthExceededFmtStd, MAX_RECURSION_DEPTH), 0, 0,
534 DiagCodes::kRecursionDepth);
535 }
536 RecursionGuard recursionGuard{recursionDepth_};
537
538 Value result = Value::nullValue();
539 bool reachedYield = false;
540
541 // AUDIT-R3 P2-8 fix: savedStackDepth 已入 CoroutineContextGuard（构造时记录），
542 // 任何退出路径（含 RuntimeError/ThrowException/DebugStopException 逃逸）
543 // 均由守卫析构统一收缩调用栈，不再依赖 try/catch 之后的手动弹帧循环。
544
545 try {
546 // 创建生成器函数环境：父级为定义时闭包环境（若有），否则为当前环境
547 std::shared_ptr<Environment> funEnv =
548 std::make_shared<Environment>(cd->closureEnv ? cd->closureEnv : currentEnv_);
549 guard.funEnv = funEnv; // 守卫持有以便异常路径 closeCapturedVariables
550
551 // 绑定调用参数（重放时每次重新绑定，参数值在创建协程时已固定）
552 for (size_t i = 0; i < cd->generatorDecl->params.size() && i < cd->args.size(); ++i) {
553 funEnv->define(cd->generatorDecl->params[i], cd->args[i]);
554 }
555
556 // 压入调用栈（用于调试器显示调用层次）
557 callStack_.emplace_back(cd->generatorDecl->name, funEnv, cd->generatorDecl->line, recursionDepth_);
558
559 currentEnv_ = funEnv;
560 currentCoroutineTargetYieldId_ = cd->currentYieldId;
561 // R164 fix: 每次重放开始时重置运行时 yield 执行计数器
562 currentYieldExecutionCount_ = 0;
563 currentFunctionReturnType_ = cd->generatorDecl->returnType;
564
565 // 重放执行函数体（从头开始）
566 // B1 TCO: 生成器重放非蹦床，禁用尾调用上下文（信号不可穿透 .next() 边界）
567 TcoScopeGuard tcoGuard{*this, nullptr, std::string(), /*isMethod=*/false, /*enabled=*/false};
568 executeFunctionBody(static_cast<Block&>(*cd->generatorDecl->body));
569 // 函数体自然结束（无 return，无未耗尽的 yield）：协程耗尽
570 cd->done = true;
571 cd->currentValueBox.clear();
572 cd->currentValueBox.push_back(result);
573 } catch (const YieldSignal& e) {
574 // 命中目标 yield：返回 yield 值，递增 currentYieldId
575 result = e.yieldValue;
576 reachedYield = true;
577 cd->currentValueBox.clear();
578 cd->currentValueBox.push_back(result);
579 cd->currentYieldId++;
580 if (cd->currentYieldId >= cd->yieldCount) {
581 cd->done = true;
582 }
583 } catch (const ReturnException& e) {
584 // 生成器函数显式 return：协程耗尽，返回 return 值
585 result = std::move(e.returnValue);
586 cd->currentValueBox.clear();
587 cd->currentValueBox.push_back(result);
588 cd->done = true;
589 }
590
591 // AUDIT-R3 P2-8 fix: 弹帧由 CoroutineContextGuard 析构统一处理（含异常路径）
592
593 (void)reachedYield; // 防止未使用警告
594 return result;
595#endif // MINILANG_CORO_FIBER
596}
