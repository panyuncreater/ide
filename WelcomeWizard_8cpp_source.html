404/// 切换到指定向导步骤并刷新页面。
405void WelcomeWizard::goToStep(int index) {
406 if (index < 0 || index >= pages_->count())
407 return;

409 updateStepIndicator();
410 updateNavButtons();
411
412 // 进入 Step 2 时默认高亮第一个 token

415 highlightCodeRange(tokenSpans_[0].start, tokenSpans_[0].end);
416 }
417}

504/// 在示例编辑器中高亮 [start,end) 区间代码。
505void WelcomeWizard::highlightCodeRange(int start, int end) {
506 if (!codeEdit_)
507 return;
508 QTextCursor cursor(codeEdit_->document());
509 QList<QTextEdit::ExtraSelection> selections;
510
511 QTextEdit::ExtraSelection sel;
512 sel.cursor = cursor;
513 sel.cursor.setPosition(start);
514 sel.cursor.setPosition(end, QTextCursor::KeepAnchor);
515 sel.format.setBackground(TeachingTheme::primary());
516 sel.format.setForeground(QColor("#ffffff"));
517 selections.append(sel);
518
519 codeEdit_->setExtraSelections(selections);
520}
