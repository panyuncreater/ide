43// UnusedVariable / UnusedFunction / UnusedParameter /

51 static const std::vector<LintRuleInfo> kRules = {

54 "移除未使用的变量声明，或在变量名前加 _ 前缀表示有意未使用"},
55 {"lint-unused-function", "UnusedFunction", "未使用函数", "🔇",
56 "函数定义后未调用（main/init/test 约定入口跳过）", "未使用的函数 '{name}'", "fun foo() { return 1; }\n",
57 "移除未使用的函数，或调用它"},
58 {"lint-unused-parameter", "UnusedParameter", "未使用参数", "🔇", "函数参数未使用", "未使用的参数 '{name}'",
59 "fun foo(a) { return 1; }\nprint(foo(1));\n", "移除未使用的参数，或在参数名前加 _ 前缀"},
60 {"lint-assignment-in-condition", "AssignmentInCondition", "条件中赋值", "⚠️",
61 "if/while 条件中含赋值（可能误将 == 写成 =）", "条件中含赋值 '{name}='，可能误写 ==",

63 {"lint-dead-code", "DeadCodeAfterReturn", "return 后死代码", "💀", "return/break/continue 之后的语句",
64 "return 后的不可达代码", "fun foo() {\n return 1;\n print(2);\n}\nprint(foo());\n",

79const LintRuleInfo* LintRuleLibrary::findByCode(const std::string& code) {
80 for (const auto& r : rules()) {
81 if (r.code == code) {
82 return &r;
83 }
84 }
85 return nullptr;
86}

334/// 运行 Lint：构建 LintOptions（规则开关 + 圈复杂度阈值），调用
335/// minilang_lint::lintSource 三步管线，将诊断填入表格并更新统计。
336/// 词法/语法错误在详情区提示。
337void LintExplorerPanel::onRunLint() {
338 diagTable_->setRowCount(0);
339 diagDetail_->clear();
340 lastDiagnostics_.clear();
341 statsLabel_->setText(mlTr("共 0 条警告，0 条错误"));
342
343 QString raw = sourceEdit_->toPlainText();
344 if (raw.trimmed().isEmpty()) {
345 // 空源码陷阱：空源码经 Parser 产生空 Block，会触发 EmptyBlock 警告。
346 // 用户清空后给出提示而非运行分析。
347 diagDetail_->setHtml(QString::fromUtf8("<div style='color:#C0392B; padding:8px;'><b>源码为空</b>，"
348 "请输入 MiniLang 源码或点击预设样例按钮。<br>"
349 "<span style='color:#6E6E6E;'>(注意：空源码经 Parser 会产生空 Block，"
350 "可能触发 EmptyBlock 警告)</span></div>"));
351 return;
352 }
353 std::string source = raw.toStdString();
354
355 // 构建 LintOptions：未勾选的规则加入 disabledRules
356 minilang::lint::LintOptions opts;
357 opts.maxCyclomaticComplexity = complexitySpin_->value();
358 const size_t ruleCount = static_cast<size_t>(minilang::lint::LintRule::Count);
359 for (size_t i = 0; i < ruleChecks_.size() && i < ruleCount; ++i) {
360 if (!ruleChecks_[i]->isChecked()) {
361 opts.disabledRules.insert(static_cast<minilang::lint::LintRule>(i));
362 }
363 }
364
365 minilang_lint::LintSourceResult result = minilang_lint::lintSource(source, opts);
366
367 if (!result.ok) {
368 // 词法 / 语法错误阶段，分析中止
369 QString phase = result.errorPhase == "lex" ? mlTr("词法错误") : mlTr("语法错误");
370 QString errs = QString::fromUtf8(result.errorMessage.c_str()).toHtmlEscaped();
371 diagDetail_->setHtml(
372 QString::fromUtf8("<div style='color:#C0392B; padding:8px;'><b>%1:</b> %2</div>").arg(phase, errs));
373 statsLabel_->setText(mlTr("分析失败：%1").arg(phase));
374 return;
375 }
376
377 // 填充诊断表格
378 const auto& diags = result.lint.diagnostics.all();
379 lastDiagnostics_ = diags; // 拷贝副本供选中查看
380
381 int warns = 0, errs = 0, infos = 0;
382 for (const auto& d : diags) {
383 int row = diagTable_->rowCount();
384 diagTable_->insertRow(row);
385 auto* cLine = new QTableWidgetItem(QString::number(d.line));
386 auto* cCol = new QTableWidgetItem(QString::number(d.column));
387 auto* cLevel = new QTableWidgetItem(QString::fromUtf8(d.levelString().c_str()));
388 auto* cMsg = new QTableWidgetItem(QString::fromUtf8(d.message.c_str()));
389 cLine->setTextAlignment(Qt::AlignCenter);
390 cCol->setTextAlignment(Qt::AlignCenter);
391 cLevel->setTextAlignment(Qt::AlignCenter);
392 diagTable_->setItem(row, 0, cLine);
393 diagTable_->setItem(row, 1, cCol);
394 diagTable_->setItem(row, 2, cLevel);
395 diagTable_->setItem(row, 3, cMsg);
396
397 if (d.isError())
398 ++errs;
399 else if (d.isWarning())
400 ++warns;
401 else
402 ++infos;
403 }
404
405 // 统计：警告 + 错误（+ 提示，若有）
406 QString statText = mlTr("共 %1 条警告，%2 条错误").arg(warns).arg(errs);
407 if (infos > 0) {
408 statText += mlTr("，%1 条提示").arg(infos);
409 }
410 statsLabel_->setText(statText);
411
412 if (!diags.empty()) {
413 diagTable_->selectRow(0);
414 } else {
415 diagDetail_->setHtml(QString::fromUtf8("<div style='color:#27AE60; padding:8px;'><b>未触发任何 Lint 诊断</b>，"
416 "当前源码符合所有启用的规则。</div>"));
417 }
418}

420/// 选中诊断表格某行时，根据诊断规则码查找规则文档，渲染诊断详情
421/// （规则说明 + 触发条件 + 修复建议）。未知规则码仅展示诊断本身。
422void LintExplorerPanel::onDiagnosticSelected() {
423 int row = diagTable_->currentRow();
424 if (row < 0 || row >= static_cast<int>(lastDiagnostics_.size())) {
425 return;
426 }
427 const auto& d = lastDiagnostics_[row];
428 const LintRuleInfo* info = LintRuleLibrary::findByCode(d.code);
429
430 QString html;
431 if (info) {
432 html = QString::fromUtf8("<h3>%1 %2</h3>"
433 "<p><b>规则码:</b> <code>%3</code> | <b>级别:</b> %4 | "
434 "<b>位置:</b> 行 %5, 列 %6</p>"
435 "<h4>诊断消息</h4>"
436 "<p>%7</p>"
437 "<h4>触发条件</h4>"
438 "<p>%8</p>"
439 "<h4>修复建议</h4>"
440 "<p>%9</p>")
441 .arg(htmlEsc(info->emoji))
442 .arg(htmlEsc(info->title))
443 .arg(htmlEsc(info->code))
444 .arg(QString::fromUtf8(d.levelString().c_str()))
445 .arg(d.line)
446 .arg(d.column)
447 .arg(htmlEsc(d.message))
448 .arg(htmlEsc(info->trigger))
449 .arg(htmlEsc(info->fix));
450 } else {
451 html = QString::fromUtf8("<h3>诊断详情</h3>"
452 "<p><b>规则码:</b> <code>%1</code> | <b>级别:</b> %2 | "
453 "<b>位置:</b> 行 %3, 列 %4</p>"
454 "<h4>诊断消息</h4>"
455 "<p>%5</p>")
456 .arg(htmlEsc(d.code))
