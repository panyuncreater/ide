83/// 用最新操作数栈内容刷新栈视图。
84void VmStackPanel::updateStack(std::vector<Value> stack) {
85 // BUG-VSP-6 fix: 模式切换时标题适配——栈式 VM 模式
86 if (stackTitle_) {
87 stackTitle_->setText(QString::fromUtf8("操作数栈 (栈顶 ↑)"));
88 }
89
90 stackList_->clear();
91
92 // 栈顶在上，栈底在下
93 for (int i = static_cast<int>(stack.size()) - 1; i >= 0; --i) {
94 // BUG-VSP-1 fix: toString() 可能抛异常（如 NaN-boxing 解码失败），用 try/catch 兜底
95 std::string valStr;
96 try {
97 valStr = stack[i].toString();
98 ErrorFormat::truncateForError(valStr);
99 } catch (...) {
100 valStr = "<error>";
101 }
102 QString itemText;
103 if (i == static_cast<int>(stack.size()) - 1) {
104 itemText = QString("TOP [%1] %2").arg(i).arg(QString::fromStdString(valStr));
105 } else {
106 itemText = QString(" [%1] %2").arg(i).arg(QString::fromStdString(valStr));
107 }
108 auto* item = new QListWidgetItem(itemText);
109
110 // 栈顶项高亮
111 if (i == static_cast<int>(stack.size()) - 1) {
112 // D12 fix: 使用 palette Link 角色替代硬编码 #00ff88，适配主题
113 item->setForeground(palette().link().color());
114 }
115
116 stackList_->addItem(item);
117 }
118
119 if (stack.empty()) {
120 auto* item = new QListWidgetItem("(栈为空)");
121 // D12 fix: 使用 palette PlaceholderText 角色替代硬编码 #666
122 item->setForeground(palette().placeholderText().color());
123 stackList_->addItem(item);
124 }
125}

159/// 刷新全局变量区视图。
160void VmStackPanel::updateGlobals(std::unordered_map<std::string, Value> globals) {
161 // BUG-VSP-4 fix: 原实现用 unordered_map 元素裸指针排序，若 globals 在排序后
162 // 被修改/销毁，指针即悬垂。改为按值接收参数（BUG-VSP-2），并 move 到本地 vector，
163 // 排序与访问均基于本地 vector，生命周期完全独立。
164 std::vector<std::pair<std::string, Value>> entries;
165 entries.reserve(globals.size());
166 for (auto& kv : globals) {
167 entries.push_back(std::move(kv));
168 }
169 std::sort(entries.begin(), entries.end(), [](const auto& a, const auto& b) { return a.first < b.first; });
170
171 // G-P2-4 fix: 仅在行数变化时调用 resizeColumnsToContents（O(n) 操作），避免每次更新都重算列宽
172 const int newRowCount = static_cast<int>(entries.size());
173 const bool rowCountChanged = (globalsTable_->rowCount() != newRowCount);
174 globalsTable_->setRowCount(newRowCount);
175
176 int row = 0;
177 int maxValueWidth = 0; // BUG-VSP-3 fix: 跟踪值列最大文本长度
178 for (const auto& entry : entries) {
179 // BUG-VSP-1 fix: toString() 可能抛异常，用 try/catch 兜底
180 std::string valStr;
181 try {
182 valStr = entry.second.toString();
183 ErrorFormat::truncateForError(valStr);
184 } catch (...) {
185 valStr = "<error>";
186 }
187 QString valQStr = QString::fromStdString(valStr);
188 globalsTable_->setItem(row, 0, new QTableWidgetItem(QString::fromStdString(entry.first)));
189 globalsTable_->setItem(row, 1, new QTableWidgetItem(valQStr));
190 if (valQStr.size() > maxValueWidth)
191 maxValueWidth = valQStr.size();
192 ++row;
193 }
194
195 // BUG-VSP-3 fix: 除行数变化外，值列最大文本长度增长超过阈值时也重算列宽，
196 // 否则同行数下列宽不会随内容变长而扩展，导致长值被截断。
197 if (rowCountChanged || maxValueWidth > lastMaxValueWidth_ + 5) {
198 globalsTable_->resizeColumnsToContents();
199 lastMaxValueWidth_ = maxValueWidth;
200 }
201}
