68/// 统一结果类型：持有成功值 T 或错误 ErrorInfo
69/// AUDIT-R4 BUG-08 fix: 标注 [[nodiscard]]——调用方丢弃返回的 Result 即静默
70/// 吞错误（无任何诊断）。加标注后编译器在 /W4+/WX 下强制所有调用点
71/// 显式处理或显式丢弃（(void) 转换）。
72template <typename T> class [[nodiscard]] Result {
73public:
74 /// 成功构造
75 /// AUDIT-R3 P2 fix: 用 in_place_index 直接构造 variant，消除"先默认构造
76 /// Result 再 emplace"的模式——原实现隐式要求 T 可默认构造（variant 默认
77 /// 构造第 0 备选 T），且 err 路径白白构造再销毁一个 T。
78 static Result ok(T value) { return Result(std::variant<T, ErrorInfo>(std::in_place_index<0>, std::move(value))); }
79
80 /// 错误构造
81 static Result err(ErrorInfo error) {
82 return Result(std::variant<T, ErrorInfo>(std::in_place_index<1>, std::move(error)));
83 }
84
85 /// 错误构造（便捷版）
86 static Result err(std::string message, int line = 0, int column = 0) {
87 return err(ErrorInfo(std::move(message), line, column));
88 }
89
90 bool is_ok() const { return data_.index() == 0; }
91 bool is_err() const { return data_.index() == 1; }
92
93 /// 获取成功值（仅 is_ok() 时调用）
94 T& value() { return std::get<0>(data_); }
95 const T& value() const { return std::get<0>(data_); }
96
97 /// 获取错误信息（仅 is_err() 时调用）
98 const ErrorInfo& error() const { return std::get<1>(data_); }
99
100 /// 获取成功值或默认值
101 /// AUDIT-R3 P2 fix: 改为 if/else——原三目运算符混合 const 左值与右值，
102 /// 公共类型退化为纯右值导致 fallback 分支也多一次拷贝；现 fallback
103 /// 分支走函数参数的隐式 move。
104 T unwrap_or(T fallback) const {
105 if (is_ok())
106 return std::get<0>(data_);
107 return fallback;
108 }
109
110private:
111 /// AUDIT-R3 P2 fix: 私有 variant 构造（供 ok/err 工厂使用，不要求 T 可默认构造）
112 explicit Result(std::variant<T, ErrorInfo>&& v) : data_(std::move(v)) {}
113
114 std::variant<T, ErrorInfo> data_;
115};
