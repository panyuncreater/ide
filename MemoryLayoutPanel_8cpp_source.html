31/// 将 64 位整数格式化为 0x%016llX 字符串
32std::string formatHex64(uint64_t bits) {
33 char buf[32];
34 std::snprintf(buf, sizeof(buf), "0x%016llX", static_cast<unsigned long long>(bits));
35 return buf;
36}

38/// 生成 8 字节位模式可视化 HTML
39/// isPointer 为 true 时高 2 字节（tag）用红色，低 6 字节（payload）用蓝色；
40/// isPointer 为 false 时全 8 字节均为 payload（蓝色）。
41QString formatBitPatternHtml(const std::string& bitPatternHex, bool isPointer) {
42 std::string hex = bitPatternHex;
43 if (hex.size() >= 2 && hex[0] == '0' && (hex[1] == 'x' || hex[1] == 'X')) {
44 hex = hex.substr(2);
45 }
46 while (hex.size() < 16) {
47 hex = "0" + hex;
48 }
49
50 QString html = QStringLiteral(
51 "<h3>内存布局可视化（8 字节 / 64 位）</h3>"
52 "<table border='1' cellpadding='8' cellspacing='0' style='font-family:monospace;font-size:14px;'>");
53
54 // 字节序号行
55 html += QStringLiteral("<tr>");
56 for (int i = 7; i >= 0; --i) {
57 QString label = (i == 7) ? QStringLiteral("byte 7 (MSB)")
58 : (i == 0) ? QStringLiteral("byte 0 (LSB)")
59 : QStringLiteral("byte %1").arg(i);
60 html +=
61 QStringLiteral("<th style='background:%1;'>%2</th>").arg(TeachingTheme::surfaceHover().name()).arg(label);
62 }
63 html += QStringLiteral("</tr>");
64
65 // 字节值行
66 html += QStringLiteral("<tr>");
67 for (int i = 0; i < 8; ++i) {
68 std::string byteStr = hex.substr(i * 2, 2);
69 QString color;
70 if (isPointer && i < 2) {
71 color = QStringLiteral("#FFB3BA"); // tag — 粉红
72 } else {
73 color = QStringLiteral("#BAE1FF"); // payload — 浅蓝
74 }
75 html += QStringLiteral("<td bgcolor='%1' style='font-weight:bold;text-align:center;'>%2</td>")
76 .arg(color)
77 .arg(QString::fromUtf8(byteStr.c_str()).toUpper());
78 }
79 html += QStringLiteral("</tr></table>");
80
81 // 图例
82 if (isPointer) {
83 html += QStringLiteral("<p style='margin-top:8px;'>"
84 "<span style='background:#FFB3BA;padding:2px 8px;'>&nbsp;tag&nbsp;</span> "
85 "高 16 位类型标签 &nbsp;&nbsp;"
86 "<span style='background:#BAE1FF;padding:2px 8px;'>&nbsp;payload&nbsp;</span> "
87 "低 48 位数据（指针 / 整数 / 布尔值）"
88 "</p>");
89 } else {
90 html += QStringLiteral("<p style='margin-top:8px;'>"
91 "<span style='background:#BAE1FF;padding:2px 8px;'>&nbsp;payload&nbsp;</span> "
92 "全 64 位为 IEEE 754 double 编码（无 tag）"
93 "</p>");
94 }
95
96 return html;
97}

280 QString html =
281 QStringLiteral(
282 "<h2>MiniLang 内存布局原理</h2>"
283 "<p>MiniLang 的 <b>Value</b> 类型采用 <b>NaN-boxing</b> 技术，将所有类型的值统一编码为 8 字节。"
284 "堆类型（数组 / 字典 / 闭包 / 字符串）通过侵入式 <b>RefCounted</b> 基类管理生命周期，"
285 "数组与字典使用 <b>Copy-On-Write (COW)</b> 优化，闭包通过 <b>upvalue</b> 捕获外层变量。</p>"
286
287 "<h3>一、NaN-boxing（8 字节统一编码）</h3>"
288 "<p>NaN-boxing 利用 IEEE 754 double 的 NaN（Not-a-Number）空间编码非浮点类型：</p>"
289 "<ul>"
290 "<li><b>double</b>：原样存储（全 64 位 IEEE 754）。若为 NaN，整个 64 位替换为 "
291 "NAN_BOXED_FLOAT_MARKER（0x7FFC...）</li>"
292 "<li><b>bool</b>：tag=0x7FF9，最低位 0/1 表示 false/true</li>"

295 "</ul>"
296 "<p>高 16 位为类型标签（tag），低 48 位为数据载荷（payload）。"
297 "由于 IEEE 754 的 quiet NaN 高 16 位为 0x7FF8~0x7FFF，"
298 "可安全复用此空间编码非浮点类型，实现 8 字节统一存储。</p>"
299
300 "<h3>二、Copy-On-Write (COW)</h3>"
301 "<p>数组（ArrayObject）和字典（DictObject）使用 COW 策略：</p>"
302 "<ul>"
303 "<li><b>共享</b>：赋值时浅拷贝指针，refcount 加 1，不复制底层数据</li>"
304 "<li><b>写前 detach</b>：修改前检查 refcount，若 >1 则复制底层数据（detach），"
305 "使修改方独占副本</li>"
306 "<li><b>释放</b>：refcount 减至 0 时触发析构，释放堆内存</li>"
307 "</ul>"
308 "<p>COW 避免了不必要的深拷贝，同时保证语义正确性。"
309 "MiniLang 的 COW 基于 RefCounted 引用计数，而非 GC。</p>"
310
311 "<h3>三、upvalue 捕获链</h3>"
312 "<p>闭包（ClosureObject）通过 upvalue 捕获外层作用域的变量：</p>"
313 "<ul>"
314 "<li><b>open upvalue</b>：指向栈槽（栈地址 + 偏移），栈未弹出。"
315 "闭包通过 upvalue 间接读取栈上的变量。</li>"
316 "<li><b>closed upvalue</b>：值已拷贝到 upvalue 对象，栈已弹出。"
317 "upvalue 将值独立保存在堆上，闭包仍可访问。</li>"
318 "</ul>"
319 "<p>当变量离开作用域（栈弹出）时，open upvalue 转为 closed upvalue，"

434void MemoryLayoutPanel::analyzeValue(const std::string& input, std::vector<MemorySimField>& out,
435 std::string& bitPattern, bool& isPointer) {
436 out.clear();
437 bitPattern.clear();
438 isPointer = false;
439
440 if (input.empty()) {
441 out.push_back({"错误", "输入为空"});
442 out.push_back({"提示", "请输入值或选择预设场景"});
443 return;
444 }
445
446 // 1. 精确匹配预设场景
447 for (const auto& scenario : MemoryLayoutLibrary::scenarios()) {
448 if (input == scenario.input) {
449 out = scenario.fields;
450 bitPattern = scenario.bitPattern;
451 isPointer = scenario.isPointer;
452 return;
453 }
454 }
455
456 // 2. 关键词匹配：bool / null
457 if (input == "true" || input == "false") {
458 bool val = (input == "true");
459 uint64_t bits = 0x7FF9000000000000ULL | (val ? 1ULL : 0ULL);
460 bitPattern = formatHex64(bits);
461 isPointer = false;
462 out.push_back({"类型", "bool"});
463 out.push_back({"位模式", bitPattern});
464 out.push_back({"引用计数", "N/A（非堆分配）"});
465 out.push_back({"是否堆分配", "否"});
466 out.push_back({"COW 状态", "N/A"});
467 out.push_back({"说明", std::string("bool ") + input + "：tag=0x7FF9，最低位 " + (val ? "1" : "0")});
468 return;
469 }
470
471 if (input == "null") {
472 uint64_t bits = 0x7FFA000000000000ULL;
473 bitPattern = formatHex64(bits);
474 isPointer = false;
475 out.push_back({"类型", "null"});
476 out.push_back({"位模式", bitPattern});
477 out.push_back({"引用计数", "N/A（非堆分配）"});
478 out.push_back({"是否堆分配", "否"});
479 out.push_back({"COW 状态", "N/A"});
480 out.push_back({"说明", "null：tag=0x7FFA，无 payload"});
481 return;
482 }
483
484 // 3. 关键词匹配：array / closure
485 if (input.find("array") != std::string::npos || input.find('[') != std::string::npos ||
486 input.find("数组") != std::string::npos) {
487 bitPattern = "0x7FFB00000040A230";
488 isPointer = true;
489 out.push_back({"类型", "array（堆分配，pointer tag）"});
490 out.push_back({"位模式", bitPattern});
491 out.push_back({"引用计数", "1（独占）→ 2（共享）→ 1（detach 后）"});
492 out.push_back({"是否堆分配", "是"});
493 out.push_back({"COW 状态", "写前 detach：refcount>1 时复制底层数据"});
494 out.push_back({"说明", "数组通过 pointer tag（0x7FFB）引用堆上的 ArrayObject，COW 共享时 refcount>1"});
495 return;
496 }
497
498 if (input.find("closure") != std::string::npos || input.find("fun") != std::string::npos ||
499 input.find("闭包") != std::string::npos) {
500 bitPattern = "0x7FFB00000040A3B0";
501 isPointer = true;
502 out.push_back({"类型", "closure（堆分配，pointer tag）"});
503 out.push_back({"位模式", bitPattern});
504 out.push_back({"引用计数", "1"});
505 out.push_back({"是否堆分配", "是"});
506 out.push_back({"COW 状态", "N/A（闭包不参与 COW）"});
507 out.push_back({"说明", "闭包通过 pointer tag（0x7FFB）引用堆上的 ClosureObject，含 upvalue 捕获链"});
508 return;
509 }
510
511 // 4. 尝试解析为数字（double 编码）
512 try {
513 size_t pos = 0;
514 long long intVal = std::stoll(input, &pos);
515 if (pos == input.size()) {
516 double dval = static_cast<double>(intVal);
517 uint64_t bits;
518 std::memcpy(&bits, &dval, sizeof(bits));
519 bitPattern = formatHex64(bits);
520 isPointer = false;
521 out.push_back({"类型", "double（整数按 double 编码）"});
522 out.push_back({"位模式", bitPattern});
523 out.push_back({"引用计数", "N/A（非堆分配）"});
524 out.push_back({"是否堆分配", "否"});
525 out.push_back({"COW 状态", "N/A"});
526 out.push_back({"说明", "整数 " + input + " 按 IEEE 754 double 编码，8 字节全为 payload，无 tag"});
527 return;
528 }
529 } catch (...) {
530 // 不是整数，继续尝试浮点
531 }
532
533 try {
534 size_t pos = 0;
535 double dval = std::stod(input, &pos);
536 if (pos == input.size()) {
537 uint64_t bits;
538 std::memcpy(&bits, &dval, sizeof(bits));
539 bitPattern = formatHex64(bits);
540 isPointer = false;
541 out.push_back({"类型", "double"});
542 out.push_back({"位模式", bitPattern});
543 out.push_back({"引用计数", "N/A（非堆分配）"});
544 out.push_back({"是否堆分配", "否"});
545 out.push_back({"COW 状态", "N/A"});
546 out.push_back({"说明", "浮点 " + input + " 按 IEEE 754 double 编码，8 字节全为 payload，无 tag"});
547 return;
548 }
549 } catch (...) {
550 // 不是数字
551 }
552
553 // 5. 无法识别
554 out.push_back({"错误", "无法识别的输入"});
555 out.push_back({"提示", "请输入数字（如 42, 3.14）、true/false/null、array、closure，或选择预设场景"});
556}

558void MemoryLayoutPanel::renderAnalysis(const std::vector<MemorySimField>& fields, const std::string& bitPattern,
559 bool isPointer) {
560 // 值类型分析表
561 fieldTable_->setRowCount(static_cast<int>(fields.size()));
562 for (int i = 0; i < static_cast<int>(fields.size()); ++i) {
563 const auto& f = fields[i];
564 fieldTable_->setItem(i, 0, new QTableWidgetItem(QString::fromUtf8(f.field.c_str())));
565 fieldTable_->setItem(i, 1, new QTableWidgetItem(QString::fromUtf8(f.value.c_str())));
566 fieldTable_->item(i, 1)->setToolTip(QString::fromUtf8(f.value.c_str()));
567 }
568
569 // 内存布局可视化
570 if (bitPattern.empty()) {
571 layoutBrowser_->setHtml(tr("<p style='color:red;'>无法生成位模式可视化，请检查输入</p>"));
572 return;
573 }
574
575 QString html = formatBitPatternHtml(bitPattern, isPointer);
576 html += QStringLiteral("<hr><p style='color:%1;font-size:small;'>"
577 "说明：byte 7 为最高位字节（MSB），byte 0 为最低位字节（LSB）。"
578 "NaN-boxing 利用 IEEE 754 quiet NaN 的高 16 位空间（0x7FF8~0x7FFF）编码非浮点类型。"
579 "</p>")
580 .arg(TeachingTheme::textHint().name());
581 layoutBrowser_->setHtml(html);
582}
