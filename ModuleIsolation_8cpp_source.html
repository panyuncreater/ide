15/// 提取顶层声明节点的 name（VarDecl/FunDecl/ClassDecl），非顶层声明返回空
16std::string getTopLevelDeclName(const ASTNode* node) {
17 if (!node)
18 return "";
19 switch (node->nodeType) {
20 case NodeType::NODE_VAR_DECL:
21 return static_cast<const VarDecl*>(node)->name;
22 case NodeType::NODE_FUN_DECL:
23 return static_cast<const FunDecl*>(node)->name;
24 case NodeType::NODE_CLASS_DECL:
25 return static_cast<const ClassDecl*>(node)->name;
26 default:
27 return "";
28 }
29}

56std::string ModuleTopLevelRenamer::pathHash(const std::string& modulePath) {
57 // AUDIT-R5 R6 fix: 规范化改用 normalizeModulePathKey（common/ModulePath.h 单一
58 // 事实源），与 Interpreter/Compiler/AstIRBuilder 的缓存键规范化完全一致，
59 // 避免隔离前缀（__mod_<hash>__）与缓存键因规范化规则不同而不匹配。
60 // 原实现仅在此处折叠连续斜杠（AUDIT-P2-ROUND53），与其他三处不同；
61 // 现共享函数同时处理双斜杠折叠与 Windows 大小写折叠。
62 std::string normalized = normalizeModulePathKey(modulePath);
63 // AUDIT-R4 BUG-09 fix: FNV-1a 由 32 位升级为 64 位（输出 16 字符 hex）。
64 // 32 位哈希碰撞概率 2^-32，两个不同模块路径碰撞时隔离前缀相同，
65 // 非导出顶层符号跨模块串名且无任何诊断。64 位将碰撞概率降至 2^-64。
66 uint64_t h = 14695981039346656037ull;
67 for (char c : normalized) {
68 h ^= static_cast<uint8_t>(c);
69 h *= 1099511628211ull;
70 }
71 std::ostringstream oss;
72 oss << std::hex << std::setfill('0') << std::setw(16) << h;
73 return oss.str();
74}

76void ModuleTopLevelRenamer::collectNonExportTopLevelNames(Block& moduleAst) {
77 // 第 1 步：收集所有导出名（ExportStmt 包装的声明名）
78 std::unordered_set<std::string> exportNames;
79 for (const auto& stmt : moduleAst.statements) {
80 if (!stmt || stmt->nodeType != NodeType::NODE_EXPORT_STMT)
81 continue;
82 auto* exp = static_cast<ExportStmt*>(stmt.get());
83 if (!exp->declaration)
84 continue;
85 std::string n = getTopLevelDeclName(exp->declaration.get());
86 if (!n.empty())
87 exportNames.insert(n);
88 }
89
90 // 第 2 步：收集所有非导出顶层声明名，构建 renameMap
91 const std::string prefix = "__mod_" + pathHash(modulePath_) + "__";
92 for (const auto& stmt : moduleAst.statements) {
93 if (!stmt)
94 continue;
95 // 跳过 ExportStmt（其内部声明是导出的，不重命名）
96 if (stmt->nodeType == NodeType::NODE_EXPORT_STMT)
97 continue;
98 // 跳过 ImportStmt（嵌套模块有自己的命名空间）
99 if (stmt->nodeType == NodeType::NODE_IMPORT_STMT)
100 continue;
101 // AUDIT-R6 F8 fix: 顶层解构绑定的每个名字也参与隔离重命名。原实现
102 // getTopLevelDeclName 不识别该节点，解构名以原名占据全局槽——导入方可用
103 // 原名直接读到模块内部值（实证：VM=7，Interpreter=未定义变量，三后端不一致），
104 // 且反向覆写导入方同名全局。解构名不可能被导出（export 仅包装
105 // VarDecl/FunDecl/ClassDecl），故无需 exportNames 豁免。
106 if (stmt->nodeType == NodeType::NODE_DESTRUCTURE_BINDING) {
107 auto* db = static_cast<DestructureBinding*>(stmt.get());
108 for (const auto& bn : db->names) {
109 renameMap_[bn] = prefix + bn;
110 scopeStack_.front().insert(bn);
111 }
112 continue;
113 }
114 std::string n = getTopLevelDeclName(stmt.get());
115 if (n.empty())
116 continue;
117 if (exportNames.count(n))
118 continue; // 导出名不重命名
119 renameMap_[n] = prefix + n;
120 // 同时登记到模块顶层作用域
121 scopeStack_.front().insert(n);
122 }
123 // 导出名也登记到模块顶层作用域（用于作用域分析，但不重命名）
124 for (const auto& name : exportNames) {
125 scopeStack_.front().insert(name);
126 }
127}

129bool ModuleTopLevelRenamer::resolveToModuleTopLevel(const std::string& name) const {
130 // 从内向外查找（栈顶到栈底），跳过栈底（模块顶层）
131 if (scopeStack_.size() <= 1)
132 return true; // 仅在模块顶层作用域
133 for (size_t i = scopeStack_.size() - 1; i >= 1; --i) {
134 if (scopeStack_[i].count(name))
135 return false; // 内层作用域定义了同名变量
136 }
137 return true; // 所有内层作用域都未定义，解析到模块顶层
138}

140void ModuleTopLevelRenamer::renameInBlock(Block* node) {
141 if (!node)
142 return;
143 // Block 进入新作用域（除非是模块顶层 Block，已在构造时压栈）
144 bool isModuleTopLevel = (scopeStack_.size() == 1);
145 if (!isModuleTopLevel)
146 pushScope();
147 for (const auto& stmt : node->statements) {
148 renameInNode(stmt.get());
149 }
150 if (!isModuleTopLevel)
151 popScope();
152}

154void ModuleTopLevelRenamer::renameInNode(ASTNode* node) {
155 if (!node)
156 return;
157 switch (node->nodeType) {
158 // ---- 顶层声明节点：重命名 name（如果在 renameMap 且当前是模块顶层）----
159 case NodeType::NODE_VAR_DECL: {
160 auto* n = static_cast<VarDecl*>(node);
161 // 模块顶层的 VarDecl 才重命名；内层 VarDecl 是局部变量，登记到当前作用域
162 if (scopeStack_.size() == 1 && renameMap_.count(n->name)) {
163 n->name = renameMap_[n->name];
164 }
165 // BUG-INH-AUDIT-5 fix: 类字段声明（inClassBody_=true）不登记到作用域。
166 // 字段只能通过 this.field 访问（MemberAccess），不参与变量名解析。
167 // 原实现将字段名登记到类作用域，遮蔽同名模块顶层变量，导致方法体内
168 // 引用该变量时不重命名，运行时报"未定义变量"。
169 if (!inClassBody_) {
170 defineInCurrentScope(n->name);
171 }
172 if (n->initializer)
173 renameInNode(n->initializer.get());
174 break;
175 }
176 case NodeType::NODE_FUN_DECL: {
177 auto* n = static_cast<FunDecl*>(node);
178 if (scopeStack_.size() == 1 && renameMap_.count(n->name)) {
179 n->name = renameMap_[n->name];
180 }
181 defineInCurrentScope(n->name);
182 // 函数体进入新作用域，参数登记到函数作用域
183 // BUG-INH-AUDIT-5 fix: 进入函数体后不再处于类成员上下文
184 // （方法内的局部变量是普通变量，应正常登记到作用域）
185 bool savedInClassBody = inClassBody_;
186 inClassBody_ = false;
187 pushScope();
188 for (const auto& param : n->params)
189 defineInCurrentScope(param);
190 if (n->body)
191 renameInNode(n->body.get());
192 popScope();
193 inClassBody_ = savedInClassBody;
194 // 默认参数表达式在函数作用域外求值，但也需要重命名其中的引用
195 for (const auto& dv : n->defaultValues) {
196 if (dv)
197 renameInNode(dv.get());
198 }
199 break;
200 }
201 case NodeType::NODE_CLASS_DECL: {
202 auto* n = static_cast<ClassDecl*>(node);
203 if (scopeStack_.size() == 1 && renameMap_.count(n->name)) {
204 n->name = renameMap_[n->name];
205 }
206 // 父类名引用：如果父类是模块顶层非导出类，重命名
207 if (!n->superClassName.empty() && resolveToModuleTopLevel(n->superClassName) &&
208 renameMap_.count(n->superClassName)) {
209 n->superClassName = renameMap_[n->superClassName];
210 }
211 defineInCurrentScope(n->name);
212 // 类成员（方法）进入新作用域，this 隐式绑定
213 pushScope();
214 defineInCurrentScope("this");
215 // BUG-INH-AUDIT-5 fix: 标记进入类成员上下文，字段声明不登记到作用域
216 bool savedInClassBody = inClassBody_;
217 inClassBody_ = true;
218 for (const auto& m : n->members) {
219 if (m)
220 renameInNode(m.get());
221 }
222 inClassBody_ = savedInClassBody;
223 popScope();
224 break;
225 }
226 // ---- 引用节点：根据作用域分析决定是否重命名 ----
227 case NodeType::NODE_VAR_REF: {
228 auto* n = static_cast<VarRef*>(node);
229 if (resolveToModuleTopLevel(n->name) && renameMap_.count(n->name)) {
230 n->name = renameMap_[n->name];
231 }
232 break;
233 }
234 case NodeType::NODE_ASSIGNMENT: {
235 auto* n = static_cast<Assignment*>(node);
236 if (resolveToModuleTopLevel(n->name) && renameMap_.count(n->name)) {
237 n->name = renameMap_[n->name];
238 }
239 if (n->value)
240 renameInNode(n->value.get());
241 break;
242 }
243 case NodeType::NODE_FUN_CALL: {
244 auto* n = static_cast<FunCall*>(node);
245 // FunCall.name 引用全局函数（当 callee 为 null 时）
246 if (!n->callee && resolveToModuleTopLevel(n->name) && renameMap_.count(n->name)) {
247 n->name = renameMap_[n->name];
248 }
249 if (n->callee)
250 renameInNode(n->callee.get());
251 for (const auto& arg : n->arguments) {
252 if (arg)
253 renameInNode(arg.get());
254 }
255 break;
256 }
257 // ---- 控制流节点：递归处理子节点 ----
258 case NodeType::NODE_IF_STMT: {
259 auto* n = static_cast<IfStmt*>(node);
260 if (n->condition)
261 renameInNode(n->condition.get());
262 if (n->thenBranch)
263 renameInNode(n->thenBranch.get());
264 if (n->elseBranch)
265 renameInNode(n->elseBranch.get());
266 break;
267 }
268 case NodeType::NODE_WHILE_STMT: {
269 auto* n = static_cast<WhileStmt*>(node);
270 if (n->condition)
271 renameInNode(n->condition.get());
272 if (n->body)
273 renameInNode(n->body.get());
274 break;
275 }
276 case NodeType::NODE_FOR_STMT: {
277 auto* n = static_cast<ForStmt*>(node);
278 // for (init; cond; upd) { body }
279 // init 可能是 VarDecl，进入新作用域
280 pushScope();
281 if (n->initializer)
282 renameInNode(n->initializer.get());
283 if (n->condition)
284 renameInNode(n->condition.get());
285 if (n->update)
286 renameInNode(n->update.get());
287 if (n->body)
288 renameInNode(n->body.get());
289 popScope();
290 break;
291 }
292 case NodeType::NODE_BLOCK: {
293 renameInBlock(static_cast<Block*>(node));
294 break;
295 }
296 case NodeType::NODE_RETURN_STMT: {
297 auto* n = static_cast<ReturnStmt*>(node);
298 if (n->value)
299 renameInNode(n->value.get());
300 break;
301 }
302 case NodeType::NODE_PRINT_STMT: {
303 auto* n = static_cast<PrintStmt*>(node);
304 for (const auto& v : n->values) {
305 if (v)
306 renameInNode(v.get());
307 }
308 break;
309 }
310 // ---- 表达式节点：递归处理子节点 ----
311 case NodeType::NODE_BINARY_OP: {
312 auto* n = static_cast<BinaryOp*>(node);
313 if (n->left)
314 renameInNode(n->left.get());
315 if (n->right)
316 renameInNode(n->right.get());
317 break;
318 }
319 case NodeType::NODE_UNARY_OP: {
320 auto* n = static_cast<UnaryOp*>(node);
321 if (n->operand)
322 renameInNode(n->operand.get());
323 break;
324 }
325 case NodeType::NODE_ARRAY_LITERAL: {
326 auto* n = static_cast<ArrayLiteral*>(node);
327 for (const auto& e : n->elements) {
328 if (e)
329 renameInNode(e.get());
330 }
331 break;
332 }
333 case NodeType::NODE_DICT_LITERAL: {
334 auto* n = static_cast<DictLiteral*>(node);
335 for (const auto& p : n->pairs) {
336 if (p.first)
337 renameInNode(p.first.get());
338 if (p.second)
339 renameInNode(p.second.get());
340 }
341 break;
342 }
343 case NodeType::NODE_INDEX_ACCESS: {
344 auto* n = static_cast<IndexAccess*>(node);
345 if (n->object)
346 renameInNode(n->object.get());
347 if (n->index)
348 renameInNode(n->index.get());
349 break;
350 }
351 case NodeType::NODE_INDEX_ASSIGN: {
352 auto* n = static_cast<IndexAssign*>(node);
353 if (n->object)
354 renameInNode(n->object.get());
355 if (n->index)
356 renameInNode(n->index.get());
357 if (n->value)
358 renameInNode(n->value.get());
359 break;
360 }
361 case NodeType::NODE_MEMBER_ACCESS: {
362 auto* n = static_cast<MemberAccess*>(node);
363 if (n->object)
364 renameInNode(n->object.get());
365 // fieldName 不是变量名，不重命名
366 break;
367 }
368 case NodeType::NODE_MEMBER_ASSIGN: {
369 auto* n = static_cast<MemberAssign*>(node);
370 if (n->object)
371 renameInNode(n->object.get());
372 if (n->value)
373 renameInNode(n->value.get());
374 break;
375 }
376 case NodeType::NODE_METHOD_CALL: {
377 auto* n = static_cast<MethodCall*>(node);
378 if (n->object)
379 renameInNode(n->object.get());
380 for (const auto& a : n->arguments) {
381 if (a)
382 renameInNode(a.get());
383 }
384 // methodName 不是变量名，不重命名
385 break;
386 }
387 case NodeType::NODE_TRY_STMT: {
388 auto* n = static_cast<TryStmt*>(node);
389 if (n->tryBlock)
390 renameInNode(n->tryBlock.get());
391 // catch 块进入新作用域，catchVarName 是局部变量
392 pushScope();
393 if (!n->catchVarName.empty())
394 defineInCurrentScope(n->catchVarName);
395 if (n->catchBlock)
396 renameInNode(n->catchBlock.get());
397 popScope();
398 // AUDIT-P1 fix: finally 块必须引入新作用域（与 catch 块一致）。
399 // 原注释"finally 块不引入新作用域（无 catch 变量）"是误解：finally 块内可包含
400 // VarDecl，这些局部变量应限制在 finally 块作用域内。若不 pushScope，finally 块内
401 // 声明的局部变量会污染外层作用域，导致 resolveToModuleTopLevel 误判，finally 块内
402 // 对模块非导出顶层名的引用不被重命名，VM/IR 路径运行时报"未定义变量"。
403 if (n->finallyBlock) {
404 pushScope();
405 renameInNode(n->finallyBlock.get());
406 popScope();
407 }
408 break;
409 }
410 case NodeType::NODE_THROW_STMT: {
411 auto* n = static_cast<ThrowStmt*>(node);
412 if (n->expression)
413 renameInNode(n->expression.get());
414 break;
415 }
416 case NodeType::NODE_INTERPOLATED_STRING: {
417 auto* n = static_cast<InterpolatedString*>(node);
418 for (const auto& expr : n->expressions) {
419 if (expr)
420 renameInNode(expr.get());
421 }
422 break;
423 }
424 // ---- R98/R99/R164 新增节点（AUDIT-R5 BUG-09 fix）----
425 // 原 switch 缺失以下分支且无兜底递归，子表达式被整体跳过：模块内
426 // match/元组/enum variant/解构/yield 中对非导出顶层名的引用不被重命名，
427 // VM/IR 路径运行时报“未定义的变量”（与历史遗漏 FunDecl/ExportStmt 同模式）。
428 case NodeType::NODE_TUPLE_LITERAL: {
429 auto* n = static_cast<TupleLiteral*>(node);
430 for (const auto& e : n->elements) {
431 if (e)
432 renameInNode(e.get());
433 }
434 break;
435 }
436 case NodeType::NODE_DESTRUCTURE_BINDING: {
437 auto* n = static_cast<DestructureBinding*>(node);
438 if (n->initializer)
439 renameInNode(n->initializer.get());
440 // AUDIT-R6 F8 fix: 模块顶层解构名重命名（collectNonExportTopLevelNames 已入
441 // renameMap），内层解构名仍为局部变量不重命名（与 VarDecl 处理一致）。
442 // 登记用重命名后的名字，使后续引用解析一致。
443 if (!inClassBody_) {
444 for (auto& bn : n->names) {
445 if (scopeStack_.size() == 1 && renameMap_.count(bn)) {
446 bn = renameMap_[bn];
447 }
448 defineInCurrentScope(bn);
449 }
450 }
451 break;
452 }
453 case NodeType::NODE_ENUM_DECL: {
454 auto* n = static_cast<EnumDecl*>(node);
455 // enum 名不参与重命名，仅登记作用域；variant 参数是类型注解，无表达式子节点。
456 defineInCurrentScope(n->name);
457 break;
458 }
459 case NodeType::NODE_ENUM_VARIANT_EXPR: {
460 auto* n = static_cast<EnumVariantExpr*>(node);
461 for (const auto& a : n->arguments) {
462 if (a)
463 renameInNode(a.get());
464 }
465 break;
466 }
467 case NodeType::NODE_YIELD_EXPR: {
468 auto* n = static_cast<YieldExpr*>(node);
469 if (n->value)
470 renameInNode(n->value.get());
471 break;
472 }
473 // 七特性 MVP 阶段 2：宏系统。
474 // MacroDecl 的 bodyExpr 是未展开的模板（运行期不求值），不参与重命名；
475 // MacroCallExpr 仅重命名 expanded 语义子树（args 子树与 expanded 内替换点
476 // 共享同一节点，重命名 expanded 已同步覆盖 args，重复访问会因 map 查不到
477 // 已改名的新名而幂等，但避免二次遍历更稳健）。
478 case NodeType::NODE_MACRO_DECL:
479 break;
480 // 七特性 MVP 阶段 3：trait 声明——方法已在 parse 期合入混入类的 members
481 // （共享节点经 ClassDecl 分支处理），TraitDecl 本身运行期不求值，不参与重命名。
482 case NodeType::NODE_TRAIT_DECL:
483 break;
484 // 七特性 MVP 阶段 4：await 表达式——重命名 operand 子树。
485 case NodeType::NODE_AWAIT_EXPR: {
486 auto* n = static_cast<AwaitExpr*>(node);
487 if (n->operand)
488 renameInNode(n->operand.get());
489 break;
490 }
491 case NodeType::NODE_MACRO_CALL: {
492 auto* n = static_cast<MacroCallExpr*>(node);
493 if (n->expanded)
494 renameInNode(n->expanded.get());
495 break;
496 }
497 case NodeType::NODE_MATCH_EXPR: {
498 // 注：MatchPattern 复用 NODE_MATCH_EXPR 类型标记，但 pattern 仅经
499 // MatchExpr::cases 可达（本函数不以 pattern 为入口被调用），
500 // 此处 static_cast<MatchExpr*> 安全。
501 auto* n = static_cast<MatchExpr*>(node);
502 if (n->scrutinee)
503 renameInNode(n->scrutinee.get());
504 for (auto& c : n->cases) {
505 // 每个 case 引入新作用域：pattern 绑定变量局部于 guard 与 body
506 pushScope();
507 if (c.pattern)
508 collectMatchPatternBindings(c.pattern.get());
509 if (c.guard)
510 renameInNode(c.guard.get());
511 if (c.body)
512 renameInNode(c.body.get());
513 popScope();
514 }
515 break;
516 }
517 // ---- 不处理的节点 ----
518 case NodeType::NODE_IMPORT_STMT:
519 // 嵌套 import 不递归（嵌套模块有自己的命名空间）
520 break;
521 case NodeType::NODE_EXPORT_STMT:
522 // ExportStmt 在 collectNonExportTopLevelNames 中已处理（导出名不重命名）
523 // 但其内部声明的 initializer 需要递归处理
524 if (auto* exp = static_cast<ExportStmt*>(node); exp->declaration) {
525 // 导出声明本身不重命名 name，但其 initializer/value 中的引用需要重写
526 // 例如: export var x = internalVar + 1; — internalVar 需要重命名
527 renameInNode(exp->declaration.get());
528 }
529 break;
530 // ---- 叶子节点：无子节点 ----
531 case NodeType::NODE_NUMBER_LITERAL:
532 case NodeType::NODE_STRING_LITERAL:
533 case NodeType::NODE_BOOL_LITERAL:
534 case NodeType::NODE_NULL_LITERAL:
535 case NodeType::NODE_SUPER_EXPR:
536 case NodeType::NODE_BREAK_STMT:
537 case NodeType::NODE_CONTINUE_STMT:
538 break;
539 }

542// AUDIT-R5 BUG-09 fix: 递归登记 match pattern 的绑定变量到当前作用域。
543// VARIABLE 模式绑定 variableName；VARIANT/TUPLE/OR 递归 subPatterns；
544// LITERAL 模式的字面量（数字/字符串/布尔/null）无变量引用，无需处理。
545void ModuleTopLevelRenamer::collectMatchPatternBindings(const MatchPattern* pattern) {
546 if (!pattern)
547 return;
548 if (pattern->kind == MatchPatternKind::VARIABLE && !pattern->variableName.empty()) {
