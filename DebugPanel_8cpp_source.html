119/// 仅处理变量叶子节点（标记 UserRole="var"）的值列（column 1），
120/// 分组标题列 0 与程序化刷新（updatingVariables_）不触发写回。
121void DebugPanel::onVariableItemChanged(QTreeWidgetItem* item, int column) {
122 if (!item || updatingVariables_)
123 return;
124 if (item->data(0, Qt::UserRole).toString() != QStringLiteral("var"))
125 return; // 分组节点不可编辑
126 if (column == 0) {
127 // 名称列只读：恢复原名（toolTip(0) 由 populate 保存），不触发写回
128 const QString orig = item->toolTip(0);
129 if (item->text(0) != orig)
130 item->setText(0, orig);
131 return;
132 }
133 if (column != 1)
134 return;
135 const QString name = item->text(0);
136 const QString newText = item->text(1);
137 const QString oldText = item->toolTip(1); // populate 时 toolTip 保存原值
138 if (newText == oldText)
139 return; // 未实际修改
140
141 bool ok = false;
142 if (variableEditCallback_) {
143 ok = variableEditCallback_(name, newText);
144 }
145 if (!ok) {
146 // 解析失败/不可写：恢复原值并提示
147 item->setText(1, oldText);
148 QToolTip::showText(
149 QCursor::pos(),
150 QStringLiteral("写入变量 %1 失败（支持 int / float / true / false / null / \"字符串\"，且需调试暂停）")
151 .arg(name));
152 return;
153 }
154 // 写入成功：同步 toolTip，供后续未修改检测与失败恢复
155 item->setToolTip(1, newText);

274/// 更新调用栈列表，截断超长栈并恢复选中行状态。
275void DebugPanel::updateCallStack(const std::vector<CallStackEntry>& stack) {
276 // AUDIT-P2 fix: 截断 currentStack_ 到显示上限，避免存储完整调用栈的全部 Value 拷贝。
277 // 列表仅显示前 200 帧（见下方 MAX_CALL_STACK_DISPLAY），但原实现 currentStack_ = stack
278 // 拷贝完整 stack（深度递归如 fib(200) 接近 MAX_RECURSION_DEPTH=256 时为 256 帧 × 10 变量），
279 // 200+ 帧的 Value 拷贝永不访问却持有 RefCounted 引用计数阻止 GC 回收堆对象。
280 constexpr int MAX_CALL_STACK_STORE = 200;
281 if (stack.size() > MAX_CALL_STACK_STORE) {
282 currentStack_.assign(stack.begin(), stack.begin() + MAX_CALL_STACK_STORE);
283 } else {
284 currentStack_ = stack;
285 }
286
287 // 保存当前选中行，刷新后恢复
288 int savedRow = callStackList_->currentRow();
289
290 // D10 fix: QSignalBlocker RAII 替代手工 blockSignals(true/false) 配对——
291 // 析构时恢复进入前状态，提前返回/异常路径不再泄漏阻塞状态。
292 // 阻塞信号防止 clear/addItem 触发 currentRowChanged 级联更新变量树
293 const QSignalBlocker callStackSignalBlocker(callStackList_);
294 callStackList_->clear();
295
296 // BUG-DBG-G3 fix (P2): 限制调用栈显示帧数上限，避免深度递归（如未优化的
297 // 斐波那契 fib(40)）产生上万帧导致 QListWidget 卡顿与内存膨胀。
298 // 上限 200 帧对调试场景足够（用户通常只关心最近的调用层级）。
299 constexpr int MAX_CALL_STACK_DISPLAY = 200;
300 int displayed = 0;
301 for (size_t i = 0; i < stack.size() && displayed < MAX_CALL_STACK_DISPLAY; ++i) {
302 const auto& frame = stack[i];
303 QString text = QString("函数: %1 @ 行 %2 (深度: %3)")
304 .arg(QString::fromStdString(frame.functionName))
305 .arg(frame.line)
306 .arg(frame.depth);
307 callStackList_->addItem(text);
308 ++displayed;
309 }
310 if (static_cast<int>(stack.size()) > MAX_CALL_STACK_DISPLAY) {
311 // 添加提示项标注未显示的帧数
312 auto* item = new QListWidgetItem(
313 QString::fromUtf8("... (还有 %1 帧未显示)").arg(static_cast<int>(stack.size()) - MAX_CALL_STACK_DISPLAY));
314 // 提示项设为不可选中，避免与真实栈帧混淆
315 item->setFlags(item->flags() & ~Qt::ItemIsSelectable);
316 callStackList_->addItem(item);
317 }
318
319 // 恢复选中行（若仍在有效范围内）
320 // BUG-DBG-G1 fix (P2): setCurrentRow(savedRow) 若在信号未阻塞时调用，
321 // 会触发 currentRowChanged 信号 → onStackFrameSelected → populateVariableTree，
322 // 覆盖刚由 updateVariables 设置的完整变量视图（仅显示选中帧的局部变量）。
323 // 此处保持 QSignalBlocker 阻塞（至函数退出），仅恢复视觉选中状态，不触发
324 // 变量树更新，让用户保留 updateVariables 提供的完整作用域视图。
325 if (savedRow >= 0 && savedRow < callStackList_->count()) {
326 callStackList_->setCurrentRow(savedRow);
327 }
328}

330/// 选中栈帧时按作用域展示其局部变量并请求跳转源码行。
331void DebugPanel::onStackFrameSelected(int index) {
332 if (index < 0 || index >= static_cast<int>(currentStack_.size()))

334
335 const auto& frame = currentStack_[index];
336 // Round 7: 选中栈帧时按作用域分组展示
337 // depth==0 的帧标记为"局部"，其他帧标记为"外层"（闭包）
338 const QString scopeLabel = (frame.depth == 0) ? QStringLiteral("局部") : QStringLiteral("外层");
339 std::vector<std::tuple<QString, QString, QString>> rows;
340 rows.reserve(frame.locals.size());

342 // R52-1 fix: 对齐 updateVariables / VariableInspectorPanel 的 try/catch 防护。

348 } catch (...) {
