26bool BytecodeIRBackend::lower(const IRFunction& ir) {
27 // 全新 chunk，避免旧哈希表/状态残留
28 chunk_ = std::make_unique<BytecodeChunk>();
29 chunk_->name = ir.name;
30 // 栈式 VM 约定：方法的 arity/requiredArity 不含 this（this 作为 slot 0 单独推送）。
31 // 但 IR 在 visitFunDecl 中为 RegVM 约定对方法 arity/requiredArity += 1（含 this）。
32 // 此处还原为栈式 VM 约定：用 IRFunction::isMethod 显式标志（D6 fix，不再从函数名
33 // 含 '.' 推断——嵌套类/模块函数名同样可能含 '.'），减去 this。
34 // 否则栈 VM 的 executeCall/executeClassNew 会因 argCount < requiredArity 触发
35 // "构造函数 init 期望 1-1 个参数，但传入了 0 个" 错误（回归：NestedMemberAccessPush）。
36 const bool isMethod = ir.isMethod;
37 chunk_->arity = isMethod ? (ir.arity > 0 ? ir.arity - 1 : 0) : ir.arity;
38 chunk_->requiredArity = isMethod ? (ir.requiredArity > 0 ? ir.requiredArity - 1 : 0) : ir.requiredArity;
39 chunk_->localCount = ir.localCount;
40 chunk_->defaultConstIndices = ir.defaultConstIndices;
41 chunk_->upvalues = ir.upvalues; // VM-05/06: 复制 upvalue 描述符
42 // BUG-IDE-12 fix: 复制 slot→name 映射，供栈式 VM 条件断点求值反查变量名
43 chunk_->localSlotNames = ir.localSlotNames;
44 // R164 协程/生成器：复制生成器标记和 yieldCount
45 chunk_->isGenerator = ir.isGenerator;
46 chunk_->yieldCount = ir.yieldCount;
47 vregStackDepth_.clear();
48 labelToOffset_.clear();
49 pendingJumps_.clear();
50 irToBytecodeOffset_.clear(); // 方向四：重置映射
51 // D1 fix: 重置栈平衡校验器状态
52 simStackDepth_ = 0;
53 labelJumpDepth_.clear();
54 labelCatchDepth_.clear();
55 traceName_ = ir.name; // 临时诊断用
56
57 // 通过 addConstant 复制常量池（保持索引一致，并填充哈希表以便后续去重）
58 for (const auto& c : ir.constants) {
59 chunk_->addConstant(c);
60 }
61
62 // 遍历所有基本块的所有指令，逐条 lowering
63 size_t instrIndex = 0; // 方向四：展平的 IR 指令序号（与 IrViewer 一致）
64 for (const auto& block : ir.blocks) {
65 for (const auto& instr : block.instructions) {
66 // 方向四：记录 IR 指令 → 当前字节码偏移映射（lowering 前）
67 irToBytecodeOffset_.push_back({instrIndex, chunk_->code.size()});
68 if (!lowerInstruction(instr, ir)) {
69 return false;
70 }
71#ifndef NDEBUG
72 // D1 fix: debug 构建逐指令校验栈平衡（Release 零开销）。
73 // 在 lowerInstruction 成功后调用——操作数布局问题由既有边界检查先行报错。
74 if (!checkStackBalance(instr)) {
75 return false;
76 }
77#endif
78 // 填充行号表（每条 IR 指令对应若干字节，统一用 instr.line）
79 // D12 fix: BUG-IBACKEND-2 修复——columns 填充 instr.column（原恒 0）
80 while (chunk_->lines.size() < chunk_->code.size()) {
81 chunk_->lines.push_back(instr.line);
82 chunk_->columns.push_back(instr.column);
83 }
84 ++instrIndex;
85 }
86 }
87
88 // 第二遍：回填跳转目标
89 if (!patchJumps())
90 return false;
91
92 // L1 fix: 将 IR 指令范围的 slotNameRanges 翻译为字节码 IP 范围。
93 // irToBytecodeOffset_ 是 (instrIndex, bytecodeOffset) 的有序映射，
94 // 用二分查找将 startInstr/endInstr 转为字节码 IP。endInstr=0 表示函数级
95 // 变量（未关闭），startIp=对应字节码偏移，endIp=chunk 末尾（fallback 到 localSlotNames）。
96 if (!ir.slotNameRanges.empty() && !irToBytecodeOffset_.empty()) {
97 auto translateInstr = [this](size_t instrIdx) -> size_t {
98 // 在 irToBytecodeOffset_ 中找 instrIdx 对应的字节码偏移
99 // lower 级查找：找最大的 (instrIndex <= instrIdx) 的 bytecodeOffset
100 size_t lo = 0, hi = irToBytecodeOffset_.size();
101 while (lo < hi) {
102 size_t mid = (lo + hi) / 2;
103 if (irToBytecodeOffset_[mid].first <= instrIdx)
104 lo = mid + 1;
105 else
106 hi = mid;
107 }
108 // lo 指向第一个 first > instrIdx 的位置，回退一个是 <= instrIdx 的最大值
109 if (lo == 0)
110 return irToBytecodeOffset_[0].second;
111 return irToBytecodeOffset_[lo - 1].second;
112 };
113 for (const auto& range : ir.slotNameRanges) {
114 BytecodeChunk::SlotNameRange out;
115 out.slot = static_cast<uint8_t>(range.slot);
116 out.name = range.name;
117 out.startIp = translateInstr(range.startInstr);
118 // endInstr=0 表示函数级变量（参数/this/字段/内嵌函数），endIp 设为 0
119 // 使 resolveSlotName 不匹配此 range，fallback 到 localSlotNames。
120 if (range.endInstr == 0) {
121 out.endIp = 0;
122 } else {
123 // endInstr 是 CLOSE_UPVALUE 之后的指令序号，翻译为对应字节码偏移
124 out.endIp = translateInstr(range.endInstr);
125 }
126 chunk_->slotNameRanges.push_back(std::move(out));
127 }
128 }
129 return true;
130}

210bool BytecodeIRBackend::lowerInstruction(const IRInstruction& instr, const IRFunction& ir) {
211 // ============================================================
212 // 单条 IR 指令 → 栈式 VM 字节码 的 lowering 核心（按 IROp 类别分发）。
213 // ------------------------------------------------------------
214 // 设计要点（关键算法，逐步说明）：
215 // 1. 栈式 VM 是「操作数栈机」：每条 IROp 对应固定次数的压栈/弹栈。
216 // 例如 ADD 弹 2 压 1，LOAD_CONST 压 1，POP 弹 1，JUMP 不碰栈。
217 // 因为 IR 是 SSA 风格（每个 vreg 只被定义一次、按定义顺序被使用），
218 // 操作数在运行时的入栈顺序与 IR 指令顺序天然一致，lowering 无需
219 // 显式维护「vreg→栈偏移」映射也能保证栈平衡——这正是栈式后端比
220 // 寄存器式后端简单的原因（寄存器式需在 vregToReg 阶段物化每个 vreg）。
221 // 2. 发射布局：每个 case 直接 push_back opcode，再按该指令的定长格式
222 // push 操作数（2 字节小端 short 或 1 字节 slot）。变长指令（仅
223 // OP_CLOSURE）在编译期已展开为「nameIdx + upvalueCount + 2*描述符」。
224 // 3. vregStackDepth_ 辅助表：对每条产生 dest vreg 的 load 指令，记录其
225 // 发射位置 (code offset)。它实际不用于正确性判定（见要点 1），而是作为
226 // 调试/潜在栈深度分析的辅助信息维护；绝不可据其反向推算栈偏移。
227 // 4. 控制流两遍法：LABEL 不产生字节码，仅写入 labelToOffset_；跳转类
228 // （JUMP / JUMP_IF_FALSE）先 push opcode 并占位 2 字节操作数，把待回填
229 // 项加入 pendingJumps_。本函数（第一遍）结束后由 patchJumps()（第二遍）
230 // 查 labelToOffset_ 回填绝对/相对偏移（见 BUG-EXC-1：OP_TRY_BEGIN 用
231 // 相对偏移，其余用绝对偏移）。
232 // 5. 防御性边界检查：槽位/索引 >=256 或常量池索引越界时 Logger::Error
233 // 并返回 false，让 lower() 中止并向上传播错误，避免静态截断生成坏字节码。
234 // 6. WRITEBACK_*_VAR 的特殊处理：其 GLOBAL_SLOT 以 IMM_UINT 编码槽位号，
235 // 但栈式 VM 的 OP_WRITEBACK_*_VAR 把操作数当作「常量池中的变量名索引」，
236 // 故需经 slotToNameConstant() 转换为变量名常量索引（否则误读为未定义变量）。
237 //
238 // 重构：原 841 行单 switch 拆分为 8 个类别 helper（lowerConstOp/lowerVarOp/
239 // lowerArithOp/lowerCompareOp/lowerControlOp/lowerCallOp/lowerContainerOp/
240 // lowerMiscOp），主函数仅保留 switch 外壳做类别分发。语义零变更，所有审计
241 // 注释（BUG-IR-BOUND-1/AUDIT-STACKCLOSURE/BUG-EXC-1/AUDIT-P1.1/AUDIT-P2/
242 // BUG-INH-1/BUG-INH-IR-1/BUG-NEW/BUG-IR-SLOTNAME-1/BUG-IR-TRY-1 等）原样
243 // 保留至各 helper 对应 case。原 globalName lambda 改为 globalNameOf 静态方法。
244 // ============================================================
245 switch (instr.op) {
246 // ---- 常量加载 ----
247 case IROp::LOAD_CONST:
248 case IROp::LOAD_NULL:
249 case IROp::LOAD_TRUE:
250 case IROp::LOAD_FALSE:
251 return lowerConstOp(instr, ir);
252
253 // ---- 变量访问 ----
254 case IROp::LOAD_LOCAL:
255 case IROp::STORE_LOCAL:
256 case IROp::LOAD_GLOBAL:
257 case IROp::STORE_GLOBAL:
258 case IROp::DEFINE_GLOBAL:
259 case IROp::DELETE_VAR:
260 case IROp::LOAD_UPVALUE:
261 case IROp::STORE_UPVALUE:
262 case IROp::CLOSE_UPVALUE:
263 return lowerVarOp(instr, ir);
264
265 // ---- 算术 + 逻辑 ----
266 case IROp::ADD:
267 case IROp::SUB:
268 case IROp::MUL:
269 case IROp::DIV:
270 case IROp::MOD:
271 case IROp::NEGATE:
272 case IROp::NOT:
273 return lowerArithOp(instr, ir);
274
275 // ---- 比较 ----
276 case IROp::EQ:
277 case IROp::NEQ:
278 case IROp::LT:
279 case IROp::GT:
280 case IROp::LTE:
281 case IROp::GTE:
282 return lowerCompareOp(instr, ir);
283
284 // ---- 控制流 ----
285 case IROp::LABEL:
286 case IROp::JUMP:
287 case IROp::JUMP_IF_FALSE:
288 // R164 协程/生成器：YIELD 可中断执行（抛 VMYieldSignal），归入控制流组
289 case IROp::YIELD:
290 // 七特性 MVP 阶段 4：AWAIT 同步 drain（可能抛异常/错误），归入控制流组
291 case IROp::AWAIT:
292 return lowerControlOp(instr, ir);
293
294 // ---- 调用 + 闭包 ----
295 case IROp::CALL:
296 case IROp::CALL_EXPR:
297 case IROp::TAIL_CALL: // L18 eng-tailcall
298 case IROp::RETURN:
299 case IROp::RETURN_NULL:
300 case IROp::MAKE_CLOSURE:
301 return lowerCallOp(instr, ir);
302
303 // ---- 容器 + 成员访问 + 方法调用 ----
304 case IROp::BUILD_ARRAY:
305 case IROp::BUILD_DICT:
306 case IROp::BUILD_TUPLE:
307 case IROp::INDEX_GET:
308 case IROp::INDEX_SET:
309 case IROp::MEMBER_GET:
310 case IROp::MEMBER_SET:
311 case IROp::MEMBER_SET_LOCAL: // L7 fix
312 case IROp::SUPER_MEMBER_GET:
313 case IROp::METHOD_CALL:
314 case IROp::SUPER_CALL:
315 // R99 枚举与 ADT：enum variant 构造/检查/取字段（与容器指令同属 lowerContainerOp）
316 case IROp::BUILD_ENUM_VARIANT:
317 case IROp::ENUM_VARIANT_NAME:
318 case IROp::ENUM_VARIANT_FIELD:
319 // R134 模式匹配扩展：容器长度（与容器指令同属 lowerContainerOp，复用 OP_LEN）
320 case IROp::LEN:
321 return lowerContainerOp(instr, ir);
322
323 // ---- 其他（类/异常/写回/杂项）----
324 case IROp::DEFINE_CLASS:
325 case IROp::CLASS_NEW:
326 case IROp::INIT_FIELD:
327 case IROp::TRY_BEGIN:
328 case IROp::TRY_END:
329 case IROp::THROW:
330 case IROp::LOAD_EXCEPTION:
331 case IROp::PUSH_JUMP_TARGET:
332 case IROp::FINALLY_END:
333 case IROp::WRITEBACK_MEMBER_VAR:
334 case IROp::WRITEBACK_MEMBER_LOCAL:
335 case IROp::WRITEBACK_INDEX_VAR:
336 case IROp::WRITEBACK_INDEX_LOCAL:
337 case IROp::WRITEBACK_MEMBER_UPVALUE:
338 case IROp::WRITEBACK_INDEX_UPVALUE:
339 case IROp::PRINT:
340 case IROp::POP:
341 case IROp::DUP:
342 case IROp::LOAD_MUTATED:
343 case IROp::TYPE_CHECK:
344 case IROp::TYPE_TEST: // R134: 软类型测试，路由到 lowerMiscOp
345 return lowerMiscOp(instr, ir);
346
347 default:
348 Logger::Error("BytecodeIRBackend: 未知 IR 操作码 " + std::to_string(static_cast<int>(instr.op)), "IR");
349 return false;
350 }
351}

358bool BytecodeIRBackend::lowerConstOp(const IRInstruction& instr, const IRFunction& ir) {
359 // ---- 常量加载 ----
360 switch (instr.op) {
361 case IROp::LOAD_CONST: {
362 // 按 Value 类型分发 OP_INT/OP_FLOAT/OP_STRING
363 if (instr.operands.size() < 2)
364 return false;
365 uint32_t constIdx = instr.operands[1].index;
366 if (constIdx >= ir.constants.size())
367 return false;
368 const Value& v = ir.constants[constIdx];
369 vregStackDepth_[instr.operands[0].index] = static_cast<uint32_t>(chunk_->code.size());
370 switch (v.getType()) {
371 case ValueType::VAL_INT:
372 chunk_->code.push_back(static_cast<uint8_t>(OpCode::OP_INT));
373 break;
374 case ValueType::VAL_FLOAT:
375 chunk_->code.push_back(static_cast<uint8_t>(OpCode::OP_FLOAT));
376 break;
377 case ValueType::VAL_STRING:
378 chunk_->code.push_back(static_cast<uint8_t>(OpCode::OP_STRING));
379 break;
380 default:
381 chunk_->code.push_back(static_cast<uint8_t>(OpCode::OP_CONSTANT));
382 break;
383 }
384 emitUint16(chunk_->code, static_cast<uint16_t>(constIdx));
385 break;
386 }
387 case IROp::LOAD_NULL: {
388 if (instr.operands.empty())
389 return false;
390 vregStackDepth_[instr.operands[0].index] = static_cast<uint32_t>(chunk_->code.size());
391 chunk_->code.push_back(static_cast<uint8_t>(OpCode::OP_NULL));
392 break;
393 }
394 case IROp::LOAD_TRUE: {
395 if (instr.operands.empty())
396 return false;
397 vregStackDepth_[instr.operands[0].index] = static_cast<uint32_t>(chunk_->code.size());
398 chunk_->code.push_back(static_cast<uint8_t>(OpCode::OP_TRUE));
399 break;
400 }
401 case IROp::LOAD_FALSE: {
402 if (instr.operands.empty())
403 return false;
404 vregStackDepth_[instr.operands[0].index] = static_cast<uint32_t>(chunk_->code.size());
405 chunk_->code.push_back(static_cast<uint8_t>(OpCode::OP_FALSE));
406 break;
407 }
408 default:
409 return false; // 非本类别（不应触达，主 dispatcher 已按类别分发）
410 }
411 return true;
412}

414bool BytecodeIRBackend::lowerVarOp(const IRInstruction& instr, const IRFunction& ir) {
415 // ---- 变量访问 ----
416 switch (instr.op) {
417 case IROp::LOAD_LOCAL: {
418 if (instr.operands.size() < 2)
419 return false;
420 // slot 编码为 1 字节，超 255 静默截断会读写错误栈槽
421 if (instr.operands[1].index >= 256) {
422 Logger::Error(
423 "IR lowering: 局部变量槽位索引超出 255 上限 (slot=" + std::to_string(instr.operands[1].index) + ")",
424 "IR");
425 return false;
426 }
427 vregStackDepth_[instr.operands[0].index] = static_cast<uint32_t>(chunk_->code.size());
428 chunk_->code.push_back(static_cast<uint8_t>(OpCode::OP_GET_LOCAL));
429 chunk_->code.push_back(static_cast<uint8_t>(instr.operands[1].index)); // slot(1B)
430 break;
431 }
432 case IROp::STORE_LOCAL: {
433 if (instr.operands.size() < 2)
434 return false;
435 if (instr.operands[0].index >= 256) {
436 Logger::Error(
437 "IR lowering: 局部变量槽位索引超出 255 上限 (slot=" + std::to_string(instr.operands[0].index) + ")",
438 "IR");
439 return false;
440 }
441 chunk_->code.push_back(static_cast<uint8_t>(OpCode::OP_SET_LOCAL));
442 chunk_->code.push_back(static_cast<uint8_t>(instr.operands[0].index)); // slot(1B)
443 // 注意：OP_SET_LOCAL 用 peek(0) 不消费栈顶值（与 Compiler.cpp 一致）。
444 // IR 的 and/or 短路逻辑依赖此行为：STORE_LOCAL 将值写入 temp slot 后值仍在栈上，
445 // 后续 LOAD_LOCAL 从同一 slot 读取。需要消费源值的场景（如 super.method() 的
446 // LOAD_MUTATED + STORE_LOCAL）应在 IR 层面显式 emit IROp::POP。
447 break;
448 }
449 case IROp::LOAD_GLOBAL: {
450 // 限制3：按操作数 kind 分发
451 // IMM_UINT → OP_GET_GLOBAL slot(2B)（槽位版）
452 // GLOBAL_NAME → OP_GET_VAR nameIdx(2B)（名称版）
453 if (instr.operands.size() < 2)
454 return false;
455 vregStackDepth_[instr.operands[0].index] = static_cast<uint32_t>(chunk_->code.size());
456 if (instr.operands[1].kind == IROperandKind::IMM_UINT) {

622 // ---- 控制流 ----
623 switch (instr.op) {
624 case IROp::LABEL: {
625 // 记录标签对应的字节码偏移（不产生字节码）
626 if (!instr.operands.empty() && instr.operands[0].kind == IROperandKind::LABEL) {
627 labelToOffset_[instr.operands[0].index] = chunk_->code.size();
628 }
629 break;
630 }
631 case IROp::JUMP: {
632 if (instr.operands.empty())
633 return false;

642 chunk_->code.push_back(static_cast<uint8_t>(OpCode::OP_JUMP_IF_FALSE));

645 break;
646 }
647 // R164 协程/生成器：YIELD dest, src → OP_YIELD（1 字节无操作数，值在栈顶）
648 // 语义：src 的值已由前序 LOAD_* 指令压栈，OP_YIELD peek 栈顶：
649 // - 命中目标 yieldId：抛 VMYieldSignal 返回 yield 值

651 case IROp::YIELD: {
652 chunk_->code.push_back(static_cast<uint8_t>(OpCode::OP_YIELD));
653 break;
654 }
655 // 七特性 MVP 阶段 4：AWAIT dest, src → OP_AWAIT（1 字节无操作数，值在栈顶）
656 // 语义：src 的值已由前序指令压栈，OP_AWAIT pop 后同步 drain 协程并 push 最终值
657 // （dest 继承 src 的栈位置）。
658 case IROp::AWAIT: {
659 chunk_->code.push_back(static_cast<uint8_t>(OpCode::OP_AWAIT));
660 break;
661 }
662 default:
663 return false; // 非本类别
664 }
665 return true;

669 // ---- 调用 + 闭包 ----
670 switch (instr.op) {
671 case IROp::CALL: {
672 // CALL dest, name_idx, arg_count, args... → OP_CALL nameIdx argCount
673 if (instr.operands.size() < 3)
674 return false;
675 // BUG-IR-BOUND-1 fix: 检查 argCount 8 位上限，对齐 RegisterBytecodeBackend L410-414
676 if (instr.operands[2].index > 255) {
677 Logger::Error("BytecodeIRBackend: CALL argCount 超出 255 上限 (" + std::to_string(instr.operands[2].index) +
678 ")",
679 "IR");
680 return false;
681 }
682 uint16_t nameConstIdx = addStringConstant(globalNameOf(ir, instr.operands[1].index), ir);
683 vregStackDepth_[instr.operands[0].index] = static_cast<uint32_t>(chunk_->code.size());
684 chunk_->code.push_back(static_cast<uint8_t>(OpCode::OP_CALL));
685 emitUint16(chunk_->code, nameConstIdx);
686 chunk_->code.push_back(static_cast<uint8_t>(instr.operands[2].index & 0xFF)); // argCount
687 break;
688 }
689 case IROp::CALL_EXPR: {
690 if (instr.operands.size() < 3)
691 return false;
692 // BUG-IR-BOUND-1 fix: 检查 argCount 8 位上限，对齐 RegisterBytecodeBackend L435-439
693 if (instr.operands[2].index > 255) {
694 Logger::Error("BytecodeIRBackend: CALL_EXPR argCount 超出 255 上限 (" +
695 std::to_string(instr.operands[2].index) + ")",
696 "IR");
697 return false;
698 }
699 vregStackDepth_[instr.operands[0].index] = static_cast<uint32_t>(chunk_->code.size());
700 chunk_->code.push_back(static_cast<uint8_t>(OpCode::OP_CALL_EXPR));
701 chunk_->code.push_back(static_cast<uint8_t>(instr.operands[2].index & 0xFF)); // argCount
702 break;
703 }
704 case IROp::TAIL_CALL: {
705 // L18 eng-tailcall: 布局同 CALL，lower 为 OP_TAIL_CALL（后随的 IROp::RETURN
706 // 由 AstIRBuilder 保证紧跟，lower 为 OP_RETURN 供降级路径使用）。
707 if (instr.operands.size() < 3)
708 return false;
709 if (instr.operands[2].index > 255) {
710 Logger::Error("BytecodeIRBackend: TAIL_CALL argCount 超出 255 上限 (" +
711 std::to_string(instr.operands[2].index) + ")",
712 "IR");
713 return false;
714 }
715 uint16_t nameConstIdx = addStringConstant(globalNameOf(ir, instr.operands[1].index), ir);
716 vregStackDepth_[instr.operands[0].index] = static_cast<uint32_t>(chunk_->code.size());
717 chunk_->code.push_back(static_cast<uint8_t>(OpCode::OP_TAIL_CALL));
718 emitUint16(chunk_->code, nameConstIdx);
719 chunk_->code.push_back(static_cast<uint8_t>(instr.operands[2].index & 0xFF)); // argCount
720 break;
721 }
722 case IROp::RETURN: {
723 chunk_->code.push_back(static_cast<uint8_t>(OpCode::OP_RETURN));
724 break;
725 }
726 case IROp::RETURN_NULL: {
727 // RETURN_NULL 前先 emit OP_NULL（栈顶作为返回值）
728 chunk_->code.push_back(static_cast<uint8_t>(OpCode::OP_NULL));
729 chunk_->code.push_back(static_cast<uint8_t>(OpCode::OP_RETURN));
730 break;
731 }
732 case IROp::MAKE_CLOSURE: {
733 // ---- 闭包 ----
734 // MAKE_CLOSURE dest, name_idx, uv_count, [isLocal, idx]×uv_count
735 if (instr.operands.size() < 3)
736 return false;
737 // AUDIT-STACKCLOSURE fix: 对齐 RegisterBytecodeBackend 的 P2-3 fix，
738 // uvCount/isLocal/idx 编码为 1 字节，原 & 0xFF 静默截断会生成错误 upvalue 描述符。
739 if (instr.operands[2].index >= 256) {
740 Logger::Error("BytecodeIRBackend: MAKE_CLOSURE uvCount 超出 255 上限 (" +
741 std::to_string(instr.operands[2].index) + ")",
742 "IR");
743 return false;
744 }
745 uint16_t nameConstIdx = addStringConstant(globalNameOf(ir, instr.operands[1].index), ir);
746 vregStackDepth_[instr.operands[0].index] = static_cast<uint32_t>(chunk_->code.size());
747 chunk_->code.push_back(static_cast<uint8_t>(OpCode::OP_CLOSURE));
748 emitUint16(chunk_->code, nameConstIdx);
749 uint32_t uvCount = instr.operands[2].index;
750 chunk_->code.push_back(static_cast<uint8_t>(uvCount));
751 for (uint32_t i = 0; i < uvCount; ++i) {
752 size_t base = 3 + i * 2;
753 if (base + 1 >= instr.operands.size())
754 return false;
755 if (instr.operands[base].index >= 256) {
756 Logger::Error("BytecodeIRBackend: MAKE_CLOSURE isLocal 超出 255 上限 (" +
757 std::to_string(instr.operands[base].index) + ")",
758 "IR");
759 return false;
760 }
761 if (instr.operands[base + 1].index >= 256) {
762 Logger::Error("BytecodeIRBackend: MAKE_CLOSURE upvalue idx 超出 255 上限 (" +
763 std::to_string(instr.operands[base + 1].index) + ")",
764 "IR");
765 return false;
766 }
767 chunk_->code.push_back(static_cast<uint8_t>(instr.operands[base].index)); // isLocal
768 chunk_->code.push_back(static_cast<uint8_t>(instr.operands[base + 1].index)); // idx
769 }
770 break;
771 }
772 default:
773 return false; // 非本类别
774 }
775 return true;

1118 }

1120 chunk_->code.push_back(static_cast<uint8_t>(exprLocalSlot));
1121 } else if (defaultConstIdx == UINT32_MAX) {
1122 chunk_->code.push_back(static_cast<uint8_t>(OpCode::OP_NULL));
1123 } else {

1148 }

1166 }

1368bool BytecodeIRBackend::lowerMiscPureOps(const IRInstruction& instr, const IRFunction& /*ir*/) {
1369 // lowerMiscOp 子阶段：杂项 IROp（PRINT/POP/DUP/LOAD_MUTATED/TYPE_CHECK）
1370 switch (instr.op) {
1371 case IROp::PRINT: {
1372 chunk_->code.push_back(static_cast<uint8_t>(OpCode::OP_PRINT));
1373 break;
1374 }
1375 case IROp::POP: {
1376 chunk_->code.push_back(static_cast<uint8_t>(OpCode::OP_POP));
1377 break;
1378 }
1379 case IROp::DUP: {
1380 if (!instr.operands.empty()) {
1381 vregStackDepth_[instr.operands[0].index] = static_cast<uint32_t>(chunk_->code.size());
1382 }
1383 chunk_->code.push_back(static_cast<uint8_t>(OpCode::OP_DUP));
1384 break;
1385 }
1386 case IROp::LOAD_MUTATED: {
1387 // MEDIUM-1/2 fix: 读取 lastMutatedReceiver_ 到栈顶（不清除）
1388 if (!instr.operands.empty()) {
1389 vregStackDepth_[instr.operands[0].index] = static_cast<uint32_t>(chunk_->code.size());
1390 }
1391 chunk_->code.push_back(static_cast<uint8_t>(OpCode::OP_LOAD_MUTATED));
1392 break;
1393 }
1394 // 2026-06-29: 类型注解运行时检查
1395 // operands: [src_vreg, type_const_idx] src_vreg 已在栈顶，peek 不弹栈
1396 case IROp::TYPE_CHECK: {
1397 if (instr.operands.size() < 2) {
1398 Logger::Error("BytecodeIRBackend: TYPE_CHECK 操作数不足", "IR");
1399 return false;
1400 }
1401 // type_const_idx 已在常量池复制阶段（lower() 的 for 循环）同步到 BytecodeChunk
1402 uint16_t typeIdx = static_cast<uint16_t>(instr.operands[1].index);
1403 // AUDIT-P2 fix: 与 DUP/LOAD_MUTATED 保持一致，更新 src_vreg 的栈深度映射。
1404 // 虽然 OP_TYPE_CHECK 是 peek 不 push，但记录 src_vreg 在此处"仍然有效"的栈深度，
1405 // 避免后续指令通过 vregStackDepth_ 查询时拿到过期记录点。
1406 vregStackDepth_[instr.operands[0].index] = static_cast<uint32_t>(chunk_->code.size());
1407 chunk_->code.push_back(static_cast<uint8_t>(OpCode::OP_TYPE_CHECK));
1408 chunk_->code.push_back(static_cast<uint8_t>(typeIdx & 0xFF));
1409 chunk_->code.push_back(static_cast<uint8_t>((typeIdx >> 8) & 0xFF));
1410 break;
1411 }
1412 case IROp::TYPE_TEST: {
1413 // R134: 软类型测试 lowering（与 OP_TYPE_TEST 对应）
1414 // IR: [dest_vreg, src_vreg, type_const_idx] 字节码：src 已在栈上，emit OP_TYPE_TEST pop src push bool
1415 if (instr.operands.size() < 3) {
1416 Logger::Error("BytecodeIRBackend: TYPE_TEST 操作数不足", "IR");
1417 return false;
1418 }
1419 uint16_t typeIdx = static_cast<uint16_t>(instr.operands[2].index);
1420 vregStackDepth_[instr.operands[0].index] = static_cast<uint32_t>(chunk_->code.size());
1421 chunk_->code.push_back(static_cast<uint8_t>(OpCode::OP_TYPE_TEST));
1422 chunk_->code.push_back(static_cast<uint8_t>(typeIdx & 0xFF));
1423 chunk_->code.push_back(static_cast<uint8_t>((typeIdx >> 8) & 0xFF));
1424 break;
1425 }
1426 default:
1427 return false; // 非本类别
1428 }
1429 return true;
1430}

1432bool BytecodeIRBackend::patchJumps() {
1433 // 遍历所有待回填跳转，查 labelToOffset 写回目标偏移
1434 for (const auto& pj : pendingJumps_) {
1435 auto it = labelToOffset_.find(pj.targetLabel);
1436 if (it == labelToOffset_.end()) {
1437 Logger::Error("BytecodeIRBackend: 未找到标签 " + std::to_string(pj.targetLabel), "IR");
1438 return false;
1439 }
1440 // BUG-EXC-1 fix: OP_JUMP/OP_JUMP_IF_FALSE 用绝对偏移（ip = target），
1441 // 但 OP_TRY_BEGIN 用相对偏移（catchIp = ip + 3 + catchOffset）。
1442 // pj.codeOffset 指向操作数首字节（opcode 在 codeOffset-1），
1443 // StackVM 执行时 ip = codeOffset-1，故相对偏移 = target - (ip+3) = target - (codeOffset+2)。
1444 size_t targetOffset = it->second;
1445 uint16_t target;
1446 if (pj.isTryBegin) {
1447 if (targetOffset < pj.codeOffset + 2) {
1448 Logger::Error(
1449 "BytecodeIRBackend: TRY_BEGIN catch 目标在 TRY_BEGIN 之前 (catch=" + std::to_string(targetOffset) +
1450 ", tryBegin=" + std::to_string(pj.codeOffset - 1) + ")",
1451 "IR");
1452 return false;
1453 }
1454 // BUG-IR-PATCH-1 fix: 检查 catch 相对偏移 64KB 上限，对齐 Compiler.cpp visitTryStmt 的 cleanupThrowOffset >
1455 // 65535 检查。 原实现 static_cast<uint16_t> 静默截断，导致 VM
1456 // 跳转到错误位置执行乱码字节码（三后端不一致）。
1457 size_t relOff = targetOffset - (pj.codeOffset + 2);
1458 if (relOff > 65535) {
1459 Logger::Error(
1460 "BytecodeIRBackend: TRY_BEGIN catch 相对偏移超过 64KB 限制 (relOff=" + std::to_string(relOff) + ")",
1461 "IR");
1462 return false;
1463 }
1464 target = static_cast<uint16_t>(relOff);
1465 } else {
1466 // 16-bit 编码限制：字节码体积超过 64KB 时跳转目标会截断，需显式检查
1467 // （对齐 Compiler::safeCodeOffset() 的 65535 上限保护）
1468 if (targetOffset > 65535) {
1469 Logger::Error("BytecodeIRBackend: 跳转目标偏移超过 64KB 限制 (offset=" + std::to_string(targetOffset) +
1470 ")",
1471 "IR");
1472 return false;
1473 }
1474 target = static_cast<uint16_t>(targetOffset);
1475 }
1476 if (pj.codeOffset + 1 >= chunk_->code.size()) {
1477 Logger::Error("BytecodeIRBackend: 跳转操作数越界", "IR");
1478 return false;
1479 }
1480 chunk_->code[pj.codeOffset] = static_cast<uint8_t>(target & 0xFF);
1481 chunk_->code[pj.codeOffset + 1] = static_cast<uint8_t>((target >> 8) & 0xFF);
1482 }
1483 return true;
1484}
