176 // B1 fix: VM RUN 模式活跃时拒绝 runCompiler，避免替换 CompileResult 致 frame.chunk 悬垂 UAF。
177 // 调用方（onShowBytecode/onShowIR）在 VM 运行时应先 vmStop()。

205/// 准备运行：失效管线缓存、运行前端管线产出 AST，供 WorkerManager 启动解释器。
206bool IdeController::prepareRun(bool isDebug, const std::string& source, const std::string& filePath) {
207 // E2 fix: 各静默 return false 路径补 emit genericError，避免 UI 已 clearAll 后
208 // 用户看不到任何反馈（onRun/onDebug 调用 prepareRun 前已 clearAll 输出面板）。
209 if (workerMgr_.isRunning()) {
210 emit genericError("已有运行在进行，请先停止当前运行");
211 return false;
212 }
213 // BUG-IDE-02 fix: 检查 VM 是否正在运行。VM RUN 模式异步执行期间 isRunning() 仅反映
214 // workerMgr 状态，VM 步进/RUN 仍可能活跃。若不同步停止，runCompiler/编译结果同步
215 // 会让 frame.chunk 悬垂，且 workerMgr 启动新 worker 时 VM 仍在动旧 CompileResult。
216 if (vmStepper_.isRunning()) {
217 emit genericError("VM RUN 模式正在执行，请先停止 VM 再启动新运行");
218 return false;
219 }
220
221 // VM-IMPORT: 保存文件路径，并为 Compiler 设置模块加载器（VM 编译路径需要）
222 currentFilePath_ = filePath;
223 setupCompilerModuleLoader(filePath);
224
225 // C9 fix: 使用统一前端管线（Lexer + Parser）
226 auto pipelineResult = pipeline_.runFrontendPipeline(source);
227 if (pipelineResult.status != PipelineRunner::PipelineStatus::OK) {
228 if (!pipelineResult.errorMessage.empty()) {
229 emit genericError(QString("%1: %2")
230 .arg(pipelineResult.status == PipelineRunner::PipelineStatus::LexerFailed
231 ? "词法分析异常"
232 : "解析异常")
233 .arg(QString::fromStdString(pipelineResult.errorMessage)));
234 }
235 return false;
236 }
237
238 if (!pipeline_.astRoot()) {
239 emit genericError("内部错误：前端管线返回成功但 AST 为空");
240 return false;
241 }
242
243 // BUG-IDE-PREP-1 fix: prepareRun 必须重新编译当前源码，否则同步到 VmStepper 的
244 // 是上一次编译结果（pipeline_.lastCompileResult() / getLastRegisterResult()），
245 // 与本次源码不对应——用户修改源码后不点"编译"直接点"运行/调试"会执行旧字节码。
246 // 注意：runCompiler 内部已检查 vmStepper_.isRunning()（上方 BUG-IDE-02 已先拦截），
247 // 且编译错误通过 diagnosticsReady 信号报告，此处不再 emit 重复错误。
248 if (!runCompiler()) {
249 return false;
250 }
251
252 // 委托 WorkerManager 设置模块加载器、调试模式、创建 worker 线程
253 if (!workerMgr_.prepareRun(isDebug, pipeline_.astRoot(), filePath)) {
254 // WorkerManager 内部已 emit 错误或抛异常（catch 内回滚），此处不重复
255 return false;
256 }
257
258 // 同步编译结果到 VmStepper（供 VM 模式调试使用）
259 // A1 fix: 根据 useRegisterVM_ 选择同步栈式或寄存器式结果
260 // BUG-DBG-2 fix: 先重置 VM 状态，防止旧 isVmInitialized_=true 时 frame.chunk 悬垂
261 vmStepper_.reset();
262 if (pipeline_.compiler().getUseRegisterVM()) {
263 vmStepper_.setRegisterCompileResult(pipeline_.compiler().getLastRegisterResult());
264 } else {
265 vmStepper_.setCompileResult(pipeline_.lastCompileResult());
266 }
267
268 // R121: 新运行启动时清除上一轮选中帧（不应继承到新调试会话）
269 clearSelectedFrame();
270
271 // OPT-1: 新运行已就绪，通知订阅面板刷新（CallStack/VariableInspector 等显示运行状态）。
272 notifyVmStateChanged();
273 return true;
274}

277// VM-IMPORT: Compiler 模块加载器设置
278// ------------------------------------------------------------
279// 对齐 WorkerManager::prepareRun 中为 Interpreter 设置的模块加载器逻辑，
280// 使 VM 编译路径（栈式 / IR / 寄存器式）也能处理 import 语句。
281// 模块路径解析策略：
282// 1. 以当前文件所在目录为基准查找 <modulePath>.mini
283// 2. 若 baseDir 为空，则按字面路径查找
284// 3. 自动补充 .mini 后缀
285// 找不到时返回空字符串，由 Compiler::visitImportStmt 报错
286// ============================================================
287void IdeController::setupCompilerModuleLoader(const std::string& filePath) {
288 QString baseDir;
289 if (!filePath.empty()) {
290 QFileInfo fi(QString::fromStdString(filePath));
291 baseDir = fi.absolutePath();
292 }
293 // 设置当前文件路径（Compiler 用于相对路径解析）
294 pipeline_.compiler().setCurrentFilePath(filePath);
295 // 捕获 baseDir 副本到 lambda（独立于 IdeController 生命周期，
296 // Compiler 持有 std::function 直到下次设置或析构）
297 pipeline_.compiler().setModuleLoader([baseDir](const std::string& modulePath) -> std::string {
298 // P2-11: 内建标准库模块拦截（std/math, std/string, std/list）
299 if (BuiltinModuleRegistry::isBuiltinModule(modulePath)) {
300 return BuiltinModuleRegistry::getSource(modulePath);
301 }
302 QString qPath = QString::fromStdString(modulePath);
303 if (!qPath.endsWith(".mini", Qt::CaseInsensitive)) {
304 qPath += ".mini";
305 }
306 QStringList candidates;
307 if (baseDir.isEmpty()) {
308 candidates << qPath;
309 } else {
310 candidates << QDir(baseDir).filePath(qPath) << qPath;
311 }
312 for (const QString& candidate : candidates) {
313 QFile file(candidate);
314 if (file.exists() && file.open(QIODevice::ReadOnly | QIODevice::Text)) {
315 // P3.4 fix: 区分"文件不存在"与"空文件"。文件存在但内容为空时
316 // 返回 "\n"（合法空白源码，Lexer 产出 [TK_EOF]，Parser 产出空 Block），
317 // 而非 ""（被三后端视为加载失败）。这样 import 一个 0 字节 .mini 文件
318 // 会得到一个合法的空模块（无导出），而非 ModuleNotFound 错误。
319 std::string content = QString::fromUtf8(file.readAll()).toStdString();
320 return content.empty() ? "\n" : content;
321 }
322 }
323 return "";
324 });
325}

348// P0-3 fix (F11): 实现 getReplScopeVariableNames，供 GUI ErrorHintEngine 拼写建议使用
349// R53-1 fix: 对齐 DebugCoordinator 的 AUDIT-P1 修复模式，使用 currentEnvironmentShared()
350// 替代裸 currentEnvironment()。该方法可能在 worker 线程 emit runtimeError 后被主线程
351// QueuedConnection 处理器调用，裸指针遍历 parent 链违反 Interpreter.h 跨线程契约。
352// shared_ptr 副本通过赋值延长 parent 链每个节点生命周期，防止 worker 线程修改或析构
353// Environment 时主线程解引用悬挂指针。
354std::vector<std::string> IdeController::getReplScopeVariableNames() const {
355 std::vector<std::string> names;
356 auto env = interpreter_->currentEnvironmentShared();
357 if (!env)
358 return names;
359 // 手动遍历 parent 链（用 shared_ptr 赋值延长每个节点生命周期），
360 // 顺序与 allVariables() 一致：父作用域在前，子作用域追加末尾。
361 std::unordered_set<std::string> seen;
362 while (env) {
363 // BUG-64 fix (P2 线程安全): 原实现以 const 引用绑定 env->localVariables() 返回的
364 // SmallMap，遍历期间若 worker 线程修改/析构该 Environment（QueuedConnection 跨线程
365 // 调用本方法时与 worker 并发），引用会悬挂或读到撕裂状态。改为值拷贝快照
366 // (snapshotLocalVariables 返回 unordered_map by value)，遍历完全脱离原容器，
367 // 与 R53-1 shared_ptr 父链延长策略对齐，保证主线程遍历期间数据有效。
368 auto locals = env->snapshotLocalVariables();
369 for (const auto& kv : locals) {
370 if (seen.insert(kv.first).second) {
371 names.push_back(kv.first);
372 }
373 }
374 env = env->parent; // shared_ptr 赋值，延长 parent 生命周期
375 }
376 return names;
377}

411// ============================================================
412// R117 Watch 表达式求值（调试器拓展）
413// ------------------------------------------------------------
414// 在当前调试暂停上下文中求值用户输入的表达式，返回完整 Value 信息供
415// WatchPanel 显示。实现策略与 setConditionEvaluator 一致：
416// 1. 自动补充分号（用户输入通常不带分号）
417// 2. LRU 缓存 AST（32 条上限），避免循环内重复 Lexer+Parser
418// 3. 创建临时 Interpreter + 沙箱 Environment，注入 globals/locals
419// 4. 若存在 this 且为实例，绑定 boundInstance_ 使裸字段名可访问
420// 5. evaluateCondition 返回 Value（沙箱化，状态自动恢复），
421// WatchPanel 用完整 Value 而非 isTruthy() 展示类型+值
422//
423// 与条件断点求值的区别：
424// - 条件断点求值仅返回 bool（isTruthy），Watch 返回完整 Value
425// - 条件断点求值仅覆盖 VM 路径（VmStepper 回调），Watch 同时覆盖
426// VM 路径与 Interpreter 调试暂停路径（直接调用，无 worker 线程冲突）
427// - 条件断点求值的 LRU 缓存在 lambda 闭包内（VmStepper 持有），
428// Watch 的 LRU 缓存在 IdeController 成员中（多次调用共享）
429//
430// 线程安全：本方法在 GUI 线程调用，访问的 vmStepper_/debugCoord_ 状态
431// 在暂停期间稳定（VM 暂停 / Interpreter 在 worker 线程暂停中），
432// 与 setConditionEvaluator 的线程模型一致。
433// ============================================================

448IdeController::WatchResult IdeController::evaluateWatchExpression(const std::string& expr) {
449 WatchResult result;
450 if (expr.empty()) {
451 result.error = "表达式为空";
452 return result;
453 }
454
455 // 判定当前暂停模式，收集要注入的变量
456 bool vmMode = vmStepper_.isRunning() || vmStepper_.isInitialized();
457 bool interpPaused = debugCoord_.isPaused();
458 if (!vmMode && !interpPaused) {
459 result.error = "未在调试暂停状态（请先调试运行并命中断点）";
460 return result;
461 }
462
463 try {
464 // ---- 1. 自动补充分号 + LRU AST 缓存 ----
465 std::string normExpr = expr;
466 if (!normExpr.empty() && normExpr.back() != ';') {
467 normExpr += ';';
468 }
469 std::shared_ptr<Block> ast;
470 auto lookupIt = watchAstCacheLookup_.find(normExpr);
471 if (lookupIt != watchAstCacheLookup_.end()) {
472 // 命中：move_to_front（最近使用）
473 watchAstCacheList_.splice(watchAstCacheList_.begin(), watchAstCacheList_, lookupIt->second);
474 ast = lookupIt->second->second;
475 } else {
476 Lexer lexer;
477 auto tokens = lexer.scan(normExpr);
478 Parser parser;
479 auto parsed = parser.parse(tokens);
480 if (!parsed || parsed->statements.empty() || parser.getDiagnostics().hasErrors()) {
481 result.error = "表达式解析失败";
482 return result;
483 }
484 ast = std::shared_ptr<Block>(std::move(parsed));
485 // LRU 淘汰：超上限时移除最久未使用（链表尾部）
486 if (watchAstCacheList_.size() >= WATCH_AST_CACHE_MAX) {
487 watchAstCacheLookup_.erase(watchAstCacheList_.back().first);
488 watchAstCacheList_.pop_back();
489 }
490 watchAstCacheList_.emplace_front(normExpr, ast);
491 watchAstCacheLookup_[normExpr] = watchAstCacheList_.begin();
492 }
493
494 // ---- 2. 创建临时 Interpreter + 注入变量 ----
495 Interpreter tempInterp;
496 auto env = std::make_shared<Environment>();
497 if (vmMode) {
498 // VM 模式：注入 VM globals + 所选帧 locals（局部变量遮蔽同名全局）
499 for (const auto& kv : vmStepper_.getGlobals()) {
500 env->define(kv.first, kv.second);
501 }
502 // R121: 用 getSelectedFrameLocals() 替代 getCurrentFrameLocals()，
503 // 支持用户切换到调用栈任意帧查看该帧上下文。selectedFrame_ == -1 时
504 // 返回栈顶 locals（与原 getCurrentFrameLocals 行为一致，向后兼容）。
505 for (const auto& kv : getSelectedFrameLocals()) {
506 env->define(kv.first, kv.second);
507 }
508 } else {
509 // Interpreter 调试暂停模式：
510 // - 注入全局 + 栈顶 upvalue/locals 快照（保持 R117 行为）
511 // - 若 selectedFrame_ >= 0，用所选帧 locals 覆盖（R121 帧切换）
512 // env->define 同名变量后写覆盖前写，实现"切换到帧 N 时看到帧 N 的 locals"
513 for (const auto& snap : debugCoord_.getDebugVariableSnapshot()) {
514 env->define(snap.name, snap.value);
515 }
516 if (selectedFrame_ >= 0) {
517 for (const auto& kv : getSelectedFrameLocals()) {
518 env->define(kv.first, kv.second);
519 }
520 }
521 }
522 // this 绑定：若存在 this 变量且为实例，绑定 boundInstance_ 使裸字段名
523 // 可回退解析实例字段（与方法体执行语义一致）。
524 auto& vars = env->localVariables();
525 auto thisIt = vars.find("this");
526 if (thisIt != vars.end() && thisIt->second.isInstance()) {
527 env->bindInstance(const_cast<Value*>(&thisIt->second));
528 }
529 tempInterp.setGlobalEnvironment(env);
530
531 // ---- 3. 沙箱求值，返回完整 Value ----
532 Value v = tempInterp.evaluateCondition(ast->statements[0].get());
533 result.ok = true;
534 result.value = v;
535 result.typeName = v.typeName();
536 result.valueRepr = v.toString();
537 return result;
538 } catch (const std::exception& e) {
539 result.error = std::string("求值异常: ") + e.what();
540 LOG_WARNING("Watch 表达式求值异常: " + result.error + "（表达式: " + expr + "）", "IdeController");
541 return result;
542 } catch (...) {
543 result.error = "求值异常: 未知错误";
544 return result;
545 }
546}

548// ---- R161 调试器 REPL 求值 ----
549// 与 evaluateWatchExpression 的核心区别：
550// 1. 允许多条语句（Block 而非单表达式），用 executeRepl 执行整个 Block
551// 2. 允许 var/print/赋值语句，临时 Interpreter 独立实例天然隔离副作用
552// 3. 捕获 print 输出到 output 参数（供 ReplPanel 显示）
553// 4. 不做沙箱恢复（evaluateCondition 的 SandboxGuard 会撤销赋值，调试器 REPL 不需要）
554IdeController::WatchResult IdeController::evaluateDebuggerRepl(const std::string& source, std::string* output) {
555 WatchResult result;
556 if (source.empty()) {
557 result.error = "输入为空";
558 return result;
559 }
560
561 // 判定当前暂停模式，收集要注入的变量
562 bool vmMode = vmStepper_.isRunning() || vmStepper_.isInitialized();
563 bool interpPaused = debugCoord_.isPaused();
564 if (!vmMode && !interpPaused) {
565 result.error = "未在调试暂停状态（请先调试运行并命中断点）";
566 return result;
567 }
568
569 try {
570 // ---- 1. 自动补充分号 + LRU AST 缓存 ----
571 std::string normSource = source;
572 if (!normSource.empty() && normSource.back() != ';') {
573 normSource += ';';
574 }
575 std::shared_ptr<Block> ast;
576 auto lookupIt = watchAstCacheLookup_.find(normSource);
577 if (lookupIt != watchAstCacheLookup_.end()) {
578 watchAstCacheList_.splice(watchAstCacheList_.begin(), watchAstCacheList_, lookupIt->second);
579 ast = lookupIt->second->second;
580 } else {
581 Lexer lexer;
582 auto tokens = lexer.scan(normSource);
583 Parser parser;
584 auto parsed = parser.parse(tokens);
585 if (!parsed || parsed->statements.empty() || parser.getDiagnostics().hasErrors()) {
586 // 解析失败时尝试提取诊断信息
587 const auto& diags = parser.getDiagnostics();
588 if (!diags.all().empty()) {
589 result.error = "语法错误: " + diags.all()[0].message;
590 } else {
591 result.error = "语法解析失败";
592 }
593 return result;
594 }
595 ast = std::shared_ptr<Block>(std::move(parsed));
596 if (watchAstCacheList_.size() >= WATCH_AST_CACHE_MAX) {
597 watchAstCacheLookup_.erase(watchAstCacheList_.back().first);
598 watchAstCacheList_.pop_back();
599 }
600 watchAstCacheList_.emplace_front(normSource, ast);
601 watchAstCacheLookup_[normSource] = watchAstCacheList_.begin();
602 }
603
604 // ---- 2. 创建临时 Interpreter + 注入变量（与 evaluateWatchExpression 一致）----
605 Interpreter tempInterp;
606 auto env = std::make_shared<Environment>();
607 if (vmMode) {
608 for (const auto& kv : vmStepper_.getGlobals()) {
609 env->define(kv.first, kv.second);
610 }
611 for (const auto& kv : getSelectedFrameLocals()) {
612 env->define(kv.first, kv.second);
613 }
614 } else {
615 for (const auto& snap : debugCoord_.getDebugVariableSnapshot()) {
616 env->define(snap.name, snap.value);
617 }
618 if (selectedFrame_ >= 0) {
619 for (const auto& kv : getSelectedFrameLocals()) {
620 env->define(kv.first, kv.second);
621 }
622 }
623 }
624 // this 绑定
625 auto& vars = env->localVariables();
626 auto thisIt = vars.find("this");
627 if (thisIt != vars.end() && thisIt->second.isInstance()) {
628 env->bindInstance(const_cast<Value*>(&thisIt->second));
629 }
630 tempInterp.setGlobalEnvironment(env);
631
632 // ---- R161 阶段 2：注入函数/类/枚举注册表 ----
633 // 从主 Interpreter 复制注册表，使调试器 REPL 中可调用用户定义的
634 // 函数/类/枚举。shared_ptr<FunDecl> 共享所有权，AST 在主 Interpreter
635 // 析构前有效（主 Interpreter 由 IdeController 持有，生命期长于 tempInterp）。
636 //
637 // 线程安全：Interpreter 调试暂停期 worker 阻塞在 pauseCV_，
638 // VM 暂停期主 Interpreter 空闲，复制注册表安全。
639 //
640 // 边界：主 Interpreter 未运行过（如直接用 VM 调试从未 Run Interpreter）
641 // 时 hasRegistries()=false，跳过注入，REPL 仅能访问变量不能调用函数。
642 if (interpreter_ && interpreter_->hasRegistries()) {
643 tempInterp.injectRegistriesFrom(*interpreter_);
644 }
645
646 // ---- 3. 捕获 print 输出 ----
647 std::string capturedOutput;
648 if (output) {
649 tempInterp.setOutputCallback([&capturedOutput](const std::string& s) {
650 capturedOutput += s;
651 capturedOutput += '\n';
652 });
653 }
654
655 // ---- 4. 用 executeRepl 执行整个 Block（多条语句，不做沙箱恢复）----
656 // 临时 Interpreter 是独立实例，赋值/var 声明的副作用仅影响 tempInterp 的 env，
657 // 不写回主程序的 VM/Interpreter 状态（阶段1只读语义）。
658 Value v = tempInterp.executeRepl(*ast);
659 result.ok = true;
660 result.value = v;
661 result.typeName = v.typeName();
662 result.valueRepr = v.toString();
663 if (output) {
664 *output = capturedOutput;
665 }
666 return result;
667 } catch (const RuntimeError& e) {
668 result.error = std::string("运行时错误: ") + e.what();
669 return result;
670 } catch (const std::exception& e) {
671 result.error = std::string("求值异常: ") + e.what();
672 LOG_WARNING("调试器 REPL 求值异常: " + result.error + "（源码: " + source + "）", "IdeController");
673 return result;
674 } catch (...) {
675 result.error = "求值异常: 未知错误";
676 return result;
677 }
678}

680/// 设置 REPL 模块加载回调：将相对 import 路径解析到当前活动文件所在目录。
681void IdeController::setupReplModuleCallbacks() {
682 // 若 Interpreter 已有 moduleLoader_（先 Run 过），不覆盖，保持 Run 时建立的
683 // baseDir 与模块缓存基准一致（避免 Run 后 REPL 用不同的 baseDir）。
684 if (interpreter_->hasModuleLoader()) {
685 return;
686 }
687
688 // 基于 currentFilePath_ 解析相对模块路径（对齐 WorkerManager::prepareRun 逻辑）
689 const std::string& filePath = currentFilePath_;
690 QString baseDir;
691 if (!filePath.empty()) {
692 QFileInfo fi(QString::fromStdString(filePath));
693 baseDir = fi.absolutePath();
694 }
695 interpreter_->setCurrentFilePath(filePath);
696
697 // 模块路径解析辅助函数，供 loader 和 mtime checker 共用
698 auto resolveModulePath = [baseDir](const std::string& modulePath) -> QString {
699 QString qPath = QString::fromStdString(modulePath);
700 if (!qPath.endsWith(".mini", Qt::CaseInsensitive)) {
701 qPath += ".mini";
702 }
703 QStringList candidates;
704 if (baseDir.isEmpty()) {
705 candidates << qPath;
706 } else {
707 candidates << QDir(baseDir).filePath(qPath) << qPath;
708 }
709 for (const QString& candidate : candidates) {
710 if (QFile::exists(candidate)) {
711 return candidate;
712 }
713 }
714 return {};
715 };
716 interpreter_->setModuleLoader([resolveModulePath](const std::string& modulePath) -> std::string {
717 // P2-11: 内建标准库模块拦截（std/math, std/string, std/list）
718 if (BuiltinModuleRegistry::isBuiltinModule(modulePath)) {
719 return BuiltinModuleRegistry::getSource(modulePath);
720 }
721 QString resolved = resolveModulePath(modulePath);
722 if (resolved.isEmpty())
723 return "";
724 QFile file(resolved);
725 if (file.open(QIODevice::ReadOnly | QIODevice::Text)) {
726 // P3.4 fix: 区分"文件不存在"与"空文件"——空文件返回 "\n"（合法空模块）
727 std::string content = QString::fromUtf8(file.readAll()).toStdString();
728 return content.empty() ? "\n" : content;
729 }
730 return "";
731 });
732 interpreter_->setModuleMtimeChecker([resolveModulePath](const std::string& modulePath) -> int64_t {
733 QString resolved = resolveModulePath(modulePath);
734 if (resolved.isEmpty())
735 return 0;
736 QFileInfo fi(resolved);
737 if (!fi.exists())
738 return 0;
739 return fi.lastModified().toMSecsSinceEpoch();
740 });
741}
