1// ============================================================
2// ShareCodec — 代码片段分享链接编解码实现（拓展二期·平台）
3// ------------------------------------------------------------
4// 负载格式：minilang://share/v1/<base64url(qCompress(utf8) || CRC16)>
5//
6// CRC-16 尾部校验（2 字节，big-endian，Qt qChecksum/ISO 3309）：
7// 实测发现 qUncompress 对"被截断的 zlib 流"会进入缓冲区反复加倍的
8// 近似无限循环（Z_BUF_ERROR 重试直到内存耗尽，表现为挂死）。
9// 因此解压前必须先做完整性校验：截断/篡改的负载在 CRC 阶段
10// 即被 O(n) 拒绝，qUncompress 只处理校验通过的数据。
11// 另对负载声明的解压尺寸设 16MB 上限，拒绝恶意构造的 zip bomb。
12// ============================================================

23/// 解压后源码尺寸上限（16MB）：教学片段远小于此，超限视为恶意负载
24constexpr quint32 kMaxDecompressedSize = 16u * 1024u * 1024u;

27QString encode(const QString& code) {
28 if (code.isEmpty())
29 return {};
30 // UTF-8 → zlib 压缩（级别 9：分享链接一次编码多次粘贴，压得越小越好）
31 QByteArray blob = qCompress(code.toUtf8(), 9);
32 // 追加 CRC-16 尾部校验（防截断/篡改，见文件头说明）
33 quint16 crc = qChecksum(QByteArrayView(blob));
34 blob.append(static_cast<char>((crc >> 8) & 0xFF));
35 blob.append(static_cast<char>(crc & 0xFF));
36 // base64url（无 + / 字符）+ 省略尾部 = 填充，负载可直接嵌入 URL/聊天消息
37 QByteArray payload = blob.toBase64(QByteArray::Base64UrlEncoding | QByteArray::OmitTrailingEquals);
38 return QString::fromLatin1(kSharePrefix) + QString::fromLatin1(payload);
39}

41bool decode(const QString& url, QString& codeOut, QString& errorOut) {
42 codeOut.clear();
43 errorOut.clear();
44
45 // 容忍聊天工具粘贴引入的首尾空白/换行
46 QString trimmed = url.trimmed();
47 const QString prefix = QString::fromLatin1(kSharePrefix);
48 if (!trimmed.startsWith(prefix)) {
49 errorOut = QStringLiteral("链接格式无效：应以 %1 开头").arg(prefix);
50 return false;
51 }
52
53 QString payloadStr = trimmed.mid(prefix.size());
54 if (payloadStr.isEmpty()) {
55 errorOut = QStringLiteral("链接负载为空");
56 return false;
57 }
58
59 // AbortOnBase64DecodingErrors：负载被截断/篡改时立即失败而非静默产出垃圾
60 auto decoded = QByteArray::fromBase64Encoding(payloadStr.toLatin1(), QByteArray::Base64UrlEncoding |
61 QByteArray::AbortOnBase64DecodingErrors);
62 if (!decoded) {
63 errorOut = QStringLiteral("base64 解码失败：链接可能被截断或损坏");
64 return false;
65 }
66
67 // 结构下限：qCompress 4 字节长度头 + 至少 1 字节 zlib 流 + 2 字节 CRC
68 QByteArray blob = *decoded;
69 if (blob.size() < 7) {
70 errorOut = QStringLiteral("链接负载过短：不是有效的分享负载");
71 return false;
72 }
73
74 // CRC-16 完整性校验——必须先于 qUncompress（截断流会使其挂死，见文件头）
75 const auto n = blob.size();
76 quint16 storedCrc =
77 static_cast<quint16>((static_cast<quint8>(blob[n - 2]) << 8) | static_cast<quint8>(blob[n - 1]));
78 QByteArray compressed = blob.left(n - 2);
79 if (qChecksum(QByteArrayView(compressed)) != storedCrc) {
80 errorOut = QStringLiteral("校验和不匹配：链接被截断或篡改");
81 return false;
82 }
83
84 // qCompress 前 4 字节为解压后尺寸（big-endian），超限拒绝（zip bomb 防护）
85 quint32 expectedSize = (static_cast<quint32>(static_cast<quint8>(compressed[0])) << 24) |
86 (static_cast<quint32>(static_cast<quint8>(compressed[1])) << 16) |
87 (static_cast<quint32>(static_cast<quint8>(compressed[2])) << 8) |
88 static_cast<quint32>(static_cast<quint8>(compressed[3]));
89 if (expectedSize > kMaxDecompressedSize) {
90 errorOut = QStringLiteral("负载声明尺寸异常（%1 字节）：拒绝解压").arg(expectedSize);
91 return false;
92 }
93
94 QByteArray decompressed = qUncompress(compressed);
95 if (decompressed.isEmpty()) {
96 // 合法空源码在 encode 端已被拒绝，空结果只可能是解压失败
97 errorOut = QStringLiteral("解压失败：链接负载损坏");
98 return false;
99 }
100
101 codeOut = QString::fromUtf8(decompressed);
102 return true;
103}
